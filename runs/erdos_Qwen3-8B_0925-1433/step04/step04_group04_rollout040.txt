import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()

    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        scale_x = (n_points / 2) / sum_x if sum_x != 0 else 1.0
        x_scaled = np.clip(x * scale_x, 0, 1)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx

    def local_minimizer(x0):
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

        res = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)

        best_c5 = compute_c5(best_h, dx, n_points)
