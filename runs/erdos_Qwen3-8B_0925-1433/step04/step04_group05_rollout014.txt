            scale_update = (n_points / 2) / sum_update
            h_update = h_update * scale_update
        else:
            h_update = np.zeros_like(h)
        new_corr = np.correlate(h_update, 1 - h_update, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < current_c5:
            h = h_update.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            if best_c5 < 0.3807:
                break
    return (best_h, best_c5, n_points)
