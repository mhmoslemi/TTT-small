import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
    h = np.clip(initial_h_values * scale, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_max_lag = np.argmax(corr)
        current_c5 = corr[current_max_lag] * dx

        def obj(x):
            h_vec = x
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return corr[current_max_lag] * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1)] * n_points

        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
        except:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
