import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    h = scale_h(initial_h_values.copy())
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    tol = 1e-5
    max_iter = 500
    alpha = 0.01
    perturb = 0.1

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        sensitivity = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += perturb
            h_perturb = scale_h(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            sensitivity[i] = (corr_perturb[k_max] - corr[k_max]) / perturb
        grad = -sensitivity
        h_new = h + alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = scale_h(h_new)
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if new_c5 < best_c5 and np.abs(new_c5 - best_c5) > tol:
            best_c5 = new_c5
            best_h = h_new.copy()
            alpha = max(alpha * 0.9, 1e-5)
        else:
            alpha *= 0.9
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
