from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    global initial_h_values

    target_sum = n_points / 2.0

    # Initialize and normalize
    h = initial_h_values.copy()

    scale = (target_sum / sum_h) if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    h = h / np.sum(h) * target_sum
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    iter_count = 0
    max_iters = 100

    while time.time() - start_time < budget_s and iter_count < max_iters:
        iter_count += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        lags = np.arange(-len(h) + 1, len(h))
        values = corr[len(h)-1:]  # Only consider positive lags
        top_lags = np.argsort(values)[-5:][::-1]  # Top 5 worst lags
        obj_lags = [lags[i] for i in top_lags]
        obj_values = [values[i] for i in top_lags]

        # Objective is max of top 5
        def objective(x):
            h_new = np.clip(x, 0, 1)
            h_new = h_new / np.sum(h_new) * target_sum
            corr = np.correlate(h_new, 1.0 - h_new, mode='full')
            lags = np.arange(-len(h_new) + 1, len(h_new))
            values = corr[len(h_new)-1:]
            return np.max([values[i] for i in top_lags])

        # Constraints: sum(h) = target_sum
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        try:
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * target_sum
        except:
            pass

        # Perturbation

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr[len(h)-1:]) * dx
