def evaluate(h, n_points, dx):
    if np.any(h < 0) or np.any(h > 1):
        return np.inf
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        return np.inf

    max_corr = np.max(corr) * dx

def crossover(parent1, parent2, n_points):
