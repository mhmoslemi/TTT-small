import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    h_initial = scale_h(initial_h_values)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    time_used = 0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_c5(x)

    def optimize_candidate(h):
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, bounds=bounds, constraints=constraints, method='SLSQP', tol=1e-7)
        if res.success:
            return res.x
        else:
            return scale_h(h + np.random.normal(0, 0.05, n_points))

    for _ in range(300):
        if time_used > budget_s:
            break
        start = time.time()
        best_h = optimize_candidate(best_h)
        best_c5 = compute_c5(best_h)
        time_used += time.time() - start
        if best_c5 < 0.3807:
            break

    best_h = scale_h(best_h)
    return (best_h, best_c5, n_points)
