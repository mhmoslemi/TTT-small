            term1 = 1.0 if i_plus < 0 or i_plus >= n_points else 1.0 - h[i_plus]
            i_minus = i - k_max
            term2 = 0.0 if i_minus < 0 or i_minus >= n_points else h[i_minus]
            grad[i] = (term1 - term2) * dx
        h = h - 0.01 * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / (sum_h if sum_h > 1e-10 else 1)
        current_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
