import numpy as np

def run(seed=42, budget_s=600):
    import time
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.astype(np.float64).copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = h * scale
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = h * scale
    h = np.clip(h, 0, 1)

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(h, dx):
        epsilon = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += epsilon
            h_p = np.clip(h_p, 0, 1)
            c_p = compute_c5(h_p, dx)
            h_m = h.copy()
            h_m[i] -= epsilon
            h_m = np.clip(h_m, 0, 1)
            c_m = compute_c5(h_m, dx)
            grad[i] = (c_p - c_m) / (2 * epsilon)
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        g = grad(h, dx)
        h = h - 0.1 * g
