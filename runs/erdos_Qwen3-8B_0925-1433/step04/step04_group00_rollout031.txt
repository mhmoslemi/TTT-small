        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=10,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=int((budget_s - 10) / 0.2)
    )
    best_h = result.x
    best_c5 = objective(best_h)

    h_local = best_h.copy()
    h_local = np.clip(h_local, 0, 1)
    sum_h = np.sum(h_local)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h_local = np.clip(h_local * scale, 0, 1)
    shifted = 1 - h_local
    corr = np.correlate(h_local, shifted, mode='full')
    max_corr_idx = np.argmax(corr)

    def local_objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = x_clipped * scale
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return corr[max_corr_idx] * dx

    def constraint(x):
        x_clipped = np.clip(x, 0, 1)
        return np.sum(x_clipped) - (n_points / 2)

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds_local = [(0.0, 1.0) for _ in range(n_points)]
    res_local = minimize(local_objective, h_local, method='SLSQP', bounds=bounds_local, constraints=cons, tol=1e-6)
    best_h = res_local.x
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
