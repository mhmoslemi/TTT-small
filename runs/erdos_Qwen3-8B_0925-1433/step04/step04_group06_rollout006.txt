import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / s
        return np.clip(h * scale, 0, 1)

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, mutation_rate=0.1, mutation_amp=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-mutation_amp, mutation_amp) * h[i]
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)

    def crossover(parent1, parent2):
        idx = np.random.randint(1, len(parent1)-1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        return normalize(child)

    def local_refine(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - len(h) + 1
        if shift < 0:
            shift += len(h)
        k = shift
        h_shifted = np.zeros_like(h)
        for i in range(len(h)):
            x = i * dx
            xk = x + k * dx
            if 0 <= xk <= 2:
                h_shifted[i] = 1 - h[i]
        return normalize(h_shifted)

    h_pop = [normalize(initial_h_values.copy())]
    for _ in range(49):
        rand_h = np.random.uniform(0, 1, n_points)
        h_pop.append(normalize(rand_h))
    h_pop = np.array(h_pop)

    best_h = h_pop[0]
    best_c5 = fitness(best_h)
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        fitnesses = np.array([fitness(h) for h in h_pop])
        idxs = np.argsort(fitnesses)
        best_idx = idxs[0]
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = h_pop[best_idx]

            break
        parents = h_pop[idxs[:int(len(h_pop)*0.3)]]
        new_pop = []
        for _ in range(len(h_pop)//2):
            p1, p2 = np.random.choice(parents, 2, replace=False)

        h_pop = np.array(new_pop)
        for h in h_pop:
            if np.random.rand() < 0.2:
                h = local_refine(h)
        h_pop = np.array([normalize(h) for h in h_pop])
