        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                h = res.x
        except:
            pass
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
        perturb = np.random.normal(0, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        perturb_sum = np.sum(h_perturbed) - target_sum
        h_perturbed += (perturb_sum / n_points) * np.ones(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h = h_perturbed

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
