    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    max_iterations = int(200 * (budget_s / 600))
    for _ in range(max_iterations):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not ones.size or not zeros.size:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
