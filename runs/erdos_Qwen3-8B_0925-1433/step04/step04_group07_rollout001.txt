import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Initialize h
    h = initial_h_values.copy()
    sum_initial = np.sum(h)
    scale = (n_points / 2) / sum_initial
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    # Optimization parameters
    alpha = 0.1
    iterations = 1000
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        # Update h
        h = h - alpha * grad
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize sum to n_points/2
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.full(n_points, 0.5)
        # Compute current c5
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Decay alpha
        if _ >= 500:
            alpha *= 0.1
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, dx, n_points, eps=1e-4):
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Perturb up
        h_plus = h.copy()
        h_plus[i] += eps
        sum_p = np.sum(h_plus)
        if sum_p == 0:
            c5_p = 0
        else:
            scale_p = (n_points / 2) / sum_p
            h_plus = np.clip(h_plus * scale_p, 0, 1)
            c5_p = compute_c5(h_plus, dx, n_points)
        # Perturb down
        h_minus = h.copy()
        h_minus[i] -= eps
        sum_m = np.sum(h_minus)
        if sum_m == 0:
            c5_m = 0
        else:
            scale_m = (n_points / 2) / sum_m
            h_minus = np.clip(h_minus * scale_m, 0, 1)
            c5_m = compute_c5(h_minus, dx, n_points)
        grad[i] = (c5_p - c5_m) / (2 * eps)
    return grad
