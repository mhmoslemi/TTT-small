            h = h_new.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        T *= alpha
    return (best_h, best_c5, n_points)
