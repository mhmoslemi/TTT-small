import numpy as np
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = basinhopping(
        objective,
        initial_h_values,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': cons
        },
        niter=150,
        T=1.5,
        stepsize=0.6,
        accept_prob='norm',
        callback=lambda x: None
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h_values, compute_c5(initial_h_values, dx, n_points), n_points)

    best_h = result.x
    best_c5 = compute_c5(best_h, dx, n_points)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
