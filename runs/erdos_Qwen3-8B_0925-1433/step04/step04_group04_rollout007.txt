import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s and iterations < 1000:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx

        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            scale = (n_points / 2) / sum_x if sum_x != 0 else 1.0
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = minimize(obj, h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-5)
        if res.success:

            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
