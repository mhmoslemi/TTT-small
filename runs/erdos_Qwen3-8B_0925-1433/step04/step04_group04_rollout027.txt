import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target = 0.3807
    population_size = 50
    generations = 300
    elite_size = 2
    mutation_rate = 0.1
    mutation_steps = 3

    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr * dx)

    def scale_h(h):
        total = np.sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        h = np.clip(h * scale, 0, 1)
        return h

    def mutate(h):
        h = h.copy()
        for _ in range(mutation_steps):
            idxs = np.random.choice(n_points, size=3, replace=False)
            h[idxs] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
        return scale_h(h)

    def crossover(parent1, parent2):
        child = parent1.copy()
        idxs = np.random.choice(n_points, size=5, replace=False)
        child[idxs] = parent2[idxs]
        return scale_h(child)

    initial_population = []
    for _ in range(population_size):
        if np.random.rand() < 0.7:
            h = initial_h_values.copy()
            h = np.clip(h, 0, 1)
            h = scale_h(h)
        else:
            h = np.random.rand(n_points)
            h = scale_h(h)
        initial_population.append(h)

    pop = initial_population
    best_h = scale_h(initial_h_values)
    best_score = evaluate(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(ind) for ind in pop]
        sorted_indices = np.argsort(scores)
        sorted_pop = [pop[i] for i in sorted_indices]
        elite = sorted_pop[:elite_size]
        new_pop = elite.copy()

        while len(new_pop) < population_size:
            parent1, parent2 = np.random.choice(sorted_pop[:population_size//2], size=2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.rand() < mutation_rate:
                child = mutate(child)
            new_pop.append(child)

        pop = new_pop
        current_best = sorted_pop[0]
        current_score = scores[0]
        if current_score < best_score:
            best_score = current_score
            best_h = current_best
        if best_score < target:
            break

    return (best_h, -best_score, n_points)
