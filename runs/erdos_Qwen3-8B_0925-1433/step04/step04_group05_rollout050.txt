import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    candidates = [scale_h(initial_h_values)]

    x = np.arange(n_points) * dx
    sine = 0.5 + 0.5 * np.sin(2 * np.pi * x)
    candidates.append(scale_h(sine))
    cosine = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    candidates.append(scale_h(cosine))

    for _ in range(5):
        h = np.zeros(n_points)
        cnt = 0
        for i in range(n_points):
            if cnt % 2 == 0:
                h[i] = 1
            cnt += 1
        h = scale_h(h)
        candidates.append(h)

    def optimize(h):
        def obj(x):
            h_scaled = scale_h(x)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        return res.x

    best_h = candidates[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for cand in candidates:
            curr = compute_c5(cand)
            if curr < best_c5:
                best_c5 = curr
                best_h = cand.copy()
        if best_c5 < 0.3807:
            break
        best_h = optimize(best_h)
        if compute_c5(best_h) < best_c5:
            best_c5 = compute_c5(best_h)
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
