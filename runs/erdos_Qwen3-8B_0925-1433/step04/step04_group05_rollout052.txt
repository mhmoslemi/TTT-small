import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = scale_h(initial_h_values.copy())
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

    for _ in range(5):
        res = minimize(obj, initial_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        if res.success:
            initial_h = res.x
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = initial_h.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    break
        else:
            break

    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            if best_c5 < 0.3807:
                break
        else:
            res = minimize(obj, best_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                best_h = res.x
                best_c5 = res.fun
            else:
                break

    return (best_h, best_c5, n_points)
