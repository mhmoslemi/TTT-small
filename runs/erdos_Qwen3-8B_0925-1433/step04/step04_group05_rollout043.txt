import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

            if (i % 2 == 0):
                mask[start:end] = 1
        h_bin = scale_h(mask * (n_points / 2) / np.sum(mask))
        candidates.append(h_bin)

    # Periodic
    x = np.linspace(0, 2, n_points)
    h_per = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    candidates.append(scale_h(h_per))

        return compute_c5(x)
