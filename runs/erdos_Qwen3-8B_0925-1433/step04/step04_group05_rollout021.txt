        res = minimize(lambda x: compute_c5(scale_h(x)), h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        if res.success:
            h_candidate = res.x
        else:
            h_candidate = scale_h(h + np.random.normal(0, 0.05, n_points))
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Try local optimization
        h = best_h.copy()

        current_c5 = compute_c5(h_candidate)

            best_h = h_candidate.copy()
            best_c5 = current_c5
        # Random perturbation
        if np.random.rand() < 0.1:

            current_c5 = compute_c5(h)
            if current_c5 < best_c5:

        if best_c5 < 0.3807:
