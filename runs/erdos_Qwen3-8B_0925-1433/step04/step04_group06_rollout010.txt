import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
    else:
        scale = 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    perturb_freq = 10
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full')
        integral_values = corr * dx
        top_indices = np.argsort(integral_values)[::-1][:5]
        weights = [np.exp(-i) for i in range(1, 6)]

        def obj(x):
            h_val = x
            corr_val = np.correlate(h_val, 1 - h_val, mode='full')
            integrals = corr_val * dx
            current_indices = np.argsort(integrals)[::-1][:5]
            return np.sum(weights * integrals[current_indices])

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                h = res.x
        except:
            pass

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
        else:
            scale = 1.0
        h = h * scale

        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
        if iter_count % perturb_freq == 0:
            perturb = np.random.normal(0, 0.05, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
            else:
                scale = 1.0
            h = h * scale

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
