import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 50
    particles = []
    if initial_h_values is not None:
        for _ in range(population_size):
            h = initial_h_values.copy()
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)
            particles.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)
            particles.append(h)
    particles = np.array(particles)

    # PSO parameters
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    max_vel = 0.5
    velocities = np.zeros((population_size, n_points))

    pbest = particles.copy()
    gbest = particles[0].copy()
    best_c5 = compute_c5(gbest)

    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s and iterations < 200:
        iterations += 1
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            scale = (n_points / 2) / sum_p if sum_p != 0 else 1.0
            particles[i] = np.clip(particles[i] * scale, 0, 1)

        current_c5 = np.array([compute_c5(p) for p in particles])
        for i in range(population_size):
            if current_c5[i] < compute_c5(pbest[i]):
                pbest[i] = particles[i].copy()
        best_idx = np.argmin(current_c5)
        gbest = pbest[best_idx].copy()
        best_c5 = current_c5[best_idx]

        if best_c5 < 0.3807:
            break

    return (gbest, best_c5, n_points)
