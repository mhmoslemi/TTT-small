import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.round(initial_h_values).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum != target_sum:
        delta = target_sum - current_sum
        zeros = np.where(h == 0)[0]
        if delta > 0:
            h[zeros[:delta]] = 1
        else:
            ones = np.where(h == 1)[0]
            h[ones[:abs(delta)]] = 0
    start_time = time.time()
    max_iter = 500
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        if current_max < 0.3807:
            break
        max_index = np.argmax(corr)
        current_k = max_index - (n_points - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        found = False
        best_i, best_j = -1, -1
        for i in ones:
            shifted_i = (i + current_k) % n_points
            if h[shifted_i] == 0:
                for j in zeros:
                    shifted_j = (j + current_k) % n_points
                    if h[shifted_j] == 1:
                        best_i, best_j = i, j
                        found = True
                        break
                if found:
                    break
        if found:
            h[best_i] = 0
            h[best_j] = 1
        else:
            best_delta = 0
            best_i, best_j = -1, -1
            for i in ones:
                shifted_i = (i + current_k) % n_points
                val_i = h[shifted_i]
                for j in zeros:
                    shifted_j = (j + current_k) % n_points
                    val_j = h[shifted_j]
                    delta = val_i - val_j
                    if delta < best_delta:
                        best_delta = delta
                        best_i, best_j = i, j
            if best_delta < 0:
                h[best_i] = 0
                h[best_j] = 1
        if time.time() - start_time >= budget_s:
            break
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return (h, c5, n_points)
