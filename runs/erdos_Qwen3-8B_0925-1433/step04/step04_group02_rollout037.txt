        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_perturbed.copy()
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if best_c5 < early_termination:
            return (best_h, best_c5, n_points)
        T *= alpha
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
