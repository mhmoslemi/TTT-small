        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    best_h = initial_candidates[0]
    best_c5 = compute_c5(best_h)
    best_time = 0

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        scale = (n_points / 2) / sum_x if sum_x != 0 else 1
        x = np.clip(x * scale, 0, 1)
        return compute_c5(x)

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for h_candidate in initial_candidates:
            result = minimize(objective, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                optimized_h = result.x
                optimized_h = np.clip(optimized_h, 0, 1)
                sum_opt = np.sum(optimized_h)
                scale_opt = (n_points / 2) / sum_opt if sum_opt != 0 else 1
                optimized_h = np.clip(optimized_h * scale_opt, 0, 1)
                current_c5 = compute_c5(optimized_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized_h
                    best_time = time.time()
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)
        # Fallback to less intensive refinement if time permits
        if time.time() - best_time > 60:
            break
    return (best_h, best_c5, n_points)
