    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 50
    population = []
    # Start with initial_h_values scaled
    initial_sum = np.sum(initial_h_values)
    scale = target_sum / initial_sum
    h = np.clip(initial_h_values * scale, 0, 1)
    population.append(h)
    # Add other random individuals
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale = target_sum / h_sum
        h = np.clip(h * scale, 0, 1)
        population.append(h)

    best_h = population[0].copy()
    best_c5 = compute_c5(population[0])
    start_time = time.time()

    for generation in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Evaluate and select parents
        fitness = [compute_c5(h) for h in population]
        indices = np.argsort(fitness)
        parents = [population[i] for i in indices[:int(population_size * 0.3)]]
        new_population = []
        for _ in range(population_size):
            # Crossover
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = np.random.uniform(0, 1, n_points)
                child_sum = np.sum(child)
                scale = target_sum / child_sum
                child = np.clip(child * scale, 0, 1)
            # Mutate
            if np.random.rand() < 0.2:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.uniform(-0.2, 0.2) * child[i]
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                if child_sum != 0:
                    scale = target_sum / child_sum
                    child *= scale
            new_population.append(child)
        population = new_population
        # Local refinement on top solutions
        for h_candidate in population[:5]:
            curr_c5 = compute_c5(h_candidate)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_candidate.copy()
                # Refine worst lag
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                max_idx = np.argmax(corr)
                lag = max_idx - (n_points - 1)
                k = lag
                start_idx = max(0, int((-k) * n_points / 2))
                end_idx = min(n_points - 1, int((2 - k) * n_points / 2))
                modified = h_candidate.copy()
                for i in range(start_idx, end_idx + 1):
                    modified[i] = 1 - modified[i]
                modified_sum = np.sum(modified)
                if modified_sum != 0:
                    scale = target_sum / modified_sum
                    modified *= scale
                modified = np.clip(modified, 0, 1)
                improved_c5 = compute_c5(modified)
                if improved_c5 < best_c5:
                    best_c5 = improved_c5
                    best_h = modified.copy()
            if best_c5 < 0.3807:
                break

    return (best_h, best_c5, n_points)
