import time

    np.random.seed(seed)
    global initial_h_values

    dx = 2.0 / n_points

        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    stagnation_count = 0
    max_stagnation = 100

        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
