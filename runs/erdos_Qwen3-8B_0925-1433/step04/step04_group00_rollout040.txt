from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    h_best = initial_h_values.copy()
    h_best = np.clip(h_best, 0, 1)
    sum_h = np.sum(h_best)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h_best = np.clip(h_best * scale, 0, 1)
    else:
        h_best = np.ones(n_points) * 0.5
    best_c5 = compute_c5(h_best)
    
    num_particles = 50
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    for _ in range(num_particles):

            h = initial_h_values.copy()
        elif np.random.rand() < 0.3:

            h = np.sin(np.linspace(0, np.pi, n_points)) + 1

            scale = (n_points / 2) / sum_h

            h = np.ones(n_points) * 0.5
