        lambda h: compute_c5(np.clip(h, 0, 1)),
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=150,
        tol=1e-6,
        disp=False,
        polish=True
    )
    best_h = result_de.x
    
    # Local refinement
    h_local = best_h.copy()
    h_local = np.clip(h_local, 0, 1)
    sum_h = np.sum(h_local)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h_local = np.clip(h_local * scale, 0, 1)
    else:
        h_local = np.ones(n_points) * 0.5
    
    shifted = 1 - h_local
    corr = np.correlate(h_local, shifted, mode='full')
    max_corr_idx = np.argmax(corr)
    max_corr_val = corr[max_corr_idx]
    
    def local_obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = np.clip(x * scale, 0, 1)
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return corr[max_corr_idx] * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - n_points / 2}]
    res = minimize(local_obj, h_local, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
    else:
        best_h = np.ones(n_points) * 0.5
    
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
