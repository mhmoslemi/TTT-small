import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def generate_sine(freq, phase):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * (x - phase))
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        return h

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x = np.clip(x * scale, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h_initial = np.clip(h_initial * scale, 0, 1)
    else:
        h_initial = np.ones(n_points) * 0.5
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    candidates = [best_h]
    for freq in range(1, 6):
        for phase in [0, 0.25, 0.5, 0.75, 1.0]:
            h = generate_sine(freq, phase)
            candidates.append(h)

    for h in candidates:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    while time.time() - start_time < budget_s:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
        if res.success:
            optimized_h = res.x
            optimized_h = np.clip(optimized_h, 0, 1)
            sum_opt = np.sum(optimized_h)
            if sum_opt != 0:
                scale = (n_points / 2) / sum_opt
                optimized_h = np.clip(optimized_h * scale, 0, 1)
            else:
                optimized_h = np.ones(n_points) * 0.5
            new_c5 = compute_c5(optimized_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = optimized_h.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
