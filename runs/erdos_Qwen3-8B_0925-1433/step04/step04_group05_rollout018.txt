import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    initial_scaled = scale_h(initial_h_values.copy())
    candidates = [initial_scaled]

    # Binary slice candidates
    for shift in range(5):
        binary = np.zeros(n_points)
        binary[::2] = 1.0
        binary = np.roll(binary, shift)
        binary = scale_h(binary)
        candidates.append(binary)

    # Periodic candidates
    for freq in [1.0, 0.5, 0.333, 0.25, 0.2]:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        h = scale_h(h)
        candidates.append(h)

    # Interval spacing candidates
    for shift in range(5):
        interval = np.zeros(n_points)
        spacing = n_points // 2 + (1 if n_points % 2 else 0)
        for i in range(n_points):
            if (i + shift) % spacing == 0:
                interval[i] = 1.0
        interval = scale_h(interval)
        candidates.append(interval)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h = initial_scaled.copy()

    start_time = time.time()
    for candidate in candidates:
        def obj(x):
            h = scale_h(x)
            return compute_c5(h)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        optimized_h = res.x if res.success else candidate
        current_c5 = compute_c5(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
