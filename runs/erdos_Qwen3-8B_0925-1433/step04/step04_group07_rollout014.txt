import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2.0) / sum_h

    best_h = h.copy()

        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        contrib = h * np.roll(h, int(k_max))
        top_indices = np.argsort(contrib)[::-1][:5]
        delta = 0.01
        for i in top_indices:
            j = (i + k_max) % n_points
            h[i] -= delta
            h[j] -= delta
        h = np.clip(h, 0, 1)

        current_c5 = compute_c5(h, dx, n_points)
