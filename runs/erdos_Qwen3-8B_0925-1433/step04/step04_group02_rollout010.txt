import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / np.sum(h)
        return np.clip(h * scale, 0, 1)

    x = np.linspace(0, 2, n_points)
    candidates = [initial_h_values.copy()]
    for nu in range(1, 9):
        h = 0.5 + 0.5 * np.sin(2 * np.pi * nu * x)
        candidates.append(h.copy())

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    t0 = time.time()
    for i in range(5):
        if time.time() - t0 > budget_s:
            break
        h = candidates[i % len(candidates)]
        h = clip_scale(h)
        res = minimize(compute_c5, h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            h_opt = res.x
            h_opt = clip_scale(h_opt)
            curr_c5 = compute_c5(h_opt)
            if curr_c5 < best_c5:
                best_h, best_c5 = h_opt, curr_c5
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
