    def compute_c5(h):

        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    h = initial_h_values.copy()

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]

        def obj(x):
            current_h = x
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            return corr[max_corr_idx]

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
