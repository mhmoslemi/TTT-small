        if time.time() - start_time > budget_s:
            break
        grad = gradient_func(best_h)
        best_h -= alpha * grad
        best_h = np.clip(best_h, 0.0, 1.0)
        h_sum = np.sum(best_h)
        if h_sum != 0:
            best_h *= target_sum / h_sum
        best_h = np.clip(best_h, 0.0, 1.0)
        current_c5 = obj_func(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        if np.abs(current_c5 - best_c5) < 1e-6:
            break
        alpha *= 0.98

    return (best_h, best_c5, n_points)
