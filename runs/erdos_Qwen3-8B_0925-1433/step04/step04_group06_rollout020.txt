            perturb = np.random.normal(0, 0.1, size=n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = h * scale

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
