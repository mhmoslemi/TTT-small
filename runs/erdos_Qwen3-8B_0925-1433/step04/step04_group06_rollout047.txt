import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def scale_to_sum(h, target_sum):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target_sum / s
        h = np.clip(h * scale, 0, 1)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = []
    h = initial_h_values.copy()
    h = scale_to_sum(h, target_sum)
    population.append(h)
    for _ in range(49):
        if np.random.rand() < 0.5:
            h = np.random.rand(n_points)
            h = scale_to_sum(h, target_sum)
        else:
            h = initial_h_values.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = scale_to_sum(h, target_sum)
        population.append(h)

    best_h = population[0]
