    # Initial guess from initial_h_values, scaled
    initial_guess = scale_h(initial_h_values.copy())

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Objective function for optimization
    def objective(h):
        return compute_c5(h)

    # Constraints
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    h_initial = initial_guess.copy()

    # Optimization
    start_time = time.time()
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()

    for _ in range(100):  # Limit to 100 iterations to stay within budget
        if time.time() - start_time > budget_s:
