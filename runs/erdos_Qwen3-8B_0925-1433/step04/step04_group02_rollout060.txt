import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5
    
    T = 1.0
    alpha = 0.998
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while i == j:
            j = np.random.randint(n_points)
        original_i = current_h[i]
        original_j = current_h[j]
        delta_i = np.random.uniform(-0.12, 0.12)
        new_i = original_i + delta_i
        new_i = np.clip(new_i, 0, 1)
        new_j = original_j - (new_i - original_i)
        new_j = np.clip(new_j, 0, 1)
        new_h = current_h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = new_h.copy()

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        T *= alpha
    return (best_h, best_c5, n_points)

def compute_c5(h):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / len(h))
