import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize initial_h_values
    h = initial_h_values.copy()
    sum_initial = np.sum(h)
    if sum_initial == 0:
        h = np.ones(n_points) * 0.5
    else:
        scale = target_sum / sum_initial
        h = np.clip(h * scale, 0, 1)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    def compute_c5(h):
        shifted = 1.0 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    temperature = 1.0
    cooling_rate = 0.995
    max_iter = 10000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Perturb h
        num_perturb = np.random.randint(2, 5)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        deltas = np.random.uniform(-0.1, 0.1, size=num_perturb)
        h_perturb = h.copy()
        for i, idx in enumerate(indices):
            h_perturb[idx] += deltas[i]
        h_perturb = np.clip(h_perturb, 0, 1)
        current_sum_perturb = np.sum(h_perturb)
        if current_sum_perturb == 0:
            h_scaled = np.ones(n_points) * 0.5
        else:
            scale_factor = target_sum / current_sum_perturb
            h_scaled = h_perturb * scale_factor
        h_scaled = np.clip(h_scaled, 0, 1)
        # Compute new_c5
        new_c5 = compute_c5(h_scaled)
        delta_c5 = new_c5 - current_c5
        # Acceptance
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            h = h_scaled.copy()
            current_c5 = new_c5
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        # Cool down
        temperature *= cooling_rate
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
