from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = initial_h_values.copy()

    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    iterations = 0
    step_size = 0.1
    decay_rate = 0.995
    early_stop = 10

    while time.time() - start_time < budget_s and iterations < 1000:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx

        if current_max < best_c5:

            best_h = h.copy()

        if best_c5 < 0.3807:

        k_max = np.argmax(corr) - (n_points - 1)
        if k_max < 0:
            k_max += 2 * (n_points - 1) - 1

        def grad_func(h, k):
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k + n_points - 1] * dx
