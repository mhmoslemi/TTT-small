        if it % perturb_interval == 0:
            noise = np.random.normal(0, perturb_strength, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h_perturbed = np.sum(h)
            if sum_h_perturbed != 0:
                scale_perturb = (n_points / 2) / sum_h_perturbed
                h = h * scale_perturb
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
