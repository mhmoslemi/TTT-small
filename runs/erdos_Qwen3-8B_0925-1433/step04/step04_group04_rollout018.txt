import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr * dx)

    def mutate(h):
        h = h.copy()
        for _ in range(3):
            idx = np.random.randint(n_points)
            h[idx] += np.random.uniform(-0.1, 0.1)
            h[idx] = np.clip(h[idx], 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        return h

    def crossover(p1, p2):
        offspring = p1.copy()
        for _ in range(5):
            idx = np.random.randint(n_points)
            offspring[idx] = p2[idx]
        return offspring

    initial_pop = [initial_h_values.copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h)
        initial_pop.append(np.clip(h * scale, 0, 1))

    best_h = initial_pop[0]
    best_c5 = evaluate(best_h)
    gen = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and gen < 300:
        gen += 1
        fitness = [evaluate(h) for h in initial_pop]
        valid_idx = np.isfinite(fitness)
        if np.sum(valid_idx) == 0:
            break
        valid_pop = [initial_pop[i] for i in np.where(valid_idx)[0]]
        valid_fitness = fitness[valid_idx]
        best_idx = np.argmin(valid_fitness)
        best_h = valid_pop[best_idx]
        best_c5 = valid_fitness[best_idx]

        if best_c5 < -0.3807:
            break

        parents = np.argsort(valid_fitness)[:20]
        new_pop = []
        for _ in range(25):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(valid_pop[p1], valid_pop[p2])
            child = mutate(child)
            new_pop.append(child)
        initial_pop = valid_pop + new_pop
        if len(initial_pop) > 50:
            initial_pop = initial_pop[:50]

    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h * scale, 0, 1)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = -np.max(corr * dx)
    return (best_h, best_c5, n_points)
