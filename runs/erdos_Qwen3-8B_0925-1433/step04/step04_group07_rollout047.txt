        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        # Target positions contributing to the worst lag
        target_indices = set()
        for i in range(n_points):
            j = (i + lag) % n_points
            target_indices.add(i)
            target_indices.add(j)
        # Decrease values at target positions
        for i in target_indices:
            if h[i] > 0.1:
                h[i] -= 0.01
        # Normalize to maintain sum constraint
        sum_h = np.sum(h)
        if sum_h > 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
