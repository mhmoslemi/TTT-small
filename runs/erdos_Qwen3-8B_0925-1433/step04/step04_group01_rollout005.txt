import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    pop = [initial_scaled.copy() for _ in range(50)]
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all in population
        fitness = [compute_c5(ind, dx, n_points) for ind in pop]
        top_idx = np.argsort(fitness)[:15]
        new_pop = []
        for i in range(15):
            parent = pop[top_idx[i]]
            # Small noise perturbation
            noise = np.random.normal(0, 0.02 * dx, n_points)
            mut = parent + noise
            s = np.sum(mut)
            mut = np.clip(mut * target_sum / s, 0, 1)
            # Local optimization focused on worst shift
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            res = minimize(obj, mut, method='COBYLA', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            new_pop.append(res.x)
        # Refine diversity
        pop = new_pop + pop[:30]
        # Update best
        for h in new_pop:
            curr = compute_c5(h, dx, n_points)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
