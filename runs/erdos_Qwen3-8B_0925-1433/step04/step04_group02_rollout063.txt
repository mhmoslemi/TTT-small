    # Function to compute C5
    def compute_c5(h):

        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Gradient descent parameters
    alpha = 0.1
    lambd = 1.0
    eps = 1e-7

        max_idx = np.argmax(corr)

        grad = np.zeros_like(h)
