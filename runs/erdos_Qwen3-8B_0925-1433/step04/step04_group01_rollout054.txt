import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    alpha = 0.05
    decay = 0.999
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        w = np.argmax(corr)
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += eps
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[w] - corr[w]) / eps
        h -= alpha * grad
        current_sum = np.sum(h)
        if current_sum == 0:
            continue
        h = np.clip(h * target_sum / current_sum, 0, 1)
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        alpha *= decay
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
