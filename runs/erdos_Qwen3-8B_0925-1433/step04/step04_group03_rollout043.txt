    scale = (n_points / 2) / sum_sinus if sum_sinus != 0 else 1.0
    h_sinus = np.clip(h_sinus * scale, 0, 1)
    current_best = h_sinus.copy()

    h_binary = np.random.choice([0, 1], size=n_points)
    sum_binary = np.sum(h_binary)
    scale = (n_points / 2) / sum_binary if sum_binary != 0 else 1.0
    h_binary = np.clip(h_binary * scale, 0, 1)
    current_best = h_binary.copy()

    def compute_c5(h, dx, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[(n_points - 1):2 * n_points - 1]) * dx

    best_c5 = compute_c5(current_best, dx, n_points)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(current_best, 1 - current_best, mode='full')
        max_corr = np.max(corr[(n_points - 1):2 * n_points - 1]) * dx
        if max_corr < 0.3807:
            break
        max_index = np.argmax(corr[(n_points - 1):2 * n_points - 1])
        k = max_index

        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            scale = (n_points / 2) / sum_x if sum_x != 0 else 1.0
            x = np.clip(x * scale, 0, 1)
            j = (np.arange(n_points) + k) % n_points
            overlap = np.dot(x, 1 - x[j])
            return overlap * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]

        try:
            result = minimize(objective, current_best, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                current_best = result.x
        except:
            perturb = np.random.normal(0, 0.03, n_points)
            new_h = current_best + perturb
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            scale = (n_points / 2) / sum_new if sum_new != 0 else 1.0
            new_h = np.clip(new_h * scale, 0, 1)
            current_best = new_h

        c5_new = compute_c5(current_best, dx, n_points)
        if c5_new < best_c5:
            best_c5 = c5_new

    return (current_best, best_c5, n_points)
