import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    pop = [initial_scaled.copy() for _ in range(20)]
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    temp = 1.0
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        if temp < 1e-5:
            break
        for i in range(len(pop)):
            h = pop[i].copy()
            noise = np.random.normal(0, 0.05 * dx, n_points)
            h += noise
            h = np.clip(h * target_sum / np.sum(h), 0, 1)
            f_curr = compute_c5(pop[i], dx, n_points)
            f_new = compute_c5(h, dx, n_points)
            if f_new < f_curr or np.random.rand() < np.exp((f_curr - f_new) / temp):
                pop[i] = h.copy()
        temp *= 0.995
        
        top_indices = np.argsort([compute_c5(p, dx, n_points) for p in pop])[:5]
        refined = []
        for i in top_indices:
            h = pop[i].copy()
            def smooth_obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(np.stack([np.roll(corr, -1), corr, np.roll(corr, 1)], axis=0).mean(axis=0)) * dx
            res = minimize(smooth_obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            refined.append(res.x)
        pop = refined + pop[:len(pop)-5]
        for h in refined:
            curr = compute_c5(h, dx, n_points)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
