import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def generate_sine_candidates():
        candidates = []
        for ν in range(1, 9):
            for phi in [0.25, 0.5, 0.75]:
                x = np.linspace(0, 2, n_points)
                h = 0.5 + 0.5 * np.sin(2 * np.pi * ν * (x - phi))
                h = np.clip(h, 0, 1)
                candidates.append(h)
        return candidates
    
    def optimize_candidate(h_initial):
        def objective(x):
            h = x
            shifted = 1 - h
            corr = np.correlate(h, shifted, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if result.success:
            return result.x
        return h_initial
    
    candidates = generate_sine_candidates()
    candidates.append(initial_h_values.copy())
    
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for h_initial in candidates:
        h_opt = optimize_candidate(h_initial)
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
