import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initial scaling
    h = initial_h_values.copy()
    h_sum = np.sum(h)
    if h_sum:
        scale = target_sum / h_sum
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    penalty = 1e6

    def obj_DE(x):
        c5 = evaluate_erdos_solution(x)
        sum_x = np.sum(x)
        return c5 + penalty * abs(sum_x - target_sum)

    start_time = time.time()
    result_DE = differential_evolution(obj_DE, bounds, strategy='best1bin', popsize=20, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.9)
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    best_h_DE = result_DE.x
    sum_h = np.sum(best_h_DE)
    if sum_h:
        scale = target_sum / sum_h
        best_h_DE = np.clip(best_h_DE * scale, 0, 1)
    else:
        best_h_DE = np.ones(n_points) * 0.5
    c5_de = evaluate_erdos_solution(best_h_DE)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = best_h_DE.copy()

    def obj_SLSQP(x):
        return evaluate_erdos_solution(x)

    def cons_SLSQP(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': cons_SLSQP}]
    bounds_SLSQP = [(0.0, 1.0) for _ in range(n_points)]

    result_SLSQP = minimize(obj_SLSQP, best_h_DE, method='SLSQP', bounds=bounds_SLSQP, constraints=cons, tol=1e-6)
    if result_SLSQP.success:
        best_h = result_SLSQP.x
        sum_h = np.sum(best_h)
        if sum_h:
            scale = target_sum / sum_h
            best_h = np.clip(best_h * scale, 0, 1)
        else:
            best_h = np.ones(n_points) * 0.5
        c5_final = evaluate_erdos_solution(best_h)
        if c5_final < best_c5:
            best_c5 = c5_final

    return (best_h, best_c5, n_points)
