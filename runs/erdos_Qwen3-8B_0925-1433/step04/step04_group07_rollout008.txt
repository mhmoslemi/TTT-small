def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize
    import time
    
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    sum_h = np.sum(h)

    else:

    h = h * scale

        scale = T / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    max_iterations = 1000
    learning_rate = 0.1
    for iteration in range(max_iterations):
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = compute_c5(h_plus, dx, n_points)
            
            h_minus = h.copy()

            h_minus = np.clip(h_minus, 0, 1)

            gradient[i] = grad
        
        # Update h
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Project to maintain sum
        sum_h = np.sum(h)
        while abs(sum_h - T) > 1e-6:
            scale = T / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
        
        # Check if current h is better
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check for convergence
        if iteration > 500:
            learning_rate *= 0.9
            if learning_rate < 1e-5:
                break
    
    final_c5 = compute_c5(best_h, dx, n_points)
    return (best_h, final_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
