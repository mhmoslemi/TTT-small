import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective_function(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraints(h):
        return np.sum(h) - target_sum
    
    def scale_and_clip(h):
        s = np.sum(h)
        if abs(s) < 1e-6:
            return np.full_like(h, 0.5)
        scaled = h * target_sum / s
        return np.clip(scaled, 0, 1)
    
    # Initial population: 80% from initial_h_values with perturbations, 20% random
    initial_pop = []
    for _ in range(50):
        if np.random.rand() < 0.8:
            h = initial_h_values.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
        else:
            h = np.random.uniform(0, 1, n_points)
        initial_pop.append(scale_and_clip(h))
    
    best_h = scale_and_clip(initial_pop[0])
    best_c5 = objective_function(best_h)
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Differential Evolution
        bounds = [(0, 1)] * n_points
        result = differential_evolution(objective_function, bounds, strategy='best1bin', 
                                        popsize=50, mutation=(0.8, 1.0), recombination=0.9, 
                                        tol=1e-6, seed=seed, disp=False, polish=True, 
                                        constraints=[{'type': 'eq', 'fun': constraints}])
        if result.success:
            candidate_h = scale_and_clip(result.x)
            curr_c5 = objective_function(candidate_h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = candidate_h
    
    # Local refinement with SLSQP
    def slsqp_objective(x):
        return objective_function(x)
    
    def slsqp_constraints(x):
        return np.sum(x) - target_sum
    
    result_slsqp = minimize(slsqp_objective, best_h, method='SLSQP', 
                            bounds=[(0, 1)]*n_points, 
                            constraints=[{'type': 'eq', 'fun': slsqp_constraints}])
    best_h = result_slsqp.x
    best_c5 = objective_function(best_h)
    
    return (best_h, best_c5, n_points)
