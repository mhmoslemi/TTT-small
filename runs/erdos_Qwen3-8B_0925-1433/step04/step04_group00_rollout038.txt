        sum_h = np.sum(h)
        if sum_h != 0:
            scale = target_sum / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    shifted = 1 - h_best
    corr = np.correlate(h_best, shifted, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]

    def local_objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x != 0:
            scale = target_sum / sum_x
            x = np.clip(x * scale, 0, 1)
        else:
            x = np.ones(n_points) * 0.5
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return corr[max_idx] * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds_local = [(0.0, 1.0)] * n_points

    result_local = minimize(local_objective, h_best, method='SLSQP',
                            bounds=bounds_local, constraints=cons, tol=1e-6)

    h_opt = result_local.x
    c5_bound = evaluate_c5(h_opt)
    return (h_opt, c5_bound, n_points)
