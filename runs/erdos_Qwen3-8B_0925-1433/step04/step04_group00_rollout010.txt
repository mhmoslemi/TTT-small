import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    iteration = 0
    max_iterations = int(budget_s / 0.1)
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted = 1 - x
            corr = np.correlate(x, shifted, mode='full')
            target_corr = corr[max_idx]
            return target_corr
        
        def constraint(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
        iteration += 1
    
    return (best_h, best_c5, n_points)
