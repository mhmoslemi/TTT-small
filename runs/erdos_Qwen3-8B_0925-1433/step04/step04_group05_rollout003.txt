from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    candidates = []
    m = 4
    for shift in range(m):
        idx = np.argsort(np.abs(np.linspace(0, 2, n_points) - shift / m))
        h = np.zeros(n_points)
        h[idx[:n_points//2]] = 1
        candidates.append(scale_h(h))
    for freq in [1, 2, 3]:
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.linspace(0, 2, n_points))
        candidates.append(scale_h(h))
    for _ in range(5):
        h = np.zeros(n_points)
        h[np.argsort(np.random.rand(n_points))[:n_points//2]] = 1
        candidates.append(scale_h(h))

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = min(compute_c5(c) for c in candidates)
    best_h = candidates[np.argmin([compute_c5(c) for c in candidates])].copy()

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(c) for c in candidates]
        top_idx = np.argsort(c5_values)[:3]
        top_candidates = [candidates[i] for i in top_idx]
        new_candidates = []
        for h in top_candidates:
            res = minimize(obj, h, bounds=[(0, 1) for _ in range(n_points)], 
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}], 
                          method='SLSQP', tol=1e-5)
            if res.success:
                new_candidates.append(res.x)

                new_candidates.append(h)
