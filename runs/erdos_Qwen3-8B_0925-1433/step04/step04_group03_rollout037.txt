import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Scale initial_h_values to meet the sum constraint
    sum_initial = np.sum(initial_h_values)
    scale = (n_points / 2) / sum_initial if sum_initial != 0 else 1.0
    h = np.clip(initial_h_values * scale, 0, 1)

    # Convert to binary
    h = np.round(h)
    sum_h = np.sum(h)
    if sum_h < n_points // 2:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[: (n_points//2) - sum_h]] = 1
    elif sum_h > n_points // 2:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[: sum_h - (n_points//2)]] = 0

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Local optimization
    max_iterations = 300
    for _ in range(max_iterations):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not ones.size or not zeros.size:
            break
        i, j = np.random.choice(ones), np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
