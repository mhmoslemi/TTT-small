            p2 = np.random.choice(sorted_pop[:top_size], size=1)[0]
            off = p1.copy()
            idxs = np.random.choice(n_points, size=5, replace=False)
            for i in idxs:
                off[i] = p2[i]
            # Mutate
            for i in np.random.choice(n_points, size=3, replace=False):
                off[i] += np.random.uniform(-0.1, 0.1)
                off[i] = np.clip(off[i], 0, 1)
            sum_off = np.sum(off)
            scale = (n_points / 2) / sum_off
            off = np.clip(off * scale, 0, 1)
            new_pop.append(off)

        population = new_pop
        curr_best = sorted_pop[0]
        curr_c5 = compute_c5(curr_best, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_best.copy()
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
