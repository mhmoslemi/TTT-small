        max_corr = np.max(corr)
        return max_corr * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    alpha = 0.1
    lambda_lag = 1.0
    max_iter = 2000
    decay_rate = 0.995
    freq_decay = 100

    for iter in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)

        argmax_index = np.argmax(corr)
        k_max = argmax_index - (n_points - 1)

        grad = np.zeros(n_points)
        for j in range(n_points):
            index1 = (j + k_max) % n_points
            term1 = (1 - h[index1]) * dx
            index2 = (j - k_max) % n_points
            term2 = h[index2] * dx
            grad[j] = term1 - term2

        grad_const = np.ones(n_points)
        total_grad = grad + lambda_lag * grad_const
        h -= alpha * total_grad

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5

        if iter % freq_decay == 0:
            alpha *= decay_rate
            constraint_violation = np.abs(sum_h - n_points / 2)
            if constraint_violation > 1e-6:
                lambda_lag *= 1.1
            else:
                lambda_lag *= 0.9

    return (best_h, best_c5, n_points)
