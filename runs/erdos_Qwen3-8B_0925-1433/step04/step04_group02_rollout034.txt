import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Generate initial sinusoidal candidates
    initial_candidates = [initial_h_values.copy()]
    for v in range(1, 9):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * v * x)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        initial_candidates.append(h_scaled)

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for candidate in initial_candidates:
        h = candidate.copy()

        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            sum_h = np.sum(h_vec)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h_vec = np.clip(h_vec * scale, 0, 1)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')

        bounds = [(0, 1) for _ in range(n_points)]
