        corr = np.correlate(h, 1 - h, mode='full')

            j = i + index_k
            if 0 <= j < n_points:
                gradient[i] += 1.0 - h[j]
            # Check for i - index_k
            m = i - index_k
            if 0 <= m < n_points:
                gradient[i] -= h[m]
        # Update h
        h = h - learning_rate * gradient
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Project to maintain sum
        sum_h = np.sum(h)
        delta = sum_h - (n_points / 2)
        if delta != 0:
            # Adjust the elements with highest gradient magnitudes
            gradient_abs = np.abs(gradient)
            sort_indices = np.argsort(gradient_abs)
            for i in sort_indices:
                if delta > 1e-6:
                    # Need to increase sum
                    if h[i] < 1:
                        adjust = min(1 - h[i], delta)
                        h[i] += adjust
                        delta -= adjust
                else:
                    # Need to decrease sum
                    if h[i] > 0:
                        adjust = min(h[i], -delta)
                        h[i] -= adjust
                        delta += adjust
                if abs(delta) < 1e-6:
                    break
        # Clip again after adjustment
        h = np.clip(h, 0, 1)

import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    sum_h = np.sum(h)
    assert np.isclose(sum_h, n_points / 2)

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.1
    iterations = 0
    max_iterations = 1000
    start_time = time.time()

    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        index_k = np.argmax(corr)
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = i + index_k
            if 0 <= j < n_points:
                gradient[i] += 1.0 - h[j]
            m = i - index_k
            if 0 <= m < n_points:
                gradient[i] -= h[m]
        
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        delta = sum_h - (n_points / 2)
        if delta != 0:
            gradient_abs = np.abs(gradient)
            sort_indices = np.argsort(gradient_abs)
            for i in sort_indices:
                if delta > 1e-6:
                    if h[i] < 1:
                        adjust = min(1 - h[i], delta)
                        h[i] += adjust
                        delta -= adjust

                    if h[i] > 0:
                        adjust = min(h[i], -delta)
