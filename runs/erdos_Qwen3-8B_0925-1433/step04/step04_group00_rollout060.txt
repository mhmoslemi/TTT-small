import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Initialize h with initial_h_values and scale to meet the integral constraint
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    
    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute cross-correlation and find the maximum lag
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        # Prepare optimization bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        # Objective function for this iteration
        def objective(x):
            cross_corr_new = np.correlate(x, 1 - x, mode='full')
            return cross_corr_new[max_index] * dx
        
        # Optimize
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0.0, 1.0)
            sum_h_new = np.sum(h_new)
            if sum_h_new != 0:
                scale = (n_points / 2) / sum_h_new
                h_new = np.clip(h_new * scale, 0.0, 1.0)
            else:
                h_new = np.ones(n_points) * 0.5
            h = h_new
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if best_c5 < 0.3807:
                break
        else:
            break  # Fallback to best solution if optimization fails
    
    return (best_h, best_c5, n_points)
