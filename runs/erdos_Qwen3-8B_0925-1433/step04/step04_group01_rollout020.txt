import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    pop = [initial_scaled.copy() for _ in range(10)]
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness and track max k
        fitness = []
        k_list = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)
            k_list.append(k)
            fitness.append(corr[max_idx] * dx)
        # Select top individuals
        top_idx = np.argsort(fitness)[:5]
        new_pop = []
        for i in top_idx:
            h = pop[i].copy()
            noise = np.random.normal(0, 0.03, n_points)
            mut = h + noise
            s = np.sum(mut)
            mut = np.clip(mut * target_sum / s, 0, 1)
            new_pop.append(mut)
        refined = []
        for h in new_pop:
            bounds = [(0, 1)] * n_points
            def obj(x):
                k = k_list[pop.index(h)]
                # Direct overlap computation for k
                overlap = 0.0
                for i in range(n_points):
                    x_val = dx * i
                    x_plus_k = x_val + k
                    if 0 <= x_plus_k <= 2:
                        j = int(np.floor(x_plus_k / dx))
                        if 0 <= j < n_points:

                        else:
                            overlap += x[i] * 1.0 * dx

                        overlap += x[i] * 1.0 * dx

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
