    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for _ in range(200):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if not ones.size or not zeros.size:
            break
        i, j = np.random.choice(ones), np.random.choice(zeros)
        best_h[i], best_h[j] = best_h[j], best_h[i]
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
        if best_c5 < 0.3807:
            break
    
    return (best_h, best_c5, n_points)
