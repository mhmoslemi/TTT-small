import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    h = initial_h_values.copy()
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    pop = [h.copy() for _ in range(20)]
    best_h, best_c5 = h.copy(), compute_c5(h, dx, n_points)
    time_start = time.time()
    T, cooling = 1.0, 0.995
    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        new_pop = []
        for ind in pop:
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = ind + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum > 0:
                perturbed = perturbed * target_sum / perturbed_sum
            new_pop.append(perturbed)
        for i in range(len(new_pop)):
            curr = compute_c5(new_pop[i], dx, n_points)
            if curr < best_c5:
                best_c5 = curr
                best_h = new_pop[i].copy()
        rand_idx = np.random.randint(len(new_pop))
        x = new_pop[rand_idx]
        bounds = [(0, 1)] * n_points
        def obj(x):
            corr = np.correlate(x, 1-x, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, x, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        pop[rand_idx] = res.x
        T *= cooling
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
