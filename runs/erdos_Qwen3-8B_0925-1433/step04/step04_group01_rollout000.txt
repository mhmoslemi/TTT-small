import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 50
    pop = [initial_h_values.copy() for _ in range(pop_size)]
    for _ in range(20):
        perturbed = initial_h_values + np.random.normal(0, 0.05, n_points)
        s = np.sum(perturbed)
        if s == 0:
            continue
        scaled = perturbed * (target_sum / s)
        clipped = np.clip(scaled, 0, 1)
        pop.append(clipped)
    pop = pop[:pop_size]

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            x1, x2, x3 = pop[a], pop[b], pop[c]
            mutant = x1 + 0.8 * (x2 - x3)
            trial = np.random.rand(n_points) < 0.9
            trial = np.where(trial, mutant, pop[i])
            s = np.sum(trial)
            if s == 0:
                continue
            trial = trial * (target_sum / s)
            trial = np.clip(trial, 0, 1)
            new_pop.append(trial)
        combined = pop + new_pop
        combined.sort(key=compute_c5)
        pop = combined[:pop_size]
        if compute_c5(pop[0]) < best_c5:
            best_h = pop[0].copy()
            best_c5 = compute_c5(best_h)

    # Local optimization
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    def cons(x):
        return np.sum(x) - target_sum
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}], tol=1e-8)
    best_h = res.x

    return (best_h, best_c5, n_points)
