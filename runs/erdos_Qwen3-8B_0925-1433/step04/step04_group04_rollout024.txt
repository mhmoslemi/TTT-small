import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        scale = 1.0
    else:
        scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2
    cons = [{'type': 'eq', 'fun': constraint}]

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = basinhopping(
        objective,
        h,
        niter=100,
        T=1.0,
        stepsize=0.1,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': cons
        }
    )

    best_h = result.x
    best_c5 = compute_c5(best_h, dx, n_points)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
