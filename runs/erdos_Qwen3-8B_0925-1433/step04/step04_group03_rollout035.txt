import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

        worst_k = np.argmax(corr)
        def objective(x):
            h_new = x.copy()
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            return corr[worst_k] * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1)] * n_points
        try:
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
