import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    candidates = []
    for _ in range(5):
        idx = np.random.permutation(n_points)
        bin_mask = np.zeros(n_points, dtype=bool)
        bin_mask[idx[:n_points//2]] = True
        h = np.zeros(n_points)
        h[bin_mask] = 1
        candidates.append(scale_h(h))
    for _ in range(5):
        freq = np.random.uniform(0.3, 1.5)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n_points) / n_points)
        candidates.append(scale_h(h))
    for _ in range(5):
        h = np.zeros(n_points)
        h[::2] = 1
        candidates.append(scale_h(h))
    for _ in range(5):
        h = np.zeros(n_points)
        h[::3] = 1
        candidates.append(scale_h(h))

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = min(compute_c5(c) for c in candidates)
    best_h = candidates[np.argmin([compute_c5(c) for c in candidates])].copy()

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(c) for c in candidates]
        elite_idx = np.argsort(c5_values)[:3]
        elites = [candidates[i] for i in elite_idx]
        new_candidates = []
        for i in range(len(elites)//2):
            p1, p2 = elites[2*i], elites[2*i+1]
            child = p1 * np.random.rand() + p2 * (1 - np.random.rand())
            new_candidates.append(scale_h(child))
        for p in elites:
            noise = np.random.normal(0, 0.05, n_points)
            new_candidates.append(scale_h(p + noise))
        for c in new_candidates:
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2})
            bounds = [(0,1) for _ in range(n_points)]
            res = minimize(lambda x: compute_c5(x), c, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                new_candidates.append(res.x)
        candidates = elites + new_candidates
        current_best = min(c5_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = candidates[np.argmin(c5_values)].copy()
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
