import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    h = initial_scaled.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Compute worst shift for current h
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        def obj(x):
            corr_x = np.correlate(x, 1 - x, mode='full')
            return corr_x[k] * dx
        # Local optimization
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='SLSQP', constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], bounds=bounds, tol=1e-6)
        h = res.x
        # Perturb and reoptimize
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + noise, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        k_perturbed = np.argmax(corr_perturbed)
        def obj_perturbed(x):
            corr_x = np.correlate(x, 1 - x, mode='full')
            return corr_x[k_perturbed] * dx
        res_perturbed = minimize(obj_perturbed, h_perturbed, method='SLSQP', constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], bounds=bounds, tol=1e-6)
        h = res_perturbed.x
        curr_c5 = compute_c5(h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
