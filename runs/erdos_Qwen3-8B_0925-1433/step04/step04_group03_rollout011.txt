def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial_h_values
    h_initial = initial_h_values.copy()
    sum_initial = np.sum(h_initial)
    scale = (n_points / 2) / sum_initial if sum_initial != 0 else 1.0
    h_initial = np.clip(h_initial * scale, 0, 1)

    # Generate a perturbed candidate with sinusoidal variation
    perturbation = 0.01 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    h_perturbed = np.clip(h_initial + perturbation, 0, 1)
    sum_perturbed = np.sum(h_perturbed)
    scale_perturbed = (n_points / 2) / sum_perturbed
    h_perturbed = np.clip(h_perturbed * scale_perturbed, 0, 1)

    # Generate a binary candidate
    h_binary = np.random.choice([0, 1], size=n_points)
    sum_binary = np.sum(h_binary)
    scale_binary = (n_points / 2) / sum_binary
    h_binary = np.clip(h_binary * scale_binary, 0, 1)

    # Initial candidates
    candidates = [h_initial, h_perturbed, h_binary]
    best_c5 = np.inf
    best_h = h_initial

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = np.clip(h * scale, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x, max_index, dx, n_points):
        h_opt = np.clip(x, 0, 1)
        sum_h = np.sum(h_opt)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_opt = np.clip(h_opt * scale, 0, 1)
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        return corr[max_index] * dx

    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_c5s = [compute_c5(candidate) for candidate in candidates]
        best_idx = np.argmin(current_c5s)
        current_best = candidates[best_idx]
        current_best_c5 = current_c5s[best_idx]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()

        if best_c5 < 0.3807:
            break

        h = current_best.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

        try:
            res = minimize(lambda x: objective(x, max_index, dx, n_points), 
                           current_best, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                new_h = res.x
                candidates[0] = new_h
        except:
            new_h = current_best + 0.01 * np.random.randn(n_points)
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            scale_new = (n_points / 2) / sum_new
            new_h = np.clip(new_h * scale_new, 0, 1)
            candidates[0] = new_h

    return (best_h, best_c5, n_points)
