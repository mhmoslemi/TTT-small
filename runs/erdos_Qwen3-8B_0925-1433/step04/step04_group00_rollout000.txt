    pbest_c5 = [compute_c5(h) for h in pbest]
    gbest = h_best.copy()
    gbest_c5 = best_c5
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            vel = velocities[i]
            vel = 0.7 * vel + 1.5 * np.random.rand(n_points) * (pbest[i] - particles[i]) + 1.5 * np.random.rand(n_points) * (gbest - particles[i])
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                particles[i] = np.clip(particles[i] * scale, 0, 1)
            else:
                particles[i] = np.ones(n_points) * 0.5
            current_c5 = compute_c5(particles[i])
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
                    if gbest_c5 < 0.3807:
                        return (gbest, gbest_c5, n_points)
        
        top_indices = np.argsort(pbest_c5)[:3]
        for i in top_indices:
            h = pbest[i].copy()
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5
            
            def objective(x):
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)
                if sum_x == 0:
                    return 1e6
                scale = (n_points / 2) / sum_x
                x = np.clip(x * scale, 0, 1)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                return np.max(corr) * dx
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                sum_h_opt = np.sum(h_opt)
                if sum_h_opt != 0:
                    scale = (n_points / 2) / sum_h_opt
                    h_opt = np.clip(h_opt * scale, 0, 1)
                else:
                    h_opt = np.ones(n_points) * 0.5
                particles[i] = h_opt.copy()
                current_c5 = compute_c5(h_opt)
                if current_c5 < pbest_c5[i]:
                    pbest[i] = h_opt.copy()
                    pbest_c5[i] = current_c5
                    if current_c5 < gbest_c5:
                        gbest = h_opt.copy()
                        gbest_c5 = current_c5
                        if gbest_c5 < 0.3807:
                            return (gbest, gbest_c5, n_points)
    
    return (gbest, gbest_c5, n_points)
