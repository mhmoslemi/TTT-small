                        new_h = best_h.copy()
                        new_h[i] = 0
                        new_h[j] = 1
                        new_overlaps = compute_lags(new_h)
                        new_max = np.max(new_overlaps)
                        delta = max_overlap - new_max
                        if delta > best_delta:
                            best_delta = delta
                            best_idx = (i, j)
        if best_idx == -1:
            break
        i, j = best_idx
        best_h[i], best_h[j] = best_h[j], best_h[i]

        current_c5 = compute_c5(best_h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        if best_c5 < 0.3807:
            break
        if time.time() - start_time > budget_s * 0.9:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_lags(h):
    n = len(h)
    overlaps = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        if k < 0:
            shifted = np.roll(h, k)
        else:
            shifted = np.roll(h, -k)
        overlaps[k + n - 1] = np.dot(h, 1 - shifted)
    return overlaps
