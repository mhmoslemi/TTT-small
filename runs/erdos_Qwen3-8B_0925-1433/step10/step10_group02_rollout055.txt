    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.8, 1), recombination=0.9, maxiter=200)

    a_current = [1.0] + best_coeffs[:N_terms]
    b_current = [0.0] + best_coeffs[N_terms:]
    h = 0.5 * np.ones(n_points)
    for m in range(1, N_terms + 1):
        h += a_current[m] * np.cos(m * np.pi * x) + b_current[m] * np.sin(m * np.pi * x)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h) * dx
    if sum_h == 0:
        return (np.zeros(n_points), float('inf'), n_points)
    h_scaled = h * (1.0 / sum_h)
    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
    c5 = np.max(corr) * dx
    return (h_scaled, c5, n_points)
