    initial_h.append(np.random.rand(n_points))
    initial_h[-1] *= target_sum / initial_h[-1].sum()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    best_h, best_c5 = initial_h[0].copy(), float('inf')

    for h in initial_h:
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_clipped = h_clipped * target_sum / sum_h
        result = minimize(obj, h_clipped, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (result.x, result.fun, n_points)
