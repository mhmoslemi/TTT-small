import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing
    initial_temperature = 0.2 * current_c5
    cooling_rate = 0.995
    sa_iterations = 1000
    start_time = time.time()
    h_sa = h.copy()
    
    for _ in range(sa_iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        candidate_h = h_sa + perturb
        candidate_h = np.clip(candidate_h, 0, 1)
        sum_candidate = np.sum(candidate_h)
        if sum_candidate > 0:
            candidate_h = candidate_h * target_sum / sum_candidate
        else:
            candidate_h = h_sa
        candidate_c5 = compute_c5(candidate_h)
        
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / initial_temperature):
            h_sa = candidate_h.copy()
            current_c5 = candidate_c5
        
        if current_c5 < best_c5:
            best_h = h_sa.copy()
            best_c5 = current_c5
        
        initial_temperature *= cooling_rate
    
    # Local refinement
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    return (best_h, best_c5, n_points)
