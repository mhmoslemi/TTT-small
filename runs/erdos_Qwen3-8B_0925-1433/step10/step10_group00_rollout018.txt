        lag = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_lag = (i + lag) % n_points
            i_minus_lag = (i - lag) % n_points
            grad[i] = (1 - current_h[i_plus_lag]) - current_h[i_minus_lag]

        updated_h = best_h - step_size * grad
        updated_h = np.clip(updated_h, 0, 1)
        sum_updated = np.sum(updated_h)
        if sum_updated > 0:
            updated_h = updated_h * target_sum / sum_updated

        corr_updated = np.correlate(updated_h, 1 - updated_h, mode='full')
        candidate_c5 = np.max(corr_updated) * dx

        if candidate_c5 < best_c5:
            best_h = updated_h.copy()
            best_c5 = candidate_c5
        else:
            step_size *= 0.5

        step_size *= step_decay
        iter_count += 1

    return (best_h, best_c5, n_points)
