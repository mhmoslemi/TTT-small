        return 0.5 * (1 + np.sin(np.pi * np.arange(n) / (n / 2)))

    base = [gen_uniform(n_points), gen_alt(n_points), gen_sine(n_points), initial_h]
    noisy = []
    for p in base:
        p = p + np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        s = np.sum(p)
        if s > 0:
            p = p * target_sum / s
        noisy.append(p)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start = time.time()
    for p in noisy:
        bounds = [(0, 1)] * n_points
        def obj(x):
            x = np.clip(x, 0, 1)
            s = np.sum(x)
            if s > 0:
                x = x * target_sum / s
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        try:
            res = differential_evolution(obj, bounds, strategy='best1bin',
                                        popsize=30, maxiter=100, tol=1e-8,
                                        mutation=(0.7, 1.0), recombination=0.8,
                                        seed=seed, disp=False)
        except:
            res = minimize(obj, p, bounds=bounds, method='L-BFGS-B', tol=1e-10)
        res_local = minimize(res.x, method='SLSQP', bounds=bounds,
                             constraints=LinearConstraint(np.ones(n_points), [target_sum], [target_sum]))
        if res_local.fun < best_c5:
            best_c5 = res_local.fun
            best_h = res_local.x
        if time.time() - start > budget_s - 100:
            break
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if s > 0:
        best_h = best_h * target_sum / s
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
