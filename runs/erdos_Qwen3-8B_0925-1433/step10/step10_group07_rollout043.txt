import time

        h_ = np.clip(h_vec, 0, 1)
        h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points

    # Genetic algorithm with mutation refinement

    gen_max = 100
    elite_frac = 0.2
    mut_strength = 0.1

    pop = np.array([v * target_sum / np.sum(v) if np.sum(v) > 0 else np.zeros(n_points) for v in pop])
    pop[0, :] = h
    
    best_h = h.copy()
    best_obj = obj(h)
    
    for gen in range(gen_max):
        if time.time() - start > budget_s - 60:
            break
        fitness = np.array([obj(ind) for ind in pop])

        elite_pop = pop[elite_idx]
        
        # Crossover
        new_pop = np.zeros_like(pop)

            child2 = b * elite_pop[p1] + (1 - b) * elite_pop[p2]
            new_pop[i] = child1

        # Mutation

            new_pop[i] = ind
        
        pop = new_pop
        cur_best = np.argmin(fitness)

    # SQP refinement
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
