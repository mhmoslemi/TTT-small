import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return np.inf
        x_norm = x_clipped * target_sum / sum_x
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    # Generate initial population
    initial_population = []
    parent = np.clip(initial_h_values, 0, 1)
    parent_sum = np.sum(parent)
    if parent_sum > 0:
        parent = parent * target_sum / parent_sum
    initial_population.append(parent)

    for _ in range(5):
        rand = np.random.uniform(0, 1, n_points)
        rand = np.clip(rand, 0, 1)
        rand_sum = np.sum(rand)
        if rand_sum > 0:
            rand = rand * target_sum / rand_sum
        initial_population.append(rand)

    for m in [3, 4, 5]:
        pattern = np.zeros(n_points)
        for i in range(n_points):
            pattern[i] = 1.0 if (i % m == 0) else 0.0
        pattern_sum = np.sum(pattern)
        if pattern_sum > 0:
            pattern = pattern * target_sum / pattern_sum
        initial_population.append(pattern)

    even = np.zeros(n_points)
    even[::2] = 1.0
    even_sum = np.sum(even)
    if even_sum > 0:
        even = even * target_sum / even_sum
    initial_population.append(even)

    for sigma in [0.1, 0.2]:
        perturb = np.random.normal(0, sigma, n_points)
        mod_parent = np.clip(parent + perturb, 0, 1)
        mod_parent_sum = np.sum(mod_parent)
        if mod_parent_sum > 0:
            mod_parent = mod_parent * target_sum / mod_parent_sum
        initial_population.append(mod_parent)

    initial_population = np.array(initial_population)
    best_h = initial_population[0]
    best_c5 = objective(best_h)

    # Differential evolution
    bounds = [(0, 1)] * n_points
    result_de = differential_evolution(objective, bounds, popsize=20, strategy='best1bin', maxiter=100, tol=1e-8, mutation=(0.5, 1), recombination=0.9)
    best_c5 = result_de.fun
    best_h = result_de.x

    # Local refinement with SLSQP
    def slsqp_refine(x0):
        def obj(x):
            x_clipped = np.clip(x, 0, 1)
            sum_x = np.sum(x_clipped)
            if sum_x == 0:
                return np.inf
            x_norm = x_clipped * target_sum / sum_x
            corr = np.correlate(x_norm, 1 - x_norm, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        return result.x, result.fun

    for _ in range(10):
        if time.time() - time.time() > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        refined_h, refined_c5 = slsqp_refine(new_h)
        if refined_c5 < best_c5:
            best_h, best_c5 = refined_h, refined_c5

    final_result = minimize(objective, best_h, method='SLSQP', bounds=bounds, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
