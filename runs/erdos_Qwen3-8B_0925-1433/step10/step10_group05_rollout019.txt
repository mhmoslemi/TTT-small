```python
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})

    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
```
