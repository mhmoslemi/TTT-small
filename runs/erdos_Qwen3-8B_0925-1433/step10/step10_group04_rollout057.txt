            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    final_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
