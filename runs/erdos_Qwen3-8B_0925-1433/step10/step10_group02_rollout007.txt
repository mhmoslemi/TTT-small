        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(
            obj, new_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-10
        )
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final evaluation
    final_res = minimize(
        obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-10
    )
    return (final_res.x, final_res.fun, n_points)
