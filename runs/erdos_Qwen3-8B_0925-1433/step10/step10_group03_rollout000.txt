                                           popsize=15, maxiter=120, tol=1e-8,
                                           recombination=0.9, mutation=0.2,
                                           polish=True, disp=False)
    except:
        result_de = minimize(obj, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-8)

    best_h_de = result_de.x
    best_c5_de = result_de.fun

    # Local refinement
    result_lbgf = minimize(obj, best_h_de, method='L-BFGS-B', bounds=bounds, tol=1e-8)
    best_h = result_lbgf.x
    best_c5 = result_lbgf.fun

    # Check initial guess
    initial_c5 = obj(h_initial)
    if initial_c5 < best_c5:
        best_h = h_initial
        best_c5 = initial_c5

    # Early termination
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
