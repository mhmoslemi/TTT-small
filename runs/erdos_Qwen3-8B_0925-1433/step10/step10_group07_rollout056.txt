import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return h * target_sum / s

    h = np.clip(initial_h_values, 0, 1)
    h = rescale_h(h)

    def obj_de(x):
        h_ = np.clip(x, 0, 1)
        h_ = rescale_h(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h
    best_c5 = obj_de(h)

    # Global search with DE
    try:
        result = differential_evolution(
            obj_de, bounds, strategy='rand1bin',
            popsize=50, maxiter=200, tol=1e-8,
            mutation=(0.8, 1.2), recombination=0.9,
            vectorized=True, disp=False
        )
        best_h = result.x
        best_c5 = result.fun
    except:
        pass

    # Local refinement with SLSQP
    try:
        def objective(x):
            h_ = np.clip(x, 0, 1)
            h_ = rescale_h(h_)
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(np.clip(x, 0, 1)) - target_sum

        cons = [{'type': 'eq', 'fun': constraint}]
        result = minimize(
            objective, best_h, method='SLSQP',
            bounds=bounds, constraints=cons,

        )
        best_h = result.x
        best_c5 = result.fun
    except:
        pass

    elapsed = time.time() - start_time

    # Validate
    h_valid = np.clip(best_h, 0, 1)
    h_valid = rescale_h(h_valid)
    final_c5 = np.max(np.correlate(h_valid, 1 - h_valid, mode='full')) * dx
    return (h_valid, final_c5, n_points)
