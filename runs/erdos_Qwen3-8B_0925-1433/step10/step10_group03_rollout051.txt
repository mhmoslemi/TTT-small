import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial_sum = np.sum(h_initial)

        sum_h = np.sum(h_)
        if sum_h == 0:
            return float('inf')
        h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    
    start_time = time.time()
    result = differential_evolution(
        obj, bounds, strategy='best1bin', 
        popsize=15, mutation=(0.8, 1.2), 
        recombination=0.9, tol=1e-6,

        disp=False,
