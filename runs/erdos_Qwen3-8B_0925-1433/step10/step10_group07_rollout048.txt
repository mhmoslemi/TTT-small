import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:

    else:

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    # Global search with adaptive DE
    try:
        result_de = differential_evolution(
            obj, bounds, strategy='rand1bin',
            popsize=50, maxiter=150, tol=1e-8,
            mutation=(0.7, 1.3), recombination=0.8,
            vectorized=True
        )
        best_h, best_c5 = result_de.x, result_de.fun
    except:
        best_h, best_c5 = h, obj(h)

    # Local refinement with SLSQP
    try:
        def objective(x):
            h_ = np.clip(x, 0, 1)
            if np.sum(h_) > 0:
                h_ = h_ * target_sum / np.sum(h_)
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(np.clip(x, 0, 1)) - target_sum

            constraints=cons,

        )
        best_h, best_c5 = res_slsqp.x, res_slsqp.fun
    except:

    elapsed = time.time() - start_time
