import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def _normalize(arr):
        s = np.sum(arr)
        return np.clip(arr * target_sum / max(1e-12, s), 0, 1)

    def _objective(x):
        h = np.clip(x, 0, 1)
        h = _normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def _generate_patterns():
        patterns = []
        # Even spacing
        even = np.zeros(n_points)
        even[::2] = 1
        patterns.append(_normalize(even))
        # Symmetric periodic (period 3)
        sym = np.zeros(n_points)
        for i in range(n_points):
            sym[i] = 1 if (i % 3 < 2) else 0
        patterns.append(_normalize(sym))
        # Modular repeat first half
        mod = np.repeat(initial_h_values[:n_points//2], 2)
        patterns.append(_normalize(mod))
        # Gaussian smoothed
        gaussian = np.convolve(initial_h_values, np.ones(5)/5, mode='same')
        patterns.append(_normalize(gaussian))
        # Original
        patterns.append(_normalize(initial_h_values))
        return patterns

    patterns = _generate_patterns()
    best_h = None
    best_c5 = np.inf
    start_time = time.time()

    # Initial optimization
    for h in patterns:
        if np.sum(h) == 0:
            continue
        res = minimize(_objective, h, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-10)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    # Refinement
    if best_h is None:
