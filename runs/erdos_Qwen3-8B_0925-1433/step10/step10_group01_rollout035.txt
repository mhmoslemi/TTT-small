import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values
    initial_h = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * target_sum / sum_initial
    else:
        initial_h = np.zeros(n_points)

    # Initialize population
    population = [initial_h.copy()]
    for _ in range(49):
        perturb = np.random.normal(0, 0.01, n_points)
        h = np.clip(initial_h + perturb, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)

    # Evaluate function
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # GA parameters
    num_generations = 100
    population_size = 50
    best_h = np.copy(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(num_generations):
        # Evaluate all in population
        fitness = [evaluate(h) for h in population]
        # Sort indices by fitness
        sorted_indices = np.argsort(fitness)
        top_candidates = [population[i] for i in sorted_indices[:10]]
        new_population = []

        while len(new_population) < population_size:
            # Crossover
            parent1 = np.random.choice(top_candidates)
            parent2 = np.random.choice(top_candidates)
            child = (parent1 + parent2) / 2
            # Mutation
            perturb = np.random.normal(0, 0.01, n_points)
            child += perturb

            sum_h = np.sum(child)
            if sum_h > 0:
                child = child * target_sum / sum_h
            new_population.append(child)
