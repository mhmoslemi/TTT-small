import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    population = [h.copy()]
    for _ in range(20):
        new_pop = []
        for _ in range(50):
            if np.random.rand() < 0.6:
                x = np.random.uniform(0, 1, n_points)
                x_sum = np.sum(x)
                x = x * target_sum / x_sum
                new_pop.append(x)
            else:
                idx = np.random.randint(len(population))
                x = population[idx].copy()
                delta = np.sum(x) - target_sum
                if abs(delta) < 1e-8:
                    new_pop.append(x)
                    continue
                i, j = np.random.choice(n_points, 2, replace=False)
                x[i] += 0.05 * delta
                x[j] -= 0.05 * delta
                x = np.clip(x, 0, 1)
                x_sum = np.sum(x)
                delta_new = x_sum - target_sum
                x += delta_new / n_points
                x = np.clip(x, 0, 1)
                new_pop.append(x)
        population += new_pop
        population.sort(key=obj)
        population = population[:50]

    best_h = population[0]
    best_c5 = obj(best_h)
    for h_candidate in population:
        c5 = obj(h_candidate)
        if c5 < best_c5:
            best_h, best_c5 = h_candidate, c5

    start_time = time.time()
    while time.time() - start_time < budget_s - 100:
        result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-10, options={'maxiter': 10})
        if result.success:
            best_h, best_c5 = result.x, result.fun
        else:
            break

    return (best_h, best_c5, n_points)
