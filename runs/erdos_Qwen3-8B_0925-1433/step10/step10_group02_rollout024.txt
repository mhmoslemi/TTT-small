```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def norm(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            return h_clipped * target_sum / sum_h
        return np.zeros_like(h_clipped)

    def obj(x):
        h_ = norm(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def opt(start):
        bounds = [(0, 1)] * n_points
        res = minimize(obj, start, method='SLSQP', bounds=bounds, tol=1e-10)
        return res.x, res.fun

    h0 = norm(initial_h_values)
    best, best_c5 = opt(h0)

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s - 100:
            break
        pert = np.random.normal(0, 0.1, n_points)
        new_h = norm(best + pert)
        res = opt(new_h)
        if res[1] < best_c5:
            best, best_c5 = res

    for _ in range(30):
        if time.time() - start_time > budget_s - 50:
            break
        pert = np.random.normal(0, 0.05, n_points)
        new_h = norm(best + pert)
        res = opt(new_h)
        if res[1] < best_c5:
            best, best_c5 = res

    return (best, best_c5, n_points)
```
