import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(h_initial)
    if sum_initial > 0:
        h_initial = h_initial * target_sum / sum_initial
    best_h = h_initial.copy()
    best_c5 = obj(best_h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    try:
        result_de = differential_evolution(
            obj, bounds, strategy='best1bin', popsize=50, maxiter=100,
            tol=1e-8, mutation=(0.5, 1), recombination=0.8,
            mutation_scale=0.5, vectorized=True
        )
        if result_de.fun < best_c5:
            best_h = result_de.x
            best_c5 = result_de.fun
    except:
        pass

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = tuple((0, 1) for _ in range(n_points))
    try:
        result_slsqp = minimize(
            obj, best_h, method='SLSQP', bounds=bounds_slsqp,
            constraints=cons, tol=1e-8, options={'maxiter': 200}
        )
        if result_slsqp.fun < best_c5:
            best_h = result_slsqp.x
            best_c5 = result_slsqp.fun
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        pass

    return (best_h, best_c5, n_points)
