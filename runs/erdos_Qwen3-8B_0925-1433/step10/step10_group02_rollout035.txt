import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    # Validate initial guess
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)

    # Generate structured patterns
    patterns = []
    even_pattern = np.zeros(n_points)
    even_pattern[::2] = 1.0
    even_pattern = np.clip(even_pattern, 0, 1)
    even_pattern = even_pattern * target_sum / np.sum(even_pattern)
    patterns.append(even_pattern)

    sym_pattern = np.zeros(n_points)
    sym_pattern[::3] = 1.0
    sym_pattern = np.clip(sym_pattern, 0, 1)
    sym_pattern = sym_pattern * target_sum / np.sum(sym_pattern)
    patterns.append(sym_pattern)

    mod_pattern = np.tile(initial_h_values[:n_points//2], 2)
    mod_pattern = np.clip(mod_pattern, 0, 1)
    mod_pattern = mod_pattern * target_sum / np.sum(mod_pattern)
    patterns.append(mod_pattern)

    def smooth_pattern(h):
        kernel = np.hanning(5)
        kernel = kernel / np.sum(kernel)
        return np.convolve(h, kernel, mode='same')
    smoothed = smooth_pattern(h_initial)
    smoothed = np.clip(smoothed, 0, 1)
    smoothed = smoothed * target_sum / np.sum(smoothed)
    patterns.append(smoothed)

    best_c5 = float('inf')
    best_h = h_initial

    # Initial optimization
    for pattern in patterns:
        res = minimize(obj, pattern, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-10)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    # Perturbation refinement
    start_time = time.time()
    for _ in range(3):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=[(0,1)]*n_points,

            best_c5 = res.fun
            best_h = res.x

                         constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-10)
    return (final_res.x, final_res.fun, n_points)
