import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured initial guess
    def create_structured(n):
        structured = np.zeros(n)
        structured[::2] = 1.0
        current_sum = np.sum(structured)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            structured[-1] += diff
            structured[-1] = np.clip(structured[-1], 0, 1)
        return structured

    initial_guesses = [
        np.copy(initial_h_values),
        create_structured(n_points),
        np.random.rand(n_points)
    ]

    # Rescale to target sum
    def rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            return h * target_sum / np.sum(h)
        return h

    def obj(x):
        h_ = rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(rescale(x)) - target_sum}]

    best_h, best_c5 = None, float('inf')
    start_time = time.time()

    # Initial optimization
    for guess in initial_guesses:
        res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Perturbations
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        h_perturbed = best_h + np.random.normal(0, 0.03, n_points)
        h_perturbed = rescale(h_perturbed)
        res = minimize(obj, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    final_res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    return (final_res.x, final_res.fun, n_points)
