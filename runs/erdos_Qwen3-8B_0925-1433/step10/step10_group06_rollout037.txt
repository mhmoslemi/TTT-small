    # Generate the final h_values
    x = np.linspace(0, 2, n_points)
    h_values = 0.5 * np.sin(2 * np.pi * 1 * x + best_phi) + 0.5
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if sum_h > 0:
        h_values = h_values * target_sum / sum_h
    return (h_values, best_c5, n_points)
