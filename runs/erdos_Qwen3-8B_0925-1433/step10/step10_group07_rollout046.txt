import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial construction
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def compute_c5(h_):
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def mutate(ind):
        delta = np.sum(ind) - target_sum
        i, j = np.random.randint(n_points), np.random.randint(n_points)
        ind[i] += 0.05 * delta
        ind[j] -= 0.05 * delta
        ind = np.clip(ind, 0, 1)
        new_sum = np.sum(ind)
        error = new_sum - target_sum
        ind += error / n_points
        ind = np.clip(ind, 0, 1)
        return ind

    # Genetic Algorithm
    pop_size = 50
    generations = 200
    mutation_rate = 0.7
    population = [h.copy() for _ in range(pop_size)]
    for _ in range(generations):
        fitness = np.array([compute_c5(p) for p in population])
        parents = np.argsort(fitness)[:pop_size//2]
        new_pop = []
        for i in range(pop_size//2):
            p1, p2 = population[parents[i]], population[parents[pop_size//2-i-1]]
            child = (p1 + p2) / 2
            if np.random.rand() < mutation_rate:
                child = mutate(child.copy())
            new_pop.append(child)
        population = new_pop

    # Local refinement with SLSQP
    best_h = min(population, key=compute_c5)
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if abs(sum_x - target_sum) > 1e-8:
            return compute_c5(x) + 1e6 * abs(sum_x - target_sum)
        return compute_c5(x)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 150})
    best_h = result.x
    best_c5 = compute_c5(best_h)

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else np.zeros(n_points)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
