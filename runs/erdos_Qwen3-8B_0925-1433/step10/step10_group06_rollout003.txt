from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = h_initial * target_sum / np.sum(h_initial)

        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def smooth_objective(h, epsilon=0.01):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        O_values = corr * dx
        sum_exp = np.sum(np.exp(-epsilon * O_values))
        return (1.0 / epsilon) * np.log(sum_exp)

    # Generate initial diversity
    candidates = [h_initial.copy() for _ in range(5)]

        candidates[i] = np.clip(h_initial + noise, 0, 1)
        candidates[i] = candidates[i] * target_sum / np.sum(candidates[i])

    best_h = h_initial
    best_c5 = objective_func(h_initial)

    for candidate in candidates:
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
        bounds = Bounds(0, 1)
        result = minimize(smooth_objective, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
