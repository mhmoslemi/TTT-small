import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h == 0:
            return np.inf
        h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, maxiter=300, tol=1e-10, seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        def obj_for_slsqp(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h == 0:
                return np.inf
            h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        bounds_slsqp = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result_slsqp = minimize(obj_for_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp,
                               constraints=cons, tol=1e-10, options={'maxiter': 200})
        if result_slsqp.fun < best_c5:
            best_h, best_c5 = result_slsqp.x, result_slsqp.fun

    final_res = minimize(obj_for_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp,
                         constraints=cons, tol=1e-10, options={'maxiter': 200})
    return (final_res.x, final_res.fun, n_points)
