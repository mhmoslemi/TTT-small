import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1).copy()
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return 1e6
        x_norm = x_clipped * target_sum / sum_x
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)

    if time.time() - start_time < budget_s - 200:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', 
            popsize=15, tol=1e-8, mutation=(0.4, 0.9), 
            recombination=0.9, maxiter=100, polish=True
        )
        best_h = result.x
        best_c5 = result.fun

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_lbgf = minimize(
        objective, best_h, method='L-BFGS-B', bounds=bounds, 
        constraints=cons, tol=1e-10, options={'maxiter': 200}
    )
    best_h = result_lbgf.x
    best_c5 = result_lbgf.fun

    return (best_h, best_c5, n_points)
