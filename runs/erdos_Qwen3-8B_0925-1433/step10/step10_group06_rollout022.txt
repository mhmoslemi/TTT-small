import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values to satisfy the integral constraint
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def objective(h_vec):
        h = h_vec.copy()
        # Compute the full cross-correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # Smooth approximation of max(corr)
        eps = 0.01
        sum_exp = np.sum(np.exp(eps * corr))
        return (1 / eps) * np.log(sum_exp)

    # Equality constraint: sum(h) == target_sum
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    initial_guess = h.copy()

    # Optimization with SLSQP
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

    # Final refinement with tighter epsilon to improve accuracy
    final_h = result.x
    final_h = np.clip(final_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final > 0:
        final_h = final_h * target_sum / sum_final
    else:
        final_h = np.zeros(n_points)

    # Evaluate exact C₅
    corr_exact = np.correlate(final_h, 1 - final_h, mode='full') * dx
    c5_bound = np.max(corr_exact)

    return (final_h, c5_bound, n_points)
