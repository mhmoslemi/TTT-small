import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 1e-12:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points) + 0.5

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.1 * best_c5
    alpha = 0.995
    max_iters = 1000
    start_time = time.time()

    for _ in range(max_iters):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        candidate = h + perturb
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > 1e-12:
            candidate = candidate * target_sum / sum_candidate
        else:
            candidate = np.zeros(n_points) + 0.5
        c5_candidate = compute_c5(candidate)
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = candidate.copy()
            if delta < 0:
                best_c5 = c5_candidate
                best_h = h.copy()
        T *= alpha

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 1e-12:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 1e-12:
        best_h = best_h * target_sum / sum_best
    return (best_h, best_c5, n_points)
