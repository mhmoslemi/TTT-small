def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def generate_patterns(initial_h):
        patterns = []
        # Even spacing
        even = np.zeros(n_points)
        even[::2] = 1
        even = even * target_sum / np.sum(even)
        patterns.append(even)
        # Symmetric periodic
        symmetric = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                symmetric[i] = 1
            else:
                symmetric[i] = 0
        symmetric = symmetric * target_sum / np.sum(symmetric)
        patterns.append(symmetric)
        # Modular repeat
        mod_half = n_points // 2
        modular = np.tile(initial_h[:mod_half], 2)
        modular = modular * target_sum / np.sum(modular)
        patterns.append(modular)
        # Gaussian smoothed
        sigma = 0.2
        kernel = np.exp(- (np.arange(-5, 6)) ** 2 / (2 * sigma**2)) / (np.sqrt(2 * np.pi) * sigma)
        kernel = kernel / np.sum(kernel)
        smoothed = np.convolve(initial_h, kernel, mode='same')
        smoothed = np.clip(smoothed, 0, 1)

        return patterns

    best_c5 = float('inf')
    best_h = None

    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    start_time = time.time()
    for _ in range(100):
