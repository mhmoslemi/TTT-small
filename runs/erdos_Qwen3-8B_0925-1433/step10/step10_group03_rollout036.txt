import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros(n_points)
        return h * target_sum / sum_h

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def smooth_h(h, sigma):
        h = np.clip(h, 0, 1)
        kernel = np.exp(-np.linspace(-sigma, sigma, 16)**2 / 2)
        kernel /= np.sum(kernel)
        h = np.convolve(h, kernel, mode='full')[n_points-1:2*n_points-1]
        return scale_h(h)

    # Initialize population
    population = [scale_h(initial_h_values)]
    for _ in range(49):
        candidate = np.random.rand(n_points)
        population.append(scale_h(candidate))

    best_h = scale_h(initial_h_values)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        # Evaluate fitness
        fitness = np.array([evaluate(h) for h in population])
        elites = population[np.argsort(fitness)[:int(len(population)*0.2)]]
        # Tournament selection
        parents = []
        for _ in range(len(population)):
            candidates = np.random.choice(population, 5, replace=False)
            parents.append(candidates[np.argmin([evaluate(h) for h in candidates])])
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            idx = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:idx], p2[idx:]])
            offspring.append(child)
        # Mutation
        for h in offspring:
            mask = np.random.rand(n_points) < 0.1
            h[mask] += np.random.uniform(-0.1, 0.1, np.sum(mask))
            h = scale_h(h)
        # Replace
        new_population = elites + offspring
        population = np.array([h for h in new_population if h is not None])
        # Update best
        current_best = population[np.argmin([evaluate(h) for h in population])]
        if evaluate(current_best) < best_c5:
            best_h = current_best
            best_c5 = evaluate(current_best)

    # Final refinement
    best_h = smooth_h(best_h, n_points / 12)
    return (best_h, best_c5, n_points)
