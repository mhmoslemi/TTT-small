        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = obj(current_h)

    T = 0.1 * best_c5
    cooling_rate = 0.995
    start_time = time.time()

    while True:
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        candidate = current_h + perturb
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > 0:
            candidate = candidate * target_sum / sum_candidate
        else:
            candidate = current_h.copy()
        new_c5 = obj(candidate)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = candidate.copy()
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
        T *= cooling_rate

    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h, best_c5 = res.x, res.fun
    return (best_h, best_c5, n_points)
