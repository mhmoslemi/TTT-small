import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with initial_h_values and perturbation
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(100)]
    for h in population:
        # Rescale to meet target_sum
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h[:] = 0.5 * np.ones(n_points)  # Fallback to uniform if sum is zero

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(200):  # Reduce iterations to save time
        if time.time() - start_time > budget_s - 50:
            break
        # Evaluate all candidates
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        # Select parents
        parents = []
        for _ in range(50):  # Generate 50 children
            idxs = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.7:
                # Crossover with adaptive alpha
                alpha = np.random.uniform(0.1, 0.9)  # Wider range for more diversity
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Mutation with adaptive step
                noise = np.random.normal(0, 0.1, n_points)  # Larger initial step
                child = parent1 + noise
            # Clip and rescale
            child = np.clip(child, 0, 1)
            sum_ch = np.sum(child)
            if sum_ch > 0:
                child = child * target_sum / sum_ch
            parents.append(child)

        # Replace worst half with offspring
        population.sort(key=lambda x: evaluate(x))
        population = population[:50] + parents
        population = population[:100]  # Ensure population size is 100

    # Final refinement with local optimization
    # Use SLSQP with equality constraint
    def objective(x):
        h = x.copy()
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=[(0, 1)] * n_points,
