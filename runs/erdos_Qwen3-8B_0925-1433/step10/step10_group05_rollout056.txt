import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def objective(x):
        x_clip = np.clip(x, 0, 1)
        x_norm = x_clip * target_sum / np.sum(x_clip)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 0:
        best_h = h
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
        return (h, best_c5, n_points)

    try:
        result_de = differential_evolution(
            objective, bounds, strategy='rand1bin', popsize=50,
            maxiter=300, tol=1e-8, mutation=(0.7, 1.0), recombination=0.8,
            seed=seed, disp=False, timeout=remaining_time
        )
    except:
        result_de = minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-10)

    best_h = result_de.x
    best_c5 = result_de.fun

    def obj_lbgfs(x):
        x_clip = np.clip(x, 0, 1)
        x_norm = x_clip * target_sum / np.sum(x_clip)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds_lbgfs = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_lbgfs = minimize(
        obj_lbgfs, best_h, method='SLSQP', bounds=bounds_lbgfs,
        constraints=cons, tol=1e-10, options={'maxiter': 100, 'ftol': 1e-9}
    )

    h_final = np.clip(result_lbgfs.x, 0, 1)
    h_sum_final = np.sum(h_final)
    if h_sum_final > 0:
        h_final = h_final * target_sum / h_sum_final
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (h_final, final_c5, n_points)
