import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            return h * target_sum / s
        return np.zeros(n_points)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h):
        h = h.copy()
        delta = np.sum(h) - target_sum
        if abs(delta) < 1e-10:
            return h
        i, j = np.random.choice(n_points, 2, replace=False)
        h[i] += 0.05 * delta
        h[j] -= 0.05 * delta
        h = np.clip(h, 0, 1)
        new_sum = np.sum(h)
        delta_new = new_sum - target_sum
        if abs(delta_new) > 1e-10:
            h += delta_new / n_points
            h = np.clip(h, 0, 1)
        return h
    
    def crossover(a, b):
        return 0.7 * a + 0.3 * b
    
    h_pop = [rescale(initial_h_values)]
    for _ in range(49):
        h_rand = np.random.rand(n_points)
        h_pop.append(rescale(h_rand))
    h_pop = np.array(h_pop)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 100:
            break
        fit = np.array([compute_c5(h) for h in h_pop])
        best_idx = np.argmin(fit)
        best_h = h_pop[best_idx]
        
        if np.random.rand() < 0.8:
            new_h = mutate(best_h)
            h_pop = np.vstack((h_pop, new_h))
        
        if len(h_pop) > 50:
            h_pop = np.vstack((h_pop[:50], np.random.choice(h_pop, 50, replace=False)))
    
    h_pop = np.array(h_pop)
    best_idx = np.argmin([compute_c5(h) for h in h_pop])
    best_h = h_pop[best_idx]
    
    def objective(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return compute_c5(h)
    
    try:
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-10, options={'maxiter':100})
        best_h = result.x
    except:
        pass
    
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
