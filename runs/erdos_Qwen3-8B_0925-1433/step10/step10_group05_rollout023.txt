                h_final = h_final * target_sum / np.sum(h_final)
                corr = np.correlate(h_final, 1 - h_final, mode='full')
                current_c5 = np.max(corr) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_final
        if time.time() - start_time > budget_s:
            break

    if best_h is None:
        h_final = initial_h_values.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * target_sum / np.sum(h_final)
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        best_c5 = np.max(corr) * dx
        best_h = h_final

    return (best_h, best_c5, n_points)
