import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        s = np.sum(h)
        return h * target_sum / s if s > 0 else h

    # Initialize population
    population = []
    h0 = np.clip(initial_h_values, 0, 1)
    h0 = rescale(h0)
    for _ in range(50):
        perturb = np.random.normal(0, 0.01, n_points)
        h = np.clip(h0 + perturb, 0, 1)
        h = rescale(h)
        population.append(h)

    best_h = h0.copy()
    best_c5 = evaluate(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(h) for h in population]
        idxs = np.argsort(scores)
        top10 = population[idxs[:10]]
        new_pop = []
        for _ in range(50):
            p1, p2 = np.random.choice(top10, 2, replace=False)
            child = (p1 + p2) / 2
            perturb = np.random.normal(0, 0.01, n_points)
            child += perturb
            child = np.clip(child, 0, 1)
            child = rescale(child)
            new_pop.append(child)
        population = new_pop
        current_best = population[np.argmin(scores)]
        if evaluate(current_best) < best_c5:
            best_h, best_c5 = current_best, evaluate(current_best)

    start_time = time.time()
    def obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return evaluate(h)
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h, best_c5 = res.x, res.fun
    return (best_h, best_c5, n_points)
