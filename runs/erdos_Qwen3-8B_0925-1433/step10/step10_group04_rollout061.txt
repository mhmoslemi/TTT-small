            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        for _ in range(5):
            if time.time() - start_time > budget_s - 50:
                break
            perturb_mask = np.random.choice(n_points, size=5, replace=False)
            perturb = np.zeros(n_points)
            perturb[perturb_mask] = np.random.uniform(-0.05, 0.05, size=5)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            result = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()

    final_result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    return (final_result.x, final_result.fun, n_points)
