import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population from initial_h_values with variation
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(100)]
    for h in population:
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        if np.sum(h_clip) > 0:
            h = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        # Select parents
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.7:
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1 + np.random.normal(0, 0.05, n_points)
            child = np.clip(child, 0, 1)
            if np.sum(child) > 0:
                child = child * target_sum / np.sum(child)
            parents.append(child)

        # Replace worst 50% with offspring
        population[50:] = parents
        population.sort(key=lambda x: evaluate(x))
        population = population[:100]

    # Final refinement with L-BFGS-B
    result = minimize(lambda x: evaluate(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-10)
