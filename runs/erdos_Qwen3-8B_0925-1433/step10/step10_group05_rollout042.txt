import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def norm(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            return x * target_sum / s
        return np.zeros_like(x)

    h = norm(initial_h_values)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8})
    best_h = result.x
    best_c5 = result.fun

    for _ in range(3):
        if np.random.rand() < 0.5:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturb = norm(best_h + perturb)
        else:
            h_perturb = best_h
        try:
            res = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8})
            if res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            continue

    h_final = norm(best_h)
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (h_final, final_c5, n_points)
