import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.random.rand(45) * 2 - 1
        h_initial = np.clip(h_initial, 0, 1)

    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) > 0:
            h_clipped = h_clipped * target_sum / np.sum(h_clipped)
        return h_clipped

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    pop = [h_initial + np.random.normal(0, 0.01, n_points) for _ in range(pop_size - 1)]
    pop = [h_initial] + pop
    pop = [clip_rescale(p) for p in pop]

    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fit = [evaluate(p) for p in pop]
        idx = np.argsort(fit)
        top = [pop[i] for i in idx[:10]]
        new_pop = []
        for _ in range(5):
            a, b = np.random.choice(top, 2, replace=False)
            child = (a + b) / 2
            new_pop.append(clip_rescale(child))
        for _ in range(5):

            c5 = evaluate(h)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
