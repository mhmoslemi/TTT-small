import numpy as np
from scipy.optimize import minimize
from scipy.stats import latinhypercube
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_vec(v):
        return np.clip(v, 0, 1) * target_sum / np.sum(np.clip(v, 0, 1)) if np.sum(np.clip(v, 0, 1)) > 0 else v

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def smooth_h(h, sigma):
        return np.convolve(h, np.exp(-np.arange(-sigma, sigma+1)**2 / (2*sigma**2)), mode='same')

    def generate_population(size):
        pop = []
        for _ in range(size):
            if np.random.random() < 0.7:
                v = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
            else:
                v = np.random.uniform(0, 1, n_points) if np.random.random() < 0.5 else latinhypercube.rvs(n_points)
            pop.append(scale_vec(v))
        return pop

    initial_pop = generate_population(50)
    best_h = scale_vec(initial_h_values)
    best_c5 = obj(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = np.array([obj(ind) for ind in initial_pop])
        elites = np.array([initial_pop[i] for i in np.argsort(fitness)[:int(len(initial_pop)*0.2)]])

        new_pop = []

            p1, p2 = np.random.choice(parents, 2, replace=False)
            idx = np.random.randint(n_points)
            child = np.concatenate([p1[:idx], p2[idx:]])
            child = scale_vec(child)
            new_pop.append(child)
        new_pop += elites
        initial_pop = new_pop

        for i in range(len(initial_pop)):

        current_best = min(zip(initial_pop, fitness), key=lambda x: x[1])
