import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    def gen_spaced(n_blocks=5):
        block_len = 1.0 / n_blocks
        spacing = (2.0 - n_blocks * block_len) / (n_blocks + 1)
        h = np.zeros(n_points)
        for i in range(n_blocks):
            start = i * (block_len + spacing) + spacing
            end = start + block_len
            for k in range(n_points):
                if start <= k * dx <= end:
                    h[k] = 1
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        return h

    def gen_periodic(n_blocks=5, shift=0.5):
        h = np.zeros(n_points)
        for i in range(n_blocks):
            start = i * (1.0 / n_blocks) + shift
            end = start + 1.0 / n_blocks
            for k in range(n_points):
                if start <= k * dx <= end:
                    h[k] = 1
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        return h

    initial_points = [initial_h_values.copy()]
    initial_points.append(gen_spaced(5))
    initial_points.append(gen_periodic(5))

    for init in initial_points:
        init = np.clip(init, 0, 1)
        init_sum = np.sum(init)
        if init_sum > 0:
            init = init * target_sum / init_sum
        result = minimize(objective, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    start_time = time.time()
    while time.time() - start_time < budget_s - 50:
        idx = np.random.choice(n_points)
        perturbation = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturbation, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * target_sum / new_sum
        result = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    final_result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    return (final_result.x, final_result.fun, n_points)
