import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_points = [initial_h_values.copy()]

    # Regular pattern: two intervals of 0.5 on [0, 0.5] and [1.5, 2.0]
    regular_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5 or x >= 1.5:
            regular_h[i] = 1
    sum_regular = np.sum(regular_h)
    scale_factor = target_sum / sum_regular
    regular_h = regular_h * scale_factor
    initial_points.append(regular_h)

    # Random initial
    random_h = np.random.rand(n_points)
    sum_random = np.sum(random_h)
    scale_factor = target_sum / sum_random
    random_h = random_h * scale_factor
    initial_points.append(random_h)

    best_h = initial_points[0].copy()
    best_c5 = obj(best_h)

    for h in initial_points:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.04, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    final_opt = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_opt.x, final_opt.fun, n_points)
