import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):

        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')

    bounds = [(0.0, 1.0)] * n_points

    # Differential evolution for global search
    result_de = differential_evolution(obj, bounds, strategy='best1bin',
                                      popsize=15, tol=1e-8, mutation=(0.5, 1),
                                      recombination=0.8, maxiter=150, seed=seed)

    best_candidate = result_de.x
    best_c5 = result_de.fun

    # Local refinement with L-BFGS-B
    def constraint(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]

                           bounds=bounds, constraints=cons, tol=1e-10)

    if result_lbgf.success:
