    h_norm = h * target_sum / np.sum(h)

    start_time = time.time()

        h_start = np.clip(h_start, 0, 1)

        def objective(x):

            x_norm = x_clip * target_sum / np.sum(x_clip)
            corr = np.correlate(x_norm, 1 - x_norm, mode='full')
            return np.max(corr) * dx

        except:
            result_de = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, tol=1e-10)

        def obj_lbgfs(x):

            x_norm = x_clip * target_sum / np.sum(x_clip)
