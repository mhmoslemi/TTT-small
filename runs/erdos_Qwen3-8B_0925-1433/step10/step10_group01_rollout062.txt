import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    h = h * target_sum / (sum_h if sum_h > 0 else 1)

    def compute_c5(h_candidate):
        h_ = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # SA parameters
    initial_temp = 0.1 * current_c5
    temp = initial_temp
    alpha = 0.995
    sa_iterations = 1000
    start_time = time.time()

    for _ in range(sa_iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, temp * 0.05, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > 0:
            candidate = candidate * target_sum / sum_candidate
        candidate_c5 = compute_c5(candidate)
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = candidate.copy()
            current_c5 = candidate_c5
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        temp *= alpha

    # Local refinement with SLSQP
    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, maxiter=100)
        if result.success:
            best_h = result.x
            best_c5 = result.fun
    except:
        pass

    # Final clipping and rescaling
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    else:
        best_h = np.zeros(n_points) + 0.5

    return (best_h, best_c5, n_points)
