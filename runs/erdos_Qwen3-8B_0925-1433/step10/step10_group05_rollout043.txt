import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,

    )

    best_h = result.x
