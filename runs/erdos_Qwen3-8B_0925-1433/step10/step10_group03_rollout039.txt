import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        return h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial population
    h_pop = []
    h_pop.append(initial_h_values)
    for _ in range(49):
        if np.random.rand() < 0.4:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.linspace(0, 1, n_points)
        h_pop.append(scale_h(h))

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        # Evaluate fitness
        fitness = [obj(h) for h in h_pop]
        # Select elites
        elite_idx = np.argsort(fitness)[:int(len(h_pop)*0.2)]
        elites = [h_pop[i] for i in elite_idx]
        # Tournament selection
        new_pop = elites[:]
        for _ in range(len(h_pop) - len(elites)):
            p1, p2 = np.random.choice(len(h_pop), 2, replace=False)
            if fitness[p1] < fitness[p2]:
                parent = h_pop[p1]
            else:
                parent = h_pop[p2]
            # Crossover
            idx = np.random.randint(1, n_points-1)
            child = np.concatenate([parent[:idx], h_pop[np.random.choice(len(h_pop))][idx:]])
            # Mutation
            mut = np.random.normal(0, 0.1, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child = scale_h(child)
            new_pop.append(child)
        h_pop = new_pop
        # Update best
        current_best = min(zip(fitness, h_pop), key=lambda x: x[0])[1]
        if obj(current_best) < best_c5:
            best_h = current_best
            best_c5 = obj(current_best)
    # Local refinement
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    if res.success:
        best_h = res.x
    return (best_h, best_c5, n_points)
