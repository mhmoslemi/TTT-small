import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return h

    def obj(x):
        h = rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    pop_size = 50
    population = [rescale(initial_h_values + np.random.normal(0, 0.01, n_points)) for _ in range(pop_size)]
    best_h = min(population, key=obj)
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [obj(ind) for ind in population]
        elite = [best_h] + [population[i] for i in np.argsort(fitness)[:9]]
        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = (p1 + p2) / 2
            new_pop.append(rescale(child))
            if np.random.rand() < 0.1:
                mut = np.random.normal(0, 0.01, n_points)
                new_pop.append(rescale(child + mut))
        population = new_pop + [population[i] for i in np.argsort(fitness)[::-1][:pop_size // 2]]
        if obj(population[0]) < best_c5:
            best_h, best_c5 = population[0], obj(population[0])

    if time.time() - start_time < budget_s - 10:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
