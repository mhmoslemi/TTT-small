import numpy as np

import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    initial_guesses.append(periodic)
    # Random uniform
    uniform = np.random.rand(n_points)
