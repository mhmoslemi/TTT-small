import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        n_points = 50
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0

    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * T / np.sum(h)
    else:
        h = np.zeros(n_points)
        h[0] = T

    def obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return np.inf
        x = x * T / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    try:
        result_de = differential_evolution(
            obj, bounds, strategy='best1bin',
            popsize=50, maxiter=200, tol=1e-8, vectorized=True,
            mutation=(0.5, 1), recombination=0.8, mutation_scale=0.5
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        best_h = h
        best_c5 = obj(h)

    try:
        result = minimize(
            obj, best_h, method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - T},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result.x
        best_c5 = result.fun
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        pass

    return (best_h, best_c5, n_points)
