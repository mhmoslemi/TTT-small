        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]

        A_current = 1 - h
        grad = np.roll(A_current, -max_k) - np.roll(h, max_k)

        h -= step_size * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h

        A_new = 1 - h
        overlaps_new = np.zeros(n_points)
        for k in range(n_points):
            overlaps_new[k] = np.dot(h, np.roll(A_new, k)) * dx
        new_c5 = np.max(overlaps_new)

        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            early_stop = 0
        else:
            early_stop += 1

        if early_stop >= 20:
            break

        step_size *= decay_rate

    A_final = 1 - best_h
    overlaps_final = np.zeros(n_points)
    for k in range(n_points):
        overlaps_final[k] = np.dot(best_h, np.roll(A_final, k)) * dx
    best_c5 = np.max(overlaps_final)

    return (best_h, best_c5, n_points)
