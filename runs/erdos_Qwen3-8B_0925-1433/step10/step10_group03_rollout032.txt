import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    # Differential evolution with early stopping
    try:
        result = differential_evolution(
            obj, bounds, strategy='best1bin', popsize=15,
            tol=1e-8, mutation=(0.5, 1), recombination=0.8, maxiter=100
        )
        best_h_de = result.x
        best_c5_de = result.fun
        if best_c5_de < best_c5:
            best_h = best_h_de
            best_c5 = best_c5_de
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 100:
            return (best_h, best_c5, n_points)
    except:
        pass

    # L-BFGS-B refinement
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    if res_lbgf.success and res_lbgf.fun < best_c5:
        best_h = res_lbgf.x
        best_c5 = res_lbgf.fun

    return (best_h, best_c5, n_points)
