import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Scale initial guess to meet sum constraint
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    if np.sum(h_initial) > 0:
        h_initial = h_initial * target_sum / np.sum(h_initial)
    else:
        h_initial = np.full(n_points, 0.5)

    # Objective function with normalization
    def objective(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Differential evolution parameters
    bounds = [(0, 1)] * n_points
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15,
        mutation=(0.5, 1), recombination=0.9, tol=1e-8,
        maxiter=100, seed=seed, polish=True, disp=False
    )
    best_h = result.x
    best_c5 = result.fun

    # Local refinement with L-BFGS-B
    def objective_refine(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_lbgf = minimize(
        objective_refine, best_h, method='L-BFGS-B',
        bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-8
    )
    best_h = res_lbgf.x
    best_c5 = res_lbgf.fun

    return (best_h, best_c5, n_points)
