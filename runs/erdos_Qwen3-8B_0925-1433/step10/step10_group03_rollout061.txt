            h_vals = h_vals * target_sum / np.sum(h_vals)

        return np.max(corr) * dx

    def smooth_h(h_vals, sigma):
        kernel = np.exp(-0.5 * np.arange(-sigma, sigma + 1)**2 / sigma**2)
        kernel = kernel / np.sum(kernel)
        return np.convolve(h_vals, kernel, mode='same')
