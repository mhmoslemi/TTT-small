from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):

        return h * target_sum / s if s > 0 else np.zeros(n_points)

    # Initialize population with initial_h_values and perturbations
    npop = 100
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(npop)]
    for h in population:
        h = rescale(h)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        # Selection and crossover
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[idxs[0]], population[idxs[1]]
