import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

        corr = np.correlate(x, 1 - x, mode='full')
        O = corr * dx
        return np.log(np.sum(np.exp(-0.01 * O))) / 0.01
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons)
    corr_exact = np.correlate(result.x, 1 - result.x, mode='full')
    exact_c5 = np.max(corr_exact) * dx
