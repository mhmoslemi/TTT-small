    a_opt = final_params[:N_terms]
    b_opt = final_params[N_terms:]
    h_final = 0.5
    for i in range(N_terms):
        h_final += a_opt[i] * np.cos((i+1)*np.pi*x) + b_opt[i] * np.sin((i+1)*np.pi*x)
    h_final = np.clip(h_final, 0, 1)
    sum_h = np.sum(h_final)
    if sum_h > 0:
        h_final *= (n_points / 2.0) / sum_h
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5 = np.max(corr) * dx
    return h_final, c5, n_points
