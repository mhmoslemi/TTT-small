import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros_like(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    best_c5 = obj(h)
    best_h = h.copy()

    T = best_c5 * 0.1
    alpha = 0.995
    n_iterations = 1000
    current_h = h.copy()
    current_c5 = best_c5
    start_time = time.time()

    for _ in range(n_iterations):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.01, n_points)
        candidate = current_h + noise
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > 0:
            candidate = candidate * target_sum / sum_candidate
        else:
            candidate = current_h.copy()
        candidate_c5 = obj(candidate)
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0:
            current_h = candidate.copy()
            current_c5 = candidate_c5
        else:
            if np.random.rand() < np.exp(-delta_c5 / T):
                current_h = candidate.copy()
                current_c5 = candidate_c5
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        T *= alpha

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    sum_h_final = np.sum(best_h)
    if sum_h_final > 0:
        best_h = best_h * target_sum / sum_h_final
    else:
        best_h = np.zeros_like(best_h)
    return (best_h, best_c5, n_points)
