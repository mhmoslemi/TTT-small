from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_k = np.argmax(corr)
        k = max_k
        if k > n_points - 1:
            k = n_points - 1
        else:
            k = int(k)
        if k < 0:
            k = 0
        
        perturb = np.zeros_like(best_h)
        for i in range(max(0, k - 5), min(n_points, k + 6)):
            if i < n_points and (i + k) < n_points:
                perturb[i] += -0.02 * best_h[i] * (1 - best_h[i + k])
                perturb[i + k] += 0.02 * (1 - best_h[i]) * best_h[i + k]
        
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
