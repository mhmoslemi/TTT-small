import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def initial_points():
        yield initial_h_values.copy()
        yield np.zeros(n_points)
        for i in range(n_points):
            yield np.roll(np.array([1.0]*n_points//2 + [0.0]*n_points//2), i)
        for _ in range(5):
            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)
            yield h

    best_h = initial_h_values.copy()
    best_c5 = np.inf

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for init in initial_points():
        res = minimize(obj, init, method='SLSQP', bounds=[(0,1)]*n_points, 
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-10)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if time.time() - start_time > budget_s - 50:
            break

    for _ in range(50):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=[(0,1)]*n_points, 
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-10)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
