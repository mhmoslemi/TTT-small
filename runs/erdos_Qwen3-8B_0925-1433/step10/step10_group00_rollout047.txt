import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial guess
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    h_initial = np.clip(h_initial, 0, 1)
    
    def evaluate(h):
        h_norm = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    pop_size = 20
    population = [h_initial]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        h = np.clip(h_initial + perturb, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    # Genetic algorithm
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([evaluate(h) for h in population])
        elite_idx = np.argsort(fitness)[:int(0.2 * len(population))]
        elites = [population[i] for i in elite_idx]
        
        # Generate new candidates
        new_pop = []
        for _ in range(len(population) - len(elites)):
            # Tournament selection
            a, b = np.random.choice(population, 2, replace=False)
            parent1, parent2 = a, b
            # Crossover
            idx = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:idx], parent2[idx:]])
            # Mutation
            mutation = np.random.normal(0, 0.02, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        
        population = elites + new_pop
        # Track best
        for h in population:
            curr_c5 = evaluate(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    
    # Local refinement
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        # Pick top 3 candidates
        fitness = np.array([evaluate(h) for h in population])
        top_idx = np.argsort(fitness)[:3]
        for i in top_idx:
            h = population[i].copy()
            # Local optimization
            def objective(x):
                x_norm = x * target_sum / np.sum(x) if np.sum(x) > 0 else x
                corr = np.correlate(x_norm, 1 - x_norm, mode='full')
                return np.max(corr) * dx
