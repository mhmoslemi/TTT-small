def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h * target_sum / np.sum(initial_h)

    def smooth_obj(h, eps=0.01):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        ovals = corr * dx
        return (1 / eps) * np.log(np.sum(np.exp(-eps * ovals)))

    def exact_obj(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    # Generate 5 perturbations
    candidates = []
    for _ in range(5):
        noise = np.random.normal(0, 0.1, n_points)
        tmp = initial_h + noise
        tmp = np.clip(tmp, 0, 1)
        tmp = tmp * target_sum / np.sum(tmp)

    # Evaluate and select top 2
    scores = [exact_obj(c) for c in candidates]

    best_h = initial_h.copy()
