import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    # Differential Evolution
    try:
        result_de = differential_evolution(
            obj, bounds,
            strategy='rand1bin',
            popsize=50,
            maxiter=200,
            tol=1e-8,
            vectorized=True,
            mutation=(0.8, 1.2),
            recombination=0.9
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        best_h, best_c5 = h, obj(h)

    # SLSQP refinement
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    try:
        result_slsqp = minimize(
            obj, best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        best_h, best_c5 = result_slsqp.x, result_slsqp.fun
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        pass

    return (best_h, best_c5, n_points)
