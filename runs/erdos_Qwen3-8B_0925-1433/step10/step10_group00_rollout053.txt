import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)

    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    T = 1.0
    alpha = 0.995
    max_iter = 1000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = best_h.copy()
        new_c5 = compute_c5(new_h, dx)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            best_c5 = new_c5
        T *= alpha

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
