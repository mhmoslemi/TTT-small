            if D != 0:
                i_idx, j_idx = np.random.choice(n_points, 2, replace=False)
                h[i_idx] += 0.05 * D
                h[j_idx] -= 0.05 * D
                h = np.clip(h, 0, 1)
                new_sum = np.sum(h)
                D_new = new_sum - target_sum
                h += D_new / n_points
                h = np.clip(h, 0, 1)
            new_population[i] = h
        population = new_population
        best_h = min(population, key=obj)
        if obj(best_h) < 0.381:
            break

    # Local Refinement
    best_h = min(population, key=obj)

    def lbfgs_obj(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return float('inf')
        h_scaled = h_clipped * target_sum / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(lbfgs_obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h_local = result.x
    best_c5 = result.fun

    return (best_h_local, best_c5, n_points)
