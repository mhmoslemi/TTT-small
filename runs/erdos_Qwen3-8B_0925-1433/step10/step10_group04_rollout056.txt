import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x, dx, n_points):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values)
    h_regular = np.zeros(n_points)
    h_regular[:n_points // 2] = 1.0
    h_random = np.random.rand(n_points)
    h_random = h_random * target_sum / np.sum(h_random)
    initial_points = [initial_h, h_regular, h_random]

    best_h = initial_h
    best_c5 = objective(best_h, dx, n_points)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for x in initial_points:
        res = minimize(objective, x, args=(dx, n_points), method='SLSQP', bounds=[(0, 1)] * n_points,
                       constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, args=(dx, n_points), method='SLSQP', bounds=[(0, 1)] * n_points,
                       constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    res = minimize(objective, best_h, args=(dx, n_points), method='SLSQP', bounds=[(0, 1)] * n_points,
                   constraints=constraints, tol=1e-8)
    return (res.x, res.fun, n_points)
