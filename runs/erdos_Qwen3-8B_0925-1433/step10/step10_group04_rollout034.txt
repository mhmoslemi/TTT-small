import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_guesses = [initial_h_values]
    binary_h = np.zeros(n_points)
    binary_h[::2] = 1
    initial_guesses.append(binary_h)
    random_h = np.random.uniform(0, 1, n_points)
    initial_guesses.append(random_h)

    best_c5 = float('inf')
    best_h = initial_guesses[0]

    for guess in initial_guesses:
        result = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    final_result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
