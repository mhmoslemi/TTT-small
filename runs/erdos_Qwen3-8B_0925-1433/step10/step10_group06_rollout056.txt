import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(100)]
    for h in population:
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.7:
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1 + np.random.normal(0, 0.03, n_points)
            child = np.clip(child, 0, 1)
            sum_ch = np.sum(child)
            if sum_ch > 0:
                child = child * target_sum / sum_ch
            parents.append(child)

        population[50:] = parents
        for h in population:
            if np.random.rand() < 0.1:
                h = np.clip(h * np.random.uniform(0.9, 1.1), 0, 1)
                sum_h = np.sum(h)
                if sum_h > 0:
                    h = h * target_sum / sum_h
        population.sort(key=lambda x: evaluate(x))
        population = population[:100]

    result = minimize(lambda x: evaluate(x), best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-10)
    return (result.x, result.fun, n_points)
