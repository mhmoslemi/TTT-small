import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Validate and normalize initial solution
    h = np.array(initial_h_values)
    sum_h = np.sum(h)
    if sum_h < 1e-6:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / sum_h
    h = np.clip(h, 0, 1)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    # Differential evolution
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-5,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        top_candidate = h
    else:
        top_candidate = result_de.x

    # SLSQP refinement
    def obj_slsqp(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        x = x * target_sum / (sum_x + 1e-12)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Phase 1: Global refinement
    result_phase1 = minimize(

        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        tol=1e-10,
        options={'maxiter': 200}
    )
    best_h, best_c5 = result_phase1.x, result_phase1.fun

    # Phase 2: Local refinement
    for _ in range(3):

        if res.fun < best_c5:

    final_result = minimize(

        method='SLSQP',
