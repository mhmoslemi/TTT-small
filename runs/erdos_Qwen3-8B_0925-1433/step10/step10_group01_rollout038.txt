import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    T = 0.1 * best_c5
    cooling = 0.995
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        candidate = np.clip(best_h + perturb, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > 0:
            candidate = candidate * target_sum / sum_candidate
        else:
            candidate = best_h.copy()
        corr = np.correlate(candidate, 1 - candidate, mode='full')
        candidate_c5 = np.max(corr) * dx
        if candidate_c5 < best_c5 or np.random.random() < np.exp((best_c5 - candidate_c5) / T):
            best_h = candidate.copy()
            best_c5 = candidate_c5
        T *= cooling

    def obj(x):
        h_ = np.clip(x, 0, 1)
        s = np.sum(h_)
        if s > 0:
            h_ = h_ * target_sum / s
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if s > 0:
        best_h = best_h * target_sum / s
    return (best_h, best_c5, n_points)
