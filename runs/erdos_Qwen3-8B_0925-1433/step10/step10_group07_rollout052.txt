import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial population
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    h = h * target_sum / sum_h if sum_h > 0 else np.zeros(n_points) * target_sum / n_points

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Genetic Algorithm
    pop_size = 60
    gen_limit = 100
    elite = 5
    mutate_rate = 0.3

    pop = [h + np.random.normal(0, 0.1, n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        pop[i] = np.clip(pop[i], 0, 1)
        s = np.sum(pop[i])
        if s > 0:
            pop[i] = pop[i] * target_sum / s
        else:
            pop[i] = np.zeros(n_points) * target_sum / n_points

    pop_fitness = [obj(p) for p in pop]
    best_idx = np.argmin(pop_fitness)
    best_h = pop[best_idx]
    best_c5 = pop_fitness[best_idx]

    start_time = time.time()
    for gen in range(gen_limit):
        if time.time() - start_time > budget_s - 60:
            break
        # Selection
        parents = np.random.choice(pop_size, size=pop_size, replace=True)
        children = []
        for _ in range(pop_size // 2):
            a, b = pop[parents[0]], pop[parents[1]]
            child = (a + b) * 0.5

            child = np.clip(child, 0, 1)

                child = np.zeros(n_points) * target_sum / n_points
            children.append(child)
        # Merge and sort

    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points) * target_sum / n_points
    corr = np.correlate(h, 1 - h, mode='full')
