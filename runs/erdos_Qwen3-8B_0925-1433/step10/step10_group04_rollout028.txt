    initial_patterns = [h_initial, generate_periodic(), generate_random()]
    best_h, best_c5 = None, float('inf')

    start_time = time.time()
    for h in initial_patterns:
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        h_opt, c5 = optimize(h)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    for _ in range(30):
        if time.time() - start_time > budget_s - 100:
            break
        h = perturb(best_h)
        h_opt, c5 = optimize(h)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    final_opt, final_c5 = optimize(best_h)
    return (final_opt, final_c5, n_points)
