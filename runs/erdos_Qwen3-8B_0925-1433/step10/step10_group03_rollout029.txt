        if current_sum == 0:
            return np.zeros_like(x)
        scale = target_sum / current_sum
        return x * scale

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.random.rand(n, size)

    def smooth(h, sigma):
        sigma = sigma / n_points
        return np.convolve(h, np.exp(-np.arange(200)/(2*sigma)), mode='same')[:n_points]
