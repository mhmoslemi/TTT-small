import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    sum_initial = np.sum(h)

        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    T = 1.0
    cooling_rate = 0.995
    max_steps = 1000
    time_start = time.time()

    for step in range(max_steps):

            break
        std = T * 0.05
        delta = np.random.normal(0, std, n_points)
        new_h = current_h + delta
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.zeros_like(new_h)
        new_c5 = compute_c5(new_h)
        current_c5 = compute_c5(current_h)

        if new_c5 < current_c5:
