import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        return x

    # Generate initial guesses
    initial_guesses = []
    # Initial provided values
    h0 = np.array(initial_h_values)
    h0 = clip_rescale(h0)
    initial_guesses.append(h0)
    # Binary pattern
    binary_h = np.zeros(n_points)
    binary_h[::2] = 1.0
    binary_h = clip_rescale(binary_h)
    initial_guesses.append(binary_h)
    # Random guess
    random_h = np.random.uniform(0, 1, n_points)
    random_h = clip_rescale(random_h)
    initial_guesses.append(random_h)

    def obj(x):
        h_ = clip_rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_guesses[0].copy()
    best_c5 = obj(best_h)
    for guess in initial_guesses[1:]:
        result = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = clip_rescale(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    final_result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
