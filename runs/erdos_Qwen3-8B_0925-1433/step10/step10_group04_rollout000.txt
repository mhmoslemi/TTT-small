import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x

    def obj(x):
        h_ = clip_rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def evaluate(h):
        return obj(clip_rescale(h))

    # Initial points: provided, structured, random
    initial_points = [initial_h_values]
    structured_h = np.zeros(n_points)
    structured_h[:n_points//2] = 1
    initial_points.append(structured_h)
    random_h = (np.random.rand(n_points) > 0.5).astype(float)
    initial_points.append(random_h)

    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    start_time = time.time()

    for ip in initial_points:
        h = clip_rescale(ip)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}], tol=1e-10)
        current_h = res.x
        current_c5 = res.fun
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5

        # Perturbations
        for _ in range(100):
            if time.time() - start_time > budget_s - 50:
                break
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = clip_rescale(current_h + perturb)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}], tol=1e-10)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    # Final refinement
    final_res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}], tol=1e-10)
    return (final_res.x, final_res.fun, n_points)
