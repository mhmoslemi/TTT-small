import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_initial_guesses():
        # Initial guess from parent
        h0 = np.array(initial_h_values)
        # Periodic binary
        h1 = np.zeros(n_points)
        for i in range(n_points):
            if (i // 2) % 2 == 0:
                h1[i] = 1.0
        # Evenly spaced
        m = int(np.round(target_sum))
        x = np.linspace(0, 2, m, endpoint=False)
        idx = np.round(x / dx).astype(int)
        h2 = np.zeros(n_points)
        for i in idx:
            if 0 <= i < n_points:
                h2[i] = 1.0
        # Fallback random
        h3 = np.random.rand(n_points)
        h3 = h3 * target_sum / np.sum(h3)
        return [h0, h1, h2, h3]

    initial_guesses = make_initial_guesses()
    best_h, best_c5 = initial_guesses[0].copy(), np.inf

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    for h in initial_guesses:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        # Perturb
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8)
        if res.fun < best_c5:

    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8)
