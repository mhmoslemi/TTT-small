import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = objective(h)

    start_time = time.time()

    # Differential Evolution with global search
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=15, tol=1e-8, mutation=(0.5, 1), recombination=0.8,
        maxiter=50, disp=False, polish=False, callback=lambda x: None
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    # Local refinement with L-BFGS-B
    res = minimize(
        objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-10, options={'maxiter': 100, 'disp': False}
    )
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun

    # Early exit if time is up
    if time.time() - start_time > budget_s - 50:
        return (best_h, best_c5, n_points)

    # Use initial guess as fallback
    return (best_h, best_c5, n_points)
