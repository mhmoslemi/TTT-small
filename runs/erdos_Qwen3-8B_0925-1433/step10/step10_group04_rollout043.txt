import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    desired_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_normalize(x):
        sum_x = np.sum(x)
        if sum_x > 0:
            return np.clip(x, 0, 1) * desired_sum / sum_x
        return np.zeros(n_points)

    # Initial guesses
    guesses = [np.array(initial_h_values)]
    # Binary pattern
    binary = np.zeros(n_points)
    binary[::2] = 1.0
    s = np.sum(binary)
    diff = s - desired_sum
    if diff > 0:
        ones = np.where(binary == 1)[0]
        binary[ones[-int(diff):]] = 0.0
    elif diff < 0:
        zeros = np.where(binary == 0)[0]
        binary[zeros[:int(-diff)]] = 1.0
    guesses.append(clip_normalize(binary))
    # Random initial
    guesses.append(clip_normalize(np.random.rand(n_points)))

    def obj(x):
        h = clip_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - desired_sum}]

    best_h = guesses[0]
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Perturbations
    for _ in range(150):
        if time.time() - start_time > budget_s - 50:
            break
        pert = np.random.normal(0, 0.02, n_points)
        new_h = clip_normalize(best_h + pert)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_res.x, final_res.fun, n_points)
