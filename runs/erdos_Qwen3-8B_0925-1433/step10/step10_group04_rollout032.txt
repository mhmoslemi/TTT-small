    h_random = h_random * target_sum / np.sum(h_random) if np.sum(h_random) > 0 else np.zeros(n_points)
    initial_points.append(h_random)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h_start in initial_points:
        result = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        for _ in range(100):
            if time.time() - start_time > budget_s - 50:
                break
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(h_start + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h_start
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

    final_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
