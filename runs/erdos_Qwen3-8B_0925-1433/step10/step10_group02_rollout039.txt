    h_normalized = h_clipped * (1 / sum_h)
    
    def slsqp_obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h == 0:
            return np.inf
        h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(slsqp_obj, h_normalized, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-10)
    return (result_slsqp.x, result_slsqp.fun, n_points)
