import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evolve(h):
        h = h.copy()
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    initial = initial_h_values.copy()
    initial = initial * target_sum / np.sum(initial)
    best = initial
    best_c5 = evaluate(initial)
    timestamps = [time.time()]

    for _ in range(200):
        if time.time() - timestamps[-1] > budget_s - 100:
            break
        timestamps.append(time.time())
        candidates = [evolve(best) for _ in range(10)]
        for c in candidates:
            c_val = evaluate(c)
            if c_val < best_c5:
                best = c
                best_c5 = c_val
        best_c5 = min(best_c5, evaluate(best))

    result = minimize(lambda x: evaluate(x), best, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-8)
    return (result.x, result.fun, n_points)
