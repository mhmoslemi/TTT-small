import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 50
    population = [np.clip(initial_h_values + np.random.normal(0, 0.01, n_points), 0, 1) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] * target_sum / np.sum(population[i])

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    # GA loop
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        current_evals = np.array([compute_c5(ind) for ind in population])
        top_idx = np.argsort(current_evals)[:int(population_size * 0.2)]
        top_candidates = [population[i] for i in top_idx]

        new_population = []
        for _ in range(int(population_size * 0.5)):
            p1, p2 = np.random.choice(top_candidates, 2, replace=False)
            child = (p1 + p2) / 2
            mutate = np.random.normal(0, 0.01, n_points)
            child += mutate
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)

        population = top_candidates + new_population
        for cand in population:
            current_c5 = compute_c5(cand)
            if current_c5 < best_c5:
                best_h, best_c5 = cand, current_c5

    # Local refinement
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    return (best_h, best_c5, n_points)
