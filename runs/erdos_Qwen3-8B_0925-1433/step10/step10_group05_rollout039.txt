import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def gen_patterns():
        res = []
        h = initial_h_values.copy()
        res.append(h)
        alt = np.zeros(n_points)
        alt[::2] = 1.0
        res.append(alt)
        clust = np.zeros(n_points)
        clust[::3] = 1.0
        clust[1::3] = 1.0
        clust[2::3] = 1.0
        res.append(clust)
        sin_p = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / (n_points/2)))
        res.append(sin_p)
        rand = np.random.permutation(h)
        res.append(rand)
        return res

    patterns = gen_patterns()
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for p in patterns:
        p_clip = np.clip(p, 0, 1)
        p_norm = p_clip * target_sum / np.sum(p_clip) if np.sum(p_clip) > 0 else np.zeros_like(p_clip)
        def obj_de(x):
            x_clip = np.clip(x, 0, 1)
            x_norm = x_clip * target_sum / np.sum(x_clip)
            corr = np.correlate(x_norm, 1 - x_norm, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        try:
            res_de = differential_evolution(obj_de, bounds, strategy='best1bin', popsize=40, maxiter=200, tol=1e-9, mutation=(0.7, 1.0), recombination=0.8, seed=seed, timeout=max(0, budget_s - (time.time() - start_time)))
        except:
            res_de = minimize(obj_de, p_norm, method='L-BFGS-B', bounds=bounds, tol=1e-10)

        def obj_slsqp(x):
            x_clip = np.clip(x, 0, 1)
            x_norm = x_clip * target_sum / np.sum(x_clip)
            corr = np.correlate(x_norm, 1 - x_norm, mode='full')
            return np.max(corr) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res_slsqp = minimize(obj_slsqp, res_de.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})

    best_h = np.clip(best_h, 0, 1)

    if h_sum > 0:
