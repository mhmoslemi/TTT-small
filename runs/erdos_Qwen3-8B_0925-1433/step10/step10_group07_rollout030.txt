import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Parent validation
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    h = h * target_sum / h_sum if h_sum > 0 else np.zeros(n_points)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Function to maintain sum constraint
    def enforce_sum(x):
        x = np.clip(x, 0, 1)
        delta = np.sum(x) - target_sum
        if abs(delta) < 1e-10:
            return x
        # Distribute delta proportionally
        return x + delta / n_points

    # Fitness function with constraint penalty
    def fitness(x):
        x = enforce_sum(x)
        return compute_c5(x)

    # Initialize population
    population = [h.copy()]
    for _ in range(49):
        perturb = np.random.rand(n_points) * 0.1
        trial = h + perturb
        trial = enforce_sum(trial)
        population.append(trial)

    best_h = h
    best_c5 = compute_c5(h)
    start_time = time.time()

    # Genetic Algorithm parameters
    pop_size = 50
    max_iter = 100
    mutation_rate = 0.2
    elite_size = 5

    for gen in range(max_iter):
        if time.time() - start_time > budget_s - 60:
            break
        # Evaluate fitness
        fitness_vals = [fitness(ind) for ind in population]
        # Sort by fitness
        sorted_idx = np.argsort(fitness_vals)
        sorted_pop = [population[i] for i in sorted_idx]
        # Select elites
        elites = sorted_pop[:elite_size]
        # Generate new population
        new_pop = elites[:]
        for _ in range(pop_size - elite_size):
            # Select parents
            p1, p2 = np.random.choice(elite_size, 2, replace=False)
            parent1 = sorted_pop[p1]
            parent2 = sorted_pop[p2]
            # Blend crossover
            child = (parent1 + parent2) / 2.0
            # Mutation
            if np.random.rand() < mutation_rate:
                idxs = np.random.choice(n_points, 2, replace=False)
                delta = np.sum(child) - target_sum
                child[idxs[0]] += 0.05 * delta
                child[idxs[1]] -= 0.05 * delta
                child = enforce_sum(child)
            new_pop.append(child)
        population = new_pop
        # Update best
        current_best = sorted_pop[0]
        current_c5 = fitness_vals[0]
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

    # Local refinement
    def objective(x):
        x = enforce_sum(x)
        return compute_c5(x)

    for _ in range(3):

            pass
