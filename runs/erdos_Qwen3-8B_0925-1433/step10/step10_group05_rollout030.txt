from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def objective(x):
        x_norm = x * target_sum / np.sum(x)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return np.zeros_like(x)
        return x * target_sum / np.sum(x)

    base_patterns = [
        np.zeros(n_points), np.zeros(n_points), np.zeros(n_points),
        np.zeros(n_points), np.zeros(n_points)
    ]
    base_patterns[0][:n_points//2] = 1.0
    base_patterns[1] = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)])
    base_patterns[2][::3] = 1.0
    base_patterns[2][n_points//3::3] = 1.0
    base_patterns[2][2*n_points//3::3] = 1.0
    base_patterns[3] = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / (n_points//2)))
    if 'initial_h_values' in globals():

    start_time = time.time()
