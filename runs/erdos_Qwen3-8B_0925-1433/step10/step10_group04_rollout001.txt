import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def generate_initial():

        reg[-1] += 0.5
        return reg
    def generate_random():

    initial_points = [initial_h_values, generate_initial(), generate_random()]

    best = (initial_h_values, float('inf'))
