        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for f in range(1, 7):
        initial_guess = [0.5, 0.0]
        res = minimize(
            lambda p: objective(p, f, n_points, dx),
            initial_guess,
            method='L-BFGS-B',
            bounds=[(0, 0.5), (0, 2 * np.pi)],
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                h = 0.5 + res.x[0] * np.sin(2 * np.pi * f * np.linspace(0, 2, n_points) + res.x[1])
                h = np.clip(h, 0, 1)
                best_h = h

    return (best_h, best_c5, n_points)
