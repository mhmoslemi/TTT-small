import numpy as np
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points)
    
    def compute_c5(phi):
        h = 0.5 + 0.5 * np.sin(np.pi * x + phi)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Multi-start optimization to avoid local minima
    best_c5 = float('inf')
    best_phi = 0
    for _ in range(5):
        result = minimize_scalar(compute_c5, bounds=(0, 2 * np.pi), method='bounded', options={'xatol': 1e-5})
        if result.fun < best_c5:
            best_c5 = result.fun
            best_phi = result.x
    
    h = 0.5 + 0.5 * np.sin(np.pi * x + best_phi)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1) > 1e-5:
        h *= 1 / sum_h * target_sum / dx
    return (h, best_c5, n_points)
