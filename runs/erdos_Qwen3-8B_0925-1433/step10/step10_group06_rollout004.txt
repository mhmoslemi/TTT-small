import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points)

    def objective(phi):
        h = 0.5 * np.sin(2 * np.pi * x + phi) + 0.5
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(objective, x0=0.0, bounds=[(0, 2 * np.pi)], method='L-BFGS-B', tol=1e-8)
    best_phi = result.x[0]
    best_h = 0.5 * np.sin(2 * np.pi * x + best_phi) + 0.5
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    best_h = best_h * target_sum / sum_h
    c5_bound = result.fun
    return (best_h, c5_bound, n_points)
