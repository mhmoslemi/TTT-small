import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = obj(h)

    initial_T = 0.2 * best_c5
    T = initial_T
    alpha = 0.998
    max_iter = 1000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        h_candidate = np.clip(h + perturb, 0, 1)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate > 0:
            h_candidate = h_candidate * target_sum / sum_candidate
        else:
            h_candidate = h.copy()

        delta = c5_candidate - best_c5

            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
