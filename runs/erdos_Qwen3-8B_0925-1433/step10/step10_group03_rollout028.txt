import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def scale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s

    def eval_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h):
        h = h.copy()
        mask = np.random.rand(*h.shape) < 0.1
        h[mask] += np.random.uniform(-0.1, 0.1, np.sum(mask))
        h = np.clip(h, 0, 1)
        return scale(h)

    def crossover(a, b):
        p = np.random.rand(n_points) < 0.5
        return np.where(p, a, b)

    def tournament_select(pop, k=5):
        return pop[np.random.choice(len(pop), k, replace=False)][0]

    pop = [scale(np.clip(initial_h_values, 0, 1))] + [scale(np.random.rand(n_points)) for _ in range(10)] + \
          [scale(np.random.rand(n_points)) for _ in range(20)]
    pop = [scale(h) for h in pop if np.sum(h) > 0]
    best_h = h
    best_val = eval_fitness(h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        if gen > 0 and abs(best_val - eval_fitness(best_h)) < 1e-6:
            break
        elites = [best_h]
        candidates = [tournament_select(pop) for _ in range(40)]
        offspring = []
        for a, b in zip(candidates[::2], candidates[1::2]):
            child = crossover(a, b)
            if np.random.rand() < 0.8:
                child = mutate(child)
            offspring.append(child)
        pop = elites + offspring
        pop = [scale(h) for h in pop if np.sum(h) > 0]
        pop = sorted(pop, key=eval_fitness)[:50]
        current_val = eval_fitness(pop[0])
        if current_val < best_val:

            best_h = pop[0]

    # Final refinement
    def smooth(h, sigma):
