from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = []

    base = base * target_sum / np.sum(base) if np.sum(base) > 0 else np.zeros(n_points)
    for _ in range(50):
        perturb = np.random.normal(0, 0.08, n_points)
        h = base + perturb
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)
    
    # Additional patterns
    uniform = np.full(n_points, 2.0 / n_points)
    peak = np.zeros(n_points); peak[0] = 1.0
    two_peak = np.zeros(n_points); two_peak[0] = two_peak[1] = 0.5
    population.extend([uniform, peak, two_peak])

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

        if time.time() - start_time > budget_s - 50:

        # Evaluate
        fitness = [evaluate(h) for h in population]
