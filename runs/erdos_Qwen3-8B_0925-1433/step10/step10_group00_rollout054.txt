from scipy.optimize import minimize
import time

        sum_h = np.sum(h_)

            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Population-based GA
    population = [h.copy()]
    for _ in range(19):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
        population.append(new_h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()

        if time.time() - start_time > budget_s:

        # Generate new candidates
        new_population = elites[:]
        while len(new_population) < len(population):

            new_population.append(child)
        population = new_population
        current_c5 = min(fitness)
