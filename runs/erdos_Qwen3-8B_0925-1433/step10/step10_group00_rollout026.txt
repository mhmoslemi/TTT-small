def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    def compute_corr_max_lag(h):

        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        lag_k = max_idx - (n_points - 1)
        return lag_k, max_corr

    best_h = h.copy()
    best_c5 = compute_corr_max_lag(best_h)[1]

    start_time = time.time()
    step_size = 0.1
    decay_factor = 0.995
    for iteration in range(1000):
        if time.time() - start_time > budget_s:

        lag_k, curr_c5 = compute_corr_max_lag(best_h)
        grad = np.zeros(n_points)
        if lag_k > 0:
            for i in range(n_points - lag_k):

                grad[i] += 1 - best_h[j]
                grad[j] -= best_h[i]
        elif lag_k < 0:
            m = -lag_k
            for i in range(m, n_points):
                j = i - m
                grad[i] += 1 - best_h[j]
                grad[j] -= best_h[i]
