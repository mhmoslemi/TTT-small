import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in locals():
        h = initial_h_values
    else:
        n_points = 45
        h = np.zeros(n_points)
        m = n_points // 5
        h[:m//2] = 1
        h[m//2:] = 0

        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)

        h = np.zeros(n_points)

    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], tol=1e-8, options={'maxiter': 100})
