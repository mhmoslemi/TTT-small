import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Validate initial guess
    h_initial = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(h_initial)
    if sum_initial > 0:
        h_initial = h_initial * target_sum / sum_initial
    else:
        h_initial = np.zeros(n_points)
    best_h = h_initial
    best_c5 = obj(best_h)

    # Generate diverse initial guesses
    guesses = [h_initial]
    for _ in range(10):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        if np.sum(h_rand) > 0:
