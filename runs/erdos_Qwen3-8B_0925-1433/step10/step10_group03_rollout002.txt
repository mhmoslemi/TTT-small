import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        # Differential evolution with transformed bounds
        def transformed_obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x > 0:
                x = x * target_sum / sum_x
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        res = differential_evolution(transformed_obj, bounds, strategy='best1bin',
                                    popsize=15, mutation=(0.5, 1.0), recombination=0.8,
                                    tol=1e-6, disp=False, polish=True, xtol=1e-8)
