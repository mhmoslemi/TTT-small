import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # Iterative refinement
    max_iterations = 500
    alpha = 0.1
    for _ in range(max_iterations):
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        if max_corr <= best_c5:
            # If no improvement, break early
            break
        # Find the lag with maximum value
        k_max_index = np.argmax(corr)
        lag = k_max_index - (n_points - 1)
        # Compute derivative terms for this lag
        derivative = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = i + lag
            if 0 <= shifted_i < n_points:
                derivative[i] = (1 - h[shifted_i]) * dx
        # Compute delta_h
        delta_h = -alpha * derivative
        # Adjust h
        h += delta_h
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        # Update best solution
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (best_h, best_c5, n_points)

        k_max_index = np.argmax(corr)
        lag = k_max_index - (n_points - 1)
        # Compute derivative terms for this lag
        derivative = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = i + lag
            if 0 <= shifted_i < n_points:
                derivative[i] = (1 - h[shifted_i]) * dx
        # Compute delta_h
        delta_h = -alpha * derivative
        # Add perturbation with some probability
        if np.random.rand() < perturb_prob:
            delta_h += np.random.normal(0, 0.05, size=n_points)
        # Adjust h
        h += delta_h
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        # Update best solution
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        # Decay alpha
        alpha *= 0.999
    return (best_h, best_c5, n_points)
