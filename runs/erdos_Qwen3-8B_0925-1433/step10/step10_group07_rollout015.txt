import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    population = [h_initial.copy()]
    for _ in range(49):
        h = h_initial.copy()
        delta = np.random.uniform(-0.1, 0.1, n_points)
        h += delta
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def ga_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h):
        delta = np.sum(h) - target_sum
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = h.copy()
        h_new[i] += 0.05 * delta
        h_new[j] -= 0.05 * delta
        h_new = np.clip(h_new, 0, 1)
        new_sum = np.sum(h_new)
        delta_new = new_sum - target_sum
        h_new += delta_new / n_points
        h_new = np.clip(h_new, 0, 1)
        return h_new

    def crossover(parent1, parent2):
        alpha = np.random.uniform(0.5, 1.0)
        return np.clip((1 - alpha) * parent1 + alpha * parent2, 0, 1)

    n_gen = 100
    population_size = 50
    best_fitness = float('inf')
    best_h = h_initial

    start_time = time.time()
    for gen in range(n_gen):
        fitness = [ga_objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_fitness:
            best_fitness = fitness[best_idx]
            best_h = population[best_idx]
        if time.time() - start_time > budget_s - 100:
            break
        parent_indices = np.argsort(fitness)[:int(population_size / 2)]
        parents = [population[i] for i in parent_indices]
        new_population = []
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            new_population.append(mutate(child))
        population = new_population

    fitness = [ga_objective(ind) for ind in population]

    best_c5 = best_fitness
    best_h_opt = best_h

                best_h_opt = res.x
        except:

        pass
