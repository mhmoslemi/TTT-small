import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate structured patterns
    patterns = []
    # Even spacing
    even = np.zeros(n_points)
    even[::2] = 1.0
    even = np.clip(even, 0, 1)
    even = even * target_sum / np.sum(even)
    patterns.append(even)
    # Symmetric
    sym = np.zeros(n_points)
    sym[:n_points//2] = 1.0
    sym = np.clip(sym, 0, 1)
    sym = sym * target_sum / np.sum(sym)
    patterns.append(sym)
    # Modular repeat
    mod = np.tile(initial_h_values[:n_points//2], 2)
    mod = np.clip(mod, 0, 1)
    mod = mod * target_sum / np.sum(mod)
    patterns.append(mod)
    # Gaussian smooth
    kernel = np.exp(-np.arange(n_points)**2 / (2*(0.2)**2))
    smooth = np.convolve(initial_h_values, kernel, mode='same')
    smooth = np.clip(smooth, 0, 1)
    smooth = smooth * target_sum / np.sum(smooth)
    patterns.append(smooth)

    best_h = patterns[0]
    best_c5 = obj(best_h)

    start_time = time.time()
    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_res.x, final_res.fun, n_points)
