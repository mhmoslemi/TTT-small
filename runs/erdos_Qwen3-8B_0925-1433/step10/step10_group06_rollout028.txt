import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_candidate():
        h = np.copy(initial_h_values)
        idxs = np.random.choice(n_points, min(n_points // 4, 20), replace=False)
        for i in idxs:
            h[i] = np.random.uniform(0, 1)
        h = h * target_sum / np.sum(h)
        return h

    population = [generate_candidate() for _ in range(50)]
    best_h = np.copy(population[0])
    best_c5 = np.inf

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()
        # Select parents and evolve
        population = [np.clip(h, 0, 1) for h in population]
        new_population = []
        for _ in range(50):
            a, b = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[a], population[b]
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0, 1)
            sum_ch = np.sum(child)
            if sum_ch > 0:
                child = child * target_sum / sum_ch
            new_population.append(child)
        population = new_population + [np.copy(best_h) for _ in range(50)]
        population.sort(key=lambda x: evaluate(x))
        population = population[:50]

    result = minimize(lambda x: evaluate(x), best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-10)
    return (result.x, result.fun, n_points)
