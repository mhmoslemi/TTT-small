import time

    def obj(x):

        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0
    perturb_amp = 0.02

    for _ in range(300):
        if time.time() - start_time > budget_s:

        sum_h = np.sum(h_)
        if sum_h > 0:

        if k < n_points - 1:

                perturb[i] += np.random.normal(0, perturb_amp, 1)
