import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points) * target_sum

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break

        perturbations = []

        # Perturbation 1: Random Walk
        pert = np.random.normal(0, 0.05, n_points)
        pert_h = h + pert
        pert_h = np.clip(pert_h, 0, 1)
        pert_h_sum = np.sum(pert_h)
        if pert_h_sum > 0:
            pert_h = pert_h * target_sum / pert_h_sum
        perturbations.append(pert_h)

        # Perturbation 2: Spatial Transition
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        val1, val2 = h[i1], h[i2]
        new_h = h.copy()
        new_h[i1:i2+1] = np.linspace(val1, val2, i2 - i1 + 1)
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        perturbations.append(new_h)

        # Perturbation 3: Gaussian Smoothing
        sigma = 1.0
        kernel = np.exp(- (np.arange(n_points) - n_points//2)**2 / (2 * sigma**2))
        kernel = kernel / np.sum(kernel)
        smooth_h = np.convolve(h, kernel, mode='same')
        smooth_h = np.clip(smooth_h, 0, 1)
        smooth_h_sum = np.sum(smooth_h)
        if smooth_h_sum > 0:
            smooth_h = smooth_h * target_sum / smooth_h_sum
        perturbations.append(smooth_h)

        # Evaluate candidates
        candidates = []
        for p in perturbations:
            c = compute_c5(p)
            candidates.append((c, p))
        candidates.sort()
        best_candidate, best_p = candidates[0]

        if best_candidate < best_c5:
            best_c5 = best_candidate
            best_h = best_p.copy()

        # Local optimization
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        def obj(x):
            x_clipped = np.clip(x, 0, 1)
            x_scaled = x_clipped * target_sum / np.sum(x_clipped) if np.sum(x_clipped) > 0 else x_clipped * target_sum
            return compute_c5(x_scaled)

        res = minimize(obj, best_p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
