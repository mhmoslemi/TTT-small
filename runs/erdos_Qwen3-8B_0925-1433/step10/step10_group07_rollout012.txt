import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(best_h)
    if sum_initial > 0:
        best_h = best_h * target_sum / sum_initial
    best_c5 = obj(best_h)

    try:
        result_de = differential_evolution(
            obj, bounds, strategy='best1bin',
            popsize=30, maxiter=50, tol=1e-8, vectorized=True,
            mutation=(0.5, 1), recombination=0.8, mutation_scale=0.5
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        pass

    try:
        def cons(h):
            return np.sum(h) - target_sum
        cons_dict = {'type': 'eq', 'fun': cons}
        result_slsqp = minimize(
            obj, best_h, method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=cons_dict,
            tol=1e-10,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    except:
        pass

    if elapsed > budget_s - 100:
        pass
