import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1.0 - h_val, mode='full')
        return np.max(corr) * dx

    def mutate(h_val):
        delta = np.sum(h_val) - target_sum
        if abs(delta) < 1e-10:
            return h_val.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        alpha = 0.05
        h_val[i] += alpha * delta
        h_val[j] -= alpha * delta
        h_val = np.clip(h_val, 0, 1)
        new_sum = np.sum(h_val)
        delta_new = new_sum - target_sum
        if abs(delta_new) > 1e-10:
            h_val += delta_new / n_points
            h_val = np.clip(h_val, 0, 1)
        return h_val.copy()

    # Initialize population
    population = [h.copy()]
    for _ in range(49):
        rand_h = np.random.rand(n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)

    # GA parameters
    generations = 200
    pop_size = 50
    elite_size = 5

    for gen in range(generations):
        fitnesses = np.array([-compute_c5(ind) for ind in population])
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        elite = population[:elite_size]
        new_pop = elite.copy()
        for _ in range(pop_size - elite_size):
            p1, p2 = np.random.choice(elite_size, 2, replace=False)

            new_pop.append(child)
        population = new_pop

    best_h = population[0]

    def objective(x):
        return compute_c5(x)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8, options={'maxiter': 100})
    final_h = result.x
