import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    population = [initial_scaled]
    for _ in range(49):
        rand_h = np.random.uniform(0, 1, n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)
    
    best_h = initial_scaled
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for gen in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        parents = []
        for _ in range(50):
            tournament = np.random.choice(len(population), 5, replace=False)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])
        
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points - 1)
                o1 = np.concatenate([p1[:split], p2[split:]])
                o2 = np.concatenate([p2[:split], p1[split:]])
                offspring.append(o1)
                offspring.append(o2)
        
        mutated = []
        for child in offspring:
            noise = np.random.normal(0, 0.03, n_points)
            child_mut = child + noise
            child_mut = np.clip(child_mut, 0, 1)
            total = np.sum(child_mut)
            if total > 0:
                child_mut *= target_sum / total
            mutated.append(child_mut)
        
        population = mutated
        if gen % 5 == 0:
            for _ in range(3):
                smoothed = np.convolve(best_h, np.ones(5)/5, mode='same')
                smoothed = np.clip(smoothed, 0, 1)
                total = np.sum(smoothed)
                if total > 0:
                    smoothed *= target_sum / total
                if evaluate(smoothed) < best_c5:
