import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_individual(h_initial):
        h = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    def smooth_h(h, sigma):
        kernel = np.hanning(sigma)
        kernel = kernel / np.sum(kernel)
        h_smoothed = np.zeros_like(h)
        for i in range(n_points):
            start = max(0, i - sigma + 1)
            end = min(n_points, i + sigma)
            h_smoothed[i] = np.sum(h[start:end] * kernel[:end - start])
        return h_smoothed

    population = []
    for _ in range(20):
        h = initial_h_values.copy()
        h = create_individual(h)
        population.append(h)
    for _ in range(30):
        h = np.random.rand(n_points)
        h = create_individual(h)
        population.append(h)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break

        fitness = []
        for h in population:
            h_clip = np.clip(h, 0, 1)
            sum_h = np.sum(h_clip)
            if sum_h > 0:
                h_norm = h_clip * target_sum / sum_h
            else:
                h_norm = np.zeros_like(h_clip)
            corr = np.correlate(h_norm, 1 - h_norm, mode='full')
            fit = np.max(corr) * dx
            fitness.append(fit)

        elite_indices = np.argsort(fitness)[:int(len(population)*0.2)]
        elites = [population[i] for i in elite_indices]

        new_population = elites.copy()

    return (best_h_smoothed, best_c5, n_points)
