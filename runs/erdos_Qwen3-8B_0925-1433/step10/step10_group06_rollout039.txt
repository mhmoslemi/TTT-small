import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h * target_sum / np.sum(initial_h)
    h_initial += np.random.normal(0, 0.05, n_points)
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h > 0:
        h_initial = h_initial * target_sum / sum_h

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        sum_exp = np.sum(np.exp(-0.01 * corr))
        return (1 / 0.01) * np.log(sum_exp)

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})

    if not result.success:
        best_h = initial_h.copy()
        sum_best = np.sum(best_h)
        if sum_best > 0:
            best_h = best_h * target_sum / sum_best
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        c5_bound = np.max(corr) * dx
        return (best_h, c5_bound, n_points)

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best

    corr = np.correlate(best_h, 1 - best_h, mode='full')
