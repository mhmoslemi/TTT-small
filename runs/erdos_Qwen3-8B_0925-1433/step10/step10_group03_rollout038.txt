def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale_and_norm(h, target_sum):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            h = np.random.uniform(0, 1, n_points)
            s = np.sum(h)
        scale = target_sum / s
        return h * scale

    population = []
    h0 = initial_h_values.copy() * target_sum / np.sum(initial_h_values)
    population.append(h0)
    for _ in range(10):
        rand = np.random.uniform(0, 1, n_points)
        population.append(rand * target_sum / np.sum(rand))

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h0
    best_c5 = fitness(h0)

    max_gen = 100
    for gen in range(max_gen):
        fit = np.array([fitness(h) for h in population])

                best_c5 = fit[best_idx]

        offspring = []
        for _ in range(len(population)//2):

            p2 = population[np.random.choice(len(population))]
            c = (p1 + p2) / 2
