import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Differential evolution for global search
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, mutation=(0.5, 1), recombination=0.8,
        tol=1e-5, maxiter=80, seed=seed
    )

    best_h_de = result_de.x
    best_c5_de = result_de.fun
    if best_c5_de < best_c5:
        best_h, best_c5 = best_h_de, best_c5_de

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_lbgf = minimize(
        objective, best_h, method='L-BFGS-B',
        bounds=bounds, constraints=cons, tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h, best_c5 = res_lbgf.x, res_lbgf.fun

    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, best_c5, n_points)
