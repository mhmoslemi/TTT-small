import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        h = h * target_sum / (sum_h + 1e-10) if sum_h > 0 else x * target_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_guesses = [
        initial_h_values.copy(),
        np.zeros(n_points) + 0.5,
        np.random.rand(n_points) * 0.5,
        np.zeros(n_points),
        np.random.rand(n_points)
    ]

    best_h, best_c5 = initial_guesses[0], float('inf')

    for guess in initial_guesses:
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5, best_h = res.fun, res.x

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5, best_h = res.fun, res.x

    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    return (final_res.x, final_res.fun, n_points)
