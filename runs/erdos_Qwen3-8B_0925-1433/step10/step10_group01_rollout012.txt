import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_current = np.array(initial_h_values, dtype=np.float64)
    h_current = np.clip(h_current, 0, 1)
    sum_h = np.sum(h_current)
    if sum_h > 0:
        h_current = h_current * target_sum / sum_h
    else:
        h_current = np.zeros(n_points) + 0.5
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)

    T = 0.1 * best_c5
    alpha = 0.995
    sa_iterations = 1000
    start_time = time.time()

    for _ in range(sa_iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        h_candidate = h_current + perturb
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_cand = np.sum(h_candidate)
        if sum_cand > 0:
            h_candidate = h_candidate * target_sum / sum_cand
        else:
            h_candidate = np.zeros(n_points) + 0.5
        c5_candidate = compute_c5(h_candidate)
        delta = c5_candidate - compute_c5(h_current)
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_candidate.copy()
            if c5_candidate < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_candidate
        T *= alpha

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    return (best_h, best_c5, n_points)
