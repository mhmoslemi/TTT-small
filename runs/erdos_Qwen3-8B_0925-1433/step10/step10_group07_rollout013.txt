import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Genetic Algorithm setup
    pop_size = 50
    generations = 100
    mutation_scale = 0.1
    best_h = h.copy()
    best_c5 = obj(h)

    # Initialize population with initial values and random perturbations
    population = [np.copy(h) for _ in range(pop_size)]
    for i in range(pop_size):
        perturb = np.random.normal(0, mutation_scale, n_points)
        population[i] += perturb
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])

    start_time = time.time()

    for _ in range(generations):
        new_population = []
        for ind in population:
            # Constraint-aware mutation
            mutated = ind + np.random.normal(0, mutation_scale, n_points)
            mutated = np.clip(mutated, 0, 1)
            sum_mut = np.sum(mutated)
            if sum_mut > 0:
                mutated = mutated * target_sum / sum_mut
            new_population.append(mutated)
        population = new_population

        # Evaluate and select best
        for ind in population:
            current_c5 = obj(ind)
            if current_c5 < best_c5:
                best_h = np.copy(ind)
                best_c5 = current_c5
            if best_c5 < 0.381:
                break
        if best_c5 < 0.381 or time.time() - start_time > budget_s - 100:
            break

    # Local refinement with SLSQP
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(obj, best_h, method='SLSQP', bounds=Bounds(0, 1, shape=(n_points,)), constraints=constraints, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = obj(best_h)
    return (best_h, final_c5, n_points)
