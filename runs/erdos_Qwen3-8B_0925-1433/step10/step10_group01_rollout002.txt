import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def rescale(arr):
        return np.clip(arr, 0, 1) * target_sum / np.sum(arr) if np.sum(arr) > 0 else np.zeros_like(arr)
    
    # Initialize population
    population_size = 50
    initial = rescale(initial_h_values)
    population = [rescale(initial + np.random.normal(0, 0.01, n_points)) for _ in range(population_size)]
    
    best_c5 = float('inf')
    best_h = initial.copy()
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate C₅ for all candidates
        ys = [np.max(np.correlate(cand, 1 - cand, mode='full')) * dx for cand in population]
        indices = np.argsort(ys)
        top_indices = indices[:10]
        top_candidates = [population[i] for i in top_indices]
        
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            # Crossover
            parent1, parent2 = np.random.choice(top_candidates, size=2, replace=False)
            child = (parent1 + parent2) / 2.0
            # Mutation
            mutate_mask = np.random.rand(n_points) < 0.1
            child[mutate_mask] += np.random.normal(0, 0.01, np.sum(mutate_mask))
            child = rescale(child)
            new_population.append(child)
        population = new_population
        
        # Track best candidate
        current_best_idx = np.argmin(ys)
        if ys[current_best_idx] < best_c5:
            best_c5 = ys[current_best_idx]
            best_h = population[current_best_idx]
    
    # Local refinement
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    return (best_h, best_c5, n_points)
