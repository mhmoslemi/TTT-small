            x = i * dx
            if np.floor(x / 0.4) % 2 == 0:
                h[i] = 1
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros(n_points)
        return h * (n_points / 2) / sum_h

    initial_guesses = [initial_h_values.copy()]
    for _ in range(3):
        initial_guesses.append(generate_pattern(n_points))

    best_h = initial_guesses[0]
    best_c5 = compute_c5(best_h)

    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    for guess in initial_guesses:
        res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
