import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def compute_c5(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return float('inf')
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.5 * best_c5
    alpha = 0.995
    sa_iter = 500
    start_time = time.time()

    for _ in range(sa_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        h_candidate = np.clip(best_h + perturb, 0, 1)
        if np.sum(h_candidate) == 0:
            h_candidate = np.zeros(n_points)
        else:
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
        T *= alpha

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    if res.success:
        best_h = res.x
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) == 0:
        best_h = np.zeros(n_points)
    else:
        best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
