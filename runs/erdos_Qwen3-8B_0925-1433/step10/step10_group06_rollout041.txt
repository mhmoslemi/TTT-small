from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial population
    population = [np.clip(initial_h_values, 0, 1).copy()]
    # Add alternating pattern
    alt = np.zeros(n_points)
    alt[::2] = 1.0
    alt = alt * target_sum / np.sum(alt)
    population.append(alt)
    # Add other patterns
    for _ in range(98):
        h = np.clip(initial_h_values * np.random.uniform(0.8, 1.2), 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s - 50:
            break
        # Evaluate all candidates
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        # Adaptive mutation and crossover
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.6:
                # Crossover
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Mutation
                noise = np.random.normal(0, 0.03, n_points)
                child = parent1 + noise
            child = np.clip(child, 0, 1)
            sum_ch = np.sum(child)

        population = population[:100]

    # Final refinement using L-BFGS-B
