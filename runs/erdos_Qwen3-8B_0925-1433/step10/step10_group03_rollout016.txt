import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    try:
        res_de = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=15,
            maxiter=100, tol=1e-8, mutation=(0.5, 1), recombination=0.8,
            seed=seed, disp=False, callback=lambda x: None
        )
        best_h = res_de.x
        best_c5 = res_de.fun
    except:
        pass
    
    if time.time() - start_time > budget_s - 10:
        return (best_h, best_c5, n_points)

    res_lbgf = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8)
    best_h = res_lbgf.x
    best_c5 = res_lbgf.fun

    return (best_h, best_c5, n_points)
