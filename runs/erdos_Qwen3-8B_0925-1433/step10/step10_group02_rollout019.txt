import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()

        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')

    patterns = []

    patterns.append(np.flip(h))
    patterns.append(gaussian_filter1d(h, sigma=1.0))
    patterns.append(np.concatenate([h[:n_points//2], h[:n_points//2]]))

    best_h, best_c5 = h, float('inf')

        p = p * target_sum / np.sum(p) if np.sum(p) > 0 else p

        if res.success and res.fun < best_c5:

            best_h = res.x

    for _ in range(100):

        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    final_res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-9)
