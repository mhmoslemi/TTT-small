            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, final_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
    best_h = res.x
    best_c5 = res.fun
    return (best_h, best_c5, n_points)
