```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with given starting vector
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    try:
        # Run differential evolution
        res_de = differential_evolution(
            obj, bounds, strategy='best1bin',
            popsize=15, tol=1e-8, maxiter=100,
            mutation=(0.5, 1), recombination=0.8,
            seed=seed
        )
        if res_de.success:
            best_h = res_de.x.copy()
            best_c5 = res_de.fun

        # Final refinement with L-BFGS-B
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res_lbgf = minimize(
            obj, best_h, method='L-BFGS-B',
            bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        if res_lbgf.success and res_lbgf.fun < best_c5:
            best_h = res_lbgf.x
            best_c5 = res_lbgf.fun
    except:
        pass  # Fallback to initial guess if optimization fails

    # Safety check
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    return (best_h, best_c5, n_points)
```
