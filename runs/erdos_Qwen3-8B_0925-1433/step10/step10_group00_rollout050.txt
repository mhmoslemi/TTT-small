        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        step_size *= 0.99
    return (best_h, best_c5, n_points)
