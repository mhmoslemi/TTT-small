import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def validate(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros(n_points)
        h = h * target_sum / np.sum(h)
        return h

    def obj(x):
        h = validate(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_patterns(h0):
        patterns = []
        h0 = validate(h0)
        patterns.append(h0)
        h_even = np.zeros(n_points)
        h_even[::2] = 1
        patterns.append(validate(h_even))
        h_sym = np.zeros(n_points)
        h_sym[::3] = 1
        patterns.append(validate(h_sym))
        h_mod = np.concatenate([h0[:n_points//2], h0[:n_points//2]])
        patterns.append(validate(h_mod))
        h_gauss = np.convolve(h0, np.exp(-np.linspace(0, 5, 5)**2 / 2), mode='same')
        patterns.append(validate(h_gauss))
        return patterns

    patterns = generate_patterns(initial_h_values)
    best_h = patterns[0].copy()
    best_c5 = obj(best_h)

    start_time = time.time()

    for p in patterns:
        res = minimize(obj, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    def perturb_refine(h, scale, iters):
        h = h.copy()
        for _ in range(iters):
            noise = np.random.normal(0, scale, n_points)
            h = np.clip(h + noise, 0, 1)
            h = validate(h)
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-10)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        return best_h

    best_h = perturb_refine(best_h, 0.1, 100)
    best_h = perturb_refine(best_h, 0.05, 100)
    best_h = perturb_refine(best_h, 0.01, 100)
