from scipy.optimize import minimize
import time

    def fitness(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vals) - target_sum

        i, j = np.random.choice(n_points, 2, replace=False)

        parent[j] -= 0.05 * delta
        parent = np.clip(parent, 0, 1)

        parent += delta_new / n_points
        parent = np.clip(parent, 0, 1)
        return parent
    
    population = [mutate(h.copy()) for _ in range(50)]
    best = min(population, key=fitness)
    best_fit = fitness(best)
    best_idx = 0

        if time.time() - start_time > budget_s - 100:
