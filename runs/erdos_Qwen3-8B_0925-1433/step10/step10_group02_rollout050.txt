            best_h = res.x

    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    if final_res.success:
        best_c5 = final_res.fun
        best_h = final_res.x

    return (best_h, best_c5, n_points)
