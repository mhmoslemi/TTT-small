import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    # Differential evolution with adaptive restarts
    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        result = differential_evolution(
            obj, bounds, constraints=cons,
            popsize=20, tol=1e-8,
            mutation=(0.5, 1), recombination=0.8,
            seed=seed + _,
            polish=True, disp=False
        )
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

        # Local refinement
        res_lbgf = minimize(obj, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           tol=1e-10, options={'maxiter': 100})
        if res_lbgf.success and res_lbgf.fun < best_c5:
            best_h = res_lbgf.x
            best_c5 = res_lbgf.fun

    return (best_h, best_c5, n_points)
