import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with a mix of initial and random samples
    population = [initial_h_values.copy()]
    for _ in range(99):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(500):
        if time.time() - start_time > budget_s - 50:
            break
        # Evaluate all candidates
        fitness = [evaluate(h) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Adaptive mutation step
        mut_step = 0.05 * (1 - gen / 500)
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.7:
                # Crossover with large alpha for better diversity
                alpha = np.random.uniform(0.1, 0.9)
                child = alpha * p1 + (1 - alpha) * p2
            else:
                # Mutation with adaptive step
                child = p1 + np.random.normal(0, mut_step, n_points)
            child = np.clip(child, 0, 1)
            sum_ch = np.sum(child)
            if sum_ch > 0:
                child = child * target_sum / sum_ch
            parents.append(child)

        # Replace worst half with offspring
        population[50:] = parents
        population.sort(key=lambda x: evaluate(x))
        population = population[:100]

    # Final refinement with local optimization
    result = minimize(lambda x: evaluate(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-8)
    return (result.x, result.fun, n_points)
