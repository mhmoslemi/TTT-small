import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return h * target_sum / s

    h = np.clip(initial_h_values, 0, 1)
    h = rescale(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_ = rescale(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def ga_step(population, gen, max_gen):
        if gen >= max_gen:
            return population
        new_pop = []
        for _ in range(len(population)):
            a, b = np.random.choice(population, 2, replace=False)
            alpha = np.random.uniform(0.5, 1.5)
            child = a + alpha * (b - a)
            child = rescale(np.clip(child, 0, 1))
            new_pop.append(child)
        return np.array(new_pop)

    population = [h.copy() for _ in range(50)]
    for gen in range(200):
        population = ga_step(population, gen, 200)
        vals = np.array([obj(ind) for ind in population])
        best_idx = np.argmin(vals)
        best_h, best_c5 = population[best_idx].copy(), vals[best_idx]
        if time.time() - start_time > budget_s - 100:
            break

    def calc_c5(x):
        h_ = np.clip(x, 0, 1)
        h_ = rescale(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
