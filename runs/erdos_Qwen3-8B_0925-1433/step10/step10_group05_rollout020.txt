            i_minus = (i - lag) % n_points
            grad = 1.0 - h[i_plus] - h[i_minus]
            gradients.append(grad)
        delta_h = -alpha * np.array(gradients)
        h = np.clip(h + delta_h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        if np.random.random() < random_perturbation_prob:
            perturb = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturb, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * target_sum / h_sum
            else:
                h = np.zeros(n_points)
        current_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_corr < best_c5:
            best_c5 = current_corr
            best_h = h.copy()
        alpha *= decay_rate

    final_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_corr, n_points)
