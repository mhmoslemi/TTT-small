import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def objective(x):
        x_norm = x * target_sum / np.sum(x)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 150})

    if result.success:
        best_h = result.x
    else:
        best_h = h

    best_h = np.clip(best_h, 0, 1)
    h_sum_final = np.sum(best_h)
    if h_sum_final > 0:
        best_h = best_h * target_sum / h_sum_final

    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx

    return (best_h, final_c5, n_points)
