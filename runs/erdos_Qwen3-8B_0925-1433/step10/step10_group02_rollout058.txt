import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    patterns = []
    # Even spacing
    even = np.zeros(n_points)
    even[::2] = 1.0
    even = np.clip(even, 0, 1)
    if np.sum(even) > 0:
        even = even * target_sum / np.sum(even)
    patterns.append(even)
    # Symmetric periodic
    m = max(1, n_points // 3)
    symmetric = np.zeros(n_points)
    for i in range(n_points):
        if i % m == 0:
            symmetric[i] = 1.0
    symmetric = np.clip(symmetric, 0, 1)
    if np.sum(symmetric) > 0:
        symmetric = symmetric * target_sum / np.sum(symmetric)
    patterns.append(symmetric)
    # Modular pattern
    modular = np.tile(initial_h_values[:n_points//2], 2)
    modular = np.clip(modular, 0, 1)
    if np.sum(modular) > 0:
        modular = modular * target_sum / np.sum(modular)
    patterns.append(modular)
    # Gaussian smoothed
    kernel = np.exp(-np.linspace(-0.5, 0.5, 10)**2 / 0.04)
    kernel /= np.sum(kernel)
    smoothed = np.convolve(initial_h_values, kernel, mode='same')
    smoothed = np.clip(smoothed, 0, 1)
    if np.sum(smoothed) > 0:
        smoothed = smoothed * target_sum / np.sum(smoothed)
    patterns.append(smoothed)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-10)
