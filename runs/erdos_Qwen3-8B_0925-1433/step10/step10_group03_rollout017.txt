import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    # Bounds for differential evolution
    bounds = [(0, 1)] * n_points

    # Run differential evolution
    start_time = time.time()
    res_de = differential_evolution(obj, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), 
                                   recombination=0.8, 
                                   tol=1e-6, 
                                   maxiter=100, 
                                   disp=False)
    # Check if differential evolution found a solution
    if res_de.success:
        best_h = res_de.x
        best_c5 = res_de.fun
    else:
        best_h = h
        best_c5 = obj(h)

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = res_lbgf.x
    best_c5 = res_lbgf.fun

    return (best_h, best_c5, n_points)

from scipy.optimize import differential_evolution, minimize
import time
