import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate binary pattern initial guess
    def gen_binary():
        pattern = np.zeros(n_points)
        pattern[::2] = 1.0
        sum_p = np.sum(pattern)
        delta = target_sum - sum_p
        pattern[-1] += delta
        return np.clip(pattern, 0.0, 1.0)

    # Bounds and constraints
    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Objective function
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # Initial guesses
    initial_guesses = [initial_h_values.copy(), gen_binary(), np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))]

    # Optimization function
    def optimize(guess):
        res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x, res.fun

    best_h, best_c5 = initial_guesses[0].copy(), float('inf')
    for guess in initial_guesses:
        h_opt, c5 = optimize(guess)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Perturbation loop
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        h_opt, c5 = optimize(new_h)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    final_h, final_c5 = optimize(best_h)
    return (final_h, final_c5, n_points)
