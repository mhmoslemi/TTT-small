import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def validate_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def generate_candidates():
        candidates = []
        # Symmetric pattern
        sym = np.zeros(n_points)
        for i in range(n_points):
            sym[i] = 1 if i % 2 == 0 else 0
        candidates.append(validate_h(sym))
        # Periodic pattern (every 3)
        per = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                per[i] = 1
        candidates.append(validate_h(per))
        # Random pattern
        rand = np.random.rand(n_points)
        candidates.append(validate_h(rand))
        # Initial pattern
        candidates.append(validate_h(initial_h_values))
        return candidates

    candidates = generate_candidates()

    def obj(x):
        h = validate_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = candidates[0]
    best_c5 = float('inf')

    # Phase 1: Multi-start
    for h in candidates:
        if time.time() - start_time > budget_s - 100:
            break
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    # Phase 2: Perturbative refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 100:
            break
        perturb = np.random.normal(0, 0.15, n_points)
        new_h = validate_h(best_h + perturb)
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    # Phase 3: Local optimization

        if result.fun < best_c5:
