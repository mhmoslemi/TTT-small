        def obj_de(x):
            x_clip = np.clip(x, 0, 1)
            x_norm = x_clip * target_sum / np.sum(x_clip)
            corr = np.correlate(x_norm, 1 - x_norm, mode='full')
            return np.max(corr) * dx

        try:
            res_de = differential_evolution(
                obj_de, [(0, 1)] * n_points, strategy='best1bin',
                popsize=40, maxiter=50, tol=1e-6,
                mutation=(0.8, 1.0), recombination=0.9,
                seed=seed, disp=False,
                timeout=max(0, budget_s - (time.time() - start_time)) // 2
            )
        except:
            res_de = minimize(obj_de, pat, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-10)

        def obj_sqp(x):
            x_clip = np.clip(x, 0, 1)
            x_norm = x_clip * target_sum / np.sum(x_clip)
            corr = np.correlate(x_norm, 1 - x_norm, mode='full')
            return np.max(corr) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res_sqp = minimize(
            obj_sqp, res_de.x, method='SLSQP',
            bounds=[(0, 1)] * n_points, constraints=cons,
            tol=1e-8, options={'maxiter': 100},
            timeout=max(0, budget_s - (time.time() - start_time)) // 2
        )

        if res_sqp.fun < best_c5:
            best_c5 = res_sqp.fun
            best_h = res_sqp.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (best_h, final_c5, n_points)
