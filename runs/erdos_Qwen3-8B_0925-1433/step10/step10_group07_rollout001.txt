            h = h * target_sum / np.sum(h)
        return h

    def obj(x):

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Differential Evolution with adaptive parameters
    try:
        result_de = differential_evolution(

            callback=lambda x: None
        )
