import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):

        if np.sum(h) > 0:

    def eval_c5(h):

        return np.max(corr) * dx

    h_pop = [rescale(np.random.rand(n_points)) for _ in range(50)]
    h_pop = [rescale(h) for h in h_pop]
    best_h = h_pop[0]
    best_c5 = eval_c5(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:

        fitness = [eval_c5(h) for h in h_pop]
