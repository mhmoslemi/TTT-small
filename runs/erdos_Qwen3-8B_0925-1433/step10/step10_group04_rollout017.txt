        # Reduce some values
        diff = sum_alt - target_sum
        alt_init[::2] -= diff / 2
    elif sum_alt < target_sum:
        # Add some values
        diff = target_sum - sum_alt
        alt_init[::2] += diff / 2
    initial_points.append(alt_init)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    # Prepare objective function
    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for init in initial_points:
        # Initial optimization
        h = np.clip(init, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

        # Perturbation loop with early stopping
        for _ in range(50):
            if time.time() - start_time > budget_s - 60:
                break
            perturb = np.random.normal(0, 0.03, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

        if time.time() - start_time > budget_s - 60:
            break

    final_result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
