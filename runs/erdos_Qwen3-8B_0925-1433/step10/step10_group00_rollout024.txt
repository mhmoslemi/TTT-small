import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0
    iterations = 0

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Compute current max overlap and its lag
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')
        max_k = np.argmax(corr)
        if max_k >= len(corr) - 1:

        # Targeted perturbation: adjust h[x] and h[x + max_k]
        perturb = np.zeros_like(best_h)
        for x in range(n_points):
            if x + max_k < n_points:
                perturb[x] -= 0.005

        perturb += np.random.normal(0, 0.003, n_points)

        # Local optimization
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
