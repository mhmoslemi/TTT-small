import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    h_sum = np.sum(h)
    if h_sum > 0:

    max_iterations = 500
    alpha = 0.1

    best_h = h.copy()
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        if max_corr < best_corr:
            best_corr = max_corr
            best_h = h.copy()
        if np.abs(max_corr - best_corr) < 1e-8:
            break

        k_index = np.argmax(corr)
        shift = k_index - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            if 0 <= i_plus < n_points:
                term1 = 1.0 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            term2 = 0.0 if not (0 <= i_minus < n_points) else h[i_minus]
