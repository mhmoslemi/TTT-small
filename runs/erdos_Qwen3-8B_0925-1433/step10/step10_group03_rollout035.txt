import numpy as np

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    try:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6,

        )
        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = initial_h_values.copy()
        if np.sum(best_h) > 0:
            best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    elapsed = time.time() - start_time
    if elapsed < budget_s:

        bounds_lsq = [(0, 1)] * n_points
        res_lsq = minimize(objective, best_h, method='SLSQP', bounds=bounds_lsq, constraints=cons, tol=1e-8, options={'maxiter': 50})

        best_c5 = res_lsq.fun
