import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * target_sum / sum_initial
    initial_h = np.clip(initial_h, 0, 1)

    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return float('inf')
        h_norm = h_clipped * target_sum / sum_h
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Generate diverse initial population
    pop = [initial_h]
    for _ in range(5):
        pop.append(np.clip(np.random.rand(n_points), 0, 1))
    for m in [3, 4, 5]:
        p = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0:
                p[i] = 1
        pop.append(p)
    ev = np.zeros(n_points)
    ev[::2] = 1
    pop.append(ev)
    for s in [0.1, 0.2]:
        p = initial_h + np.random.normal(0, s, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        pop.append(p)

    # Differential Evolution
    t0 = time.time()
    result_de = differential_evolution(obj, bounds, popsize=20, mutation=(0.8, 1), recombination=0.9, tol=1e-10, maxiter=100, polish=True)
    best_de = result_de.x

    # SLSQP refinement
    def refine(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * target_sum / sum_h
        for _ in range(100):
            noise = np.random.normal(0, 0.01, n_points)

            sum_nh = np.sum(new_h)
            new_h = new_h * target_sum / sum_nh

            if res.fun < obj(h):
                h = res.x
        return h

    best_h = refine(best_de)
    final_c5 = obj(best_h)
    return best_h, final_c5, n_points
