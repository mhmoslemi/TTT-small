import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')

        h = np.clip(h_initial + perturb, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_fitness = float('inf')
    best_h = h_initial.copy()
    num_generations = 500

    for gen in range(num_generations):
        fitness = [obj(h) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        elites = [population[i] for i in elite_indices]
        new_population = elites.copy()

        while len(new_population) < len(population):
            indices = np.random.choice(len(population), 4, replace=False)
            candidates = [population[i] for i in indices]

            parents = sorted_candidates[:2]
