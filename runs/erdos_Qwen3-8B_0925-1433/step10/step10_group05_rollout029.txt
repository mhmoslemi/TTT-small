import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    h = np.clip(h, 0, 1)
    
    max_iterations = 500
    learning_rate = 0.15
    time_start = time.time()
    
    for _ in range(max_iterations):
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        # Determine lag from index
        lag = max_idx - (n_points - 1)
        # Compute gradient for current lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + lag
            term1 = 1.0 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - lag
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = term1 - term2
        # Update h
        h -= learning_rate * grad
        # Clip and renormalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        # Stop if max correlation is negligible
        if max_corr < 1e-6:
            break
        # Check time
        if time.time() - time_start > budget_s:
            break
    
    # Final evaluation
    corr_final = np.correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(corr_final)
    return (h, c5_bound, n_points)
