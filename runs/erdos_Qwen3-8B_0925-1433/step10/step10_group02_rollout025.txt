import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial patterns
    starts = [initial_h_values.copy()]
    for i in range(1, 4):
        if i == 1:
            starts.append(np.array([1 if j % 2 == 0 else 0 for j in range(n_points)]))
        elif i == 2:
            starts.append(np.array([1 if j % 3 == 0 else 0 for j in range(n_points)]))
        else:
            starts.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    # Initial optimization
    best_h = None
    best_c5 = float('inf')
    for start in starts:
        h_start = np.clip(start, 0, 1)
        sum_h_start = np.sum(h_start)
        if sum_h_start > 0:
            h_start = h_start * target_sum / sum_h_start
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Refinement with decreasing perturbations
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.15, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement
    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    return (final_res.x, final_res.fun, n_points)
