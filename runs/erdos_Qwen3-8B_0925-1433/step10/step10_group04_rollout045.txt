import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_patterns = [initial_h_values]

    def generate_spaced_binary(n_points):
        block_length = n_points // 5
        spacing = (n_points - 5 * block_length) // 6
        h = np.zeros(n_points)
        for i in range(5):
            start = i * (block_length + spacing)
            h[start:start + block_length] = 1.0
        return h

    def generate_periodic(n_points, period):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period < period // 2:
                h[i] = 1.0
        return h

    initial_patterns.append(generate_spaced_binary(n_points))
    initial_patterns.append(generate_periodic(n_points, n_points // 5))
    initial_patterns.append(np.random.rand(n_points))

    best_h = None
    best_c5 = float('inf')

    for pattern in initial_patterns:
        pattern = pattern / np.sum(pattern) * target_sum
        res = minimize(objective, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h / np.sum(new_h) * target_sum
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    final_res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
    if final_res.success and final_res.fun < best_c5:
        best_h, best_c5 = final_res.x, final_res.fun

    return (best_h, best_c5, n_points)
