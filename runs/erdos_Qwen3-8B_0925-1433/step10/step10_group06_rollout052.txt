from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = [initial_h_values.copy() for _ in range(100)]
    for h in population:
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()
