import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

            end = start + block_length
            s = int(np.clip(start / dx, 0, n_points - 1))
            e = int(np.clip(end / dx, 1, n_points))
            h[s:e] = 1.0
        return h

    def periodic_binary():
        h = np.zeros(n_points)
        for i, x in enumerate(np.linspace(0, 2, n_points)):
            if (x % 0.5) < 0.25:
                h[i] = 1.0
        return h

    def random_binary():
        h = np.random.rand(n_points)
        return h * target_sum / np.sum(h)

    for pt in initial_points:

                       bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success and res.fun < best_c5:
