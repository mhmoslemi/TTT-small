import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_temp = 0.1 * best_c5

    sa_iterations = 1000
    sa_perturb_std = 0.01
    start_time = time.time()

    for _ in range(sa_iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, sa_perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / initial_temp):
            best_h = new_h.copy()
            best_c5 = new_c5

    # Local Refinement with SLSQP
