            if time.time() - start_time > budget_s - 50:
                break
            perturb = np.random.normal(0, 0.03, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun

    # Final optimization
    final_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    return (final_result.x, final_result.fun, n_points)
