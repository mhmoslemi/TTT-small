import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    initial_h = normalize(initial_h_values.copy())
    population = [initial_h.copy() for _ in range(50)]
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()

    # Initial population evaluation
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Genetic algorithm loop
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        evaluations = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            evaluations.append((c5, h))
        evaluations.sort()
        top_options = evaluations[:int(len(evaluations) * 0.2)]
        top_h = [h for (c5, h) in top_options]
        new_pop = []
        for _ in range(len(population) // 2):
            parent1, parent2 = np.random.choice(top_h, 2, replace=False)
            child = (parent1 + parent2) / 2
            new_pop.append(normalize(child))
            mut = parent1 + np.random.normal(0, 0.01, len(parent1))
            new_pop.append(normalize(mut))
        combined = population + new_pop
        combined.sort(key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
        population = combined[:len(population)]
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Local refinement with SLSQP
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
