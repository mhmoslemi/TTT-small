import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    best_h = h.copy()
    best_c5 = current_c5

    T = T_initial

    n_iter = 10000

    start_time = time.time()

    while iter_count < n_iter and time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.01, n_points)
        h_candidate = np.clip(h + perturb, 0, 1)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate > 0:
            h_candidate = h_candidate * target_sum / sum_candidate
        else:
            h_candidate = h.copy()

        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        T *= alpha
        iter_count += 1
