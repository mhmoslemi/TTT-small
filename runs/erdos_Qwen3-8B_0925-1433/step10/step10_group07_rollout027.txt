import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_p = x
        if np.abs(np.sum(h_p)) > 1e-8:
            h_p = h_p * target_sum / np.sum(h_p)
        corr = np.correlate(h_p, 1 - h_p, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    try:
        result_de = differential_evolution(
            obj, bounds, strategy='best1bin',
            popsize=50, maxiter=150, tol=1e-8, vectorized=True,
            mutation=(0.5, 1), recombination=0.8, mutation_scale=0.5
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        best_h, best_c5 = h, obj(h)

    try:
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result_sqp = minimize(
            obj, best_h, method='SLSQP', bounds=bounds,
            constraints=constraints, tol=1e-8, options={'maxiter': 150}
        )
        best_h, best_c5 = result_sqp.x, result_sqp.fun
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        pass

    return (best_h, best_c5, n_points)
