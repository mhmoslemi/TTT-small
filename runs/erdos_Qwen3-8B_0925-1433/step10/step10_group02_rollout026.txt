import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h == 0:
            return np.inf
        h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Generate patterns
    def generate_patterns():
        patterns = []
        # Even spacing
        even = np.zeros(n_points)
        even[::2] = 1
        patterns.append(even)
        # Symmetric
        sym = np.zeros(n_points)
        sym[::2] = 1
        sym[1::2] = 0
        patterns.append(sym)
        # Modular
        mod = np.concatenate([initial_h_values[:n_points//2], initial_h_values[:n_points//2]])
        patterns.append(mod)
        # Smoothed
        def gauss_smooth(x, sigma=0.2):
            kernel = np.exp(-np.linspace(-3,3,100)**2/(2*sigma**2))
            kernel /= kernel.sum()
            return np.convolve(x, kernel, mode='same')
        smooth = np.clip(gauss_smooth(initial_h_values, sigma=0.2), 0, 1)
        patterns.append(smooth)
        # Add initial
        patterns.append(initial_h_values.copy())
        return patterns
    
    patterns = generate_patterns()
    
    best_h = np.copy(initial_h_values)
    best_c5 = np.inf
    
    # Initial optimization
    for p in patterns:
        p = np.clip(p, 0, 1)
        if np.sum(p) == 0:
            continue
        p = p * target_sum / np.sum(p)
        res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Perturbation phases
    start_time = time.time()
    phases = [(0.2, 200), (0.05, 500), (0.01, 1000)]
