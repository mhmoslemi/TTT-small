import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h_ = np.clip(x, 0, 1)
        h_sum = np.sum(h_)
        if h_sum > 0:
            h_ = h_ * target_sum / h_sum
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    temp = 0.1 * best_c5
    cool_rate = 0.995
    sa_iter = 1000
    for _ in range(sa_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)

        new_c5 = objective(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        temp *= cool_rate

    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})

    return (np.clip(best_h, 0, 1), best_c5, n_points)
