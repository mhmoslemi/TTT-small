import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    
    def obj(x):
        h_ = x
        sum_h = np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        penalty = 10000 * (sum_h - target_sum)**2
        return np.max(corr) * dx + penalty
    
    bounds = [(0, 1)] * n_points
    start_time = time.time()
    
    try:
        result_de = differential_evolution(
            obj, bounds, strategy='best1bin', popsize=50, maxiter=150,
            tol=1e-8, vectorized=True, mutation=(0.5, 1), recombination=0.8,
            mutation_scale=0.5
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        best_h, best_c5 = h, obj(h)
    
    try:
        result_slsqp = minimize(
            obj, best_h, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            tol=1e-10, options={'maxiter': 100}
        )
        best_h, best_c5 = result_slsqp.x, result_slsqp.fun
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        pass
    
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
