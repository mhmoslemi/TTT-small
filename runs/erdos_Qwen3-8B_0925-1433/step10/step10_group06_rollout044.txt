import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points, dx):
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    population = [np.clip(initial_h_values.copy(), 0, 1).copy() for _ in range(50)]
    for h in population:
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()
        
        new_population = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            h1, h2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.4:
                alpha = np.random.uniform(0.4, 0.6)
                child = alpha * h1 + (1 - alpha) * h2
            else:
                child = h1 + np.random.normal(0, 0.07, n_points)
            child = np.clip(child, 0, 1)
            sum_ch = np.sum(child)
            if sum_ch > 0:
                child = child * target_sum / sum_ch
            new_population.append(child)
        population = new_population + population[:50]
        population.sort(key=lambda x: evaluate(x))
        population = population[:50]
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(evaluate, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-10, options={'maxiter': 50})
    final_h = result.x
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
