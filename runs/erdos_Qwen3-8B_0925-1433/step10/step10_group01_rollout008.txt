import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    no_improve_count = 0

    phases = [
        ('COBYLA', 0.05, 100, 1e-5),
        ('L-BFGS-B', 0.005, 120, 1e-6),
        ('SLSQP', 0.001, 80, 1e-8)
    ]

    for phase_name, perturb_std, iter_count, tol in phases:
        if time.time() - start_time > budget_s:
            break
        for _ in range(iter_count):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            if phase_name == 'COBYLA':
                res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=tol)
            elif phase_name == 'L-BFGS-B':
                res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=tol)
            else:  # SLSQP
                res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=tol)
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = res.x, current_c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
            if no_improve_count >= 20:
                break

    return (best_h, best_c5, n_points)
