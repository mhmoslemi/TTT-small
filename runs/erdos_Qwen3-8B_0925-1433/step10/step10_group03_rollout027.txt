from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 30:
            break
        # Random perturbation

        pert_h = h + pert
        pert_h = np.clip(pert_h, 0, 1)
        pert_h = pert_h * target_sum / np.sum(pert_h) if np.sum(pert_h) > 0 else h

        # Global optimization
        try:
            res = minimize(obj, pert_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        except:

        if res.success and res.fun < best_c5:
