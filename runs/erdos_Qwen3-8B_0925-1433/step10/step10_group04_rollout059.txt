            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

    for h_start in initial_guesses:
        h_start = h_start.copy()
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start = h_start * target_sum / sum_start
        h_start = np.clip(h_start, 0, 1)
        result = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

        for _ in range(100):
            if time.time() - start_time > budget_s - 60:
                break
            perturb = np.random.normal(0, 0.1, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
                current_h = res.x

    final_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    return (final_result.x, final_result.fun, n_points)
