import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        # Random perturbation
        pert = np.random.normal(0, 0.05, n_points)
        pert_h = h + pert
        pert_h = np.clip(pert_h, 0, 1)
        pert_h = pert_h * target_sum / np.sum(pert_h) if np.sum(pert_h) > 0 else h

        # Global optimization with SLSQP
        res = minimize(obj, pert_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

        # Local refinement with L-BFGS-B
        res_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 50})
        best_h = res_lbgf.x
        best_c5 = res_lbgf.fun

        # Periodic reinitialization to escape plateaus
        if _ % 20 == 0:
            pert = np.random.uniform(-0.1, 0.1, n_points)
            pert_h = h + pert
            pert_h = np.clip(pert_h, 0, 1)
            pert_h = pert_h * target_sum / np.sum(pert_h) if np.sum(pert_h) > 0 else h
            res = minimize(obj, pert_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 20})
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun

    return (best_h, best_c5, n_points)
