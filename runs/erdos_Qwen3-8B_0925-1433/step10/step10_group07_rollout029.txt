from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) > 1e-8:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - target_sum

    initial_h = np.clip(initial_h_values, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros(n_points)

    bounds = Bounds(0, 1)

    start_time = time.time()
    elapsed = 0

    # Genetic Algorithm with constraint-aware mutations
    population = [initial_h + np.random.randn(n_points) * 0.1 for _ in range(50)]
    population.append(initial_h)
    best_h = initial_h
    best_c5 = objective(best_h)

    for _ in range(200):
        if elapsed > budget_s - 60:
            break
        offspring = []
        for parent in population:
            mutant = parent.copy()
            idx = np.random.choice(n_points)
            mutant[idx] += np.random.uniform(-0.2, 0.2)
            mutant = np.clip(mutant, 0, 1)
            sum_mut = np.sum(mutant)
            if abs(sum_mut - target_sum) > 1e-8:
                diff = (target_sum - sum_mut) / n_points
                mutant += diff
                mutant = np.clip(mutant, 0, 1)
            offspring.append(mutant)
        population += offspring
        population = np.array(population)
        scores = np.array([objective(p) for p in population])
        order = np.argsort(scores)
        population = population[order[:50]]
        best_h = population[0]

        elapsed = time.time() - start_time

    # Local optimization
