    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def objective(x):
        x_clip = np.clip(x, 0, 1)
        x_sum = np.sum(x_clip)
        if x_sum == 0:
            return 1e6  # Penalize zero sum
        x_norm = x_clip * (target_sum / x_sum)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)

    result = minimize(
        objective, h, method='L-BFGS-B', bounds=bounds,
        tol=1e-8, options={'maxiter': 100}
    )
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
