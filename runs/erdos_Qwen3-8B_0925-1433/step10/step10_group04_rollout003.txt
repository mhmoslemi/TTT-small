import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Generate structured initial guesses
    guesses = [initial_h_values.copy()]
    binary = np.zeros(n_points)
    binary[::2] = 1.0
    idx = np.where(binary == 0)[0][0]
    binary[idx] += target_sum - np.sum(binary)
    binary = np.clip(binary, 0, 1)
    guesses.append(binary)
    guesses.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()
    for guess in guesses:
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    final_res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_res.x, final_res.fun, n_points)
