            obj, best_h, method='SLSQP',
            bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 200}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        pass

    h_values = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_values)
    if abs(sum_h - target_sum) > 1e-6:
        h_values = h_values * target_sum / sum_h
    best_c5 = evaluate_erdos_solution(h_values)

    return (h_values, best_c5, n_points)
