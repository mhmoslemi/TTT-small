import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(arr):
        arr = np.clip(arr, 0, 1)
        s = np.sum(arr)
        if s == 0:
            return np.zeros(n_points)
        return arr * target_sum / s

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        if _ < 20:
            h = initial_h_values.copy()
        elif _ < 30:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.random.laplace(0.5, 0.2, n_points)
        population.append(rescale(h))

    best_h = population[0]
    best_c5 = fitness(best_h)
    generation = 0

    start_time = time.time()
    while time.time() - start_time < budget_s - 50:
        generation += 1
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            selected.append(min([a, b, c], key=fitness))
        selected = selected[:pop_size]

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if i+1 >= pop_size:
                break
            p1 = selected[i]
            p2 = selected[i+1]
            cross_point = np.random.randint(1, n_points)
            child1 = np.concatenate((p1[:cross_point], p2[cross_point:]))
            child2 = np.concatenate((p2[:cross_point], p1[cross_point:]))
            offspring.append(rescale(child1))
            offspring.append(rescale(child2))
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < 0.1:
                idx = np.random.choice(n_points, 5, replace=False)
                offspring[i][idx] += np.random.uniform(-0.1, 0.1)
                offspring[i] = rescale(offspring[i])

        # Replace population
        population = selected[:pop_size//2] + offspring
        population = [rescale(h) for h in population]

        # Update best
        for h in population:
            c5 = fitness(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
        if generation % 10 == 0:
            print(f"Generation {generation}: C5 = {best_c5}")

    # Final smoothing
    from scipy.ndimage import gaussian_filter1d
    best_h = gaussian_filter1d(best_h, sigma=n_points/12)
    best_h = rescale(best_h)
    return (best_h, fitness(best_h), n_points)
