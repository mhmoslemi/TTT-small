import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_symmetric(n_points):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            pattern[i] = 1.0 if i % 2 == 0 else 0.0
        sum_p = np.sum(pattern)
        if sum_p > target_sum:
            pattern[-1] = 0.0
        pattern = pattern * target_sum / sum_p
        return pattern

    def generate_periodic(n_points, m):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if (i % m) == 0:
                pattern[i] = 1.0
        sum_p = np.sum(pattern)
        pattern = pattern * target_sum / sum_p
        return pattern

    initial_guesses = [initial_h_values.copy(), generate_symmetric(n_points), generate_periodic(n_points, 3), np.random.rand(n_points)]
    for i in range(len(initial_guesses)):
        initial_guesses[i] = np.clip(initial_guesses[i], 0, 1)
        initial_guesses[i] = initial_guesses[i] * target_sum / np.sum(initial_guesses[i])

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize(h0):
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        return result.x, result.fun

    best_h, best_c5 = initial_guesses[0], obj(initial_guesses[0])
    for guess in initial_guesses[1:]:
        h_opt, c5_opt = optimize(guess)
        if c5_opt < best_c5:

    start_time = time.time()
