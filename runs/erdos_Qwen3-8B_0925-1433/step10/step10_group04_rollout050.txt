import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial setup with multiple starting points
    best_h = None
    best_c5 = float('inf')

    # Define a few initial guess patterns
    def gen_binary():
        res = np.zeros(n_points)
        res[::2] = 1.0
        res = res * target_sum / np.sum(res) if np.sum(res) > 0 else res
        return res

    def gen_random():
        res = np.random.rand(n_points)
        res = res * target_sum / np.sum(res) if np.sum(res) > 0 else res
        return res

    # Evaluate initial guesses
    for guess in [initial_h_values, gen_binary(), gen_random()]:
        h = np.clip(guess, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Local refinement with perturbations
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    final_result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
