import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    def normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            return h * target_sum / np.sum(h)
        return np.zeros_like(h)

    h_pop = [normalize(initial_h_values)]
    for _ in range(19):
        pert = np.random.normal(0, 0.1, n_points)
        h = normalize(initial_h_values + pert)
        h_pop.append(h)

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = normalize(initial_h_values)
    best_c5 = eval_c5(best_h)

    # Genetic Algorithm
    pop_size = 20
    gen_max = 500
    elite_frac = 0.2
    mut_std = 0.02

    for gen in range(gen_max):
        if time.time() - start_time > budget_s:
            break
        scores = [eval_c5(h) for h in h_pop]
        elite_idx = np.argsort(scores)[:int(elite_frac * pop_size)]
        elites = [h_pop[i] for i in elite_idx]

        new_pop = elites[:]
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(elite_idx, 2, replace=False)
            parent1, parent2 = h_pop[p1], h_pop[p2]
            idx = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:idx], parent2[idx:]])
            child = np.clip(child + np.random.normal(0, mut_std, n_points), 0, 1)
            child = normalize(child)
            new_pop.append(child)

        h_pop = new_pop
        current_best = min(scores)
        if current_best < best_c5:
            best_h = h_pop[scores.index(current_best)]
            best_c5 = current_best

    # Local refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h = best_h.copy()
        res = minimize(lambda x: eval_c5(x), h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
