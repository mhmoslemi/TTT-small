def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=np.float64)
    # Ensure sum is correct
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    # Parameters
    step_size = 0.01
    step_decay = 0.999
    max_iterations = 500
    improvements = 0

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        if max_corr == 0:
            break  # No more improvements
        # Find the index m of the maximum
        max_index = np.argmax(corr)
        # Compute lag k
        k = max_index - (n_points - 1)
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h[j]) * dx
        # Update h
        h -= step_size * grad
        # Clip and normalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        # Compute new C5
        new_max_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_max_corr < best_c5:

            improvements = 0
        else:
            improvements += 1
            if improvements >= 50:
                break
        step_size *= step_decay

    return (best_h, best_c5, n_points)
