import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj_de(x):
        h_ = np.clip(x, 0, 1)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h, best_c5 = h, obj_de(h)

    try:
        result = differential_evolution(
            obj_de, bounds, strategy='rand1bin', popsize=50,
            maxiter=200, tol=1e-8, mutation=(0.7, 1.3),
            recombination=0.8, vectorized=True, polish=True, disp=False
        )
        best_h, best_c5 = result.x, result.fun
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed < budget_s - 100:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        try:
            result = minimize(
                lambda x: np.max(np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')) * dx,
                best_h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            best_h, best_c5 = result.x, result.fun
        except:
            pass

    h_final = np.clip(best_h, 0, 1)
    if np.abs(np.sum(h_final) - target_sum) > 1e-8:
        h_final = h_final * target_sum / np.sum(h_final)
    return (h_final, best_c5, n_points)
