import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_norm(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return h

    def obj(x):
        h = clip_norm(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def optimize(initial):
        res = minimize(obj, initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            return res.x, res.fun
        return initial, np.inf

    h = clip_norm(initial_h_values.copy())
    best_h = h
    best_c5 = obj(h)

    for _ in range(3):
        if np.random.rand() < 0.5:
            h = np.random.rand(n_points)
            h = clip_norm(h)
        else:
            h = np.roll(initial_h_values.copy(), np.random.randint(n_points))
            h = clip_norm(h)
        curr_h, curr_c5 = optimize(h)
        if curr_c5 < best_c5:
            best_h, best_c5 = curr_h, curr_c5

    start_time = time.time()
    for _ in range(50):

        perturbed = np.clip(best_h + noise, 0, 1)
        perturbed = clip_norm(perturbed)
        curr_h, curr_c5 = optimize(perturbed)
        if curr_c5 < best_c5:

    final_h, final_c5 = optimize(best_h)
