from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

        patterns.append(h_base)

        h_alt = np.zeros(n_points)
        h_alt[::2] = 1.0
        h_alt = h_alt * target_sum / np.sum(h_alt)
        patterns.append(h_alt)

        h_cluster = np.zeros(n_points)
        h_cluster[::n_points//3] = 1.0
        h_cluster = h_cluster * target_sum / np.sum(h_cluster)
        patterns.append(h_cluster)

        h_sinusoidal = 0.5 * (1 + np.sin(np.linspace(0, np.pi, n_points)))
        h_sinusoidal = np.clip(h_sinusoidal, 0, 1)
        h_sinusoidal = h_sinusoidal * target_sum / np.sum(h_sinusoidal)
        patterns.append(h_sinusoidal)

        h_rand = np.random.permutation(initial_h_values)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        patterns.append(h_rand)

    patterns = generate_patterns()
    best_h = patterns[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)

    def objective(x):
        x_clip = np.clip(x, 0, 1)

        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    def refine(x0):
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, x0, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8, options={'maxiter': 100})
        return res.x

    for h in patterns:
        h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
        h = h * target_sum / np.sum(h)
        try:
            res_de = differential_evolution(
                objective, [(0,1)]*n_points, strategy='best1bin', popsize=30, maxiter=100,
                mutation=(0.7, 1.0), recombination=0.8, seed=seed, tol=1e-8, disp=False,
                timeout=remaining_time / len(patterns)
            )
            h_de = res_de.x
