import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def generate_initial_points():
        initial = np.array(initial_h_values)
        random = np.random.rand(n_points)
        m = n_points // 3
        periodic = np.zeros(n_points)
        periodic[::m] = 1
        symmetric = np.zeros(n_points)
        symmetric[:n_points//2] = 1
        return [initial, random, periodic, symmetric]

    initial_points = generate_initial_points()
    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    start_time = time.time()

    for h in initial_points:
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros(n_points)
        res = minimize(obj, h_norm, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for scale in [0.15, 0.05, 0.01]:
        if time.time() - start_time > budget_s - 100:
            break
        perturb = np.random.normal(0, scale, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        res = minimize(obj, new_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    if time.time() - start_time < budget_s - 50:
        final_res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-10)
        if final_res.success:
            best_h = final_res.x
            best_c5 = final_res.fun

    return (best_h, best_c5, n_points)
