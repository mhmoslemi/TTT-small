import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def process_candidate(candidate):
        candidate = np.clip(candidate, 0, 1)
        sum_c = np.sum(candidate)
        if sum_c > 0:
            candidate = candidate * target_sum / sum_c
        return candidate

    initial = np.array(initial_h_values)
    initial = process_candidate(initial)
    population = [initial.copy()]
    for _ in range(49):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial + perturb
        candidate = process_candidate(candidate)
        population.append(candidate)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([evaluate(h) for h in population])
        indices = np.argsort(scores)[:len(population) // 2]
        top_candidates = [population[i] for i in indices]
        new_population = []
        for _ in range(len(population) - len(top_candidates)):
            p1, p2 = np.random.choice(top_candidates, 2, replace=False)
            child = (p1 + p2) / 2
            mutate = np.random.normal(0, 0.01, n_points)
            child += mutate
            child = process_candidate(child)
            new_population.append(child)
        population = top_candidates + new_population
        current_scores = np.array([evaluate(h) for h in population])
        best_index = np.argmin(current_scores)
        if current_scores[best_index] < best_c5:
            best_h = population[best_index].copy()
            best_c5 = current_scores[best_index]

    def obj(x):
        h = process_candidate(x)
        return evaluate(h)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    return (best_h, best_c5, n_points)
