from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = [initial_h_values.copy() for _ in range(50)]
    population += [np.clip(np.random.uniform(0, 1, n_points), 0, 1).copy() for _ in range(25)]
    for h in population:
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    alpha_decay = 0.995
    sigma = 0.07

    for _ in range(300):
        if time.time() - start_time > budget_s - 100:
            break
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        new_population = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[idxs[0]], population[idxs[1]]
            if np.random.rand() < 0.8:
                alpha = np.random.uniform(0.4, 0.6)
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1 + np.random.normal(0, sigma, n_points)
            child = np.clip(child, 0, 1)

            new_population.append(child)
        population = new_population + [best_h for _ in range(25)]
        population.sort(key=lambda x: evaluate(x))

    result = minimize(lambda x: evaluate(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-10)
