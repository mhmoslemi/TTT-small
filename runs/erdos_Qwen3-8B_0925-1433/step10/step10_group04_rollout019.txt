            best_c5 = current_c5
        for _ in range(20):
            if time.time() - start_time > budget_s - 50:
                break
            perturb = np.random.normal(0, 0.02, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_h
            res = minimize(evaluate, new_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
            if res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun

    final_res = minimize(evaluate, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
    return final_res.x, final_res.fun, n_points
