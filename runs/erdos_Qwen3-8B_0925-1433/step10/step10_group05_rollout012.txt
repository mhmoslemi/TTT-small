    h = np.array(initial_h_values)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    start_time = time.time()
    
    for iteration in range(500):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += epsilon
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full') * dx
            grad[i] = (corr_perturb[k_max] - corr[k_max]) / epsilon
        
        step = -0.05 * grad
        h += step
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        if iteration % 10 == 0:
            h += np.random.normal(0, 0.01, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        current_corr = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_corr < best_c5:
            best_c5 = current_corr
            best_h = h.copy()
    
    final_corr = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    return (best_h, final_corr, n_points)
