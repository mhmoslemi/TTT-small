import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        h_scaled = h * target_sum / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Perform DE without local refinement first
    try:
        opt_result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 0.8),
            recombination=0.9,
            maxiter=100,
            polish=False,
            disp=False,
            vectorized=True

        best_h = opt_result.x
