import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    population = []
    initial_scaled = np.array(initial_h_values) * target_sum / np.sum(initial_h_values)
    population.extend([initial_scaled] * 20)
    for _ in range(10):
        rand = np.random.rand(n_points)
        scaled = rand * target_sum / np.sum(rand)
        population.append(scaled)
    for _ in range(20):
        rand = np.random.rand(n_points)
        scaled = rand * target_sum / np.sum(rand)
        population.append(scaled)
    np.random.shuffle(population)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_scaled
    best_c5 = evaluate(best_h)

    max_generations = 100
    mutation_std = 0.05
    tournament_size = 5
    elite_fraction = 0.2
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 100:
            break
        fitnesses = [evaluate(h) for h in population]
        best_idx = np.argmin(fitnesses)
        current_best = population[best_idx]
        if fitnesses[best_idx] < best_c5:
            best_h = current_best
            best_c5 = fitnesses[best_idx]

        elite_count = int(elite_fraction * len(population))
        elite_indices = np.argsort(fitnesses)[:elite_count]
        elites = [population[i] for i in elite_indices]

        new_population = elites[:]
        while len(new_population) < len(population):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            cross_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            mutation = np.random.normal(0, mutation_std, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            scaling_factor = target_sum / current_sum if current_sum else 1.0
            child = child * scaling_factor
            new_population.append(child)

        population = new_population

    # Local refinement
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    scaling_factor = target_sum / current_sum if current_sum else 1.0
    best_h = best_h * scaling_factor

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    best_h = result.x
    best_c5 = result.fun

    return (best_h, best_c5, n_points)
