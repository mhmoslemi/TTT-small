    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        s = np.sum(new_h)
        if s > 0:
            new_h = new_h * target_sum / s
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Final evaluation
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    return (result.x, result.fun, n_points)
