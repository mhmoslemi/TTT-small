import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with initial_h_values and perturbations
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(100)]
    for h in population:
        if np.sum(h) > 0:
            h[:] = h * target_sum / np.sum(h)
        else:
            h[:] = np.zeros(n_points)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(700):
        if time.time() - start_time > budget_s - 50:
            break
        # Evaluate all candidates
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

        # Tournament selection
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 5, replace=False)
            parent = population[np.argmin([fitness[i] for i in idxs])]
            parents.append(parent)

        # Crossover and mutation
        offspring = []
        for i in range(50):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            if np.random.rand() < 0.7:
                alpha = np.random.uniform(0.1, 0.9)
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1 + np.random.normal(0, 0.03, n_points)
            child = np.clip(child, 0, 1)
            if np.sum(child) > 0:
                child = child * target_sum / np.sum(child)
            offspring.append(child)

        # Elitism: keep top 20%
        combined = population + offspring
        combined.sort(key=lambda x: evaluate(x))
        population = combined[:100]

    # Local refinement
    result = minimize(lambda x: evaluate(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-10)
    return (result.x, result.fun, n_points)
