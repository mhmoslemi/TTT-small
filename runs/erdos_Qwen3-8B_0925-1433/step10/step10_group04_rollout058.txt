        # Perturb and optimize
        current_h = result.x
        for _ in range(100):
            if time.time() - start_time > budget_s - 50:
                break
            perturb = np.random.normal(0, 0.03, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_h
            result_perturb = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
            if result_perturb.fun < best_c5:
                best_c5 = result_perturb.fun
                best_h = result_perturb.x
            current_h = result_perturb.x

    # Final refinement
    result_final = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
    best_h = result_final.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else np.zeros(n_points)
    return (best_h, result_final.fun, n_points)
