import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h)

    T_initial = 0.1 * best_c5
    alpha = 0.995
    iterations = 1000
    start_time = time.time()

    for i in range(iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        h_candidate = h + perturb
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate > 0:
            h_candidate = h_candidate * target_sum / sum_candidate
        c5_candidate = obj(h_candidate)
        prev_best_c5 = best_c5
        if c5_candidate < prev_best_c5:
            best_c5 = c5_candidate
            h = h_candidate
        else:
            delta = c5_candidate - prev_best_c5
            if np.random.rand() < np.exp(-delta / T_initial):
                h = h_candidate
        T_initial *= alpha

    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    else:
        best_h = np.zeros(n_points)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
