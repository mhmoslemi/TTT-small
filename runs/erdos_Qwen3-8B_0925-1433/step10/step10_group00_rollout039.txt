import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(45)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clamp(x): return np.clip(x, 0, 1)
    def normalize(v): return v * target_sum / np.sum(v)

    def compute_c5(h_vals):
        h = clamp(h_vals)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 20
    population = [normalize(clamp(h + np.random.normal(0, 0.1, n_points))) for _ in range(population_size)]

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    # GA parameters
    generations = 500
    tournament_size = 4
    mutation_std = 0.02
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        elite_idx = np.argsort(fitness)[:population_size // 5]
        elites = [population[i] for i in elite_idx]
        new_population = elites.copy()

        while len(new_population) < population_size:
            a, b = np.random.randint(0, population_size, 2)
            parent1, parent2 = population[a], population[b]
            crossover = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover], parent2[crossover:]])
            mutation = np.random.normal(0, mutation_std, n_points)
            child += mutation
            child = normalize(clamp(child))
            new_population.append(child)
        population = new_population

        for ind in population:
            c5 = compute_c5(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()

    # Local refinement
    candidates = [population[i] for i in np.argsort([compute_c5(p) for p in population])[:3]]
    for cand in candidates:
        def obj(x):
            return compute_c5(x)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, cand, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
