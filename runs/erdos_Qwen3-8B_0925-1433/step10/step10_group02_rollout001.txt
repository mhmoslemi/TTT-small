import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate initial patterns
    initial_patterns = [initial_h_values.copy()]
    # Alternating pattern
    alt = np.zeros(n_points)
    alt[::2] = 1.0
    if n_points % 2 == 1:
        alt[-1] = 0.5
    if np.sum(alt) > 0:
        alt = alt * target_sum / np.sum(alt)
    initial_patterns.append(alt)
    # Random pattern
    ran = np.random.rand(n_points)
    ran = ran * target_sum / np.sum(ran)
    initial_patterns.append(ran)
    # Periodic pattern
    per = np.zeros(n_points)
    for i in range(n_points):
        if (i % 3) == 0:
            per[i] = 1.0
    if np.sum(per) > 0:
        per = per * target_sum / np.sum(per)
    initial_patterns.append(per)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h in initial_patterns:
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) > 0:
            h_clipped = h_clipped * target_sum / np.sum(h_clipped)
        result = minimize(obj, h_clipped, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    for _ in range(200):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    final_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (final_result.x, final_result.fun, n_points)
