import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    pop_size, generations = 50, 100
    population = [h.copy() for _ in range(pop_size)]

    for gen in range(generations):
        fitness = []
        for ind in population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            fitness.append(-np.max(corr) * dx)
        population.sort(key=lambda x: -np.max(np.correlate(x, 1 - x, mode='full')) * dx)
        parents = population[:int(pop_size * 0.2)]
        offspring = []
        while len(offspring) < pop_size:
            if np.random.rand() < 0.7:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = (p1 + p2) / 2
            else:
                child = np.random.rand(n_points)
            mut = np.random.normal(0, 0.05, n_points)
            child += mut
            sum_c = np.sum(child)
            child *= target_sum / sum_c if sum_c else 0
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            child *= target_sum / sum_c if sum_c else 0
            offspring.append(child)
        population = offspring
        curr_best = population[0]
        curr_c5 = np.max(np.correlate(curr_best, 1 - curr_best, mode='full')) * dx
        if curr_c5 < best_c5:
            best_h = curr_best.copy()
            best_c5 = curr_c5

    def objective(x):
        h_ = x
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        lag = np.argmax(corr) - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = (1 - x[j]) * dx
        return grad

    result = minimize(
        objective, best_h, method='L-BFGS-B',
        jac=gradient, bounds=[(0, 1)] * n_points,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    final_h = result.x
    final_c5 = result.fun

    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5

    final_h = np.clip(final_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final:
        final_h = final_h * target_sum / sum_final

    return (final_h, best_c5, n_points)
