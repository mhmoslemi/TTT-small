    initial_params = [1.0, 0.0]
    result = minimize(objective, initial_params, bounds=[(0.5, 2.0), (0, 2 * np.pi)], method='L-BFGS-B', tol=1e-10)
    best_h = 0.5 + 0.5 * np.sin(2 * np.pi * result.x[0] * np.linspace(0, 2, n_points) + result.x[1])
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
