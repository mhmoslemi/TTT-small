        rand = np.clip(rand, 0, 1)
        sum_r = np.sum(rand)
        if sum_r > 0:
            rand = rand * target_sum / sum_r
        population.append(rand)
    for m in [3, 4, 5]:
        pat = np.zeros(n_points)
        for i in range(n_points):
            if i % m == 0:
                pat[i] = 1
        sum_p = np.sum(pat)
        if sum_p > 0:
            pat = pat * target_sum / sum_p
        population.append(pat)
    even = np.zeros(n_points)
    even[::2] = 1
    sum_e = np.sum(even)
    if sum_e > 0:
        even = even * target_sum / sum_e
    population.append(even)
    for s in [0.1, 0.2]:
        mod = h + np.random.normal(0, s, n_points)
        mod = np.clip(mod, 0, 1)
        sum_m = np.sum(mod)
        if sum_m > 0:
            mod = mod * target_sum / sum_m
        population.append(mod)

    result = differential_evolution(obj, [(0, 1)] * n_points, popsize=len(population), strategy='best1bin', maxiter=100, tol=1e-6)
    best_h = result.x
    best_c5 = obj(best_h)

    def refine(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    for _ in range(30):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(refine, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        perturb = np.random.normal(0, 0.05, n_points)

        res = minimize(refine, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    final_res = minimize(refine, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    return (final_res.x, final_res.fun, n_points)
