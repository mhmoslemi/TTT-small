from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate a better initial guess based on a sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    scale_factor = target_sum / current_sum
    h = h * scale_factor

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Apply structured perturbations
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
