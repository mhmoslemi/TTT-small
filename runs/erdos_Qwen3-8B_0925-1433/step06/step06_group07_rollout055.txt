        top_lags = int(top_lags_percent * n_points)
        selected_lags = top_indices[:top_lags]
        
        combined_gradient = np.zeros(n_points)
        for lag in selected_lags:
            g = np.zeros(n_points)
            i_plus_k = np.arange(n_points) + lag
            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            g[valid_plus] += (1 - h[i_plus_k[valid_plus]]) * dx
            i_minus_k = np.arange(n_points) - lag
            valid_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
            g[valid_minus] -= h[i_minus_k[valid_minus]] * dx
            weight = 1.0 - np.abs(corr[lag])
            combined_gradient += g * weight
        
        best_new_h = None
        best_new_c5 = current_c5
        for step in step_sizes:
            trial_h = h - step * combined_gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_new_h = trial_h.copy()
        
        if best_new_c5 < current_c5:
            h = best_new_h
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
        
        if iteration % 100 == 0:
            top_lags_percent = min(top_lags_percent + 0.05, 0.5)
            step_sizes = [0.01, 0.05]
        
        iteration += 1
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
