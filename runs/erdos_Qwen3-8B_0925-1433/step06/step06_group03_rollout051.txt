import numpy as np
import time
from scipy.optimize import Bounds, LinearConstraint, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h_sum = np.sum(h)
    h *= target_sum / h_sum
    best_h = h
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    mutation_amp = 0.05
    iter_limit = 1000
    success_count = 0
    
    for _ in range(iter_limit):
        if time.time() - start_time > budget_s:

        perturbed = best_h + np.random.normal(0, mutation_amp, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed *= target_sum / perturbed_sum
        
        def obj(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        
        try:
            res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)

                    mutation_amp *= 1.05

        except:

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = correlate(h, 1 - h, mode='full')
