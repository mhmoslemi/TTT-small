            for i in range(n_points):
                if i + lag < n_points:
                    grad[i] += (1 - h[i + lag]) * dx
                if i - lag >= 0:
                    grad[i] -= h[i - lag] * dx
            grads.append(grad)
        averaged_grad = np.mean(grads, axis=0)
        h = h - learning_rate * averaged_grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                learning_rate *= 0.8
                no_improvement = 0
    return (best_h, best_c5, n_points)
