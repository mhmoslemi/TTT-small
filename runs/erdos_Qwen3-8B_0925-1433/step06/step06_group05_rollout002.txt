            best_h = h.copy()
        
        if step % 50 == 0:
            perturb = np.random.normal(0, 0.15, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        learning_rate *= decay_rate
        step += 1
    
    return (best_h, best_c5, n_points)
