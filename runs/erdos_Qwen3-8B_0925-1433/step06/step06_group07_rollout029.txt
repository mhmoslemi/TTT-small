        grads = []
        for l in top_mask[::-1]:
            m = l - (n_points - 1)
            grad = np.zeros(n_points)
            for j in range(n_points):
                k = j + m
                if 0 <= k < n_points:
                    grad[j] = 1.0 - h[k]
            grads.append(grad)
        total_grad = np.sum(grads, axis=0)
        best_new_max = current_max
        best_trial = h.copy()
        for alpha in [0.01, 0.05, 0.1, 0.2]:
            trial_h = h - alpha * total_grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_max:
                best_new_max = new_max
                best_trial = trial_h
        if best_new_max < best_c5:
            best_c5 = best_new_max
            best_h = best_trial.copy()
            h = best_trial
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.15, 0.15, size=n_points)
            h = h * (1 + noise)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    return (best_h, best_c5, n_points)
