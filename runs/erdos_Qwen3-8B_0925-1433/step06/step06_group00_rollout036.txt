    if 'initial_h_values' in locals() or 'initial_h_values' in globals():
        s = np.array(initial_h_values, dtype=float)
        N = len(s)
        temp_s = (s > 0.5).astype(int)
        cnt = np.sum(temp_s)
        targ = N // 2
        if cnt < targ:
            zeros = np.where(temp_s == 0)[0]
            flip_indices = np.random.choice(zeros, size=targ - cnt, replace=False)
            temp_s[flip_indices] = 1
        elif cnt > targ:
            ones = np.where(temp_s == 1)[0]
            flip_indices = np.random.choice(ones, size=cnt - targ, replace=False)
            temp_s[flip_indices] = 0
        s = temp_s
    else:
        N = 100
        s = np.random.choice([0, 1], size=N)
        while np.sum(s) != N // 2:
            s = np.random.choice([0, 1], size=N)
    best_s = s.copy()
    best_c5 = compute_c5(best_s, N)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for _ in range(50):
            i, j = np.random.randint(0, N, size=2)
            if s[i] == 1 and s[j] == 0:
                new_s = s.copy()
                new_s[i], new_s[j] = 0, 1
                current_c5 = compute_c5(new_s, N)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_s = new_s
                    s = new_s
    return (best_s.astype(float), best_c5, N)

def compute_c5(s, N):
    cor = np.correlate(s, 1 - s, mode='full')
    return np.max(cor) * (2.0 / N)
