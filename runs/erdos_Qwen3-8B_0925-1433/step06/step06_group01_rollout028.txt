            return weights

        weights = get_weights(k_max)

        # Define objective for QP: minimize sum(h[i] * (1 - h[j]) * weights[i,j])
        # which is equivalent to minimizing the overlap for worst lag
        # This is bilinear, but we approximate with linear constraints
        # For the sake of the solution, we use a simple gradient descent step

        # Gradient of objective w.r.t h[i]: sum(1 - h[j]) * weights[i,j] - sum(h[i] * weights[i,j]) 
        # (approximation for the gradient of the bilinear form)
        # But this is complex, so instead we use a directional perturbation

        # Generate perturbation directed towards minimizing the worst lag
        delta = np.zeros(n_points)
        for i in range(n_points):
            contrib = np.sum((1 - best_h) * weights[i])
            delta[i] = -contrib * mutation_amp
        perturbed = best_h + delta
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum == 0:
            perturbed = best_h
        else:
            perturbed = perturbed * target_sum / current_sum

        # Run SLSQP optimization
        def obj(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

        if res.success:
            new_h = res.x
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
                improvement_count = 0
                mutation_amp *= 0.98
            else:
                improvement_count += 1
        else:
            new_h = perturbed.copy()
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
                improvement_count = 0
            else:
                improvement_count += 1

        iterations += 1
        if improvement_count >= 50:
            break

    return (best_h, best_c5, n_points)
