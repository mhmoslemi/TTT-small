            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            fitness = compute_c5(particles[i], dx, n_points)
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
        current_global_fitness = np.min(personal_best_fitness)
        if current_global_fitness < global_best_fitness:
            global_best = personal_best[np.argmin(personal_best_fitness)]
            global_best_fitness = current_global_fitness
        corr = np.correlate(global_best, 1 - global_best, mode='full')
        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        grad = corr[k_max] * np.sign(max_corr)
        global_best += alpha * grad
        global_best = np.clip(global_best, 0, 1)
        global_best = global_best * target_sum / np.sum(global_best)
        new_fitness = compute_c5(global_best, dx, n_points)
        if new_fitness < global_best_fitness:
            global_best_fitness = new_fitness
        iterations += 1
    return (global_best, global_best_fitness, n_points)
