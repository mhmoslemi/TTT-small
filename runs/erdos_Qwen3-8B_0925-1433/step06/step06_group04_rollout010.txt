            parent2 = np.random.choice(selected_parents)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:split], parent2[split:]])
            mut_child = child + np.random.normal(0, 0.05, n_points)
            mut_child = np.clip(mut_child, 0, 1)
            mut_child = mut_child * target_sum / np.sum(mut_child)
            new_population.append(mut_child)
        combined = population + new_population
        c5_new = [compute_c5(h) for h in combined]
        sorted_indices = np.argsort(c5_new)
        population = combined[sorted_indices[:len(population)]]
        current_best_idx = np.argmin(c5_new[:len(population)])
        if c5_new[current_best_idx] < best_c5:
            best_c5 = c5_new[current_best_idx]
            best_h = population[current_best_idx]
    return (best_h, best_c5, n_points)
