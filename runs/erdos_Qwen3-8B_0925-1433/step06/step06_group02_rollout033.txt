import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    for _ in range(100):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_lag = np.argmax(corr)
        worst_corr = corr[worst_lag]
        if worst_corr <= best_c5:
            break  # No improvement
        # Define objective for this lag
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[worst_lag] * dx
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            obj,
            bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-5,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed + 1
        )

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
