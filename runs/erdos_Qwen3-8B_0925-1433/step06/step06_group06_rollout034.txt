import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population = np.array([initial_h_values.copy() * target_sum / np.sum(initial_h_values)])
    for _ in range(9):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population = np.vstack((population, h))
    
    best_h = population[0].copy()
    best_c5 = np.inf
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        c5_vals = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            c5_vals.append(c5)
        
        elite_idx = np.argsort(c5_vals)[:int(0.2 * len(population))]
        elites = population[elite_idx]
        
        non_elite_idx = np.setdiff1d(np.arange(len(population)), elite_idx)
        new_pop = []
        for _ in range(len(non_elite_idx) // 2):
            a, b = np.random.choice(non_elite_idx, 2, replace=False)
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([population[a][:split], population[b][split:]])
            else:
                child = np.mean([population[a], population[b]], axis=0)
            
            corrupted = False
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    child[j] = np.random.rand()
                elif np.random.rand() < 0.05:
                    child[j] *= np.random.uniform(0.95, 1.05)
                child[j] = np.clip(child[j], 0, 1)
                if child[j] > 1 or child[j] < 0:
                    corrupted = True
            if corrupted:
                child = population[a].copy()
            new_pop.append(child)
        
        population = np.vstack((elites, np.array(new_pop)))
        c5_vals = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in population]
        best_idx = np.argmin(c5_vals)
        if c5_vals[best_idx] < best_c5:
            best_c5 = c5_vals[best_idx]
            best_h = population[best_idx].copy()
    
    return (best_h, best_c5, n_points)
