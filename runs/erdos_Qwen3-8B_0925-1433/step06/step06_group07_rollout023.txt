            grad_k = np.zeros(n_points)
            grad_k[valid_mask] = (1 - h[shifted[valid_mask]]) * dx
            total_gradient += grad_k * weight

        best_alpha = 0.0
        best_new_max = best_c5 + 1e-6

        for alpha in alpha_steps:
            trial_h = h - alpha * total_gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_max:
                best_new_max = new_max
                best_alpha = alpha

        if best_alpha > 0:
            h = h - best_alpha * total_gradient
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        if _ % perturb_interval == 0:
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * (target_sum / np.sum(h))
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()

    return (best_h, best_c5, n_points)
