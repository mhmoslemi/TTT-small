    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        epsilon = 1e-4
        vec_grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            h_perturbed = h_perturbed * target_sum / current_sum if current_sum > 0 else h.copy()
            new_c5 = compute_c5(h_perturbed, dx, n_points)
            vec_grad[i] = (new_c5 - current_c5) / epsilon
        
        learning_rate = 0.1
        h = h - learning_rate * vec_grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * target_sum / current_sum if current_sum > 0 else h
        
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * target_sum / current_sum if current_sum > 0 else h
        
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)
