import time

def compute_c5(h, dx, n_points):

    return np.max(corr) * dx

        h = np.full(n_points, 1.0 / n_points)
    else:
        h = h * target_sum / h_sum

    start_time = time.time()
    learning_rate = 0.2
    momentum = 0.8
    decay_rate = 0.999
    perturb_freq = 500
    step_count = 0
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        max_lag_idx = np.argmax(corr)
