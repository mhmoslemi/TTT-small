            split = np.random.randint(1, n_points)
            child = np.concatenate([winner[:split], parent2[split:]])
            
            # Mutation
            mutate_mask = np.random.rand(n_points) < 0.15
            child[mutate_mask] = np.random.rand(np.sum(mutate_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            new_pop.append(child)
        
        population = np.vstack([elites, np.array(new_pop)])
        
        # Update best
        for cand in population:
            corr = np.correlate(cand, 1 - cand, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = cand.copy()
    
    return (best_h, best_c5, n_points)
