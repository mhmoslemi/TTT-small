import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    K = 3

    no_improvement = 0
    start_time = time.time()
    for _ in range(max_iters):

            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        top_lags = np.argsort(corr)[-K:]
        grad = np.zeros_like(h)
        for lag in top_lags:
            for i in range(n_points):
                a = 0.0
                if i + lag < n_points:
                    a = 1.0 - h[i + lag]
                b = 0.0

                grad[i] += a - b
        grad *= dx

        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < current_c5:
            h = h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
