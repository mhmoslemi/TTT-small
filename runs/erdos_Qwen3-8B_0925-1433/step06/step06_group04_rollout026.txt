import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    np.random.seed(seed)
    population = []
    for _ in range(100):
        if np.random.rand() < 0.2:  # 20% random
            h = np.random.uniform(0, 1, n_points)
        elif np.random.rand() < 0.5:  # 50% from initial
            h = np.copy(initial_h_values)
        else:  # 30% mutated from best
            h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        population.append(h)
    population = np.array(population)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    for _ in range(1500):
        if time.time() - start_time > budget_s - 1.5:
            break
        # Evaluate current population
        c5_vals = np.array([obj(h) for h in population])
        idxs = np.argsort(c5_vals)
        top_idx = idxs[0]
        current_best = population[top_idx]
        current_c5 = c5_vals[top_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(current_best)

        # Select top 10% for breeding
        selected = population[idxs[:int(len(population)*0.1)]]
        new_population = []

        # Crossover and mutation
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            # Uniform crossover
            if np.random.rand() < 0.8:
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            else:
                # Single-point crossover
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            # Mutation
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child) if np.sum(child) > 0 else np.zeros(n_points)
            new_population.append(child)

        # Replace worst 30% with new candidates
        new_population += [population[i] for i in idxs[int(len(population)*0.7):]]
        population = np.array(new_population)

    # Final evaluation
    c5_vals = np.array([obj(h) for h in population])
    best_idx = np.argmin(c5_vals)
    final_h = population[best_idx]
    final_c5 = c5_vals[best_idx]
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h

    return (best_h, best_c5, n_points)
