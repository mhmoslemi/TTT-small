import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 30
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        rand_vals = np.random.uniform(0.4, 0.6, n_points)
        rand_vals = rand_vals * target_sum / np.sum(rand_vals)
        population.append(rand_vals)
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx
    best_h = initial_h_values.copy()
    F = 0.8
    CR = 0.9
    max_iter = 300
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            h_a, h_b, h_c = population[a], population[b], population[c]
            mutant = h_a + F * (h_b - h_c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(corr) * dx
            original_c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx
            if trial_c5 < original_c5:
                population[i] = trial.copy()
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
