import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    initial_sum = np.sum(initial_h)
    initial_scaled = initial_h * target_sum / initial_sum
    initial_clip = np.clip(initial_scaled, 0, 1)
    clip_sum = np.sum(initial_clip)
    if clip_sum > 0:
        initial_clip = initial_clip * target_sum / clip_sum
    initial_h = initial_clip

    # Initialize population
    population = [initial_h.copy()]
    velocities = [np.random.uniform(-0.03, 0.03, n_points)]
    for _ in range(19):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)
        velocities.append(np.random.uniform(-0.03, 0.03, n_points))

    # Track best solution
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h, dx, n_points)

    start_time = time.time()
    for step in range(500):
        if time.time() - start_time > budget_s:

        current_c5 = []
        for i in range(len(population)):

            current_c5.append(compute_c5(h, dx, n_points))
        # Update best
        new_best = np.argmin(current_c5)
        if current_c5[new_best] < best_c5:

        # Update velocities and positions
