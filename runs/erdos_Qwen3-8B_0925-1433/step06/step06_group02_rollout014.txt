import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
        k = np.argmax(corr) - n_points + 1
        def obj(x):
            shifted = np.roll(x, k)
            return np.sum(x * (1 - shifted))
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:

            if new_c5 < best_c5:

        if np.random.rand() < 0.1:

            s = np.sum(h)

            c = compute_c5(h, dx, n_points)
