import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        x_norm = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    def evaluate(x):
        return obj(x)

    # Initial simplex
    simplex = [h + np.random.normal(0, 0.05, n_points) for _ in range(5)]
    for i in range(len(simplex)):
        simplex[i] = np.clip(simplex[i], 0, 1)
        simplex[i] = simplex[i] * target_sum / np.sum(simplex[i])

    res = minimize(evaluate, simplex[0], method='Nelder-Mead', initial_simplex=simplex,
                   options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 300})
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    # Final evaluation
    final_h = best_h
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = compute_c5(final_h, dx, n_points)

    return (final_h, final_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
