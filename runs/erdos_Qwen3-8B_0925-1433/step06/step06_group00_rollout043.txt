import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.random.rand(45)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = LinearConstraint(np.ones(n_points_high), [n_points_high / 2], [n_points_high / 2])
    bounds = Bounds(0, 1)

    best_h = h_high.copy()
    best_c5 = objective(best_h)

    result = minimize(objective, best_h, method='COBYLA', bounds=bounds, constraints=cons)
    if result.success:

        curr_c5 = objective(curr_h)
        if curr_c5 < best_c5:
            best_h = curr_h
            best_c5 = curr_c5
