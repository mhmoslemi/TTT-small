import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    max_iterations = int(budget_s * 1.5)
    decay = 0.9999
    learning_rate = 0.2
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[-5:]
        avg_grad = np.zeros_like(h)
        for lag in top_lags:
            term1 = np.zeros_like(h)
            valid_term1 = np.arange(n_points) + lag < n_points
            term1[valid_term1] = 1 - h[valid_term1]
            term2 = np.zeros_like(h)
            valid_term2 = np.arange(n_points) - lag >= 0
            term2[valid_term2] = h[valid_term2]
            avg_grad += (term1 - term2) * dx
        avg_grad /= len(top_lags)
        h = h - learning_rate * avg_grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay
    return (best_h, best_c5, n_points)
