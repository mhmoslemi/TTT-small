import numpy as np
import time
import cvxpy as cp

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        m = max_idx
        A = np.zeros((n_points, n_points))
        valid = np.arange(0, n_points - m)
        for i in valid:
            A[i, i + m] = 1.0
        
        h = cp.Variable(n_points)
        objective = cp.Maximize(cp.quad_form(h, A))
        constraints = [cp.sum(h) == target_sum, 0 <= h, h <= 1]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                new_h = h.value
                new_h = np.clip(new_h, 0, 1)
                if np.sum(new_h) == 0:
                    continue
                new_h = new_h * target_sum / np.sum(new_h)
                new_c5 = compute_c5(new_h, dx, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        except:
            pass
    return (best_h, best_c5, n_points)
