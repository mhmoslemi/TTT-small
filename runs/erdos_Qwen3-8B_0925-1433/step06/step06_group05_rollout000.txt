import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    initial_clip = np.clip(initial_scaled, 0, 1)
    clip_sum = np.sum(initial_clip)
    if clip_sum > 0:
        initial_clip = initial_clip * target_sum / clip_sum
    h = initial_clip.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    momentum = 0.8
    velocity = np.zeros(n_points)
    learning_rate = 0.1
    best_improvement_steps = 0
    max_steps = int(budget_s / 0.1)
    
    for _ in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_improvement_steps = 0
        else:
            best_improvement_steps += 1
        if best_improvement_steps > 100:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            best_improvement_steps = 0
        learning_rate *= 0.99
    return (best_h, best_c5, n_points)
