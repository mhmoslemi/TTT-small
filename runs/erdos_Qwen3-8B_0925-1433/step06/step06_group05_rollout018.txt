import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    if corr.size == 0:
        return 0.0
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    initial_sum = np.sum(h)
    if initial_sum == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / initial_sum
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    velocity = np.zeros(n_points)
    learning_rate = 0.6
    momentum = 0.9
    perturb_freq = 200
    perturb_amp = 0.3
    steps = 0
    while time.time() - start_time < budget_s:
        steps += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / h_sum
        if steps % perturb_freq == 0:
            noise = np.random.normal(0, perturb_amp, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)
