import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    mutation_amp = 0.03
    iterations = 0
    max_iterations = 500
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        target_c5 = corr[k_max] * dx
        noise = np.random.normal(0, mutation_amp, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum == 0:
            perturbed = best_h
        else:
            perturbed = perturbed * target_sum / current_sum
        def obj(x):

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)

        if res.success:

                elif best_c5 > 0.385:
                    mutation_amp *= 1.05
        iterations += 1
    return (best_h, best_c5, n_points)

def corr_at_k_max(x, k_max, dx):
    corr = np.correlate(x, 1 - x, mode='full')
    return corr[k_max] * dx
