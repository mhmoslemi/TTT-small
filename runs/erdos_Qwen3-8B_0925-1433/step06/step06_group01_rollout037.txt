    max_iterations = 200
    iteration = 0
    while iteration < max_iterations and time.time() - start_time < budget_s:

        max_corr = np.max(corr)

        # Estimate lag in domain units
        lag_index = max_index - n_points + 1
        lag_k = lag_index * dx
        
        # Compute overlap weights for this lag
        weights = np.zeros((n_points, n_points))

            x_i = i * dx
            for j in range(n_points):
