    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Prepare parent
    h_parent = np.clip(initial_h_values, 0, 1)
    if np.sum(h_parent) > 0:
        h_parent = h_parent * target_sum / np.sum(h_parent)
    else:
        h_parent = np.zeros(n_points)

    # Objective function
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    population_size = 50
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.1:  # 10% random
            h = np.random.uniform(0, 1, n_points)
        elif np.random.rand() < 0.3:  # 30% perturb parent
            h = np.clip(h_parent + np.random.normal(0, 0.1, n_points), 0, 1)
        else:  # 60% clones with small changes
            h = np.copy(h_parent) + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        # Rescale
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        population.append(h)
    population = np.array(population)

    best_h = h_parent.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            trial = np.random.rand(n_points) < 0.7
            trial = np.where(trial, mutant, population[i])
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = h_parent.copy()
            else:
                trial = trial * target_sum / trial_sum
            new_population.append(trial)
        new_population = np.array(new_population)
        combined = np.vstack((population, new_population))
        fitness = np.array([obj(h) for h in combined])
        elite_indices = np.argsort(fitness)[:population_size]
        population = combined[elite_indices]
        current_best = population[np.argmin(fitness)]
        current_best_c5 = obj(current_best)
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        # Early stopping if no improvement for 100 iterations
        if iterations > 100 and current_best_c5 >= best_c5:
            break
    return (best_h, best_c5, n_points)
