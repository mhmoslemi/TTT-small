import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):

    global initial_h_values

        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    momentum = 0.8
    learning_rate = 0.1
    decay_rate = 0.99
    velocity = np.zeros(n_points)
    no_improvement = 0
    step = 0

    while time.time() - start_time < budget_s:

        if current_c5 < best_c5:
