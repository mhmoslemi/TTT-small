import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    scaled = initial_h_values * target_sum / initial_sum
    best_h = scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h.copy()
        indices = np.random.choice(n_points, 10, replace=False)
        perturbed[indices] += np.random.normal(0, 0.02, 10)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        current_c5 = compute_c5(perturbed, dx, n_points)
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = perturbed.copy()
            best_c5 = current_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
