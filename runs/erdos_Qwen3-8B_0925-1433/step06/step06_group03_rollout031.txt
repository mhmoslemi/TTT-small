    h_initial = np.zeros(n_points)
    h_initial[ones_indices] = 1
    h_initial = h_initial * target_sum / np.sum(h_initial)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    mutation_amplitude = 0.02
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Perturb best_h
        perturbed = best_h + np.random.normal(0, mutation_amplitude, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        # Define objective
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        # Constraints
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        # Optimize
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                if best_c5 < 0.381:
                    mutation_amplitude *= 0.95
                elif best_c5 > 0.382:
                    mutation_amplitude *= 1.05

    return (best_h, best_c5, n_points)
