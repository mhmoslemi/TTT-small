    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    mutation_amp = 0.05
    iterations = 0
    max_iterations = 500
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_c5_idx = np.argmax(corr)
        max_c5_val = corr[max_c5_idx]
        
        if max_c5_val <= best_c5:
            break
        
        # Focus on reducing the max lag
        if max_c5_idx >= n_points:
            shift = max_c5_idx - n_points + 1
            shifted_h = np.roll(best_h, shift)
        else:
            shifted_h = np.roll(best_h, max_c5_idx)
        
        # Create the objective for this lag
        objective = cp.Minimize(cp.sum(cp.multiply(best_h, 1 - shifted_h)) * dx)
        constraints = [
            cp.sum(best_h) == target_sum,
            cp.Minimize(cp.sum(best_h)) >= 0,
            cp.Maximize(cp.sum(best_h)) <= n_points
        ]
        problem = cp.Problem(objective, constraints)
        try:
            result = problem.solve(solver=cp.ECOS, eps=1e-8)
            if result is not None:
                new_h = np.array(cp.reshape(best_h, (n_points,)))
                new_h = np.clip(new_h, 0, 1)
                new_sum = np.sum(new_h)
                if new_sum == 0:
                    new_h = best_h
                else:
                    new_h = new_h * target_sum / new_sum
                new_c5 = compute_c5(new_h, dx, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
                    mutation_amp *= 0.95
                else:
                    mutation_amp *= 1.05
        except:
            # Fallback to random mutation
            noise = np.random.normal(0, mutation_amp, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum == 0:
                perturbed = best_h
            else:
                perturbed = perturbed * target_sum / perturbed_sum
            new_c5 = compute_c5(perturbed, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = perturbed
            mutation_amp *= 1.05
        
        iterations += 1
    
    return (best_h, best_c5, n_points)
