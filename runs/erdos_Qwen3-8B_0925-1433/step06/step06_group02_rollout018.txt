import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    
    population_size = 50
    population = []
    for _ in range(population_size):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = initial_scaled + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        population.append(perturbed)
    
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(h) for h in population]
        worst_idx = np.argmax(c5_values)
        worst_h = population[worst_idx]
        
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(objective, worst_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        population[worst_idx] = res.x
        
        new_c5 = compute_c5(res.x)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = res.x.copy()
        
        if np.random.rand() < 0.1:
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(best_h + perturbation, 0, 1)
            perturbed_sum = np.sum(perturbed)
            perturbed = perturbed * target_sum / perturbed_sum
            population.append(perturbed)
            population.pop(0)
    
    return (best_h, best_c5, n_points)
