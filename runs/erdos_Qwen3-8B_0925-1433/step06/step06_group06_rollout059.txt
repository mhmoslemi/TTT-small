import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    # Initialize population
    pop_size = 50
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array([h * target_sum / np.sum(h) for h in population])
    population = np.clip(population, 0, 1)
    best_h = population[0]
    best_c5 = np.inf
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.zeros(pop_size)
        for i, h in enumerate(population):
            corr = np.correlate(h, 1 - h, mode='full')
            fitness[i] = np.max(corr) * dx
        # Select elites
        elite_idx = np.argsort(fitness)[:int(0.2 * pop_size)]
        elites = population[elite_idx]
        # Generate new population
        new_pop = np.zeros((pop_size, n_points))
        new_pop[:len(elites)] = elites
        for j in range(len(elites), pop_size):
            # Tournament selection
            tournament = np.random.choice(population, size=5)
            parents = tournament[np.argmin(np.array([np.max(np.correlate(t, 1 - t, mode='full')) * dx for t in tournament]))]
            p1, p2 = parents[0], parents[1]
            # Crossover
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate((p1[:split], p2[split:]))
            # Mutation
            mask = np.random.rand(n_points) < 0.1
            child[mask] = np.random.rand(sum(mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop[j] = child
        # Replace worst half
        new_pop[0:int(0.5 * pop_size)] = elites
        population = new_pop
        # Update best
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]
    return (best_h, best_c5, n_points)
