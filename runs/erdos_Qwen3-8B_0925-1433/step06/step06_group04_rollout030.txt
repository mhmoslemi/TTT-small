from scipy.optimize import minimize

    n_points = len(initial_h_values)

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return h
        return h * target_sum / np.sum(h)

    population = []

            h = rescale_h(h)
        else:
            x = np.linspace(0, 2, n_points)
            h = 0.5 * (1 + np.sign(np.sin(np.pi * x)))
            h = rescale_h(h)
        population.append(h)

    best_h = rescale_h(np.copy(initial_h_values))
