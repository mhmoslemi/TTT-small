import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Symmetrize initial guess
    h = initial_h_values.copy()
    if n_points % 2 == 0:
        h = np.concatenate([h[:n_points//2], h[::-1][:n_points//2]])
    else:
        h = np.concatenate([h[:n_points//2], [h[n_points//2]], h[:n_points//2][::-1]])
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    for _ in range(200):  # Run 200 iterations
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturbed = best_h + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        res = minimize(lambda x: np.max(np.correlate(x, 1-x, mode='full')) * dx, 
                       perturbed, method='SLSQP', bounds=Bounds(0,1), constraints=cons)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
