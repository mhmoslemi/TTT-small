import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    
    # Improve initial guess with a smooth sinusoidal spread
    spread = 1.0 + np.sin(np.linspace(np.pi/2, 3*np.pi/2, n_points))
    initial_scaled = np.clip(initial_scaled * spread / np.max(spread), 0, 1)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    mutation_amp = 0.1
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current best
        perturbed = best_h * (1 - mutation_amp) + np.random.uniform(0, mutation_amp, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        
        # Define objective with max correlation
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        # Constraints
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)

        if res.success:
