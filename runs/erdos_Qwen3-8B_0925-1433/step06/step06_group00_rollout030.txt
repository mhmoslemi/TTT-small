import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy() * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx, n_points)
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    max_iterations = 5000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        perturbed = h.copy()
        idxs = np.random.choice(n_points, size=10, replace=False)
        perturbed[idxs] += np.random.normal(0, 0.02, 10)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed = perturbed * target_sum / sum_perturbed
        else:
            perturbed = np.random.rand(n_points) * 1.0
            perturbed = perturbed * target_sum / np.sum(perturbed)
        new_c5 = evaluate_erdos_solution(perturbed, dx, n_points)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = perturbed.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)
