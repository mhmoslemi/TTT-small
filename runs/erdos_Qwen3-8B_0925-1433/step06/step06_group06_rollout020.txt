import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    # Initialize population with initial_h_values and random candidates
    pop_size = 30
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        rand_h = np.random.uniform(0.4, 0.6, n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)
    
    best_h = population[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Evaluate current population
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        # Select top 1% for elitism
        elite_idx = np.argsort(fitness)[:int(np.ceil(pop_size * 0.01))]
        elite = [population[i] for i in elite_idx]
        
        # Generate new candidates
        new_population = elite.copy()
        for _ in range(pop_size - len(elite)):
            # Select 3 random parents
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            h_a = population[a]
            h_b = population[b]
            h_c = population[c]
            # Mutation
            mutant = h_a + 0.8 * (h_b - h_c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            # Crossover
            h_trial = mutant.copy()
            for i in range(n_points):
                if np.random.rand() < 0.9:
                    h_trial[i] = mutant[i]
            # Ensure sum constraint
            h_trial = h_trial * target_sum / np.sum(h_trial)
            new_population.append(h_trial)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            new_fitness.append(c5)
        # Select best from new population
        combined = list(zip(new_population, new_fitness))
        combined.sort(key=lambda x: x[1])
        new_population = [pair[0] for pair in combined[:pop_size]]
        
        # Update best solution
        current_best = new_population[0]
        current_c5 = new_fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
    
    return (best_h, best_c5, n_points)
