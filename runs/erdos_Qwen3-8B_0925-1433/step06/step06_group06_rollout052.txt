import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    # Initialize population with initial guess and 29 random candidates
    population = [initial_h_values.copy()]
    for _ in range(29):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    F, CR, elites = 0.8, 0.9, 5
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_population = []
        for idx in range(len(population)):
            a_idx = idx
            b_idx = np.random.randint(len(population))
            c_idx = np.random.randint(len(population))
            while b_idx == a_idx or c_idx == a_idx or b_idx == c_idx:
                b_idx = np.random.randint(len(population))
                c_idx = np.random.randint(len(population))
            a = population[a_idx]
            b = population[b_idx]
            c = population[c_idx]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            trial = a.copy()
            mask = np.random.rand(n_points) < CR
            trial[mask] = mutant[mask]
            c5_trial = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
            c5_orig = np.max(np.correlate(a, 1 - a, mode='full')) * dx
            if c5_trial < c5_orig:
                new_population.append(trial)
            else:
                new_population.append(a)
        # Elitism: retain top elites from previous population
        new_population += [x for x in population if np.max(np.correlate(x, 1 - x, mode='full')) * dx <= best_c5][:elites]
        population = new_population
        # Update best
        for h in population:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)
