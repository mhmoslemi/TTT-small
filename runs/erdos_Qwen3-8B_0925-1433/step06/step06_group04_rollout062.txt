            res = minimize(local_obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                population[i] = res.x

        current_best = sorted_scores[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = sorted_pop[0]

    return (best_h, best_c5, n_points)
