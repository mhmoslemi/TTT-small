import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        return np.clip(h * target_sum / s, 0, 1) if s > 0 else np.zeros_like(h)

    def mutate(h, std=0.05):
        return normalize(h + np.random.normal(0, std, n_points))

    def crossover(a, b):
        if np.random.rand() < 0.7:
            return np.random.permutation(np.concatenate([a, b]))[:n_points]
        idx = np.random.randint(1, n_points-1)
        return np.concatenate([a[:idx], b[idx]])

    initial_pop = [np.array(initial_h_values)]
    for _ in range(49):
        initial_pop.append(np.random.uniform(0, 1, n_points))
        initial_pop.append(mutate(initial_h_values))
    initial_pop = [normalize(h) for h in initial_pop]
    pop = initial_pop[:100]
    best_h = min(pop, key=eval_c5)
    best_c5 = eval_c5(best_h)
    start_time = time.time()

    for _ in range(1000):
