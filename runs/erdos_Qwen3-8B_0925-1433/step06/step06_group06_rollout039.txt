import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    for _ in range(100000):
        if time.time() - start_time > budget_s:

        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.2, 0.2)
        new_h = h.copy()
        new_h[i] += delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:

            best_h = new_h.copy()
