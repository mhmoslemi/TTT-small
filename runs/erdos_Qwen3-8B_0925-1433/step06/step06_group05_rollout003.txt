import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    momentum = 0.8
    velocity = np.zeros(n_points)
    learning_rate = 0.1
    for step in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if step % 100 == 0:
            perturb = np.random.normal(0, 0.15, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
