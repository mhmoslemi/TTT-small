        res_p = minimize(obj_high, perturbed_scaled, method='COBYLA', bounds=bounds, constraints=cons)

            curr_c5 = res_p.fun
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res_p.x

    return (best_h, best_c5, n_points_high)
