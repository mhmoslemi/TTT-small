        current_max = np.max(corr) * dx
        
        top_lags = np.argsort(np.abs(corr))[-10:]
        top_lags = top_lags[::-1]
        total_grad = np.zeros(n_points)
        
        for lag_idx in top_lags:
            lag = lag_idx - (n_points - 1)
            grad_k = np.zeros(n_points)
            for x in range(n_points):
                shifted_x = x + lag
                if 0 <= shifted_x < n_points:
                    grad_k[x] = (1 - h[shifted_x]) * dx
            weight = np.exp(-np.abs(lag) / sigma)
            total_grad += grad_k * weight
        
        best_new_c5 = current_max
        best_new_h = h.copy()
        for alpha in line_search_alphas:
            new_h = h - alpha * total_grad
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new_h = np.sum(new_h)
            if sum_new_h == 0:
                continue
            new_h = new_h * (target_sum / sum_new_h)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_c5:
                best_new_c5 = new_max
                best_new_h = new_h.copy()
        
        if best_new_c5 < current_max:
            h = best_new_h
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
        
        if iterations % perturb_freq == 0:
            noise = np.random.normal(0, 0.1, n_points)
            perturbed_h = h + noise
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            sum_perturbed = np.sum(perturbed_h)
            if sum_perturbed != 0:
                perturbed_h = perturbed_h * (target_sum / sum_perturbed)
            h = perturbed_h
        
        if iterations < 100:
            sigma = 0.5 * (n_points / 2)
        elif iterations < 200:
            sigma = 0.25 * (n_points / 2)
        else:
            sigma = 0.1 * (n_points / 2)
        
        if iterations >= max_iter:
            break
    
    return (best_h, best_c5, n_points)
