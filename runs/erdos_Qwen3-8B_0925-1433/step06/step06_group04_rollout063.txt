import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s

    def eval_c5(h):
        try:
            return np.max(np.correlate(h, 1 - h, mode='full')) * dx
        except:
            return np.inf

    def mutate(h, F, sigma):
        h = np.clip(h, 0, 1)
        h = rescale(h)
        m = np.random.choice(n_points, 2, replace=False)
        h1, h2 = h[m[0]], h[m[1]]
        h_m = h + F * (h2 - h1) + np.random.normal(0, sigma, n_points)
        h_m = np.clip(h_m, 0, 1)
        return rescale(h_m)

    def crossover(h1, h2, p):
        return np.where(np.random.rand(n_points) < p, h1, h2)

    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = rescale(h_initial)

    pop_size = 100
    pop = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(30)]
    for _ in range(30):
        pop.append(np.clip(0.5 * (1 + np.sign(np.sin(np.linspace(0, 2, n_points) * np.pi))), 0, 1))
    pop = [rescale(p) for p in pop]
    pop = [np.clip(p, 0, 1) for p in pop]
    pop = list(set(pop))[:pop_size]

    best_h = h_initial
    best_c5 = eval_c5(best_h)
    start_time = time.time()

    for _ in range(1500):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([eval_c5(p) for p in pop])
        idxs = np.argsort(fitness)
        selected = [pop[i] for i in idxs[:int(0.2 * len(pop))]]
        new_pop = []
        for _ in range(int(0.7 * len(pop))):
            a, b = np.random.choice(selected, 2, replace=False)
            h_m = mutate(a, np.random.uniform(0.5, 1.0), 0.05)
            h_c = crossover(h_m, b, 0.8)
            new_pop.append(h_c)
        new_pop += [np.random.uniform(0, 1, n_points) for _ in range(int(0.3 * len(pop)))]
        pop = [rescale(p) for p in new_pop]
        pop = [np.clip(p, 0, 1) for p in pop]
        pop = list(set(pop))[:pop_size]
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_h = pop[idx]
            best_c5 = fitness[idx]

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        res = minimize(lambda x: eval_c5(x), best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':lambda x: np.sum(x) - target_sum}])
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
