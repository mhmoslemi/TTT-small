            child = np.concatenate([p1[:split], p2[split:]])
            child = mutate(child)
            new_population.append(child)
        population = new_population
        for c5, h in evaluated:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)
