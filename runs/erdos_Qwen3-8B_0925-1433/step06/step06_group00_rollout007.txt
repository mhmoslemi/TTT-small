import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = initial_h_values.copy()
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=15, replace=False)
        perturbed = current_h.copy()
        perturbed[indices] += np.random.normal(0, 0.025, 15)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        new_c5 = compute_c5(perturbed, dx, n_points)
        if new_c5 < best_c5:
            current_h = perturbed
            best_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - best_c5) / T)
            if np.random.rand() < prob:
                current_h = perturbed
        if best_c5 < compute_c5(best_h, dx, n_points):
            best_h = current_h.copy()
        T *= cooling_rate
    return (best_h, best_c5, n_points)
