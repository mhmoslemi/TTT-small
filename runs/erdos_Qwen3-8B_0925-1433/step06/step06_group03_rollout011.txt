import time
from scipy.optimize import Bounds, LinearConstraint, minimize

    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    mutation_amp = 0.05
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, mutation_amp, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        def obj(x):

            return np.max(corr) * dx
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(lambda x: -obj(x), perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
