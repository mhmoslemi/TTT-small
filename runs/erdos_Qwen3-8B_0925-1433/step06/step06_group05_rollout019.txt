import numpy as np
import time
import scipy.optimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)

    initial_sum = np.sum(initial_h_values)
    initial_scaled = np.clip(initial_h_values * target_sum / initial_sum, 0, 1)
    h = initial_scaled * target_sum / np.sum(initial_scaled)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    iterations = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        if last_improvement > 100:
            perturb = np.random.normal(0, 0.2, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            last_improvement = 0
            continue
        
        max_lag = np.argmax(corr)
        k = max_lag - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        
        learning_rate = max(0.01, 0.5 * (0.995 ** iterations))
        momentum = 0.9

        h += velocity
        h = np.clip(h, 0, 1)
