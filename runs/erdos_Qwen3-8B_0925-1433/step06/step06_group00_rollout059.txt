import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.random.rand(100)
    n_points_initial = len(h_initial)
    target_sum_initial = n_points_initial / 2
    h_initial = h_initial * target_sum_initial / np.sum(h_initial)
    
    n_points_low = 100
    n_points_high = 200
    target_high_sum = n_points_high / 2
    
    def evaluate_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def obj(h):
        dx = 2.0 / len(h)
        return evaluate_c5(h, dx)
    
    low_candidates = [np.random.rand(n_points_low) * (n_points_low / 2) / np.sum(np.random.rand(n_points_low)) for _ in range(10)]
    best_low = min(low_candidates, key=lambda x: evaluate_c5(x, 2.0 / n_points_low))
    
    high_init = np.repeat(best_low, 2)
    high_init = high_init * (target_high_sum / np.sum(high_init))
    
    bounds = [(0.0, 1.0) for _ in range(n_points_high)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_high_sum}
    
    h_high = high_init.copy()
    best_c5 = obj(h_high)
    best_h = h_high.copy()
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 10:
        res = minimize(obj, h_high, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_current = res.x
            curr_c5 = obj(h_current)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_current.copy()
            perturb = np.random.normal(0, 0.05, n_points_high)
            perturbed = np.clip(h_current + perturb, 0, 1)
            perturbed = perturbed * (target_high_sum / np.sum(perturbed))
            res_p = minimize(obj, perturbed, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res_p.success:
                h_p = res_p.x
                curr_c5_p = obj(h_p)
                if curr_c5_p < best_c5:
                    best_c5 = curr_c5_p
                    best_h = h_p.copy()
    
    return (best_h, best_c5, n_points_high)
