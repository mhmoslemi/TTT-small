def run(seed=42, budget_s=600):
    np.random.seed(seed)

    best_h = h_initial.copy()

    start_time = time.time()

    for _ in range(500):

            break

        perturbed = best_h + np.random.normal(0, mu, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        # Define objective
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        # Optimize
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
