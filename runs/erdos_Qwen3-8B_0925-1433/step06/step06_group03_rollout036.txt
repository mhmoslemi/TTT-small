        perturbed = perturbed * target_sum / perturbed_sum
        # Define objective
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        # Constraints
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        # Optimize
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
            if best_c5 < 0.381:
                mutation_amplitude *= 0.95
            elif best_c5 > 0.382:
                mutation_amplitude *= 1.05
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
