    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        lag_max_idx = np.argmax(corr)
        def objective(x):
            h_candidate = x
            corr_current = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            return corr_current[lag_max_idx]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-5,
            maxiter=50,
            polish=True,
            disp=False
        )
        h_candidate = result.x
        clipped_h = np.clip(h_candidate, 0, 1)
        sum_clipped = np.sum(clipped_h)
        if sum_clipped != 0:
            scaling_factor = target_sum / sum_clipped
            new_h = clipped_h * scaling_factor
        else:
            new_h = np.zeros(n_points)
        new_c5 = compute_c5(new_h, 2.0 / n_points, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
