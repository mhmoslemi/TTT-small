        best_h = structured_init(n_points)
    else:
        best_h = (initial_scaled + structured_init(n_points)) / 2
    
    # Define objective
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    # Constraints
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    
    # Adaptive optimization loop
    mutation_amplitude = 0.03
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Perturb current best
        perturbed = best_h + np.random.normal(0, mutation_amplitude, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        
        # Optimize with SLSQP
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
            # Dynamic mutation control
            if best_c5 < 0.381:
                mutation_amplitude *= 0.96
            elif best_c5 > 0.382:
                mutation_amplitude *= 1.04
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
