import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint, Bounds
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    population = [normalize(initial_h_values)]
    for _ in range(49):
        noise = np.random.normal(0, 0.1, n_points)
        h = normalize(initial_h_values + noise)
        population.append(h)
    
    best_h = population[0]
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    # Main loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates
        c5_values = [compute_c5(h, dx, n_points) for h in population]
        worst_idx = np.argmax(c5_values)
        worst_h = population[worst_idx]
        
        # Find worst lag
        corr = np.correlate(worst_h, 1 - worst_h, mode='full')
        idx = np.argmax(corr)
        worst_k = idx - (n_points - 1)
        
        # Define objective for this lag
        def objective(x):
            indices = (np.arange(n_points) + worst_k) % n_points
            return np.sum(x * (1 - x[indices])) * dx
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(objective, worst_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        optimized_h = res.x
        
        # Update population
        population[worst_idx] = optimized_h
        
        # Update best solution
        current_c5 = compute_c5(optimized_h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
        
        # Inject random variants
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = normalize(initial_h_values + noise)
            population.append(h)
            population.pop(0)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
