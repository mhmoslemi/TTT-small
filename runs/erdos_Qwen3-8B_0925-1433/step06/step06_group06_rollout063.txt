import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    T = 1000.0
    cooling = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        idx = np.random.randint(n_points)
        delta = np.random.uniform(-0.2, 0.2)
        new_h = h.copy()
        new_h[idx] += delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            prob = np.exp(- (new_c5 - best_c5) / T)
            if np.random.rand() < prob:
                h = new_h.copy()
                best_c5 = new_c5
                best_h = h.copy()
        
        T *= cooling
    
    return (best_h, best_c5, n_points)
