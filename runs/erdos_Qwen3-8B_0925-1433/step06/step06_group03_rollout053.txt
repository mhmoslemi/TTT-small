import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()

    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        # Perturb with adaptive noise
        perturbed = best_h + np.random.normal(0, mutation_amplitude, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        # Objective function
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        # Constraints
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        # Optimize
        try:
            res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:

                    best_h = res.x

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
