import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    else:
        n_points = 45  # Fallback for testing
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Preprocess initial_h_values if available
    if 'initial_h_values' in globals():
        initial_scaled = initial_h_values.copy()
    else:
        initial_scaled = np.ones(n_points) * (target_sum / n_points)
    initial_scaled = np.clip(initial_scaled, 0, 1)
    initial_scaled_sum = np.sum(initial_scaled)
    if initial_scaled_sum != 0:
        initial_scaled = initial_scaled * target_sum / initial_scaled_sum
    else:
        initial_scaled = np.ones(n_points) * (target_sum / n_points)

    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    # DE parameters
    population_size = 30
    mutation_factor = 0.9
    crossover_prob = 0.8
    max_iterations = 300

    # Initialize population
    population = [initial_scaled.copy()]
    for _ in range(population_size - 1):
        rand_h = np.random.rand(n_points)
        rand_h = np.clip(rand_h, 0, 1)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        current_c5s = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(current_c5s)
        best_c5 = current_c5s[best_idx]

        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[best_idx].copy()
            mutant += mutation_factor * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_prob:

            trial = rescale(trial)

        population = new_population
        if best_c5 < evaluate(best_h):

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
