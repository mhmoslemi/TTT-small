import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    initial_clip = np.clip(initial_scaled, 0, 1)
    clip_sum = np.sum(initial_clip)
    if clip_sum > 0:
        initial_clip = initial_clip * target_sum / clip_sum
    h = initial_clip.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    momentum = 0.8
    velocity = np.zeros(n_points)
    learning_rate = 0.1
    steps = 0
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = i + k
            if 0 <= j1 < n_points:
                term1 = (1 - h[j1]) * dx
            else:
                term1 = 0.0
            j2 = i - k
            if 0 <= j2 < n_points:
                term2 = -h[j2] * dx
            else:
                term2 = 0.0
            grad[i] = term1 + term2
        velocity = momentum * velocity + grad
        h -= learning_rate * velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        steps += 1
        if steps % 100 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
