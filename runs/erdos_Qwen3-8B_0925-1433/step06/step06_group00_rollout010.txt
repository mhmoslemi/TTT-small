import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_low = 100
    n_high = 200
    dx = 2.0 / n_high
    target_sum = n_high / 2
    if 'initial_h_values' in locals():
        h_low = np.stack([initial_h_values] * 2)[:n_low]
        h_low = h_low * target_sum / np.sum(h_low)
        h_high = np.repeat(h_low, 2)
    else:
        h_high = np.random.rand(n_high)
        h_high = h_high * target_sum / np.sum(h_high)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    cons = LinearConstraint(np.ones(n_high), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    best_h = h_high.copy()
    best_c5 = compute_c5(best_h, dx, n_high)
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 30:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_high)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h
        np.random.seed(int(time.time()))
        pert = np.random.normal(0, 0.05, n_high)
        h_perturb = np.clip(best_h + pert, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res_p = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons)
        if res_p.success:
            curr_h_p = res_p.x
            curr_c5_p = compute_c5(curr_h_p, dx, n_high)
            if curr_c5_p < best_c5:
                best_c5 = curr_c5_p
                best_h = curr_h_p
    
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), target_sum)
    return (best_h, compute_c5(best_h, dx, n_high), n_high)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
