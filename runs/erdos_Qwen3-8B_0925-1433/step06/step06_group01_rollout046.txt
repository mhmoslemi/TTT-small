    particles = [initial_scaled.copy() for _ in range(50)]
    for i in range(50):
        noise = np.random.normal(0, 0.1, n_points)
        h = initial_scaled + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles[i] = h
    best_h = np.min(particles, axis=0)

    start_time = time.time()
