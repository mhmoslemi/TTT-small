import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in locals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(45)
        h = h * (45.0/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * (45.0/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    T = 1000.0
    cooling_ratio = 0.995
    max_iterations = 10000
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        new_h = h.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        new_h[i] += delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5)/T):
            h, current_c5 = new_h, new_c5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        T *= cooling_ratio
    return (best_h, best_c5, n_points)
