    trial_sum = np.sum(trial_h)
    if trial_sum != 0:
        trial_h = trial_h * (target_sum / trial_sum)
        new_corr = np.correlate(trial_h, 1 - trial_h, mode='full') * dx
        new_max = np.max(new_corr)
        if new_max < best_c5:
            best_c5 = new_max
            best_h = trial_h
    
    return (best_h, best_c5, n_points)
