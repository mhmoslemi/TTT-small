import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    max_iter = 2000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb_indices = np.random.choice(n_points, size=10, replace=False)
        noise = np.random.normal(0, 0.02, n_points)
        h_perturbed = h.copy()
        h_perturbed[perturb_indices] += noise[perturb_indices]
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        new_c5 = compute_c5(h_perturbed, 2.0 / n_points, n_points)
        if new_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = h_perturbed.copy()
                best_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
