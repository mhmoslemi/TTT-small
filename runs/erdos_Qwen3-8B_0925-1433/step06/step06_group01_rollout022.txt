import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    N_p = 50
    particles = [initial_h_values + np.random.normal(0, 0.1, n_points) for _ in range(N_p)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(p) for p in particles]
    personal_bests = [p.copy() for p in particles]
    global_best = particles[0].copy()
    best_c5 = compute_c5(global_best, dx, n_points)
    velocities = np.random.rand(N_p, n_points) * 0.01
    w, c1, c2 = 0.75, 1.5, 1.5
    start_time = time.time()
    max_iterations = 1000

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(N_p):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            curr_c5 = compute_c5(particles[i], dx, n_points)
            if curr_c5 < compute_c5(personal_bests[i], dx, n_points):
                personal_bests[i] = particles[i].copy()
                if curr_c5 < best_c5:
                    global_best = personal_bests[i].copy()
                    best_c5 = curr_c5
        corr = np.correlate(global_best, 1 - global_best, mode='full')
        max_corr = np.max(corr)
        if max_corr > 1e-8:
            delta = -0.1 * max_corr
            g_best = global_best.copy()
            g_best += delta
            g_best = np.clip(g_best, 0, 1)
            g_best = g_best * target_sum / np.sum(g_best)
            new_c5 = compute_c5(g_best, dx, n_points)
            if new_c5 < best_c5:
                global_best = g_best
                best_c5 = new_c5
    return (global_best, best_c5, n_points)
