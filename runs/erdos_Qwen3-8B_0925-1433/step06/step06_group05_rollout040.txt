import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()

    if h_sum > 0:
        h = h * target_sum / h_sum
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    current_h = h.copy()
    velocity = np.zeros(n_points)
    learning_rate = 0.1
    momentum = 0.8
    steps_without_improvement = 0
    max_steps_without_improvement = 100
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
