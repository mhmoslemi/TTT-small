    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    current_h = h.copy()
    best_h = h.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    current_c5 = best_c5
    temp = 1000.0
    cooling_rate = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        idx = np.random.randint(n_points)
        delta = np.random.uniform(-0.2, 0.2)
        new_h = current_h.copy()
        new_h[idx] += delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < current_c5:
            current_h = new_h.copy()
            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() < prob:
                current_h = new_h.copy()
                current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        temp *= cooling_rate
    
    return (best_h, best_c5, n_points)
