import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    for _ in range(200):  # Increase iterations for better exploration
        if time.time() - start_time > budget_s:
            break
        # Generate diverse candidates
        candidates = []
        # Perturb best
        perturbed_h = np.clip(best_h + np.random.normal(0, 0.005, n_points), 0, 1)
        perturbed_sum = np.sum(perturbed_h)
        perturbed_h = perturbed_h * target_sum / perturbed_sum
        candidates.append(perturbed_h)
        # Random uniform
        random_h = np.random.uniform(0, 1, n_points)
        random_h = random_h * target_sum / np.sum(random_h)
        candidates.append(random_h)
        # Segmented pattern
        seg_h = np.zeros(n_points)
        seg_h[:n_points//3] = 1.0
        seg_h[n_points//3:2*n_points//3] = 0.5

        seg_h = seg_h * target_sum / np.sum(seg_h)
        candidates.append(seg_h)
        # Circular binary
        cir_h = np.zeros(n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
