import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h_values
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    
    # Population parameters
    pop_size = 100
    mut_factor = 0.8
    crossover_prob = 0.8
    elite_ratio = 0.2
    refine_ratio = 0.1
    max_iter = int(budget_s * 0.9 / 0.39)
    time_start = time.time()
    
    # Generate initial population
    def generate_wave():
        x = np.linspace(0, 2, n_points)
        return 0.5 * (1 + np.sign(np.sin(np.pi * x)))

    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.3:
            # Clone with noise
            vec = h + np.random.normal(0, 0.05, n_points)
        elif np.random.rand() < 0.4:
            # Random vector
            vec = np.random.rand(n_points)
        else:
            # Wave pattern
            vec = generate_wave()
        vec = np.clip(vec, 0, 1)
        if np.sum(vec) > 0:
            vec = vec * target_sum / np.sum(vec)
        population.append(vec)
    
    def evaluate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        
        # Evaluate all
        fitness = np.array([evaluate(ind) for ind in population])
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs[:int(pop_size * elite_ratio)]]
        
        # Mutation and crossover
        new_population = []
        for i in range(len(population)):
            a, b = np.random.choice(len(population), 2, replace=False)
            ha, hb = population[a], population[b]
            mutant = ha + mut_factor * (hb - ha)
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) > 0:
                mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
        
        # Crossover
        for i in range(len(new_population)):
            if np.random.rand() < crossover_prob:
                parent = population[np.random.randint(len(population))]
                child = new_population[i] * 0.5 + parent * 0.5
                new_population[i] = np.clip(child, 0, 1)
                if np.sum(new_population[i]) > 0:
                    new_population[i] = new_population[i] * target_sum / np.sum(new_population[i])
        
        # Replace worst
        population += new_population
        population.sort(key=lambda x: evaluate(x))
        population = population[:pop_size]
        
        # Local refinement
        top_indices = np.argsort(fitness)[:int(pop_size * refine_ratio)]
        for i in top_indices:
            h = population[i].copy()
            res = minimize(evaluate, h, method='SLSQP',
                           bounds=[(0,1)]*n_points,

                           tol=1e-8)
