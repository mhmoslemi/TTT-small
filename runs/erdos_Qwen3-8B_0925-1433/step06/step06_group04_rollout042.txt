import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    population = []
    initial_h = np.clip(initial_h_values, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros(n_points)
    population.append(initial_h)

    for _ in range(99):
        rand_h = np.random.rand(n_points)
        rand_h = np.clip(rand_h, 0, 1)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        scores = [compute_c5(h) for h in population]
        sorted_indices = np.argsort(scores)
        sorted_pop = [population[i] for i in sorted_indices]
        num_parents = int(len(population) * 0.5)
        parents = sorted_pop[:num_parents]

        new_pop = []
        for _ in range(len(population) - num_parents):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = np.random.choice(parents)
            noise = np.random.normal(0, 0.1, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child) if np.sum(child) > 0 else np.zeros(n_points)
            new_pop.append(child)

        population = sorted_pop[:num_parents] + new_pop
        current_best = sorted_pop[0]
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        if time.time() - start_time >= budget_s:
            break

    return (best_h, best_c5, n_points)
