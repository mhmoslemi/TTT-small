import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Symmetrize initial_h_values
    symmetric_h = np.zeros(n_points)
    for i in range(n_points):
        j = n_points - 1 - i
        if i < j:
            symmetric_h[i] = initial_h_values[i]
            symmetric_h[j] = initial_h_values[i]
        elif i == j:
            symmetric_h[i] = initial_h_values[i]

    scaled = symmetric_h * target_sum / np.sum(symmetric_h)
    best_h = scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Perturb current best solution
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum == 0:
            perturbed = np.zeros_like(perturbed) + 0.5
        else:
            perturbed = perturbed * target_sum / perturbed_sum

        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, perturbed, method='COBYLA', bounds=bounds, constraints=cons)

        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
