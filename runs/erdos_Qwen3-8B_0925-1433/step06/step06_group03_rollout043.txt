import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Structured initial guess: evenly spaced 1s
    m = int(np.round(target_sum))
    positions = np.linspace(0, 2, m + 1)[:-1]
    samples = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for x in positions:
        idx = np.argmin(np.abs(samples - x))
        h[idx] = 1
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    mutation_amplitude = 0.05
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, mutation_amplitude, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum

        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)

        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
            if best_c5 < 0.381:
                mutation_amplitude *= 0.95
            elif best_c5 > 0.382:
                mutation_amplitude *= 1.05
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
