import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, target_sum):
        h = h * (target_sum / sum_h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    elapsed = 0.0
    iterations = 0
    max_iter = int(budget_s / 1.5)

    while elapsed < budget_s:
        elapsed = time.time() - start_time
        if iterations >= max_iter:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        max_idx = np.argmax(corr)
        s = max_idx - (n_points - 1)
        shifted = np.arange(n_points) + s
        valid_mask = (shifted >= 0) & (shifted < n_points)
        grad = np.zeros(n_points)
        grad[valid_mask] = (1 - h[shifted[valid_mask]]) * dx

        best_trial = None
        best_new_max = current_max
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3]:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0.0, 1.0)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_max:
                best_new_max = new_max
                best_trial = trial_h

        if best_trial is not None:
            h = best_trial.copy()

        if iterations % 50 == 0 and np.random.rand() < 0.2:
            perturb = np.random.uniform(-0.15, 0.15, size=n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))

        iterations += 1

    return (best_h, best_c5, n_points)
