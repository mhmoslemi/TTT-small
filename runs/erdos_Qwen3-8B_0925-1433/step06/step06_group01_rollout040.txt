        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
        
        if best_c5 < 0.38:
            learning_rate *= 0.95
        elif best_c5 > 0.385:
            learning_rate *= 1.05
        
        iterations += 1
    
    return (best_h, best_c5, n_points)
