import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e6 * np.abs(np.sum(x) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0)] * n_points
    result = differential_evolution(
        obj, bounds, strategy='best1bin',
        maxiter=int(budget_s * 0.8), popsize=50,
        tol=1e-8, mutation=(0.5, 1), recombination=0.7,
        seed=seed, disp=False
    )
    best_h = result.x
    best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
