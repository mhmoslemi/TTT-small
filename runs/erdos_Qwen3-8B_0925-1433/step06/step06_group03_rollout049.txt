import time
from scipy.optimize import minimize, Bounds, LinearConstraint

    while iterations < max_iterations and time.time() - start_time < budget_s:
        # Create perturbed guess

        # Define objective function
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        # Constraints and bounds
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
