import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    max_iterations = 1000
    learning_rate = 0.2
    decay_rate = 0.995
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]
        top_lags = list(max_indices[:3])  # Use top 3 lags
        avg_grad = np.zeros(n_points)
        for lag in top_lags:
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i + lag < n_points:
                    grad[i] += 1 - h[i + lag]
                if i - lag >= 0:
                    grad[i] -= h[i - lag]
            avg_grad += grad
