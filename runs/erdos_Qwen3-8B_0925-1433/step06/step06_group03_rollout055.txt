        scale = target_sum / total
        return x_clipped * scale
    
    def callback(x):
        x[:] = rescale(x)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    options = {'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-8}
    res = minimize(objective, initial_scaled, method='Nelder-Mead', bounds=bounds, callback=callback, options=options)
    end_time = time.time()
    
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h, dx, n_points)
    else:
        best_h = initial_scaled
        best_c5 = compute_c5(best_h, dx, n_points)
    
    best_h = rescale(best_h)
    best_c5 = compute_c5(best_h, dx, n_points)
    
    return (best_h, best_c5, n_points)
