        if current_sum > 0:
            h = h * target_sum / current_sum
        else:
            h = np.full(n_points, target_sum / n_points)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 100:
            break
    return (best_h, best_c5, n_points)
