import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        return h

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, sigma=0.05):
        noise = np.random.normal(0, sigma, n_points)
        return clip_rescale(h + noise)

    def crossover(parent1, parent2):
        idx = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        return clip_rescale(child)

    # Initialize population
    pop_size = 100
    population = []
    base = clip_rescale(initial_h_values.copy())
    population.append(base)

    for _ in range(pop_size - 1):
        if np.random.rand() < 0.5:
            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = mutate(base)
        population.append(h)

    best_h = base
    best_c5 = evaluate_c5(base)
    start_time = time.time()

    for _ in range(1500):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        c5s = np.array([evaluate_c5(h) for h in population])
        # Select top 30% as parents
        parent_indices = np.argsort(c5s)[:int(0.3 * pop_size)]
        parents = [population[i] for i in parent_indices]
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            offspring.append(mutate(child))
        # Replace worst half
        combined = population + offspring
        c5s_combined = np.array([evaluate_c5(h) for h in combined])
        sort_indices = np.argsort(c5s_combined)
        population = [combined[i] for i in sort_indices[:pop_size]]
        # Update best
        if evaluate_c5(population[0]) < best_c5:
            best_h, best_c5 = population[0].copy(), evaluate_c5(population[0])

    return (best_h, best_c5, n_points)
