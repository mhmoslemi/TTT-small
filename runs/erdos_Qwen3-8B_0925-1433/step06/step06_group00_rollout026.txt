import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in locals():

        h = np.interp(x_new, x_initial, initial_h_values)
        h = h * target_sum / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    
    # Optimization with COBYLA
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:

                best_h = curr_h.copy()
