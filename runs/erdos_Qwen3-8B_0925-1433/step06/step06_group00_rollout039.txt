import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_normalized = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_normalized.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    T = 1.0
    cooling_rate = 0.995
    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        new_c5 = compute_c5(perturbed, dx, n_points)
        if new_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = perturbed.copy()
                best_c5 = new_c5
        T *= cooling_rate
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
