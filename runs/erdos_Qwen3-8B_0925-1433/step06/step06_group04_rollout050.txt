from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        return h * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 100
    population = []

    h0 = clip_and_rescale(np.array(initial_h_values))
    population.append(h0)
    # Random vectors
    for _ in range(pop_size - 1):

        h = clip_and_rescale(h)
        population.append(h)
    
    best_h = h0
    best_c5 = compute_c5(best_h)
    elapsed = 0
    
    for _ in range(1000):
        if elapsed > budget_s - 1:
            break
        start = time.time()
        
        # Evaluate all
        scores = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(scores)
        top_idx = sorted_idx[:int(pop_size * 0.1)]
        top_candidates = [population[i] for i in top_idx]
        
        # Crossover
        new_population = []
        for _ in range(int(pop_size * 0.5)):
            p1, p2 = np.random.choice(top_candidates, 2, replace=False)

            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            child = clip_and_rescale(child)
            new_population.append(child)
        
        # Mutation
        for _ in range(pop_size - len(new_population)):
            h = np.random.choice(top_candidates)
            noise = np.random.normal(0, 0.05, n_points)

        # Replace worst
