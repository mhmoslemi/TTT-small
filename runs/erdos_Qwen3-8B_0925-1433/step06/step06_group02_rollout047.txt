from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    h = initial_scaled.copy()

    start_time = time.time()

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Find the worst lag
        worst_k = np.argmax(corr)
        # Optimize for the worst lag
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        # Define the objective function
        def obj(x):

            # Compute the correlation for this lag
            corr_k = np.correlate(x, 1 - x, mode='full')[k]
            return corr_k
        # Solve the optimization problem
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        # Update h to the solution
        h = res.x
        # Early stopping if no improvement
        if np.allclose(h, best_h, atol=1e-5):
