            grad[valid_mask] += (1 - h[valid[valid_mask]])
        
        # Line search
        trials = []
        for alpha in [0.01, 0.05, 0.1, 0.2]:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            trials.append((alpha, trial_h))
        
        best_trial = None
        best_new_max = current_max
        for alpha, trial_h in trials:
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_corr_vals = new_corr * dx
            new_max = np.max(new_corr_vals)
            if new_max < best_new_max:
                best_new_max = new_max
                best_trial = trial_h
        
        if best_trial is not None:
            h = best_trial
            if best_new_max < best_c5:
                best_c5 = best_new_max
                best_h = h.copy()
        
        # Random perturbation every 100 iterations
        if _ % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            perturbed_h = h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_sum = np.sum(perturbed_h)
            if perturbed_sum != 0:
                perturbed_h = perturbed_h * (target_sum / perturbed_sum)
            new_corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
            new_corr_vals = new_corr * dx
            new_max = np.max(new_corr_vals)
            if new_max < best_c5:
                h = perturbed_h.copy()
                best_c5 = new_max
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
