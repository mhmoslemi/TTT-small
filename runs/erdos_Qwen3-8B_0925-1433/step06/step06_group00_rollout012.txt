                zeros_indices = np.where(s == 0)[0]
                s[zeros_indices[:abs(diff)]] = 1
    else:
        target = N // 2
        s = np.zeros(N, dtype=int)
        s[:target] = 1
    dx = 2.0 / N
    
    def compute_c5(s, dx):
        corr = np.correlate(s, 1 - s, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(s, dx)
    best_s = s.copy()
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        i, j = np.random.randint(0, N, size=2)
        if i == j:
            continue
        if s[i] != s[j]:
            s[i], s[j] = s[j], s[i]
            new_c5 = compute_c5(s, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_s = s.copy()
            else:
                s[i], s[j] = s[j], s[i]
    
    return (best_s.astype(np.float64), best_c5, N)
