import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    population = [initial_scaled.copy()]
    for _ in range(29):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(initial_scaled + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        current_c5s = [compute_c5(h, dx, n_points) for h in population]
        best_idx = np.argmin(current_c5s)
        best_candidate = population[best_idx]

            mutant = population[a] + 0.9 * (population[b] - population[c])

            trial = np.where(trial, mutant, population[i])
            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)

        new_c5s = [compute_c5(h, dx, n_points) for h in new_population]
        best_new_idx = np.argmin(new_c5s)
        worst_idx = np.argmax(current_c5s)
        if new_c5s[best_new_idx] < current_c5s[worst_idx]:

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
