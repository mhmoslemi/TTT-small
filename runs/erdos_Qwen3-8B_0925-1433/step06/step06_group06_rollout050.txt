        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all candidates
        scores = [evaluate(h) for h in population]
        # Sort population by score
        sorted_indices = np.argsort(scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_scores = np.array(scores)[sorted_indices]
        
        # Select elites
        num_elites = int(elitism_rate * population_size)
        elites = sorted_population[:num_elites]
        
        # Generate new candidates
        new_population = elites.copy()
        remaining = population_size - num_elites
        # Use tournament selection for the remaining
        while len(new_population) < population_size:
            # Select tournament candidates
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            tournament_scores = sorted_scores[tournament]
            winner_index = tournament[np.argmin(tournament_scores)]
            parent1 = sorted_population[winner_index]
            # Select another parent
            tournament2 = np.random.choice(population_size, tournament_size, replace=False)
            tournament_scores2 = sorted_scores[tournament2]
            winner_index2 = tournament2[np.argmin(tournament_scores2)]
            parent2 = sorted_population[winner_index2]
            
            # Crossover
            split = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:split], parent2[split:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.random.rand()
            # Clamp and normalize
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child > 0:
                child = child * target_sum / sum_child
            
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Update best solution
        current_best_score = sorted_scores[0]
        if current_best_score < best_c5:
            best_c5 = current_best_score
            best_h = sorted_population[0]
    
    return (best_h, best_c5, n_points)
