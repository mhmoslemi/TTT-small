import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)
    pop_size = 100
    pop = [np.clip(initial_h_values, 0, 1) * target_sum / np.sum(initial_h_values) if np.sum(initial_h_values) > 0 else np.zeros(n_points)]
    for _ in range(pop_size - 1):
        m = np.random.rand(n_points)
        h = m / np.sum(m) * target_sum
        pop.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = pop[0]
    best_c5 = evaluate(best_h)

    time_start = time.time()
    while time.time() - time_start < budget_s:
        pop_sorted = sorted(pop, key=evaluate)
        top_10 = pop_sorted[:int(len(pop_sorted)*0.1)]
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(top_10, 2, replace=False)
            if np.random.rand() < 0.8:
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            else:
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        new_pop += pop_sorted[:int(len(pop_sorted)*0.7)]
        pop = new_pop
        for h in pop:
            c = evaluate(h)
            if c < best_c5:
                best_c5 = c
                best_h = h

    return (best_h, best_c5, n_points)
