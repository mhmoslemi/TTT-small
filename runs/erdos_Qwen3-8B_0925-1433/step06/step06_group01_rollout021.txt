import numpy as np
import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    start_time = time.time()
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        n = len(best_h)
        lag = max_index - (n - 1)
        W = np.zeros((n, n), dtype=np.float64)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                W[i, j] = 1.0
        h_var = cp.Variable(n)
        obj = cp.Maximize(h_var @ W @ h_var)
        constraints = [cp.sum(h_var) == target_sum, 0 <= h_var, h_var <= 1]
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS, eps=1e-6)
            if prob.status == 'optimal':
                new_h = np.clip(h_var.value, 0, 1)
                if np.isclose(np.sum(new_h), target_sum):
                    new_c5 = evaluate_erdos_solution(new_h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h
        except:
            pass
    return (best_h, best_c5, n_points)
