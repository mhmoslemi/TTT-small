def run(seed=42, budget_s=600):
    import numpy as np
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * (target_sum / np.sum(h))

    if sum_h != target_sum:
        h = h * (target_sum / sum_h)
    best_h = h.copy()

    start_time = time.time()

        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        top_lags = np.argsort(np.abs(corr))[-int(0.2 * n_points):]
        grads = []
        for lag in top_lags:
            s = lag - (n_points - 1)
            shifted_indices = np.arange(n_points) + s
            valid_mask = (shifted_indices >= 0) & (shifted_indices < n_points)
            grad_k = (1 - h[shifted_indices[valid_mask]]) * dx

            grads.append(grad_k * weight_k)
        gradient = np.sum(grads, axis=0)
        alphas = [0.01, 0.05, 0.1, 0.2]
        best_alpha = 0.0
        best_new_max = current_max
        for alpha in alphas:
            trial_h = h - alpha * gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_max = np.max(new_corr) * dx

                best_new_max = new_max
