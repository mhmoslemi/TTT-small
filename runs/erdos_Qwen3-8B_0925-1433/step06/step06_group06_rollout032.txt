import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iter = 1000
    learning_rate = 0.1
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(-corr)[:5]
        grad = np.zeros_like(h)
        for lag in top_lags:
            valid1 = np.arange(n_points) + lag < n_points
            term1 = np.zeros(n_points)
            term1[valid1] = 1 - h[valid1]
            valid2 = np.arange(n_points) - lag >= 0
            term2 = np.zeros(n_points)
            term2[valid2] = h[valid2]
            grad_current = (term1 - term2) * dx
            grad += grad_current
        grad /= len(top_lags)
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        learning_rate *= 0.9995
    return (best_h, best_c5, n_points)
