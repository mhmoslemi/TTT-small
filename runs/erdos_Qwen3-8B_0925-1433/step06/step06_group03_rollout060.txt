import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guess as a sine wave
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 + 0.5 * np.sin(np.pi * x)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    mutation_amplitude = 0.03
    iteration = 0

    while time.time() - start_time < budget_s:
        # Perturb best_h
        perturbed = best_h + np.random.normal(0, mutation_amplitude, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum

        # Define objective
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        # Constraints
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)

        # Optimize
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                if best_c5 < 0.381:
                    mutation_amplitude *= 0.98
                elif best_c5 > 0.382:
                    mutation_amplitude *= 1.02
            else:
                mutation_amplitude *= 1.01
        iteration += 1

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
