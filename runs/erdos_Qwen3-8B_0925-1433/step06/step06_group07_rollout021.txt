import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = initial_h_values.copy()
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, target_sum):
        h = h * (target_sum / sum_h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    for _ in range(1200):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        max_lag = np.argmax(corr)
        s = max_lag - (n_points - 1)
        
        shifted_indices = (np.arange(n_points) + s) % n_points
        grad = np.zeros(n_points)
        grad[shifted_indices] = (1 - h[shifted_indices]) * dx
        
        # Line search
        alpha = 0.05
        trial_h = h - alpha * grad
        trial_h = np.clip(trial_h, 0, 1)
        trial_sum = np.sum(trial_h)

        trial_h = trial_h * (target_sum / trial_sum)

        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < current_max:
