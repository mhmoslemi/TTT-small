import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    max_iter = 10000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb_indices = np.random.choice(n_points, size=5, replace=False)
        perturbed_h = best_h.copy()
        perturbed_h[perturb_indices] += np.random.normal(0, 0.02, size=5)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h_sum = np.sum(perturbed_h)
        if perturbed_h_sum == 0:
            perturbed_h = np.zeros_like(perturbed_h)
        else:
            perturbed_h = perturbed_h * target_sum / perturbed_h_sum
        new_c5 = compute_c5(perturbed_h, dx, n_points)
        if new_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                best_h = perturbed_h.copy()
                best_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
