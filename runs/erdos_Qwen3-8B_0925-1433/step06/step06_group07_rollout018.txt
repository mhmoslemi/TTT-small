    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if np.abs(sum_h - target_sum) > 1e-6:
        h = h * (target_sum / sum_h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    for iteration in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(corr)
        active_lags = np.argsort(corr)[-30:]  # Select top 30 lags
        
        grad = np.zeros(n_points)
        for idx in active_lags:
            lag = idx - (n_points - 1)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
        
        # Line search
        best_alpha = 0.0
        best_c5_new = current_max
        alphas = [0.01, 0.05, 0.1, 0.2]
        for alpha in alphas:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full') * dx
            new_c5 = np.max(new_corr)
            if new_c5 < best_c5_new:
                best_c5_new = new_c5
                best_alpha = alpha
        
        if best_alpha > 0:
            h = h - best_alpha * grad
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        if iteration % 50 == 0:
            perturb = np.random.rand(n_points) * 0.15 - 0.075
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    
    return (best_h, best_c5, n_points)
