import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    population_size = 50
    population = []
    
    # Initialize with initial_h_values
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population.append(h_initial.copy())
    
    # Generate additional random candidates
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand.copy())
    
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    
    start_time = time.time()
    
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        
        elite_idx = np.argsort(fitness)[:int(0.2 * population_size)]
        elites = [population[i] for i in elite_idx]
        
        new_population = elites.copy()
        remaining = population_size - len(elites)
        
        for _ in range(remaining):
            t_idx = np.random.choice(len(population), size=5, replace=False)
            t_fitness = [fitness[i] for i in t_idx]
            winner = t_idx[np.argmin(t_fitness)]
            p1 = population[winner]
            
            while True:
                p2_idx = np.random.randint(len(population))
                if p2_idx != winner:
                    break
            p2 = population[p2_idx]
            
            cp = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:cp], p2[cp:]])
            
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% mutation rate per gene
                    child[i] = np.random.rand()
            
            child = child * target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            new_population.append(child)
        
        population = new_population
        
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
