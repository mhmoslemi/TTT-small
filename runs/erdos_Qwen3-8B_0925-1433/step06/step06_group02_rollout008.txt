        def objective(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[k_max]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            maxiter=10, popsize=15, tol=1e-5, mutation=(0.5, 1),
            recombination=0.8, polish=True, disp=False
        )
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_sum = np.sum(current_h)
        current_h = current_h * target_sum / (current_sum + 1e-12)
        new_c5 = compute_c5(current_h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
    return (best_h, best_c5, n_points)
