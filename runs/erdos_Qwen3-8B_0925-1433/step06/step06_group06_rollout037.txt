    max_iter = 500
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        idx1, idx2, idx3 = np.random.choice(len(population), 3, replace=False)
        h_a, h_b, h_c = population[idx1], population[idx2], population[idx3]
        mutant = h_a + F * (h_b - h_c)
        mutant = np.clip(mutant, 0, 1)
        mutant = mutant * target_sum / np.sum(mutant)
        trial = np.copy(h_a)
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0, 1)
        trial = trial * target_sum / np.sum(trial)
        trial_c5 = evaluate_erdos_solution(trial)
        if trial_c5 < evaluate_erdos_solution(h_a):
            population[idx1] = trial.copy()
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial.copy()

    return (best_h, best_c5, n_points)
