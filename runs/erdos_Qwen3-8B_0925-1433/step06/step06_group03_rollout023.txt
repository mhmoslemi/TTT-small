import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    positions = np.linspace(0, 2, n_points)

    h_initial[np.argsort(-tri_wave)[:int(n_points/2)]] = 1
    h_initial = h_initial * (target_sum / np.sum(h_initial))
    h_initial = np.clip(h_initial, 0, 1)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iter = 300
    # Objective with max correlation
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    # Constrained optimization
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    for _ in range(max_iter):

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
