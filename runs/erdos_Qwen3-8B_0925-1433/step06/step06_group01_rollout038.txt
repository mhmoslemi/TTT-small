    best_c5 = compute_c5(best_h, dx, n_points)

    omega = 0.75
    c1, c2 = 1.5, 1.5
    alpha = 0.1
    iterations = 0
    start_time = time.time()

    while iterations < 500 and time.time() - start_time < budget_s:
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = omega * np.random.normal(0, 0.01, n_points) + \
                  c1 * r1 * (particles[i] - particles[i]) + \
                  c2 * r2 * (best_h - particles[i])
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])

        # Track global best
        for p in particles:
            current_c5 = compute_c5(p, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = p.copy()

        # Gradient descent step on global best
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr, k_max = np.max(corr), np.argmax(corr)
        delta = alpha * max_corr * np.sign(max_corr)
        best_h += delta
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)

        iterations += 1

    return (best_h, best_c5, n_points)
