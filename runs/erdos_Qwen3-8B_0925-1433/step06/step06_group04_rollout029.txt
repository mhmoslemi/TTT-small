        scores = [obj(h) for h in population]
        elites = sorted(zip(population, scores), key=lambda x: x[1])[:int(population_size * 0.5)]
        elites = [e[0] for e in elites]
        new_population = []
        for _ in range(population_size):
            a, b, c = random.sample(elites, 3)
            mutant = a + F * (b - c)
            target = random.choice(population)
            trial = mutant.copy()
            for i in range(n_points):
                if random.random() < CR:
                    trial[i] = mutant[i]
                else:
                    trial[i] = target[i]
            trial = np.clip(trial, 0, 1)
            total = np.sum(trial)
            trial = trial * target_sum / total if total > 0 else np.zeros(n_points)
            new_population.append(trial)
        combined = population + new_population
        all_scores = [obj(h) for h in combined]
        combined_sorted = sorted(zip(combined, all_scores), key=lambda x: x[1])
        population = [x[0] for x in combined_sorted[:population_size]]
        current_best = min(scores + [obj(h) for h in new_population])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[0]

    final_eval = obj(best_h)
    return (best_h, final_eval, n_points)
