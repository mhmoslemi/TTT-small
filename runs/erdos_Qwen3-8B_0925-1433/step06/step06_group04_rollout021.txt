import numpy as np
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with parent and variations
    pop_size = 50
    pop = []
    np.random.seed(seed)
    for _ in range(pop_size):
        if np.random.rand() < 0.3:
            h = np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1)
        elif np.random.rand() < 0.5:
            h = np.random.rand(n_points)
        else:
            h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        pop.append(h)
    pop = np.array(pop)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return h * target_sum / s

    start_time = time.time()
    best_h = pop[0].copy()
    best_c5 = obj(best_h)

    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        scores = np.array([obj(h) for h in pop])
        elites = pop[np.argsort(scores)[:int(pop_size / 2)]]
        # Generate new candidates
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(elites, 3, replace=False)
            mutant = a + 0.8 * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant) if np.sum(mutant) > 0 else mutant

            mask = np.random.rand(n_points) < 0.7
            trial = np.where(mask, mutant, pop[i])
            trial = clip_and_rescale(trial)

        pop = np.array(new_pop)
        # Update best
        curr_best = pop[np.argmin(scores)]
        if obj(curr_best) < best_c5:
            best_h, best_c5 = curr_best, obj(curr_best)
