import time

        h = h * target_sum / np.sum(h)
        h_pop.append(h)
    h_pop = np.array(h_pop)
    best_c5 = np.inf
    best_h = np.zeros(n_points)

            h_b = h_pop[b]
            h_c = h_pop[c]
            mutant = h_a + 0.8 * (h_b - h_c)
