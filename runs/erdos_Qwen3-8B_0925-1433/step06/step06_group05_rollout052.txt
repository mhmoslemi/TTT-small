import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    initial_scaled = initial_h_values * target_sum / initial_sum
    initial_clip = np.clip(initial_scaled, 0, 1)
    clip_sum = np.sum(initial_clip)
    if clip_sum > 0:
        initial_clip = initial_clip * target_sum / clip_sum
    h = initial_clip.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    for step in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        h = np.clip(h - 0.1 * grad, 0, 1)

    return (best_h, best_c5, n_points)
