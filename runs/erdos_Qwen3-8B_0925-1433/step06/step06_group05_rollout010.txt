    h = h * target_sum / current_sum
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    learning_rate = 0.3
    momentum = 0.9
    velocity = np.zeros(n_points)
    perturb_interval = 100
    perturb_amp = 0.2
    step = 0
    
    while time.time() - start_time < budget_s:
        step += 1
        cross_corr = scipy.fft.ifft(scipy.fft.fft(h) * np.conj(scipy.fft.fft(1.0 - h))).real
        current_c5 = np.max(cross_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        k_max = np.argmax(cross_corr)
        j = (np.arange(n_points) + k_max) % n_points
        grad = (1.0 - h[j]) * dx
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        if step % perturb_interval == 0:
            h += np.random.normal(0, perturb_amp, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
