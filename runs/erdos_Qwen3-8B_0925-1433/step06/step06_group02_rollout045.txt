import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    population = [initial_scaled]
    for _ in range(29):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_scaled + perturbation, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        mutants = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + np.random.rand(n_points) * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            mutants.append(mutant)
        scores = [compute_c5(h, dx, n_points) for h in mutants]
        mutants, scores = zip(*sorted(zip(mutants, scores)))
        for i in range(len(population)):

                population[i] = mutants[i]
        curr_h = min(population, key=lambda h: compute_c5(h, dx, n_points))

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
