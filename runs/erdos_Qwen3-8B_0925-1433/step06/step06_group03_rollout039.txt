import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values.copy()
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
            h_initial = res.x + np.random.normal(0, 0.01, n_points)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * target_sum / np.sum(h_initial)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
