import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    np.random.seed(seed)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    population = []

    for _ in range(50):
        if np.random.rand() < 0.1:
            h = np.random.uniform(0, 1, n_points)
        elif np.random.rand() < 0.3:
            h = np.clip(h_initial + np.random.normal(0, 0.1, n_points), 0, 1)
        else:
            h = np.clip(h_initial + np.random.normal(0, 0.05, n_points), 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)
    best_h = h_initial.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = np.array([obj(ind) for ind in population])
        elite_indices = np.argsort(fitness)[:int(len(population)/2)]
        elites = population[elite_indices]

        new_population = []
        for _ in range(len(population)):
            idx1, idx2, idx3 = np.random.choice(len(elites), 3, replace=False)
            a, b, c = elites[idx1], elites[idx2], elites[idx3]
            mutant = a + np.random.rand(n_points)*(b - c)*0.8
            trial = np.copy(mutant)
            for i in range(n_points):
                if np.random.rand() < 0.7:
                    trial[i] = mutant[i]
            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)
            new_population.append(trial)

        fitness_new = np.array([obj(ind) for ind in new_population])
        best_new = new_population[np.argmin(fitness_new)]
        if obj(best_new) < best_c5:
            best_h = best_new
            best_c5 = obj(best_new)
        population = new_population

    return (best_h, best_c5, n_points)
