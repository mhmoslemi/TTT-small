import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_low = 100
    n_high = 200
    dx_high = 2.0 / n_high
    target_sum_high = n_high / 2
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.random.rand(45)
        h_initial = h_initial * (45/2) / np.sum(h_initial)
    def compute_c5(h, dx, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    # Generate low-res candidates
    candidates_low = []
    for _ in range(10):
        h_low = np.random.rand(n_low)
        h_low = h_low * (n_low/2) / np.sum(h_low)
        c5 = compute_c5(h_low, 2.0 / n_low, n_low)
        candidates_low.append((c5, h_low))
    candidates_low.sort()
    top_low = [c[1] for c in candidates_low[:5]]
    # Upsample to high-res
    candidates_high = []
    for h_low in top_low:
        h_high = np.repeat(h_low, 2)
        current_sum = np.sum(h_high)
        h_high = h_high * (target_sum_high / current_sum)
        candidates_high.append(h_high)
    best_h = None
    best_c5 = np.inf
    for h in candidates_high:
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx_high
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_high), [target_sum_high], [target_sum_high])
        res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons)
        if res.success:
            curr_c5 = compute_c5(res.x, dx_high, n_high)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x.copy()
    # Perturb and refine
    start = time.time()
    for _ in range(300):
        if time.time() - start > budget_s:
            break
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx_high
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_high), [target_sum_high], [target_sum_high])
        res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons)
        if res.success:
            curr_c5 = compute_c5(res.x, dx_high, n_high)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res.x.copy()
        # Perturb
        pert = np.random.normal(0, 0.01, n_high)
        pert_h = np.clip(best_h + pert, 0, 1)
        pert_h = pert_h * target_sum_high / np.sum(pert_h)
        res_p = minimize(obj, pert_h, method='COBYLA', bounds=bounds, constraints=cons)
        if res_p.success:
            curr_c5_p = compute_c5(res_p.x, dx_high, n_high)
            if curr_c5_p < best_c5:
                best_c5 = curr_c5_p
                best_h = res_p.x.copy()
    return (best_h, best_c5, n_high)
