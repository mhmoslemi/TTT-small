        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        pop.append(h)

    best_c5 = float('inf')
    best_h = None
    start = time.time()
    for _ in range(500):
        if time.time() - start > budget_s:
            break
        c5s = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            max_val = np.max(corr)
            c5s.append(max_val * (2.0 / n_points))
        sorted_idx = np.argsort(c5s)
        sorted_pop = [pop[i] for i in sorted_idx]
        sorted_c5 = np.sort(c5s)
        num_elites = int(0.2 * len(pop))
        elites = sorted_pop[:num_elites]
        new_pop = elites.copy()
        for _ in range(len(sorted_pop) - num_elites):
            p1 = tournament_selection(pop, c5s)
            p2 = tournament_selection(pop, c5s)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        pop = new_pop
        if sorted_c5[0] < best_c5:
            best_c5 = sorted_c5[0]
            best_h = sorted_pop[0]
    return (best_h, best_c5, n_points)
