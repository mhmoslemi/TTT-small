import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    
    bounds = [(0.0, 1.0)] * n_points
    max_iter_outer = 200
    
    for _ in range(max_iter_outer):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_max] * dx
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=15,
            tol=1e-5, maxiter=10, mutation=(0.5, 1), recombination=0.8, disp=False
        )
        new_h = np.clip(result.x, 0.0, 1.0)
        new_h_sum = np.sum(new_h)
        new_h = new_h * target_sum / new_h_sum
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
    return (best_h, best_c5, n_points)
