import numpy as np
import time

from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h_values.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    for _ in range(500):  # Increased iteration count for better search
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_sum = np.sum(perturbed_h)
        if perturbed_sum == 0:
            perturbed_h = np.full_like(perturbed_h, 0.5)
        else:
            max_clipped = np.max(perturbed_h)
            possible_scale = target_sum / perturbed_sum
            actual_scale = min(possible_scale, 1.0 / max_clipped)
            perturbed_h = perturbed_h * actual_scale
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, perturbed_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        curr_h = res.x
        curr_c5 = compute_c5(curr_h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
