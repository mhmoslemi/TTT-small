import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(h, scale=0.05):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        noise = np.random.normal(0, scale, n_points)
        return np.clip(h + noise, 0, 1) * target_sum / np.sum(h) if np.sum(h) > 0 else h

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Structural change

        h = best_h.copy()
        h[idx] = 1 - h[idx]

        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
