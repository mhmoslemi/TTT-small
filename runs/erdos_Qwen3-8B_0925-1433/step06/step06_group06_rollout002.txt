import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize population with initial and random candidates
    population = [initial_h_values.copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = compute_c5(population[0])

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        fitness = [compute_c5(h) for h in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        elite_count = int(len(sorted_pop) * 0.2)
        elites = sorted_pop[:elite_count]
        remaining = sorted_pop[elite_count:]
        new_pop = elites.copy()
        for _ in range(len(remaining)):
            parents = np.random.choice(remaining, 2)
            # Crossover
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([parents[0][:split], parents[1][split:]])
            # Mutation
            mutate = np.random.rand(n_points) < 0.1
            child[mutate] = np.random.rand(np.sum(mutate))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop[:50]
        current_best = sorted_pop[0]
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    return (best_h, best_c5, n_points)
