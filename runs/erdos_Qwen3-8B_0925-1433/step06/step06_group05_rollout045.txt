    pbest = []

            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        pbest.append(h.copy())
        velocities.append(np.random.uniform(-0.03, 0.03, n_points))

    def evaluate_particle(i):
        h = particles[i]
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    gbest = min(particles, key=evaluate_particle)
    best_c5 = evaluate_particle(0)
