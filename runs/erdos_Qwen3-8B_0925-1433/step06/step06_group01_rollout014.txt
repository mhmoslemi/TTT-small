            new_gbest = gbest + delta
            new_gbest = np.clip(new_gbest, 0, 1)
            new_gbest = new_gbest * target_sum / np.sum(new_gbest)
            new_c5 = compute_c5(new_gbest, dx, n_points)
            if new_c5 < current_c5:
                gbest = new_gbest
                best_c5 = new_c5
                grad_steps += 1
        
        # Early termination if no progress
        if grad_steps > 100:
            break
    
    return (gbest, best_c5, n_points)
