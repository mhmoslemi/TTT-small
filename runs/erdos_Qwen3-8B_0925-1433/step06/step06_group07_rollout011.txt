            h = h * (target_sum / np.sum(h))
        if _ % 100 == 0 and _ > 0:
            perturb = np.random.normal(0, 0.05, size=n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    return (best_h, best_c5, n_points)
