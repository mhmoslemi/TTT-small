    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-6:
        h = h * (target_sum / sum_h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        corr_vals = corr[20:-20]  # Focus on middle lags
        top_indices = np.argsort(corr_vals)[-20:]
        lags = [idx - (n_points - 1) for idx in top_indices]
        grad = np.zeros(n_points)
        for lag in lags:
            lag_idx = lag + (n_points - 1)
            scale = corr[lag_idx]
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx * scale
                j = i - lag
                if 0 <= j < n_points:
                    grad[i] -= h[j] * dx * scale
        trial_h = h.copy()
        best_alpha = 0.0
        best_trial = h.copy()
        best_new_max = best_c5
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2]:
            trial = h - alpha * grad
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                continue
            trial = trial * (target_sum / trial_sum)
            new_corr = np.correlate(trial, 1 - trial, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_max:
                best_new_max = new_max
                best_trial = trial.copy()
                best_alpha = alpha
        if best_new_max < best_c5:
            h = best_trial
            best_c5 = best_new_max
            best_h = h.copy()
        if _ % 50 == 0:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    return (best_h, best_c5, n_points)
