            new_corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            delta = (new_corr[k_idx] - corr[k_idx]) / eps
            grad[j] = delta
        best_h = best_h - learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        
        # Random mutation
        noise = np.random.normal(0, mutation_amp, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        # Evaluate
        new_c5 = compute_c5(perturbed, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = perturbed.copy()
            if best_c5 < 0.38:
                mutation_amp *= 0.95
                learning_rate *= 0.95
            elif best_c5 > 0.385:
                mutation_amp *= 1.05
                learning_rate *= 1.1
        iterations += 1
    return (best_h, best_c5, n_points)
