import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()

    start_time = time.time()

    for _ in range(200):

            break

        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_sum = np.sum(perturbed_h)
        perturbed_h = perturbed_h * target_sum / perturbed_sum

        corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
        k_max = np.argmax(corr)
        def sub_obj(x):
            idx = (np.arange(n_points) + k_max) % n_points
            return np.sum(x * (1 - x[idx]))
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(sub_obj, perturbed_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

        if curr_c5 < best_c5:

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
