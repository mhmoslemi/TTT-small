import numpy as np
import time
import cvxpy as cp

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    max_iterations = 200
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        # Compute the correlation and find the worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_index = np.argmax(corr)
        lag = max_corr_index - (n_points - 1)
        # Construct matrix M
        M = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                M[i][j] = 1
        # Set up QP
        h_var = cp.Variable(n_points)
        Q = -2 * M
        c = np.ones(n_points)
        constraints = [cp.sum(h_var) == target_sum, 0 <= h_var, h_var <= 1]
        prob = cp.Problem(cp.Minimize(0.5 * cp.quad_form(h_var, Q) + c @ h_var), constraints)
        try:

            if prob.status == 'optimal':
