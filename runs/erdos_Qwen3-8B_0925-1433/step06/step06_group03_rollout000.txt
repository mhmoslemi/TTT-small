import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    mutation_amplitude = 0.05
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, mutation_amplitude, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        def obj(x):
            return compute_c5(x)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        try:
            res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        except:
            continue
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                if best_c5 < 0.381:
                    mutation_amplitude *= 0.95
                elif best_c5 > 0.382:
                    mutation_amplitude *= 1.05
    return (best_h, best_c5, n_points)
