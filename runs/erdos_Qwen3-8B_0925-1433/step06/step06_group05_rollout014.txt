import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    initial_clip = np.clip(initial_scaled, 0, 1)
    clip_sum = np.sum(initial_clip)
    if clip_sum > 0:
        initial_clip = initial_clip * target_sum / clip_sum
    h = initial_clip.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    momentum = 0.8
    learning_rate = 0.1
    decay_rate = 0.999
    velocity = np.zeros(n_points)
    perturb_count = 0
    
    for step in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        if current_c5 >= best_c5:
            perturb_count += 1
            if perturb_count % 10 == 0:
                h = h * np.random.choice([0.8, 1.2], size=n_points)
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
            continue
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k])
            if i - k >= 0:
                grad[i] -= h[i - k]
        velocity = momentum * velocity + learning_rate * grad
        h = h + velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if step % 100 == 0:
            learning_rate *= decay_rate
            if learning_rate < 1e-5:
                learning_rate = 1e-5
    return (best_h, best_c5, n_points)
