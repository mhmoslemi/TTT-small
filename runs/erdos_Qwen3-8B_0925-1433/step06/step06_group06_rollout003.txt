import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    npop = 50
    pop = np.vstack([initial_h_values, np.random.rand(npop-1, n_points)]).copy()
    for i in range(npop-1):
        pop[i+1] = np.random.rand(n_points) * target_sum / np.sum(pop[i+1])
    pop = np.clip(pop, 0, 1)
    pop = (pop * target_sum) / np.sum(pop, axis=1, keepdims=True)
    
    best_h = pop[0].copy()
    best_c5 = np.max(np.correlate(pop[0], 1 - pop[0], mode='full')) * dx
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([np.max(np.correlate(h, 1-h, mode='full')) * dx for h in pop])
        elite_idx = np.argsort(scores)[:int(npop*0.2)]
        elites = pop[elite_idx]
        
        candidates = []
        for _ in range(npop - len(elites)):
            p1, p2 = np.random.choice(elite_idx, 2, replace=False)
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([pop[p1][:split], pop[p2][split:]])
            candidates.append(child)
        
        for j in range(len(candidates)):
            mut = np.random.rand(n_points) < 0.1
            candidates[j][mut] = np.random.rand(np.sum(mut))
            candidates[j] = np.clip(candidates[j], 0, 1)
            candidates[j] = (candidates[j] * target_sum) / np.sum(candidates[j])
        
        pop[elite_idx] = elites
        pop[~np.isin(np.arange(npop), elite_idx)] = candidates
        
        curr_scores = np.array([np.max(np.correlate(h, 1-h, mode='full')) * dx for h in pop])
        if np.min(curr_scores) < best_c5:
            best_c5 = np.min(curr_scores)
            best_h = pop[np.argmin(curr_scores)].copy()
    
    return (best_h, best_c5, n_points)
