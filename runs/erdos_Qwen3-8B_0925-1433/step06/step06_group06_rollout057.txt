import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    for iteration in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        lag = np.argmax(corr)
        valid_term1 = np.arange(n_points) + lag < n_points
        term1 = np.zeros(n_points)
        term1[valid_term1] = 1 - h[valid_term1]
        valid_term2 = np.arange(n_points) - lag >= 0
        term2 = np.zeros(n_points)
        term2[valid_term2] = h[valid_term2]
        grad = (term1 - term2) * dx
        learning_rate = 0.1 * (0.99 ** iteration)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * target_sum / total
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, size=n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * target_sum / total
    return (best_h, best_c5, n_points)
