import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initialize population with initial_h_values and random candidates
    pop_size = 30
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h, dx)

    F, CR, elite_frac = 0.8, 0.9, 0.01
    max_iter = 500
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(len(population)):
            a, b, c = population[np.random.choice(len(population), 3, replace=False)]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_c5 = evaluate_erdos_solution(trial, dx)
            orig_c5 = evaluate_erdos_solution(population[i], dx)
            new_pop.append(trial if trial_c5 < orig_c5 else population[i])
        
        # Evaluate new population for elitism
        c5_vals = np.array([evaluate_erdos_solution(h, dx) for h in new_pop])
        best_idx = np.argmin(c5_vals)
        if c5_vals[best_idx] < best_c5:
            best_c5 = c5_vals[best_idx]
            best_h = new_pop[best_idx]
        population = new_pop

    return (best_h, best_c5, n_points)
