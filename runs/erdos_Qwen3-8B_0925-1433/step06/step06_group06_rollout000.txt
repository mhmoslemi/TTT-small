import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    max_iterations = 1000
    learning_rate = 0.2
    decay_rate = 0.999
    min_lr = 1e-5
    start_time = time.time()
    
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        current_c5 = max_val * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        lag = np.argmax(corr)
        valid_term1 = np.arange(n_points) + lag < n_points
        term1 = np.zeros(n_points)
        term1[valid_term1] = 1 - h[valid_term1]
        valid_term2 = np.arange(n_points) - lag >= 0
        term2 = np.zeros(n_points)
        term2[valid_term2] = h[valid_term2]
        grad = (term1 - term2) * dx
        
        # Adaptive learning rate
        learning_rate *= decay_rate
        if learning_rate < min_lr:
            learning_rate = min_lr
        
        new_h = h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:

        new_h = new_h * target_sum / new_sum
        
        # Evaluate new candidate
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_max_val = np.max(new_corr)
        new_c5 = new_max_val * dx
        if new_c5 < current_c5:
            h = new_h.copy()
    
    return (best_h, best_c5, n_points)
