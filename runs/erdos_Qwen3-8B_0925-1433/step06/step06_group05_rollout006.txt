import numpy as np

def compute_c5(h, dx, n_points):

def run(seed=42, budget_s=600):

    decay_rate = 0.9999
    momentum = 0.8
    velocity = np.zeros(n_points)

    for step in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        current_c5 = current_max * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_lag_index = np.argmax(corr)

            if 0 <= j < n_points:

        velocity = momentum * velocity + learning_rate * grad
        h = h + velocity
