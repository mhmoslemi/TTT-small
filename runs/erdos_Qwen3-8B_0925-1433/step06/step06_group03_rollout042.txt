import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    mutation_amp = 0.05
    no_improvement = 0
    max_iterations = 1000

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.uniform(-mutation_amp, mutation_amp, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                no_improvement = 0
            else:
                no_improvement += 1
            if best_c5 < 0.381:
                mutation_amp *= 0.98
            elif best_c5 > 0.382:
                mutation_amp *= 1.02
            else:
                mutation_amp *= 0.99
        else:
            no_improvement += 1
        if no_improvement > 150:
            break
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
