import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    population = [initial_scaled.copy()]
    for _ in range(49):
        if np.random.rand() < 0.5:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = initial_scaled.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5s.append(np.max(corr) * dx)
        worst_idx = np.argmax(current_c5s)
        worst_h = population[worst_idx]
        corr = np.correlate(worst_h, 1 - worst_h, mode='full')
        max_idx = np.argmax(corr)
        lag_values = np.arange(-n_points + 1, n_points)
        worst_k = lag_values[max_idx]
        j = (np.arange(n_points) + worst_k) % n_points
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var * (1 - h_var[j])))
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
        except:
            pass
        new_h = np.clip(h_var.value, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        population[worst_idx] = new_h
        for h_candidate in population:
            c5 = compute_c5(h_candidate, dx, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
