        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        delta = -alpha * max_corr
        gbest += delta
        gbest = np.clip(gbest, 0, 1)
        gbest *= target_sum / np.sum(gbest)
        new_fitness = compute_c5(gbest, dx, n_points)
        if new_fitness < best_c5:
            best_c5 = new_fitness
            best_h = gbest
        iters += 1
    return (best_h, best_c5, n_points)
