import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values * (target_sum / np.sum(initial_h_values))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start = time.time()
    top_ratio = 0.3
    step_sizes = [0.01, 0.05, 0.1, 0.2]
    perturb = 0.1
    for _ in range(1000):
        if time.time() - start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        if np.max(corr) < best_c5:
            best_c5 = np.max(corr)
            best_h = h.copy()
        top_lags = np.argsort(-np.abs(corr))[:int(top_ratio * len(corr))]
        grad = np.zeros(n_points)
        for lag in top_lags:
            k = lag - (n_points - 1)
            i = np.arange(n_points)
            valid_plus = (i + k >= 0) & (i + k < n_points)
            valid_minus = (i - k >= 0) & (i - k < n_points)
            term1 = (1 - h[i + k][valid_plus]) * dx
            term2 = -h[i - k][valid_minus] * dx
            weight = (1 - np.abs(corr[lag]))
            grad[valid_plus] += term1 * weight
            grad[valid_minus] += term2 * weight
        best_alpha = 0
        best_val = np.inf
        for a in step_sizes:
            trial = h - a * grad
            trial = np.clip(trial, 0, 1)
            s = np.sum(trial)
            if s == 0:
                continue
            trial = trial * (target_sum / s)
            val = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
            if val < best_val:
                best_val = val
                best_alpha = a
        if best_alpha != 0:
            h = h - best_alpha * grad
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        if _ % 150 == 0:
            noise = np.random.normal(0, perturb, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    return (best_h, best_c5, n_points)
