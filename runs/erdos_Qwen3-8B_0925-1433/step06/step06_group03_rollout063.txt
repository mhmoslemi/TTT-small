import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate a sine-wave inspired initial guess
    positions = np.linspace(0, 2, n_points)
    sin_wave = 0.5 + 0.5 * np.sin(np.pi * positions)
    h_initial = sin_wave.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    # Objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    # Constraints: ∑h = target_sum
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    
    start_time = time.time()
    
    # First optimization pass
    res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons)
    best_h = res.x
    best_c5 = res.fun
    
    # Iterative refinement
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    return (best_h, best_c5, n_points)
