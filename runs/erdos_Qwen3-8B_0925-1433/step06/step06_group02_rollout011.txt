import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    # Normalize initial_h_values
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    initial_population = []
    for _ in range(50):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_scaled + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_population.append(perturbed)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        pop_c5 = [compute_c5(h, dx, n_points) for h in initial_population]
        worst_idx = np.argmax(pop_c5)
        worst_h = initial_population[worst_idx]
        corr = np.correlate(worst_h, 1 - worst_h, mode='full')
        max_idx = np.argmax(corr)
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[max_idx] * dx
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, worst_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            new_h = res.x
        else:
            new_h = worst_h
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        initial_population[worst_idx] = new_h
        curr_c5 = compute_c5(new_h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = new_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
