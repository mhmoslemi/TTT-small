            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5)/T)
            if np.random.random() < prob:
                h = new_h.copy()
                current_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)
