import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def normalize(h):
        return h * target_sum / np.sum(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [normalize(initial_h_values.copy())]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = normalize(h)
        population.append(h)
    population = np.array(population)
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        elites = population[np.argsort(fitness)[:int(len(population)*0.2)]]
        new_pop = elites.copy()
        remaining = len(population) - len(elites)
        
        for _ in range(remaining):
            t1 = np.random.choice(len(population), 5, replace=False)
            t1_fitness = fitness[t1]
            p1 = population[t1[np.argmin(t1_fitness)]]
            
            t2 = np.random.choice(len(population), 5, replace=False)
            t2_fitness = fitness[t2]
            p2 = population[t2[np.argmin(t2_fitness)]]
            
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            
            if np.random.rand() < 0.1:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] = np.random.rand()
            
            child = normalize(child)
            new_pop = np.vstack([new_pop, child])
        
        population = new_pop[:len(population)]
    
    return (best_h, best_c5, n_points)
