        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        learning_rate *= decay_rate
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
