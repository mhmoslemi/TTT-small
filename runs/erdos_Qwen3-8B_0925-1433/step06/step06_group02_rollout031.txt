import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    population = [initial_scaled.copy()]
    for _ in range(200):  # Reduced iterations for efficiency

            break

        new_candidates = []
        # Random uniform samples
        for _ in range(10):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            new_candidates.append(h)

            h = initial_scaled.copy() + np.random.normal(0, 0.02, n_points)

            h = h * target_sum / np.sum(h)
            new_candidates.append(h)
        # Random segments
        for _ in range(10):
            h = np.zeros(n_points)
            length = np.random.randint(5, 10)
            start = np.random.randint(0, n_points - length)
            h[start:start+length] = 1

            new_candidates.append(h)
        # Circular binary

            h = np.random.choice([0, 1], size=n_points)
            h = h * target_sum / np.sum(h)

        # Evaluate all
        all_candidates = population + new_candidates
        all_candidates = [np.clip(c, 0, 1) for c in all_candidates]
        all_candidates = [c * target_sum / np.sum(c) for c in all_candidates]
        # Evaluate and keep top performers
        candidates = []
        for c in all_candidates:
            c5 = compute_c5(c, dx, n_points)
            if c5 < best_c5 or np.random.random() < 0.2:

        best_h, best_c5 = min([(c[0], c[1]) for c in candidates], key=lambda x: x[1])

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
