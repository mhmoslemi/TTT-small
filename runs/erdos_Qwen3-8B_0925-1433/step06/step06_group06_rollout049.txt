    random_h = np.random.rand(n_points)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_configs = []
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    initial_configs.append(h)
    perturb = np.random.normal(0, 0.1, n_points)
    perturbed = initial_h_values + perturb
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    initial_configs.append(perturbed)
    random_h = np.random.rand(n_points)
    random_h = random_h * target_sum / np.sum(random_h)
    initial_configs.append(random_h)
    best_h = None
    best_c5 = float('inf')
    for config in initial_configs:
        h = np.copy(config)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        start_time = time.time()
        max_iterations = 1000
        learning_rate = 0.2
        for iter in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            max_lags = np.where(corr == max_corr)[0]
            grads = []
            for lag in max_lags:
                valid1 = (np.arange(n_points) + lag) < n_points
                term1 = np.zeros(n_points)
                term1[valid1] = 1 - h[valid1]
                valid2 = (np.arange(n_points) - lag) >= 0

                term2[valid2] = h[valid2]
                grad_k = (term1 - term2) * dx
                grads.append(grad_k)
            grad = np.mean(grads, axis=0) if grads else np.zeros(n_points)
            alpha = learning_rate * (1 / (1 + iter * 0.01))
            new_h = h - alpha * grad
