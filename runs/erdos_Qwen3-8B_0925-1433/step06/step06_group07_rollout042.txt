def run(seed=42, budget_s=600):

        h = h * (target_sum / sum_h)

    start_time = time.time()
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(corr)
        top_indices = np.argsort(corr)[-20:]
        combined_grad = np.zeros(n_points)
        for idx in top_indices:
            lag = idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx
            combined_grad += grad
