from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        perturbed = initial_scaled + np.random.normal(0, 0.05, n_points)

        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    population = np.array(population)[:pop_size]
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    best_h = population[0]
    best_c5 = np.inf
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break

            corrs.append(np.max(corr) * dx)
