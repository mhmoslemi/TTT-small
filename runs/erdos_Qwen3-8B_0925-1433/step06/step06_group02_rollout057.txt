    target_sum = n_points / 2.0
    initial_sum = np.sum(initial_h_values)
    scale_factor = target_sum / initial_sum
    best_h = initial_h_values * scale_factor
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr_full)
        def objective(h_candidate):
            h = np.clip(h_candidate, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h == 0:
                return np.inf
            h_scaled = h * (target_sum / sum_h)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return corr[k_max] * dx
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = differential_evolution(objective, bounds, strategy='best1bin',
                                    popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        h_candidate = res.x
        h = np.clip(h_candidate, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points) + 1e-10
        else:
            h = h * (target_sum / sum_h)
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_h = h
            best_c5 = new_c5
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
