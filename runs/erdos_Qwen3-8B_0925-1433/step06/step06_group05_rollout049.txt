import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    velocity = np.zeros(n_points)
    learning_rate = 0.1
    momentum = 0.8
    perturb_freq = 50
    perturb_amp = 0.05
    start_time = time.time()
    for step in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / sum_h
        if step % perturb_freq == 0:
            perturb = np.random.normal(0, perturb_amp, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros(n_points)
            else:
                h = h * target_sum / sum_h
    return (best_h, best_c5, n_points)
