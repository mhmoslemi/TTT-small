import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_scaled = h * target_sum / np.sum(h)
    best_h = h_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    temperature = 1.0
    cooling_rate = 0.995
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        perturb = np.random.choice(n_points, size=10, replace=False)
        noisy = best_h.copy()
        noisy[perturb] += np.random.normal(0, 0.02, 10)
        noisy = np.clip(noisy, 0, 1)
        noisy = noisy * target_sum / np.sum(noisy)
        current_c5 = compute_c5(noisy, dx, n_points)
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            best_h = noisy.copy()
            best_c5 = current_c5
        temperature *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
