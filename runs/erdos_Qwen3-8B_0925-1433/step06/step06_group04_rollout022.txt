import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = np.clip(initial_h_values, 0, 1)
    h0 = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else np.zeros(n_points)

    pop_size = 100
    pop = []
    # Generate initial diversity
    for _ in range(pop_size):
        if np.random.rand() < 0.3:  # Clones with noise
            h = h0 + np.random.normal(0, 0.05, n_points)
        elif np.random.rand() < 0.4:  # Random vectors
            h = np.random.uniform(0, 1, n_points)
        else:  # Wave-like patterns
            x = np.linspace(0, 2, n_points)
            h = 0.5 * (1 + np.sign(np.sin(np.pi * x)))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    pop = np.array(pop)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h0
    best_c5 = eval_c5(h0)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        scores = np.array([eval_c5(h) for h in pop])
        indices = np.argsort(scores)
        elites = pop[indices[:int(0.2 * pop_size)]]
        new_pop = []
        for _ in range(pop_size - len(elites)):
            a, b = np.random.choice(len(elites), 2, replace=False)
            parent1, parent2 = elites[a], elites[b]
            F = np.random.uniform(0.5, 1.0)
            mutant = parent1 + F * (parent2 - parent1)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_pop.append(mutant)
        combined = np.vstack((elites, new_pop))
        scores_combined = np.array([eval_c5(h) for h in combined])
        pick = np.argsort(scores_combined)[:pop_size]
        pop = combined[pick]

        # Local refinement
        top_10 = pop[:int(0.1 * pop_size)]
        for h in top_10:
            def obj_vec(x):
                return eval_c5(x)
            res = minimize(obj_vec, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

    return (best_h, best_c5, n_points)
