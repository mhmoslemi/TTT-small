            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        velocity = momentum * velocity + learning_rate * grad
        h = h + velocity
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros_like(h)
        else:
            h = h * target_sum / sum_h
        if step % perturb_freq == 0:
            noise = np.random.normal(0, perturb_amp, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros_like(h)
            else:
                h = h * target_sum / sum_h
        learning_rate *= decay_rate
    return (best_h, best_c5, n_points)
