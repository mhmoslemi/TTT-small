import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    population = [h.copy()]
    for _ in range(pop_size - 1):
        if np.random.rand() < 0.3:
            perturb = np.random.normal(0, 0.1, n_points)
            new_h = h + perturb
        elif np.random.rand() < 0.6:
            new_h = h.copy()
            perturb = np.random.normal(0, 0.05, n_points)

            new_h = np.random.rand(n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else new_h
        population.append(new_h)

    best_h = h
