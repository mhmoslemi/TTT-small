    target_sum = n_points / 2.0

    population = []
    initial_h = initial_h_values.copy()
    initial_h_sum = np.sum(initial_h)
    initial_h = initial_h * target_sum / initial_h_sum
    population.append(initial_h)

    for _ in range(5):
        rand_vec = np.random.uniform(0, 1, n_points)
        rand_sum = np.sum(rand_vec)
        rand_vec = rand_vec * target_sum / rand_sum
        population.append(rand_vec)

    population = np.array(population)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    F = 0.8
    CR = 0.9
    iterations = 500
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            mutant = mutant * target_sum / mutant_sum
            original = population[i].copy()
            trial = original.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            corr_trial = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr_trial) * dx
            corr_original = np.correlate(original, 1 - original, mode='full')
            c5_original = np.max(corr_original) * dx
            if c5_trial < c5_original:
                new_population.append(trial)
            else:
                new_population.append(original)
        population = np.array(new_population)
        for candidate in population:
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
    return (best_h, best_c5, n_points)
