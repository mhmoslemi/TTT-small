import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def normalize(h):
        return h * target_sum / np.sum(h)
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        return normalize(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population = [initial_h_values.copy()]

        rand_h = np.random.rand(n_points)

    best_c5 = evaluate(best_h)

    for _ in range(500):
        if elapsed > budget_s:
            break
        start_time = time.time()
        
        # Evaluate all candidates
        fitness = np.array([evaluate(h) for h in population])
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(len(population)*0.2)]

        new_population = elites.copy()
