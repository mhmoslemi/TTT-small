import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(9):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(h) for h in population]
        elite_idx = np.argsort(scores)[:int(0.2 * population_size)]
        elites = [population[i] for i in elite_idx]
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            tournament = np.random.choice(population, size=5)
            winner = tournament[np.argmin([evaluate(h) for h in tournament])]
            parent2 = np.random.choice(population)
            point = np.random.randint(1, n_points - 1)
            child = np.concatenate([winner[:point], parent2[point:]])
            
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] = np.random.rand()
            
            child = np.clip(child, 0, 1)

        for h in population:
