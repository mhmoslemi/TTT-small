import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    population = []
    current_h = initial_h_values.copy()
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    population.append(current_h)
    for _ in range(29):
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = initial_h_values + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        population.append(new_h)
    np.random.shuffle(population)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    F, CR, max_iters = 0.9, 0.8, 300

    for _ in range(max_iters):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            rand1, rand2, rand3 = population[a], population[b], population[c]
            mutant = rand3 + F * (rand1 - rand2)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:

        population = new_pop
