        noise = np.random.normal(0, sigma, len(perturb_indices))
        new_h = current_h.copy()
        new_h[perturb_indices] += noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = compute_c5(new_h, 2.0 / n_points, n_points)
        
        if new_c5 < current_c5:
            current_h, current_c5 = new_h, new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / max(1e-6, T))
            if np.random.rand() < prob:
                current_h, current_c5 = new_h, new_c5
        
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
        
        T *= cooling_rate
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
