    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for iteration in range(200):  # Increase iterations for better exploration
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturbed = np.random.rand(n_points // 2)
        if n_points % 2 == 0:

        perturbed = perturbed * target_sum / np.sum(perturbed)

        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:

        # Local optimization on current best
        res_local = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
