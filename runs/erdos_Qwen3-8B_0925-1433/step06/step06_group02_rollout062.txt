        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
        
        worst_k_idx = np.argmax(corr)
        lag = worst_k_idx - (n_points - 1)
        
        def sub_obj(x):
            shifted = np.roll(x, -lag)
            return np.dot(x, 1 - shifted)
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(sub_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        best_h = res.x
        
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
