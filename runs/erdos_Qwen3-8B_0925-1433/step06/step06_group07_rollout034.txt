import numpy as np
import time

def run(seed=42, budget_s=600):
    import numpy as np
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != target_sum:
        h = h * (target_sum / sum_h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    active_lags = 20
    for iteration in range(500):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        top_lags_indices = np.argsort(corr)[-active_lags:]
        grad = np.zeros(n_points)
        for idx in top_lags_indices:
            lag_k = idx - (n_points - 1)
            for i in range(n_points):
                if 0 <= i + lag_k < n_points:
                    grad[i] += (1 - h[i + lag_k]) * dx
                if 0 <= i - lag_k < n_points:
                    grad[i] -= h[i - lag_k] * dx

        possible_alphas = [0.01, 0.05, 0.1, 0.2]
        best_new_c5 = current_max
        best_trial_h = h.copy()
        for alpha in possible_alphas:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_c5:
                best_new_c5 = new_max
                best_trial_h = trial_h

        if best_new_c5 < current_max:
            h = best_trial_h
        elif best_new_c5 < best_c5:
            h = best_trial_h
            best_c5 = best_new_c5
            best_h = h.copy()

            h = h + noise
            h = np.clip(h, 0, 1)

    return (best_h, best_c5, n_points)
