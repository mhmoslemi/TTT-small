import numpy as np
import time
import cvxpy as cp
from scipy.optimize import Bounds, LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    best_h = initial_h_values * target_sum / initial_sum
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    max_iterations = 100
    iterations = 0
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_index = np.argmax(corr)
        lag = max_corr_index - (n_points - 1)
        
        # Construct matrix A for lag
        matrix_A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                matrix_A[i][j] = 1.0
        
        h = cp.Variable(n_points)
        obj = cp.Maximize(cp.quad_form(h, matrix_A))
        constraints = [cp.sum(h) == target_sum, 0 <= h, h <= 1]
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
        except:
            prob.solve(solver=cp.SCS, eps=1e-6)
        
        if prob.status == 'OPTIMAL':
            new_h = h.value
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
            if best_c5 < 0.38:

        iterations += 1
