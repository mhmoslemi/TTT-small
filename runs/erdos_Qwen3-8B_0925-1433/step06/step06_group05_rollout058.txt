            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] = (1 - h[j])
                else:
                    grad[i] = 0.0
            velocity = momentum * velocity + learning_rate * grad
            h = h + velocity
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros(n_points)
            else:
                h = h * target_sum / sum_h
        if np.random.random() < 0.05:
            perturb = np.random.normal(0, 0.3, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros(n_points)
            else:
                h = h * target_sum / sum_h
    return (best_h, best_c5, n_points)
