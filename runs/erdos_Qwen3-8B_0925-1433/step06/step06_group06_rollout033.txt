import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    max_iterations = 600
    learning_rate = 0.1
    no_improvement = 0
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        top_lags = np.argsort(corr)[-5:]
        grad = np.zeros(n_points)
        for lag in top_lags:
            valid_term1 = np.arange(n_points) + lag < n_points
            term1 = np.zeros(n_points)
            term1[valid_term1] = 1 - h[valid_term1]
            valid_term2 = np.arange(n_points) - lag >= 0
            term2 = np.zeros(n_points)
            term2[valid_term2] = h[valid_term2]
            grad += (term1 - term2) * dx
        grad /= len(top_lags)
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:

            best_h = h.copy()

            if no_improvement > 50:
                learning_rate *= 0.5
                no_improvement = 0
        else:
            no_improvement = 0
    return (best_h, best_c5, n_points)
