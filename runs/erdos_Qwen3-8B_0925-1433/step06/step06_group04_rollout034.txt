def run(seed=42, budget_s=600):

    population.append(initial_h)
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        population.append(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def process_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        return h
    
    best_c5 = float('inf')
    best_h = initial_h
    
    start_time = time.time()

        # Evaluate population

        idx = np.argsort(c5_values)
        top_idx = idx[:int(population_size * 0.2)]
        top_pop = [population[i] for i in top_idx]
        
        new_pop = []
