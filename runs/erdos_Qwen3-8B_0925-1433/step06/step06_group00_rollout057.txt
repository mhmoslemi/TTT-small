import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum_full = n_points / 2
    
    # Symmetrize initial guess
    sym_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            sym_initial[i] = initial_h_values[i]
            sym_initial[n_points - 1 - i] = initial_h_values[i]
        elif n_points % 2 == 1 and i == n_points // 2:
            sym_initial[i] = initial_h_values[i]
    
    # Prepare variables for optimization
    if n_points % 2 == 0:
        half = n_points // 2
        x_initial = sym_initial[:half]
        target_sum_x = target_sum_full / 2
        constraint_matrix = np.ones(half)
    else:
        mid = n_points // 2
        x_initial = sym_initial[:mid + 1]
        target_sum_full = n_points / 2
        constraint_matrix = np.zeros(mid + 1)
        for i in range(mid + 1):
            constraint_matrix[i] = 2
        constraint_matrix[mid] = 1
        target_sum_x = (target_sum_full + x_initial[mid]) / 2
    
    # Initial solution
    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    best_x = x_initial.copy()
    
    def build_full_h(x):
        if n_points % 2 == 0:
            full = np.zeros(n_points)
            full[:half] = x
            full[half:] = x[::-1]
            return full
        else:
            full = np.zeros(n_points)
            full[:mid] = x[:mid]
            full[mid] = x[mid]
            full[mid+1:] = x[:mid][::-1]
            return full
    
    def obj(x):
        h = build_full_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Main optimization loop
    start_time = time.time()
    for _ in range(200):  # 200 iterations
        if time.time() - start_time > budget_s:
            break
        
        # Optimize
        bounds = Bounds(0, 1)
        cons = LinearConstraint(constraint_matrix, [target_sum_x], [target_sum_x])
        res = minimize(obj, best_x, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            curr_x = res.x
            curr_h = build_full_h(curr_x)
            curr_c5 = obj(curr_x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_x = curr_x.copy()
                best_h = curr_h.copy()
        
        # Perturb and optimize
        perturbed = best_x + np.random.normal(0, 0.015, len(best_x))
        perturbed = np.clip(perturbed, 0, 1)
        cons_pert = LinearConstraint(constraint_matrix, [target_sum_x], [target_sum_x])
        res_p = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons_pert)
        if res_p.success:
            curr_x_p = res_p.x
            curr_h_p = build_full_h(curr_x_p)
            curr_c5_p = obj(curr_x_p)

                best_c5 = curr_c5_p
                best_x = curr_x_p.copy()
                best_h = curr_h_p.copy()
    
    return (best_h, best_c5, n_points)
