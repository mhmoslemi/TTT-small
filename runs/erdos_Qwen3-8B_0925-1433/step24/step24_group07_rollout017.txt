from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, max_idx):
        K = max_idx - (n_points - 1)
        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - h[pos]) - h[neg]
        return grad

    def optimize_with_bounds(method='L-BFGS-B'):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, best_h, method=method, bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = compute_gradient(h, max_idx)
        alpha = 0.05 * (1 - (time.time() - start_time) / budget_s)
        h -= alpha * grad

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
