import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def refine(x):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, x, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    def perturb(x):
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(x + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        return res.x

    seeds = [h.copy()]
    x = np.linspace(0, 2, n_points)
    seeds.append(0.5 + 0.5 * np.sin(np.pi * x) * 0.5)
    seeds.append(np.where((np.arange(n_points) % 2) == 0, 1, 0).astype(float))

    seeds.append(np.random.choice([0, 1], size=n_points, p=[0.75, 0.25]))

    for seed in seeds:
        h = seed * target_sum / np.sum(seed)
        best_h_candidate = h.copy()
        best_c5_candidate = np.max(np.correlate(h, 1 - h, mode='full')) * dx
