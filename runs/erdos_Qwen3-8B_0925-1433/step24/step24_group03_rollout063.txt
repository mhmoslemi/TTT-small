import time

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.3807)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        for i in range(n_points):
            shifted = (i - k_max) % n_points
            grad[i] = (1 - h[(i + k_max) % n_points]) - h[shifted]
        return grad

    start_time = time.time()
