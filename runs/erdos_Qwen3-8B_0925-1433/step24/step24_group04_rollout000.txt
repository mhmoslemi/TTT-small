import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)
        g = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k_max
            term1 = 1 - x[i_plus] if 0 <= i_plus < n_points else 0
            i_minus = i - k_max
            term2 = x[i_minus] if 0 <= i_minus < n_points else 0
            g[i] = term1 - term2
        return g

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        perturb = np.random.normal(0, 0.05, n_points)
        h = np.clip(res.x + perturb, 0, 1)
        h = h * target_sum / np.sum(h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
