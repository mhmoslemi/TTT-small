import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def evaluate_c5(h):

        return np.max(corr) * dx

    population = [h.copy()]
    pop_size = 50
    gen_limit = 200
    mutation_rate = 0.1
    start_time = time.time()

    for _ in range(gen_limit):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(ind) for ind in population]
        selected = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(range(pop_size), size=5, replace=False)
            idx = idxs[np.argmin(fitness[idxs])]
            selected.append(population[idx])
        new_pop = []
        for i in range(pop_size // 2):
            p1, p2 = selected[i], selected[i+1]
            c1, c2 = (p1 + np.random.rand(n_points) * p2) / 2, (p2 + np.random.rand(n_points) * p1) / 2
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    c1[j] += np.random.normal(0, 0.05)
                    c2[j] += np.random.normal(0, 0.05)

        population = new_pop
        for ind in population:
            curr = evaluate_c5(ind)
            if curr < best_c5:
