import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    particles = []
    for i in range(30):
        h = np.array(initial_h_values, dtype=float)
        if i > 0:
            h += np.random.normal(0, 0.15, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        np.random.shuffle(h)
        particles.append(h)
    best_h = particles[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    omega, phi1, phi2 = 0.7, 1.5, 1.5
    velocities = [np.zeros(n_points) for _ in range(len(particles))]
    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        for i, h in enumerate(particles):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        pbest = [h.copy() for h in particles]
        gbest = best_h.copy()
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + phi1 * r1 * (pbest[i] - particles[i]) + phi2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        particles.sort(key=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx)
        for i in range(int(len(particles)*0.2)):
            h = np.array(initial_h_values, dtype=float)
            h += np.random.normal(0, 0.15, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            np.random.shuffle(h)
            particles[i] = h
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    candidates = [h.copy() for h in particles]
    candidates.sort(key=lambda h: obj(h))
    for h in candidates[:5]:

        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons)
        if res.success:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
