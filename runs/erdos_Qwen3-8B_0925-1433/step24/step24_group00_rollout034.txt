    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.clip(h_initial, 0, 1)
    sum_initial = np.sum(h_initial)
    if abs(sum_initial - target_sum) > 1e-8:
        h_initial = h_initial * target_sum / sum_initial

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-5,
        mutation=0.5,
        recombination=0.9,
        seed=seed,
        polish=True
    )

    best_h = result.x
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
