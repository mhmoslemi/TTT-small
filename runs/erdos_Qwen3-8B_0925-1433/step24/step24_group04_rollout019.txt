        for ind in population:
            current_c5 = compute_c5(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5

    # Refinement
    sorted_pop = population[np.argsort(compute_c5(population))]
    for i in range(5):
        if time.time() - start_time > budget_s:
            break
        h = sorted_pop[i]
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_refined = res.x
            current_c5 = compute_c5(h_refined)
            if current_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
