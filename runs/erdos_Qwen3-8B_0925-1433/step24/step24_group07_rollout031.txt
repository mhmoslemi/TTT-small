import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    pop_size = 30
    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append((c5, h))
        fitness.sort()
        selected = [h for c5, h in fitness[:int(len(fitness)*0.2)]]
        new_pop = []
        for _ in range(pop_size):
            if len(selected) < 2:
                break
            p1, p2 = np.random.choice(selected, 2, replace=False)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            mutate_idx = np.random.choice(n_points, size=min(int(n_points/10), 10), replace=False)
            child[mutate_idx] += np.random.normal(0, 0.01, size=len(mutate_idx))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        for idx in range(len(new_pop)):
            x0 = new_pop[idx]
            def obj(x):
                return compute_c5(x)
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
            new_pop[idx] = res.x
        population = new_pop
        for c5, h in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
