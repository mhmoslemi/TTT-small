                if 0 <= i_minus < n_points:
                    term2 = h[i_minus]
                else:
                    term2 = 0.0
                grad[i] = term1 - term2
            h += 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        res = minimize(lambda x: np.max(np.correlate(x, 1-x, mode='full')) * dx, h, method='SLSQP',
                       bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (np.clip(best_h, 0, 1), best_c5, n_points)
