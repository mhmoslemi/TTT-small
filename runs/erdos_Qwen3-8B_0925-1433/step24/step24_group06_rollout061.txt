import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def optimize_pattern(pattern):
        pattern = np.clip(pattern * target_sum / np.sum(pattern), 0, 1)
        def objective(x):
            x = np.clip(x * target_sum / np.sum(x), 0, 1)
            return compute_c5(x)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(objective, pattern, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        return result.x if result.success else None
    
    def perturb(h):
        return np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1) * target_sum / np.sum(h)
    
    initial_patterns = [
        np.array(initial_h_values, dtype=float),
        0.5 + 0.5 * np.sin(np.pi * x),
        np.where((x % 1) < 0.4, 1.0, 0.0),
        0.5 + 0.5 * np.abs(2 * (x % 1) - 1),
        np.zeros_like(x) + 1.0 / (n_points // 4)
    ]
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for p in initial_patterns:
        if time.time() - start_time > budget_s:
            break
        h = optimize_pattern(p)
        if h is not None:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = perturb(best_h)
        h = optimize_pattern(h)
        if h is not None and compute_c5(h) < best_c5:
            best_h = h
            best_c5 = compute_c5(h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
