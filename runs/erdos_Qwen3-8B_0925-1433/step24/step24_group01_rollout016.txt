from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    population = [h.copy()]
    for _ in range(8):
        pert = np.random.normal(0, 0.05, n_points)
        h = np.clip(initial_h_values + pert, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for i in range(9):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='COBYLA', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-7)
            if res.success:

        for cand in population:
