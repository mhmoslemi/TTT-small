import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Clip and scale initial_h
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    # Initialize population
    population_size = 20
    particles = [initial_h.copy() for _ in range(population_size)]
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # PSO parameters
    inertia = 0.7
    inertia_decay = (0.7 - 0.4) / 1000
    cognitive = 1.5
    social = 1.5
    max_iter = 1000
    converge_threshold = 25
    
    # Personal best and global best
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    global_best = best_h.copy()
    global_best_fitness = best_c5
    
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update inertia
        inertia = 0.7 - inertia_decay * iter
        
        # Update velocities and positions
        for i in range(len(particles)):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            # Update velocity
            vel = velocities[i] * inertia + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            velocities[i] = vel
            # Update position
            new_pos = particles[i] + vel
            # Clip and scale
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = np.sum(new_pos)
            if current_sum != 0:
                new_pos = new_pos * target_sum / current_sum
            particles[i] = new_pos.copy()
        
        # Evaluate fitness and update personal best
        for i in range(len(particles)):
            fitness = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
        
        # Update global best
        for i in range(len(particles)):
            if personal_best_fitness[i] < global_best_fitness:
                global_best = personal_best[i].copy()
                global_best_fitness = personal_best_fitness[i]
        
        # Check for convergence
        if iter > converge_threshold and global_best_fitness == best_c5:
            break
    
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
