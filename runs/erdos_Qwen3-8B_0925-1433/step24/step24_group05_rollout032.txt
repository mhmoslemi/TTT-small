import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def generate_patterns():
        patterns = [np.array(initial_h_values, dtype=float)]
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
        patterns.append(sin_h)
        square_wave = 0.5 + 0.5 * np.sign(np.sin(np.pi * x))
        patterns.append(square_wave)
        random_h = np.random.uniform(0.3, 0.7, n_points)
        patterns.append(random_h)
        triangle_wave = 0.5 + 0.5 * (2 * (x - 0.5) * (x < 1) - 1)
        patterns.append(triangle_wave)
        return patterns

    patterns = generate_patterns()
    for seed in patterns:
        seed = seed * target_sum / np.sum(seed)
        h = seed.copy()
        for _ in range(500):
            grad = compute_gradient(h)
            learning_rate = 0.02 * (1 / (1 + 0.001 * _))
            h += learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        res = minimize(obj, h, method='SLSQP', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
