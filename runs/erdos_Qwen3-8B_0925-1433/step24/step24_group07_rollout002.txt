import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        corr = np.correlate(x * target_sum / np.sum(x), 1 - x * target_sum / np.sum(x), mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n_points - 1)
        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - h[pos]) - h[neg]
        return grad

    start_time = time.time()
    perturbation_scale = 0.1
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        delta = np.random.normal(0, perturbation_scale, n_points)
        tentative_h = h + delta
        tentative_h = np.clip(tentative_h, 0, 1)
        tentative_h = tentative_h * target_sum / np.sum(tentative_h)
        current_c5 = np.max(np.correlate(tentative_h, 1 - tentative_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = tentative_h.copy()
            bounds = Bounds(0, 1)
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(obj, tentative_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            best_h = res.x
            best_c5 = np.max(np.correlate(best_h * target_sum / np.sum(best_h), 1 - best_h * target_sum / np.sum(best_h), mode='full')) * dx
        perturbation_scale *= 0.995
    return (np.clip(best_h, 0, 1), best_c5, n_points)
