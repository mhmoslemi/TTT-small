import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, max_idx):
        K = max_idx - (n_points - 1)
        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - h[pos]) - h[neg]
        return grad

    start_time = time.time()
    scale = 0.1
    perturb_steps = 50
    local_steps = 10

    for _ in range(perturb_steps):
        if time.time() - start_time > budget_s:
            break
        delta = np.random.normal(0, scale, n_points)
        tentative_h = h + delta
        tentative_h = np.clip(tentative_h, 0, 1)
        tentative_h = tentative_h * target_sum / np.sum(tentative_h)
        current_c5 = compute_c5(tentative_h)
        if current_c5 < best_c5:
            bounds = Bounds(0, 1)
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            try:
                res = minimize(obj, tentative_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=local_steps)
                new_h = res.x
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
            except:
                pass
        scale *= 0.99

    for _ in range(local_steps):
        if time.time() - start_time > budget_s:
            break
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, maxiter=10)
            new_h = res.x
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        except:
            pass

    return (np.clip(best_h, 0, 1), best_c5, n_points)
