import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize population
    initial_h = np.array(initial_h_values, dtype=float)
    population_size = 40

        population.append(h)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for _ in range(200):  # Evolutionary iterations
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            trial = np.zeros_like(population[i])
            for j in range(n_points):
                if np.random.rand() < 0.7:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            trial = np.clip(trial, 0, 1)
            new_population.append(trial)
        new_population = np.array(new_population)
        for i in range(population_size):
            if evaluate_c5(new_population[i]) < evaluate_c5(population[i]):

                if evaluate_c5(population[i]) < best_c5:
                    best_h = population[i].copy()
                    best_c5 = evaluate_c5(population[i])

    # Local refinement with L-BFGS-B
    def obj(x):
        return evaluate_c5(x)
