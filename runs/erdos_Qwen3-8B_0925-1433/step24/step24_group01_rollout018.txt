import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population with initial_h_values and random samples
    def init_population(size=20):
        pop = np.copy(initial_h_values)
        pop = np.vstack((pop, np.random.rand(size-1, n_points)))
        pop = np.clip(pop, 0, 1)
        for i in range(size):
            pop[i] = pop[i] * target_sum / np.sum(pop[i])
        return pop
    
    pop = init_population()
    best_h = pop[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def eval_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def update_velocity(v, pos, pbest, gbest, omega, phi_p, phi_g):
        r1, r2 = np.random.rand(2, n_points)
        v = omega * v + phi_p * r1 * (pbest - pos) + phi_g * r2 * (gbest - pos)
        return v
    
    omega_start, omega_end = 0.7, 0.4
    phi_p, phi_g = 1.5, 1.5
    max_iter = 1000
    iter = 0
    time_start = time.time()
    
    # Initial personal and global best
    pbest = pop.copy()
    pbest_fitness = np.array([eval_fitness(h) for h in pbest])
    gbest = pop[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    
    velocities = np.random.rand(20, n_points) * 0.01
    
    while iter < max_iter and time.time() - time_start < budget_s:
        omega = omega_start - (omega_start - omega_end) * iter / max_iter
        for i in range(20):
            v = update_velocity(velocities[i], pop[i], pbest[i], gbest, omega, phi_p, phi_g)
            new_pos = pop[i] + v
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * target_sum / np.sum(new_pos)
            new_fitness = eval_fitness(new_pos)
            if new_fitness < pbest_fitness[i]:
                pbest[i] = new_pos
                pbest_fitness[i] = new_fitness
            if new_fitness < gbest_fitness:
                gbest = new_pos
                gbest_fitness = new_fitness
            pop[i] = new_pos
            velocities[i] = v
        iter += 1
    
    best_h = gbest
    best_c5 = gbest_fitness
    return (best_h, best_c5, n_points)
