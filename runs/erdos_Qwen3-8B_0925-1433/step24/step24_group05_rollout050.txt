import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial population
    initial_population = []
    h0 = np.array(initial_h_values, dtype=float)
    initial_population.append(normalize(h0))
    x = np.linspace(0, 2, n_points)
    sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_population.append(normalize(sin_h))
    random_h = np.random.uniform(0, 1, n_points)
    initial_population.append(normalize(random_h))
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = h0 + noise
        perturbed = np.clip(perturbed, 0, 1)
        initial_population.append(normalize(perturbed))
    for _ in range(30):
        random_h = np.random.uniform(0, 1, n_points)
        initial_population.append(normalize(random_h))
    while len(initial_population) < 50:
        random_h = np.random.uniform(0, 1, n_points)
        initial_population.append(normalize(random_h))
    population = initial_population

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    # Genetic algorithm
    for _ in range(100):
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        elite_size = int(len(population) * 0.05)
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        new_population = elites[:]  # Copy elites
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(elite_indices, 2, replace=False)
            child = (population[p1] + population[p2]) / 2
            mutation = np.random.normal(0, 0.2, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        population = new_population

    # Local optimization
    res = minimize(compute_c5, best_h, method='SLSQP', bounds=[(0, 1)]*n_points,
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
