import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
        n_points = len(initial_h)
    else:
        n_points = 45
        initial_h = np.random.rand(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clamp_and_scale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    initial_h_clamped = clamp_and_scale(initial_h.copy())

    population_size = 20
    particles = []
    velocities = []
    personal_bests = []
    global_best = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals():
        particles.append(initial_h_clamped)
        velocities.append(np.zeros(n_points))
        personal_bests.append(initial_h_clamped.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = clamp_and_scale(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
            personal_bests.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = clamp_and_scale(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
            personal_bests.append(h.copy())

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for i, h in enumerate(particles):
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            global_best = h.copy()

    inertia_weight = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter = 1000
    start_time = time.time()

    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        inertia_weight = 0.7 - 0.3 * it / max_iter
        for i in range(population_size):
            h = particles[i]
            vel = velocities[i]
            pbest = personal_bests[i]
            r1, r2 = np.random.rand(2)
            vel += inertia_weight * vel
            vel += cognitive * r1 * (pbest - h)
            vel += social * r2 * (global_best - h)
            new_h = h + vel
            new_h = clamp_and_scale(new_h)
            particles[i] = new_h
            velocities[i] = vel
            c5 = compute_c5(new_h)
            if c5 < best_c5:
                best_c5 = c5
                global_best = new_h.copy()
            corr_pbest = np.correlate(pbest, 1 - pbest, mode='full')
            c5_pbest = np.max(corr_pbest) * dx
            if c5 < c5_pbest:
                personal_bests[i] = new_h.copy()

    final_h = clamp_and_scale(global_best)
    return (final_h, best_c5, n_points)
