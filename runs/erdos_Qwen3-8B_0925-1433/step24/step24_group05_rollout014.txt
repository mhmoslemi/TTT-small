    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                term1 = 1 - h[i + lag]
            else:
                term1 = 0.0
            if 0 <= i - lag < n_points:
                term2 = h[i - lag]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    learning_rate = 0.05
    for _ in range(500):
        grad = compute_gradient(h)
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    for _ in range(200):
        grad = compute_gradient(best_h)
        best_h -= 0.01 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
