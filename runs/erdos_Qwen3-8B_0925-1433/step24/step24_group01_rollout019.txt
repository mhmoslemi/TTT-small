import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    n_particles = 20
    max_iter = 1000
    c1, c2 = 1.5, 1.5
    omega = 0.7
    omega_decay = 0.001

    particles = []
    for _ in range(n_particles):
        pert = np.random.normal(0, 0.1, n_points)
        new_h = np.array(initial_h_values) + pert
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        particles.append(new_h)
    particles = np.array(particles)

    personal_best = [p.copy() for p in particles]
    global_best = particles[0].copy()
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx
    for p in particles:
        curr_c5 = np.max(np.correlate(p, 1 - p, mode='full')) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            global_best = p.copy()

    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            curr_c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if curr_c5 < np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full')) * dx:
                personal_best[i] = particles[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    global_best = particles[i].copy()
        omega = max(0.4, omega - omega_decay)

    return (global_best, best_c5, n_points)
