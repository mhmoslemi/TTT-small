    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()

    result = minimize(compute_c5, h, method='SLSQP', bounds=bounds, constraints=cons, jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
    best_h, best_c5 = result.x, compute_c5(result.x, dx)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result_perturb = minimize(compute_c5, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        current_c5 = compute_c5(result_perturb.x, dx)
        if current_c5 < best_c5:
            best_h, best_c5 = result_perturb.x, current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
