import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h_curr = x * target_sum / np.sum(x)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * dx

    def local_minimize(x):
        res = minimize(
            obj,
            x,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-7,
            options={'maxiter': 100}
        )
        if res.success:
            return res.x
        return x

    population = [h.copy() for _ in range(50)]
    for p in population:
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)

    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            # Directed mutation: perturb best_h
            mutant = best_h + np.random.normal(0, 0.1, n_points)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
        population = new_population

        # Local refine top 10
        for i in range(10):
            population[i] = local_minimize(population[i])
        
        # Evaluate and select
        evaluated = [(obj(p), p) for p in population]
        evaluated.sort()
        current_c5, current_p = evaluated[0]
        if current_c5 < best_c5:
            best_h = current_p.copy()
            best_c5 = current_c5
        gen += 1
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
