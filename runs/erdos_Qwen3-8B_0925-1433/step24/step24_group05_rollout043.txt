import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h_init = np.array(initial_h_values, dtype=float)
    h_init = h_init * target_sum / np.sum(h_init)  # Normalize
    pop_size = 50
    generations = 100
    mutation_rate = 0.2
    elite_ratio = 0.2
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx
    
    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_initial_population():
        pop = []
        pop.append(h_init)
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
        pop.append(sin_h)
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = h_init + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        pop.append(perturbed)
        for _ in range(pop_size - 3):
            rand = np.random.uniform(0, 1, n_points)
            rand = rand * target_sum / np.sum(rand)
            pop.append(rand)
        return pop
    
    pop = generate_initial_population()
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
        scores = [eval_c5(h) for h in pop]
        elite_size = int(elite_ratio * len(pop))
        elite = [h for _, h in sorted(zip(scores, pop), key=lambda x: x[0])[:elite_size]]
        new_pop = elite[:]
        for _ in range(len(pop) - elite_size):
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = (p1 + np.random.rand(n_points) * (p2 - p1)) / 2
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.choice(n_points, 5, replace=False)
                child[mutate_idx] += np.random.uniform(-0.1, 0.1, size=5)
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
        for h in pop:
            if eval_c5(h) < best_c5:
                best_c5 = eval_c5(h)
                best_h = h.copy()

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')

        return np.sum(x) - target_sum
