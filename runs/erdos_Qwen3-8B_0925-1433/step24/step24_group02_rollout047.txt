        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if abs(current_c5 - best_c5) < tolerance:
            break
        if time.time() - start_time > budget_s:
            break
    h_values = best_h.copy()
    return (h_values, best_c5, n_points)
