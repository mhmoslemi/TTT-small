    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')

        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k_max
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - k_max
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad
    
    def de_step(population, elite_ratio=0.2):
        elites = np.argsort(obj(population))[:int(len(population)*elite_ratio)]

            new_pop.append(mutant)

    population = [h.copy()]
