import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)
    learning_rate = 0.1
    decay_rate = 0.995
    perturb_interval = 100
    perturb_strength = 0.05
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points, dtype=float)
        for i in range(n_points):
            j = (i + k_max) % n_points
            l = (i - k_max) % n_points
            grad[i] = (1. - h[j]) - h[l]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        current_c5 = np.max(corr) * dx

            best_h = h.copy()
        learning_rate *= decay_rate
