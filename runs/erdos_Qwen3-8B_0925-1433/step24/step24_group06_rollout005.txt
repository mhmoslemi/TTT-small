import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        x = np.linspace(0, 2, n_points)
        patterns = []
        # Sine wave
        patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
        # Square wave
        patterns.append(np.where((x % 1) < 0.5, 1, 0))
        # Triangle wave
        patterns.append(np.abs(x - 1) % 1 * 2 - 1)
        # Random binary
        patterns.append(np.random.rand(n_points))
        # Initial pattern
        patterns.append(initial_h_values)
        return patterns

    patterns = generate_patterns()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf

    def evaluate(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for p in patterns:
        h = p.copy()
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    def optimize(h):
        def objective(x):
            x = x * target_sum / np.sum(x)
            return evaluate(x)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        return minimize(objective, h, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons)

    for _ in range(3):
        if time.time() > time.time() + budget_s:
            break
        h = best_h.copy()
        res = optimize(h)
        if res.success:
            best_h, best_c5 = res.x, evaluate(res.x)
        h_rand = np.random.rand(n_points) * 0.5
        h_rand = h_rand * target_sum / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand_c5 = evaluate(h_rand)
        if h_rand_c5 < best_c5:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
