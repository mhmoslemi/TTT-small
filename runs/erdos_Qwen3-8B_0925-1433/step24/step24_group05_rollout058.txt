    initial_pop = []

    # Generate initial population with multiple patterns
    def generate_pattern():
        x = np.linspace(0, 2, n_points)
        if np.random.rand() < 0.3:
            return np.array(initial_h_values, dtype=float)
        elif np.random.rand() < 0.2:
            return 0.5 + 0.5 * np.sin(np.pi * x)
        elif np.random.rand() < 0.2:
            return np.random.uniform(0, 1, n_points)
        else:
            return np.random.normal(0.5, 0.2, n_points)

    for _ in range(50):
        h = generate_pattern()
        h = h * target_sum / np.sum(h)
        initial_pop.append(h)

    best_c5 = float('inf')
    best_h = np.array(initial_h_values, dtype=float)

    def evaluate(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    population = np.array(initial_pop)
    for generation in range(100):
        scores = np.array([evaluate(h) for h in population])
        elite_indices = np.argsort(scores)[:5]
        elites = population[elite_indices]

        for _ in range(len(population)):
            parent1, parent2 = np.random.choice(elites, 2, replace=False)
            child = (parent1 + np.random.rand(n_points) * (parent2 - parent1)) / 2
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = np.array(new_population)
        if generation % 10 == 0:
            res = minimize(lambda x: evaluate(x), population[0], method='SLSQP',
                           bounds=[(0, 1) for _ in range(n_points)],

            if res.success:

    final_h = np.clip(best_h, 0, 1)
