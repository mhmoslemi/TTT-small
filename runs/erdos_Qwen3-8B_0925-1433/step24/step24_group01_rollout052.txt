    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def compute_smooth_c5(h, T):
        corr = np.correlate(h, 1 - h, mode='full')
        terms = corr * dx
        exp_terms = np.exp(T * terms)
        sum_exp = np.sum(exp_terms)
        smooth_c5 = (1.0 / T) * np.log(sum_exp)
        return smooth_c5
    
    def objective(x, T):
        h = x.copy()
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        return compute_smooth_c5(h, T)
    
    T = 1.0
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        def objective_with_T(x):
            return objective(x, T)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        x0 = h.copy()
        res = minimize(objective_with_T, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            h = res.x.copy()
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
            current_c5 = compute_smooth_c5(h, T)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        T *= 1.1
        if T > 100:
            T = 100
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h, final_c5, n_points)
