from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_points = [np.array(initial_h_values, dtype=float)]

        h_rand = h_rand * target_sum / np.sum(h_rand)
        initial_points.append(h_rand)

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)

        grad = np.zeros_like(x)
        for i in range(n_points):
