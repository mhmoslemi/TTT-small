import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(45)
        h[:22] = 1.0
        h = h * 45 / 2 / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def obj(x):
        hx = x * target_sum / np.sum(x)
        corr = np.correlate(hx, 1 - hx, mode='full')
        T = 10.0
        smooth_vals = T * (corr * dx)
        return (1 / T) * np.log(np.sum(np.exp(smooth_vals)))
    
    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    
    start = time.time()
    for _ in range(50):
        if time.time() - start > budget_s:
            break
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
