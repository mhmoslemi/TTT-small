import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def generate_patterns():
        patterns = [np.array(initial_h_values, dtype=float)]
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
        patterns.append(sin_h)
        noise = np.random.normal(0, 0.05, n_points)
        jittered = np.array(initial_h_values, dtype=float) + noise
        patterns.append(np.clip(jittered, 0, 1))
        patterns.append(0.5 * np.ones(n_points))  # Uniform
        patterns.append(0.4 + 0.2 * np.sin(np.pi * x))
        return patterns
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k >= 0:
                if i + k < n_points:
                    grad[i] = 1 - h[i + k]
                else:
                    grad[i] = 0.0
            else:
                k_neg = -k
                if i + k_neg < n_points:
                    grad[i] = -h[i + k_neg]
                else:
                    grad[i] = 0.0
        return grad
    
    patterns = generate_patterns()
    
    for pattern in patterns:
        h = pattern.copy() * target_sum / np.sum(pattern)
        h = np.clip(h, 0, 1)
        for _ in range(500):
            grad = compute_gradient(h)
            h += 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        res = minimize(obj, h, method='SLSQP', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
