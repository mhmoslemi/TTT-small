        while len(new_pop) < len(population):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = np.random.rand() * p1 + (1 - np.random.rand()) * p2
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1) * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
        for h in refined:
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    return (np.clip(best_h, 0, 1), best_c5, n_points)
