import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    time_start = time.time()

    # Generate alternative initial configurations
    def generate_alt_h():
        alt = np.roll(h, 1)
        alt = alt * target_sum / np.sum(alt)
        return alt

    def generate_random_h():
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        return h_rand

    alt_h = [h, generate_alt_h(), generate_random_h()]
    for init in alt_h:
        corr = np.correlate(init, 1 - init, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(init)
        for i in range(n_points):
            i_plus = (i + k_max) % n_points
            i_minus = (i - k_max) % n_points
            grad[i] = (1 - init[i_plus]) - init[i_minus]
        h_perturb = init - 0.1 * grad
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        current_c5 = np.max(np.correlate(h_perturb, 1 - h_perturb, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h_perturb.copy(), current_c5

    # Refinement with L-BFGS-B
    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    while time.time() - time_start < budget_s:
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, 
                       best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        # Add random perturbation
        if np.random.rand() < 0.2:
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturb = np.clip(best_h + perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            res_perturb = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, 
                                   h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res_perturb.success and res_perturb.fun < best_c5:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
