import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize(h_initial):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(lambda x: compute_c5(x * target_sum / np.sum(x)), h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        optimized = optimize(perturbed)
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_h, best_c5 = optimized, current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
