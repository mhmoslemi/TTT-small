        max_idx = np.argmax(corr)
        return compute_gradient(x, max_idx)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = Bounds(0, 1)

    start_time = time.time()
    eta = 0.1
    improvement_count = 0

    while time.time() - start_time < budget_s:
        delta = np.random.normal(0, eta, n_points)
        tentative_h = h + delta
        tentative_h = np.clip(tentative_h, 0, 1)
        tentative_h = tentative_h * target_sum / np.sum(tentative_h)
        corr = np.correlate(tentative_h, 1 - tentative_h, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            result = minimize(objective, tentative_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad_func, tol=1e-6)
            optimized_h = result.x
            optimized_c5 = objective(optimized_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h.copy()
                best_c5 = optimized_c5
                improvement_count = 0
            else:
                improvement_count += 1
        else:
            improvement_count += 1

        h = tentative_h
        if improvement_count >= 5:
            eta *= 0.9
            improvement_count = 0

    return (np.clip(best_h, 0, 1), best_c5, n_points)
