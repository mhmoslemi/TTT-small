import numpy as np
from scipy.optimize import minimize, Bounds
import time
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h_val):
        return np.max(np.correlate(h_val, 1 - h_val, mode='full')) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()
    for _ in range(5):  # Multi-phase optimization
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        best_h = res.x
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

    for _ in range(10):  # Gradient refinement
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        K = max_idx - (n_points - 1)
        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - best_h[pos]) - best_h[neg]
        h_step = best_h - 0.05 * grad
        h_step = np.clip(h_step, 0, 1)
        h_step = h_step * target_sum / np.sum(h_step)
        current_c5 = compute_c5(h_step)
        if current_c5 < best_c5:
            best_h, best_c5 = h_step.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
