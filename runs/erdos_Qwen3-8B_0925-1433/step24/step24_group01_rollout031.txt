    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        for i in range(9):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='SLSQP', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-7)
            if res.success:
                new_h = res.x
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                population.append(new_h)
                population.pop(np.argmin([evaluate(h) for h in population]))
        for cand in population:
            curr_c5 = evaluate(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
