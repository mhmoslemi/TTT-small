from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    population = [h.copy()]
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.03, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        population.append(h_pert)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        candidates = []
        for i in range(len(population)):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='COBYLA',
                          bounds=[(0, 1)] * n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-8)
            if res.success:
                new_h = res.x
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                candidates.append(new_h)
        if candidates:
            population += candidates
            population = sorted(population, key=lambda x: obj(x))[:10]
            best_h = population[0].copy()
            best_c5 = obj(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
