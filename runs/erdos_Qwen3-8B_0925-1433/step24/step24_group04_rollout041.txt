from scipy.optimize import minimize
import time

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k_max = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):

            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k_max
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=constraints, tol=1e-6)

    for _ in range(5):
