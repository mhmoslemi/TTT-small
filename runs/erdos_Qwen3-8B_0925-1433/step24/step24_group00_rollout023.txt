import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    learning_rate = 0.1
    decay_factor = 0.995
    perturb_interval = 30
    perturb_strength = 0.1
    iteration = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        pos = (np.arange(n_points) + k_max) % n_points
        neg = (np.arange(n_points) - k_max) % n_points
        grad = (1 - h[pos]) - h[neg]
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if iteration % perturb_interval == 0:
            h += np.random.uniform(-perturb_strength, perturb_strength, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        learning_rate *= decay_factor
        iteration += 1
    return (best_h, best_c5, n_points)
