import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        h_reg = np.roll(h, 1) - h
        R = np.sum(h_reg**2)
        return max_corr + 0.1 * R

    def compute_gradient(h, max_idx):
        K = max_idx - (n_points - 1)
        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - h[pos]) - h[neg]
        return grad

    def compute_regularization_gradient(h):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            h_prev = h[i-1] if i > 0 else h[-1]
            h_next = h[i+1] if i < n-1 else h[0]
            grad[i] = 2 * (h[i] - h_prev) + 2 * (h[i] - h_next)
        return grad

    def optimize_with_bounds(method='L-BFGS-B'):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, best_h, method=method, bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad_corr = compute_gradient(h, max_idx)
        grad_reg = compute_regularization_gradient(h)
        grad_total = grad_corr + 0.1 * grad_reg
        h -= 0.05 * grad_total
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        best_h = optimize_with_bounds('L-BFGS-B')
        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
