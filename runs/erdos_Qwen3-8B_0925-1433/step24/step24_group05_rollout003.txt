from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    pop.append(h_sine)
    
    h_uniform = np.full(n_points, 0.5)
    h_uniform = h_uniform * target_sum / np.sum(h_uniform)
    pop.append(h_uniform)
    
    noise = np.random.normal(0, 0.05, n_points)
    h_perturb = np.array(initial_h_values, dtype=float) + noise
    h_perturb = np.clip(h_perturb, 0, 1)
    h_perturb = h_perturb * target_sum / np.sum(h_perturb)
    pop.append(h_perturb)
    
    # Fill remaining with random
    for _ in range(pop_size - 4):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        pop.append(h_rand)
    pop = np.array(pop)

    best_c5 = np.inf
    best_h = pop[0]
    elite_ratio = 0.2
    mut_rate = 0.2
    mut_sigma = 0.2
    gens = 100

    for _ in range(gens):
        # Evaluate and track best
        c5s = []
        for h in pop:

        c5s = np.array(c5s)
        if np.min(c5s) < best_c5:

        elites = pop[elite_idx]

        # Generate new population
