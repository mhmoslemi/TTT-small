import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # Fallback

    learning_rate = 0.1
    decay_factor = 0.995
    perturb_interval = 30
    perturb_strength = 0.1
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute correlation to find k_max
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k_max_in_grid = (k_max_idx - (n_points - 1)) % n_points

        # Compute gradient based on k_max_in_grid
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k_max_in_grid) % n_points
            i_minus = (i - k_max_in_grid) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]

        # Update h_new using gradient descent
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)

        # Check if new h_new is better
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()

        # Apply perturbation
        if iteration % perturb_interval == 0:
            perturb = np.random.uniform(-perturb_strength, perturb_strength, size=n_points)
            h_perturbed = h_new + perturb

            h_new = h_perturbed

        # Update h to h_new
        h = h_new.copy()
        learning_rate *= decay_factor
        iteration += 1

    return (best_h, best_c5, n_points)
