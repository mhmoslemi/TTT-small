        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
