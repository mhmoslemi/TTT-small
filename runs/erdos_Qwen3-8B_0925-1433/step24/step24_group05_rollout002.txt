import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    best_h = initial_h_norm.copy()
    best_c5 = np.max(np.correlate(initial_h_norm, 1 - initial_h_norm, mode='full')) * dx

        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    population_size = 50
    population = [initial_h_norm]
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    population.append(sine * target_sum / np.sum(sine))
    for _ in range(population_size - 2):
        rand_h = np.random.rand(n_points)
        population.append(rand_h * target_sum / np.sum(rand_h))

    generations = 100

    mutation_std = 0.1
    crossover_rate = 0.8

        elites = np.array(population)[np.argsort(scores)[:int(len(population)*0.1)]]
        new_population = elites.tolist()
        while len(new_population) < population_size:
