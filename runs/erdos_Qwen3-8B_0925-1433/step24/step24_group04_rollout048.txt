import numpy as np
from scipy.optimize import minimize, Bounds
import time
from skopt import BayesSearch
from skopt.utils import use_named_args

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k_max = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k_max
            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k_max
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        # Perturb and restart
        perturb = np.random.normal(0, 0.05, n_points)
        h = np.clip(h + perturb, 0, 1)
        h = h * target_sum / np.sum(h)

    @use_named_args
    def surrogate_func(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    search_space = [Real(0, 1, prior='uniform') for _ in range(n_points)]
    opt = BayesSearch(surrogate_func, search_space, n_initial_points=5, n_iter=20)
    opt.run()
    best_h_bayes = opt.x
    best_c5_bayes = surrogate_func(best_h_bayes)
    if best_c5_bayes < best_c5:

    best_h = np.clip(best_h, 0, 1)

    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
