import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    # Generate initial patterns
    patterns = [np.array(initial_h_values)]
    x = np.linspace(0, 2, n_points)
    patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
    noise = np.random.normal(0, 0.05, n_points)
    patterns.append(np.clip(np.array(initial_h_values) + noise, 0, 1))
    patterns.append(np.array([0.4 if i % 3 == 0 else 0.2 for i in range(n_points)]))
    patterns.append(np.clip(np.random.uniform(0.4, 0.6, n_points), 0, 1))
    
    # Optimization parameters
    max_iter = 500
    steps_per_pattern = 100
    decay_rate = 1.0 / max_iter
    
    start_time = time.time()
    for pattern in patterns:
        h = pattern * target_sum / np.sum(pattern)
        for step in range(max_iter):

                break

            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
