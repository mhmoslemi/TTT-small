import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h, best_c5 = h.copy(), np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x_scaled = x * target_sum / np.sum(x)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def optimize_start(x0):
        res = minimize(obj, x0, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x, res.fun

    best_h, best_c5 = optimize_start(h)

    x = np.linspace(0, 2, n_points)
    patterns = [
        0.5 + 0.5 * np.sin(np.pi * x),
        np.where((np.arange(n_points) % 2) == 0, 1, 0),
        np.random.rand(n_points),
    ]
    for p in patterns:
        p_scaled = p * target_sum / np.sum(p)
        h_p, c5_p = optimize_start(p_scaled)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    for _ in range(5):

        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        h_p, c5_p = optimize_start(h_perturb)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    return (np.clip(best_h, 0, 1), best_c5, n_points)
