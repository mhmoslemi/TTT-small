        best_h_candidate = initial_h
    else:
        scaling = target_sum / sum_candidate
        best_h_candidate = best_h_candidate * scaling
        best_h_candidate = np.clip(best_h_candidate, 0, 1)

    return (best_h_candidate, best_c5, n_points)
