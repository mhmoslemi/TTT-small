import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    pop_size = 30
    particles = [h_initial.copy()]
    for _ in range(pop_size-1):
        p = h_initial + np.random.normal(0, 0.15, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        p = p[np.random.permutation(n_points)]
        particles.append(p)
    
    inertia = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 150
    velocity = np.zeros((pop_size, n_points))
    pbest = particles.copy()
    gbest = particles[np.argmin([compute_c5(p) for p in particles])]
    best_c5 = compute_c5(gbest)
    
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        fits = [compute_c5(p) for p in particles]
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocity[i] = inertia * velocity[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocity[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            if compute_c5(particles[i]) < compute_c5(pbest[i]):
                pbest[i] = particles[i].copy()
        gbest = pbest[np.argmin([compute_c5(p) for p in pbest])]
        if compute_c5(gbest) < best_c5:
            best_c5 = compute_c5(gbest)
            best_h = gbest.copy()
    
    def refine(p):
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(lambda x: compute_c5(x), p, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons)
        return res.x
    
    best_h = refine(best_h)
    best_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
