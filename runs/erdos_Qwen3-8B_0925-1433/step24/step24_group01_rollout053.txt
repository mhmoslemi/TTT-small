import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    time_start = time.time()
    
    for iter in range(300):
        if time.time() - time_start > budget_s:
            break
        # Smooth max objective
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            vals = corr * dx
            T = max(0.1, 1.0 + iter * 0.01)
            return (1 / T) * np.log(np.sum(np.exp(T * vals)))
        
        # Constraints and bounds
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if not res.success:
            continue
        h = res.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Evaluate true C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation with original metric
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, final_c5, n_points)
