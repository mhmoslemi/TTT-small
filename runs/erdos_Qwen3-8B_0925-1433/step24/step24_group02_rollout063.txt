        best_lag = np.argmax(current_c5)
        p = best_lag - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + p < n_points:
                grad[i] += (1 - h[i + p]) * (2.0 / n_points)
            if 0 <= i - p < n_points:
                grad[i] -= h[i - p] * (2.0 / n_points)
        alpha = 1.0
        for _ in range(50):
            h_test = h + alpha * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * target_sum / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            corr_test_val = corr_test[best_lag] * (2.0 / n_points)
            if corr_test_val < current_c5[best_lag]:
                alpha *= 1.1
            else:
                alpha *= 0.9
            if alpha < 1e-6:
                break
        h = h + alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * (2.0 / n_points))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iteration > 0 and abs(current_c5 - best_c5) < 1e-6:
            break
        iteration += 1
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
