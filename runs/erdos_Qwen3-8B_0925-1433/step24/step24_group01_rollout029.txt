        corr = np.correlate(h, 1 - h, mode='full')
        values = corr * dx
        T = 50.0
        max_value = 100.0 / T
        values = np.clip(values, -max_value, max_value)
        sum_exp = np.sum(np.exp(T * values))
        return (1.0 / T) * np.log(sum_exp)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    try:
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        best_h = result.x
    except:
        best_h = h
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    
    return (best_h, c5_bound, n_points)
