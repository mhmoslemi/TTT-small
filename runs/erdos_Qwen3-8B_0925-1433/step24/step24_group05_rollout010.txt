import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    population = [initial_h]
    x = np.linspace(0, 2, n_points)
    sine_h = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_h = sine_h * (target_sum / np.sum(sine_h))
    population.append(sine_h)
    perturbed = initial_h + np.random.normal(0, 0.05, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * (target_sum / np.sum(perturbed))
    population.append(perturbed)
    for _ in range(3):
        noise = np.random.normal(0, 0.1, n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    
    best_c5 = evaluate(initial_h)
    best_h = initial_h.copy()
    sigma = 0.2
    gen = 0
    
    while gen < 100 and time.time() < budget_s - 10:
        evals = [(evaluate(h), h) for h in population]
        evals.sort()
        current_best_c5, current_best_h = evals[0]
        if current_best_c5 < best_c5:
