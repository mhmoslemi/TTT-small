import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial scaling
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(h)
    
    def obj(x):
        h_curr = x * target_sum / np.sum(x)
        return compute_c5(h_curr)
    
    def local_minimize(x):
        res = minimize(
            obj,
            x,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 50, 'ftol': 1e-7}
        )
        return res.x
    
    population = [h.copy() for _ in range(50)]
    for _ in range(50):
        p = population[np.random.choice(50)]
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
    population = np.array(population)
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            idxs = np.random.choice(50, 3, replace=False)
            x1, x2, x3 = population[idxs]
            mutant = x1 + 0.5 * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
        population = np.array(new_population)
        population.sort(key=obj)
        
        # Local refinement
        for i in range(min(10, len(population))):
            population[i] = local_minimize(population[i])

            best_c5 = current_c5

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
