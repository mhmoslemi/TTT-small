        current_c5 = np.max(np.correlate(best_h_candidate, 1 - best_h_candidate, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = best_h_candidate.copy(), current_c5

        for _ in range(5):
            if time.time() - time.time() > budget_s:
                break
            h_perturb = perturb(best_h_candidate)
            current_c5 = np.max(np.correlate(h_perturb, 1 - h_perturb, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h_perturb.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
