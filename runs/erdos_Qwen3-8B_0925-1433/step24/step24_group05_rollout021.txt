import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 - (h[i_plus] if 0 <= i_plus < n_points else 0.0)
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    # Generate diverse initial candidates
    candidates = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)
    candidates.append(0.5 + 0.5 * np.sin(np.pi * x))
    candidates.append(np.clip(np.array(initial_h_values) + np.random.normal(0, 0.05, n_points), 0, 1))
    block_h = np.array([0.4 if i % 2 == 0 else 0.6 for i in range(n_points)])
    candidates.append(block_h)
    uniform_h = np.random.uniform(0.4, 0.6, n_points)
    candidates.append(uniform_h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = compute_c5(best_h)

    # Evaluate initial candidates
    for h in candidates:
        h_copy = h.copy()
        h_opt = h_copy * target_sum / np.sum(h_copy)
        h_opt = np.clip(h_opt, 0, 1)
        for _ in range(100):
            grad = compute_gradient(h_opt)
            h_opt += 0.01 * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Refine the best candidate with SLSQP
    def objective(x):
        return compute_c5(x)

    def gradient(x):
        return compute_gradient(x)

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=gradient, tol=1e-6)
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
