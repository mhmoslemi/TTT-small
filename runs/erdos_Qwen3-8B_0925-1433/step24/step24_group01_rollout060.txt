import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

            h = x.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            integral_values = corr * dx
            max_int = np.max(integral_values)
