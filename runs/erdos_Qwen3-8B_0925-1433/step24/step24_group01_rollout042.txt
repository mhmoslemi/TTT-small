    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    particles.append(h_initial.copy())
    velocities.append(np.zeros(n_points))
    personal_best.append(h_initial.copy())
    for _ in range(n_particles - 1):
        new_p = np.random.rand(n_points)
        new_p = np.clip(new_p, 0, 1)
        new_p = new_p * target_sum / np.sum(new_p)
        particles.append(new_p.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(new_p.copy())
    fitness = []
    for p in particles:
        one_minus_p = 1 - p
        cor = np.correlate(p, one_minus_p, mode='full')
        fit = np.max(cor) * dx
        fitness.append(fit)
    global_best_idx = np.argmin(fitness)
    best_c5 = fitness[global_best_idx]
    best_h = particles[global_best_idx].copy()
    start_time = time.time()
    inertia_start = 0.7
    inertia_end = 0.4
    max_iterations = 200
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            inertia = inertia_start - (iter / max_iterations) * (inertia_start - inertia_end)
            velocities[i] = inertia * velocities[i] + 1.5 * r1 * (personal_best[i] - particles[i]) + 1.5 * r2 * (particles[global_best_idx] - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= target_sum / np.sum(particles[i])
            one_minus_p = 1 - particles[i]
            cor = np.correlate(particles[i], one_minus_p, mode='full')
            new_fitness = np.max(cor) * dx
            if new_fitness < fitness[i]:
                fitness[i] = new_fitness
                personal_best[i] = particles[i].copy()
            if new_fitness < best_c5:
                best_c5 = new_fitness
                best_h = particles[i].copy()
                global_best_idx = i
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
