from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_initial = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_h = h_initial.copy()

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) > 1e-6:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    
    # Differential evolution with adaptive parameter tweaking
    result = differential_evolution(
