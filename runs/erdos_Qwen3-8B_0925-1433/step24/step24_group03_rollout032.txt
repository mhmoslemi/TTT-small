        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def jac(x):
        h = x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_m = np.argmax(corr)
        k = max_m - (n_points - 1)
        shifted_1_minus_h = np.roll(1 - h, k)
        shifted_h = np.roll(h, -k)
        return shifted_1_minus_h - shifted_h

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)

    initial_configs = [np.array(initial_h_values, dtype=float)]
    for _ in range(3):
        rand_config = np.random.rand(n_points)
        rand_config = rand_config * target_sum / np.sum(rand_config)
        initial_configs.append(rand_config)

    start_time = time.time()
    for config in initial_configs:
        if time.time() - start_time > budget_s:
            break
        config = config * target_sum / np.sum(config)
        res = minimize(objective, config, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=jac, tol=1e-6)
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
        else:
            current_c5 = compute_c5(config)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = config.copy()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=jac, tol=1e-6)
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
