import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    population_size = 20
    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        h = np.clip(np.array(initial_h_values) + noise, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    np.random.shuffle(population)

    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = compute_fitness(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_fitness(h) for h in population]
        parents = []
        for _ in range(population_size // 2):
            t = np.random.choice(population, 5)
            best_in_t = min(t, key=compute_fitness)
            parents.append(best_in_t)
        new_pop = []
        for i in range(population_size // 2):
            if np.random.rand() < 0.8:
                p1, p2 = parents[i], parents[population_size//2 - 1 - i]
                child = (p1 + p2) / 2
            else:
                child = parents[i].copy()
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
        for h in population:
            curr = compute_fitness(h)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()

    top = sorted(population, key=compute_fitness)[:5]
    for h in top:
        res = minimize(lambda x: compute_fitness(x), h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            curr = compute_fitness(res.x)
            if curr < best_c5:
                best_c5 = curr
                best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
