import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population_size = 20
    population = [initial_h.copy()]

        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = (h / np.sum(h)) * target_sum
        population.append(h)
    velocities = np.zeros((population_size, n_points))
    personal_best = [np.copy(p) for p in population]
    global_best = np.copy(population[0])
    best_c5 = np.inf
    for p in population:
        c5 = np.max(np.correlate(p, 1 - p, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
            global_best = p.copy()
    for _ in range(1000):
        if time.time() - time.time() > budget_s:
            break
        omega = 0.7 - (0.7 - 0.4) * (_ / 1000)
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + 1.5 * r1 * (personal_best[i] - population[i]) + 1.5 * r2 * (global_best - population[i])
