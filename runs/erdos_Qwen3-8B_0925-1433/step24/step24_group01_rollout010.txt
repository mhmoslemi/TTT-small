import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(19):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = evaluate_c5(population[0])
    best_h = population[0].copy()
    
    # PSO parameters
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_stagnation = 50
    stagnation = 0
    
    # Personal best and global best
    personal_best = [h.copy() for h in population]
    personal_best_values = [evaluate_c5(h) for h in population]
    global_best = population[0].copy()
    global_best_value = best_c5
    
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            r1 = np.random.rand()
            r2 = np.random.rand()
            new_h = population[i] + cognitive * r1 * (personal_best[i] - population[i]) + social * r2 * (global_best - population[i])
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            population[i] = new_h
            curr_c5 = evaluate_c5(population[i])
            if curr_c5 < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = curr_c5
            if curr_c5 < global_best_value:
                global_best = population[i].copy()
                global_best_value = curr_c5
        if evaluate_c5(global_best) == best_c5:
            stagnation += 1
        else:
            stagnation = 0
        if stagnation > max_stagnation:
            population = [np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(len(population))]
            personal_best = [h.copy() for h in population]
            personal_best_values = [evaluate_c5(h) for h in population]
            global_best = population[0].copy()
            global_best_value = evaluate_c5(global_best)
            stagnation = 0
        if global_best_value < best_c5:
            best_c5 = global_best_value
            best_h = global_best.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
