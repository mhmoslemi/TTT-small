def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)
    
    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k
            if 0 <= i_plus < n_points:
                term1 = 1 - x[i_plus]
            else:
                term1 = 0.0
            i_minus = i - k
            if 0 <= i_minus < n_points:
                term2 = x[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def generate_patterns():
        patterns = []
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
        patterns.append(sin_h)
        patterns.append(np.random.uniform(0.4, 0.6, n_points))
        return [np.array(initial_h_values, dtype=float), *patterns]

    patterns = generate_patterns()
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for p in patterns:

        p = p * target_sum / np.sum(p)
        h = p.copy()
