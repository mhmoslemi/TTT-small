from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    patterns = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)
    patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
    patterns.append(np.clip(np.random.uniform(0.4, 0.6, n_points), 0, 1))
    patterns.append(np.tile([0.4, 0.6, 0.4, 0.6], n_points // 4 + 1)[:n_points])
    for pattern in patterns:
        pattern = pattern * target_sum / np.sum(pattern)
        h = pattern.copy()
        for step in range(500):
            corr = np.correlate(h, 1 - h, mode='full')
            k_max_idx = np.argmax(corr)
            shift = k_max_idx - (n_points - 1)
            grad = np.zeros_like(h)
            for i in range(n_points):
                i_plus = i + shift
                term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0

                grad[i] = term1 - term2
            lr = 0.05 * (0.995 ** step)
            h += lr * grad
