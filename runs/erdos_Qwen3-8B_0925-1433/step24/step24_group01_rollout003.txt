import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Initialize PSO
    n_particles = 20
    particles = []
    velocities = []
    pbest = []
    for _ in range(n_particles):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(h.copy())

    # Evaluate initial population
    for i in range(n_particles):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = particles[i].copy()

    iters = 0
    max_iter = 1000
    start_time = time.time()

    # PSO parameters
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    decay_rate = 0.003

    while iters < max_iter and time.time() - start_time < budget_s:
        for i in range(n_particles):
            # Update velocity
            v = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = inertia * v + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (best_h - particles[i])
            velocities[i] = v
            # Update position
            new_pos = particles[i] + v
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            if sum_new != 0:
                new_pos = new_pos * target_sum / sum_new
            particles[i] = new_pos
            # Evaluate and update pbest
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            if c5 < np.max(np.correlate(pbest[i], 1 - pbest[i], mode='full')) * dx:
                pbest[i] = particles[i].copy()
        # Update global best
        for i in range(n_particles):
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = particles[i].copy()
        # Decay inertia
        inertia = max(inertia - decay_rate, 0.4)
        iters += 1

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
