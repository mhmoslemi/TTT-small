            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-5,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            polish=True,
            disp=False
        )
        result_h = result.x
        result_c5 = objective(result_h)
        if result_c5 < best_c5:
            best_h = result_h
            best_c5 = result_c5
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Add random perturbation to top candidates for diversity
        for _ in range(5):
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            c5_perturb = objective(h_perturb)
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb

    return (best_h, best_c5, n_points)
