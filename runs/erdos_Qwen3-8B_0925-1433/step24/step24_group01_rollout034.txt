import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population = [h_initial.copy()]
    for _ in range(19):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    personal_best = [p.copy() for p in population]
    global_best = population[0].copy()
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx
    velocities = [np.zeros(n_points) for _ in population]
    start = time.time()
    for iter in range(1000):
        if time.time() - start > budget_s:
            break
        inertia = 0.7 - (iter / 1000) * 0.3
        for i in range(len(population)):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = inertia * velocities[i] + 1.5 * r1 * (personal_best[i] - population[i]) + 1.5 * r2 * (global_best - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        for i in range(len(population)):
            curr_c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                global_best = population[i].copy()
            if curr_c5 < np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full')) * dx:
                personal_best[i] = population[i].copy()
    global_best = np.clip(global_best, 0, 1)
    global_best = global_best * target_sum / np.sum(global_best)
    return (global_best, best_c5, n_points)
