import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k_max = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k_max
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k_max
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    result = minimize(
        fun=obj,
        x0=h,
        method='SLSQP',
        bounds=[(0, 1)] * n_points,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        jac=compute_gradient,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    best_h, best_c5 = result.x, obj(result.x)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(
            fun=obj,
            x0=h_perturb,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient,
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = obj(best_h)
    return (best_h, final_c5, n_points)
