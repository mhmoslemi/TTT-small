import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [h]
    for _ in range(9):
        perturbed = h * (1 + np.random.uniform(-0.2, 0.2, n_points))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    population = [np.array(p) for p in population]

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        current_fitness = [compute_overlap(p) for p in population]
        best_idx = np.argmin(current_fitness)
        best_c5 = current_fitness[best_idx]
        elite = sorted(zip(current_fitness, population), key=lambda x: x[0])[:int(len(population)*0.2)]
        new_population = []
        for _ in range(len(population) // 2):
            idxs = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + (b - c) * np.random.uniform(0.5, 1.0)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
            noisy = mutant + np.random.normal(0, 0.05, mutant.shape)
            noisy = np.clip(noisy, 0, 1)
            noisy = noisy * target_sum / np.sum(noisy)
            new_population.append(noisy)
        new_population = sorted(new_population, key=lambda x: compute_overlap(x))[:len(population)//2]
        population = [x[1] for x in elite] + new_population
        if compute_overlap(population[0]) < best_c5:

    res = minimize(compute_overlap, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
