import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    population = [h.copy() for _ in range(10)]
    for i in range(10):
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        population[i] = h_pert

    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_candidates = []
        for h in population:
            pert = np.random.normal(0, 0.03, n_points)
            h_pert = np.clip(h + pert, 0, 1)
            h_pert = h_pert * target_sum / np.sum(h_pert)
            new_candidates.append(h_pert)
        for cand in new_candidates:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, cand, method='COBYLA', bounds=bounds, constraints=cons)
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * target_sum / np.sum(h_opt)
                curr_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = h_opt.copy()
                population.append(h_opt)
        evals = [(np.max(np.correlate(h, 1 - h, mode='full')) * dx, h) for h in population]
        evals.sort()
        population = [h for (c5, h) in evals[:len(population)//2]]
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
