import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        if np.max(corr) == 0:
            return np.zeros_like(x)
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def constraint(x):
        return np.sum(x) - target_sum

    cons = {'type': 'eq', 'fun': constraint}
    bounds = Bounds(0, 1)

    start_time = time.time()

    result = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, size=n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result = minimize(obj, h_perturb, method='L-BFGS-B', jac=grad, bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    result = minimize(obj, best_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
    if result.fun < best_c5:
        best_h, best_c5 = result.x, result.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
