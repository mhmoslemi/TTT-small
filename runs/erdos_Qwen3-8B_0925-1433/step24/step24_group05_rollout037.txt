import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points)
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1) * target_sum / np.sum(h)
    best_h, best_c5 = h.copy(), np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h, k_max_idx):
        grad = np.zeros_like(h)
        k_max = k_max_idx - (n_points - 1)
        for i in range(n_points):
            i_plus = i + k_max
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k_max
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def refine(h):
        for _ in range(100):
            corr = np.correlate(h, 1 - h, mode='full')
            k_max_idx = np.argmax(corr)
            grad = compute_gradient(h, k_max_idx)
            h += 0.05 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        return h

    patterns = [
        h.copy(), 
        0.5 + 0.5 * np.sin(np.pi * x), 
        np.random.normal(0.5, 0.1, n_points), 
        np.tile([0.4, 0.2], n_points // 2), 
        np.random.uniform(0.4, 0.6, n_points)
    ]
    for p in patterns:
        p = np.clip(p, 0, 1) * target_sum / np.sum(p)
        r = refine(p)
        c = np.max(np.correlate(r, 1 - r, mode='full')) * dx
        if c < best_c5:
            best_c5, best_h = c, r

    for _ in range(100):
        if time.time() > budget_s - 5: break
        new_pop = []
        for _ in range(50):
            a, b = np.random.choice([best_h] + [np.random.rand(n_points) * 0.8 for _ in range(4)], 2)
            child = (a + b) / 2
            child = np.clip(child, 0, 1) * target_sum / np.sum(child)
            child = refine(child)
            new_pop.append(child)
        best_pop = sorted(new_pop, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)[:10]
        best_h = best_pop[0]
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
