def run(seed=42, budget_s=600):

        corr = np.correlate(h, 1 - h, mode='full')

        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx1 = (i + lag) % n_points
