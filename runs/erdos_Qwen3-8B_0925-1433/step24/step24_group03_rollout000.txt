from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad_val = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad_val[i] = term1 - term2
        return grad_val * dx
    
    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    start_time = time.time()
    
    if time.time() - start_time < budget_s:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad, tol=1e-6)
        if res.success and res.fun < best_c5:
