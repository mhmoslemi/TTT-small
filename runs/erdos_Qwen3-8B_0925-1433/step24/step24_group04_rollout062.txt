import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # PSO setup
    population_size = 30
    positions = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.15, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        perm = np.random.permutation(n_points)
        new_h = new_h[perm]
        positions.append(new_h)
    
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    personal_best = [pos.copy() for pos in positions]
    global_best = positions[0].copy()
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx
    
    def compute_fitness(pos):
        corr = np.correlate(pos, 1 - pos, mode='full')
        return np.max(corr) * dx
    
    max_iterations = 500
    w = 0.7
    c1, c2 = 1.5, 1.5
    elapsed = 0
    
    for _ in range(max_iterations):
        if elapsed > budget_s:
            break
        start = time.time()
        
        # Update personal bests
        for i in range(population_size):
            curr = compute_fitness(positions[i])
            if curr < compute_fitness(personal_best[i]):
                personal_best[i] = positions[i].copy()
        
        # Update global best
        best_idx = np.argmin([compute_fitness(p) for p in personal_best])
        global_best = personal_best[best_idx].copy()
        best_c5 = compute_fitness(global_best)
        
        # Update velocities and positions
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (personal_best[i] - positions[i]) + c2 * r2 * (global_best - positions[i])
            velocities[i] = vel
            positions[i] += vel
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * target_sum / np.sum(positions[i])
        
        elapsed += time.time() - start
    
    # Local refinement with SLSQP
    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def cons(x):
        return np.sum(x) - target_sum
    
    bounds = Bounds(0, 1)
    candidates = [global_best]
    for _ in range(5):
        if elapsed > budget_s:
            break
        start = time.time()
        res = minimize(obj, candidates[-1], method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons})
        if res.success:
            candidates.append(res.x)
        elapsed += time.time() - start
    
    # Final evaluation with best candidate
    best_h = candidates[0]
    for c in candidates:
        curr_c5 = compute_fitness(c)
        if curr_c5 < best_c5:
            best_h = c
            best_c5 = curr_c5
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
