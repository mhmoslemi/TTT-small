import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        if np.sum(h) == 0:

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * target_sum / np.sum(initial_guess)

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False
    )

    elapsed = time.time() - start_time
    best_h = result.x if elapsed < budget_s else initial_guess

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
