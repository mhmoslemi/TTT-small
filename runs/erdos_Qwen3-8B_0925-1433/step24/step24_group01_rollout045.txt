from scipy.optimize import minimize
import time

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def smooth_obj(h_vec, T):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        vals = (1 - np.take(corr, np.arange(len(corr))) * dx) * T
        return np.log(np.sum(np.exp(vals))) / T
    
    def grad_smooth_obj(h_vec, T):
        h = h_vec * target_sum / np.sum(h_vec)
