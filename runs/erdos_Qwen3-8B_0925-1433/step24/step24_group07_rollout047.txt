        return np.max(corr) * dx

        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - h[pos]) - h[neg]
        return grad

    def normalize(x):
        scale = target_sum / np.sum(x)
        return np.clip(x * scale, 0, 1)

    def evolve_population(population, n_elites=10):

            alpha = np.random.rand()
            child = a * alpha + b * (1 - alpha)
            child = normalize(child)
            new_pop.append(child)
        return new_pop
