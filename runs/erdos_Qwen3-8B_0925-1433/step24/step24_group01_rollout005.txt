import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    T = 1.0
    max_iterations = 200
    time_used = 0

    def smooth_obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        values = corr * dx
        max_val = np.max(values)
        exp_sum = np.sum(np.exp(T * (values - max_val)))
        return (np.log(exp_sum) + max_val) / T

    def grad_func(h):
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_p = h.copy()
            h_p[i] += eps
            h_p = h_p * target_sum / np.sum(h_p)
            val_p = smooth_obj(h_p)
            h_m = h.copy()
            h_m[i] -= eps
            h_m = h_m * target_sum / np.sum(h_m)
            val_m = smooth_obj(h_m)
            grad[i] = (val_p - val_m) / (2 * eps)
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
