from scipy.optimize import minimize, Bounds, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, max_idx):
        K = max_idx - (n_points - 1)
        pos = (np.arange(n_points) + K) % n_points
        neg = (np.arange(n_points) - K) % n_points
        grad = (1 - h[pos]) - h[neg]
        return grad

    def optimize_with_bounds(method='SLSQP'):
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
        res = minimize(obj, best_h, method=method, bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
        return res.x

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = compute_gradient(h, max_idx)
        h -= 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
