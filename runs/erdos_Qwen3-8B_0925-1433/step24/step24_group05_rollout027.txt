import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def evaluate(ind):
        return compute_c5(ind)

    def update_sum(ind):
        s = np.sum(ind)
        if s != 0:
            ind = ind * target_sum / s
        return ind

    def generate_patterns():
        patterns = []
        x = np.linspace(0, 2, n_points)
        patterns.append(initial_h_values.copy())
        patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
        noise = np.random.normal(0, 0.05, n_points)
        patterns.append(np.clip(initial_h_values + noise, 0, 1))
        return patterns

    population = [np.array(p, dtype=float) for p in generate_patterns()]
    for p in population:
        p = update_sum(p)
    population = [p for p in population if np.sum(p) > 1e-6]

    pop_size = 100
    elite_ratio = 0.1
    mut_rate = 0.2
    mut_step = 0.2
    gen = 0
    max_gen = 100

    while gen < max_gen and time.time() < budget_s - 10:
        fitness = np.array([evaluate(p) for p in population])
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [population[i] for i in elite_idx]
        new_pop = elites[:]
        for _ in range(pop_size - len(elites)):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(elites, 2, replace=False)
                child = 0.5 * p1 + 0.5 * p2
            else:
                p = np.random.choice(elites)
                child = p + np.random.normal(0, mut_step, n_points)
            child = update_sum(child)
            new_pop.append(child)
        population = new_pop
        curr_best = min(fitness)
        if curr_best < best_c5:
            best_c5 = curr_best
            best_h = population[np.argmin(fitness)].copy()
        gen += 1

    def obj(x):
        return evaluate(x)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='SLSQP', bounds=[(0, 1) for _ in range(n_points)], constraints=cons, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
