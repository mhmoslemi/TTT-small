def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    learning_rate = 0.1
    iterations = 0
    max_iterations = 600  # Adjust based on time budget
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        kmax_idx = np.argmax(corr)
        # Convert kmax_idx to actual lag kmax
        # For full mode, the lags are -n_points+1 to n_points-1
        kmax = kmax_idx - (n_points - 1)
        
        # Compute gradient for this kmax
        grad = np.zeros_like(h)
        for i in range(n_points):
            j1 = i + kmax
            if 0 <= j1 < n_points:
                term1 = 1 - h[j1]
            else:
                term1 = 0
            j2 = i - kmax
            if 0 <= j2 < n_points:
                term2 = h[j2]
            else:
                term2 = 0
            grad[i] = (term1 - term2) * dx
        
        # Update h
        # Apply gradient descent step
        h = h - learning_rate * grad

            best_h = h.copy()
        
        # Decrease learning rate
