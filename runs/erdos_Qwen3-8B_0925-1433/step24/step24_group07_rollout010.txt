import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    best_c5 = min(best_c5, 0.381)
    
    eta = 0.15
    decay = 0.96
    min_eta = 1e-4
    no_improvement = 0
    max_no_improvement = 30
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, eta, n_points)
        tentative_h = h + perturbation
        tentative_h = np.clip(tentative_h, 0, 1)
        tentative_h = tentative_h * target_sum / np.sum(tentative_h)

                h_opt = x

                return np.max(corr) * dx

                else:

        else:
            no_improvement += 1
        
        if no_improvement >= max_no_improvement:
