    def obj(x):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')

        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def refine(x):

        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, x, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
