            best_h = h.copy()

        index = np.argmax(corr)
        kmax = index - (n_points - 1)

        gradient = np.zeros_like(h)
        for y in range(n_points):
            y_plus = (y + kmax) % n_points
            gradient[y] += (1 - h[y_plus]) * dx
            y_minus = (y - kmax) % n_points
            gradient[y] -= h[y_minus] * dx

        learning_rate = 0.1 / (1 + _ * 0.001)
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if _ % 100 == 0:
            perturb = np.random.normal(0, 0.05, h.shape)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

    return (best_h, best_c5, n_points)
