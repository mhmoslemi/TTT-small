    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        lag_val = k_max_index - (n_points - 1)
        j = (np.arange(n_points) + lag_val) % n_points
        l = (np.arange(n_points) - lag_val) % n_points
        grad = (1 - h[j]) - h[l]
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if iteration % perturb_interval == 0:
            h += np.random.uniform(-perturb_strength, perturb_strength, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_factor
        iteration += 1

    return (best_h, best_c5, n_points)
