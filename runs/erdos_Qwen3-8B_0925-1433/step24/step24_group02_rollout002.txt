    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h_curr = x * target_sum / np.sum(x)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    population = [h.copy()]
    for _ in range(49):
        pert = np.random.normal(0, 0.1, n_points)
        p = np.clip(h + pert, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
    population = np.array(population)

    while time.time() - start_time < budget_s:

            mutant = np.clip(mutant, 0, 1)

            new_pop.append(mutant)
        population = np.array(new_pop)
        for i in range(50):
            x = population[i].copy()
            res = minimize(lambda x: obj(x), x, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-7, options={'maxiter': 100})
