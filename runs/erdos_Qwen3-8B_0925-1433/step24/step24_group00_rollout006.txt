import time

    decay_factor = 0.995
    perturb_interval = 30
    perturb_strength = 0.1

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        # Calculate gradient for k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            l = (i - k_max) % n_points
            grad[i] = (1 - h[j]) - h[l]
        
        # Update h
        h -= learning_rate * grad
