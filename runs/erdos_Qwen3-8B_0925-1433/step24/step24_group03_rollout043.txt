import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = (0, 1)
    start_time = time.time()

    for _ in range(500):  # 500 steps with early termination
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        res = minimize(compute_c5, h_perturb, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        # Random swap perturbation
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_perturb_swap = h.copy()
        h_perturb_swap[[idx1, idx2]] = h_perturb_swap[[idx2, idx1]]
        res_swap = minimize(compute_c5, h_perturb_swap, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-6)
        if res_swap.success:
            current_h = res_swap.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
