from scipy.optimize import minimize, Bounds

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                grad[i] -= (1 - x[i_plus])
            i_minus = i - shift

                grad[i] += x[i_minus]
        return grad
