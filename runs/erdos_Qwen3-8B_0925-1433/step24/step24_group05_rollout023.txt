import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        return h * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = rescale(x)
        return compute_c5(h)

    def grad_finite_diff(x):
        eps = 1e-5
        grad = np.zeros_like(x)
        h = rescale(x)
        base_c5 = compute_c5(h)
        for i in range(len(x)):
            perturbed = x.copy()
            perturbed[i] += eps
            perturbed = rescale(perturbed)
            perturbed_c5 = compute_c5(perturbed)
            grad[i] = (perturbed_c5 - base_c5) / eps
        return grad

    def generate_patterns():
        patterns = []
        patterns.append(initial_h)
        x = np.linspace(0, 2, n_points)
        patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
        noise = np.random.normal(0, 0.05, n_points)
        patterns.append(np.clip(initial_h + noise, 0, 1))
        patterns.append(np.clip(np.random.uniform(0.4, 0.6, n_points), 0, 1))
        return patterns

    patterns = generate_patterns()
    candidates = [rescale(pattern) for pattern in patterns]
    best_h = rescale(initial_h)
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for h in candidates:
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(objective, h, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=constraints, jac=grad_finite_diff, tol=1e-6)
            if res.success and compute_c5(res.x) < best_c5:
                best_c5 = compute_c5(res.x)
                best_h = res.x
        candidates = [rescale(p) for p in candidates]
        for _ in range(50):
            p1, p2 = np.random.choice(candidates, 2, replace=False)
            child = p1 * 0.5 + p2 * 0.5
            child = rescale(child)
            if compute_c5(child) < best_c5:
                candidates.append(child)
        candidates = [rescale(p) for p in candidates[:len(candidates)//2]]
        candidates.append(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
