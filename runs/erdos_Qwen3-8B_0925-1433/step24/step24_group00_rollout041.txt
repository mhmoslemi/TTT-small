import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    learning_rate = 0.1
    decay_rate = 0.999
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        kmax = max_index - (n_points - 1)

        grad = np.zeros_like(h)
        for y in range(n_points):
            shifted_y_plus = (y + kmax) % n_points
            shifted_y_minus = (y - kmax) % n_points
            grad[y] = (1 - h[shifted_y_plus]) - h[shifted_y_minus]

        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            h[:] = 0.5 / n_points
        else:
            h = h * target_sum / np.sum(h)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if np.random.rand() < 0.05:
            h += np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h, 0, 1)
            if np.sum(h) == 0:
                h[:] = 0.5 / n_points
            else:
                h = h * target_sum / np.sum(h)

        learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
