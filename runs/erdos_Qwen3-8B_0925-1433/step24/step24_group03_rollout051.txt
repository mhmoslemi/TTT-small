from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h_vals = x * target_sum / np.sum(x)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def compute_grad(x):
        h_vals = x * target_sum / np.sum(x)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - n_points + 1
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = (i + shift) % n_points
            term1 = 1 - h_vals[i_plus]
            i_minus = (i - shift) % n_points
            term2 = h_vals[i_minus]
            grad[i] = term1 - term2
        return grad

    start_time = time.time()

    momentum = 0.9
    max_iter = 300
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        grad = compute_grad(h)
        m = momentum * m + (1 - momentum) * grad
        h = h - learning_rate * m
        h = np.clip(h, 0, 1)

        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def refine(x):
