import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)

        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    result = differential_evolution(
        obj,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun

    bounds_local = [(0.0, 1.0) for _ in range(n_points)]

    best_h = res.x
    best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = obj(best_h)
