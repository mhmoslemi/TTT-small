import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)

    # Normalize initial guess
    h = h * target_sum / np.sum(h)

    def obj(x):
        h_val = x * target_sum / np.sum(x)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h_val = x * target_sum / np.sum(x)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        k_max_index = np.argmax(corr)
        k_max = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k_max
            if 0 <= i_plus < n_points:
                term1 = 1 - h_val[i_plus]
            else:
                term1 = 0.0
            i_minus = i - k_max
            if 0 <= i_minus < n_points:
                term2 = h_val[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def evaluate(h_vec):
        h_val = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time.time()

    # Initial SLSQP optimization
    res = minimize(obj, h, method='SLSQP', jac=grad, bounds=[(0,1)]*n_points, 
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, 
                   tol=1e-6, options={'maxiter': 100})
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    # Multi-restart with perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='SLSQP', jac=grad, bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate(best_h)
    best_c5 = min(best_c5, final_c5)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
