    initial_pop = np.array(initial_pop)
    best_h = initial_pop[0].copy()
    best_c5 = np.max(np.correlate(initial_pop[0], 1 - initial_pop[0], mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    pop_size = 20
    pop = initial_pop[:pop_size]
    start_time = time.time()
    for gen in range(60):
        if time.time() - start_time > budget_s:
            break
        fitness = np.zeros(pop.shape[0])

            corr = np.correlate(h, 1 - h, mode='full')
            fitness[i] = np.max(corr) * dx
        elite_idx = np.argsort(fitness)[:10]
        elite = pop[elite_idx]
        new_pop = []
        for _ in range(pop_size - len(elite)):

            parent2 = elite[b]
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            child = child * target_sum / np.sum(child)
