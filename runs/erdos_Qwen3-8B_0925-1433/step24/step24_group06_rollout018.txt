            if i_minus >= 0:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def optimize_with_gradient_descent(start_h, max_steps=500, learning_rate=0.05, momentum=0.9):
        h = start_h.copy()
        velocity = np.zeros_like(h)
        best_h = h.copy()
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        start_time = time.time()
        for step in range(max_steps):
            grad = compute_gradient(h)
            velocity = momentum * velocity - learning_rate * grad
            h += velocity
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            if time.time() - start_time > budget_s:
                break
        return best_h, best_c5

    seeds = [
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)),
        np.where((np.arange(n_points) % 4) == 0, 1, 0).astype(float),
        np.where((np.arange(n_points) % 2) == 0, 1, 0).astype(float)
    ]

    start_time = time.time()
    for seed in seeds:
        h_initial = seed * target_sum / np.sum(seed)
        h_opt, c5_opt = optimize_with_gradient_descent(h_initial)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            if time.time() - start_time > budget_s:
                break

    def refine_solution(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    best_h = refine_solution(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
