import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    start_time = time.time()

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def perturb(h, shift):
        pert = np.zeros_like(h)
        for i in range(n_points):
            i_plus = (i + shift) % n_points
            i_minus = (i - shift) % n_points
            pert[i_minus] += 0.1
            pert[i_plus] -= 0.1
        return np.clip(h + pert, 0, 1) * target_sum / np.sum(np.clip(h + pert, 0, 1))

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        h = perturb(h, shift)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:

            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
