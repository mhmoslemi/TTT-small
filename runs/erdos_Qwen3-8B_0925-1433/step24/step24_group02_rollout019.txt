import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h_curr = x * target_sum / np.sum(x)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * dx
    
    def local_minimize(x):
        corr = np.correlate(x, x, mode='full')
        if np.isnan(corr).any():
            return x
        k = np.argmax(corr)
        k = k - len(x) + 1
        if k < 0:
            k = 0
        elif k >= len(x) - 1:
            k = len(x) - 2
        max_val = corr[k + len(x) - 1]
        
        def cost(x):
            return max_val - np.correlate(x, x, mode='full')[k + len(x) - 1]
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(
            cost,
            x,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=constraints,
            tol=1e-7,
            options={'maxiter': 100}
        )
        if res.success:
            return res.x
        return x
    
    population = [h.copy() for _ in range(50)]
    for p in population:
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            idxs = np.random.choice(range(50), 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = x1 + 0.5 * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
        population = new_population
        population.sort(key=obj)
        for i in range(10):
            population[i] = local_minimize(population[i])
        current_c5 = obj(population[0])
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5
        gen += 1
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
