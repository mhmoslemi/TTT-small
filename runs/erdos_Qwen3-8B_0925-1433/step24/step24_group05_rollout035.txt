            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop = np.vstack((new_pop, child))
        population = new_pop
        # Update best
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)]
        # Time check
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final optimization
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - k
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad
    
    res = minimize(obj, best_h, method='SLSQP', bounds=(0,1), 
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, 
                   tol=1e-6)
    if res.success:
        best_h = res.x
    # Ensure bounds and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    # Final C5 calculation
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
