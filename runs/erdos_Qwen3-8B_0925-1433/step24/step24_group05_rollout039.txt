    perturbed = initial_h + noise
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    patterns.append(perturbed)
    block_h = np.zeros(n_points)
    block_size = n_points // 3
    block_h[:block_size] = 0.4
    block_h[block_size:2 * block_size] = 0.6
    block_h[2 * block_size:] = 0.2
    block_h = block_h * target_sum / np.sum(block_h)
    patterns.append(block_h)
    uniform_h = np.random.uniform(0.4, 0.6, n_points)
    uniform_h = uniform_h * target_sum / np.sum(uniform_h)
    patterns.append(uniform_h)

    for h_start in patterns:
        h = np.clip(h_start, 0, 1)
        h = h * target_sum / np.sum(h)
        best_c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = h.copy()
        for _ in range(500):
            corr = np.correlate(h, 1 - h, mode='full')
            m = np.argmax(corr)
            k_max = m - (n_points - 1)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                i_plus = i + k_max
                term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
                i_minus = i - k_max
                term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
                gradient[i] = term1 - term2
            h -= 0.02 * gradient
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def const(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, bounds=bounds, constraints={'type': 'eq', 'fun': const}, method='SLSQP', tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
