    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(h_candidate)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - h_candidate[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h_candidate[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def local_optimize(h_candidate, steps=100, step_size=0.05):

            grad = compute_gradient(h_candidate)
            h_candidate += step_size * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)

    def generate_patterns():

        patterns.append(uniform)
        return patterns
