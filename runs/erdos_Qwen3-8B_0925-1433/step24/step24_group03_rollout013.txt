import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial_scaled = h_initial * target_sum / np.sum(h_initial)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(x):
        sum_x = np.sum(x)
        if sum_x == 0:
            return np.inf
        scaling = target_sum / sum_x
        x_scaled = x * scaling
        x_clipped = np.clip(x_scaled, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return np.inf
        scaling2 = target_sum / sum_clipped
        x_final = x_clipped * scaling2
        corr = np.correlate(x_final, 1 - x_final, mode='full')
        c5 = np.max(corr) * dx
        return c5

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=200,
        polish=True
    )

    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
