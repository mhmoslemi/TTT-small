import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 20
    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    best_c5 = None
    best_h = None
    start_time = time.time()
    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        current_candidates = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            current_candidates.append((current_c5, h))
        current_candidates.sort()
        top_candidates = current_candidates[:10]
        new_population = []
        for i in range(10):
            for j in range(i+1, 10):
                alpha = np.random.rand()
                h1, h2 = top_candidates[i][1], top_candidates[j][1]
                offspring = alpha * h1 + (1 - alpha) * h2
                offspring = offspring * target_sum / np.sum(offspring)
                new_population.append(offspring)
        for _ in range(5):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            new_population.append(h)
        optimized = []
        for h in new_population:
            def obj(x):
                h_opt = x * target_sum / np.sum(x)
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                return np.max(corr) * dx
            bounds = np.array([[0., 1.] for _ in range(n_points)])
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
            if res.success:
                optimized.append((np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx, res.x))
        optimized.sort()
        population = [x[1] for x in optimized[:10]]
        if best_c5 is None or optimized[0][0] < best_c5:
            best_c5 = optimized[0][0]
            best_h = optimized[0][1]
    return (np.clip(best_h, 0, 1), best_c5, n_points)
