import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

        if iteration % 100 == 0:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)

        corr = np.correlate(h, 1 - h, mode='full')

        grad = np.zeros_like(h)
