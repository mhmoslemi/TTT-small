from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h_curr = x * target_sum / np.sum(x)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * dx

    def local_minimize(x):
        res = minimize(obj, x, method='SLSQP',
                      bounds=[(0, 1)] * n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            return res.x
        return x

    population = [h.copy()]

        p = np.random.rand(n_points)
        p = p * target_sum / np.sum(p)
        population.append(p)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            a, b = np.random.choice(range(50), 2, replace=False)
            x1, x2 = population[a], population[b]
            x = 0.5 * x1 + 0.5 * x2 + 0.1 * np.random.randn(n_points)
