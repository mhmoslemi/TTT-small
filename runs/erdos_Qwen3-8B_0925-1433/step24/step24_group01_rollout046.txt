import numpy as np
from scipy.optimize import minimize

    dx = 2.0 / n_points

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    population = [h.copy()]
    for _ in range(9):
        pert = np.random.normal(0, 0.03, n_points)
        h = np.clip(initial_h_values + pert, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(9):
            x0 = population[i].copy()
