import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h_initial.copy()
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h):
        return max(np.correlate(h, 1 - h, mode='full')) * dx

    population = [h_initial.copy()]

        perturb = np.random.rand(n_points) * 2 - 1
        perturb = np.clip(perturb, 0, 1)
        perturb = perturb * target_sum / np.sum(perturb)
        population.append(perturb)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break

        candidates.sort(key=lambda x: x[1])
        selected = [c[0] for c in candidates[:10]]
        new_pop = []

                alpha = np.random.rand()

                crossed = np.clip(crossed, 0, 1)

                new_pop.append(crossed)
        for h in selected:
