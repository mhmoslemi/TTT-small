    np.random.seed(seed)
    n_points = len(initial_h_values)

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.99
    perturb_interval = 200
    perturb_strength = 0.1

    for it in range(int(budget_s / 0.01)):  # Approximate iteration count
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            pos = (i + lag) % n_points
            neg = (i - lag) % n_points
            grad[i] = (1 - h[pos]) - h[neg]
