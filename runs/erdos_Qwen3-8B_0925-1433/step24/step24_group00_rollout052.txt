import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    learning_rate = 0.1
    decay_factor = 0.995
    perturb_interval = 100
    perturb_strength = 0.05
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute correlation to find the worst lag
        full_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(full_corr)
        lag = max_index - (n_points - 1)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            l = (i - lag) % n_points
            grad[i] = (1 - h[j]) - h[l]
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Compute current_corr after update
        current_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_corr < best_c5:
            best_c5 = current_corr
            best_h = h.copy()
        
        # Perturb every perturb_interval iterations
        if iteration % perturb_interval == 0:
            h += np.random.uniform(-perturb_strength, perturb_strength, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        # Decay learning rate
        learning_rate *= decay_factor
    
    return (best_h, best_c5, n_points)
