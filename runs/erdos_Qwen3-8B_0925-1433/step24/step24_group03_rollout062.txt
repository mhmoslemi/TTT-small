import numpy as np
from scipy.optimize import minimize, Bounds, L-BFGS-B
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k_max = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = (i + k_max) % n_points
            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = (i - k_max) % n_points
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def optimize_h(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(compute_c5, h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6)
        return result.x

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        h += 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturb = np.clip(perturb, -0.2, 0.2)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result = optimize_h(h_perturb)
        current_c5 = compute_c5(result)
        if current_c5 < best_c5:
            best_h, best_c5 = result.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
