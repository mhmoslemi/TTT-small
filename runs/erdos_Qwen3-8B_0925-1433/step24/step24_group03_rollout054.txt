import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = (i + shift) % n_points
            i_minus = (i - shift) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]
        return grad

    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=compute_gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.randn(n_points) * 0.05
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(objective, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=compute_gradient, tol=1e-6)
