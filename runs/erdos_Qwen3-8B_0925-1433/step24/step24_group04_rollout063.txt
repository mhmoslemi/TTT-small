from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    particles = [h.copy()]
    for _ in range(29):
        noise = np.random.normal(0, 0.15, n_points)
        p = h + noise
        p = clip_rescale(p)
        particles.append(p)
    particles = np.array(particles)
    velocities = np.random.uniform(-0.1, 0.1, (30, n_points))
    pbest = particles.copy()
    pbest_vals = np.array([obj(p) for p in particles])
    gbest = particles[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    start_time = time.time()

            curr_val = obj(particles[i])

                pbest_vals[i] = curr_val
        if np.min(pbest_vals) < gbest_val:

    def refine(h):
