import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_h = np.array(initial_h, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0., 1.) for _ in range(n_points)]
    start_time = time.time()
    
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, seed=seed, maxiter=500
    )
    h = result.x
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    bounds_l = Bounds(0, 1)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds_l, constraints=cons, tol=1e-6)
    h_opt = res.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * target_sum / np.sum(h_opt)
    
    c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
    return (h_opt, c5, n_points)
