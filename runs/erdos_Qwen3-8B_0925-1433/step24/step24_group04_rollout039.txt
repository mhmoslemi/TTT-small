import numpy as np
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return h * target_sum / sum_h

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def pso_step(pop, vel, pbest, pbest_fitness, gbest, gbest_fitness):
        for i in range(len(pop)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel[i] = 0.7 * vel[i] + 1.5 * r1 * (pbest[i] - pop[i]) + 1.5 * r2 * (gbest - pop[i])
            pop[i] = pop[i] + vel[i]
            pop[i] = normalize(pop[i])
            fit = objective(pop[i])
            if fit < pbest_fitness[i]:
                pbest[i] = pop[i].copy()
                pbest_fitness[i] = fit
        new_fitness = np.array([objective(p) for p in pop])
        gbest_idx = np.argmin(new_fitness)
        gbest = pop[gbest_idx].copy()
        gbest_fitness = new_fitness[gbest_idx]
        return pop, vel, pbest, pbest_fitness, gbest, gbest_fitness

    pop_size = 30
    h_base = normalize(np.array(initial_h_values))
    pop = [h_base.copy()]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.15, n_points)
        h_perturb = h_base + perturb
        h_perturb = normalize(h_perturb)
        perm = np.random.permutation(n_points)
        h_perturb = h_perturb[perm]
        pop.append(h_perturb)
    vel = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(pop_size)]
    pbest = [p.copy() for p in pop]
    pbest_fitness = [objective(p) for p in pop]
    gbest = pop[np.argmin(pbest_fitness)].copy()
    gbest_fitness = np.min(pbest_fitness)
    time_start = time.time()
    for _ in range(150):
        if time.time() - time_start > budget_s:
            break
        pop, vel, pbest, pbest_fitness, gbest, gbest_fitness = pso_step(pop, vel, pbest, pbest_fitness, gbest, gbest_fitness)
        w = np.argmin(pbest_fitness)
        if pbest_fitness[w] < gbest_fitness:
            gbest = pbest[w].copy()
            gbest_fitness = pbest_fitness[w]
    best_h = gbest
    best_c5 = gbest_fitness

    def refine_candidates(h):
        cons = {'type': 'eq', 'fun': lambda x: np.sum(normalize(x)) - target_sum}
        res = minimize(objective, h, method='SLSQP', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        return res.x

    best_h = refine_candidates(best_h)
    best_c5 = objective(best_h)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
