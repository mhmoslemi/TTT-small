import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.sum(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        grad = np.zeros_like(x)
        for i in range(n_points):
            for k in range(1, n_points):
                i_k = (i + k) % n_points
                grad[i] += (1 - h[i_k]) - h[i] * k
        return grad

    def nelder_mead_refine(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        idxs = np.random.choice(n_points, size=200, replace=False)
        x0 = np.zeros(n_points)
        x0[idxs] = h[idxs].copy()
        x0 = x0 * target_sum / np.sum(x0)
        res = minimize(obj, x0, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            h = res.x
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = nelder_mead_refine(best_h)
    current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
