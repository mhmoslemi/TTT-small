    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        population = [h]
    
    for _ in range(100):
        if time.time() - start_time > budget_s:

        # Evaluate and select candidates
        candidates = []
        for pop in population:
            curr_c5 = obj(pop)
            if curr_c5 < best_c5:

        for _ in range(5):
