import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_scaled = h * target_sum / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.9,
        polish=False
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
