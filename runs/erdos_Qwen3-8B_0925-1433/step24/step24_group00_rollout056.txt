    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    learning_rate = 0.1
    learning_rate_decay = 0.998
    max_iterations = 1000
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        k = max_corr_idx - (n_points - 1)
        
        shifted_plus = np.arange(n_points) + k
        term1 = np.where((shifted_plus >= 0) & (shifted_plus < n_points), 1 - h[shifted_plus], 0)
        
        shifted_minus = np.arange(n_points) - k
        term2 = np.where((shifted_minus >= 0) & (shifted_minus < n_points), h[shifted_minus], 0)
        
        grad = term1 - term2
        
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        learning_rate *= learning_rate_decay
    
    return (best_h, best_c5, n_points)
