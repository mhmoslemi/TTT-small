import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial scaling
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = Bounds(0, 1)

    # Generate additional patterns
    patterns = []
    x = np.linspace(0, 2, n_points)
    patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
    patterns.append(np.where((np.arange(n_points) % 3) == 0, 1, 0.5).astype(float))
    patterns.append(np.random.rand(n_points))
    patterns.append(np.where((np.arange(n_points) % 2) == 0, 1, 0).astype(float))

    for pattern in patterns:
        p = pattern * target_sum / np.sum(pattern)
        result = minimize(obj, p, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-7)
        if result.success:
            current_h = result.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5
                best_c5 = min(best_c5, 0.381)

    # Perturbation search
    for _ in range(100):
        if time.time() > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result = minimize(obj, h_perturb, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-7)
        if result.success:
            current_c5 = obj(result.x)
            if current_c5 < best_c5:
                best_h, best_c5 = result.x.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
