            new_val = compute_c5(swarm[i])
            if new_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = new_val
        current_global_val = np.min(personal_best_vals)
        if current_global_val < global_best_val:
            global_best_val = current_global_val
            global_best = personal_best[np.argmin(personal_best_vals)]
        if np.random.rand() < mutation_rate:
            mutation = np.random.normal(0, mutation_scale, n_points)
            global_best += mutation
            global_best = np.clip(global_best, 0, 1)
            global_best = global_best * target_sum / np.sum(global_best)
            new_global_val = compute_c5(global_best)
            if new_global_val < global_best_val:
                global_best_val = new_global_val
    
    best_h = np.clip(global_best, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, global_best_val, n_points)
