import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def mutator(x, alpha=0.1):
        x = x.copy()
        x += np.random.normal(0, alpha, size=n_points)
        x = clip_and_normalize(x)
        return x

    population = [mutator(h) for _ in range(50)]
    population = [clip_and_normalize(p) for p in population]
    for p in population:
        p = clip_and_normalize(p)

    for _ in range(200):
        if time.time() - time.time() > budget_s:
            break
        candidates = []
        for ind in population:
            if np.random.rand() < 0.8:
                new_ind = mutator(ind)
                candidates.append(new_ind)
            else:
                idx1, idx2 = np.random.choice(len(population), 2, replace=False)
                parent1, parent2 = population[idx1], population[idx2]
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                child = clip_and_normalize(child)
                candidates.append(child)
        population += candidates
        population = sorted(population, key=lambda x: obj(x))
        population = population[:50]

    best_h = min(population, key=obj)
    best_c5 = obj(best_h)

    def local_refine(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    best_h = local_refine(best_h)
    best_c5 = obj(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
