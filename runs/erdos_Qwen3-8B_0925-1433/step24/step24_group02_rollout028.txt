import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():

    swarm_size = 50
    particles = [np.clip(h_initial + np.random.normal(0, 0.1, n_points), 0, 1) * target_sum / np.sum(np.clip(h_initial + np.random.normal(0, 0.1, n_points), 0, 1)) for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    personal_best = particles.copy()
    personal_best_values = [evaluate_c5(p) for p in personal_best]
    global_best = np.argmin(personal_best_values)
    global_best_val = personal_best_values[global_best]
    
    inertia = 0.8
    cogn = 1.5
    soc = 1.5
    vmax = 0.1
    mut_rate = 0.05
    mut_scale = 0.05
    max_its = 500
    start = time.time()
    
    for _ in range(max_its):
        if time.time() - start > budget_s:
            break
        for i in range(swarm_size):

            v_new = inertia * v + cogn * r1 * (personal_best[i] - particles[i]) + soc * r2 * (personal_best[global_best] - particles[i])
            v_new = np.clip(v_new, -vmax, vmax)
            velocities[i] = v_new

            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            particles[i] = h_new
            new_c5 = evaluate_c5(particles[i])
            if new_c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = new_c5
        new_global = np.argmin(personal_best_values)

            global_best = new_global

        if np.random.rand() < mut_rate:
            g_best = personal_best[global_best]
