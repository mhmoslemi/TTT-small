import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    target_sum = n_points / 2.0
    dx = 2.0 / n_points

            np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points))
        ]
    
    pop = [np.clip(h, 0, 1) for h in generate_initial_population()]
    pop = [h * target_sum / np.sum(h) for h in pop]
    pop = [h for h in pop if np.all(h >= 0) and np.all(h <= 1)]
    
    best_h = pop[0]
    best_c5 = float('inf')
    
    # GA parameters
    gen_limit = 100
    pop_size = 50
    mut_rate = 0.2
    cross_rate = 0.8
    sigma = 0.15
    
    for _ in range(gen_limit):
        # Fitness evaluation
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in pop]
        best_idx = np.argmin(fitness)
        best_c5 = min(best_c5, fitness[best_idx])
        best_h = pop[best_idx].copy()
        
        # Elitism
        elites = [pop[i] for i in np.argsort(fitness)[:int(len(pop)*0.1)]]
        
        # Generate new population
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(elites, 2, replace=False)
            if np.random.rand() < cross_rate:

                child = p1.copy()

            new_pop.append(child)

    # Local refinement
    def obj(x):
        h = x * target_sum / np.sum(x)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
