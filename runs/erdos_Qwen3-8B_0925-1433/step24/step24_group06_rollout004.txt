import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_h[:] = best_h[:] * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 0.0
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            i_minus = i - shift
            term2 = 0.0
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            grad[i] = term1 - term2
        return grad

    seeds = [
        np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values),
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)),
        np.abs(np.sin(np.pi * np.linspace(0, 2, n_points))) * 0.5 + 0.5,
        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)),
        (np.arange(n_points) % 3 < 1).astype(float) * target_sum / np.sum((np.arange(n_points) % 3 < 1).astype(float)),
        np.where((np.arange(n_points) % 5 == 0) | (np.arange(n_points) % 5 == 4), 1, 0).astype(float) * target_sum / np.sum(np.where((np.arange(n_points) % 5 == 0) | (np.arange(n_points) % 5 == 4), 1, 0).astype(float)),
        np.where((np.arange(n_points) % 4 == 0) | (np.arange(n_points) % 4 == 3), 1, 0).astype(float) * target_sum / np.sum(np.where((np.arange(n_points) % 4 == 0) | (np.arange(n_points) % 4 == 3), 1, 0).astype(float)),
    ]

    for seed in seeds:
        h = np.copy(seed)
        for _ in range(100):
            if time.time() > time.time():
                break
            grad = compute_gradient(h)
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
