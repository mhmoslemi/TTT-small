import time

        s = np.sum(h)

            scale = (n_points / 2) / s
            return np.clip(h * scale, 0, 1)
        return np.ones(n_points) * 0.5

    best_c5 = compute_c5(h_initial)

    n_pop = 100
    pop = [scale_h(h_initial + np.random.uniform(-0.1, 0.1, n_points)) for _ in range(n_pop)]
    best_pop = pop.copy()
    best_pop_c5 = [compute_c5(h) for h in pop]

    generation = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        mutant = []
        for i in range(n_pop):
            a, b, c = np.random.choice(n_pop, 3, replace=False)
