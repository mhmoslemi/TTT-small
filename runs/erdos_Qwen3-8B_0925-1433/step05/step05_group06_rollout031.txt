from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

        return (h_best, best_c5, n_points)
    
    def clip_scale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    population = [clip_scale(h) for h in [

    ]]
