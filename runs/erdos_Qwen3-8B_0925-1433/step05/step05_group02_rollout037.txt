from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Objective function
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints
    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Perform optimization
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})

    # Evaluate result
    best_h = result.x
    best_c5 = objective(best_h)
