                current_c5 = compute_c5(h_opt)
                if current_c5 < pbest_c5[i]:
                    pbest[i] = h_opt.copy()
                    pbest_c5[i] = current_c5
                    if current_c5 < gbest_c5:
                        gbest = h_opt.copy()
                        gbest_c5 = current_c5
                        if gbest_c5 < 0.3807:
                            return (gbest, gbest_c5, n_points)

    return (gbest, gbest_c5, n_points)
