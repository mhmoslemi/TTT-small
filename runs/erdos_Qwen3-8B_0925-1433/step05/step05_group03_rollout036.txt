            temp_h[idx2] ^= 1
        elif np.sum(temp_h) == target_sum - 1:
            idx2 = np.random.choice(np.where(temp_h == 0)[0])
            temp_h[idx2] ^= 1
        current_c5 = compute_c5(temp_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = temp_h.copy()
    return (best_h, best_c5, n_points)
