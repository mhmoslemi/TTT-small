        for ind in new_pop:
            for _ in range(20):
                for i in range(n_points):
                    new_ind = ind.copy()
                    new_ind[i] ^= 1
                    if compute_c5(new_ind) < compute_c5(ind):
                        ind = new_ind
                        break
            temp_pop.append(ind)
        population = temp_pop
        current_best = min(population, key=compute_c5)
        curr_c5 = compute_c5(current_best)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()
    return (best_h, best_c5, n_points)
