import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with initial_h_values and random perturbations
    population = [initial_h_values.copy() for _ in range(20)]
    for _ in range(30):
        candidate = np.random.rand(n_points)
        candidate /= np.sum(candidate)
        candidate *= target_sum
        population.append(candidate)
    population = [np.clip(p, 0, 1) for p in population]
    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            if np.random.rand() < 0.6:
                j = np.random.randint(len(population))
                k = np.random.randint(len(population))
                mutant = population[j] + 0.5 * (population[k] - population[i])
                mutant = np.clip(mutant, 0, 1)
                mutant_sum = np.sum(mutant)
                if mutant_sum == 0:
                    new_pop = population[i]
                else:
                    new_pop = mutant * target_sum / mutant_sum
            else:
                new_pop = population[i]
            new_population.append(new_pop)
        population = new_population

    for candidate in population:
        c5 = compute_c5(candidate, dx, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    local_result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints)
    best_h = local_result.x
    best_c5 = compute_c5(best_h, dx, n_points)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
