import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Generate 5 mutants from current best
        mutants = [best_h + np.random.normal(0, 0.01, n_points) for _ in range(5)]
        mutants = [np.clip(m, 0, 1) for m in mutants]
        mutants = [m * target_sum / np.sum(m) for m in mutants]
        refined = []
        for h in mutants:
            cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
            bounds = Bounds(0, 1)
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons)
            refined.append(res.x)
        for h in refined:
            curr = compute_c5(h, dx, n_points)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
