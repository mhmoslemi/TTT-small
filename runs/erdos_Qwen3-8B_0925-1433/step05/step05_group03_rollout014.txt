import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 100

        temp = 100
        corr_clipped = np.clip(temp * corr, -100, 100)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, initial_scaled, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-8})
    
    best_h = result.x
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
