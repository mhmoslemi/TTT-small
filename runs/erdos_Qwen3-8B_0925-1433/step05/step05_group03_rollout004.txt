                indices = np.where(h == 1)[0]
                flip = np.random.choice(indices, diff, replace=False)
                h[flip] = 0
            else:
                indices = np.where(h == 0)[0]
                flip = np.random.choice(indices, -diff, replace=False)
                h[flip] = 1
    target = n_points // 2
    pop_size = 50
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_new = np.random.randint(0, 2, n_points)
        ones = np.sum(h_new)
        diff = ones - target
        if diff != 0:
            if diff > 0:
                indices = np.where(h_new == 1)[0]
                flip = np.random.choice(indices, diff, replace=False)
                h_new[flip] = 0
            else:
                indices = np.where(h_new == 0)[0]
                flip = np.random.choice(indices, -diff, replace=False)
                h_new[flip] = 1
        population.append(h_new)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            j, k = np.random.choice(pop_size, 2, replace=False)
            a, b = population[i], population[j]
            cross = np.random.randint(1, n_points - 1)
            child = np.concatenate((a[:cross], b[cross:]))
            ones = np.sum(child)
            diff = ones - target
            if diff != 0:
                if diff > 0:
                    indices = np.where(child == 1)[0]
                    flip = np.random.choice(indices, diff, replace=False)
                    child[flip] = 0
                else:
                    indices = np.where(child == 0)[0]
                    flip = np.random.choice(indices, -diff, replace=False)
                    child[flip] = 1
            new_pop.append(child)
        for i in range(pop_size):
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                child = new_pop[i].copy()
                child[idx] ^= 1
                ones = np.sum(child)
                diff = ones - target
                if diff != 0:
                    if diff > 0:
                        indices = np.where(child == 1)[0]
                        flip = np.random.choice(indices, diff, replace=False)
                        child[flip] = 0
                    else:
                        indices = np.where(child == 0)[0]
                        flip = np.random.choice(indices, -diff, replace=False)
                        child[flip] = 1
                new_pop[i] = child
        new_pop = sorted(new_pop, key=lambda x: compute_c5(x, n_points))
        population = new_pop[:pop_size]
        current_best = population[0]
        curr_c5 = compute_c5(current_best, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n_points)
