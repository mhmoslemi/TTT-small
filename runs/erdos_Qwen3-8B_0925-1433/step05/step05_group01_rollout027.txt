import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_h = initial_h_values.copy()
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = (1 - current_h[j]) * dx
        step_size = 0.15 * (1 - _ / 500)
        current_h = current_h - step_size * grad
        current_h = np.clip(current_h, 0.0, 1.0)
        current_h = current_h * target_sum / np.sum(current_h)
        new_c5 = compute_c5(current_h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            perturbed_c5 = compute_c5(perturbed, dx, n_points)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
