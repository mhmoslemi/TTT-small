import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Clip and scale initial_h_values to meet constraints
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
    best_h = result.x
    best_c5 = obj(best_h)
    
    return (best_h, best_c5, n_points)
