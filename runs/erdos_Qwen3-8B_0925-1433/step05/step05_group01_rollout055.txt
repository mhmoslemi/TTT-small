import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    current_h = initial_h_values * target_sum / initial_sum
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.9975
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        noise = np.random.normal(0, 0.15, n_points)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        new_h = new_h * target_sum / new_h_sum
        new_c5 = compute_c5(new_h, dx, n_points)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        # Gradient step on max lag
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        lag_idx = np.argmax(corr)
        lag = lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = (1 - current_h[j]) * dx
        step = 0.15
        new_h = current_h - step * grad
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        new_h = new_h * target_sum / new_h_sum
        new_c5 = compute_c5(new_h, dx, n_points)
        if new_c5 < best_c5:
            current_h = new_h
            best_c5 = new_c5
            best_h = new_h.copy()
        temp *= cooling_rate
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
