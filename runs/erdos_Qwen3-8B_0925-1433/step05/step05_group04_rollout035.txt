            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            sum_t = np.sum(trial)
            if sum_t != 0:
                scale_t = (n_points / 2) / sum_t
                trial = np.clip(trial * scale_t, 0, 1)
            else:
                trial = np.full(n_points, 0.5)
            new_pop.append(trial)
        
        new_c5 = [compute_c5(h) for h in new_pop]
        for i in range(pop_size):
            if new_c5[i] < compute_c5(population[i]):
                population[i] = new_pop[i]
        
        current_c5_values = [compute_c5(h) for h in population]
        best_idx = np.argmin(current_c5_values)
        best_c5 = current_c5_values[best_idx]
        best_h = population[best_idx].copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
        
        if generations % 10 == 0:
            top_indices = np.argsort(current_c5_values)[:int(0.05 * pop_size)]
            for idx in top_indices:
                h = population[idx].copy()
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                if sum_h != 0:
                    scale_h = (n_points / 2) / sum_h
                    h = np.clip(h * scale_h, 0, 1)
                else:
                    h = np.full(n_points, 0.5)
                res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
                if res.success:
                    refined = res.x
                    refined = np.clip(refined, 0, 1)
                    sum_r = np.sum(refined)
                    if sum_r != 0:
                        scale_r = (n_points / 2) / sum_r
                        refined = np.clip(refined * scale_r, 0, 1)
                    else:
                        refined = np.full(n_points, 0.5)
                    population[idx] = refined.copy()
                    current_c5 = compute_c5(refined)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = refined.copy()
                        if best_c5 < 0.3807:
                            return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
