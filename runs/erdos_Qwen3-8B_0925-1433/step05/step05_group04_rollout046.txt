import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(45)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h):
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            return np.clip(h * scale, 0, 1)
        return np.ones(n_points) * 0.5

    def objective(x, h, dx):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = np.clip(x * scale, 0, 1)
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    pop_size = 50

        if np.random.rand() < 0.7:
            h = initial_h.copy()

            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = np.clip(h, 0, 1)
        population.append(clip_and_scale(h))

    start_time = time.time()
