import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize and clip the initial solution
    h = np.clip(initial_h_values, 0, 1).copy()
    h_sum = np.sum(h)
    h = h * target_sum / h_sum

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimize with L-BFGS-B for efficiency
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-8, maxiter=500)

    # Ensure the result is feasible and compute C5
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = h
        best_c5 = objective(h)

    return (best_h, best_c5, n_points)
