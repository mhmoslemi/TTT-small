import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h, dx, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial candidate
    h_initial = initial_h_values.copy().astype(float)
    best_h = h_initial
    best_c5 = compute_c5(best_h, dx, n_points)

    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Perturb and optimize
    num_perturbations = 5
    start_time = time.time()
    for _ in range(num_perturbations):
        if time.time() - start_time > budget_s:
            break
        # Perturb the initial candidate
        perturbed = h_initial + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Rescale to target_sum
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        # Local optimization
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=constraints)
        current_h = result.x
        current_c5 = compute_c5(current_h, dx, n_points)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        # Update h_initial for next perturbation
        h_initial = current_h

    # Final optimization on best_h
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints)
    best_h = result.x
    best_c5 = compute_c5(best_h, dx, n_points)

    return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
