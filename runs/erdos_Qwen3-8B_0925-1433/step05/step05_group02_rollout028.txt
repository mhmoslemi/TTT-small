import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize initial_h_values and compute its C5
    initial_h_norm = initial_h_values * target_sum / np.sum(initial_h_values)
    corr_initial = np.correlate(initial_h_norm, 1 - initial_h_norm, mode='full')
    c5_initial = np.max(corr_initial) * dx

    # Generate structured guess
    structured_h = np.zeros(n_points)
    if n_points % 2 == 0:
        structured_h[::2] = 1
    else:
        structured_h[:n_points//2] = 1
        structured_h[-1] = 0.5
    structured_h = structured_h * target_sum / np.sum(structured_h)
    corr_structured = np.correlate(structured_h, 1 - structured_h, mode='full')
    c5_structured = np.max(corr_structured) * dx

    # Select better initial guess
    if c5_initial < c5_structured:
        best_h = initial_h_norm.copy()
        best_c5 = c5_initial
    else:
        best_h = structured_h.copy()
        best_c5 = c5_structured

    start_time = time.time()
    n_iter = 0
    while time.time() - start_time < budget_s:
        n_iter += 1
        # Perturb
        perturbed = best_h + np.random.normal(0, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
