import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Clip and scale initial values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h)
    
    best_c5 = np.inf
    best_h = h.copy()
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturb = np.random.normal(0, 0.05, n_points)
        trial = h + perturb
        trial = np.clip(trial, 0, 1)
        trial = trial * target_sum / np.sum(trial)
        
        bounds = [(0, 1)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        result = minimize(objective, trial, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        else:
            current_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
