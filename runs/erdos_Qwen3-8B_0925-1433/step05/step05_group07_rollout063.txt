    stagnation_count = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = particles[i] * (n_points / 2) / sum_p
            else:
                particles[i] = np.ones(n_points) * 0.5
            current_c5 = compute_c5(particles[i])
            if current_c5 < compute_c5(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = particles[i].copy()
        if best_c5 == compute_c5(global_best):
            stagnation_count += 1
        else:
            stagnation_count = 0
        if stagnation_count >= stagnation_threshold:
            break
    
    return (global_best, best_c5, n_points)
