import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    def adjust_binary(arr):
        cnt = np.sum(arr)
        diff = target_sum - cnt
        if diff == 0:
            return arr
        if diff > 0:
            zeros = np.where(arr == 0)[0]
            idxs = np.random.choice(zeros, int(diff), replace=False)
            arr[idxs] = 1
        else:
            ones = np.where(arr == 1)[0]
            idxs = np.random.choice(ones, -int(diff), replace=False)
            arr[idxs] = 0
        return arr
    initial_binary = np.round(initial_h_values).astype(int)
    initial_binary = adjust_binary(initial_binary)
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    population = [initial_binary.copy()]
    for _ in range(50):
        vec = np.random.rand(n_points) > 0.5
        vec = adjust_binary(vec)
        population.append(vec)
    population = np.stack(population)
    best_h = initial_binary.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(len(population), 50, replace=False)
        parents = population[indices]
        new_pop = []
        for i in range(50):
            j, k = np.random.choice(len(parents), 2, replace=False)
            p1, p2 = parents[j], parents[k]
            split = np.random.randint(1, n_points)
            child = np.concatenate([p1[:split], p2[split:]])
            cnt = np.sum(child)
            if cnt != target_sum:
                if cnt > target_sum:
                    idx = np.random.choice(np.where(child == 1)[0])
                else:
                    idx = np.random.choice(np.where(child == 0)[0])
                child[idx] ^= 1
            new_pop.append(child)
        for i in range(50):
            vec = new_pop[i]
            if np.random.rand() < 0.1:
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                if vec[idx1] != vec[idx2]:
                    vec[idx1], vec[idx2] = 1 - vec[idx1], 1 - vec[idx2]
            new_pop[i] = vec
        population = np.array(new_pop)
        best_idx = np.argmin([compute_c5(p) for p in population])
        curr_h = population[best_idx]
        curr_c5 = compute_c5(curr_h)
        if curr_c5 < best_c5:
            best_h, best_c5 = curr_h, curr_c5
    return (best_h, best_c5, n_points)
