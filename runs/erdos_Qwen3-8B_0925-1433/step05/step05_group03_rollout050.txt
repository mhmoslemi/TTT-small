import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def obj(x):
        return compute_c5(x)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    # Initial local optimization
    res = minimize(obj, initial_scaled, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 100})
    best_h = res.x
    best_c5 = compute_c5(best_h)
    
    # Genetic algorithm steps
    pop_size = 30
    pop = [np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(pop_size)]
    for idx in range(pop_size):
        pop[idx] = np.clip(pop[idx] * target_sum / np.sum(pop[idx]), 0, 1)
    
    best_c5 = compute_c5(best_h)
    best_h = initial_scaled.copy()
    
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            j, k, l = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[j] + 0.8 * (pop[k] - pop[l])
            trial = pop[i].copy()
            for idx in range(n_points):
                if np.random.rand() < 0.9:
                    trial[idx] = mutant[idx]
            trial = np.clip(trial * target_sum / np.sum(trial), 0, 1)
            trial_c5 = compute_c5(trial)
            if trial_c5 < compute_c5(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        curr_best = min(pop, key=compute_c5)
        curr_c5 = compute_c5(curr_best)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_best.copy()
    
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 100})
    best_h = res.x
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
