import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    population = [initial_h_values.copy() for _ in range(20)]
    for _ in range(15):
        candidate = np.random.rand(n_points)
        candidate /= np.sum(candidate)
        candidate *= target_sum
        population.append(candidate)

    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(250):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            j, k = np.random.choice(len(population), 2, replace=False)
            a, b = population[j], population[k]
            mutant = a + 0.4 * (b - a)
            trial = np.where(np.random.rand(n_points) < 0.8, mutant, population[i])
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            original_c5 = compute_c5(population[i], dx, n_points)
            trial_c5 = compute_c5(trial, dx, n_points)
            if trial_c5 < original_c5:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population[:20]

    best_c5 = np.inf
    best_h = initial_h_values.copy()
    for candidate in population:
        c5_val = compute_c5(candidate, dx, n_points)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = candidate.copy()

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    local_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints)
    best_h = local_result.x
    best_c5 = compute_c5(best_h, dx, n_points)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
