import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    # Normalize initial_h_values
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    population_size = 50
    particles = []
    velocities = []
    for _ in range(population_size):
        h = initial_h_values.copy()
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-8:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    pbest = particles.copy()
    pbest_values = [compute_c5(p) for p in particles]
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = gbest_value = np.min(pbest_values)
    best_h = gbest.copy()
    best_c5 = gbest_value
    
    start_time = time.time()
    
    w, c1, c2 = 0.7, 1.5, 1.5
    max_iterations = 150
    stagnation = 0
    
    for _ in range(max_iterations):

            particles[i] = np.clip(particles[i], 0, 1)
