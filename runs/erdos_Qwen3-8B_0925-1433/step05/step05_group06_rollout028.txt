                h_temp = np.clip(h_temp * (n_points / 2) / sum_t, 0, 1)
            else:
                h_temp = np.zeros(n_points)
            c1 = compute_c5(h_temp)
            h_temp[i] -= 2 * eps
            sum_t = np.sum(h_temp)
            if sum_t != 0:
                h_temp = np.clip(h_temp * (n_points / 2) / sum_t, 0, 1)
            else:
                h_temp = np.zeros(n_points)
            c2 = compute_c5(h_temp)
            grad[i] = (c1 - c2) / (2 * eps)
            h_temp[i] += eps
        alpha = 0.1
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = np.clip(new_h * (n_points / 2) / sum_new, 0, 1)
        c = compute_c5(new_h)
        if c < current_c5:
            h = new_h
        if np.random.rand() < 0.1:
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
    return (best_h, best_c5, n_points)
