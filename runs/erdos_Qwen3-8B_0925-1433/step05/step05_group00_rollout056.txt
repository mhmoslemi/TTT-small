from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    num_perturbations = 20
    num_refine = 5

    for _ in range(num_perturbations):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(
            lambda x: compute_c5(x, dx, n_points),
            perturbed,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
                for _ in range(num_refine):
                    if time.time() - start_time > budget_s:
                        break
                    noise = np.random.normal(0, 0.01, n_points)
                    perturbed = best_h + noise
                    perturbed = np.clip(perturbed, 0, 1)
                    perturbed = perturbed * target_sum / np.sum(perturbed)
                    res = minimize(
                        lambda x: compute_c5(x, dx, n_points),
                        perturbed,
                        method='SLSQP',
                        bounds=[(0, 1)] * n_points,
                        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
                    )
                    if res.success:
                        curr_h = res.x
                        curr_c5 = compute_c5(curr_h, dx, n_points)
                        if curr_c5 < best_c5:
                            best_c5 = curr_c5
                            best_h = curr_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
