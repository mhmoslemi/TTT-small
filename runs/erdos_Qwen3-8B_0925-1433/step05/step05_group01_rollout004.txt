import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h_initial.copy()
    best_c5 = compute_c5(h_initial, dx, n_points)
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    iterations = 800
    eps = 0.1
    perturb_freq = 20
    
    for step in range(iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb
        perturb = np.random.choice([-1, 0, 1], size=n_points)
        h_perturb = h_initial + eps * perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb_sum = np.sum(h_perturb)
        h_perturb = h_perturb * target_sum / h_perturb_sum if h_perturb_sum > 1e-10 else h_perturb
        # Compute new C5
        corr = correlate(h_perturb, 1 - h_perturb, mode='full')
        new_c5 = np.max(corr) * dx
        # Accept
        current_c5 = compute_c5(h_initial, dx, n_points)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_initial = h_perturb
        # Update best
        current_c5 = compute_c5(h_initial, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_initial.copy()
        # Local refinement
        if step % perturb_freq == 0:
            corr = correlate(h_initial, 1 - h_initial, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx
            def obj(x):
                h = x
                if k == 0:
                    return np.sum(h * (1 - h))
                else:
                    return np.sum(h[:n_points - k] * (1 - h[k:]))
            def grad(x):
                h = x
                if k == 0:
                    return 1 - 2 * h
                else:
                    grad = np.zeros_like(h)
                    grad[:n_points - k] = 1 - h[k:]
                    return grad
            res = minimize(obj, h_initial, method='SLSQP', jac=grad, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_initial = res.x
                h_initial = np.clip(h_initial, 0, 1)
                h_initial_sum = np.sum(h_initial)
                h_initial = h_initial * target_sum / h_initial_sum if h_initial_sum > 1e-10 else h_initial
                current_c5 = compute_c5(h_initial, dx, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_initial.copy()
        temp *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
