import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    population = [initial_h_values.copy() for _ in range(10)]
    for _ in range(30):
        if np.random.rand() < 0.5:
            candidate = np.random.rand(n_points)
            candidate /= np.sum(candidate)
            candidate *= target_sum
        else:
            idx = np.random.randint(10)
            candidate = population[idx].copy()
            perturb = np.random.normal(0, 0.1, n_points)
            candidate += perturb
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        candidate *= target_sum / candidate_sum if candidate_sum > 0 else 0.0
        population.append(candidate)
    population = [np.clip(p, 0, 1) for p in population]

    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        candidates = []
        for p in population:
            p = np.clip(p, 0, 1)
            p_sum = np.sum(p)
            p *= target_sum / p_sum if p_sum > 0 else 0.0
            candidates.append(p)
        candidates = np.array(candidates)
        scores = np.array([compute_c5(c, dx, n_points) for c in candidates])
        top_idx = np.argsort(scores)[:30]
        top_candidates = candidates[top_idx]

        new_population = []
        for i in range(30):
            if np.random.rand() < 0.1:
                c = np.random.rand(n_points)
                c /= np.sum(c)
                c *= target_sum
            else:
                c = top_candidates[np.random.randint(30)].copy()
                perturb = np.random.normal(0, 0.1, n_points)
                c += perturb
            c = np.clip(c, 0, 1)
            c_sum = np.sum(c)
            c *= target_sum / c_sum if c_sum > 0 else 0.0
            new_population.append(c)
        population = new_population

    best_c5 = np.inf
    best_h = initial_h_values.copy()
    for candidate in population:
        c5_val = compute_c5(candidate, dx, n_points)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = candidate.copy()

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    local_result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints)
    best_h = local_result.x
    best_c5 = compute_c5(best_h, dx, n_points)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
