import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.6:
            h = initial_h_values.copy()
        elif np.random.rand() < 0.3:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        population.append(h)

    # Initial best solution
    best_h = initial_h_values.copy()
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
    else:
        best_h = np.ones(n_points) * 0.5
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    # Local refinement parameters
    local_refine_rate = 0.2

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        current_best_idx = np.argmin(fitness)

        # Update best solution
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)

        # Generate next population
        next_population = []
