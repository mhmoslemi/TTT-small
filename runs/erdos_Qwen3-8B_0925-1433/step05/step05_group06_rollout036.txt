        best_alpha = 0.0
        best_c5_new = np.inf
        for _ in range(10):
            h_candidate = h - alpha * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_candidate = np.sum(h_candidate)
            scale_candidate = (n_points / 2) / sum_candidate if sum_candidate != 0 else 1
            h_scaled = np.clip(h_candidate * scale_candidate, 0, 1)
            c5_new = compute_c5(h_scaled)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_alpha = alpha
            alpha *= 0.5
        h = h - best_alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_h = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = np.clip(h * scale_h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
