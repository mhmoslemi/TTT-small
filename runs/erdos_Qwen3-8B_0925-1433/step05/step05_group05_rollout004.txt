import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial = initial_h_values
    else:
        n_points = 45
        initial = np.random.uniform(0, 1, n_points)
    dx = 2.0 / n_points

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        scale = (n_points / 2) / s
        return np.clip(h * scale, 0.0, 1.0)

    population = [normalize(initial.copy())]
    while len(population) < 50:
        population.append(normalize(np.random.uniform(0, 1, n_points)))

    def compute_c5(h):
        s = np.correlate(h, 1 - h, mode='full')
        return np.max(s) * dx

    best_c5 = np.inf
    best_h = None
    start = time.time()

    for gen in range(300):
        if time.time() - start > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        curr_best_idx = np.argmin(fitness)
        curr_best = fitness[curr_best_idx]
        if curr_best < best_c5:
            best_c5, best_h = curr_best, population[curr_best_idx].copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)

        selected = np.argsort(fitness)[:40]
        selected_pop = [population[i] for i in selected]
        new_pop = []

        for i in range(0, len(selected_pop), 2):
            if i + 1 < len(selected_pop):
                p1, p2 = selected_pop[i], selected_pop[i + 1]
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, p1, p2)
                new_pop.append(child)

        for i in range(len(new_pop)):
            c = new_pop[i]
            noise = np.random.normal(0, 0.05, n_points)
            c += noise
            s = np.sum(c)
            if s == 0:
                c = np.full_like(c, 0.5)
            else:
                scale = (n_points / 2) / s
                c = np.clip(c * scale, 0.0, 1.0)
            new_pop[i] = c

        combined = selected_pop + new_pop
        combined_fitness = [compute_c5(ind) for ind in combined]
        combined.sort(key=lambda x: compute_c5(x))
        population = combined[:50]

    return (best_h, best_c5, n_points)
