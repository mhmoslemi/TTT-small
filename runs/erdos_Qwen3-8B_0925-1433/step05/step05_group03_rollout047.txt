import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = []
    # Scale initial_h_values
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    initial_guesses.append(initial_scaled)
    # Add random binary sequences
    for _ in range(3):

        count = np.sum(binary_mask)
        if count == 0:
            binary_mask = np.zeros(n_points)
        else:
            binary_mask = binary_mask * target_sum / count
        initial_guesses.append(binary_mask)
    
    # Objective with soft max
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        temp = 100.0
        soft_max = np.log(np.sum(np.exp(corr / temp))) * temp
        return soft_max * dx
    
    # Constraints
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = initial_scaled.copy()
    
    # Try each initial guess within budget
    start_time = time.time()
    for x0 in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 100})
            current_h = res.x
            current_sum = np.sum(current_h)

            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:

                best_h = current_h.copy()
        except:
            pass
