        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.zeros(n_points)
            else:
                h = h * (n_points / 2) / sum_h
    
    return (best_h, best_c5, n_points)
