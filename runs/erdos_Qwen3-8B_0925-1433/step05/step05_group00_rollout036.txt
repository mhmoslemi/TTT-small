            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        trial_h = h + learning_rate * grad
        trial_h = np.clip(trial_h, 0, 1)
        s = np.sum(trial_h)
        trial_h = trial_h * target_sum / s if s != 0 else trial_h
        trial_c5 = compute_c5(trial_h)
        if trial_c5 < compute_c5(h):
            h = trial_h
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = h.copy()
        else:
            learning_rate *= 0.9
        iteration += 1

    return (best_h, best_c5, n_points)
