                    h_initial[flip_indices] = 0
            else:
                indices = np.where(h_initial == 0)[0]
                if indices.size > 0:
                    flip_indices = np.random.choice(indices, -diff, replace=False)
                    h_initial[flip_indices] = 1
        initial_pop = [h_initial.copy()]
    else:
        h_initial = np.random.choice([0, 1], size=n_points)
        ones_current = np.sum(h_initial)
        while ones_current != target_ones:
            if ones_current > target_ones:
                indices = np.where(h_initial == 1)[0]
                if indices.size > 0:
                    idx = np.random.choice(indices)
                    h_initial[idx] = 0
            else:
                indices = np.where(h_initial == 0)[0]
                if indices.size > 0:
                    idx = np.random.choice(indices)
                    h_initial[idx] = 1
            ones_current = np.sum(h_initial)
        initial_pop = [h_initial.copy()]

    pop_size = 50
    for _ in range(pop_size - 1):
        h_new = np.random.choice([0, 1], size=n_points)
        ones_current = np.sum(h_new)
        while ones_current != target_ones:
            if ones_current > target_ones:
                indices = np.where(h_new == 1)[0]
                if indices.size > 0:
                    idx = np.random.choice(indices)
                    h_new[idx] = 0
            else:
                indices = np.where(h_new == 0)[0]
                if indices.size > 0:
                    idx = np.random.choice(indices)
                    h_new[idx] = 1
            ones_current = np.sum(h_new)
        initial_pop.append(h_new)

    pop = initial_pop.copy()
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    max_iter = 300

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            j, k, l = np.random.choice(pop_size, 3, replace=False)
            p1, p2 = pop[j], pop[k]
            crossover_point = np.random.randint(1, n_points)
            child = p1.copy()
            child[crossover_point:] = p2[crossover_point:]
            child = adjust_count(child, target_ones)
            idx1, idx2 = np.random.randint(n_points), np.random.randint(n_points)
            if idx1 == idx2:
                idx2 = (idx1 + 1) % n_points
            child[idx1] ^= 1
            child[idx2] ^= 1
            child_c5 = compute_c5(child)
            current = pop[i]
            current_c5 = compute_c5(current)
            new_pop.append(child if child_c5 < current_c5 else current)
        pop = new_pop
        current_best = min(pop, key=compute_c5)
        curr_c5 = compute_c5(current_best)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()

    def refine(h):
        for _ in range(20):
            for idx in range(n_points):
                new_h = h.copy()
                new_h[idx] ^= 1
                ones = np.sum(new_h)
                if ones != target_ones:
                    if new_h[idx] == 1:
                        ones_idx = np.where(new_h == 1)[0]
                        if ones_idx.size > 0:
                            other_idx = np.random.choice(ones_idx)
                            new_h[other_idx] ^= 1
                    else:
                        zeros_idx = np.where(new_h == 0)[0]
                        if zeros_idx.size > 0:
                            other_idx = np.random.choice(zeros_idx)
                            new_h[other_idx] ^= 1
                new_c5 = compute_c5(new_h)
                if new_c5 < compute_c5(h):
                    h = new_h
                    break
        return h

    best_h = refine(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
