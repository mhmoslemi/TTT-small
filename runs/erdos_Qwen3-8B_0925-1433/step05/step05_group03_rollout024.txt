    best_c5 = compute_c5(best_h, dx, n_points)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, dx, n_points):
    grad = np.zeros_like(h)
    eps = 1e-4
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        obj_plus = compute_c5(h_plus, dx, n_points)
        obj_minus = compute_c5(h_minus, dx, n_points)
        grad[i] = (obj_plus - obj_minus) / (2 * eps)
    return grad

def local_refine(h, dx, n_points, target_sum, max_iter=50, lr=0.05):
    for _ in range(max_iter):
        grad = compute_gradient(h, dx, n_points)
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * target_sum / h_sum
        else:
            h = np.full(n_points, target_sum / n_points)
    return h
