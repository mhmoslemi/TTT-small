        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    scale = (n_points / 2) / h_sum if h_sum != 0 else 1
    h = np.clip(h * scale, 0, 1)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)

        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx

        alpha = 0.1
        h_trial = h - alpha * grad
        h_trial = np.clip(h_trial, 0, 1)
        h_trial_sum = np.sum(h_trial)
        if h_trial_sum != 0:
            h_trial = np.clip(h_trial * (n_points / 2) / h_trial_sum, 0, 1)
        c5_trial = compute_c5(h_trial)

        while alpha > 1e-5:
            alpha *= 0.9
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum != 0:
                h_new = np.clip(h_new * (n_points / 2) / h_new_sum, 0, 1)
            c5_new = compute_c5(h_new)
            if c5_new < c5_trial:
                c5_trial = c5_new
                h = h_new
                break

        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = np.clip(h * (n_points / 2) / h_sum, 0, 1)
        current_c5 = compute_c5(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)

        if _ % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum != 0:
                h = np.clip(h * (n_points / 2) / h_sum, 0, 1)

    return (best_h, best_c5, n_points)
