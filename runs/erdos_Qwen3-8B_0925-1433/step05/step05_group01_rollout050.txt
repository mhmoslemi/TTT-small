    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_corr = np.max(corr)
        m = np.argmax(corr)
        k = m - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            xi = i * dx
            xj = xi + k
            if 0 <= xj <= 2:
                j = int(np.round(xj / dx))
                if 0 <= j < n_points:
                    grad[i] = (1 - current_h[j]) * dx
        step_size = 0.1
        current_h = current_h - step_size * grad
        current_h = np.clip(current_h, 0, 1)
        current_sum = np.sum(current_h)
        current_h = current_h * target_sum / current_sum
        current_c5 = compute_c5(current_h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        if _ % 5 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            perturbed = perturbed * target_sum / perturbed_sum
            perturbed_c5 = compute_c5(perturbed, dx, n_points)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
