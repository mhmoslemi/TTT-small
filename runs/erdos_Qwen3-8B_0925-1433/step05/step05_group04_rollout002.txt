import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.full(n_points, 0.5)
        return np.clip(h * (n_points / 2) / s, 0, 1)

    def objective(x, h, dx):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return 1e6
        x_scaled = np.clip(x * (n_points / 2) / s, 0, 1)
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 200
    h_init = clip_and_scale(initial_h_values)
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.7:
            h = h_init + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        population.append(clip_and_scale(h))
    population = np.array(population)

    best_h = h_init
    best_c5 = compute_c5(best_h)
    best_eval = best_c5
    gen = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        gen += 1
        # DE step
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            new_pop.append(clip_and_scale(trial))
        population = np.array(new_pop)

        # Update best
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_eval:
                best_h = h.copy()
                best_eval = c5
                if best_eval < 0.3807:
                    return (best_h, best_eval, n_points)

        # SLSQP refinement
        if gen % 10 == 0:
            candidates = population[np.argsort([compute_c5(h) for h in population])]
            for h in candidates[:5]:
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

                if res.success:

                        best_eval = c5_opt
                        if best_eval < 0.3807:

    return (best_h, best_eval, n_points)
