import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and random samples
    def generate_population(base, size=50):
        pop = [base.copy()]
        for _ in range(size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.ones(n_points) * 0.5
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            pop.append(h)
        return pop

    population = generate_population(initial_h_values)
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    def s_lsqp_refine(h):
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted = 1 - x
            corr = np.correlate(x, shifted, mode='full')
            return np.max(corr) * dx
        def constraint(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            return sum_x - n_points / 2
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x if res.success else h

    while time.time() - start_time < budget_s:
        new_pop = []
        for h in population:
            # Mutation
            mutated = h.copy()
            for i in range(n_points):

                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0, 1)
            # Scale

            if sum_m != 0:
                scale = (n_points / 2) / sum_m
                mutated = np.clip(mutated * scale, 0, 1)
            else:

            new_pop.append(mutated)
        # Evaluate and select

            current_c5 = compute_c5(h)
            if current_c5 < best_c5:

                best_c5 = current_c5

                best_c5 = current_c5

        population = new_pop
    return (best_h, best_c5, n_points)
