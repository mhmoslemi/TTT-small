import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    max_tries = 100
    
    for _ in range(max_tries):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed candidate
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        candidate = np.clip(initial_h_values + perturb, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate != 0:
            scale = (n_points / 2) / sum_candidate
            candidate = np.clip(candidate * scale, 0, 1)
        current_c5 = compute_c5(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate.copy()
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
        
        # Optimize candidate
        corr_initial = np.correlate(candidate, 1 - candidate, mode='full')
        max_index = np.argmax(corr_initial)
        lag_index = max_index
        
        def obj(x):
            h_new = x
            shifted = 1 - h_new
            corr = np.correlate(h_new, shifted, mode='full')
            return corr[lag_index] * dx
        
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new != 0:
                scale = (n_points / 2) / sum_new
                new_h = np.clip(new_h * scale, 0, 1)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
