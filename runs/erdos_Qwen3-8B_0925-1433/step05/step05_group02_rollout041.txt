import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = np.inf

    population = [initial_h_values.copy() for _ in range(50)]
    for _ in range(50):
        candidate = np.random.rand(n_points)

        candidate *= target_sum
        candidate = np.clip(candidate, 0, 1)
        population.append(candidate)
    population = [np.clip(np.array(p), 0, 1) for p in population]

    def compute_c5(h):
        return np.max(np.correlate(h, 1-h, mode='full')) * dx

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for h in population:
            # Local optimization
            def obj(x):
                return compute_c5(x)
            bounds = [(0,1)]*n_points
            cons = [{'type':'eq','fun':lambda x: np.sum(x)-target_sum}]

                new_h = res.x
            else:
                new_h = h
            new_pop.append(new_h)
        population = new_pop
        # Track best
        for h in population:
            c5_val = compute_c5(h)
            if c5_val < best_c5:

                best_h = h.copy()

    # Final refinement
    def obj(x):
