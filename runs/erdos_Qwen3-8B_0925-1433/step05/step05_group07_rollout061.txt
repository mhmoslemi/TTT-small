import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def normalize(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / total
        scaled = h_clipped * scale
        return np.clip(scaled, 0, 1)

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        h = normalize(h * (n_points / 2) / sum_h)
    else:
        h = np.full(n_points, 0.5)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    population_size = 50
    population = [h.copy() for _ in range(population_size)]

    for _ in range(150):
        if time.time() - time.time() > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)

        new_population = []
        for _ in range(population_size):
            a, b, c = np.random.choice(n_points, 3, replace=False)
            parent1 = population[np.random.randint(len(population))]
            parent2 = population[np.random.randint(len(population))]
            parent3 = population[np.random.randint(len(population))]
            mutant = (parent1 + parent2 + parent3) / 3 + np.random.uniform(-0.1, 0.1, n_points)
            trial = 0.5 * mutant + 0.5 * parent1
            trial = normalize(trial)
            new_population.append(trial)

        population = new_population

    return (best_h, best_c5, n_points)
