import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def local_optimize(h, dx, n_points):
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        def constraint(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_opt = res.x
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1
            h_opt = np.clip(h_opt * scale, 0, 1)
            return h_opt, compute_c5(h_opt)
        return h, compute_c5(h)

    population_size = 50
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.7:
            h = initial_h_values.copy()
        elif np.random.rand() < 0.2:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = np.clip(h * scale, 0, 1)
        population.append(h)

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        idxs = np.argsort(fitness)

                c = np.random.randint(n_points)
                child[c:] = parent2[c:]
            if np.random.rand() < 0.1:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        child[j] = np.random.uniform(0, 1)
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            scale = (n_points / 2) / sum_c if sum_c != 0 else 1

            new_population.append(child)
        population = new_population

                best_h = h.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
