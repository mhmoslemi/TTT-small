        corr = np.correlate(current_h, 1 - current_h, mode='full')
        m = np.argmax(corr)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + m - (n_points - 1)
            if 0 <= j < n_points:
                grad[i] = (1 - current_h[j]) * dx
        
        step_size = 0.1 / (_ + 1)
        new_h = current_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        new_h = new_h * target_sum / new_sum
        
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        current_h = new_h
        
        if _ % 5 == 0:
            perturb = grad * 0.1 + np.random.normal(0, 0.05, n_points)
            perturbed_h = current_h + perturb
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_sum = np.sum(perturbed_h)
            perturbed_h = perturbed_h * target_sum / perturbed_sum
            perturbed_c5 = compute_c5(perturbed_h)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed_h.copy()
    
    return (best_h, best_c5, n_points)
