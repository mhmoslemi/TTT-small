import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1e6
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = np.clip(x * scale, 0, 1)
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    h_initial = initial_h_values.copy()
    sum_h_initial = np.sum(h_initial)
    if sum_h_initial != 0:
        scale = (n_points / 2) / sum_h_initial
        h_initial = np.clip(h_initial * scale, 0, 1)
    else:
        h_initial = np.ones(n_points) * 0.5
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()

    start_time = time.time()
    for _ in range(5):
        candidates = [best_h.copy()]
        for _ in range(4):
            perturb = np.random.normal(0, 0.02, n_points)
            h = best_h + perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5
            candidates.append(h)
        for h in candidates:
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                sum_h_opt = np.sum(h_opt)
                if sum_h_opt != 0:
                    scale_opt = (n_points / 2) / sum_h_opt
                    h_opt = np.clip(h_opt * scale_opt, 0, 1)
                else:
                    h_opt = np.ones(n_points) * 0.5
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n_points)
