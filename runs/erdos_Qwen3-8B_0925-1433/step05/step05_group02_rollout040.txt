import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2

    population = [initial_h_values.copy()]
    for _ in range(49):
        candidate = np.random.rand(n_points)
        candidate /= np.sum(candidate)
        candidate *= target_sum
        population.append(candidate)
    
    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            mutant = population[i] + np.random.normal(0, 0.1, n_points)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            trial = mutant
            trial_c5 = compute_c5(trial, dx, n_points)
            if trial_c5 < compute_c5(population[i], dx, n_points):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population

    for candidate in population:
        c5_val = compute_c5(candidate, dx, n_points)
        if c5_val < best_c5:

    bounds = [(0, 1)] * n_points

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
