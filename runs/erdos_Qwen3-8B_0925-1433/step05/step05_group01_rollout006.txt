            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        
        def objective(x):
            new_h = x
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1)] * n_points
        result = minimize(objective, current_h, method='L-BFGS-B',
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          bounds=bounds, options={'maxiter': 25, 'ftol': 1e-8})
        
        if result.success:
            new_h = result.x
            new_sum = np.sum(new_h)
            new_h = np.clip(new_h * target_sum / new_sum, 0, 1)
            current_h = new_h
            current_c5 = compute_c5(current_h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        if _ % 5 == 0:
            noise = np.random.normal(0, 0.15, n_points)
            perturbed = current_h + noise
            perturbed_sum = np.sum(perturbed)
            perturbed = np.clip(perturbed * target_sum / perturbed_sum, 0, 1)
            perturbed_c5 = compute_c5(perturbed, dx, n_points)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
