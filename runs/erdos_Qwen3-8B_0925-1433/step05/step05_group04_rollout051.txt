                    x = np.clip(x, 0, 1)
                    sum_x = np.sum(x)
                    return sum_x - n_points / 2
                cons = [{'type': 'eq', 'fun': constraint}]
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
                if res.success:
                    refined_h = res.x
                    refined_h = np.clip(refined_h, 0, 1)
                    sum_refined = np.sum(refined_h)
                    if sum_refined != 0:
                        scale = (n_points / 2) / sum_refined
                        refined_h = np.clip(refined_h * scale, 0, 1)
                    else:
                        refined_h = np.ones(n_points) * 0.5
                    current_c5 = compute_c5(refined_h)
                    if current_c5 < compute_c5(h):
                        index = population.index(h)
                        population[index] = refined_h.copy()
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = refined_h.copy()
                            if best_c5 < 0.3807:
                                return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
