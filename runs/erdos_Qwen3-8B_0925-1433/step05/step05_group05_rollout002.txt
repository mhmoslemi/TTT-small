import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1e6
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = np.clip(x * scale, 0, 1)
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]
