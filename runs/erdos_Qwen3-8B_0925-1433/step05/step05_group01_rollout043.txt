    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        
        cross_corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(cross_corr)
        l_samples = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            shifted_idx = i + l_samples
            if 0 <= shifted_idx < n_points:
                grad[i] = (1 - current_h[shifted_idx]) * dx
        
        step_size = max(0.05, 0.1 * (1 - _ / 500))
        current_h = current_h - step_size * grad
        current_h = np.clip(current_h, 0.0, 1.0)
        current_sum = np.sum(current_h)
        current_h = current_h * target_sum / current_sum
        
        current_c5 = compute_c5(current_h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        if _ % 5 == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_sum = np.sum(perturbed_h)
            perturbed_h = perturbed_h * target_sum / perturbed_sum
            perturbed_c5 = compute_c5(perturbed_h, dx, n_points)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed_h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    return np.max(cross_corr) * dx
