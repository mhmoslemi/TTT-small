import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    learning_rate = 0.5
    no_improvement = 0
    max_iter = int(budget_s / 3)
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        max_lag = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
        step_sizes = [0.1, 0.5, 1.0]
        trial = None
        for s in step_sizes:
            trial_h = h + s * (-grad)
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * target_sum / trial_sum
            trial_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            trial_c5 = np.max(trial_corr) * dx
            if trial_c5 < current_max:
                trial = trial_h
                break
        if trial is not None:
            h = trial
            no_improvement = 0
        else:
            learning_rate *= 0.8
            no_improvement += 1
            if no_improvement > 50:
                break
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
