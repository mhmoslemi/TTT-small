        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        best_h = result.x
        best_c5 = compute_c5(best_h, dx, n_points)
    except:
        best_h = h
        best_c5 = compute_c5(h, dx, n_points)
    
    return (best_h, best_c5, n_points)
