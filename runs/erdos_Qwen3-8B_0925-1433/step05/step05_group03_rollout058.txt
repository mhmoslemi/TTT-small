import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    pop_size = 50
    pop = [initial_scaled.copy()]
    for _ in range(pop_size - 1):
        candidate = np.random.rand(n_points)
        candidate = candidate * target_sum / np.sum(candidate)
        candidate = np.clip(candidate, 0, 1)
        pop.append(candidate)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iter = 300
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size):
            i, j, k = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[i] + 0.8 * (pop[j] - pop[k])
            trial = pop[j].copy()
            for idx in range(n_points):
                if np.random.rand() < 0.9:
                    trial[idx] = mutant[idx]
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.full(n_points, target_sum / n_points)
            else:
                trial = np.clip(trial * target_sum / trial_sum, 0, 1)
            # Local Nelder-Mead optimization
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            bounds = [(0, 1)] * n_points
            res = minimize(obj, trial, method='Nelder-Mead', bounds=bounds, tol=1e-6, options={'maxiter': 100})
            new_pop.append(res.x)
        pop = new_pop
        current_best = min(pop, key=lambda x: compute_c5(x, dx, n_points))
        curr_c5 = compute_c5(current_best, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
