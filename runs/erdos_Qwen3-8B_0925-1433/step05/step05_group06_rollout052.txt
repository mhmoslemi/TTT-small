            perturb = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)

            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                scale = 1.0
            else:
                scale = (n_points / 2) / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)
        
        # Step size decay
        step_size *= decay_rate
    
    return (best_h, best_c5, n_points)
