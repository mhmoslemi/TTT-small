import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    for _ in range(150):  # Reduce iterations for efficiency
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturbation = np.random.normal(0, 0.01, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_sum = np.sum(perturbed_h)
        perturbed_h = perturbed_h * target_sum / perturbed_sum

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

            best_c5 = curr_c5
            best_h = curr_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
