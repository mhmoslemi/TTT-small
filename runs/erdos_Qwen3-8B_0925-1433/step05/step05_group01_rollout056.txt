import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_h = np.array(initial_h_values, dtype=np.float64)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        bounds = [(0.0, 1.0)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, current_h, method='COBYLA', bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            new_h = new_h * target_sum / new_sum
            current_c5 = compute_c5(new_h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
        noise = np.random.normal(0.0, 0.05, n_points)
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        perturbed_c5 = compute_c5(perturbed, dx, n_points)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed.copy()
    return (best_h, best_c5, n_points)
