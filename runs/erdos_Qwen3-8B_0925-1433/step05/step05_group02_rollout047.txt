import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / h_sum

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-5})
    except:
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-8})
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    return (result.x, result.fun, n_points)
