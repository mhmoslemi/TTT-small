                    child[i] += np.random.uniform(-0.04, 0.04)
                    child[i] = np.clip(child[i], 0, 1)
            
            sum_child = np.sum(child)
            scale = (n_points / 2) / sum_child if sum_child != 0 else 1
            child = np.clip(child * scale, 0, 1)
            new_population.append(child)
        
        population = new_population
        
        top_indices = np.argsort(fitness)[:5]
        for i in top_indices:
            h_opt = population[i].copy()
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1
            h_opt = np.clip(h_opt * scale, 0, 1)
            
            def obj_func(x):
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)
                if sum_x == 0:
                    return 1e6
                scale = (n_points / 2) / sum_x
                x = np.clip(x * scale, 0, 1)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                return np.max(corr) * dx
            
            def cons_func(x):
                x = np.clip(x, 0, 1)
                return np.sum(x) - n_points / 2
            
            cons = [{'type': 'eq', 'fun': cons_func}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj_func, h_opt, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                sum_h = np.sum(h_opt)
                scale = (n_points / 2) / sum_h if sum_h != 0 else 1
                h_opt = np.clip(h_opt * scale, 0, 1)
                population[i] = h_opt
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
