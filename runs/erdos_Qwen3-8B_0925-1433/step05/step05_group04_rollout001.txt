import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def rescale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    h_best = rescale_h(initial_h_values.copy())
    best_c5 = compute_c5(h_best)
    
    num_particles = 150
    particles = []
    velocities = []
    for _ in range(num_particles):
        if np.random.rand() < 0.7:
            h = initial_h_values.copy()
        elif np.random.rand() < 0.2:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = np.clip(h, 0, 1)
        particles.append(rescale_h(h))
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    gbest = h_best.copy()
    gbest_c5 = best_c5
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for i in range(num_particles):
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            vel = 0.7 * vel + 2 * r1 * (particles[i] - particles[i]) + 2 * r2 * (gbest - particles[i])
            velocities[i] = vel
            particles[i] += vel
            particles[i] = rescale_h(particles[i])
        
        for i in range(min(10, num_particles)):
            h = particles[i].copy()
            def obj_func(x):
                x = rescale_h(x)
                return compute_c5(x)
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj_func, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
