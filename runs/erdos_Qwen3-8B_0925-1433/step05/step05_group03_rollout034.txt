import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def adjust_candidate(x, target_sum):
        s = np.sum(x)
        if s == 0:
            return np.full_like(x, target_sum / len(x))
        scale = target_sum / s
        scaled = np.clip(x * scale, 0, 1)
        return scaled

    def objective(x):
        h = adjust_candidate(x, target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    def initial_guess():
        if initial_h_values is not None:
            return initial_h_values.copy()
        ones_needed = int(round(target_sum))
        spacing = n_points // ones_needed if ones_needed > 0 else n_points
        indices = np.arange(0, n_points, spacing)
        if len(indices) < ones_needed:
            extra = np.random.randint(0, spacing, ones_needed - len(indices))
            indices = np.concatenate((indices, extra))
        h = np.zeros(n_points)
        for idx in indices:
            if idx < n_points:
                h[idx] = 1.0
        h = adjust_candidate(h, target_sum)
        return h

    initial_pop = [initial_guess() for _ in range(10)]
    best_h = initial_pop[0]
    best_c5 = objective(best_h)

    start = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', maxiter=100, popsize=50,
        tol=1e-6, mutation=(0.5, 1), recombination=0.9, polish=True, seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)

    def nelder_objective(x):
        return objective(x)

    nelder_res = minimize(nelder_objective, best_h, method='Nelder-Mead',
