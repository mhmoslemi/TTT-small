import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    # Initialize with initial_h_values
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        scale = 1
    else:
        scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    if best_c5 < 0.3807:
        return (best_h, best_c5, n_points)

    # Simulated Annealing
    T = 1.0
    cooling_rate = 0.99
    iterations = 1000
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time >= budget_s:
            break
        # Perturb 3-5 indices
        perturb_count = np.random.randint(3, 6)
        indices = np.random.choice(n_points, size=perturb_count, replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1) * best_h[idx]
            best_h[idx] += delta
        # Clip and rescale
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h == 0:
            scale = 1
        else:
            scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
        new_c5 = compute_c5(best_h)

        # Acceptance
        if new_c5 < best_c5:
            best_c5 = new_c5
        elif np.random.rand() < np.exp( -(new_c5 - best_c5) / T ):
            best_c5 = new_c5
        T *= cooling_rate

    # Local optimization with SLSQP
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = np.clip(x * scale, 0, 1)
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    if res.success:
        best_h = res.x.copy()
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            best_h = np.clip(best_h * scale, 0, 1)
        best_c5 = compute_c5(best_h)

    if best_c5 < 0.3807:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
