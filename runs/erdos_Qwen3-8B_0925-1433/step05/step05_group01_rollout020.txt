import numpy as np
import time
from scipy.optimize import minimize, COBYLA

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    T = 1.0
    epsilon = 0.1
    for iteration in range(800):
        if time.time() - start_time > budget_s:
            break
        # Generate random perturbation
        perturb = np.random.choice([-1, 0, 1], size=n_points)
        new_h = current_h + perturb * epsilon
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        # Compute new C5
        new_c5 = compute_c5(new_h, dx, n_points)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            best_c5 = new_c5
        # Local refinement every 20 steps
        if iteration % 20 == 0:
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            max_k_idx = np.argmax(corr)
            def objective(x):
                return compute_c5(x, dx, n_points)
            def gradient(x):
                corr = np.correlate(x, 1 - x, mode='full')
                max_k_idx = np.argmax(corr)
                grad = np.zeros_like(x)
                for i in range(len(x)):
                    if i + max_k_idx < len(x):
                        grad[i] = (1 - x[i + max_k_idx]) * dx
                return grad
            result = minimize(objective, current_h, method='COBYLA',
                              jac=gradient, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                              options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                current_h = result.x
                current_c5 = compute_c5(current_h, dx, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        T *= 0.995
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
