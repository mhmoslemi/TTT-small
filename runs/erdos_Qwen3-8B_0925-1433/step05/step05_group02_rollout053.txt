import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Clip and normalize to satisfy constraints
    h = np.clip(initial_h_values, 0, 1)
    h /= np.sum(h)
    h *= target_sum

    # Objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints
    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Local optimization with SLSQP
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 150})

    # Evaluate and return
    best_h = result.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
