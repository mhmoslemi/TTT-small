import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Normalize initial values
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    if sum_initial != 0:
        scale_initial = (n_points / 2) / sum_initial
        initial_h = np.clip(initial_h * scale_initial, 0, 1)
    else:
        initial_h = np.ones(n_points) * 0.5
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.inf
        scale = (n_points / 2) / sum_h
        h_scaled = h_clip * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        maxiter=150, popsize=50, tol=1e-6, 
        mutation=(0.5, 1), recombination=0.7, 
        seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final constraint enforcement
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        scale = (n_points / 2) / sum_best
        best_h = np.clip(best_h * scale, 0, 1)
    else:
        best_h = np.ones(n_points) * 0.5
    
    # Early termination if needed
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
