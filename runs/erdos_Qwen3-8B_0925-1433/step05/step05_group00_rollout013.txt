        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_lag_index = np.argmax(corr)
        m = len(corr)
        lag = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        trial_h = h - learning_rate * grad
        trial_h = np.clip(trial_h, 0, 1)
        trial_sum = np.sum(trial_h)
        trial_h = trial_h * target_sum / trial_sum
        trial_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
        trial_c5 = np.max(trial_corr) * dx
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                learning_rate *= 0.9
                no_improvement = 0
    return (best_h, best_c5, n_points)
