import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

        h_scaled = np.clip(h_clipped * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_population = []
    h_init = initial_h_values.copy()
    h_init = np.clip(h_init, 0, 1)
    sum_h = np.sum(h_init)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1
    h_init = np.clip(h_init * scale, 0, 1)
    initial_population.append(h_init)

    for _ in range(10):
        h_perturb = np.random.uniform(0, 1, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h_perturb = np.clip(h_perturb * scale, 0, 1)
        initial_population.append(h_perturb)

    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, population_size=50, strategy='best1bin',

    )
