        bounds = Bounds(0, 1)

        if res.success:
            h = res.x
        curr_c5 = compute_c5(h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
        if _ > 10 and (abs(best_c5 - compute_c5(h, dx, n_points)) < 1e-5):
            break
    
    return (h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
