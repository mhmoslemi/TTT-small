        for c in pop:
            curr_c5 = compute_c5(c, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = c

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def mutate(candidate, mutation_rate):
    if np.random.rand() < mutation_rate:
        zeros = np.where(candidate == 0)[0]
        ones = np.where(candidate == 1)[0]
        if len(zeros) > 0 and len(ones) > 0:
            idx0, idx1 = np.random.choice(zeros), np.random.choice(ones)
            candidate[idx0], candidate[idx1] = 1, 0
    return candidate
