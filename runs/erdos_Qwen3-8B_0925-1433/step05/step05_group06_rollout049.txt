    initial_pop = [initial_h_values.copy(), generate_structured(n_points)]
    for _ in range(5):
        rand_h = np.random.uniform(0, 1, n_points)
        sum_rand = np.sum(rand_h)
        if sum_rand != 0:
            rand_h *= (n_points / 2) / sum_rand
        initial_pop.append(np.clip(rand_h, 0, 1))
    initial_pop = [np.array(p) for p in initial_pop]

    best_h = initial_pop[0]
    best_c5 = compute_c5(best_h)
    if best_c5 < 0.3807:
        return (best_h, best_c5, n_points)

    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    max_iterations = 100
    start_time = time.time()

    for gen in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        scores = [compute_c5(h) for h in initial_pop]
        best_idx = np.argmin(scores)
        best_c5 = scores[best_idx]
        if best_c5 < 0.3807:
            return (initial_pop[best_idx], best_c5, n_points)

        parents = []
        for _ in range(population_size):
            idxs = np.random.choice(len(initial_pop), tournament_size, replace=False)
            best_idx_tourn = np.argmin([scores[i] for i in idxs])
            parents.append(initial_pop[idxs[best_idx_tourn]])

        new_population = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
            else:
                child1, child2 = p1.copy(), p2.copy()

            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.random.uniform(0, 1)
                    child2[i] = np.random.uniform(0, 1)

            def scale_h(h):
                h_clipped = np.clip(h, 0, 1)
                sum_h = np.sum(h_clipped)
                if sum_h == 0:
                    return np.zeros_like(h)
                scale = (n_points / 2) / sum_h
                return h_clipped * scale

            child1 = scale_h(child1)
            child2 = scale_h(child2)
            new_population.extend([child1, child2])

        initial_pop = new_population[:population_size]
        for i in range(min(5, population_size)):
            h_candidate = initial_pop[i]
            h_clipped = np.clip(h_candidate, 0, 1)
            sum_h = np.sum(h_clipped)
            if sum_h == 0:
                continue
            scale = (n_points / 2) / sum_h
            h_scaled = h_clipped * scale
            shifted = 1 - h_scaled
            corr = np.correlate(h_scaled, shifted, mode='full')
            max_corr_val = np.max(corr)
            k_index = np.argmax(corr)

            def objective(x):
                x_clipped = np.clip(x, 0, 1)
                sum_x = np.sum(x_clipped)
                if sum_x == 0:
                    return 1e6
                scale_x = (n_points / 2) / sum_x
                x_scaled = x_clipped * scale_x
                shifted_x = 1 - x_scaled
                corr_x = np.correlate(x_scaled, shifted_x, mode='full')
                k_value = np.argmax(corr_x)
                return corr_x[k_value] * dx

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                h_opt_clipped = np.clip(h_opt, 0, 1)
                sum_opt = np.sum(h_opt_clipped)
                if sum_opt == 0:
                    continue
                scale_opt = (n_points / 2) / sum_opt
                h_opt_scaled = h_opt_clipped * scale_opt
                initial_pop[i] = h_opt_scaled

        for h in initial_pop:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    return (h, best_c5, n_points)

    best_h = initial_pop[np.argmin([compute_c5(h) for h in initial_pop])]
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
