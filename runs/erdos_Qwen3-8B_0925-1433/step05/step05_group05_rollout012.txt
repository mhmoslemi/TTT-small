import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values) if 'initial_h_values' in locals() else 45
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return 1e6
        scale = (n_points / 2) / np.sum(h)
        h_scaled = np.clip(h * scale, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        scale = (n_points / 2) / s
        return np.clip(h * scale, 0, 1)

    def generate_initial_population():
        if 'initial_h_values' in locals():
            h = initial_h_values.copy()
            return [normalize(h)]
        else:
            return [normalize(np.random.uniform(0, 1, n_points)) for _ in range(50)]

    pop = generate_initial_population()
    best_h = pop[0]
    best_c5 = compute_c5(best_h)
    best_time = time.time()

    for _ in range(1000):
        if time.time() - best_time > budget_s:
            break
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
        
        fitness = [compute_c5(ind) for ind in pop]
        sorted_idx = np.argsort(fitness)
        selected = [pop[i] for i in sorted_idx[:40]]
        
        offspring = []
        for _ in range(10):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)

                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            child = normalize(child)
            offspring.append(child)

            if compute_c5(ind) < best_c5:
                best_h = ind
                best_c5 = compute_c5(best_h)
                best_time = time.time()
