def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    sum_initial = np.sum(initial_h)

        initial_h = np.clip(initial_h * scale, 0, 1)
    else:
        initial_h = np.full(n_points, 0.5)
    
    population = [initial_h.copy()]

        h = np.random.uniform(0, 1, n_points)

        if sum_h:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.full(n_points, 0.5)
        population.append(h)

        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def normalize(h):
        sum_h = np.sum(h)
        if sum_h:
            scale = (n_points / 2) / sum_h
            return np.clip(h * scale, 0, 1)
