        for alpha in [0.05, 0.1, 0.2, 0.5]:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            trial_h = trial_h * target_sum / trial_sum
            corr_trial = np.correlate(trial_h, 1 - trial_h, mode='full')
            corr_s = corr_trial[s]
            trial_c5_s = corr_s * dx
            if trial_c5_s < best_trial_c5:
                best_trial_c5 = trial_c5_s
                best_alpha = alpha
        if best_alpha != 0:
            h = h - best_alpha * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
            current_c5 = compute_c5(h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5_final = np.max(corr_final) * dx
    return (best_h, best_c5_final, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
