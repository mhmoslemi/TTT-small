import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)

    def obj(x):
        corr = correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1)] * n_points
    try:
        res = minimize(obj, initial_scaled, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
    except:
        res = minimize(obj, initial_scaled, method='Nelder-Mead', bounds=bounds)
    best_h = res.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
