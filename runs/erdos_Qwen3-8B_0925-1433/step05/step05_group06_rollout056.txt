import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def scale_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / np.sum(h)
        return np.clip(h * scale, 0, 1)
    
    # Initialize structured population
    population_size = 50
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            h = np.zeros(n_points)
            h[::2] = 1
            h = scale_h(h)
        else:
            h = initial_h_values.copy()
            h = scale_h(h)
        population.append(h)
    
    best_h = scale_h(initial_h_values)
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for _ in range(200):  # Max 200 generations
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate and select
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx].copy()
            if best_c5 < 0.3807:
                break
        
        # Tournament selection
        tournament = np.random.choice(population_size, 5, replace=False)
        winners = np.argmin([fitness[i] for i in tournament])
        parent1, parent2 = population[tournament[winners]], population[tournament[winners + 1 % 5]]
        
        # Crossover
        child1 = (parent1 * np.random.rand() + parent2 * (1 - np.random.rand()))
        child2 = (parent2 * np.random.rand() + parent1 * (1 - np.random.rand()))
        new_pop = [child1, child2]
        
        # Mutation
        for h in new_pop:
            if np.random.rand() < 0.1:
                noise = np.random.uniform(-0.1, 0.1, n_points)
                h += noise
                h = scale_h(h)
        
        # Replace weakest
        new_pop.extend(population[:population_size - 2])
        population = [scale_h(h) for h in new_pop]
        
        # Local optimization
        top_indices = np.argsort(fitness)[:5]
        for i in top_indices:
            h = population[i].copy()
            h = scale_h(h)
            def objective(x):
                x = scale_h(x)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                return np.max(corr) * dx
            def constraint(x):
                return np.sum(x) - n_points / 2
            res = minimize(objective, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type':'eq','fun':constraint})
            if res.success:
                optimized = res.x
                optimized = scale_h(optimized)
                population[i] = optimized
                current_c5 = compute_c5(optimized)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized.copy()
                    if best_c5 < 0.3807:
                        break
    
    return (best_h, best_c5, n_points)
