import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values *= target_sum / np.sum(initial_h_values)
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed candidate
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed *= target_sum / np.sum(perturbed)
        
        # Local optimization
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        
        if result.success:

            c5_val = compute_c5(candidate, dx, n_points)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = candidate
    
    return (best_h, best_c5, n_points)
