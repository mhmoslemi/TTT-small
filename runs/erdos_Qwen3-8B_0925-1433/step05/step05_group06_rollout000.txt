import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / np.sum(h)
        return np.clip(h * scale, 0, 1)

    initial_h = initial_h_values.copy()
    initial_h = clip_scale(initial_h)
    best_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()

    population = []
    for _ in range(50):
        h = np.zeros(n_points)
        m = int(np.ceil(n_points / 5))
        for i in range(0, n_points, m * 2):
            h[i:i + m] = 1.0
        h = clip_scale(h)
        population.append(h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            a = np.random.randint(0, 50)
            b = np.random.randint(0, 50)
            parent1 = population[a]
            parent2 = population[b]
            child = (parent1 + parent2) / 2
            if np.random.rand() < 0.1:
                idx = np.random.randint(0, n_points)
                child[idx] = np.random.rand()
            child = clip_scale(child)
            new_population.append(child)
        population = new_population

        for h in population:
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()

        top_h = sorted(population, key=lambda x: compute_c5(x))[:5]
        for h in top_h:

            h = clip_scale(h)

                x = x.astype(float)

                return compute_c5(x)
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(clip_scale(x)) - n_points/2})
