import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial check
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Initial optimization attempt
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    res = minimize(compute_c5, initial_scaled, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    else:
        best_h = initial_scaled
        best_c5 = compute_c5(best_h)

    # Refinement loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Perturb current best
        pert = np.random.normal(0, 0.05, n_points)
        pert = np.clip(pert, -0.1, 0.1)
        mut = best_h + pert
        mut = np.clip(mut, 0, 1)
        s = np.sum(mut)
        mut = mut * target_sum / s
        # Optimize perturbed solution
        res = minimize(compute_c5, mut, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
    return (best_h, best_c5, n_points)
