import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize from initial_h_values, clip and rescale
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    num_iterations = 100
    for _ in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current solution
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum != 0:
            h_perturbed = h_perturbed * target_sum / h_perturbed_sum
        else:
            h_perturbed = np.zeros(n_points)
        # Local optimization
        result = minimize(
            lambda x: compute_c5(x),
            h_perturbed,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                h = current_h.copy()
    return (best_h, best_c5, n_points)
