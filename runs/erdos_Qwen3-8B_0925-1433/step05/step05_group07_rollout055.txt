import time

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.ones(n_points) * 0.5
        return np.clip(h * (n_points / 2) / s, 0, 1)
    
    h = initial_h_values.copy()

    best_h = h.copy()
    best_c5 = compute_c5(h)

    particles = [scale_h(p) for p in particles]
    particles = np.stack(particles)
    personal_best = particles.copy()
    global_best = particles[0].copy()
    best_c5 = compute_c5(global_best)
    
    while time.time() - start_time < budget_s:
