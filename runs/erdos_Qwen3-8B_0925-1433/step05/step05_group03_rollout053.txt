import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_initial = np.round(initial_h_values)
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    diff = current_sum - target_sum
    if diff != 0:
        if diff > 0:
            ones = np.where(h_initial == 1)[0]
            h_initial[ones[:int(diff)]] = 0
        else:
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:int(-diff)]] = 1
    h_initial = h_initial.astype(int)

    def mutate(h):
        idx1, idx2 = np.random.choice(n_points, size=2, replace=False)
        if h[idx1] == 0 and h[idx2] == 1:
            h[idx1], h[idx2] = 1, 0
        elif h[idx1] == 1 and h[idx2] == 0:
            h[idx1], h[idx2] = 0, 1
        else:
            h[idx1] ^= 1
            h[idx2] ^= 1
        return h

    pop_size = 50
    pop = [h_initial.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        h = np.zeros(n_points, dtype=int)
        h[indices] = 1
        pop.append(h)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    max_iter = 300

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(pop, size=2, replace=False)
            split = np.random.randint(1, n_points)
            child = np.concatenate([p1[:split], p2[split:]])
            child = mutate(child)
            new_pop.append(child)
        pop = new_pop[:pop_size]
        best_pop = min(pop, key=compute_c5)
        curr_c5 = compute_c5(best_pop)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = best_pop.copy()

    return (best_h, best_c5, n_points)
