import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_scaled = np.clip(h * (n_points / 2) / np.sum(h), 0, 1) if np.sum(h) else h * 0.5
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x_scaled = np.clip(x * (n_points / 2) / np.sum(x), 0, 1) if np.sum(x) else x * 0.5
        shifted = 1 - x_scaled
        corr = np.correlate(x_scaled, shifted, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    h_best = initial_h_values.copy()
    h_best = np.clip(h_best, 0, 1)
    h_best = np.clip(h_best * (n_points / 2) / np.sum(h_best), 0, 1) if np.sum(h_best) else h_best * 0.5
    best_c5 = compute_c5(h_best)

    particles = []
    for _ in range(50):
        if np.random.rand() < 0.25:
            h = np.random.uniform(0, 1, n_points)
        elif np.random.rand() < 0.2:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        elif np.random.rand() < 0.15:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1 if i % 2 == 0 else 0
        else:
            h = initial_h_values.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = np.clip(h, 0, 1)
        h = np.clip(h * (n_points / 2) / np.sum(h), 0, 1) if np.sum(h) else h * 0.5
        particles.append(h)

    gbest = h_best.copy()
    gbest_c5 = best_c5
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(len(particles)):
            res = minimize(objective, particles[i], method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = np.clip(h_opt * (n_points / 2) / np.sum(h_opt), 0, 1) if np.sum(h_opt) else h_opt * 0.5
                current_c5 = compute_c5(h_opt)
                if current_c5 < gbest_c5:
                    gbest = h_opt.copy()
                    gbest_c5 = current_c5
                    if gbest_c5 < 0.3807:
                        return (gbest, gbest_c5, n_points)
        if gbest_c5 < 0.3807:
            return (gbest, gbest_c5, n_points)

    return (gbest, gbest_c5, n_points)
