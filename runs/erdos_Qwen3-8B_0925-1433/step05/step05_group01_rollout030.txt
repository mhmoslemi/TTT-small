import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_h = np.array(initial_h_values, dtype=np.float64) * target_sum / np.sum(initial_h_values)
    best_h = current_h.copy()

        if time.time() - start_time > budget_s:

        corr = np.correlate(current_h, 1.0 - current_h, mode='full')
        max_k = np.argmax(corr)
        idx1 = (np.arange(n_points) + max_k) % n_points
        idx2 = (np.arange(n_points) - max_k) % n_points
        grad = ((1.0 - current_h[idx1]) - current_h[idx2]) * dx
        step_size = 0.1 * (1.0 - _ / 1000.0)
        h_new = current_h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new_sum = np.sum(h_new)
        if h_new_sum != 0:

        new_c5 = compute_c5(h_new, dx, n_points)
