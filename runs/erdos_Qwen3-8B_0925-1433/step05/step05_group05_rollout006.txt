import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def normalize_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        return h_scaled

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        return compute_c5(x)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    h_initial = initial_h_values.copy()
    h_initial = normalize_h(h_initial)
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()

    start_time = time.time()
    for _ in range(10):
        if np.random.rand() < 0.3:
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturb = normalize_h(h_initial + perturb)
        elif np.random.rand() < 0.4:
            x = np.linspace(0, 2, n_points)
            h_perturb = np.sin(np.pi * (x - 1)) + 1
            h_perturb = normalize_h(h_perturb)
        else:
            h_perturb = np.random.uniform(0, 1, n_points)
            h_perturb = normalize_h(h_perturb)
        
        res = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_opt = res.x
            h_opt = normalize_h(h_opt)
            current_c5 = compute_c5(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)

        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
