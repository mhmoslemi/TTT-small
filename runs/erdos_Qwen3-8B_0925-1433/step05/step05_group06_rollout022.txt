import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = np.clip(x * scale, 0, 1)
        return compute_c5(x_scaled)

    bounds = [(0, 1) for _ in range(n_points)]
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=50, tol=1e-6, mutation=(0.8, 1), 
                                   recombination=0.9, maxiter=100, disp=False, 
                                   polish=True, polish_xatol=1e-6, polish_maxiter=10)
    best_h = result.x
    best_c5 = compute_c5(best_h)

    def local_opt(h):
        corr = np.correlate(h, 1 - h, mode='full')
        worst_k = np.argmax(corr)
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x_scaled = np.clip(x * scale, 0, 1)
            corr_scaled = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return corr_scaled[worst_k] * dx
        def cons(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x if res.success else h

    best_h = local_opt(best_h)
    best_c5 = compute_c5(best_h)
