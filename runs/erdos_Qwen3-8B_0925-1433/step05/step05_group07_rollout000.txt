from scipy.optimize import minimize
import time

        max_corr = np.max(corr)
        return max_corr * dx
    
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)

        h = np.ones(n_points) * 0.5
    best_h = h.copy()

    start_time = time.time()
    
    while time.time() - start_time < budget_s:

        lag = max_index - (n_points - 1)
        
        def obj(x):
