    result = opt.differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin', 
        popsize=50, 
        maxiter=100, 
        tol=1e-6, 
        mutation=(0.5, 1), 
        recombination=0.9, 
        polish=True, 
        disp=False
    )
    
    best_h = result.x
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (best_h, result.fun, n_points)
    
    # Clip and scale best_h
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h == 0:
        scale = 1.0
    else:
        scale = (n_points / 2) / sum_h
    best_h_scaled = np.clip(best_h * scale, 0, 1)
    corr = np.correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
    worst_k = np.argmax(corr)
    
    # Local optimization
    def local_objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = x_clipped * scale
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return corr[worst_k]
    
    def constraint(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        return sum_x - n_points / 2
    
    bounds_local = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = best_h_scaled
    
    result_local = opt.minimize(
        local_objective, 
        initial_guess, 
        method='SLSQP', 
        bounds=bounds_local, 
        constraints={'type': 'eq', 'fun': constraint}, 
        tol=1e-6
    )
    
    if result_local.success:
        best_h_optimized = result_local.x
        best_h_optimized = np.clip(best_h_optimized, 0, 1)
        sum_h_opt = np.sum(best_h_optimized)
        if sum_h_opt == 0:
            scale_opt = 1.0
        else:
            scale_opt = (n_points / 2) / sum_h_opt
        best_h_optimized = np.clip(best_h_optimized * scale_opt, 0, 1)
        corr_opt = np.correlate(best_h_optimized, 1 - best_h_optimized, mode='full')
        final_c5 = np.max(corr_opt) * dx
        if final_c5 < result.fun:
            best_h = best_h_optimized
            result.fun = final_c5
    else:
        best_h = best_h
        result.fun = result.fun
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (best_h, result.fun, n_points)
    
    return (best_h, result.fun, n_points)
