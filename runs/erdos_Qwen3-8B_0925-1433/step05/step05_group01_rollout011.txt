import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = initial_h_values.copy()

        max_idx = np.argmax(corr)
        grad = np.zeros(n_points)
        m = max_idx - n_points + 1
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:

        current_h -= 0.1 * grad
        current_h = np.clip(current_h, 0, 1)
        current_sum = np.sum(current_h)
        current_h = current_h * target_sum / current_sum

def compute_c5(h, dx, n_points):
