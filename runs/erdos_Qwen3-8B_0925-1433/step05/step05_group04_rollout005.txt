from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def scale_to_sum(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return np.clip(h * (n_points / 2) / s, 0, 1)
    
    def local_refine(h):
        def objective(x):
            x = scale_to_sum(x)
            shifted = 1 - x
            corr = np.correlate(x, shifted, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}, tol=1e-6)
        return res.x
    
    h_initial = scale_to_sum(initial_h_values.copy())
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    pop_size = 50
    pop = [h_initial.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        if np.random.rand() < 0.3:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = scale_to_sum(h)
        pop.append(h)
    
    pop = np.array(pop)[:pop_size]
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        if gen % 10 == 0:
            print(f"Generation {gen}, Best C5: {best_c5}")
        
        new_pop = []
        for i in range(pop_size):

            else:
