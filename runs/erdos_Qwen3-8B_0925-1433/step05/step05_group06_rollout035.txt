    steps = 0
    max_steps = 1000

    while time.time() - start_time < budget_s and steps < max_steps:

            h[idx] *= np.random.uniform(0.9, 1.1)

            scale = (n_points / 2) / sum_h

            scale = 1

                return (best_h, current_c5, n_points)
