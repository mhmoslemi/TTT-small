def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h_sum = np.sum(h)
    if h_sum != 0:
        h *= target_sum / h_sum

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_c5(h, dx, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Local optimization
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    # Perturb and optimize
    for _ in range(3):

        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum == 0:
            continue
        perturbed = perturbed * target_sum / perturbed_sum
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
