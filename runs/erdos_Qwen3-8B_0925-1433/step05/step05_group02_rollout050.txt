import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Validate initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, target_sum):

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial optimization
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = objective(best_h)

    # Perturbation and refinement
    for _ in range(3):
        perturb = np.random.randn(n_points) * 0.05
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        result_perturb = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if objective(result_perturb.x) < best_c5:
            best_h = result_perturb.x
            best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
