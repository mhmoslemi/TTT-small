    best_h = h.copy()
    best_c5_val = best_c5
    
    def local_optimize(h_vectors):
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted = 1 - x
            corr = np.correlate(x, shifted, mode='full')
            return np.max(corr) * dx
        def constraint(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_vectors, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x
    
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    start_time = time.time()
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        refined = [local_optimize(p) for p in population]
        c5_values = [compute_c5(p) for p in refined]
        idx = np.argmin(c5_values)
        if c5_values[idx] < best_c5_val:
            best_c5_val = c5_values[idx]
            best_h = refined[idx].copy()
        parents = np.argsort(c5_values)[:population_size//2]
        parents = refined[parents]
        new_population = []
        while len(new_population) < population_size:
            if len(parents) < 2:
                new_population.append(np.random.uniform(0,1,n_points))
                continue
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
            sum_child = np.sum(child)
            if sum_child != 0:
                scale = (n_points / 2) / sum_child
                child = np.clip(child * scale, 0, 1)
            else:
                child = np.ones(n_points) * 0.5
            new_population.append(child)
            if np.random.rand() < 0.1:
                noise = np.random.uniform(-0.1, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child != 0:
                    scale = (n_points / 2) / sum_child
                    child = np.clip(child * scale, 0, 1)
                else:
                    child = np.ones(n_points) * 0.5
            new_population.append(child)
        population = new_population
    
    return (best_h, best_c5_val, n_points)
