            for idx in range(n_points):
                if np.random.rand() < CR:
                    trial[idx] = mutant[idx]
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.full(n_points, target_sum / n_points)
            else:
                trial = np.clip(trial * target_sum / trial_sum, 0, 1)
            trial_c5 = compute_c5(trial, dx, n_points)
            if trial_c5 < compute_c5(pop[i], dx, n_points):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        current_best = min(pop, key=lambda x: compute_c5(x, dx, n_points))
        curr_c5 = compute_c5(current_best, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()
    def obj(x):
        max_corr = 0.0
        for k in range(n_points):
            shifted = np.roll(x, k)
            corr = np.sum(x * (1 - shifted)) * dx
            if corr > max_corr:
                max_corr = corr
        return max_corr
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 100})
        best_h = res.x
        best_c5 = compute_c5(best_h, dx, n_points)
    except:
        pass
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    max_corr = 0.0
    for k in range(n_points):
        shifted = np.roll(h, k)
        corr = np.sum(h * (1 - shifted)) * dx
        if corr > max_corr:
            max_corr = corr
    return max_corr
