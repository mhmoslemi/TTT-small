import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def scale_clamp(h):
        s = np.sum(h)
        if s == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / s
        return np.clip(h * scale, 0, 1)
    
    h = initial_h_values.copy()
    h = scale_clamp(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    num_tries = 50
    
    for _ in range(num_tries):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h + perturb, 0, 1)
        h = scale_clamp(h)
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[np.argmax(corr)] * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)*n_points], constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = scale_clamp(h)
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
                if best_c5 < 0.3807:
                    break
        else:
            h = np.clip(h, 0, 1)
            h = scale_clamp(h)
    
    return (best_h, best_c5, n_points)
