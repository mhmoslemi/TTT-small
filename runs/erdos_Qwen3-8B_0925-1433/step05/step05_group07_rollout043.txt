        else:
            h = np.ones(n_points) * 0.5
        
        def objective(x):
            h_new = x
            shifted = 1 - h_new
            corr = np.correlate(h_new, shifted, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        def constraint(x):
            return np.sum(x) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint}
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            sum_opt = np.sum(h_opt)
            if sum_opt != 0:
                scale = (n_points / 2.0) / sum_opt
