import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    def generate_candidate(base):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        candidate = np.clip(base + perturb, 0, 1)
        sum_h = np.sum(candidate)
        if sum_h != 0:
            candidate = candidate * (n_points / 2) / sum_h
        return candidate

    candidates = [generate_candidate(initial_h_values) for _ in range(5)]
    for candidate in candidates:
        h = candidate.astype(float)
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)

        def obj(x):
            h_new = x
            shifted = 1 - h_new
            corr_val = np.correlate(h_new, shifted, mode='full')
            return corr_val[max_index] * dx

        def cons(x):
            return np.sum(x) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': cons}]
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

        if result.success:
            h_new = np.clip(result.x, 0, 1)

            current_c5 = compute_c5(h_new)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)

    remaining_time = budget_s - (time.time() - start_time)
