            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum != 0:
                h_minus = h_minus * target_sum / h_minus_sum
            else:
                h_minus = np.full(n_points, target_sum / n_points)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    pop_size = 20
    pop = []
    for _ in range(pop_size):
        if np.random.rand() < 0.8:
            perturb = np.random.normal(0, 0.05, n_points)
            candidate = initial_scaled + perturb
        else:
            candidate = np.random.rand(n_points)
            candidate = candidate * target_sum / np.sum(candidate)
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if candidate_sum != 0:
            candidate = candidate * target_sum / candidate_sum
        else:
            candidate = np.full(n_points, target_sum / n_points)
        pop.append(candidate)
    
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    max_iter = 300
    F = 0.8
    CR = 0.9
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            j, k, l = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[j] + F * (pop[k] - pop[l])
            trial = pop[i].copy()
            for idx in range(n_points):
                if np.random.rand() < CR:
                    trial[idx] = mutant[idx]
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.full(n_points, target_sum / n_points)
            else:
                trial = np.clip(trial * target_sum / trial_sum, 0, 1)
            trial_c5 = compute_c5(trial)
            if trial_c5 < compute_c5(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        current_best = min(pop, key=lambda x: compute_c5(x))
        curr_c5 = compute_c5(current_best)
        if curr_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = curr_c5
    
    learning_rate = 0.1
    for _ in range(10):
        grad = compute_gradient(best_h)
        best_h -= learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        best_h_sum = np.sum(best_h)
        if best_h_sum != 0:
            best_h = best_h * target_sum / best_h_sum
        else:
            best_h = np.full(n_points, target_sum / n_points)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
