    for _ in range(pop_size - 1):
        perturbed = initial_h_values + np.random.uniform(-perturbation, perturbation, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed != 0:
            scale = (n_points / 2) / sum_perturbed
            perturbed = np.clip(perturbed * scale, 0, 1)
        population.append(perturbed)
        velocities.append(np.zeros_like(perturbed))
        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr) * dx
        personal_best.append(perturbed)
        personal_best_c5.append(current_c5)

    # Find initial global best
    best_c5 = np.inf
    for i in range(pop_size):
        if personal_best_c5[i] < best_c5:
            best_c5 = personal_best_c5[i]
            global_best = population[i]
    best_h = global_best.copy()
    previous_best_c5 = best_c5
    stagnation_count = 0
    start_time = time.time()

    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break

        # Update each particle
        for i in range(pop_size):
            # Update velocity
            velocity = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocity = w * velocity + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            velocities[i] = velocity

            # Update position
            new_h = population[i] + velocity
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new != 0:
                scale = (n_points / 2) / sum_new
                new_h = np.clip(new_h * scale, 0, 1)
            population[i] = new_h

            # Evaluate new_h
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            current_c5 = np.max(corr) * dx

            # Update personal best
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = new_h.copy()
                personal_best_c5[i] = current_c5

        # Update global best
        for i in range(pop_size):
            if personal_best_c5[i] < best_c5:
                best_c5 = personal_best_c5[i]
                global_best = personal_best[i]

        # Check for improvement
        if best_c5 < previous_best_c5:
            previous_best_c5 = best_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= stagnation_threshold:
                break

        # Check if target is reached
        if best_c5 < 0.3807:
            return (global_best, best_c5, n_points)

    return (best_h, best_c5, n_points)
