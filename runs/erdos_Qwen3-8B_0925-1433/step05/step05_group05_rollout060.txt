import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Initialize population
    population = []
    for _ in range(50):
        h = initial_h_values.copy()

            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        population.append(h)

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[best_idx].copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
        selected = np.argsort(fitness)[:40]
        new_population = []
        for i in range(0, 40, 2):
            p1 = population[selected[i]]
            p2 = population[selected[i+1]]
            child = np.random.choice([p1, p2], size=n_points)
            sum_h = np.sum(child)
            if sum_h > 0:
                scale = (n_points / 2) / sum_h
                child = np.clip(child * scale, 0, 1)
            else:
                child = np.ones(n_points) * 0.5
            new_population.append(child)
        for i in range(len(new_population)):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                new_pop = new_population[i] + noise
                sum_h = np.sum(new_pop)
                if sum_h > 0:
                    scale = (n_points / 2) / sum_h
