from scipy.optimize import minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')

def run(seed=42, budget_s=600):

    initial_guess = np.clip(initial_h_values, 0, 1)
    initial_guess = initial_guess * target_sum / np.sum(initial_guess)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
