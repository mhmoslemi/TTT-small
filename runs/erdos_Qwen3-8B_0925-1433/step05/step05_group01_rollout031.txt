import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    current_h = np.copy(initial_h_values)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Perturbation
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = perturbed * target_sum / perturbed_sum
        perturbed_c5 = compute_c5(perturbed, 2.0 / n_points, n_points)
        if perturbed_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = perturbed_c5
        
        # Gradient step
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - n_points + 1
        if k < 0:
            k = 0
        if k >= n_points:
            k = n_points - 1
        
        # Compute gradient for lag k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
