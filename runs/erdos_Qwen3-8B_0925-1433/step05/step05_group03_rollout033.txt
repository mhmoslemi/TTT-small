import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        max_integral = 0.0
        for shift in range(n_points):
            integral = 0.0
            for x in range(n_points - shift):
                integral += h[x] * (1.0 - h[x + shift])
            integral *= dx
            if integral > max_integral:
                max_integral = integral
        return max_integral

    def obj(x):
        return compute_c5(x)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guess = initial_h_values * target_sum / np.sum(initial_h_values)

    start_time = time.time()
    try:
        result = minimize(obj, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 200})
        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = initial_guess
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
