import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.ones(n_points) * 0.5
        return np.clip(h * (n_points / 2) / s, 0, 1)
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    # Initial check and scaling
    h_initial = scale_h(initial_h_values.copy())
    h_initial = np.clip(h_initial, 0, 1)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization
    result = differential_evolution(compute_c5, bounds, strategy='best1bin',
                                   mutation=(0.8, 1), recombination=0.9,
                                   popsize=40, tol=1e-6, maxiter=200,
                                   disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_h = scale_h(best_h)
    best_c5 = compute_c5(best_h)
    
    if best_c5 < 0.3807:

    def objective_slsqp(x):
        x = np.clip(x, 0, 1)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    res = minimize(objective_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp,

        best_h = scale_h(best_h)
        best_c5 = compute_c5(best_h)
        if best_c5 < 0.3807:
