from scipy.optimize import minimize
import time

        h_scaled = np.clip(h * scale, 0, 1)

def tournament_selection(population, size):
    candidates = np.random.choice(population, size, replace=False)
    return [c for c in candidates if compute_c5(c) == min([compute_c5(c) for c in candidates])][0], [c for c in candidates if compute_c5(c) == min([compute_c5(c) for c in candidates])][0]

def crossover(parent1, parent2):
    split = np.random.randint(1, len(parent1) - 1)
    child = np.concatenate([parent1[:split], parent2[split:]])
    return child

def mutate(individual):
    for i in range(len(individual)):
        if np.random.rand() < 0.1:
            individual[i] += np.random.uniform(-0.2, 0.2)
    return individual
