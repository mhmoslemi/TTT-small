        best_c5_alpha = compute_c5(h - alpha * grad)
        for a in [0.01, 0.05, 0.1, 0.2, 0.5]:
            curr_h = h - a * grad
            curr_h = np.clip(curr_h, 0, 1)
            sum_curr = np.sum(curr_h)
            scale_curr = (n_points / 2) / sum_curr if sum_curr != 0 else 1
            curr_h = np.clip(curr_h * scale_curr, 0, 1)
            curr_c5 = compute_c5(curr_h)
            if curr_c5 < best_c5_alpha:
                best_c5_alpha = curr_c5
                best_alpha = a
        h = h - best_alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_h = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = np.clip(h * scale_h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if np.random.rand() < 0.1:
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale_h = (n_points / 2) / sum_h if sum_h != 0 else 1
            h = np.clip(h * scale_h, 0, 1)
    return (best_h, best_c5, n_points)
