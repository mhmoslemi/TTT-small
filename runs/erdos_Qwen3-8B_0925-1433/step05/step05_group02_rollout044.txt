import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full')
        return np.max(c) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0., 1.) for _ in range(n_points)]
    cons_dict = [{'type': 'eq', 'fun': cons}]

    def perturb(x):
        return np.clip(x * (1 + 0.1 * np.random.randn(n_points)), 0, 1) * target_sum / np.sum(x)

    initial_guess = np.clip(initial_h_values, 0, 1)
    initial_guess = initial_guess * target_sum / np.sum(initial_guess)

    best_h = initial_guess
    best_c5 = obj(initial_guess)
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        p = perturb(initial_guess)
        res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons_dict, tol=1e-8)
        if obj(res.x) < best_c5:
            best_h, best_c5 = res.x, obj(res.x)
        initial_guess = res.x

    return (best_h, best_c5, n_points)
