import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 50
    population = []
    # Start with initial_h_values scaled
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    if sum_initial != 0:
        scale = (n_points / 2) / sum_initial
        initial_h = np.clip(initial_h * scale, 0, 1)
    population.append(initial_h)
    
    # Generate rest of the population
    for _ in range(population_size - 1):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            scale = (n_points / 2) / sum_new
            new_h = np.clip(new_h * scale, 0, 1)
        population.append(new_h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    
    start_time = time.time()
    
    # Run for a fixed number of generations
    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
