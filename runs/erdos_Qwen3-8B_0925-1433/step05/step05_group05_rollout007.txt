    def objective(x):

        x = np.clip(x, 0, 1)

        sum_x = np.sum(x)

        if sum_x == 0:

        x_scaled = np.clip(x * scale, 0, 1)

        shifted = 1 - x_scaled

        corr = np.correlate(x_scaled, shifted, mode='full')

        return max_corr * dx

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1e6
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx

    def optimize(c):
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x_scaled = np.clip(x * scale, 0, 1)
            shifted = 1 - x_scaled
            corr = np.correlate(x_scaled, shifted, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, c, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6, options={'maxiter': 100})
            if res.success:
                h_opt = res.x

                sum_h_opt = np.sum(h_opt)
                if sum_h_opt == 0:
                    h_opt = np.full(n_points, 0.5)

                    scale_opt = (n_points / 2) / sum_h_opt
                    h_opt = np.clip(h_opt * scale_opt, 0, 1)
                return h_opt, compute_c5(h_opt)
        except:

        return c.copy(), compute_c5(c)

    h = initial_h_values.copy()
