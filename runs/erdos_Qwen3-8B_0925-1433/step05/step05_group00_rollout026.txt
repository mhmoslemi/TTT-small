import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    pop = [initial_scaled.copy() for _ in range(20)]
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind, dx, n_points) for ind in pop]
        top_idx = np.argsort(fitness)[:10]
        new_pop = []
        for i in range(10):
            parent = pop[top_idx[i]]
            if best_c5 < 0.38:
                sigma = 0.03 * (1 - best_c5)
            else:
                sigma = 0.02
            noise = np.random.normal(0, sigma, n_points)
            mut = parent + noise
            mut = np.clip(mut, 0, 1)
            s = np.sum(mut)
            mut = mut * target_sum / s
            new_pop.append(mut)
        refined = []
        for h in new_pop:
            cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
            bounds = Bounds(0, 1)
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            refined.append(res.x)
        pop = refined + pop[:10]
        for h in refined:
            curr = compute_c5(h, dx, n_points)
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
