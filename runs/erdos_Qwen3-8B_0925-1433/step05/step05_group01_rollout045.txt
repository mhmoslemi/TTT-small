import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_h = initial_h_values.copy()
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    start_time = time.time()
    T = 1.0
    for _ in range(800):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        new_h = current_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = compute_c5(new_h, dx, n_points)
        if np.random.random() < np.exp(-(new_c5 - best_c5) / T):
            current_h = new_h.copy()
            best_c5 = new_c5
        if _ % 20 == 0:
            def obj(x):
                return compute_c5(x, dx, n_points)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0, 1)] * n_points
            res = minimize(obj, current_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                current_h = res.x
                curr_c5 = compute_c5(current_h, dx, n_points)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = current_h.copy()
