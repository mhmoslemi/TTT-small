import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(initial_h_values * scale, 0, 1)
    
    pop_size = 50
    population = np.random.uniform(0, 1, (pop_size, n_points))
    population[0] = h.copy()
    
    for i in range(pop_size):
        population[i] = np.clip(population[i], 0, 1)
        population[i] *= (n_points / 2) / np.sum(population[i])
    
    personal_best = population.copy()
    global_best = population[0].copy()
    best_c5 = float('inf')
    
    def compute_c5(h_):
        shifted = 1 - h_
        corr = np.correlate(h_, shifted, mode='full')
        return np.max(corr) * dx
    
    for p in population:
        c5 = compute_c5(p)
        if c5 < best_c5:
            best_c5 = c5
            global_best = p.copy()
    
    start_time = time.time()
    w, c1, c2 = 0.7, 1.5, 1.5
    velocities = np.zeros((pop_size, n_points))
    max_iter = 150
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] *= (n_points / 2) / np.sum(population[i])
        
        for i in range(pop_size):
            curr_c5 = compute_c5(population[i])
            if curr_c5 < compute_c5(personal_best[i]):
                personal_best[i] = population[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    global_best = population[i].copy()
    
    return (global_best, best_c5, n_points)
