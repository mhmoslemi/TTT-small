def run(seed=42, budget_s=600):

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)

        scale = 1.0

        scale = (n_points / 2) / sum_h

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        k = k_index - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            if 0 <= i_plus_k < n_points:
                term1 = 1.0 - h[i_plus_k]
            else:
                term1 = 0.0
            i_minus_k = i - k
            if 0 <= i_minus_k < n_points:
                term2 = h[i_minus_k]

                term2 = 0.0
            gradient[i] = (term1 - term2) * dx
        
        best_alpha = 0.1
        best_c5_alpha = float('inf')
