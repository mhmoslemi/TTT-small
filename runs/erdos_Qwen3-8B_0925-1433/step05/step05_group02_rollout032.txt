import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

        candidate = candidate / np.sum(candidate) * target_sum
        population.append(candidate)

    best_c5 = np.inf

    start_time = time.time()

            break

        top_idx = np.argsort(scores)[:5]
        top_candidates = [population[i] for i in top_idx]

        # Local optimization on top candidates
        new_population = []
        for candidate in top_candidates:
            bounds = [(0, 1)] * n_points

            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx

        # Replace worst with mutated best
        scores = np.array([compute_c5(p, dx, n_points) for p in new_population])
        worst_idx = np.argmax(scores)
