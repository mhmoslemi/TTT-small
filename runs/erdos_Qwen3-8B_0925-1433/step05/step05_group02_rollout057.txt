import time
from scipy.optimize import minimize

            candidate = initial_h_values.copy()
        else:

            candidate /= np.sum(candidate)
            candidate *= target_sum
        population.append(candidate)
    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            # Mutation step
            j, k, l = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[j], population[k], population[l]
            mutant = a + 0.8 * (b - c)
            # Crossover step
            trial = np.where(np.random.rand(n_points) < 0.9, mutant, population[i])
            # Constraint handling
            trial_clip = np.clip(trial, 0, 1)
            trial_sum_clip = np.sum(trial_clip)
            if trial_sum_clip == 0:
                trial = population[i]
            else:
                trial = trial_clip * target_sum / trial_sum_clip
            # Selection
            original_c5 = compute_c5(population[i], dx, n_points)
            trial_c5 = compute_c5(trial, dx, n_points)
            if trial_c5 < original_c5:

            else:
                new_population.append(population[i])
        population = new_population

    # Find best solution in population
    best_c5 = np.inf

    # Local optimization with L-BFGS-B for better convergence

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
