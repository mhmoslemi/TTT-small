import numpy as np
import time
from scipy.optimize import minimize

    dx = 2.0 / n_points

    initial_h_values *= target_sum / np.sum(initial_h_values)
    
    population = [initial_h_values.copy() for _ in range(50)]
    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            j, k, l = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[j], population[k], population[l]
            mutant = a + 0.6 * (b - c)
            trial = np.where(np.random.rand(n_points) < 0.8, mutant, population[i])
            trial_clip = np.clip(trial, 0, 1)
            trial_sum_clip = np.sum(trial_clip)
            if trial_sum_clip == 0:
                trial = population[i]
            else:
                trial = trial_clip * target_sum / trial_sum_clip
            # Local optimization
            bounds = [(0, 1)] * n_points
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            local_result = minimize(obj, trial, method='L-BFGS-B', bounds=bounds, constraints=constraints)
            local_trial = local_result.x
            original_c5 = compute_c5(population[i], dx, n_points)
            trial_c5 = compute_c5(local_trial, dx, n_points)
            if trial_c5 < original_c5:
                new_population.append(local_trial)
            else:
                new_population.append(population[i])
        population = new_population

    # Find best solution
    best_c5 = np.inf

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
