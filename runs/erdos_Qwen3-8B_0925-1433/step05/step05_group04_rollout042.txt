import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def scale_to_sum(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones_like(h) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        x = scale_to_sum(x)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2
    
    h_best = initial_h_values.copy()
    h_best = scale_to_sum(h_best)
    best_c5 = compute_c5(h_best)
    
    n_pop = 100
    population = [scale_to_sum(np.random.uniform(0, 1, n_points)) for _ in range(n_pop)]
    for i in range(n_pop):
        population[i] = scale_to_sum(np.clip(population[i], 0, 1))
    
    start_time = time.time()
    
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        mutants = []
        for i in range(n_pop):
            a, b, c = np.random.choice(n_pop, 3, replace=False)
            mutant = population[a] + np.random.uniform(-0.8, 0.8, n_points) * (population[b] - population[c])
            mutants.append(scale_to_sum(np.clip(mutant, 0, 1)))
        
        for i in range(n_pop):
            trial = mutants[i]
            trial = np.clip(trial, 0, 1)
            trial = scale_to_sum(trial)
            if compute_c5(trial) < compute_c5(population[i]):
                population[i] = trial
        
        top_indices = np.argsort([compute_c5(h) for h in population])[:int(n_pop*0.1)]
        for idx in top_indices:
            h = population[idx]
            h = np.clip(h, 0, 1)
            cons = [{'type': 'eq', 'fun': constraint}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x

                if compute_c5(h_opt) < compute_c5(population[idx]):
                    population[idx] = h_opt

                h_best = h.copy()
                if best_c5 < 0.3807:
                    return (h_best, best_c5, n_points)
    
    return (h_best, best_c5, n_points)
