import time
from scipy.optimize import minimize, Bounds, LinearConstraint

    global initial_h_values
    n_points = len(initial_h_values)

    improve_count = 0

            break

        top_idx = np.argsort(fitness)[:10]
        new_pop = []
        for i in top_idx:
            parent = pop[i]
            noise = np.random.normal(0, mutation_scale, n_points)
            mut = np.clip(parent + noise, 0, 1)
            s = np.sum(mut)
            mut = mut * target_sum / s

                return np.max(corr) * dx

            res = minimize(obj, mut, method='SLSQP', bounds=bounds, constraints=cons)
