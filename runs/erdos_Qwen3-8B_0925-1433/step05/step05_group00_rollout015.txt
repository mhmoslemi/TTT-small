        possible_steps = [0.05, 0.2, 0.5]
        best_trial = None
        best_new_max = current_max
        
        for alpha in possible_steps:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            if trial_sum == 0:
                continue
            trial_h = trial_h * (target_sum / trial_sum)
            new_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_new_max:
                best_new_max = new_max
                best_trial = trial_h
        
        if best_trial is not None:
            h = best_trial
            if best_new_max < best_c5:
                best_c5 = best_new_max
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
