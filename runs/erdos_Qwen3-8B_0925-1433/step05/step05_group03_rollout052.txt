import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 44
    target_sum = n_points // 2
    def init_population():
        h = np.random.randint(0, 2, size=n_points)
        while np.sum(h) != target_sum:
            idx = np.random.choice(n_points)
            h[idx] ^= 1
        return h
    pop_size = 50
    pop = [init_population() for _ in range(pop_size)]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = pop[0]
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    max_iter = 150
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            parent1 = pop[a]
            parent2 = pop[b]
            child = parent1.copy()
            idx = np.random.randint(n_points)
            child[idx] = parent2[idx]
            while np.sum(child) != target_sum:
                idx = np.random.randint(n_points)
                child[idx] ^= 1
            new_pop.append(child)
        pop = new_pop
        curr_best = min(pop, key=compute_c5)
        curr_c5 = compute_c5(curr_best)
        if curr_c5 < best_c5:
            best_h = curr_best
            best_c5 = curr_c5
    
    def obj(x):
        return compute_c5(x)
    
    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:

    except:
        pass
    
    return (np.round(best_h, 6), best_c5, n_points)
