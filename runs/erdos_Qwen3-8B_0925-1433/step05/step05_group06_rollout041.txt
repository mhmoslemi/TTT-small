            child1 = np.concatenate([p1[:cp], p2[cp:]])
            child2 = np.concatenate([p2[:cp], p1[cp:]])
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    child1[j] += np.random.uniform(-0.1, 0.1)
                    child2[j] += np.random.uniform(-0.1, 0.1)
            for c in [child1, child2]:
                c = np.clip(c, 0, 1)
                sum_c = np.sum(c)
                scale = (n_points / 2) / sum_c if sum_c != 0 else 1.0
                c = np.clip(c * scale, 0, 1)
                new_pop.append(c)

        population = new_pop

        # Local optimization
        for h in population[:min(5, len(population))]:
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)
            def obj(x):
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)
                if sum_x == 0:
                    return 1e6
                scale_x = (n_points / 2) / sum_x
                x = np.clip(x * scale_x, 0, 1)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                return np.max(corr) * dx
            def cons(x):
                x = np.clip(x, 0, 1)
                return np.sum(x) - n_points / 2
            cons = [{'type': 'eq', 'fun': cons}]
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                sum_opt = np.sum(h_opt)
                scale_opt = (n_points / 2) / sum_opt if sum_opt != 0 else 1.0
                h_opt = np.clip(h_opt * scale_opt, 0, 1)
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
