import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):

    best_c5 = compute_c5(best_h, dx, n_points)

    for _ in range(200):
        if time.time() - start_time > budget_s:

        # Greedy selection: maintain top 10 best solutions
        top_idx = np.argsort([compute_c5(ind, dx, n_points) for ind in [best_h]] * 10)[:10]
        new_pop = []
        for _ in range(10):

            noise = np.random.normal(0, 0.015, n_points)
            mut = parent + noise
            mut = np.clip(mut, 0, 1)
            s = np.sum(mut)

            new_pop.append(mut)
        # Local optimization for top candidates

            bounds = Bounds(0, 1)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
