import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def scale_to_sum(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.ones(n_points) * 0.5
        return np.clip(h * (n_points / 2) / s, 0, 1)
    
    def optimize_local(h):
        def objective(x):
            x = np.clip(x, 0, 1)
            s = np.sum(x)
            if s == 0:
                return 1e6
            x = np.clip(x * (n_points / 2) / s, 0, 1)
            shifted = 1 - x
            corr = np.correlate(x, shifted, mode='full')
            return np.max(corr) * dx
        
        def constraint(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return scale_to_sum(res.x)
    
    initial_h = initial_h_values.copy()

        h = np.sin(np.linspace(0, np.pi, n_points)) + 1

    pop = np.array(pop)
    
    start_time = time.time()
