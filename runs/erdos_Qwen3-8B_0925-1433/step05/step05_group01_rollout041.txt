import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        
        def objective(x):
            return compute_c5(x, dx, n_points)
        
        bounds = [(0, 1)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, current_h, method='COBYLA', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            new_h = result.x
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            new_h_perturbed = np.clip(new_h + perturbation, 0, 1)
            new_h_perturbed = new_h_perturbed * target_sum / np.sum(new_h_perturbed)
            
            for h in [new_h, new_h_perturbed]:
                h_sum = np.sum(h)
                h_scaled = h * target_sum / h_sum
                c5 = compute_c5(h_scaled, dx, n_points)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_scaled.copy()
        
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(current_h + perturbation, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        c5 = compute_c5(perturbed, dx, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed.copy()
        
        current_h = best_h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
