    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    for _ in range(600):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_c5 = max_corr * dx
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        max_lag_idx = np.argmax(corr)
        lag = max_lag_idx
        grad = np.zeros(n_points)
        for j in range(n_points):
            part1 = (1.0 - h[j + lag]) * dx if j + lag < n_points else 0.0
            part2 = h[j - lag] * dx if j - lag >= 0 else 0.0
            grad[j] = part1 - part2
        alpha_candidate = [0.05, 0.1, 0.2, 0.5]
        best_trial = None
        best_trial_c5 = float('inf')
        for alpha in alpha_candidate:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            trial_h = trial_h * target_sum / trial_sum
            trial_c5 = compute_c5(trial_h, dx, n_points)
            if trial_c5 < best_trial_c5:
                best_trial_c5 = trial_c5
                best_trial = trial_h.copy()
        if best_trial_c5 < max_c5:
            h = best_trial
        else:
            noise = np.random.normal(0, 0.05, n_points)
            trial_h = h + noise
            trial_h = np.clip(trial_h, 0, 1)
            trial_sum = np.sum(trial_h)
            trial_h = trial_h * target_sum / trial_sum
            trial_c5 = compute_c5(trial_h, dx, n_points)
            if trial_c5 < max_c5:
                h = trial_h
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
