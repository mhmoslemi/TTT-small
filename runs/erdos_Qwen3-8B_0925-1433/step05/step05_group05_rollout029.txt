import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    else:
        n_points = 45
    dx = 2.0 / n_points
    population_size = 50
    
    def normalize(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [normalize(initial_h_values)]
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            population.append(normalize(h))
    else:
        population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time >= budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        
        best_idx = np.argmin(fitness)
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[best_idx].copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
        
        indices = np.argsort(fitness)
        top_individuals = [population[i] for i in indices[:40]]
        
        offspring = []
        for i in range(0, len(top_individuals), 2):
            if i + 1 < len(top_individuals):
                p1, p2 = top_individuals[i], top_individuals[i + 1]
                child = np.random.choice([p1, p2], size=n_points)
                mutate = np.random.rand(n_points) < 0.1
                child[mutate] += np.random.normal(0, 0.05, np.sum(mutate))
                child = normalize(child)
                offspring.append(child)
        
        population = top_individuals[:40] + offspring[:10]
    
    return (best_h, best_c5, n_points)
