    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1
    h = np.clip(h * scale, 0, 1)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    start_time = time.time()
    iter_count = 0
    max_iter = 10000
    step_size = 0.1
    perturb_freq = 50
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] = (1 - h[j] - h[i]) * dx
        alpha = 0.1
        improved = False
        for _ in range(5):
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if sum_h_new != 0:
                scale_new = (n_points / 2) / sum_h_new
            else:
                scale_new = 1
            h_new = np.clip(h_new * scale_new, 0, 1)
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                best_h = h.copy()
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved and iter_count % perturb_freq == 0:
            h_perturb = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h_perturb[i] += np.random.uniform(-0.1, 0.1)
            h_perturb = np.clip(h_perturb, 0, 1)
            sum_perturb = np.sum(h_perturb)
            if sum_perturb != 0:
                scale_perturb = (n_points / 2) / sum_perturb
            else:
                scale_perturb = 1
            h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
            new_c5_perturb = compute_c5(h_perturb, dx)
            if new_c5_perturb < best_c5:
                h = h_perturb.copy()
                best_c5 = new_c5_perturb
                best_h = h.copy()
        if best_c5 < 0.3807:
            return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
