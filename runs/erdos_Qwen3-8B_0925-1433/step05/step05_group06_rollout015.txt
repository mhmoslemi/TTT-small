    bounds_local = [(0.0, 1.0) for _ in range(n_points)]
    res = minimize(local_objective, best_h, method='SLSQP', 
                   bounds=bounds_local, constraints=cons, 
                   tol=1e-6, options={'maxiter': 20})
    
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h, final_c5, n_points)
