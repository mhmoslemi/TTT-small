import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Scale initial guess
    h = initial_h_values.copy() * target_sum / np.sum(initial_h_values)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    num_restarts = 100
    time_per_restart = budget_s / num_restarts

    for _ in range(num_restarts):
        # Perturb current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)

        # Optimization
        start_time = time.time()
        result = minimize(
            lambda x: compute_c5(x, dx, n_points),
            h_perturbed,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=LinearConstraint(np.ones(n_points), [target_sum], [target_sum]),
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        elapsed = time.time() - start_time

        if elapsed > time_per_restart:
            continue
        
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
