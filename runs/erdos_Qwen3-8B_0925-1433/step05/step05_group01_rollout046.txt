import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        
        # Determine the shift k based on max_idx
        if max_idx < n_points:
            k = max_idx
        else:
            k = max_idx - n_points + 1
        
        # Compute gradient vector
        gradient = np.zeros(n_points)
        if k > 0:
            end = n_points - k
            gradient[:end] = (1 - current_h[k:])[::-1] * dx
        elif k < 0:
            start = -k
            gradient[start:] = (1 - current_h[:n_points + k]) * dx
        else:
            gradient[:] = (1 - current_h) * dx
        
        # Gradient descent step
        step_size = 0.1
        current_h -= step_size * gradient
        current_h = np.clip(current_h, 0.0, 1.0)
        current_h = current_h * target_sum / np.sum(current_h)
        
        # Random perturbation every 50 steps
        if _ % 50 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            current_h += noise
            current_h = np.clip(current_h, 0.0, 1.0)
            current_h = current_h * target_sum / np.sum(current_h)
        
        current_c5 = compute_c5(current_h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
