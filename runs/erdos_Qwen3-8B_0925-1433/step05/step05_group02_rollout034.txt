import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = result.fun

    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) == 0:
            continue
        new_h = new_h * target_sum / np.sum(new_h)
        result = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return (best_h, best_c5, n_points)
