import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    initial_h = initial_h_values.copy()

    for _ in range(n_pop // 2):
        h = np.random.uniform(0, 1, n_points)
        h = scale_h(h)
        pop.append(h.copy())
    for _ in range(n_pop // 4):
        h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = np.clip(h, 0, 1)
        h = scale_h(h)
        pop.append(h.copy())
    pop = np.array(pop)
    
    def mutate(h):
        h = h.copy()
        idx = np.random.choice(n_points, size=10, replace=False)
        h[idx] += np.random.normal(0, 0.1, size=10)
        h = np.clip(h, 0, 1)
        h = scale_h(h)
        return h
    
    def crossover(a, b):
        return (a + b) / 2
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        fitness = np.array([compute_c5(p) for p in pop])

        top_pop = pop[top_idx]
        new_pop = []
        for p in top_pop:
            if np.random.rand() < 0.8:
                p = crossover(p, np.random.choice(top_pop))
            p = mutate(p)
            new_pop.append(p)
        new_pop += [np.random.uniform(0, 1, n_points) for _ in range(n_pop - len(new_pop))]

                if sum_x == 0:
                    return 1e6
                scale = (n_points / 2) / sum_x
                x = np.clip(x * scale, 0, 1)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x

                h_opt = scale_h(h_opt)

                    h = h_opt
