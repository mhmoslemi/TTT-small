        T *= 0.95
        
        # Local refinement every 20 steps
        if step % 20 == 0:
            # Find current max lag
            current_c5, current_k = compute_c5_and_max_lag(current_h, dx, n_points)
            # Define the objective function for local refinement
            def objective(x):
                overlap = 0.0
                for i in range(n_points - current_k):
                    overlap += x[i] * (1 - x[i + current_k]) * dx
                return overlap
            
            # Define constraints
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Run COBYLA optimizer
            if step >= 19:
                result = minimize(objective, current_h, method='COBYLA', 
                                  bounds=bounds, constraints=cons, options={'maxiter': 100, 'disp': False})
                if result.success:
                    current_h = result.x
                    current_c5_new, _ = compute_c5_and_max_lag(current_h, dx, n_points)
                    if current_c5_new < best_c5:
                        best_c5 = current_c5_new
                        best_h = current_h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5_and_max_lag(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr) * dx
    max_idx = np.argmax(corr)
    return max_val, max_idx
