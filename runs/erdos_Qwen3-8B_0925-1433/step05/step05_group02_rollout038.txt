import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    h = h * target_sum / h_sum if h_sum > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1)] * n_points

    # Initial optimization with SLSQP
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})

    # Perturb and re-optimize for escaping local minima
    for _ in range(3):
        if time.time() > budget_s:
            break
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        trial = np.clip(result.x + perturb, 0, 1)
        trial = trial * target_sum / np.sum(trial)
        res = minimize(objective, trial, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 50})
        if res.fun < result.fun:
            result = res

    best_h = result.x
    c5_val = objective(best_h)
    return (best_h, c5_val, n_points)
