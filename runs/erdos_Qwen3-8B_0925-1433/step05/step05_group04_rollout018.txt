from scipy.optimize import minimize
import time

        return np.max(corr) * dx
    
    def scale_h(h):
        s = np.sum(h)

            return np.full_like(h, 0.5)
        scale = (n_points / 2) / s

            mutant.append(scale_h(np.clip(base, 0, 1)))

                h = h * (n_points / 2) / s
            
            def objective(x):
                x = np.clip(x, 0, 1)

                return s - n_points / 2
