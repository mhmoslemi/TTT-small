import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values with perturbation
    def perturb(h):
        return np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)

    candidates = [initial_h_values.copy()]
    for _ in range(49):
        candidates.append(perturb(initial_h_values))
    candidates = [np.clip(c, 0, 1) * target_sum / np.sum(np.clip(c, 0, 1)) for c in candidates]

    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_candidates = []
        for c in candidates:
            # Mutation
            pert = 0.1 * np.random.randn(n_points)
            mutant = np.clip(c + pert, 0, 1)
            mutant_sum = np.sum(mutant)
            mutant = mutant * target_sum / mutant_sum
            # Local optimization
            res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                           mutant, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                           constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            new_candidates.append(res.x)
        candidates = new_candidates
        # Evaluate and select best
        for c in candidates:
            c5 = np.max(np.correlate(c, 1 - c, mode='full')) * dx
            if c5 < best_c5:
