import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Clip and normalize
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, target_sum):
        h = h * target_sum / h_sum

    def obj(x):
        corr = correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Local optimization
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons)
    best_c5 = result.fun
    best_h = result.x

    # Perturb and refine
    for _ in range(5):
        perturb = 0.1 * np.random.randn(n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons)
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
