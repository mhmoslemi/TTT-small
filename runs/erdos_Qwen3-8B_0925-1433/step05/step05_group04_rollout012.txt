from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

    # Initial h
    h = initial_h_values.copy()
    h = scale_h(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def objective(x):
        x = scale_h(x)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    # Local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    # Genetic algorithm
    population = [best_h.copy() for _ in range(50)]
    start_time = time.time()
