import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def scale_to_integral(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return np.clip(h * (n_points / 2) / s, 0, 1)

    h = scale_to_integral(initial_h_values.copy())
    best_c5 = compute_c5(h)
    best_h = h.copy()

    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        p = np.random.uniform(0, 1, n_points)
        p = scale_to_integral(p)
        population.append(p)

    def obj(x):
        x = scale_to_integral(x)
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            h = population[i].copy()
            if np.random.rand() < 0.3:
                h = np.random.uniform(0, 1, n_points)

                h = h.copy()
                idx = np.random.choice(n_points, size=3, replace=False)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.2, size=3), 0, 1)
            h = scale_to_integral(h)
            if compute_c5(h) < best_c5:
                best_h, best_c5 = h.copy(), compute_c5(h)
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)

        for i in range(min(5, pop_size)):
            h = population[i].copy()
            res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP')
            if res.success:

                h_opt = scale_to_integral(h_opt)
                if compute_c5(h_opt) < best_c5:
