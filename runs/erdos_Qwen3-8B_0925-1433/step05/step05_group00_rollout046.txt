            else:
                term2 = 0
            grad[j] = term1 + term2
        
        best_alpha = None
        best_trial_h = None
        best_trial_c5 = current_max
        for alpha in [0.05, 0.2, 0.5]:
            trial_h = h - alpha * grad
            trial_h = np.clip(trial_h, 0, 1)
            s = np.sum(trial_h)
            trial_h = trial_h * target_sum / s
            trial_c5 = compute_c5(trial_h, dx, n_points)
            if trial_c5 < best_trial_c5:
                best_trial_c5 = trial_c5
                best_alpha = alpha
                best_trial_h = trial_h
        if best_trial_c5 < current_max:
            h = best_trial_h
        if best_trial_c5 < best_c5:
            best_c5 = best_trial_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
