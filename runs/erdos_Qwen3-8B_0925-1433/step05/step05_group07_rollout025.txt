import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)

        def obj(x):
            h_new = x
            shifted = 1 - h_new
            corr = np.correlate(h_new, shifted, mode='full')
            return np.max(corr) * dx

        def cons(x):
            return np.sum(x) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': cons}]
        result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)

        else:

            h = np.clip(h, 0, 1)
