                if 0 <= j < n_points:
                    grad_perturb[i] = (1 - h_new[j]) * dx
            perturb = np.random.normal(0, 0.08, n_points)
            h_perturb = h_new + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            current_sum_perturb = np.sum(h_perturb)
            h_perturb = h_perturb * target_sum / current_sum_perturb
            current_c5_perturb = compute_c5(h_perturb, dx, n_points)
            if current_c5_perturb < best_c5:
                best_c5 = current_c5_perturb
                best_h = h_perturb.copy()
        h = h_new.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
