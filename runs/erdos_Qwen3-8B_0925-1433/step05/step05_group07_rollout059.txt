        optimized = optimize_candidate(pt)
        if optimized is not None:
            optimized = np.clip(optimized, 0, 1)
            sum_opt = np.sum(optimized)
            if sum_opt != 0:
                scale = (n_points / 2) / sum_opt
                optimized = np.clip(optimized * scale, 0, 1)
            current_c5 = compute_c5(optimized)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
