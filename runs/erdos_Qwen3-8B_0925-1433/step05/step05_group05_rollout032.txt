        h_scaled = np.clip(h * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

        max_corr = np.max(corr)
        return max_corr * dx
