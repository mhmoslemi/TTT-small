import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

    def objective(x):
        x = clip_and_scale(x)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        x = clip_and_scale(x)
        return np.sum(x) - n_points / 2

    def initialize_population(n_pop):
        pop = []
        for _ in range(n_pop):
            if np.random.rand() < 0.7:
                h = initial_h_values.copy()
            elif np.random.rand() < 0.2:
                h = np.random.uniform(0, 1, n_points)
            else:
                h = np.sin(np.linspace(0, np.pi, n_points)) + 1
            pop.append(clip_and_scale(h))
        return np.array(pop)

    pop_size = 50
    pop = initialize_population(pop_size)
    best_c5 = np.inf
    best_h = np.zeros(n_points)

    start_time = time.time()
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            if np.random.rand() < 0.5:
                j = np.random.randint(pop_size)
                parent = pop[j].copy()
            else:
                parent = np.random.uniform(0, 1, n_points)
            parent = clip_and_scale(parent)
            mutated = parent.copy()
            idxs = np.random.choice(n_points, size=min(5, n_points//2), replace=False)
            mutated[idxs] += np.random.normal(0, 0.1, size=len(idxs))
            mutated = clip_and_scale(mutated)
            new_pop.append(mutated)
        pop = np.array(new_pop)
        fitness = np.array([compute_c5(h) for h in pop])
        top_indices = np.argsort(fitness)[:int(pop_size*0.05)]
        top_candidates = pop[top_indices]
        for cand in top_candidates:
            res = minimize(objective, cand, method='SLSQP', bounds=[(0,1)*n_points], constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
            if res.success:
                refined = res.x
                refined = clip_and_scale(refined)
                new_c5 = compute_c5(refined)

                    best_c5 = new_c5
                    best_h = refined.copy()
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)
        pop = np.vstack([pop, top_candidates])
        pop = np.random.choice(pop, size=pop_size, replace=False)
    return (best_h, best_c5, n_points)
