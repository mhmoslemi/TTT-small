import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Add small noise to current h
        noise = np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(h + noise, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        # Optimize
        result = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            perturbed,
            method='L-BFGS-B',
            bounds=Bounds(0, 1),
            constraints=[LinearConstraint(np.ones(n_points), [target_sum], [target_sum])]

            if current_c5 < best_c5:

                best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
