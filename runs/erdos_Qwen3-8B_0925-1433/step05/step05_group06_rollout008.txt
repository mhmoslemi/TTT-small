    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    population = [initial_h_values.copy()]
    while len(population) < 50:
        h = np.random.uniform(0, 1, n_points)
        population.append(clip_and_scale(h))
    
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        new_population = []
        for _ in range(50):
            p1, p2 = random.choices(population, k=2)
            child = np.zeros_like(p1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            if random.random() < 0.1:
                idx = random.randint(0, n_points-1)
                child[idx] = np.random.uniform(0, 1)
            new_population.append(clip_and_scale(child))
        population = new_population
        
        for h in population:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        
        top_candidates = sorted(population, key=lambda x: compute_c5(x), reverse=True)[:10]
        for h in top_candidates:
            def obj(x):
                x = clip_and_scale(x)
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points)
            if res.success:
                h_opt = res.x
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5 = c_opt
                    best_h = h_opt
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
