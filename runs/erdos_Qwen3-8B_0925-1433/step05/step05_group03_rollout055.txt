import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

        return np.max(softmax_corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    try:
        res = minimize(obj, initial_scaled, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
        best_h = res.x
    except:
        best_h = initial_scaled.copy()
