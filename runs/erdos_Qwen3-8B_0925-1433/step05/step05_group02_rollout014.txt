import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / h_sum

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = compute_c5(best_h)

    for _ in range(3):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        result = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        current_c5 = compute_c5(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return (best_h, best_c5, n_points)
