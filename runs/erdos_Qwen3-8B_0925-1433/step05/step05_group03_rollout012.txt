            parent1 = pop[j]
            parent2 = pop[k]
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:split], parent2[split:]])
            if np.sum(child) != target_ones:
                diff = np.sum(child) - target_ones
                if diff > 0:
                    idx = np.where(child == 1)[0][0]
                    child[idx] = 0
                else:
                    idx = np.where(child == 0)[0][0]
                    child[idx] = 1
            mutant = child.copy()
            for idx in range(n_points):
                if np.random.rand() < CR:
                    mutant[idx] ^= 1
            if np.sum(mutant) != target_ones:
                diff = np.sum(mutant) - target_ones
                if diff > 0:
                    idx = np.where(mutant == 1)[0][0]
                    mutant[idx] = 0
                else:
                    idx = np.where(mutant == 0)[0][0]
                    mutant[idx] = 1
            new_pop.append(mutant)
        pop = new_pop
        current_best = min(pop, key=lambda x: compute_c5(x, dx, n_points))
        curr_c5 = compute_c5(current_best, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()
    
    for _ in range(20):
        for idx in range(n_points):
            temp = best_h.copy()
            temp[idx] ^= 1
            if np.sum(temp) != target_ones:
                diff = np.sum(temp) - target_ones
                if diff > 0:
                    idx2 = np.where(temp == 1)[0][0]
                    temp[idx2] = 0
                else:
                    idx2 = np.where(temp == 0)[0][0]
                    temp[idx2] = 1
            temp_c5 = compute_c5(temp, dx, n_points)
            if temp_c5 < best_c5:
                best_h = temp.copy()
                best_c5 = temp_c5
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
