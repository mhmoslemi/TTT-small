import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values

        return np.max(corr) * dx

            if 0 <= j < n_points:
                grad[i] = (1 - current_h[j]) * dx
        # Update step with adaptive learning rate
        step_size = 0.1 * (1 - _ / 500)
        current_h -= step_size * grad

        current_sum = np.sum(current_h)
        if current_sum > 1e-8:
            current_h = current_h * target_sum / current_sum
