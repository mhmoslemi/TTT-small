import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def scale_candidate(h):
        norm = np.sum(h)
        if norm == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / norm
        return np.clip(h * scale, 0, 1)

    pop_size = 50
    pop = [np.clip(initial_h_values.copy() + np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(pop_size)]
    for i in range(pop_size):
        pop[i] = scale_candidate(pop[i])
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    
    while time.time() - start < budget_s:
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(pop, 3, replace=False)
            mutant = a + 0.8 * (b - c)
            trial = np.clip(mutant, 0, 1)
            trial = scale_candidate(trial)
            trial_c5 = compute_c5(trial)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
            new_pop.append(trial)
        pop = new_pop
        
        for h in pop[:5]:
            shifted = 1 - h
            corr = np.correlate(h, shifted, mode='full')
            worst_k = np.argmax(corr)
            def obj(x):
                x = np.clip(x, 0, 1)
                x = scale_candidate(x)
                return np.correlate(x, 1 - x, mode='full')[worst_k] * dx
            def cons(x):
                x = np.clip(x, 0, 1)
                x = scale_candidate(x)
                return np.sum(x) - n_points / 2
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type':'eq','fun':cons})
            if res.success:
                h_opt = res.x
                h_opt = scale_candidate(np.clip(h_opt, 0, 1))
                new_c5 = compute_c5(h_opt)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_opt.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
