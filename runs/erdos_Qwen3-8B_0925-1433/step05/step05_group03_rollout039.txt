import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def optimize(start, max_iter=100, tol=1e-8):
        start_time = time.time()
        options = {'maxiter': max_iter, 'ftol': tol, 'eps': 1e-8}
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        try:
            res = minimize(compute_c5, start, method='L-BFGS-B', bounds=bounds, constraints=cons, options=options)
            h = res.x
            if np.any(h < 0) or np.any(h > 1):
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
            return h, compute_c5(h)
        except:
            h = np.clip(start, 0, 1)
            h = h * target_sum / np.sum(h)
            return h, compute_c5(h)
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            return start, compute_c5(start)
    
    best_h, best_c5 = optimize(initial_scaled)
    
    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        x0 = initial_scaled + perturb
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * target_sum / np.sum(x0)
        h, c5 = optimize(x0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    return (best_h, best_c5, n_points)
