import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    h_best = initial_h_values.copy()
    h_best = np.clip(h_best, 0, 1)
    best_c5 = compute_c5(h_best)

    particles = []
    velocities = []
    for _ in range(50):
        if np.random.rand() < 0.4:
            h = initial_h_values.copy()
        elif np.random.rand() < 0.3:
            h = np.random.uniform(0, 1, n_points)
        else:
            h = np.sin(np.linspace(0, np.pi, n_points)) + 1
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    gbest = h_best.copy()
    gbest_c5 = best_c5
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(len(particles)):
            h = particles[i].copy()
            h = np.clip(h, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                gbest = h.copy()
                gbest_c5 = current_c5
                if gbest_c5 < 0.3807:
                    return (gbest, gbest_c5, n_points)

        w, c1, c2 = 0.7, 1.5, 1.5
        for i in range(len(particles)):
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            vel = w * vel + c1 * r1 * (gbest - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)

        for i in range(5):
            h = particles[i].copy()
            h = np.clip(h, 0, 1)
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    gbest = h_opt.copy()
                    gbest_c5 = current_c5
                    if gbest_c5 < 0.3807:
                        return (gbest, gbest_c5, n_points)

    return (gbest, gbest_c5, n_points)
