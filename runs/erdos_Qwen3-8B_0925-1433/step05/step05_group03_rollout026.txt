import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    
    pop_size = 50
    pop = [initial_scaled.copy() for _ in range(pop_size)]
    
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iter = 300
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            j, k, l = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[j] + 0.8 * (pop[k] - pop[l])
            trial = pop[i].copy()
            for idx in range(n_points):
                if np.random.rand() < 0.9:
                    trial[idx] = mutant[idx]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.full(n_points, target_sum / n_points)
            else:
                trial = trial * target_sum / trial_sum
            trial_c5 = compute_c5(trial, dx, n_points)
            if trial_c5 < compute_c5(pop[i], dx, n_points):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        current_best = min(pop, key=lambda x: compute_c5(x, dx, n_points))
        curr_c5 = compute_c5(current_best, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = current_best.copy()
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    try:
        res = minimize(obj, best_h, method='Nelder-Mead', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        best_h = res.x
        best_c5 = compute_c5(best_h, dx, n_points)
    except:
        pass
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
