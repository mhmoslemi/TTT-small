            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
        temp *= cooling_rate
        iteration += 1
    
    shifted_best = 1 - best_h
    corr_best = np.correlate(best_h, shifted_best, mode='full')
    max_corr_index = np.argmax(corr_best)
    lag = max_corr_index - (n_points - 1)
    if lag < 0:
        lag = 0
    
    def local_objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x = np.clip(x * scale, 0, 1)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return corr[lag] * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    res = minimize(local_objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    if res.success:
        final_h = res.x
        final_h = np.clip(final_h, 0, 1)
        sum_final = np.sum(final_h)
        if sum_final != 0:
            scale_final = (n_points / 2) / sum_final
            final_h = np.clip(final_h * scale_final, 0, 1)
        final_c5 = compute_c5(final_h)
        return (final_h, final_c5, n_points)
    else:
        return (best_h, best_c5, n_points)
