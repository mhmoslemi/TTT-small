import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    population = [initial_h_values.copy() for _ in range(50)]
    for _ in range(50):
        candidate = np.random.rand(n_points)
        candidate *= target_sum / np.sum(candidate)
        population.append(candidate)

    best_c5 = np.inf
    best_h = initial_h_values.copy()

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            j, k, l = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[j], population[k], population[l]
            mutant = a + 0.8 * (b - c)
            trial = np.where(np.random.rand(n_points) < 0.9, mutant, population[i])
            trial_clip = np.clip(trial, 0, 1)
            trial_sum_clip = np.sum(trial_clip)
            if trial_sum_clip != 0:
                trial = trial_clip * target_sum / trial_sum_clip
            original_c5 = compute_c5(population[i], dx, n_points)
            trial_c5 = compute_c5(trial, dx, n_points)
            new_population.append(trial if trial_c5 < original_c5 else population[i])
        population = new_population

    for candidate in population:
        c5_val = compute_c5(candidate, dx, n_points)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = candidate.copy()

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_index = np.argmax(corr)
        grad = np.zeros_like(x)
        for x_idx in range(n_points):
            shift = max_index
            x_plus_k = x_idx + shift
            if 0 <= x_plus_k < n_points:
                grad[x_idx] = 1 - x[x_plus_k]
        return grad

    local_result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, jac=grad, options={'ftol': 1e-8, 'gtol': 1e-8})
    best_h = local_result.x
    best_c5 = compute_c5(best_h, dx, n_points)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
