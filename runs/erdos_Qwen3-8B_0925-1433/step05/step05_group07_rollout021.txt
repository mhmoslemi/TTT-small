import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def scale_to_constraint(h_flat):
        current_sum = np.sum(h_flat)
        if current_sum == 0:
            return np.zeros_like(h_flat) + 0.5
        scale = target_sum / current_sum
        return np.clip(h_flat * scale, 0, 1)

    def objective(h_flat):
        h = scale_to_constraint(h_flat)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess based on initial_h_values
    initial_guess = initial_h_values.copy()
    current_sum = np.sum(initial_guess)
    if current_sum != 0:
        scale = target_sum / current_sum
        initial_guess = np.clip(initial_guess * scale, 0, 1)
    else:
        initial_guess = np.full(n_points, 0.5)

    # Run differential evolution with time constraint
    start_time = time.time()
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=30,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        dispatch_evaluator=True,
        polish=True,
        disp=False
    )

    # Ensure we respect the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = scale_to_constraint(initial_guess)
        best_c5 = objective(initial_guess)
    else:
        best_h = scale_to_constraint(result.x)
        best_c5 = result.fun

    # Fallback if result is invalid
    if np.isnan(best_c5) or np.isinf(best_c5):
        best_h = scale_to_constraint(initial_guess)
        best_c5 = objective(initial_guess)
