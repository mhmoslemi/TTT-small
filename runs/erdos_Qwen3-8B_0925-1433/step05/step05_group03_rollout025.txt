import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    best_h = initial_scaled.copy()

    start_time = time.time()
    
    while time.time() - start_time < budget_s:

            mutant = pop[i] + mut_factor * (pop[j] - pop[k])
