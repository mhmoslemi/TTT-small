        result = minimize(obj, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            if sum_h != 0:
                scale = sum_target / sum_h
                h_new = np.clip(h_new * scale, 0, 1)
            current_c5 = compute_c5(h_new)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
