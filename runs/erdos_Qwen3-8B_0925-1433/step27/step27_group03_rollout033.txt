import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    temp = 100.0
    cooling_rate = 0.99
    perturb_scale = 0.1
    time_start = time.time()
    
    def compute_gradient(h, k):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += dx * (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= dx * h[m - k]
        return grad
    
    while time.time() - time_start < budget_s:
        noise = np.random.normal(0, perturb_scale, n_points)
        h_perturb = h + noise
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        
        corr = np.correlate(h_perturb, 1 - h_perturb, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_perturb.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                best_c5 = new_c5
                best_h = h_perturb.copy()
        
        temp *= cooling_rate
        k_max = np.argmax(corr)
        grad = compute_gradient(best_h, k_max)
        alpha = 0.1
        h_grad = best_h - alpha * grad
        h_grad = np.clip(h_grad, 0, 1)
        h_grad = h_grad * target_sum / np.sum(h_grad)
        corr_grad = np.correlate(h_grad, 1 - h_grad, mode='full')
        new_c5_grad = np.max(corr_grad) * dx
        if new_c5_grad < best_c5:
            best_h = h_grad
            best_c5 = new_c5_grad
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
