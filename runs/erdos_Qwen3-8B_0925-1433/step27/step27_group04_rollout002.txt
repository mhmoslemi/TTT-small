import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def objective(x):
        x_current = np.clip(x, 0, 1)
        x_current = x_current * target_sum / np.sum(x_current)
        corr = np.correlate(x_current, 1 - x_current, mode='full') * dx
        return np.max(corr)

    x0 = h.copy()
    options = {'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 1000, 'disp': False}
    start_time = time.time()
    result = minimize(objective, x0, method='Nelder-Mead', options=options)
    elapsed = time.time() - start_time

    initial_c5 = objective(h)
    final_c5 = objective(result.x) if elapsed < budget_s else initial_c5
    best_h = result.x if final_c5 < initial_c5 else h
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = min(initial_c5, final_c5)

    return (best_h, best_c5, n_points)
