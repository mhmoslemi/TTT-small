import numpy as np

def run(seed=42, budget_s=600):

    perturb_prob = 0.05

    while time.time() - start_time < budget_s:

        current_max = np.max(corr)
        current_k = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + current_k < n_points:
                grad[i] += (1 - h[i + current_k]) * dx
            if i - current_k >= 0:
                grad[i] -= h[i - current_k] * dx
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.1, n_points)
