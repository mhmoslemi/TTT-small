import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    max_iter = 1000
    line_search_steps = 50
    decay_rate = 0.999
    tolerance = 1e-6
    time_start = time.time()
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = corr * dx
        top_lags = np.argsort(current_c5)[-3:]
        grad = np.zeros_like(h)
        for lag in top_lags:
            k = lag % n_points
            for i in range(n_points):
                if (i + k) < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if (i - k) >= 0:
                    grad[i] -= h[i - k] * dx
        alpha = 1.0
        for _ in range(line_search_steps):
            h_test = h + alpha * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * target_sum / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full') * dx
            if corr_test[top_lags[0]] < current_c5[top_lags[0]]:
                alpha *= 1.1
            else:
                alpha *= 0.9
            if alpha < 1e-6:
                break
        h = h + alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if abs(current_c5 - best_c5) < tolerance:
            break
        decay_rate *= 0.9995
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
