    decay_factor = 0.9995
    perturb_interval = 100
    perturb_strength = 0.06
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_max %= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = (1 - h[(i + k_max) % n_points]) - h[j]
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
