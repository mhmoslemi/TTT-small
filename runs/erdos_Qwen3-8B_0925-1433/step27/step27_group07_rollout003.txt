import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize subpopulations
    num_subpop = 3
    pop_size = 50
    subpopulations = []
    for _ in range(num_subpop):
        subpop = initial_h.copy()
        subpop += np.random.uniform(-0.1, 0.1, n_points)
        subpop = np.clip(subpop, 0, 1)
        subpop = subpop * target_sum / np.sum(subpop)
        subpopulations.append(subpop)

    # Function to compute C₅
    def calc_c5(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    best_h = initial_h.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = calc_c5(best_h)

    max_iter = 500
    gen = 0
    start_time = time.time()

    while gen < max_iter and time.time() - start_time < budget_s:
        for subpop in subpopulations:
            # Select two parents via tournament
            indices = np.random.choice(len(subpop), 3, replace=False)
            parent_values = [calc_c5(subpop[i]) for i in indices]
            best_idx = indices[np.argmin(parent_values)]
            parent1 = subpop[best_idx].copy()
            indices = np.random.choice(len(subpop), 3, replace=False)
            parent_values = [calc_c5(subpop[i]) for i in indices]
            best_idx = indices[np.argmin(parent_values)]
            parent2 = subpop[best_idx].copy()

            # Crossover
            offspring = (parent1 + parent2) / 2
            # Mutation
            mutating = np.random.rand(n_points) < 0.1 * (1 - gen / max_iter)
            noise = np.random.normal(0, 0.05 * (1 - gen / max_iter), n_points)
            offspring[mutating] += noise[mutating]
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * target_sum / np.sum(offspring)

            # Replace worst individual
            worst_idx = np.argmax([calc_c5(subpop[i]) for i in range(len(subpop))])
            if calc_c5(offspring) < calc_c5(subpop[worst_idx]):
                subpop[worst_idx] = offspring.copy()

        # Update global best
        for subpop in subpopulations:
            current_c5 = calc_c5(subpop)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = subpop.copy()

        gen += 1

    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
