import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    candidates = [h.copy()]
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Generate new candidates based on best solution
        for _ in range(4):
            if time.time() - start_time > budget_s:
                break
            # Directional perturbation based on current best
            idx = np.argsort(np.abs(np.correlate(best_h, 1 - best_h, mode='full')))[-1]
            pert = np.zeros(n_points)
            pert[idx % n_points] = np.random.uniform(-0.05, 0.05)
            new_h = best_h + pert
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            candidates.append(new_h)
        
        # Local refinement
        new_candidates = []
        for cand in candidates:
            x0 = cand.copy()
            bounds = Bounds(0, 1)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                          tol=1e-7, options={'maxiter': 150, 'ftol': 1e-9})
            if res.success:
                new_h = res.x
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                new_candidates.append(new_h)
        candidates = new_candidates
        
        # Evaluate and update best
        for cand in candidates:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
        candidates = [best_h] + candidates[:20]  # Keep diversity
    
    best_h = np.clip(best_h, 0, 1)
