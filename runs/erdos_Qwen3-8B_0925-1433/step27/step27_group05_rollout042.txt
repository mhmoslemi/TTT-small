import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    max_iter = 500
    line_search_steps = 20
    m = 5
    decay_rate = 0.999
    tolerance = 1e-6
    time_start = time.time()
    for iter in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = corr
        top_indices = np.argsort(current_c5)[-m:]  # top 5 lags
        grad = np.zeros_like(h)
        for lag in top_indices:
            k = lag
            for i in range(n_points):
                if (i + k) < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if (i - k) >= 0:
                    grad[i] -= h[i - k] * dx
        alpha = 1.0
        for _ in range(line_search_steps):
            h_test = h + alpha * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * target_sum / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full') * dx
            best_lag_test = np.argmax(corr_test)
            if corr_test[best_lag_test] < current_c5[best_lag_test]:
                alpha *= 1.1
            else:
                alpha *= 0.9
            if alpha < 1e-6:
                break
        h = h + alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5_val = np.max(corr)
        if current_c5_val < best_c5:
