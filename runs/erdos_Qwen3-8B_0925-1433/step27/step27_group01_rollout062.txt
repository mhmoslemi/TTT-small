    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = 500
    step_size = 0.05
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        k_max = np.argmax(corr)
        actual_lag = k_max - (n_points - 1)
        grad = np.array([1 - h[(i + actual_lag) % n_points] for i in range(n_points)])
        direction = -grad
        valid_indices = np.where(grad != 0)[0]
        if len(valid_indices) == 0:
            continue
        possible_alphas = h[valid_indices] / (-grad[valid_indices])
        alpha_max = np.min(possible_alphas)
        alpha = min(alpha_max, step_size)
        new_h = h + alpha * direction
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < current_c5:
            h = new_h.copy()
        if np.random.rand() < 0.1:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * target_sum / np.sum(h)
    
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
