    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def compute_gradients(h, lags, n_points, dx):
        grad = np.zeros(n_points)
        for k in lags:
            for m in range(n_points):
                if m + k < n_points:
                    grad[m] += dx * (1 - h[m + k])
                if m - k >= 0:
                    grad[m] -= dx * h[m - k]
        return grad
    
    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
