def run(seed=42, budget_s=600):

    alpha_schedule = [1.0] * 30 + [0.1] * 30 + [0.05] * 30
    
    start_time = time.time()
    
    for step_idx in range(max_iterations):
        if time.time() - start_time > budget_s:

        current_c5 = corr * dx
        best_lag_index = np.argmax(current_c5)
