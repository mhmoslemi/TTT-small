import time

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h_curr = x * target_sum / np.sum(x)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * dx
    
    population = [h.copy() for _ in range(50)]
    for p in population:
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)

    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):

            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = x1 + 0.8 * (x2 - x3)

            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
