            overlap_plus = compute_overlap_for_k(h_plus, k_max)
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            sum_minus = np.sum(h_minus)
            h_minus = h_minus * target_sum / sum_minus
            overlap_minus = compute_overlap_for_k(h_minus, k_max)
            grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        return grad

    start_time = time.time()
    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_max = max_index - (n_points - 1)
        grad = compute_gradient(h, k_max)
        direction = -grad
        alpha_candidates = [0.1, 0.05, 0.01, 0.005, 0.001]
        best_alpha = 0.0
        best_overlap = np.inf
        for alpha in alpha_candidates:
            new_h = h + alpha * direction
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            new_h = new_h * target_sum / sum_new
            overlap = compute_overlap_for_k(new_h, k_max)
            if overlap < best_overlap:
                best_overlap = overlap
                best_alpha = alpha
        new_h = h + best_alpha * direction
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        new_h = new_h * target_sum / sum_new
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        h = new_h
        if _ % 50 == 0:
            perturb = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            h = h * target_sum / sum_h

    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * target_sum / sum_h
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (h, best_c5, n_points)
