import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    pop_size = 50
    particles = []

    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest.copy()
    best_c5 = gbest_value
    omega, c1, c2 = 0.7, 1.5, 1.5
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])

            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
