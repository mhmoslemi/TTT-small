            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            trial = trial * target_sum / sum_trial if sum_trial != 0 else trial * target_sum / 1e-6
            trial_c5 = evaluate(trial)
            if trial_c5 < evaluate(population[i]):
                population[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
