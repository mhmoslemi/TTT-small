    while time.time() - start_time < budget_s and steps < max_steps:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                gradient[i] = dx * (1 - h[i + k])
        
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        if np.random.rand() < perturb_prob:
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        
        learning_rate *= alpha_decay
        if no_improve_counter > 100:
            break
        
        steps += 1
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
