    perturb_interval = 100

        k_max = np.argmax(corr)
        k_max %= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):

            grad[i] = (1 - h[(i + k_max) % n_points]) - h[j]
        h_new = h - learning_rate * grad
