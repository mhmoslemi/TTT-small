        total = np.sum(h)
        if total == 0:
            h = np.full(n_points, 1 / n_points)
        else:
            h = h * target_sum / total
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    space = [Real(0, 1, name=f'h_{i}') for i in range(n_points)]
    
    opt = Optimizer(space, base_estimator='GP', n_jobs=-1, acq_func='EI', n_initial_points=20)
    
    start_time = time.time()
    best_c5 = objective(initial_h)
    best_h = initial_h.copy()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        opt.tell(opt.ask(), objective(opt.ask()))
        if opt.state.n_iter > 20:
            opt.refit()
    
    candidates = opt.get_history()
    best_idx = np.argmin([obj for obj in candidates])
    best_h = candidates[best_idx]
    best_c5 = objective(best_h)
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
