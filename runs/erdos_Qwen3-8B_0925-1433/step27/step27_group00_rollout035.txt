import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    pop_size = 50
    particles = [initial_h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        noise = np.random.normal(0, 0.1, n_points)
        particles[i] += noise
        particles[i] = np.clip(particles[i], 0, 1)
        sum_p = np.sum(particles[i])
        if sum_p > 0:
            particles[i] = particles[i] * (target_sum / sum_p)
    
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = particles.copy()
    pbest_values = [evaluate_erdos_solution(p) for p in particles]
    gbest = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    best_h = gbest
    best_c5 = gbest_value
    
    omega = 0.7
    c1, c2 = 1.5, 1.5
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            particles[i] += velocities[i]
            clipped = np.clip(particles[i], 0, 1)
            sum_clipped = np.sum(clipped)
            if sum_clipped > 0:
                particles[i] = clipped * (target_sum / sum_clipped)
            current_c5 = evaluate_erdos_solution(particles[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = particles[i].copy()
        gbest_idx = np.argmin(pbest_values)
        gbest = pbest[gbest_idx].copy()
        gbest_value = pbest_values[gbest_idx]
    
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        best_h = best_h * (target_sum / sum_best)
    return (best_h, best_c5, n_points)
