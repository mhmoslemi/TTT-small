import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h_curr):
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        return np.max(corr) * dx

    population = [h + np.random.normal(0, 0.1, n_points) for _ in range(50)]
    for p in population:
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)

    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            idxs = np.random.choice(range(len(population)), 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            F = np.random.uniform(0.5, 1.5)
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
        population = new_population
        population.sort(key=compute_c5)
        top_k = 15
        pop_idx = np.argsort(compute_c5(population))[:top_k]
        population = [population[i] for i in pop_idx]
        for _ in range(50 - len(population)):
            idxs = np.random.choice(range(len(population)), 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            F = np.random.uniform(0.5, 1.5)
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            population.append(mutant)
        current_c5 = compute_c5(population[0])
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5
        gen += 1

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
