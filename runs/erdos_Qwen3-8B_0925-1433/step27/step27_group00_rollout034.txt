import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * (target_sum / np.sum(h_initial))
    h_initial = np.clip(h_initial, 0, 1)
    pop_size = 50
    particles = [h_initial.copy() for _ in range(pop_size)]
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = particles.copy()
    pbest_vals = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    gbest = particles[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)
    best_h = gbest.copy()
    best_c5 = gbest_val
    iteration = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * 0.7 + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(vel, -0.1, 0.1)
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            curr_val = np.max(corr) * dx
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
        new_gbest = np.argmin(pbest_vals)
        if pbest_vals[new_gbest] < gbest_val:
            gbest = pbest[new_gbest].copy()
            gbest_val = pbest_vals[new_gbest]
            best_h = gbest.copy()
            best_c5 = gbest_val
        iteration += 1
    return (best_h, best_c5, n_points)
