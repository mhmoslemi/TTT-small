import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = _compute_c5(best_h, dx)
    best_c5 = min(best_c5, 0.381)

    n_subpop = 3
    pop_size = 50
    max_iter = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_scale = 0.05
    elite_ratio = 0.2

    subpops = []
    for _ in range(n_subpop):
        pop = h_initial + np.random.normal(0, 0.05, (pop_size, n_points))
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)
        subpops.append(pop)

    for gen in range(max_iter):
        if time.time() - time.time() > budget_s:
            break
        for i in range(n_subpop):
            parents = [subpops[i][np.random.randint(pop_size)] for _ in range(2)]
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parents[0] + (1 - alpha) * parents[1]
            else:
                child = parents[0]
            if np.random.rand() < mutation_rate:
                delta = np.random.normal(0, mutation_scale, n_points)
                child += delta
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            c5 = _compute_c5(child, dx)
            if c5 < _compute_c5(subpops[i][np.random.randint(pop_size)], dx):
                subpops[i][np.random.randint(pop_size)] = child
        current_best = np.min([_compute_c5(subpops[i][0], dx) for i in range(n_subpop)])
        if current_best < best_c5:

    h_values = h_values * target_sum / np.sum(h_values)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
