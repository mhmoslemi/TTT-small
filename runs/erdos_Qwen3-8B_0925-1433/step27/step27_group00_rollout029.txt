import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    if np.sum(initial_h) == 0:
        initial_h = np.full_like(initial_h, target_sum / n_points)
    else:
        initial_h = initial_h * target_sum / np.sum(initial_h)
    pop = []
    for _ in range(20):
        pop.append(initial_h.copy())
    for _ in range(10):
        rand_vec = np.random.rand(n_points) * (target_sum / n_points)
        rand_vec = np.clip(rand_vec, 0, 1)
        if np.sum(rand_vec) == 0:
            rand_vec = np.full_like(rand_vec, target_sum / n_points)
        else:
            rand_vec = rand_vec * target_sum / np.sum(rand_vec)
        pop.append(rand_vec)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        parents = []
        for _ in range(len(pop)):
            idx = np.random.choice(len(pop), size=5, replace=False)
            best_idx = idx[np.argmin(fitness[idx])]
            parents.append(pop[best_idx].copy())
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):

                offspring.append(child1)
                offspring.append(child2)
        mutated = []
        for h in offspring:
            mut = np.random.uniform(-0.15, 0.15, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)

        pop = new_pop
