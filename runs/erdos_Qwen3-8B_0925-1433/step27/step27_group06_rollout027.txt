        h_perturbed = h_perturbed * target_sum / current_sum
        
        new_c5 = np.max(np.correlate(h_perturbed, 1 - h_perturbed, mode='full')) * dx
        delta_c5 = new_c5 - best_c5
        
        if new_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = new_c5
            h = h_perturbed.copy()
        else:
            if np.random.rand() < np.exp(-delta_c5 / temp):
                h = h_perturbed.copy()
        temp *= cooling_rate
    
    h_values = best_h.copy()
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
