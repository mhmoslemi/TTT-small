import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    pop_size = 20
    mutation_rate = 0.1
    elite_ratio = 0.1
    crossover_rate = 0.8
    generations = 500
    early_stop = 50
    
    population = [np.copy(h) for _ in range(pop_size)]
    for _ in range(generations):
        if time.time() - time.time() > budget_s:
            break
        fitness = [np.max(np.correlate(ind, 1-ind, mode='full')) * dx for ind in population]
        elite_size = int(pop_size * elite_ratio)
        elites = [population[i] for i in np.argsort(fitness)[::-1][:elite_size]]
        selected = np.argsort(fitness)[::-1][:int(pop_size * 0.2)]
        new_population = elites.copy()
        
        for _ in range(pop_size - elite_size):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < crossover_rate:
                crossover = np.random.randint(0, n_points)
                child = np.concatenate([population[p1][:crossover], population[p2][crossover:]])
            else:
                child = np.copy(population[np.random.choice(selected)])
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
        for ind in population:
            ind = np.clip(ind, 0, 1)
            ind = ind * target_sum / np.sum(ind)
        
        for i in range(len(population)):
            if fitness[i] < best_c5:
                best_h = np.copy(population[i])
                best_c5 = fitness[i]
        
        if np.max(fitness) - best_c5 < 1e-5 and early_stop > 0:
            early_stop -= 1
            if early_stop == 0:
                break
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
