from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Haar basis construction
    def generate_haar_basis(n_points):
        x = np.linspace(0, 2, n_points)
        basis = []
        m = 1
        while True:
            intervals = int(n_points / (2**m))
            if intervals <= 0:
                break
            # Create Haar basis functions
            for i in range(1, intervals + 1):
                start = (i - 1) / (2**(m - 1))
                end = i / (2**(m - 1))
                part = np.zeros(n_points)
                part[int(start * n_points):int(end * n_points)] = 1
                part[int(end * n_points):int((end + 1) * n_points)] = -1
                basis.append(part)
            m += 1
        return np.array(basis)
    
    basis = generate_haar_basis(n_points)
    coeff = np.linalg.lstsq(basis, h, rcond=None)[0]
    coeff = np.append(0.5, coeff)
    
    def objective(coeffs):
        h_vals = np.dot(basis, coeffs[1:]) + coeffs[0]
        h_vals = np.clip(h_vals, 0, 1)
        overlaps = []
        for k in range(1, n_points):
            shifted = np.roll(h_vals, k)
            overlap = np.sum(h_vals * (1 - shifted)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    bounds = [(0, 1) for _ in range(len(coeff))]
    
    result = minimize(objective, coeff, method='L-BFGS-B', bounds=bounds, tol=1e-6)
