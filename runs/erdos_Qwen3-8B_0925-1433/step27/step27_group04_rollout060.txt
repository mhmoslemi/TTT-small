import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    P = 20
    particles = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values) for _ in range(P)]
    velocities = [np.zeros(n_points) for _ in range(P)]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    personal_bests = [h.copy() for h in particles]
    global_best = min(particles, key=compute_c5)
    best_c5 = compute_c5(global_best)

    omega_max, omega_min = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    max_iter = 500
    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(P):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            new_velocity = omega_max * (iter / max_iter) * velocities[i] + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            new_velocity = np.clip(new_velocity, -0.5 * dx, 0.5 * dx)
            velocities[i] = new_velocity
            new_pos = particles[i] + new_velocity
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * target_sum / np.sum(new_pos)
            particles[i] = new_pos
            new_c5 = compute_c5(particles[i])
            if new_c5 < compute_c5(personal_bests[i]):
                personal_bests[i] = particles[i].copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                global_best = particles[i].copy()
    
    initial_c5 = compute_c5(np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values))
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        global_best = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    global_best = np.clip(global_best, 0, 1)
    global_best = global_best * target_sum / np.sum(global_best)
    return (global_best, best_c5, n_points)
