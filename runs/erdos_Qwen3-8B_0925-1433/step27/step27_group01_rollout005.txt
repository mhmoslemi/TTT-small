    best_c5 = min(best_c5, 0.381)
    
    def compute_corr(h):
        return np.correlate(h, 1 - h, mode='full')
    
    def update_h(h, direction, alpha):
        h_new = h + alpha * direction
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        return h_new
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr) * dx
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        max_k = np.argmax(corr)
        k = max_k - (n_points - 1)
        def grad_func(i):
            eps = 1e-4
            h_p = h.copy()
            h_p[i] += eps
            h_m = h.copy()
            h_m[i] -= eps
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')[max_k]
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')[max_k]
            return (corr_p - corr_m) / (2 * eps)
        grad = np.array([grad_func(i) for i in range(n_points)])
        direction = -grad
        alpha = 0.1
        h_candidate = update_h(h, direction, alpha)
        h_candidate_corr = compute_corr(h_candidate)
        curr_c5 = np.max(h_candidate_corr) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_candidate.copy()
        if np.random.rand() < 0.05:
            pert = np.random.normal(0, 0.05, n_points)
            h = update_h(h, pert, 1.0)
        else:
            h = h_candidate
    return (best_h, best_c5, n_points)
