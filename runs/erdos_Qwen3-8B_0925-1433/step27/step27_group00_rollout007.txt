def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)

    best_c5 = min(best_c5, 0.381)
    
    learning_rate = 0.2
    decay_factor = 0.9995
    perturb_interval = 10
    perturb_strength = 0.2

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_max %= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = (1 - h[(i + k_max) % n_points]) - h[j]
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
