    population[0] = scaled_initial.copy()

    pbest = population.copy()
    pbest_c5 = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in population]
    gbest = population[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)

    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = a + F * (b - c)
            target = population[i]
            trial = np.zeros_like(target)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = target[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * (target_sum / trial_sum)
            trial_c5 = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
            if trial_c5 < pbest_c5[i]:
                population[i] = trial.copy()
                pbest_c5[i] = trial_c5
                if trial_c5 < gbest_c5:
                    gbest = trial.copy()
                    gbest_c5 = trial_c5
                    if trial_c5 < best_c5:
                        best_c5 = trial_c5
                        best_h = trial.copy()

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
