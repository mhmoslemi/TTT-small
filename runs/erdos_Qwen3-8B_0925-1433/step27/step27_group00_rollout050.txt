import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float) * (target_sum / np.sum(initial_h_values))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    pop_size = 50
    particles = np.random.rand(pop_size, n_points) * (target_sum / n_points)
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = particles.copy()
    pbest_vals = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            curr_val = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
        gbest = pbest[np.argmin(pbest_vals)]
        gbest_val = np.min(pbest_vals)
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
        iteration += 1
    
    return (best_h, best_c5, n_points)
