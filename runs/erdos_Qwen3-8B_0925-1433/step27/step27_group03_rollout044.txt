        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        k_max = np.argmax(corr)
        grad = compute_gradient(h, k_max, n_points, dx)
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < max_c5:
                h = h_new
                alpha *= 1.1
            else:
                alpha *= 0.5
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Stochastic correction
        contrib = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                contrib[i] = h[i] * (1 - h[i + k_max])
        top_indices = np.argsort(contrib)[::-1][:20]
        for idx in np.random.choice(top_indices, size=10, replace=False):
            if np.random.rand() < 0.2:
                h[idx] = 1 - h[idx]
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
