import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    pop_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 500
    start_time = time.time()

    pop = np.zeros((pop_size, n_points))
    for i in range(pop_size):
        perturb = np.random.normal(0, 0.05, n_points)
        trial = h + perturb
        trial = np.clip(trial, 0, 1)
        trial = trial * target_sum / np.sum(trial)
        pop[i] = trial

    pop_c5 = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in pop]
    best_index = np.argmin(pop_c5)
    best_h = pop[best_index].copy()
    best_c5 = pop_c5[best_index]

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)

            mask = np.random.rand(n_points) < CR

            trial = trial * target_sum / np.sum(trial)
            trial = np.clip(trial, 0, 1)
            trial_c5 = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
