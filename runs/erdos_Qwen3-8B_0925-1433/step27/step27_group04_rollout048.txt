import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    max_backtrack_steps = 50
    beta = 0.5  # Backtracking reduction factor
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = corr * dx
        best_lag_index = np.argmax(current_c5)
        k = best_lag_index - (n_points - 1)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                grad[i] -= h[i - k] * dx
        grad = -grad
        
        best_h_test = h.copy()
        alpha = 1.0
        for step in range(max_backtrack_steps):
            if alpha < 1e-6:
