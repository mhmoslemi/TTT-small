        new_h = new_h * target_sum / np.sum(new_h)
        h = new_h
        if np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + pert, 0, 1)
            h = h * target_sum / np.sum(h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
