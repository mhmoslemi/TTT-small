import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    alpha = 0.05
    decay_rate = 0.999
    perturb_prob = 0.05
    early_stop = 100
    start_time = time.time()
    iterations = 0
    improvements = 0
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if (j + k >= 0) and (j + k < n_points):
                grad[j] += (1 - h[j + k]) * dx
            if (j - k >= 0) and (j - k < n_points):
                grad[j] -= h[j - k] * dx
        
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.1, n_points)
            h += noise

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvements = 0
        else:
            improvements += 1
            if improvements >= early_stop:

        iterations += 1
        alpha *= decay_rate
    
    h_values = np.clip(best_h, 0, 1)
