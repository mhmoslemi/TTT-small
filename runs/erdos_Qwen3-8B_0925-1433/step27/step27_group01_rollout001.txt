import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Generate a new candidate via targeted perturbation
        conv = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(conv)
        perturb = np.zeros_like(best_h)
        # Adjust regions contributing to max_k
        for x in range(n_points):
            if x + max_k < n_points:
                perturb[x] -= 0.02 * best_h[x]
                perturb[x + max_k] -= 0.02 * best_h[x + max_k]
        new_h = np.clip(best_h + np.random.normal(0, 0.025, n_points) + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        # Optimize
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
        if res.success:
            curr_h = res.x
            curr_h = np.clip(curr_h, 0, 1)
            curr_h = curr_h * target_sum / np.sum(curr_h)
            curr_c5 = obj(curr_h)
            if curr_c5 < best_c5:
                best_h = curr_h
                best_c5 = curr_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
