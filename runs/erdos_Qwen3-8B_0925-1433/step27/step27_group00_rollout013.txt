import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h * (target_sum / np.sum(initial_h))

    population = []
    for _ in range(20):
        clone = initial_h.copy()
        mutation = np.random.uniform(-0.05, 0.05, n_points)
        clone += mutation
        clone = np.clip(clone, 0, 1)
        clone = clone * (target_sum / np.sum(clone))
        population.append(clone)
    for _ in range(10):
        rand_vec = np.random.rand(n_points) * (target_sum / n_points)
        rand_vec = np.clip(rand_vec, 0, 1)
        rand_vec = rand_vec * (target_sum / np.sum(rand_vec))
        population.append(rand_vec)
    population = np.array(population)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        # Select parents via tournament
        parents = []
        for _ in range(len(population)):
            idx = np.random.choice(len(population), size=5, replace=False)
            best_idx = idx[np.argmin(fitness[idx])]
            parents.append(population[best_idx])
        # Generate offspring via crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i + 1]
                split = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                offspring.append(child1)
                offspring.append(child2)
        # Mutate offspring
        mutated = []
        for h in offspring:
            mut = np.random.uniform(-0.05, 0.05, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)

        # Replace population

        # Track best solution

        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()
    
    return (best_h, best_c5, n_points)
