        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        k_max_corr = np.argmax(corr)
        k_max = k_max_corr - (n_points - 1)
        grad = compute_gradient(h, k_max, n_points, dx)
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < max_c5:
                h = h_new
                alpha *= 1.1
            else:
                alpha *= 0.5
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
