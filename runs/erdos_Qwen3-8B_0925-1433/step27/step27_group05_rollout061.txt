    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = 0.0
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        current_c5 = corr[n_points-1:2*n_points-1] * dx
        return np.max(current_c5)
    
    best_c5 = compute_c5(h)
    
    time_start = time.time()
    max_iter = 1000
    line_search_steps = 20
    tolerance = 1e-6
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = corr[n_points-1:2*n_points-1] * dx
        top_lags = np.argsort(current_c5)[-5:]
        
        grad = np.zeros_like(h)
        for k in top_lags:
            for i in range(n_points):
                if (i + k) < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if (i - k) >= 0:
                    grad[i] -= h[i - k] * dx
        
        alpha = 1.0
        for _ in range(line_search_steps):
            h_test = h + alpha * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * target_sum / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            corr_test = corr_test[n_points-1:2*n_points-1] * dx
            if corr_test[np.argmax(current_c5)] < current_c5[np.argmax(current_c5)]:
                alpha *= 1.1
            else:
                alpha *= 0.9
            if alpha < 1e-6:
                break
        
        h = h + alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5_val = compute_c5(h)
        if current_c5_val < best_c5:
            best_c5 = current_c5_val
            best_h = h.copy()
        if abs(current_c5_val - best_c5) < tolerance:
            break
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
