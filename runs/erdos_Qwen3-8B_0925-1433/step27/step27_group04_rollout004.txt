import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_base = np.array(initial_h_values, dtype=float)
    initial_base = initial_base * target_sum / np.sum(initial_base)
    initial_base = np.clip(initial_base, 0, 1)
    
    P = 20
    particles = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    
    for _ in range(P):
        h = initial_base.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_fitness.append(np.max(np.correlate(h, 1 - h, mode='full')) * dx)
    
    best_global = min(personal_best_fitness)
    best_global_h = np.array([p.copy() for p in personal_best][np.argmin(personal_best_fitness)])
    
    max_iterations = 500
    start_time = time.time()
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        omega = 0.9 - 0.5 * iteration / max_iterations
        
        for p_idx in range(P):
            h = particles[p_idx]
            v = velocities[p_idx]
            
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = c1 * r1 * (personal_best[p_idx] - h)
            social = c2 * r2 * (best_global_h - h)
            v_new = omega * v + cognitive + social
            v_new = np.clip(v_new, -1.0, 1.0) * dx
            velocities[p_idx] = v_new
            
            h_new = h + v_new
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            particles[p_idx] = h_new
            
            corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
            current_c5 = np.max(corr)
            
            if current_c5 < personal_best_fitness[p_idx]:
                personal_best[p_idx] = h_new.copy()
                personal_best_fitness[p_idx] = current_c5
            
            if current_c5 < best_global:
                best_global = current_c5
                best_global_h = h_new.copy()
    
    initial_c5 = np.max(np.correlate(initial_base, 1 - initial_base, mode='full') * dx)
    if initial_c5 < best_global:
        best_global = initial_c5
        best_global_h = initial_base.copy()
    
    h_values = best_global_h
    return (h_values, best_global, n_points)
