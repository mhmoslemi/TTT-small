import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    T = 10.0
    alpha = 0.95
    start_time = time.time()

    while (time.time() - start_time) < budget_s and T > 1e-6:
        indices = np.random.choice(n_points, 5, replace=False)
        perturb = np.zeros(n_points)
        perturb[indices] = np.random.normal(0, T * 0.1, 5)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)

        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = new_h.copy()

        T *= alpha

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
