    learning_rate = 0.1
    decay_rate = 0.995
    line_search_steps = 10
    line_search_alpha = 0.5
    perturb_freq = 100
    perturb_strength = 0.05
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(cross_corr)[-5:]
        lag_indices = top_lags
        grad = np.zeros(n_points)
        for lag in lag_indices:
            lag_val = lag - (n_points - 1)
            for i in range(n_points):
                i_plus_k = i + lag_val
                if 0 <= i_plus_k < n_points:
                    grad[i] += (1 - h[i_plus_k]) * dx
                i_minus_k = i - lag_val
                if 0 <= i_minus_k < n_points:
                    grad[i] -= h[i_minus_k] * dx
        grad /= np.linalg.norm(grad) if np.linalg.norm(grad) > 1e-8 else 1
        
        alpha = learning_rate
        for _ in range(line_search_steps):

            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                h = new_h
                learning_rate *= decay_rate
                break
            else:
                alpha *= line_search_alpha
        
        if _ % perturb_freq == 0:
