import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr)

    def gradient(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + lag
            if 0 <= i_plus < n_points:
                grad[i] += (1 - h[i_plus])
            i_minus = i - lag
            if 0 <= i_minus < n_points:
                grad[i] -= h[i_minus]
        return grad

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    start_time = time.time()

    try:
        result = minimize(objective, initial_h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        best_h = result.x
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    except:
        pass

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
