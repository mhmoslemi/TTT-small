            positive_mask = pos < n_points
            grad[positive_mask] += dx * (1 - h[pos[positive_mask]])
            neg = np.arange(n_points) - k
            negative_mask = neg >= 0
            grad[negative_mask] -= dx * h[neg[negative_mask]]
        grad /= 3
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < max_c5:
            h = h_new
            alpha *= 1.1
        else:
            alpha *= 0.9
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
