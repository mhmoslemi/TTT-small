import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    T = 100.0
    cooling_rate = 0.995
    for _ in range(100000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        # Perturb
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        # Evaluate
        new_corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(new_corr) * dx
        # Accept
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5)/T):
            best_c5 = new_c5
            best_h = h_perturbed.copy()
        T *= cooling_rate
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
