    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 100, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = h
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total > 0:
        best_h = best_h * target_sum / total
    else:
        best_h = np.zeros_like(best_h)
    
    return (best_h, best_c5, n_points)
