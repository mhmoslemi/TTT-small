import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)
    
    learning_rate = 0.25
    decay_factor = 0.998
    perturb_interval = 50
    perturb_strength = 0.15
    iteration = 0
    start_time = time.time()
    no_improvement = 0
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_max %= n_points
        # Random lag selection
        if np.random.rand() < 0.2:  # 20% chance to pick random lag
            k_max = np.random.randint(0, n_points)
        grad = np.zeros(n_points)

            j = (i + k_max) % n_points
            grad[i] = (1 - h[j]) - h[(i - k_max) % n_points]
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        if iteration % perturb_interval == 0:
            perturb = np.random.uniform(-perturb_strength, perturb_strength, n_points)
