        max_c5 = np.max(corr)
        k_max = np.argmax(corr)
        # Compute gradient for current k_max
        grad = compute_gradient(h, k_max, n_points, dx)
        # Gradient descent step
        alpha = 0.1 + 0.01 * T  # learning rate
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain the sum
        current_sum = np.sum(h_new)
        if abs(current_sum) < 1e-5:
            h_new = h.copy()
        else:
            h_new = h_new * target_sum / current_sum
        c5_new = evaluate_c5(h_new)
        # Acceptance based on simulated annealing
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_c5 = c5_new
                best_h = h_new.copy()
        # Random perturbation
        if np.random.rand() < 0.2:  # 20% chance
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum_perturbed = np.sum(h_perturbed)
            if abs(current_sum_perturbed) < 1e-5:
                h_perturbed = h.copy()
            else:
                h_perturbed = h_perturbed * target_sum / current_sum_perturbed
            c5_perturbed = evaluate_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
            else:
                delta = c5_perturbed - best_c5
                if np.random.rand() < np.exp(-delta / T):
                    best_c5 = c5_perturbed
                    best_h = h_perturbed.copy()
        # Update temperature
        T *= temp_decay
        # Update h to best_h
        h = best_h.copy()
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
