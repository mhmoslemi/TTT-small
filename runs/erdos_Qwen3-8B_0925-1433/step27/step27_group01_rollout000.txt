import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    population = [h.copy()]
    for _ in range(6):
        # Directional perturbation: increase low regions, decrease high regions
        weights = np.sqrt(np.abs(h - 0.5))
        pert = np.random.normal(0, 0.015, n_points) * weights
        h_new = np.clip(h + pert, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        population.append(h_new)
        # Random perturbation
        pert = np.random.normal(0, 0.03, n_points)
        h_new = np.clip(h + pert, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        population.append(h_new)
    
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            x0 = population[i].copy()
            bounds = Bounds(0, 1)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            try:
                res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
                if res.success:
                    new_h = res.x
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * target_sum / np.sum(new_h)
                    population.append(new_h)
                    # Keep top 50
                    population = sorted(population, key=lambda h: obj(h))[:50]
            except:
                continue
        
        # Evaluate all candidates
        curr_cands = [obj(h) for h in population]
        curr_c5 = min(curr_cands)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = population[curr_cands.index(curr_c5)].copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
