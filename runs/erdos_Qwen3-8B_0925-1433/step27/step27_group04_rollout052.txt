import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    num_particles = 20
    population = []
    for _ in range(num_particles):
        h = np.array(initial_h_values, dtype=float) * 0.5 + np.random.rand(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    global_best = None
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        if global_best is None or c5 < global_best[0]:
            global_best = (c5, h)
    best_h, best_c5 = global_best
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    for _ in range(1000):
        if time.time() - time.time() > budget_s:
            break
        for i in range(num_particles):
            h = population[i]
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = velocities[i] * 0.8 + 2 * r1 * (np.array(initial_h_values) - h) + 2 * r2 * (global_best[1] - h)
            new_h = h + velocities[i]
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            corr = np.correlate(new_h, 1 - new_h, mode='full') * dx
            new_c5 = np.max(corr)
            if new_c5 < np.max(np.correlate(population[i], 1 - population[i], mode='full') * dx):
                population[i] = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
    return (np.clip(best_h, 0, 1), best_c5, n_points)
