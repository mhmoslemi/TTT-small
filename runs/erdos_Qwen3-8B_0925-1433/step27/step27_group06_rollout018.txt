import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    T = 10.0
    alpha = 0.99
    start_time = time.time()

    while time.time() - start_time < budget_s:
        mutant = current_h.copy()
        indices = np.random.choice(n_points, 5, replace=False)
        for i in indices:
            mutant[i] += np.random.normal(0, 0.05)
        mutant = np.clip(mutant, 0, 1)
        new_sum = np.sum(mutant)
        mutant = mutant * target_sum / new_sum
        new_c5 = compute_c5(mutant)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = mutant
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
        T *= alpha

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
