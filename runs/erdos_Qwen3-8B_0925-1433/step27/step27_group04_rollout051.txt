import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        x_rescaled = x_clipped * target_sum / np.sum(x_clipped) if np.sum(x_clipped) > 0 else x_clipped
        corr = np.correlate(x_rescaled, 1 - x_rescaled, mode='full') * dx
        return np.max(corr)
    
    start_time = time.time()
    result = minimize(
        objective, 
        h0, 
        method='Nelder-Mead', 
        options={'maxiter': 1000, 'xatol': 1e-8, 'fatol': 1e-8}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        h_best = h0
        corr = np.correlate(h_best, 1 - h_best, mode='full') * dx
        c5_best = np.max(corr)
    else:
        h_best = result.x
        h_best_clipped = np.clip(h_best, 0, 1)
        h_best_rescaled = h_best_clipped * target_sum / np.sum(h_best_clipped) if np.sum(h_best_clipped) > 0 else h_best_clipped
        corr = np.correlate(h_best_rescaled, 1 - h_best_rescaled, mode='full') * dx
        c5_best = np.max(corr)
    
    return (h_best_rescaled, c5_best, n_points)
