import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float) * (target_sum / np.sum(initial_h_values))
    pop_size = 50
    particles = [np.random.rand(n_points) for _ in range(pop_size)]
    particles[0] = initial_h
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate_erdos_solution(p) for p in particles]
    gbest = particles[np.argmin(pbest_values)]
    best_h = gbest.copy()
    best_c5 = np.min(pbest_values)
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= target_sum / np.sum(particles[i])
            current_c5 = evaluate_erdos_solution(particles[i])
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
        current_gbest = particles[np.argmin(pbest_values)]
        if np.min(pbest_values) < best_c5:
            best_c5 = np.min(pbest_values)
            best_h = current_gbest.copy()
        if iteration > 100 and np.isclose(best_c5, np.min(pbest_values), atol=1e-5):
            break
        iteration += 1
    return (best_h, best_c5, n_points)
