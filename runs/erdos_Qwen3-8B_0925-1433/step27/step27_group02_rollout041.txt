import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    current_sum = np.sum(h_initial)
    scaling_factor = target_sum / current_sum
    h_initial = np.clip(h_initial * scaling_factor, 0.0, 1.0)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(cross_corr)
        return max_val * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    
    try:
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
    except:
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-8)
    
    h_values = np.clip(result.x, 0.0, 1.0)
    h_values = h_values * target_sum / np.sum(h_values)
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
