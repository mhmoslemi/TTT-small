            velocities[i] = inertia_weight * velocities[i] + \
                            cognitive_coeff * r1 * (pbest[i] - particles[i]) + \
                            social_coeff * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i], axis=0, keepdims=True)
            c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
        gbest = pbest[np.argmin(pbest_c5)]
        gbest_c5 = np.min(pbest_c5)
        if np.random.rand() < mutation_rate:
            gbest += np.random.normal(0, mutation_scale, n_points)
            gbest = np.clip(gbest, 0, 1)
            gbest = gbest * target_sum / np.sum(gbest)
            gbest_c5 = np.max(np.correlate(gbest, 1 - gbest, mode='full')) * dx
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
