import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    max_iter = 1000
    line_search_steps = 50
    decay_rate = 0.999
    tolerance = 1e-6
    time_start = time.time()
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = corr.max()
        # Select top 5 lags
        top_lags = np.argsort(corr)[-5:]
        grad = np.zeros_like(h)
        for lag in top_lags:
            for i in range(n_points):
                if (i + lag) < n_points:
                    grad[i] += (1 - h[i + lag]) * dx
                if (i - lag) >= 0:
                    grad[i] -= h[i - lag] * dx
        alpha = 1.0
        for _ in range(line_search_steps):
            h_test = h + alpha * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * target_sum / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full') * dx
            best_lag = top_lags[0]
            if corr_test[best_lag] < corr[best_lag]:
                alpha *= 1.1
            else:
                alpha *= 0.9
            if alpha < 1e-6:

        h = np.clip(h, 0, 1)
