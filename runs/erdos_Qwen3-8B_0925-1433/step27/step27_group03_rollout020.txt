from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += dx * (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= dx * h[m - k]
        return grad
    
    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        k_max = np.argmax(corr)
        grad = compute_gradient(h, k_max, n_points, dx)
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
