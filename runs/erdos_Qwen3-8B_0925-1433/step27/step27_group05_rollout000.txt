        for lag in best_lags:
            k = lag % n_points
            for i in range(n_points):

                    grad[i] += (1 - h[i + k]) * dx
                if (i - k) >= 0:
                    grad[i] -= h[i - k] * dx
        alpha = 1.0

            # Check if the best lag's value is improved
            if corr_test[best_lags[0]] < current_c5[best_lags[0]]:
                alpha *= 1.1
            else:
                alpha *= 0.9
