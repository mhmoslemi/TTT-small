        h_new = h_new * target_sum / total
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
        if _ % 10 == 0:
            pert = np.random.normal(0, 0.05, n_points)
            h_pert = h + pert
            h_pert = np.clip(h_pert, 0, 1)
            total_pert = np.sum(h_pert)
            h_pert = h_pert * target_sum / total_pert
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode="full")
            pert_c5 = np.max(corr_pert) * dx
            if pert_c5 < best_c5:
                h = h_pert.copy()
                best_c5 = pert_c5
                best_h = h.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
