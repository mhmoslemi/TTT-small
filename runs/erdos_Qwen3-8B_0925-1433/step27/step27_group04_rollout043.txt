import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h_sum = np.sum(initial_h)
    h = initial_h * (target_sum / initial_h_sum)

        x_sum = np.sum(x_clipped)
        if x_sum == 0:
            return np.inf
        x_scaled = x_clipped * (target_sum / x_sum)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full') * dx
        return np.max(corr)
    
    options = {'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 1000, 'maxfev': 1000}
    start = time.time()
    
    result = optimize.minimize(objective, h, method='Nelder-Mead', options=options)
    
    elapsed = time.time() - start

        h_values = np.clip(h, 0.0, 1.0)
        h_values = h_values * (target_sum / np.sum(h_values))

        best_c5 = objective(h_values)
