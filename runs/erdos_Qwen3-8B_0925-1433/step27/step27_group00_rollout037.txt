import time

        rand_vec = np.random.rand(n_points) * (target_sum / n_points)
        rand_vec = np.clip(rand_vec, 0, 1)
        rand_vec = rand_vec * (target_sum / np.sum(rand_vec))
        pop.append(rand_vec)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * (2.0 / n_points)
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        fitness = []

            corr = np.correlate(h, 1 - h, mode='full')
            f = np.max(corr) * (2.0 / n_points)
            fitness.append(f)
        
        parents = []
        for _ in range(pop_size):

            best_idx = idx[np.argmin(fitness[idx])]
            parents.append(pop[best_idx].copy())
        
        offspring = []
        for i in range(0, len(parents), 2):

                child1 = np.concatenate([p1[:split], p2[split:]])

                offspring.append(child2)
        
        mutated = []
        for h in offspring:
            mut = np.random.uniform(-0.1, 0.1, n_points)

            sum_mut = np.sum(h_mut)

                h_mut = h_mut * (target_sum / sum_mut)
