        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    generations = 0
    
    while time.time() - start_time < budget_s:
        generations += 1
        
        # Evaluate all members
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr * dx)
            fitness.append(c5)
        
        # Find the best individual
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Select elites (10%)
        elite_count = int(0.1 * population_size)
        elites = np.argsort(fitness)[:elite_count]
        
        # Create new population
        new_population = []
        for _ in range(population_size - elite_count):
            # Tournament selection
            a, b = np.random.choice(population_size, 2, replace=False)
            if fitness[a] < fitness[b]:
                parent1 = population[a]
            else:
                parent1 = population[b]
            parent2 = population[np.random.choice(population_size, p=np.array(fitness)/np.sum(fitness))]
            split = np.random.randint(1, n_points)
            child = np.zeros_like(parent1)
            child[:split] = parent1[:split]
            child[split:] = parent2[split:]
            
            # Mutate
            if np.random.random() < 0.1:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Add elites
        for idx in elites:
            new_population.append(population[idx].copy())
        
        population = new_population
        
        # Apply local optimization to top 5%
        top_indices = np.argsort(fitness)[:int(0.05 * population_size)]
        for idx in top_indices:
            h = population[idx].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            best_lag_idx = np.argmax(corr * dx)
            k = best_lag_idx - (n_points - 1)
            
            # Compute gradient
            grad = np.zeros_like(h)
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if i - k >= 0:
                    grad[i] -= h[i - k] * dx
            grad = -grad
            
            # Line search
            alpha = 1.0
            for _ in range(50):
                h_test = h + alpha * grad
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * target_sum / np.sum(h_test)
                corr_test = np.correlate(h_test, 1 - h_test, mode='full') * dx
                if corr_test[best_lag_idx] < corr[best_lag_idx]:
                    alpha *= 1.1
                else:
                    alpha *= 0.9
                if alpha < 1e-6:
                    break
            
            h = h + alpha * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            population[idx] = h.copy()
        
        # Early stopping
        if generations > 500:
            break
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
