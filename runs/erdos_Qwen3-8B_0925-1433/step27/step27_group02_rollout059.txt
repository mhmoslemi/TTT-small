    best_c5 = min(best_c5, 0.381)
    
    max_iter = 500
    learning_rate = 0.08
    decay_rate = 0.998
    line_search_steps = 15
    line_search_alpha = 0.5
    start_time = time.time()
    perturb_prob = 0.05
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        indices = np.argsort(cross_corr)[-5:]
        grad = np.zeros(n_points)
        for lag_idx in indices:
            lag = lag_idx - (n_points - 1)
            for i in range(n_points):
                i_plus_k = i + lag
                if 0 <= i_plus_k < n_points:
                    grad[i] += (1 - h[i_plus_k]) * dx
                i_minus_k = i - lag
                if 0 <= i_minus_k < n_points:
                    grad[i] -= h[i_minus_k] * dx
        
        alpha = learning_rate
        improved = False
        for _ in range(line_search_steps):
            new_h = h + alpha * grad
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new == 0:
                new_h = np.zeros_like(new_h)
            else:
                new_h = new_h * target_sum / sum_new
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                h = new_h
                learning_rate *= decay_rate
                improved = True
                break
            else:
                alpha *= line_search_alpha
        
        if not improved and np.random.rand() < perturb_prob:
