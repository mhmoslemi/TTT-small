import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    learning_rate = 0.3
    decay_factor = 0.9995
    perturb_interval = 50
    perturb_strength = 0.15
    iteration = 0
    start_time = time.time()
    stagnation_counter = 0
    max_stagnation = 200
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_max %= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = (1 - h[(i + k_max) % n_points]) - h[j]
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        if iteration % perturb_interval == 0:
            perturb = np.random.uniform(-perturb_strength, perturb_strength, n_points)
            h_new += perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        h = h_new
        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            if stagnation_counter >= max_stagnation:
                break
        learning_rate *= decay_factor
        iteration += 1
    
    return (best_h, best_c5, n_points)
