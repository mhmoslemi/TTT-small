    start_time = time.time()
    
    result = gp_minimize(objective, bounds, n_calls=150, n_random_starts=20,
                         acq_func='EI', n_iter=150, verbose=False)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_theta = result.x if hasattr(result, 'x') else np.zeros(n_points)
    else:
        best_theta = result.x
    
    best_h_vals = best_theta * target_sum / np.sum(best_theta)
    best_h_vals = np.clip(best_h_vals, 0, 1)
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h_vals, best_c5, n_points)
