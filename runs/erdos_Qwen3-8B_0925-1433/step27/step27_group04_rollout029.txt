            pos_new = pos + vel_new
            pos_new = np.clip(pos_new, 0.0, 1.0)
            pos_new = pos_new * target_sum / np.sum(pos_new)
            particles[p] = pos_new
        
        for p in range(P):
            pos = particles[p]
            corr = np.correlate(pos, 1 - pos, mode='full') * dx
            current_c5 = np.max(corr)
            pbest_pos = pbest[p]
            corr_pbest = np.correlate(pbest_pos, 1 - pbest_pos, mode='full') * dx
            pbest_c5 = np.max(corr_pbest)
            if current_c5 < pbest_c5:
                pbest[p] = pos.copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = pos.copy()
    
    corr_initial = np.correlate(initial_h, 1 - initial_h, mode='full') * dx
    c5_initial = np.max(corr_initial)
    if c5_initial < best_c5:
        global_best = initial_h.copy()
        best_c5 = c5_initial
    
    return (global_best, best_c5, n_points)
