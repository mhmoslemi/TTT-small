def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    K = 5  # Number of top lags to consider
    
    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if 0 <= m + k < n_points:
                grad[m] += dx * (1 - h[m + k])
            if 0 <= m - k < n_points:
                grad[m] -= dx * h[m - k]
        return grad
    
    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        top_k_indices = np.argsort(corr)[-K:]
        total_grad = np.zeros(n_points)
        for k_idx in top_k_indices:
            lag = k_idx - (n_points - 1)
            grad = compute_gradient(h, lag, n_points, dx)
            total_grad += grad
        alpha = 0.1
        h_new = h - alpha * total_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            alpha *= 1.1
        else:
            alpha *= 0.5
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
