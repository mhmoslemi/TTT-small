    for _ in range(max_iter):

            mutant = pop[a] + mut_factor * (pop[b] - pop[c])
            trial = np.copy(pop[i])
            for j in range(n_points):

            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)
            c5 = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx

    best_h = pop[np.argmin([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in pop])]
