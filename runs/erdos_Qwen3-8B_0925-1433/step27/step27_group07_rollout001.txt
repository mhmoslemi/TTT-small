        if sum_ind != 0:
            individual = individual * (target_sum / sum_ind)
        else:
            individual = np.zeros_like(individual)
        population[i] = individual
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + mut_factor * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < cross_rate:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            if sum_trial != 0:
                trial = trial * (target_sum / sum_trial)
            else:
                trial = np.zeros_like(trial)
            trial_c5 = compute_c5(trial)
            current_c5 = compute_c5(population[i])
            if trial_c5 < current_c5:
                population[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        best_h = best_h * (target_sum / sum_best)
    else:
        best_h = np.zeros_like(best_h)
    return (best_h, best_c5, n_points)
