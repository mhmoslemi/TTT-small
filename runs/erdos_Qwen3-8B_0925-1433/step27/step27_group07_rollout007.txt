        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(corr * dx)
        return c5

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100}
    )

    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
