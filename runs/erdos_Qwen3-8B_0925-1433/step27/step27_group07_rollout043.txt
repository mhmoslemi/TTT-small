def run(seed=42, budget_s=600):

    for s in subpopulations:
        s[:] = np.clip(s, 0, 1)
        s[:] = s * target_sum / np.sum(s)

    max_iter = 500
    start_time = time.time()
    global_best_c5 = best_c5
    global_best_h = best_h

                noise = np.random.normal(0, 0.1 * (1 - gen/max_iter), n_points)
