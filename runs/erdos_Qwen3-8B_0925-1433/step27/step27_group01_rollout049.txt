        candidates = population
        for h_candidate in candidates:
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            max_index = np.argmax(corr)
            lag = (max_index - (n_points - 1)) * dx
            end_x = 2 - lag
            end_idx = int(end_x / dx)
            perturb_mask = np.zeros(n_points, dtype=bool)
            perturb_mask[:end_idx + 1] = True
            pert = np.random.normal(0, 0.01, n_points)
            h_perturbed = h_candidate + pert * perturb_mask
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            population.append(h_perturbed)
        # Select and refine
        for h_candidate in population[:100]:
            bounds = Bounds(0, 1)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(obj, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                population.append(h_new)
        # Update best
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
        # Prune population
        population = np.unique(population, axis=0).tolist()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
