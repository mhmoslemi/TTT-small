import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = 600
    learning_rate = 0.15
    momentum = 0.9
    decay_rate = 0.999
    line_search_steps = 5
    line_search_alpha = 0.8
    start_time = time.time()
    grad_prev = np.zeros(n_points)
    perturb_prob = 0.2
    momentum_decay = 0.9995
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        lag = max_index - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + lag
            if 0 <= i_plus_k < n_points:
                gradient[i] += (1 - h[i_plus_k]) * dx
            i_minus_k = i - lag
            if 0 <= i_minus_k < n_points:
                gradient[i] -= h[i_minus_k] * dx
        
        grad_direction = momentum * grad_prev + gradient
        grad_prev = grad_direction
        
        alpha = learning_rate
        for _ in range(line_search_steps):
            new_h = h + alpha * grad_direction
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new == 0:
                new_h = np.zeros_like(new_h)
            else:
                new_h = new_h * target_sum / sum_new
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                h = new_h
                learning_rate *= decay_rate
                break
            else:
                alpha *= line_search_alpha
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)

            if sum_h != 0:
                h = h * target_sum / sum_h
        
        momentum *= momentum_decay
    
    h_values = np.clip(best_h, 0, 1)
