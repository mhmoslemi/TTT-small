import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = 2000
    learning_rate = 0.15
    decay_rate = 0.999
    line_search_steps = 10
    line_search_alpha = 0.5
    perturb_freq = 50
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        top_k = 5
        top_indices = np.argsort(cross_corr)[-top_k:]
        lag_values = [index - (n_points - 1) for index in top_indices]
        
        gradient = np.zeros(n_points)
        for k in lag_values:
            for i in range(n_points):
                i_plus_k = i + k
                if 0 <= i_plus_k < n_points:
                    gradient[i] += (1 - h[i_plus_k]) * dx
                i_minus_k = i - k
                if 0 <= i_minus_k < n_points:
                    gradient[i] -= h[i_minus_k] * dx
        
        alpha = learning_rate
        for _ in range(line_search_steps):
            new_h = h + alpha * gradient
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new == 0:
                new_h = np.zeros_like(new_h)
            else:
                new_h = new_h * target_sum / sum_new
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                h = new_h
                learning_rate *= decay_rate
                break
            else:
                alpha *= line_search_alpha
        
        if iter % perturb_freq == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
