    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = 1000
    start_time = time.time()
    steps = [0.1, 0.05, 0.025, 0.01, 0.005, 0.0025, 0.001]
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        overlaps = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(overlaps) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        k_max = np.argmax(overlaps)
        grad = 1 - h[(np.arange(n_points) + k_max) % n_points]
        direction = -grad
        best_alpha = 0
        best_new_overlap = float('inf')
        for alpha in steps:
            new_h = h + alpha * direction
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            new_overlaps = np.correlate(new_h, 1 - new_h, mode='full')
            new_overlap = new_overlaps[k_max] * dx
            if new_overlap < best_new_overlap:
                best_new_overlap = new_overlap
                best_alpha = alpha
        h = h + best_alpha * direction
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if it % 100 == 0:
            if np.abs(current_c5 - best_c5) < 1e-6:
                break
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
