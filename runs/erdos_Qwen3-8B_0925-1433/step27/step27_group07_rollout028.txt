        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    space = [(0.0, 1.0) for _ in range(n_points)]

    # Initialize with initial_h and some random samples
    initial_points = [initial_h]
    for _ in range(49):
        initial_points.append(np.random.rand(n_points))

    bo = BayesianOptimization(objective, space, n_initial_points=50)

    start_time = time.time()
    for i in range(150):
        if time.time() - start_time > budget_s:
            break
        bo.suggest()
        bo.expand()

    best_result = bo.get_result()
    best_params = best_result.x
    best_h = best_params * (target_sum / np.sum(best_params))
    best_h = np.clip(best_h, 0, 1)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
