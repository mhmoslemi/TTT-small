def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(theta):
        theta = np.clip(theta, 0, 1)
        theta = theta / np.sum(theta)
        h = theta
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

        if time.time() - start_time > budget_s:

        opt.suggest()
    best_result = opt.results[0]

    best_h = best_theta / np.sum(best_theta)

    h_values = np.clip(best_h, 0, 1)
