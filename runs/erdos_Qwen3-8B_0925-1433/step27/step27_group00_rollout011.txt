            velocities[i] = vel

            # Update position
            new_p = particles[i] + vel
            new_p = np.clip(new_p, 0, 1)
            new_p = new_p * target_sum / np.sum(new_p)
            particles[i] = new_p

            # Evaluate new value
            curr_c5 = evaluate_erdos_solution(particles[i])
            if curr_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = curr_c5
                if curr_c5 < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = curr_c5
                    best_h = gbest.copy()
                    best_c5 = gbest_value

        if np.min(pbest_values) == best_c5:
            break

    return (best_h, best_c5, n_points)
