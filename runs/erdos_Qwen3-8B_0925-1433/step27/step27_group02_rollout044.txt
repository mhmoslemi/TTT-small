import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    learning_rate = 0.1
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 1
    max_iter = 1000
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        lag = max_index - (n_points - 1)
        
        # Gradient for current lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + lag
            if 0 <= i_plus_k < n_points:
                grad[i] += (1 - h[i_plus_k]) * dx
            i_minus_k = i - lag
            if 0 <= i_minus_k < n_points:
                grad[i] -= h[i_minus_k] * dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h -= step
        
        # Clamp and normalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros_like(h)
        else:
            h = h * target_sum / sum_h
        
        # Evaluate
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Perturb
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            h = h * target_sum / sum_h
        
        t += 1
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
