        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        current_k = np.argmax(corr)
        lag = current_k - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Term1: (1 - h[j + lag]) * dx if j + lag is in [0, n_points - 1]
            term1 = 0.0
            if lag > 0:
                if j + lag < n_points:
                    term1 = (1 - h[j + lag]) * dx
            elif lag < 0:
                if j + lag >= 0:
                    term1 = (1 - h[j + lag]) * dx
            else:
                term1 = (1 - h[j]) * dx
            # Term2: -h[j - lag] * dx if j - lag is in [0, n_points - 1]
            term2 = 0.0
            if lag < 0:
                if j - lag >= 0:
                    term2 = h[j - lag] * dx
            elif lag > 0:
                if j - lag >= 0:
                    term2 = h[j - lag] * dx
            else:
                term2 = h[j] * dx
            grad[j] = term1 - term2
        h += alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        if current_c5 == best_c5:
            early_stop += 1
        else:
            early_stop = 0
        if early_stop > 200:
            break
        alpha *= alpha_decay
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
