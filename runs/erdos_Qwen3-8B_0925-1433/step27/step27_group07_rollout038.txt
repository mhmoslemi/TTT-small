    N = 50
    max_iter = 300
    inertia = 0.8
    cogn = 1.5
    soc = 1.5
    mut_rate = 0.03
    mut_scale = 0.05

    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.uniform(-0.1, 0.1, (N, n_points))
    
    pbest = particles.copy()
    pbest_c5 = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    gbest = particles[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)

        if time.time() - start > budget_s - 15:
            break

            velocities[i] = inertia * velocities[i] +\
                            cogn * r1 * (pbest[i] - particles[i]) +\
                            soc * r2 * (gbest - particles[i])

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if c5 < pbest_c5[i]:

        gbest = pbest[np.argmin(pbest_c5)]
