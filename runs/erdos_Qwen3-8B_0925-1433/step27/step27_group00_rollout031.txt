import time

    learning_rate = 0.2
    decay_factor = 0.998
    perturb_interval = 100
    perturb_strength = 0.15

    start_time = time.time()

        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[-3:]  # top 3 lags
        grad = np.zeros(n_points)

        grad /= len(top_lags)
