import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    alpha = 0.1
    decay_rate = 0.995
    perturb_prob = 0.05
    max_iter = 10000
    start_time = time.time()
    iter_count = 0
    learning_rate = alpha

    while time.time() - start_time < budget_s and iter_count < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx

        m = np.argmax(corr)
        k = (n_points - 1) - m

        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k] * dx

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if np.random.rand() < perturb_prob:
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        if current_max < best_c5:
            best_h = h.copy()
            best_c5 = current_max

        learning_rate *= decay_rate
        iter_count += 1

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
