import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr) * dx
        return max_corr
    
    def grad_func(h_flat):
        h = h_flat.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_lag = i + lag
            if 0 <= i_plus_lag < n_points:
                grad[i] = dx * (1 - h[i_plus_lag])
        return grad
    
    def con_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    res = minimize(objective, h.flatten(), method='SLSQP', jac=grad_func, bounds=bounds, constraints=[{'type': 'eq', 'fun': con_func}], options={'maxiter': 100, 'ftol': 1e-8})
    
    best_h = res.x.reshape(n_points)
    best_c5 = res.fun
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * target_sum / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
