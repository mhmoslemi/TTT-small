                pbest_c5[i] = c5

        gbest_c5 = np.min(pbest_c5)
        # Mutate global best
        if np.random.rand() < mutation_rate:
            gbest += np.random.normal(0, mutation_scale, n_points)
            gbest = np.clip(gbest, 0, 1)
            gbest = gbest * target_sum / np.sum(gbest)

    h_values = h_values * target_sum / np.sum(h_values)
