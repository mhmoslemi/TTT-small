import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # initial bound
    
    alpha = 0.05
    decay_rate = 0.999
    perturb_prob = 0.05
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Compute cross-correlation and find maximum lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        m = np.argmax(corr)
        k = m - (n_points - 1)
        
        # Compute gradient for lag k
        grad = np.zeros(n_points)
        for j in range(n_points):
            if (j + k) < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if (j - k) >= 0:
                grad[j] -= h[j - k] * dx
        
        # Update h using gradient descent
        step_size = -alpha * grad
        h += step_size
        
        # Clip and rescale to [0, 1] and maintain sum constraint
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Occasionally perturb
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        # Update best solution
        current_c5 = max_val * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Decay learning rate
        alpha *= decay_rate
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
