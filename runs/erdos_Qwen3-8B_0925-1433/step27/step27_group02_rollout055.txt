        bounds = [(0, 1) for _ in range(n_points)]
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        
        if result.success:
            h = result.x
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
