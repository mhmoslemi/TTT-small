            best_idx = idx[np.argmin(fitness[idx])]
            parents.append(pop[best_idx].copy())
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                offspring.append(child1)
                offspring.append(child2)
        mutated = []
        for h in offspring:
            mut = np.random.uniform(-0.1, 0.1, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)
            if sum_mut == 0:
                h_mut = np.full_like(h_mut, target_sum / n_points)
            else:
                h_mut = h_mut * (target_sum / sum_mut)
            mutated.append(h_mut)
        current_best_c5 = best_c5
        current_best_h = best_h.copy()
        for h_mut in mutated:
            corr = np.correlate(h_mut, 1 - h_mut, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < current_best_c5:
                current_best_c5 = current_c5
                current_best_h = h_mut.copy()
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
        pop = mutated
    return (best_h, best_c5, n_points)
