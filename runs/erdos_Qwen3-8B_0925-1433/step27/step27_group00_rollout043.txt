import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * (target_sum / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (target_sum / np.sum(initial_h))

    pop_size = 50
    particles = [initial_h.copy()]
    for _ in range(pop_size - 1):
        particles.append(np.random.rand(n_points) * (target_sum / n_points))
    particles = np.array(particles)

    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = particles.copy()
    pbest_values = np.array([evaluate_erdos_solution(p) for p in particles])
    gbest = pbest[np.argmin(pbest_values)]
    best_c5 = np.min(pbest_values)
    best_h = gbest.copy()

    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 500
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            current_val = evaluate_erdos_solution(particles[i])
            if current_val < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_val
                if current_val < best_c5:
                    best_c5 = current_val
                    best_h = particles[i].copy()

        gbest_idx = np.argmin(pbest_values)
        gbest = pbest[gbest_idx].copy()
        best_h = gbest.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
