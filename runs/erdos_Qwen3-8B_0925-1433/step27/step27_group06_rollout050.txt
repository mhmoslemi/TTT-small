import numpy as np
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    max_iterations = int(budget_s / 0.1)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        selected_indices = np.random.choice(n_points, 5, replace=False)
        delta = np.random.uniform(-perturbation_size, perturbation_size, size=5)
        new_h = h.copy()
        new_h[selected_indices] += delta
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:

        new_h = new_h * target_sum / new_sum
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0:
            h = new_h.copy()
            current_c5 = new_c5
        else:
            if np.random.rand() < np.exp(-delta_c5 / T):
                h = new_h.copy()
                current_c5 = new_c5
        T *= alpha

    h_values = h_values * target_sum / np.sum(h_values)
