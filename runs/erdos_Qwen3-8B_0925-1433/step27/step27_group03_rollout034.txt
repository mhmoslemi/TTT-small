        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += dx * (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= dx * h[m - k]
        return grad
    
    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        # Get top 5 lags
        top_lags = np.argsort(corr)[-5:]
        # Compute gradients for these lags
        gradients = []
        for k in top_lags:
            grad = compute_gradient(h, k, n_points, dx)
            gradients.append(grad)
        avg_grad = np.mean(gradients, axis=0)
        # Perform line search
        steps = [0.05, 0.1, 0.2, 0.3]
        best_h_new = h.copy()
        best_c5_new = max_c5
        for step in steps:
            h_new = h - step * avg_grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new
        # Update h
        h = best_h_new
        # Check if we improved
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = h.copy()
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
