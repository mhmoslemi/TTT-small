    best_c5 = min(best_c5, 0.381)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',

        recombination=0.9,
        tol=1e-6,
        maxiter=500,

        seed=seed
    )
