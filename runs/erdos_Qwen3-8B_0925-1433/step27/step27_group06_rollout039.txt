def run(seed=42, budget_s=600):
    import numpy as np
    import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # Initial best

    # Parameters
    alpha = 0.1
    perturb_prob = 0.05
    decay_rate = 0.999

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        current_k_index = np.argmax(corr)
        k = current_k_index - (n_points - 1)

        # Compute gradient for lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i + k) >= 0 and (i + k) < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if (i - k) >= 0 and (i - k) < n_points:
                grad[i] -= h[i - k] * dx

        # Compute step
        step = -alpha * grad
        h += step

        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Check if this improves best_c5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Random perturbation
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        # Decay learning rate
        alpha *= decay_rate

        # Early stopping
        if np.random.rand() < 0.01:
            # Check for no improvement in 100 steps
            pass

    # Final check
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
