        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        new_c5 = compute_c5(h_candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
        h = h_candidate
    return (best_h, best_c5, n_points)
