    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        corr = np.correlate(h, 1 - h, mode='full') * dx
        top_lags = np.argsort(corr)[-5:]
        grad = np.zeros_like(h)
        for lag in top_lags:
            k = lag % n_points
            for i in range(n_points):
                if (i + k) < n_points:

                if (i - k) >= 0:
                    grad[i] -= h[i - k] * dx
        alpha = 1.0
