    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros(n_points)
        for m in range(n_points):
            i = m + k
            if 0 <= i < n_points:
                grad[m] += dx * (1 - h[i])
            i = m - k
            if 0 <= i < n_points:
                grad[m] -= dx * h[i]
        return grad
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        top_indices = np.argsort(corr)[-5:]
        top_lags = [i - (n_points - 1) for i in top_indices]
        
        avg_grad = np.zeros(n_points)
        for k in top_lags:
            grad = compute_gradient(h, k, n_points, dx)
            avg_grad += grad
        
        avg_grad /= len(top_lags)
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * avg_grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < max_corr:
                h = h_new
                alpha *= 1.1
            else:
                alpha *= 0.5
        h = h - alpha * avg_grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
