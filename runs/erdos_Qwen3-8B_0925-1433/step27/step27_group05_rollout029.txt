import time

def run(seed=42, budget_s=600):

    max_iter = 1000
    line_search_steps = 50
    cooling_rate = 0.999
    tolerance = 1e-6
    time_start = time.time()
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')

        top_lags = np.argsort(current_c5)[-5:]
        grad = np.zeros_like(h)
        
        for lag in top_lags:
            k = lag % n_points
            for i in range(n_points):
                if (i + k) < n_points:
                    grad[i] += (1 - h[i + k]) * dx
                if (i - k) >= 0:
                    grad[i] -= h[i - k] * dx

            h_test = h + alpha * grad

            corr_test = np.correlate(h_test, 1 - h_test, mode='full') * dx
            if corr_test[np.argmax(current_c5)] < current_c5[np.argmax(current_c5)]:
