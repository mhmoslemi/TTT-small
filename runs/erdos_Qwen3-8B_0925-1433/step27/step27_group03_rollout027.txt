    def compute_grad_lags(h, lags, n_points, dx):

                if m + k < n_points:
                    grad[m] += dx * (1 - h[m + k])
                if m - k >= 0:
                    grad[m] -= dx * h[m - k]
        return grad / len(lags)
    
    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
