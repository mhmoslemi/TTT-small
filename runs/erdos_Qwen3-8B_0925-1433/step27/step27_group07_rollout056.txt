import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    pop_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 500
    particles = np.random.rand(pop_size, n_points)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)
    
    pbest = particles.copy()
    pbest_c5 = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    gbest = pbest[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)

    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = particles[a] + F * (particles[b] - particles[c])
            trial = np.random.rand(n_points) * mutant + (1 - np.random.rand(n_points)) * particles[i]
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            trial = trial * target_sum / sum_trial
            
            c5 = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
            if c5 < pbest_c5[i]:
                pbest[i] = trial.copy()
                pbest_c5[i] = c5
        gbest = pbest[np.argmin(pbest_c5)]
        gbest_c5 = np.min(pbest_c5)
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
