from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    population = [h.copy()]
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.015, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        population.append(h_pert)
    
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
