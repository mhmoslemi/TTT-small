            if alpha > 1.0:
                alpha = 1.0
        new_h = h + alpha * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * target_sum / np.sum(new_h)
        h = new_h.copy()
        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
