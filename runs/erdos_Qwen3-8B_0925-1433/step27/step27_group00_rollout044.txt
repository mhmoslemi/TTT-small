import time

def run(seed=42, budget_s=600):

    initial_h = np.clip(initial_h, 0, 1)
    
    pop = [initial_h.copy() for _ in range(20)]
    for _ in range(10):
        rand_vec = np.random.rand(n_points) * (target_sum / n_points)
        rand_vec = np.clip(rand_vec, 0, 1)
        rand_vec = rand_vec * (target_sum / np.sum(rand_vec))
        pop.append(rand_vec)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in pop]
        idxs = np.argsort(fitness)
        parents = [pop[i] for i in idxs[:15]]
        
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                offspring.append(child1)
                offspring.append(child2)
        
        mutated = []
        for h in offspring:
            mut = np.random.uniform(-0.1, 0.1, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)
            if sum_mut == 0:
                h_mut = np.full_like(h_mut, target_sum / n_points)
            else:
                h_mut = h_mut * (target_sum / sum_mut)
            mutated.append(h_mut)
        
        new_pop = mutated
        if best_c5 > fitness[np.argmin(fitness)]:
            new_pop[0] = best_h.copy()
        
        pop = new_pop
        generation += 1
    
    best_h = pop[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
