        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        top_indices = np.argsort(corr)[-K:]
        K_lags = [i - (n_points - 1) for i in top_indices]
        grad = compute_combined_gradient(h, K_lags, n_points, dx)
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new
                alpha *= 1.1
            else:
                alpha *= 0.5
        h = h_new
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
