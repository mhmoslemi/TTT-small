            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * epsilon)
        direction = -grad
        alpha = 0.1
        for _ in range(10):
            new_h = h + alpha * direction
            new_h = np.clip(new_h, 0, 1)
            new_weights = new_h * target_sum / np.sum(new_h)
            corr_new = np.correlate(new_weights, 1 - new_weights, mode='full')
            new_c5 = corr_new[max_idx] * dx
            if new_c5 < current_c5:
                h = new_weights
                current_c5 = new_c5
                alpha *= 1.1
            else:
                alpha /= 2
                break
        if _ % 100 == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
