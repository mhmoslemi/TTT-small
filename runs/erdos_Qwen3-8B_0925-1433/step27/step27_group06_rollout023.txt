def run(seed=42, budget_s=600):
    import numpy as np
    import time
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    learning_rate = 0.05
    decay_rate = 0.995
    perturb_prob = 0.05
    start_time = time.time()
    current_c5 = best_c5
    for _ in range(100000):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        current_k_idx = np.argmax(corr)
        # Compute lag k
        k = current_k_idx - (n_points -1)
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >=0:
                grad[i] -= h[i - k] * dx
        # Update h
        h -= learning_rate * grad
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to ensure sum is target_sum
        h = h * target_sum / np.sum(h)
        # Check if improvement
        new_c5 = max_c5
        if new_c5 < current_c5:
            current_c5 = new_c5
            best_h = h.copy()
        # Perturb with probability
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        # Decay learning rate
        learning_rate *= decay_rate
        # Early stopping
        if current_c5 == best_c5:
            if perturb_prob < 0.1:
                perturb_prob += 0.01
            else:
                break
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, current_c5, n_points)
