import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    if best_c5 < 0.381:
        return (best_h, best_c5, n_points)

    def objective(x):
        h_current = x.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full') * dx
        return np.max(corr)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_sum(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 1000})
    
    if result.success:

        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, best_c5, n_points)
