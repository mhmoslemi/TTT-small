import numpy as np
import time
from skopt import BayesianOptimization
from skopt.gp_kernels import Matern

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(theta):
        theta = np.array(theta, dtype=float)
        theta_sum = np.sum(theta)
        h = theta / theta_sum
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return -c5
    
    bounds = [(0.0, 1.0)] * n_points
    optimizer = BayesianOptimization(
        f=objective,
        pbounds=bounds,
        random_state=seed
    )
    
    start_time = time.time()
    max_iter = 500
    while time.time() - start_time < budget_s:
        if max_iter <= 0:
            break
        optimizer.suggest()
        optimizer.maximize(n_iter=1)
        max_iter -= 1
    
    best_result = optimizer.get_results()[0]
    best_theta = np.array(best_result["params"])
    theta_sum = np.sum(best_theta)
    best_h = best_theta / theta_sum
    best_h = np.clip(best_h, 0, 1)
    best_c5 = -best_result["target"]
    
    return (best_h, best_c5, n_points)
