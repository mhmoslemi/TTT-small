import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population = [h_initial.copy() for _ in range(20)]
    for _ in range(10):
        rand_vec = np.random.rand(n_points) * (target_sum / n_points)
        rand_vec = np.clip(rand_vec, 0, 1)
        sum_rand = np.sum(rand_vec)
        if sum_rand == 0:
            rand_vec = np.full_like(rand_vec, target_sum / n_points)
        else:
            rand_vec = rand_vec * (target_sum / sum_rand)
        population.append(rand_vec)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            fitness.append(current_c5)
        best_index = np.argmin(fitness)
        current_best = fitness[best_index]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[best_index].copy()
        parents = []
        for _ in range(len(population)):
            idxs = np.random.choice(len(population), size=5, replace=False)
            best_idx = idxs[np.argmin(fitness[idxs])]
            parents.append(population[best_idx].copy())
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else parents[i]
            split = np.random.randint(1, n_points)
            child1 = np.concatenate([p1[:split], p2[split:]])
            child2 = np.concatenate([p2[:split], p1[split:]])
            offspring.append(child1)
            offspring.append(child2)
        mutated = []
        for h in offspring:
            mut = np.random.uniform(-0.1, 0.1, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)
            if sum_mut == 0:
                h_mut = np.full_like(h_mut, target_sum / n_points)
            else:
                h_mut = h_mut * (target_sum / sum_mut)
            mutated.append(h_mut)
        population = mutated
        if best_h is not None:
            worst_idx = np.argmax([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in mutated])
            if best_c5 < [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in mutated][worst_idx]:
                population[worst_idx] = best_h.copy()
    return (best_h, best_c5, n_points)
