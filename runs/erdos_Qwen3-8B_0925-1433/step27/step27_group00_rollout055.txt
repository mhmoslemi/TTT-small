import time

def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)
    
    learning_rate = 0.15
    decay_factor = 0.997
    momentum = 0.8
    momentum_decay = 0.9995

    perturb_strength = 0.15
    iteration = 0
    start_time = time.time()
    
    velocity = np.zeros(n_points)
    top_lags = 3
    lag_weights = np.array([1.0, 0.8, 0.6], dtype=float)
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_values = np.argsort(corr)[::-1][:top_lags]
        k_max = k_values[0]
        current_c5 = corr[k_max] * dx
        
        grad = np.zeros(n_points)
        for k in k_values:
            k %= n_points
            for i in range(n_points):
                j = (i - k) % n_points
                grad[i] += (1 - h[(i + k) % n_points]) - h[j]
        grad *= lag_weights
        
        velocity = momentum * velocity + grad
        h_new = h - learning_rate * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if iteration % perturb_interval == 0:
            perturb = np.random.uniform(-perturb_strength, perturb_strength, n_points)
            h_new += perturb
            h_new = np.clip(h_new, 0, 1)
