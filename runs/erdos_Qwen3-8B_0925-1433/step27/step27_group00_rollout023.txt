            perturb = np.random.uniform(-perturb_strength, perturb_strength, n_points)

            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(np.clip(h_new, 0, 1)))
        h = h_new
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_factor
        iteration += 1
    
    return (best_h, best_c5, n_points)
