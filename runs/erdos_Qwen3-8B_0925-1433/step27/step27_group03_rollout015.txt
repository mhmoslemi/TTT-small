    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += dx * (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= dx * h[m - k]
        return grad

    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    K = 5  # number of top lags to consider
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        # Select top K lags
        top_indices = np.argsort(-corr)[:K]
        top_lags = [i - (n_points - 1) for i in top_indices]
        combined_grad = np.zeros(n_points)
        for k in top_lags:
            grad_k = compute_gradient(h, k, n_points, dx)
            combined_grad += grad_k
        # Perform gradient descent
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * combined_grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < current_max:
                h = h_new
                alpha *= 1.1
            else:
                alpha *= 0.5
        h = h - alpha * combined_grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
