        result = optimize.minimize(objective, h, method='Nelder-Mead', bounds=bounds, options={'xatol': 1e-6, 'fatol': 1e-6})
        h_values = result.x
        c5_bound = objective(h_values)
    except:
        h_values = h.copy()
        c5_bound = best_c5
    
    # Final clip and rescale
    h_final = np.clip(h_values, 0, 1)
    sum_final = np.sum(h_final)
    if sum_final > 0:
        h_final = h_final * (target_sum / sum_final)
    else:
        h_final = np.zeros_like(h_values)
    
    return (h_final, c5_bound, n_points)
