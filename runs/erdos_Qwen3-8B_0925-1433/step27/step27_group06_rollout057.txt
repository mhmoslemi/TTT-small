import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    h = initial_h * target_sum / np.sum(initial_h)
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    if best_c5 < 0.38085:
        return (h, best_c5, n_points)
    
    population = [h.copy() for _ in range(50)]
    for p in population:
        p += np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(50):
            idxs = np.random.choice(50, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = x1 + 0.8 * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_pop.append(mutant)
        combined = population + new_pop
        obj_vals = np.array([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in combined])
        order = np.argsort(obj_vals)
        population = [combined[i] for i in order[:50]]
        if obj_vals[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = obj_vals[0]
        if np.allclose(obj_vals, obj_vals[0], atol=1e-5):
            break
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
