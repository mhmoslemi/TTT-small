import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_gradient(h, k):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= h[m - k]
        return grad

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        K = 5
        indices = np.argsort(corr)[::-1]
        top_k_indices = indices[:K]
        top_lags = [idx - (n_points - 1) for idx in top_k_indices]
        grad_total = np.zeros(n_points)
        for lag in top_lags:
            grad = compute_gradient(h, lag)
            grad_total += grad
        alpha = 0.1
        for _ in range(5):

            if new_c5 < current_max:
