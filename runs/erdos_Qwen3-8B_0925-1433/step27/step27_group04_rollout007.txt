            if current_fitness[a] < current_fitness[b]:
                parent1 = remaining[a]
            else:
                parent1 = remaining[b]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent1[split:]])
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.1, size=n_points)
                    child = np.clip(child + noise, 0, 1)
                    child = child * target_sum / np.sum(child)
            else:
                child = parent1.copy()
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.1, size=n_points)
                    child = np.clip(child + noise, 0, 1)
                    child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        top_indices = np.argsort(current_fitness)[:int(population_size * 0.05)]
        top_h = [population[i] for i in top_indices]
        for i, h in enumerate(top_h):
            corr = np.correlate(h, 1 - h, mode='full')
            best_lag_index = np.argmax(corr * dx)
            k = best_lag_index - (n_points - 1)
            grad = np.zeros_like(h)
            for idx in range(n_points):
                if idx + k < n_points:
                    grad[idx] += (1 - h[idx + k])
                if idx - k >= 0:
                    grad[idx] -= h[idx - k]
            alpha = 1.0
            for _ in range(line_search_steps):
                h_test = h + alpha * grad
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * target_sum / np.sum(h_test)
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                current_c5 = corr[best_lag_index] * dx
                new_c5 = corr_test[best_lag_index] * dx
                if new_c5 < current_c5:
                    alpha *= 1.1
                else:
                    alpha *= 0.9
                if alpha < 1e-6:
                    break
            h_new = h + alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            population[top_indices[i]] = h_new
        
        population = new_population
        if time.time() - start_time > budget_s:
            break
        
        current_fitness = [evaluate_fitness(h) for h in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
