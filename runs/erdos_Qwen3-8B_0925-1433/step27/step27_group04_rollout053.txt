import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    P = 20

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    # Initialize particles
    particles = []
    for _ in range(P):
        if _ == 0:
            h = np.array(initial_h_values, dtype=float)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles.append({
            'position': h.copy(),
            'velocity': np.zeros(n_points),
            'pbest': h.copy(),
            'pbest_fitness': compute_c5(h)
        })

    global_best = min(particles, key=lambda p: p['pbest_fitness'])

    w = 0.9
    c1 = 2.0
    c2 = 2.0
    max_iter = 500
    time_start = time.time()

    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        for p in particles:
            r1, r2 = np.random.rand(), np.random.rand()
            p['velocity'] = w * p['velocity'] + c1 * r1 * (p['pbest'] - p['position']) + c2 * r2 * (global_best['pbest'] - p['position'])
            p['velocity'] = np.clip(p['velocity'], -1.0, 1.0)
            p['position'] += p['velocity']
            p['position'] = np.clip(p['position'], 0, 1)
            p['position'] = p['position'] * target_sum / np.sum(p['position'])
            new_c5 = compute_c5(p['position'])
            if new_c5 < p['pbest_fitness']:
                p['pbest'] = p['position'].copy()
                p['pbest_fitness'] = new_c5
            if new_c5 < global_best['pbest_fitness']:
                global_best = p

    h_values = global_best['pbest']
    c5_bound = global_best['pbest_fitness']
    return (h_values, c5_bound, n_points)
