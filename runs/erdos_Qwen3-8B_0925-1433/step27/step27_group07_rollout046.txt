from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    initial_h = initial_h_values
    n_points = len(initial_h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    swarm_size = 100
    max_iter = 800
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    mutation_rate = 0.05
    mutation_scale = 0.05
    
    particles = np.random.rand(swarm_size, n_points)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    pbest = particles.copy()
    pbest_c5 = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    gbest = particles[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia_weight * velocities[i] + \
                            cognitive_coeff * r1 * (pbest[i] - particles[i]) + \
                            social_coeff * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])

        gbest = pbest[np.argmin(pbest_c5)]
