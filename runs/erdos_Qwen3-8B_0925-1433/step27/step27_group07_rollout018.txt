            offspring = sbx(p1, p2, eta_crossover)
            if np.random.rand() < 0.05 * (1 - gen / max_gen):
                mutation = np.random.normal(0, mutation_scale, size=offspring.shape)
                offspring += mutation
                offspring = np.clip(offspring, 0, 1)
            
            offspring_sum = np.sum(offspring)
            if offspring_sum > 0:
                offspring = offspring * target_sum / offspring_sum
            
            c5 = compute_c5(offspring)
            c5s = np.array([compute_c5(ind) for ind in subpop])
            worst_idx = np.argmax(c5s)
            if c5 < c5s[worst_idx]:
                subpop[worst_idx] = offspring
        
        for subpop_idx in range(num_subpop):
            subpop = subpops[subpop_idx]
            c5s = np.array([compute_c5(ind) for ind in subpop])
            worst_indices = np.argsort(c5s)[:int(subpop_size * elite_ratio)]
            for idx in worst_indices:
                other_pop = subpops[np.random.choice([i for i in range(num_subpop) if i != subpop_idx])]
                best_in_other = other_pop[np.argmin([compute_c5(ind) for ind in other_pop])]
                subpop[idx] = best_in_other.copy()
        
        for subpop in subpops:
            for ind in subpop:
                current_c5 = compute_c5(ind)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = ind.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
