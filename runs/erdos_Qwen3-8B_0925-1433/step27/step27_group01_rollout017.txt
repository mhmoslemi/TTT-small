import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    population = [h.copy()]
    
    # Generate initial directional perturbations
    for _ in range(5):
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k_center = idx - (n_points - 1)
        shifted_h = np.roll(h, k_center)
        product = h * (1 - shifted_h)
        max_indices = np.where(product == product.max())[0]
        if len(max_indices) > 0:
            perturbation = np.zeros_like(h)
            perturbation[max_indices] = -0.02
        else:
            perturbation = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(h + perturbation, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        population.append(new_h)
    
    # Local optimization phase
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        if not population:
            break
        candidate = population.pop()
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            population.append(new_h)
            curr_c5 = obj(new_h)
            if curr_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = curr_c5
        else:
            population.append(candidate)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
