    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb a random subset of h
        indices = np.random.choice(n_points, size=10, replace=False)
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = h.copy()
        new_h[indices] += perturb[indices]
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        # Compute new c5
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        # Acceptance probability
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                h = new_h.copy()
        T *= cooling_rate
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
