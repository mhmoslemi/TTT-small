    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.999
    line_search_steps = 20
    line_search_alpha = 0.5
    start_time = time.time()
    grad_buffer = np.zeros(n_points)
    grad_prev = np.zeros(n_points)
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(cross_corr)[-5:]
        grad = np.zeros(n_points)
        for lag in top_lags:
            i_plus_k = (np.arange(n_points) + lag) % n_points
            i_minus_k = (np.arange(n_points) - lag) % n_points
            grad[i_plus_k] += (1 - h[i_plus_k]) * dx
            grad[i_minus_k] -= h[i_minus_k] * dx
        grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 1e-6 else grad
        grad_direction = momentum * grad_prev + grad
        grad_prev = grad_direction
        alpha = learning_rate
        for _ in range(line_search_steps):

            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new == 0:
                new_h = np.zeros_like(new_h)
            else:
                new_h = new_h * target_sum / sum_new
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                h = new_h
                learning_rate *= decay_rate
                break
            else:
                alpha *= line_search_alpha
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
