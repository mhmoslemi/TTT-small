        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        if iteration % perturb_interval == 0:
            perturb = np.random.uniform(-perturb_strength, perturb_strength, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        new_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(new_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_steps = 0
        else:
            no_improvement_steps += 1
        
        learning_rate *= decay_factor
        if no_improvement_steps >= max_no_improvement:
            perturb_strength *= 1.1
            no_improvement_steps = 0
        
        iteration += 1
    
    return (best_h, best_c5, n_points)
