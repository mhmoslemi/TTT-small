import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += dx * (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= dx * h[m - k]
        return grad
    
    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        k_max = np.argmax(corr)
        grad = compute_gradient(h, k_max, n_points, dx)
        alpha = 0.1
        for _ in range(5):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < max_c5:
                h = h_new
                alpha *= 1.1
            else:
                alpha *= 0.5
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Stochastic correction
        if np.random.rand() < 0.2:
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            contribs = h * (1 - np.roll(h, k_max))
            mask = (np.arange(n_points) + k_max) < n_points
            contribs = contribs[mask]
            idxs = np.random.choice(np.arange(len(contribs)), size=min(5, len(contribs)), replace=False)
            for i in idxs:
                h[int(np.where(mask)[0][i])] = 1 - h[int(np.where(mask)[0][i])]
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
