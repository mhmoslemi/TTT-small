import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_initial = np.array(initial_h, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    n_pop = 50
    noise = np.random.normal(0, 0.05, (n_pop, n_points))
    population = h_initial + noise
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    F = 0.8
    CR = 0.9
    max_iter = 500
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        gbest_idx = np.argmin([compute_c5(population[i]) for i in range(n_pop)])
        gbest = population[gbest_idx]
        for i in range(n_pop):
            a, b = np.random.choice(n_pop, 2, replace=False)
            mutant = gbest + F * (population[a] - population[b])
            target = population[i]
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)
            c5_trial = compute_c5(trial)
            c5_current = compute_c5(target)
            if c5_trial < c5_current:
                population[i] = trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
