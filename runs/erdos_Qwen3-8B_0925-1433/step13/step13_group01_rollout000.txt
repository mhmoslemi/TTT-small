import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = x
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def update_h(h, target_sum):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.ones_like(h) * 0.5
        return h

    # COBYLA global search with structured perturbations
    for _ in range(80):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.1 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = update_h(new_h, target_sum)
        res = minimize(
            obj, new_h, method='COBYLA', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 10}
        )
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # L-BFGS-B local refinement
    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.02 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = update_h(new_h, target_sum)
        res = minimize(
            obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-6, options={'maxiter': 20}
        )
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # SLSQP final fine-tuning
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.005 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)

            tol=1e-8, options={'maxiter': 15}
        )

            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
