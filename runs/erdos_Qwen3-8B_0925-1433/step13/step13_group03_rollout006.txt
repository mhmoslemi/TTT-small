import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) > 0:
            h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        else:
            h_scaled = h_clipped
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    # Initial guess
    h0 = np.array(initial_h_values)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else h0

    # Differential Evolution for global optimization
    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', maxiter=100, popsize=20, tol=1e-8,
        mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h_de = result_de.x
    best_c5_de = objective(best_h_de)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h_de, best_c5_de, n_points)

    # SLSQP refinement
    start_time = time.time()
    def objective_slsqp(h):
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) > 0:
            h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        else:
            h_scaled = h_clipped
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    result_slsqp = minimize(
        objective_slsqp, best_h_de, method='SLSQP', bounds=bounds, tol=1e-8)
    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = objective_slsqp(best_h_slsqp)
    elapsed_slsqp = time.time() - start_time
    if elapsed_slsqp > budget_s - elapsed:
        return (best_h_slsqp, best_c5_slsqp, n_points)

    # L-BFGS-B refinement
    start_time = time.time()
    result_lbb = minimize(
        objective_slsqp, best_h_slsqp, method='L-BFGS-B', bounds=bounds, tol=1e-8)
    best_h_lbb = result_lbb.x
    best_c5_lbb = objective_slsqp(best_h_lbb)
    elapsed_lbb = time.time() - start_time
    if elapsed_lbb > budget_s - (elapsed + elapsed_slsqp):
        return (best_h_lbb, best_c5_lbb, n_points)

    # Choose best solution
    best_h = best_h_lbb
    best_c5 = best_c5_lbb
    if best_c5_slsqp < best_c5:
        best_h, best_c5 = best_h_slsqp, best_c5_slsqp
    if best_c5_de < best_c5:
        best_h, best_c5 = best_h_de, best_c5_de

    return (best_h, best_c5, n_points)
