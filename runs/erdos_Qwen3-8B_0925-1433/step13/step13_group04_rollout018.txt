import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # PSO parameters
    particles = []
    velocities = []
    pbest = []
    pbest_obj = []
    gbest = initial_h_values.copy()
    gbest_obj = obj(gbest)

    # Initialize particles
    for _ in range(50):
        if _ == 0:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h)
        pbest_obj.append(obj(h))

    for it in range(1000):
        if time.time() > time.time() + budget_s:
            break
        for i in range(len(particles)):
            r1, r2 = np.random.rand(2)
            velocities[i] += 0.8 * (np.random.rand(n_points) - 0.5) + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            curr_obj = obj(particles[i])
            if curr_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = curr_obj
        curr_gbest = np.argmin(pbest_obj)
        if pbest_obj[curr_gbest] < gbest_obj:
            gbest = pbest[curr_gbest].copy()
            gbest_obj = pbest_obj[curr_gbest]
        if gbest_obj < 0.3807:
            break

    # Refinement
    def refine(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return obj(x)
    res = minimize(refine, gbest, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-8)
    if res.success:
        gbest = res.x

    return (gbest, gbest_obj, n_points)
