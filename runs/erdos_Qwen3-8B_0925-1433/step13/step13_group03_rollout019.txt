from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

        sum_h = np.sum(h_)

        corr = np.correlate(h_, 1 - h_, mode='full')
        alpha = 10.0
        smooth_max = (1/alpha) * np.log(np.sum(np.exp(alpha * corr)))
        return smooth_max * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h) + 1e-6

    start_time = time.time()
