def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    start_time = time.time()
    learning_rate = 0.05
    max_iterations = 10000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        worst_lag_idx = np.argmax(corr)
        current_c5 = corr[worst_lag_idx]
        if current_c5 == best_c5:
            break
        k = worst_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k) % n_points
            i_minus = (i - k) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]
        h = h - learning_rate * grad
