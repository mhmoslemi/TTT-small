import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Simulated Annealing parameters
    temp = 5.0
    cooling_rate = 0.995
    max_sa_iter = 300
    perturbations = 3
    start_time = time.time()

    for _ in range(max_sa_iter):
        if time.time() - start_time > budget_s:
            break
        candidates = []
        for _ in range(perturbations):
            h_p = h.copy()
            idx = np.random.randint(n_points)
            delta = np.random.uniform(-0.05, 0.05)
            h_p[idx] += delta
            h_p = h_p * target_sum / np.sum(h_p)
            h_p = np.clip(h_p, 0, 1)
            c5 = compute_c5(h_p)
            candidates.append((c5, h_p))
        candidates.sort()
        best_c, best_h_p = candidates[0]
        if best_c < best_c5:
            best_h = best_h_p.copy()
            best_c5 = best_c
        else:
            if np.random.rand() < np.exp((best_c5 - best_c) / temp):
                best_h = best_h_p.copy()
                best_c5 = best_c
        temp *= cooling_rate

    # Local refinement with L-BFGS-B
    def obj(x):
        h = x * target_sum / np.sum(x)
        h = np.clip(h, 0, 1)
        return compute_c5(h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
