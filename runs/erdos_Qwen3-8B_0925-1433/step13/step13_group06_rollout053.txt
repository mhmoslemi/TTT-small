        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)
        i_plus_k = (np.arange(n_points) + k_max) % n_points
        i_minus_k = (np.arange(n_points) - k_max) % n_points
        grad = dx * (1 - h[i_plus_k] - h[i_minus_k])
        momentum_prev = momentum * momentum_prev + grad
        h = h + learning_rate * momentum_prev
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate

    def obj_lbgfs(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
