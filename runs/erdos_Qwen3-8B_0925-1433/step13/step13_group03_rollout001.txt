import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()

    h = np.clip(h, 0, 1)

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        alpha = 1e-4
        temp = np.exp(alpha * corr)
        sum_exp = np.sum(temp)
        soft_max = np.log(sum_exp) / alpha
        return soft_max * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    h_best = np.clip(best_h, 0, 1)
    sum_h_best = np.sum(h_best)
    if sum_h_best > 0:
        h_best = h_best * target_sum / sum_h_best
    corr = np.correlate(h_best, 1 - h_best, mode='full')
    true_c5 = np.max(corr) * dx

    return (best_h, true_c5, n_points)
