import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 50
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.8:
            h = np.array(initial_h_values, dtype=float)
            h += np.random.normal(0, 0.1, h.shape)
        else:
            h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    max_gen = 100
    tourn_size = 5
    mut_rate = 0.2
    cross_rate = 0.8

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(corr.max() * dx)
        best_idx = np.argmin(fitness)
        curr_best = fitness[best_idx]
        if curr_best < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = curr_best

        parents = []
        for _ in range(pop_size // 2):
            tour = np.random.choice(pop_size, tourn_size, replace=False)
            winner = tour[np.argmin(fitness[tour])]
            parents.append(population[winner])

        offspring = []
        for i in range(0, len(parents), 2):
            if np.random.rand() < cross_rate:
                idx = np.random.randint(1, n_points)
                p1, p2 = parents[i], parents[i+1]
                c1 = np.concatenate([p1[:idx], p2[idx:]])
                c2 = np.concatenate([p2[:idx], p1[idx:]])
                offspring.extend([c1, c2])
            else:
                offspring.extend([parents[i], parents[i+1]])

        for j in range(len(offspring)):
            if np.random.rand() < mut_rate:
                mut = np.random.normal(0, 0.1, n_points)
                offspring[j] += mut
                offspring[j] = np.clip(offspring[j], 0, 1)
                offspring[j] = offspring[j] * target_sum / np.sum(offspring[j])

        population = offspring

    def obj_lbgfs(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
