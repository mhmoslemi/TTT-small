import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points) + 0.5

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)

    for step in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb_sum = np.sum(h_perturb)
            if h_perturb_sum != 0:
                h_perturb = h_perturb * target_sum / h_perturb_sum
            else:
                h_perturb = np.zeros_like(h_perturb) + 0.5
            c5_perturb = compute_c5(h_perturb)
            grad[i] = (c5_perturb - compute_c5(h)) / eps

        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros_like(h) + 0.5

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
