import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(x_clipped)
        if sum_h == 0:
            return float('inf')
        x_norm = x_clipped * target_sum / sum_h
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()
    
    # Global search with DE
    result = differential_evolution(
        obj, bounds, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True
    )
    
    # Local refinement with SLSQP
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result_slsqp = minimize(obj, result.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (result.x, result.fun, n_points)
    
    # Final fine-tuning with L-BFGS-B
    result_lbgfs = minimize(obj, result_slsqp.x, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if result_lbgfs.success and result_lbgfs.fun < result_slsqp.fun:
        result_slsqp = result_lbgfs
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    return (best_h, best_c5, n_points)
