    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def compute_gradient(h, k_max):
        n = len(h)
        grad = np.zeros(n)
        dx = 2.0 / n
        for i in range(n):
            j_plus = (i + k_max) % n
            j_minus = (i - k_max) % n
            grad[i] = dx * ( (1 - h[j_plus]) - h[j_minus] )
        return grad

    # Main optimization loop
    learning_rate = 0.2
    momentum = 0.9
    decay_rate = 0.999
    max_iterations = 2000
    start_time = time.time()
    velocity = np.zeros(n_points)

        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')

        velocity = momentum * velocity + grad
        # Update h
        h = h + learning_rate * velocity
        h = clip_and_normalize(h)
        # Check if improved
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        # Decay learning rate
        learning_rate *= decay_rate
