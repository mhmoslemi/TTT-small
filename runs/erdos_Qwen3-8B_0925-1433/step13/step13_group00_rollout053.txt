import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def calc_c5(h):
        h_clip = np.clip(h, 0, 1)
        if np.sum(h_clip) == 0:
            return np.inf
        h_norm = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr[1:-1]) * dx

    def perturb(h, std, momentum, prev_h=None):
        if prev_h is None:
            pert = np.random.normal(0, std, n_points)
        else:
            pert = (h - prev_h) * momentum + np.random.normal(0, std, n_points)
        new_h = np.clip(h + pert, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    def evaluate(h):
        return calc_c5(h)

    # Generate three initial guesses
    h1 = np.clip(initial_h_values.copy(), 0, 1)
    h1 = h1 * target_sum / np.sum(h1) if np.sum(h1) > 0 else np.zeros(n_points)

    h2 = np.zeros(n_points)
    ones = np.where(h1 > 0.5)[0]
    np.random.shuffle(ones)
    h2[ones[:len(ones) // 2]] = 1
    h2 = np.clip(h2, 0, 1)
    h2 = h2 * target_sum / np.sum(h2) if np.sum(h2) > 0 else np.zeros(n_points)

    h3 = np.flipud(initial_h_values)
    h3 = np.clip(h3, 0, 1)
    h3 = h3 * target_sum / np.sum(h3) if np.sum(h3) > 0 else np.zeros(n_points)

    candidates = [h1, h2, h3]
    best_h = candidates[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()

    for h_candidate in candidates:
        h = h_candidate
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            h = perturb(h, 0.02, 0.9)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
