import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def preprocess_h(h_array):
        h = np.clip(h_array, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        return h

    initial_guesses = [initial_h_values.copy()]
    for _ in range(2):
        h = np.random.permutation(initial_h_values)
        h = preprocess_h(h)
        initial_guesses.append(h)

    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for guess in initial_guesses:
        h_init = preprocess_h(guess)
        h = h_init.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.01 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = preprocess_h(new_h)
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                curr_c5 = res.fun
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = res.x.copy()
                h = res.x.copy()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = preprocess_h(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    return (best_h, best_c5, n_points)
