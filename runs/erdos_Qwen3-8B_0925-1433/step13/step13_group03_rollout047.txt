import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.clip(initial_h_values, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros_like(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()

    def objective(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Differential Evolution phase
    try:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=20, tol=1e-6,
            maxiter=100, mutation=(0.5, 1), recombination=0.8, seed=seed
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        pass

    # SLSQP refinement
    if time.time() - start_time < budget_s:
        def obj_sqp(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * target_sum / h_sum
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        res_sqp = minimize(obj_sqp, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        best_h = res_sqp.x
        best_c5 = res_sqp.fun

    return (best_h, best_c5, n_points)
