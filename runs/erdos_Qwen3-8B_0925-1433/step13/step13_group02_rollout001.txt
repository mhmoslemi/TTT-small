import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 20
    population = np.zeros((pop_size, n_points))
    population[0] = initial_h.copy()
    # Generate diverse initial candidates
    for i in range(1, pop_size):
        # Perturb initial
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        population[i] = candidate
        # Add random 1/0 pattern
        if i == 1:
            indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
            candidate = np.zeros(n_points)
            candidate[indices] = 1.0
            candidate = np.clip(candidate, 0, 1)
            if np.sum(candidate) > 0:
                candidate *= target_sum / np.sum(candidate)
            population[i] = candidate
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    F = 0.8
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + F * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) > 0:
                mutant *= target_sum / np.sum(mutant)
            new_population.append(mutant)
        new_population = np.array(new_population)
        for i in range(pop_size):
            trial = new_population[i]
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr) * dx
            current = population[i]
            corr_current = np.correlate(current, 1 - current, mode='full')
            c5_current = np.max(corr_current) * dx
            if c5_trial < c5_current:
                population[i] = trial
        # Update best solution
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        # Adaptive mutation
        if np.std(population, axis=0).mean() < 0.1:
            F = max(0.5, F - 0.1)
        else:
            F = min(1.2, F + 0.05)
    return (best_h, best_c5, n_points)
