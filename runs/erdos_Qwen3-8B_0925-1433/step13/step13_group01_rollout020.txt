            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points) + 0.5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')[1:-1]) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    return (best_h, best_c5, n_points)
