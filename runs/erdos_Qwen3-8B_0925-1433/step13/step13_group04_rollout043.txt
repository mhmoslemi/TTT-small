    velocities = [np.zeros(n) for _ in range(population_size)]
    pbest = particles.copy()
    pbest_fitness = np.full(population_size, np.inf)
    gbest = np.copy(h)
    gbest_fitness = compute_c5(h)
    
    start_time = time.time()

        for i in range(population_size):
