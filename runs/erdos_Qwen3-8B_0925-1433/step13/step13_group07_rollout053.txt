import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h, best_c5 = h.copy(), np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    T, cooling = 5.0, 0.995
    sa_iter = 0
    while time.time() - start_time < budget_s:
        sa_iter += 1
        idxs = np.random.randint(n_points, size=5)
        deltas = np.random.uniform(-0.05, 0.05, size=5)
        perturb = [h.copy() for _ in range(5)]
        for i, idx in enumerate(idxs):
            perturb[i][idx] += deltas[i]
        for p in perturb:
            p = p * target_sum / np.sum(p)
            p = np.clip(p, 0, 1)
        corr = [np.max(np.correlate(p, 1 - p, mode='full')) * (2.0 / n_points) for p in perturb]
        best_perturb = perturb[np.argmin(corr)]
        best_corr = np.min(corr)
        if best_corr < best_c5 or np.random.rand() < np.exp((best_c5 - best_corr) / T):
            h = best_perturb.copy()
            best_c5 = best_corr
        T *= cooling

    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8)
    if res.success:
        h, best_c5 = res.x, res.fun
    return np.clip(h, 0, 1), best_c5, n_points
