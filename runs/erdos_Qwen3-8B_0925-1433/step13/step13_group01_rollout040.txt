    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.ones(n_points) * 0.5

    def obj(x):
        x_ = np.clip(x, 0, 1)
        sum_x = np.sum(x_)
        if sum_x > 0:
            x_ = x_ * target_sum / sum_x
        corr = np.correlate(x_, 1 - x_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def update_h(h, target_sum):

        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.ones_like(h) * 0.5
        return h

        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = update_h(new_h, target_sum)
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
