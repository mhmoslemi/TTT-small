from scipy.optimize import minimize
import time

        variants.append(np.concatenate([h[:n_points//2], h[n_points//2:][::-1]]))
        # Inverted
        variants.append(1.0 - h)
        return variants

    variants = generate_variants(initial_h_values)
    best_h = np.zeros(n_points)
    best_c5 = np.inf

    def obj(x):

        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

        h = np.clip(v, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        best_h = h.copy()
        best_c5 = obj(h)

        # Phase 1: Global exploration with COBYLA

            if time.time() - start_time > budget_s:

            perturb_std = 0.01 * (0.95 ** it)

            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:

        # Phase 2: Refinement with SLSQP
