import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h
    best_c5 = np.inf

    def obj(h_):
        h_ = np.clip(h_, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ == 0:
            return np.inf
        h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    current_mean = best_h
    sigma = 0.1
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
