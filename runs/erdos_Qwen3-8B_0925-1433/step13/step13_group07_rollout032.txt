import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial population with perturbations
    pop_size = 50
    population = np.array([
        np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1)
        for _ in range(pop_size)
    ])
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Differential Evolution
    mutation_factor = 0.5
    crossover_prob = 0.7
    max_generations = 100
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + mutation_factor * (b - c)
            trial = np.copy(a)
            for i in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[i] = mutant[i]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            new_pop.append(trial)
        population = new_pop
        for candidate in population:
            current_c5 = compute_c5(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5

    # L-BFGS refinement
    candidates = population[:10]
    for candidate in candidates:
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            h = np.array(x, copy=True)
            h = h * target_sum / np.sum(h)
            return compute_c5(h)
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1)] * n_points
        result = minimize(
            objective,
            candidate,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h * target_sum / np.sum(current_h))
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
