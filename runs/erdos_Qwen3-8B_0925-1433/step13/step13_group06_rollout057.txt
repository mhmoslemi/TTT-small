import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        n = len(h)
        lag_val = k_max_index - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            i_plus = i + lag_val
            h_val1 = h[i_plus] if 0 <= i_plus < n else 0
            term1 = (1 - h_val1) * dx
            i_minus = i - lag_val
            h_val2 = h[i_minus] if 0 <= i_minus < n else 0
            term2 = -h_val2 * dx
            grad[i] = term1 + term2
        return -grad

    def obj_lbgfs(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        h = h + 0.1 * grad
        h = clip_and_normalize(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
