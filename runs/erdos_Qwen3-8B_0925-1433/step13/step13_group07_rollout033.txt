            return np.sum(x) - target_sum
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8, options={'maxiter': 100})
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            best_h = population[np.argmin([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population])]
            best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
