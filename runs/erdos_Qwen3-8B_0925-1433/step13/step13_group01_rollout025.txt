import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            return x * target_sum / sum_x
        return np.full_like(x, 0.5)

    h = rescale(initial_h_values.copy())

    def obj(x):
        h_ = rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_corr = np.max(corr[1:-1])
        return max_corr * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def optimize_phases(h_init, phases, max_iter=100):
        for phase, (method, pert_scale, tol, max_iter_phase) in enumerate(phases):
            if time.time() - start_time > budget_s:
                return best_h, best_c5
            for _ in range(max_iter_phase):
                if time.time() - start_time > budget_s:
                    return best_h, best_c5
                perturb = np.random.normal(0, pert_scale, n_points)
                new_h = rescale(h_init + perturb)
                res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
                if res.success and res.fun < best_c5:
                    best_h, best_c5 = res.x, res.fun
                h_init = res.x
        return best_h, best_c5

    phases = [
        ('COBYLA', 0.02, 1e-3, 150),
        ('L-BFGS-B', 0.005, 1e-4, 120),

    ]

    best_h, best_c5 = optimize_phases(h, phases)
