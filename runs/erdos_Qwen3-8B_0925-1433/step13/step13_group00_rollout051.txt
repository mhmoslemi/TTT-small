    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_initial_guess(base):
        perm = np.random.permutation(base)
        perm_clip = np.clip(perm, 0, 1)
        sum_p = np.sum(perm_clip)
        if sum_p > 0:
            perm_norm = perm_clip * target_sum / sum_p
        else:
            perm_norm = np.zeros(n_points)
        return perm_norm

    guess1 = make_initial_guess(initial_h_values)
    guess2 = np.concatenate([initial_h_values[:n_points // 2], initial_h_values[:n_points // 2][::-1]])
    guess2 = np.clip(guess2, 0, 1)
    sum_g = np.sum(guess2)
    if sum_g > 0:
        guess2 = guess2 * target_sum / sum_g
    else:
        guess2 = np.zeros(n_points)
    initial_guesses = [guess1, guess2]

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        def obj(x):

            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            corr = np.correlate(h_, 1 - h_, mode='full')
