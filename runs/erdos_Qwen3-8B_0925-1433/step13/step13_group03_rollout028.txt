import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    # Global phase: DE with limited iterations
    try:
        result_de = differential_evolution(obj, bounds, strategy='best1bin', maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        pass
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    # Refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    except:
        pass
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    # Final refinement with L-BFGS-B
    try:
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    except:
        pass
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)

    # Final normalization
    h_final = np.clip(best_h, 0, 1)
    sum_h_final = np.sum(h_final)
    if sum_h_final > 0:
        h_final = h_final * target_sum / sum_h_final
    best_c5 = obj(h_final)

    return (h_final, best_c5, n_points)
