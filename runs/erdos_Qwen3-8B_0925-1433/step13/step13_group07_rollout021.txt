import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    T = 5.0
    cooling_factor = 0.995
    sa_iterations = 500
    sa_start_time = time.time()

    for _ in range(sa_iterations):
        if time.time() - sa_start_time > budget_s:
            break
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = h + perturb
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        c5_perturbed = compute_c5(perturbed_h, dx)
        if c5_perturbed < best_c5:
            h = perturbed_h.copy()
            best_h = h.copy()
            best_c5 = c5_perturbed
        else:
            prob = np.exp((best_c5 - c5_perturbed) / T)
            if np.random.rand() < prob:
                h = perturbed_h.copy()
                best_h = h.copy()
                best_c5 = c5_perturbed
        T *= cooling_factor

    def obj(x):
        return compute_c5(x, dx)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h, dx)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
