import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Assume initial_h_values is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    # Validate initial_h_values
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    worst_lag = np.argmax(np.correlate(h, 1 - h, mode='full'))
    # Other variables
    learning_rate = 0.05
    epsilon = 1e-5
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Compute current worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        worst_lag = np.argmax(corr)
        current_c5 = corr[worst_lag] * dx
        if np.isclose(current_c5, best_c5):
            break
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Perturb h
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus *= target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_plus_worst = corr_plus[worst_lag]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus *= target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            corr_minus_worst = corr_minus[worst_lag]
            delta_plus = corr_plus_worst * dx
            delta_minus = corr_minus_worst * dx
            grad[i] = (delta_plus - delta_minus) / (2 * epsilon)
        # Update h
        h -= learning_rate * grad
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)
        # Update best_c5 if needed
        curr_corr = np.correlate(h, 1 - h, mode='full')
        curr_c5 = np.max(curr_corr) * dx
        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5
    return (best_h, best_c5, n_points)
