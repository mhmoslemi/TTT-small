import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.full(n_points, 0.5)
    
    def compute_corr(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        if corr.ndim == 1:
            max_corr = np.max(corr[1:-1])
        else:
            max_corr = np.max(corr[1:-1])
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = compute_corr(best_h)
    start_time = time.time()
    
    def update_h(h_val):
        h_val = np.clip(h_val, 0, 1)
        if np.sum(h_val) > 0:
            h_val = h_val * target_sum / np.sum(h_val)
        else:
            h_val = np.full_like(h_val, 0.5)
        return h_val
    
    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'L-BFGS-B'
            tol = 1e-6
            perturb_std = 0.02
        elif phase == 1:
            method = 'SLSQP'
            tol = 1e-8
            perturb_std = 0.005
        else:
            method = 'COBYLA'
            tol = 1e-8
            perturb_std = 0.001
        
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = update_h(new_h)
            
            def objective(x):
                h_ = np.clip(x, 0, 1)
                h_ = h_ * target_sum / np.sum(h_) if np.sum(h_) > 0 else np.full_like(h_, 0.5)
                return compute_corr(h_)
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0, 1)] * n_points
            
            res = minimize(objective, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
