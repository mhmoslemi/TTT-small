import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=np.float64)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    
    def obj_func(x):
        h_ = np.clip(x, 0, 1)
        h_sum = np.sum(h_)
        if h_sum > 0:
            h_ = h_ * target_sum / h_sum
        corr_vals = np.correlate(h_, 1 - h_, mode='full')
        alpha = 1e-3
        soft_max = np.log(np.sum(np.exp(alpha * corr_vals))) / alpha
        return soft_max * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    best_c5 = np.inf
    best_h = h.copy()
    
    start_time = time.time()
    
    for _ in range(5):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj_func, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_h = res.x
            corr_vals = np.correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(corr_vals) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    if time.time() - start_time < budget_s:
        for _ in range(10):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            if np.sum(new_h) > 0:
                new_h = new_h * target_sum / np.sum(new_h)
            res = minimize(obj_func, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                current_h = res.x
                corr_vals = np.correlate(current_h, 1 - current_h, mode='full')
                current_c5 = np.max(corr_vals) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
    
    return (best_h, best_c5, n_points)
