import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values).copy()

    sum_h = np.sum(h)
    if sum_h > 0:

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()

    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** i)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            scaling_factor = target_sum / sum_new
            new_h = np.clip(new_h * scaling_factor, 0, 1)
        else:
            new_h = np.zeros_like(new_h)
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8, options={'maxiter': 100})

    final_h = np.clip(best_h, 0, 1)
