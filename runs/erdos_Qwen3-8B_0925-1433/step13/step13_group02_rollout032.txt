import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 30
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        population.append(candidate)
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + 0.8 * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) > 0:
                mutant *= target_sum / np.sum(mutant)
            corr_mutant = np.correlate(mutant, 1 - mutant, mode='full')
            c5_mutant = np.max(corr_mutant) * dx
            corr_current = np.correlate(population[i], 1 - population[i], mode='full')
            c5_current = np.max(corr_current) * dx
            new_population.append(mutant if c5_mutant < c5_current else population[i])
        population = np.array(new_population)
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return (best_h, best_c5, n_points)
