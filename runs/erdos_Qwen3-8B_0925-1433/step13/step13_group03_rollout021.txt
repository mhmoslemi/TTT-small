import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Global search with differential evolution
    if time.time() - start_time < budget_s * 0.6:
        result = differential_evolution(
            obj, bounds, strategy='best1bin', popsize=20, mutation=(0.5, 1),
            recombination=0.9, tol=1e-5, maxiter=300,
            seed=seed, polish=False)
        best_h, best_c5 = result.x, result.fun

    # Local refinement with SLSQP
    if time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, tol=1e-8, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement with COBYLA
    if time.time() - start_time < budget_s:
        res = minimize(obj, best_h, method='COBYLA', bounds=bounds,
                       constraints=cons, tol=1e-8, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
