from scipy.optimize import minimize
import time

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    alpha = 1e-3
    def smooth_obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)

            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')

    for phase in range(3):

            tol = 1e-4
            maxiter = 100
        elif phase == 1:
            method = 'L-BFGS-B'
            tol = 1e-5
            maxiter = 50
        else:
            method = 'COBYLA'
            tol = 1e-6
