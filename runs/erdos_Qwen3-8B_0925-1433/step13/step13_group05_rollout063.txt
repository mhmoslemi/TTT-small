import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values, ensuring sum is correct
    h = initial_h_values.copy()
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    T = 1.0
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbation
        perturb = np.random.normal(0, T * 0.05, n_points)
        new_h = best_h + perturb
        # Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 0:
            # Scale to target_sum
            scale = target_sum / current_sum
            new_h = new_h * scale
            # Clip again to ensure within [0,1]
            new_h = np.clip(new_h, 0, 1)
        else:
            new_h = best_h.copy()

        # Compute new_c5
        new_c5 = evaluate_erdos_solution(new_h)

        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            best_c5 = new_c5

        T *= 0.95

    # Final check for constraints and evaluation
    # Clip again to ensure [0,1]
    best_h = np.clip(best_h, 0, 1)
    # Ensure sum is approximately target_sum
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    else:
        best_h = np.zeros(n_points)
    # Re-evaluate best_c5
    best_c5 = evaluate_erdos_solution(best_h)

    return (best_h, best_c5, n_points)
