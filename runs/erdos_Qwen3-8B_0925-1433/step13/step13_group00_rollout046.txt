import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values.copy()

    def clip_and_rescale(x):
        x_clip = np.clip(x, 0, 1)
        sum_clip = np.sum(x_clip)
        if sum_clip == 0:
            return np.zeros_like(x)
        return x_clip * target_sum / sum_clip

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    # DE parameters
    pop_size = 50
    gen_max = 100
    mut_factor = 0.5
    cross_rate = 0.8
    bounds = [(0, 1)] * n_points

    # Initialize population
    pop = [h_initial.copy() for _ in range(pop_size)]
    for p in pop:
        p += np.random.normal(0, 0.1, n_points)
        p = clip_and_rescale(p)
    best_h = min(pop, key=objective)
    best_c5 = objective(best_h)
    start_time = time.time()

    for gen in range(gen_max):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mut_factor * (pop[b] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < cross_rate:
                    trial[j] = mutant[j]
            trial = clip_and_rescale(trial)
            if objective(trial) < objective(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        current_best = min(pop, key=objective)
        if objective(current_best) < best_c5:
            best_h = current_best
            best_c5 = objective(best_h)

    # SQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
