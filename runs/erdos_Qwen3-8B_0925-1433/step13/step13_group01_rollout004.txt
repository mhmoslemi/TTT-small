        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1]))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def update_h(h, target_sum):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.ones_like(h) * 0.5
        return h

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        shift_k = (worst_idx - (n_points - 1)) * dx
        shift_idx = int(np.floor(shift_k / dx))
        overlap_end = n_points - 1 - shift_idx
        if overlap_end < 0:
            overlap_end = 0
        overlap_indices = np.arange(0, overlap_end + 1)
        alpha = 0.01
        for i in overlap_indices:
            if i + shift_idx < n_points:
                current_val = best_h[i]
                next_val = best_h[i + shift_idx]
                if current_val > 0:
                    best_h[i] = max(0, current_val - alpha * current_val)
                    best_h[i + shift_idx] = min(1, next_val + alpha * current_val)
        best_h = update_h(best_h, target_sum)

        if _ % 10 == 0:
            perturb_std = 0.05 * (0.9 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = update_h(new_h, target_sum)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
