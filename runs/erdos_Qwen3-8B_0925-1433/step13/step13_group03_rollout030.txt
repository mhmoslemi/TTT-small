from scipy.optimize import differential_evolution, minimize, Bounds
import time

def run(seed=42, budget_s=600):

    start_time = time.time()

    def objective_de(x):
        h_ = np.clip(x, 0, 1)
        h_ = h_ * target_sum / np.sum(h_) if np.sum(h_) > 0 else np.zeros_like(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
