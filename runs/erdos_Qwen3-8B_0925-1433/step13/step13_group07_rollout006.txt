        k_max = np.argmax(corr)

        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)

    learning_rate = 0.1

            trial_h = h + step * direction
