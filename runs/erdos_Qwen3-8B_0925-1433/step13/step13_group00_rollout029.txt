import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate variations
    h_variants = [initial_h_values.copy()]
    h_variants.append(initial_h_values[::-1].copy())
    h_uniform = np.zeros(n_points)
    h_uniform[::4] = 1.0
    h_uniform = h_uniform * target_sum / np.sum(h_uniform)
    h_variants.append(h_uniform)

    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    start_time = time.time()

    for variant in h_variants:

        if obj(h) < best_c5:
            best_h, best_c5 = h, obj(h)
