        h_proj = np.clip(h_proj, 0, 1)
        return np.max(np.correlate(h_proj, 1 - h_proj, mode='full')) * dx

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-8)
    best_h = res.x
    best_c5 = obj(best_h)
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
