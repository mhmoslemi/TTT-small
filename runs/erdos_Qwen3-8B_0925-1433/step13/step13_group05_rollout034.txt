    h = np.array(initial_h_values)
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / sum_h
    best_h = h.copy()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ == 0:
            return float('inf')
        h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_c5 = obj(best_h)

    start_time = time.time()
    phases = [
        (20, 0.5),
        (50, 0.1),
        (30, 0.05)
    ]
    for num_iter, std in phases:
        if time.time() - start_time > budget_s:
            break
        for _ in range(num_iter):
            perturb = np.random.normal(0, std, n_points)
            new_h = best_h + perturb
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new == 0:
                continue
            new_h = new_h * target_sum / sum_new
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun

    final_h = np.clip(best_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final == 0:
        final_c5 = float('inf')
    else:
        final_h = final_h * target_sum / sum_final
        corr = np.correlate(final_h, 1 - final_h, mode='full')
        final_c5 = np.max(corr) * dx

    if np.all((final_h >= 0) & (final_h <= 1)) and np.isclose(np.sum(final_h), target_sum):
        return (final_h, final_c5, n_points)
    else:
        return (best_h, best_c5, n_points)
