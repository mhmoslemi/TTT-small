    h = clip_and_normalize(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        grad = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = clip_and_normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]

    learning_rate = 0.05
