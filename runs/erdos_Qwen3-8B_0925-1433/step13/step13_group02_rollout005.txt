import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 50
    population = [initial_h_values.copy()]
    
    # Generate structured candidates
    for _ in range(pop_size - 1):
        phi = np.linspace(0, np.pi, n_points)
        h = 0.5 + 0.2 * np.sin(phi)
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)
        population.append(h)
    
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5s.append(np.max(corr) * dx)
        worst_idx = np.argmax(current_c5s)
        worst_c5 = current_c5s[worst_idx]
        
        # Identify worst lag
        corr = np.correlate(population[worst_idx], 1 - population[worst_idx], mode='full')
        worst_k = np.argmax(corr)
        worst_overlap = corr[worst_k]
        
        # Compute gradient
        epsilon = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = population[worst_idx].copy()
            h_perturb[i] += epsilon
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb *= target_sum / np.sum(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full') * dx
            grad[i] = (corr_perturb[worst_k] - worst_overlap) / epsilon
        
        # Gradient descent
        learning_rate = 0.1
        new_h = population[worst_idx] - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        new_h *= target_sum / np.sum(new_h)
        population[worst_idx] = new_h
        
        # Differential evolution
        new_population = []
        for idx in range(pop_size):
            if idx == worst_idx:

        for i in range(pop_size):
            trial = new_population[i]

            c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
