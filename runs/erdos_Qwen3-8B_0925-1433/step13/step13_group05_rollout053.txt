import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize with initial_h_values, rescale to meet sum constraint
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_sum = np.sum(h_)
        if h_sum > 0:
            h_ = h_ * target_sum / h_sum
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    # First phase: local search with L-BFGS-B and decreasing perturbations
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Second phase: more refined search with COBYLA and tighter constraints
    for _ in range(150):
        if time.time() - start_time > budget_s:

        perturb_std = 0.0005 * (0.9 ** _)

        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
