            p1, p2 = random.choice(parents), random.choice(parents)
            child = np.zeros(n_points, dtype=int)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = p1['h'][i]
                else:
                    child[i] = p2['h'][i]
            for i in range(n_points):
                if random.random() < 0.05:
                    child[i] ^= 1
            while np.sum(child) != target_sum:
                idx = np.random.randint(n_points)
                child[idx] ^= 1
            fit = -calc_c5(child)
            new_pop.append({'h': child, 'fitness': fit})
        pop = new_pop
    
    best = min(pop, key=lambda x: x['fitness'])
    best_h = best['h']

    x_vals = np.linspace(0, 2, n_points)
    h_cont = np.interp(x_vals, np.arange(n_points), best_h)
    sum_cont = np.sum(h_cont) * dx
    h_cont = h_cont * (1.0 / sum_cont) if sum_cont != 0 else h_cont
    h_cont = np.clip(h_cont, 0, 1)

    corr = np.correlate(h_cont, 1 - h_cont, mode='full')
    c5_bound = np.max(corr) * dx

    return (h_cont, c5_bound, n_points)
