import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_guesses():
        guesses = [initial_h_values.copy()]
        # Random uniform
        guess = np.random.rand(n_points)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)

        guess = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
        # Block pattern
        guess = np.zeros(n_points)
        guess[:n_points//3] = 1.0
        guess[n_points//3:2*n_points//3] = 0.5
        guess[2*n_points//3:] = 0.0
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
        # Gaussian thresholded
        guess = np.random.normal(0.5, 0.2, n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
        return guesses

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_guesses = generate_initial_guesses()

    best_c5 = obj(best_h)

    start_time = time.time()

    for i, guess in enumerate(initial_guesses):
