import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with diversity
    num_particles = 50
    population = []
    for _ in range(num_particles):
        if np.random.rand() < 0.3:
            h = np.copy(initial_h_values)
        elif np.random.rand() < 0.5:
            h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        population.append(h)
    velocities = [np.zeros_like(p) for p in population]
    pbest = np.copy(population)
    pbest_scores = [compute_c5(p) for p in population]
    gbest = np.copy(population[np.argmin(pbest_scores)])
    gbest_score = np.min(pbest_scores)

    time_start = time.time()
    for _ in range(100):
        if time.time() - time_start > budget_s:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (0.8 * velocities[i] + 
                             1.5 * r1 * (pbest[i] - population[i]) +
                             1.5 * r2 * (gbest - population[i]))
            population[i] = np.clip(population[i] + velocities[i], 0, 1)
            if np.sum(population[i]) > 0:
                population[i] = population[i] * target_sum / np.sum(population[i])
            score = compute_c5(population[i])
            if score < pbest_scores[i]:
                pbest[i] = np.copy(population[i])
                pbest_scores[i] = score
        idx = np.argmin(pbest_scores)
        if pbest_scores[idx] < gbest_score:
            gbest = np.copy(pbest[idx])
            gbest_score = pbest_scores[idx]

    # Frequency domain refinement
    def freq_filter(h):
        fft_h = np.fft.fft(h)
        cutoff = int(0.25 * len(h))
        fft_h[cutoff:] = 0
        filtered = np.fft.ifft(fft_h).real
        filtered = np.clip(filtered, 0, 1)
        if np.sum(filtered) > 0:
            filtered = filtered * target_sum / np.sum(filtered)
        return filtered

    best_h = freq_filter(gbest)
    best_c5 = compute_c5(best_h)

    # Local optimization
    def obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return compute_c5(x)

    bounds = [(0, 1)] * n_points

    return (best_h, final_c5, n_points)
