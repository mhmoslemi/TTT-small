def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        eps = 1e-4
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += eps
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= eps
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        return -grad

    def compute_gradient(x):
        return grad(x)

    def compute_objective(x):
        return obj(x)

    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 300
    start_time = time.time()

    for _ in range(max_iterations):

            break
        grad = compute_gradient(h)
        h_new = h - learning_rate * grad
        h_new = h_new * target_sum / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        corr_new = compute_objective(h_new)
        if corr_new < best_c5:

    def obj_lbgfs(x):
