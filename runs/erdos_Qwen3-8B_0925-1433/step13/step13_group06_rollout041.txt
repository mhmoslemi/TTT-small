from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        grad = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = clip_and_normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = clip_and_normalize(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-5)
        return -grad

    learning_rate = 0.5
    decay_rate = 0.998
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:

        grad = compute_gradient(h)
        direction = grad
        step = learning_rate
        for _ in range(20):
            trial_h = h + step * direction
