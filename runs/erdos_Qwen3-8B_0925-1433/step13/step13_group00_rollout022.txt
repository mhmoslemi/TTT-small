import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        else:
            factor = target_sum / np.sum(h)
            return h * factor

    def objective(h):
        h = clip_rescale(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(np.maximum(0, corr[1:-1])) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    # Differential Evolution Phase
    de_result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        tol=1e-8,
        maxiter=100,
        mutation=(0.5, 1.0),
        recombination=0.8,
        disp=False
    )
    best_h = de_result.x
    best_obj = de_result.fun

    # Remaining time for SQP
    if time.time() - start_time > budget_s * 0.75:
        return (best_h, best_obj, n_points)

    # Sequential Quadratic Programming Phase
    def sqp_objective(h):
        return objective(h)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    sqp_result = minimize(
        sqp_objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-9,
        options={'maxiter': 200}
    )

    if sqp_result.success:
        best_h = sqp_result.x
        best_obj = sqp_result.fun

    return (best_h, best_obj, n_points)
