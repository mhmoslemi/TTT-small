            overlap = np.dot(h, 1 - shifted) * dx
            overlaps.append(overlap)
        return np.argmax(overlaps)

    def get_gradient(h):
        k_max = compute_k_max(h)
        return compute_gradient(k_max)

    learning_rate = 0.2
    decay_rate = 0.998
    max_iterations = 2000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        k_max = compute_k_max(h)
        grad = get_gradient(h)
        h = h - learning_rate * grad
        h = clip_and_normalize(h)
        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate

    def obj_lbgfs(x):
        h = clip_and_normalize(x)
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted) * dx
            overlaps.append(overlap)
        return max(overlaps)

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
