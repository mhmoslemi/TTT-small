import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def generate_random_candidate():
        rand_vals = np.random.rand(n_points)
        rand_vals = np.round(rand_vals * 2) / 2
        sum_rand = np.sum(rand_vals)
        scale = target_sum / sum_rand
        return np.clip(rand_vals * scale, 0, 1)
    
    population = [initial_h_values.copy()]
    for _ in range(4):
        population.append(generate_random_candidate())
    population = np.array(population)
    
    pop_size = len(population)
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        F = 0.5 + 0.5 * (_ / 200)
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + F * (population[a] - population[b])
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial *= target_sum / np.sum(trial)
            new_population.append(trial)
        new_population = np.array(new_population)
        
        diversity = np.mean([np.sum((p - population[i])**2) for i, p in enumerate(new_population)])
        if diversity < 0.05:
            new_population = [generate_random_candidate() for _ in range(pop_size)]
        
        for i in range(pop_size):
            trial = new_population[i]
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr) * dx
