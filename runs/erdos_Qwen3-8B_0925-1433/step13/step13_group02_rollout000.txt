import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 20
    
    # Initialize population with initial solution and perturbations
    population = np.zeros((pop_size, n_points))
    population[0] = initial_h_values.copy()
    for i in range(1, pop_size):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        population[i] = candidate
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + 0.8 * (population[a] - population[b])
            trial = np.clip(mutant, 0, 1)
            if np.sum(trial) > 0:
                trial *= target_sum / np.sum(trial)
            trial_c5 = compute_c5(trial)
            current_c5 = compute_c5(population[i])
            if trial_c5 < current_c5:
                new_pop.append(trial)
            else:
                new_pop.append(population[i])
        population = np.array(new_pop)
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
