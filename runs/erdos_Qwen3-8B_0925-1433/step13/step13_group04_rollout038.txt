import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum_continuous = n_points / 2.0

    def evaluate_c5(h_binary):
        x = np.linspace(0, 2, n_points)
        h_cont = np.interp(x, np.arange(n_points), h_binary)
        sum_cont = np.sum(h_cont) * dx
        scale = target_sum_continuous / sum_cont
        h_cont *= scale
        corr = np.correlate(h_cont, 1 - h_cont, mode='full')
        return np.max(corr) * dx

    # Initial population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points) > 0.5
        target = int(np.round(target_sum_continuous))
        while np.sum(h) != target:
            h = np.random.rand(n_points) > 0.5
        population.append(h)

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        selected = np.random.choice(population, size=5, replace=False)
        parent1, parent2 = selected[:2]
        child = parent1.copy()
        crossover = np.random.rand(n_points) < 0.5
        child[crossover] = parent2[crossover]
        for i in range(n_points):
            if np.random.rand() < 0.05:
                child[i] ^= 1
        current_c5 = evaluate_c5(child)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = child
        pop_vals = [evaluate_c5(h) for h in population]
        idx = np.argmax(pop_vals)
        population[idx] = child

    x = np.linspace(0, 2, n_points)
    h_cont = np.interp(x, np.arange(n_points), best_h)
    sum_cont = np.sum(h_cont) * dx
    scale = target_sum_continuous / sum_cont
    h_cont *= scale
    h_cont = np.clip(h_cont, 0, 1)
    return (h_cont, best_c5, n_points)
