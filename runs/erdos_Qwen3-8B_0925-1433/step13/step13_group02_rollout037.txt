import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 20
    population = np.zeros((pop_size, n_points))
    population[0] = initial_h_values.copy()
    # Add some random perturbations
    for i in range(1, pop_size):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        population[i] = candidate
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        corr_list = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            corr_list.append(corr)
        # Find the worst candidate
        c5_list = [np.max(corr) * dx for corr in corr_list]
        worst_index = np.argmax(c5_list)
        worst_h = population[worst_index]
        worst_corr = corr_list[worst_index]
        # Find the max lag index
        max_lag_idx = np.argmax(worst_corr)
        # Compute shift
        shift = max_lag_idx - (n_points - 1)
        # Compute j and k
        j = (np.arange(n_points) + shift) % n_points
        k = (np.arange(n_points) - shift) % n_points
        # Compute gradient
        grad = (1 - worst_h[j]) - worst_h[k]
        grad *= dx
        # Perform gradient descent step
        alpha = 0.1  # learning rate
        new_h = worst_h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        new_h *= target_sum / np.sum(new_h)
        # Update the population's worst candidate
        population[worst_index] = new_h
        # Apply differential evolution to other candidates
        new_population = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + 0.8 * (population[a] - population[b])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial *= target_sum / np.sum(trial)
            new_population.append(trial)
        new_population = np.array(new_population)
        for i in range(pop_size):
            trial = new_population[i]
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr) * dx
            current = population[i]
            corr_current = np.correlate(current, 1 - current, mode='full')
            c5_current = np.max(corr_current) * dx
            if c5_trial < c5_current:
                population[i] = trial
        # Update best solution
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return (best_h, best_c5, n_points)
