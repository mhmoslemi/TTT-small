import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_guesses(h):
        guesses = []
        # Random permutation of non-zero entries
        perm = h.copy()
        non_zero = perm[perm > 0]
        np.random.shuffle(non_zero)
        perm[perm > 0] = non_zero
        guesses.append(perm)
        # Symmetric mirroring
        sym = h.copy()
        sym[:n_points//2] = h[n_points//2:]
        sym[n_points//2:] = h[:n_points//2]
        guesses.append(sym)
        # Binary pattern with uniform spacing
        h_bin = np.zeros(n_points)
        num_ones = int(np.round(target_sum))
        positions = np.random.choice(n_points, num_ones, replace=False)
        h_bin[positions] = 1
        guesses.append(h_bin)
        return guesses

    initial_guesses = generate_guesses(initial_h_values)

    best_h = initial_h_values.copy()
    best_c5 = np.inf
    start_time = time.time()

    for guess in initial_guesses:
        h = np.clip(guess, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr[1:-1] * dx)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        if time.time() - start_time > budget_s:
            break

    # Refinement phase with SLSQP
    def obj_ref(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr[1:-1] * dx)
    res_ref = minimize(obj_ref, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
    if res_ref.success and res_ref.fun < best_c5:
        best_c5 = res_ref.fun
        best_h = res_ref.x

    return (best_h, best_c5, n_points)
