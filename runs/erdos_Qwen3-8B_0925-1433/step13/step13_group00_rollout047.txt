import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    else:
        n_points = 45
        h_initial = np.linspace(0.5, 1.5, n_points) % 2.0
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h > 0:
            h_initial = h_initial * (n_points / 2.0) / sum_h
        else:
            h_initial = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_de(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_rescaled = h_clipped * target_sum / sum_h
        corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8
    )
    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)

    def objective_sqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_sqp = minimize(
        objective_sqp,
        best_h_de,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n_points,
        constraints=constraints,
        tol=1e-9,
        options={'maxiter': 200, 'ftol': 1e-9}
    )

    best_h = result_sqp.x
    best_c5 = result_sqp.fun
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, target_sum):
        best_h = best_h * target_sum / sum_h
        best_c5 = objective_sqp(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:

    return (best_h, best_c5, n_points)
