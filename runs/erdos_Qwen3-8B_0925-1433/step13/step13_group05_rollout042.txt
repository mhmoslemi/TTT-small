    def compute_c5_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.5 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(compute_c5_obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(30):
        if time.time() - start_time > budget_s:

        perturb_std = 0.1 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)

        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:

    for _ in range(50):

    return (best_h, best_c5, n_points)
