            h_new = h.copy() - grad * 0.1
            h_new = np.clip(h_new, 0, 1)
            h_new *= target_sum / np.sum(h_new)
            new_pop.append(h_new)
        population = new_pop
        new_pop = []
        for h in population:
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
            new_pop.append(h)
        population = new_pop
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return (best_h, best_c5, n_points)
