import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with rescaling
    h = np.array(initial_h_values, dtype=float)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * target_sum / h_sum
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / h_sum
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    for iteration in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** iteration)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum != 0:
            new_h = new_h * target_sum / new_h_sum
        else:
            new_h = np.zeros(n_points)
        new_h = np.clip(new_h, 0, 1)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    # Final check and cleanup
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if final_sum != 0:
        best_h = best_h * target_sum / final_sum
    best_h = np.clip(best_h, 0, 1)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
