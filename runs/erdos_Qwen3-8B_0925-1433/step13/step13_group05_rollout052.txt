import numpy as np
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_ = np.clip(x, 0.0, 1.0)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        return compute_c5(h_)

    h = np.clip(initial_h_values, 0.0, 1.0)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    try:
        res = basinhopping(objective, best_h, niter=100, T=1.0, stepsize=0.5,
                           minimizer_kwargs={"method": "L-BFGS-B", "bounds": bounds})
    except:
        res = None

    if res is None:
        final_h = best_h
        final_c5 = best_c5
    else:
        final_h = res.x
        final_c5 = res.fun

    h_final = np.clip(final_h, 0.0, 1.0)
    if np.sum(h_final) > 0:
        h_final = h_final * target_sum / np.sum(h_final)
    final_c5 = compute_c5(h_final)

    return (h_final, final_c5, n_points)
