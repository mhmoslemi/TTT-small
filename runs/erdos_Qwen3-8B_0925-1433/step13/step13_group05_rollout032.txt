import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize h with initial_h_values and scale to meet sum constraint
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = objective(best_h, dx)

    def objective(x, dx):
        x_clipped = np.clip(x, 0, 1)
        x_sum = np.sum(x_clipped)
        if x_sum == 0:
            return float('inf')
        x_scaled = x_clipped * target_sum / x_sum
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** i)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            continue
        new_h = new_h * target_sum / new_h_sum
        res = minimize(objective, new_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5

    # Final evaluation
    best_c5 = objective(best_h, dx)
    return (best_h, best_c5, n_points)
