import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)

    def clip_and_normalize(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x

    h = clip_and_normalize(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    momentum = 0.9
    learning_rate = 0.2
    decay_rate = 0.999
    velocity = np.zeros(n_points)
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        R = np.correlate(h, h, mode='full')
        min_index = np.argmin(R)
        lag = min_index - (n_points - 1)

        valid_indices = np.arange(n_points) + lag
        valid_mask = (valid_indices >= 0) & (valid_indices < n_points)
        grad = np.zeros(n_points)
        grad[valid_mask] = dx * h[valid_indices[valid_mask]]

        velocity = momentum * velocity + learning_rate * grad
        h += velocity
        h = clip_and_normalize(h)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        learning_rate *= decay_rate
        if iteration % 1000 == 0:
            learning_rate *= decay_rate

        if iteration > 1000 and abs(best_c5 - np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx) < 1e-6:
            break

    def objective(x):
        x = clip_and_normalize(x)
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
