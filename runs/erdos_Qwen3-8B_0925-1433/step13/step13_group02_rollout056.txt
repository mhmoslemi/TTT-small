import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population = [h]

        perturb = np.random.normal(0, 0.05, n_points)
        candidate = h + perturb

        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        population.append(candidate)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    F = 0.8
    patience = 0
    max_patience = 50
    while time.time() - start_time < budget_s:
        current_c5s = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in population]
        if min(current_c5s) < best_c5:
            best_c5 = min(current_c5s)
            best_h = population[current_c5s.index(min(current_c5s))]
        if np.std(current_c5s) < 1e-5:
            new_candidate = np.random.rand(n_points)
            new_candidate = np.clip(new_candidate, 0, 1)
            new_candidate *= target_sum / np.sum(new_candidate)
            population[0] = new_candidate
        if min(current_c5s) == best_c5:
            patience += 1
        else:
            patience = 0
        if patience >= max_patience:
            F *= 1.1
            patience = 0
        new_population = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
