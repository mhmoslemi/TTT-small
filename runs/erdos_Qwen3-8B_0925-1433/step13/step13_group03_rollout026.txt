import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return (1/0.05) * np.log(np.sum(np.exp(0.05 * corr))) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    # Initial SLSQP optimization
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Perturb and refine with L-BFGS-B
    for i in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.005 * (0.9 ** i)
        new_h = np.clip(best_h + np.random.normal(0, perturb_std, n_points), 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx < best_c5:
            best_h = res.x
            best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Final COBYLA refinement
    for i in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.9 ** i)
        new_h = np.clip(best_h + np.random.normal(0, perturb_std, n_points), 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx < best_c5:
            best_h = res.x
            best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, best_c5, n_points)
