from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):

        sum_h = np.sum(h_)

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
