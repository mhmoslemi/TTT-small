    guesses = [initial_h_values.copy()]
    for _ in range(2):
        perm = np.random.permutation(n_points)
        guess = initial_h_values[perm]
        guess = np.clip(guess, 0, 1)
        sum_g = np.sum(guess)
        if sum_g > 0:
            guess = guess * target_sum / sum_g
        guesses.append(guess)

    for guess in guesses:
        new_h = guess.copy()
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else new_h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
