        best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
