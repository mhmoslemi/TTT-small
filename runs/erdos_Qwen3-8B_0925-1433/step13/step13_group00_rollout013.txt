import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):

        sum_h_ = np.sum(h_)

        corr = np.correlate(h_, 1 - h_, mode='full')

    bounds = [(0, 1)] * n_points

        guesses.append(perm)
        mirrored = initial_h_values[::-1] if _ % 2 == 0 else initial_h_values
        guesses.append(mirrored)

    best_h = np.copy(initial_h_values)
    best_c5 = obj(best_h)

    # Global phase
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
