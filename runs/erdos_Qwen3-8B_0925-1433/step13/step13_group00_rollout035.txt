    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
