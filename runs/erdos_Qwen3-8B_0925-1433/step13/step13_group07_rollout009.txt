import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_vec):
        return compute_c5(h_vec)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_population = [np.array(initial_h_values, dtype=float) + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    for i in range(len(initial_population)):
        initial_population[i] = initial_population[i] * target_sum / np.sum(initial_population[i])
    best_h = np.array(initial_population[0])
    best_c5 = compute_c5(best_h)

    start = time.time()
    try:
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=50, mutation=(0.5, 1), 
                                       recombination=0.7, tol=1e-8, polish=True, maxiter=100, disp=False)
        best_h = result.x
        best_c5 = result.fun
    except:

    if time.time() - start > budget_s * 0.8:

        best_h = res.x

    best_h = np.clip(best_h, 0, 1)
