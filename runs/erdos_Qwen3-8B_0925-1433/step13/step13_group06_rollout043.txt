def run(seed=42, budget_s=600):

        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        displacement = k_max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):

            term1 = (1 - h[j]) * dx if 0 <= j < n_points else 0.0

            term2 = h[j2] * dx if 0 <= j2 < n_points else 0.0
            grad[i] = term1 - term2

            trial_h = h + step * direction
