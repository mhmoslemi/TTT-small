import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        eps = 1e-5
        grad = np.zeros_like(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = clip_and_normalize(x_plus)
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')[k_max]
            x_minus = x.copy()
            x_minus[i] -= eps
            x_minus = clip_and_normalize(x_minus)
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        return -grad

    def compute_bound(x):
        x = clip_and_normalize(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=grad, tol=1e-6, options={'maxiter': 200})
        if res.success:
            h = res.x
            current_c5 = compute_bound(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        else:
            h = clip_and_normalize(h)

        # Random perturbation to escape local minima
        h += np.random.uniform(-0.05, 0.05, size=n_points)
        h = clip_and_normalize(h)
        current_c5 = compute_bound(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    h = clip_and_normalize(best_h)
    return (np.clip(h, 0, 1), best_c5, n_points)
