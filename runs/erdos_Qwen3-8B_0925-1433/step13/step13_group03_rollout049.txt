from scipy.optimize import minimize
import time

    def obj(x):

        sum_h = np.sum(h_)
        if sum_h > 0:

        return smooth_max(corr)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
