import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 50
    gens = 100
    mut_rate = 0.2
    mut_scale = 0.1
    el_rate = 0.2
    t_size = 5

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    def init_pop():
        pop = []
        for _ in range(pop_size):
            if np.random.rand() < 0.8:
                h = np.array(initial_h_values, dtype=float)
                h += np.random.uniform(-0.1, 0.1, size=n_points)
            else:
                h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            pop.append(h)
        return pop

    pop = init_pop()
    best_h = pop[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(gens):
        if time.time() - start_time > budget_s:
            break
        fit = [compute_c5(h) for h in pop]
        sorted_idx = np.argsort(fit)
        sorted_pop = [pop[i] for i in sorted_idx]
        new_pop = sorted_pop[:int(pop_size * el_rate)]
        while len(new_pop) < pop_size:
            t = np.random.choice(range(pop_size), t_size)
            p1, p2 = sorted_pop[np.random.choice(t, 2, replace=False)]
            cp = np.random.randint(1, n_points)
            child = np.concatenate([p1[:cp], p2[cp:]])
            mutate = np.random.rand(n_points) < mut_rate
            child[mutate] += np.random.normal(0, mut_scale, size=np.sum(mutate))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
        if compute_c5(sorted_pop[0]) < best_c5:
            best_h, best_c5 = sorted_pop[0], compute_c5(sorted_pop[0])

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
