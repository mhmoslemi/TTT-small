import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guesses
    initial_guesses = [initial_h_values.copy()]
    for _ in range(2):
        perturbed = initial_h_values + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        if sum_p > 0:
            perturbed = perturbed * target_sum / sum_p
        initial_guesses.append(perturbed)
    shifted = np.roll(initial_h_values, 1)
    shifted = np.clip(shifted, 0, 1)
    sum_shift = np.sum(shifted)
    if sum_shift > 0:
        shifted = shifted * target_sum / sum_shift
    initial_guesses.append(shifted)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        h = guess.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_best = h.copy()
        current_c5 = compute_c5(h, dx)

        # Global exploration
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.01 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_best + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_best
            res = minimize(compute_c5_obj(new_h, dx, n_points, target_sum), new_h, method='COBYLA', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res.success and res.fun < current_c5:
                current_best, current_c5 = res.x, res.fun
        # Local refinement
        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.001 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_best + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_best
            res = minimize(compute_c5_obj(new_h, dx, n_points, target_sum), new_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res.success and res.fun < current_c5:
                current_best, current_c5 = res.x, res.fun
        if current_c5 < best_c5:
            best_h, best_c5 = current_best, current_c5

    return (best_h, best_c5, n_points)

def compute_c5_obj(h, dx, n_points, target_sum):
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
    return obj
