import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    def init_guesses(initial):
        h1 = initial.copy()
        ones_pos = np.where(h1 > 0)[0]
        np.random.shuffle(ones_pos)
        h1[ones_pos] = h1[ones_pos]
        h2 = initial.copy()[::-1]
        h3 = np.zeros(n_points)
        spacing = n_points / (np.sqrt(n_points) + 1)
        for i in range(n_points):
            if i % spacing < 1e-5:
                h3[i] = 1
        sum_h3 = np.sum(h3)
        if sum_h3 > 0:
            h3 = h3 * target_sum / sum_h3
        return [h1, h2, h3]

    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        guesses = init_guesses(h0)
    else:
        h0 = np.random.rand(n_points)
        h0 = h0 * target_sum / np.sum(h0)
        guesses = [h0]

    best_h = guesses[0].copy()
    best_c5 = obj(best_h)

    # Global phase
    prev_h = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.02, n_points)
        new_h = best_h + 0.9 * (best_h - prev_h) + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        val = np.log(np.sum(np.exp(10 * corr[1:-1]))) / 10 * dx
        if val < best_c5:
            best_h, best_c5 = new_h.copy(), val
        prev_h = best_h.copy()

    # Local phase
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    return (best_h, best_c5, n_points)
