import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    # Initialize population
    pop_size = 50
    population = np.array([np.random.rand(n_points) for _ in range(pop_size)])
    population = [p * target_sum / np.sum(p) for p in population]
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    # Differential Evolution
    start_time = time.time()
    max_gen = 100
    mut_factor = 0.5
    cross_prob = 0.7
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        mutant_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + mut_factor * (population[b] - population[c])
            mutant = mutant * target_sum / np.sum(mutant)
            mutant_pop.append(mutant)
        trial_pop = []
        for i in range(pop_size):
            target = population[i]
            mutant = mutant_pop[i]
            trial = np.array(target, dtype=float)
            for j in range(n_points):
                if np.random.rand() < cross_prob:
                    trial[j] = mutant[j]
            trial = trial * target_sum / np.sum(trial)
            trial_pop.append(trial)
        population = [p for p in population if np.max(p) < 1.001 and np.min(p) > -0.001]
        for trial in trial_pop:
            if np.max(trial) < 1.001 and np.min(trial) > -0.001:
                population.append(trial)
        population = population[:pop_size]
        population = [p * target_sum / np.sum(p) for p in population]
        for p in population:
            curr_c5 = objective(p)
            if curr_c5 < best_c5:
                best_h, best_c5 = p, curr_c5

    # L-BFGS refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint},
            method='L-BFGS-B',
            tol=1e-8,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
