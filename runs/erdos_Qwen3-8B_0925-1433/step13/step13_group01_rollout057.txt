from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        n_points = len(initial_h_values)
    except NameError:
        n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def init_guess():
        if 'initial_h_values' in locals():
            h = np.round(initial_h_values).astype(float)
        else:
            h = np.random.choice([0, 1], size=n_points)
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.full(n_points, 0.5)
        return h

    h = init_guess()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        else:
            x = np.full_like(x, 0.5)
        return x

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    def optimize(optimizer, method, iters, std_start, std_decay):
        for i in range(iters):
            if time.time() - start_time > budget_s:
                break
            std = std_start * (std_decay ** i)
            perturb = np.random.normal(0, std, n_points)
            new_h = clip_rescale(best_h + perturb)
            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:

    optimize(optimizer=minimize, method='COBYLA', iters=100, std_start=0.2, std_decay=0.8)
