import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initial solution
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = compute_c5(best_h)

    def compute_c5(h):
        h_ = np.clip(h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    # PSO parameters
    n_particles = 50
    particles = [np.random.normal(h.mean(), 0.1, n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0, 1) for p in particles]
    particles = [p * target_sum / np.sum(p) for p in particles] if any(np.sum(p) > 0 for p in particles) else particles

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            h_p = particles[i]
            h_p = np.clip(h_p, 0, 1)
            sum_p = np.sum(h_p)
            if sum_p > 0:
                h_p = h_p * target_sum / sum_p
            corr = np.correlate(h_p, 1 - h_p, mode='full')
            c5 = np.max(corr) * dx
            if c5 < compute_c5(best_h):
                best_h = h_p.copy()
                best_c5 = c5

    # Frequency filtering
    def freq_filter(h):
        fft_h = np.fft.fft(h)
        fft_h[np.abs(np.fft.fftfreq(n_points)) > 0.25] = 0
        h_filtered = np.fft.ifft(fft_h).real
        h_filtered = np.clip(h_filtered, 0, 1)
        sum_f = np.sum(h_filtered)
        if sum_f > 0:
            h_filtered = h_filtered * target_sum / sum_f
        return h_filtered

    best_h = freq_filter(best_h)

    # L-BFGS-B refinement
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    return (best_h, best_c5, n_points)
