import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 50

    def evaluate_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_binary():
        h = np.zeros(n_points, dtype=int)
        h[:int(target_sum)] = 1
        np.random.shuffle(h)
        return h

    # Initialize population
    population = [generate_binary() for _ in range(population_size)]

    best_h = np.copy(population[0])
    best_c5 = evaluate_C5(best_h)

    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate and select parents
        fitness = np.array([evaluate_C5(h) for h in population])
        parents = np.argsort(fitness)[:int(0.4 * population_size)]
        new_population = []
        for _ in range(population_size):
            p1, p2 = random.choices(parents, k=2)
            child = np.copy(population[p1])
            for i in range(n_points):
                if random.random() < 0.8:
                    child[i] = population[p2][i]
            # Mutation
            for i in range(n_points):
                if random.random() < 0.05:
                    child[i] ^= 1
            # Enforce sum
            while np.sum(child) != target_sum:
                if np.sum(child) < target_sum:
                    idx = np.random.choice(np.where(child == 0)[0])
                    child[idx] = 1
                else:
                    idx = np.random.choice(np.where(child == 1)[0])
                    child[idx] = 0
            new_population.append(child)
        population = new_population
        current_c5 = evaluate_C5(population[0])
        if current_c5 < best_c5:
            best_h, best_c5 = population[0], current_c5

    h_values = best_h.astype(float)
    return (h_values, best_c5, n_points)
