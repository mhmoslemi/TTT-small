from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    # Differential Evolution for global exploration
    def de_fitness(x):
        return obj(x)
    
    res_de = differential_evolution(
        de_fitness, bounds, strategy='best1bin', 
        popsize=20, mutation=(0.5, 1.0), recombination=0.8,
        tol=1e-6, maxiter=150, polish=True, 
        disp=False, callback=lambda x: None)
    
    if res_de.success:
        best_h, best_c5 = res_de.x, res_de.fun

    # Sequential Least Squares Programming refinement
    def sqp_fitness(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    res_sqp = minimize(
        sqp_fitness, best_h, method='SLSQP', bounds=bounds, 
        constraints=cons, tol=1e-8, 
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
    
    if res_sqp.success and res_sqp.fun < best_c5:
        best_h, best_c5 = res_sqp.x, res_sqp.fun

    # Final refinement with adaptive perturbations
