import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_plus = clip_and_normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            h_minus = clip_and_normalize(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad  # Negative for descent direction

    learning_rate = 0.5
    decay_rate = 0.9995
    max_iterations = 3000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        direction = grad
        step = learning_rate
        for _ in range(15):
            trial_h = h + step * direction
            trial_h = clip_and_normalize(trial_h)
            corr_trial = np.max(np.correlate(trial_h, 1 - trial_h, mode='full')) * dx
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * direction
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate
        if _ % 100 == 0:
            h = clip_and_normalize(h + 0.01 * np.random.randn(n_points))
            h = clip_and_normalize(h)
            h = clip_and_normalize(h * target_sum / np.sum(h))

    def obj_lbgfs(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
