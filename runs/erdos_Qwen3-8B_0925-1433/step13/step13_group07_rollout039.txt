    def objective(x):
        h = x * target_sum / np.sum(x)
        h = np.clip(h, 0, 1)

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        eps = 1e-5
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += eps
            h_plus = h_plus * target_sum / np.sum(h_plus)

            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')

    def constraint(x):
