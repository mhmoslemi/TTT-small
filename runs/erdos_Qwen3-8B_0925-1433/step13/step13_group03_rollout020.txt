import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = 0.0
    start_time = time.time()

    def compute_c5(h_vec):
        h_ = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    # Global phase with DE
    de_start = time.time()
    def objective(x):
        return compute_c5(x)
    try:
        de_res = differential_evolution(objective, bounds, strategy='best1bin',
                                       popsize=20, maxiter=200, tol=1e-6,
                                       mutation=(0.5, 1), recombination=0.8)
        best_h, best_c5 = de_res.x, de_res.fun
    except:
        pass
    de_end = time.time()

    # Refinement with SLSQP
    if (de_end - start_time) < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    # Final refinement
    if (time.time() - start_time) < budget_s:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    return (best_h, best_c5, n_points)
