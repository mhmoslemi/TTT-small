            mutant = population[c] + 0.8 * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            mutant *= target_sum / np.sum(mutant)
            corr_mutant = np.max(np.correlate(mutant, 1 - mutant, mode='full')) * dx
            if corr_mutant < corr[i]:
                population[i] = mutant
        for h in population:
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
    return (best_h, best_c5, n_points)
