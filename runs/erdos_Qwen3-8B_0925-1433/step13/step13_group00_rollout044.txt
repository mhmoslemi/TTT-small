import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    def obj(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    guesses = [
        initial_h_values.copy(),
        np.flip(initial_h_values),
        np.zeros(n_points),
        np.random.rand(n_points),
    ]
    guesses = [clip_rescale(g) for g in guesses]
    guesses.append(np.zeros(n_points) + 0.5)

    best_h = guesses[0].copy()
    best_c5 = obj(best_h)

    for guess in guesses:
        h = guess.copy()
        for _ in range(100):
            perturb = np.random.normal(0, 0.02, n_points)
            h_new = h + perturb
            h_new = clip_rescale(h_new)
            c5 = obj(h_new)
            if c5 < best_c5:

                h = h_new
