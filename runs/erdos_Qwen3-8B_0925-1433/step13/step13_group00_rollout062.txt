            bounds=bounds_sqp,
            constraints=cons,
            tol=1e-9,
            options={'maxiter': 200}
        )
        
        if res_sqp.success:
            best_h = res_sqp.x
            best_c5 = res_sqp.fun
        else:
            best_h = best_de_h
            best_c5 = result_de.fun
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr[1:-1]) * dx
    
    return (best_h, c5_bound, n_points)
