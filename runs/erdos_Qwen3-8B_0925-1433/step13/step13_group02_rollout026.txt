    initial_h = clip_rescale(initial_h)

    population = [initial_h]
    for _ in range(19):
        struct = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                struct[i] = 0.5 + 0.2 * np.sin(np.pi * i / n_points)
            else:
                struct[i] = 0.5 - 0.2 * np.sin(np.pi * i / n_points)
        struct = clip_rescale(struct)
        population.append(struct)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_population = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_k = np.argmax(corr)
            grad = np.zeros_like(h)
            epsilon = 0.0005
            for i in range(n_points):
                h_upper = h.copy()
                h_upper[i] += epsilon
                h_upper = clip_rescale(h_upper)
                val_upper = np.correlate(h_upper, 1 - h_upper, mode='full')[max_k]
                h_lower = h.copy()
                h_lower[i] -= epsilon
                h_lower = clip_rescale(h_lower)
                val_lower = np.correlate(h_lower, 1 - h_lower, mode='full')[max_k]
                grad[i] = (val_upper - val_lower) / (2 * epsilon)
            learning_rate = 0.05
            h_new = h - learning_rate * grad
            h_new = clip_rescale(h_new)
            new_corr = np.correlate(h_new, 1 - h_new, mode='full')
            new_max = np.max(new_corr)
            if new_max < np.max(corr):
                h = h_new
            new_population.append(h)

        population = new_population
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

        if np.random.rand() < 0.2:
            perturb = np.random.normal(0, 0.03, n_points)
            new_h = initial_h + perturb
            new_h = clip_rescale(new_h)
            population.append(new_h)
            population.pop(np.random.randint(len(population)))

    return (best_h, best_c5, n_points)
