import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    # Phase 1: Global search with DE
    if time.time() - start_time < budget_s * 0.4:
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Phase 2: Local refinement with SLSQP
    if time.time() - start_time < budget_s * 0.7:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 150})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Phase 3: Final tuning with COBYLA
    if time.time() - start_time < budget_s:
        res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Final normalization
    final_h = np.clip(best_h, 0, 1)
    if np.sum(final_h) > 0:
        final_h = final_h * target_sum / np.sum(final_h)
    else:
        final_h = best_h
    
    # Re-evaluate
    final_c5 = obj(final_h)
    return (final_h, final_c5, n_points)
