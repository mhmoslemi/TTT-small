    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        worst_lag_idx = np.argmax(corr)
        worst_c5 = corr[worst_lag_idx] * dx
        if worst_c5 < best_c5:
            best_c5 = worst_c5
            best_h = h.copy()
        grad = np.zeros(n_points)
        epsilon = 1e-3
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            overlap_plus = corr_perturbed[worst_lag_idx] * dx
            h_perturbed = h.copy()
            h_perturbed[i] -= epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            overlap_minus = corr_perturbed[worst_lag_idx] * dx
            grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        learning_rate = 0.1
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if np.random.rand() < 0.1:
            mut = np.random.normal(0, 0.05, n_points)
            h += mut
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
