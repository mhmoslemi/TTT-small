    start_time = time.time()
    generations = 0
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    while generations < 100 and (time.time() - start_time) < budget_s:
        fitness = [evaluate(h) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population
        generations += 1
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
