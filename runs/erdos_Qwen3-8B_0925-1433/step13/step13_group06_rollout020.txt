        k = k_max
        indices_plus = (j + k) % n_points
        indices_minus = (j - k) % n_points
        gradient = dx * ( (1 - h[indices_plus]) - h[indices_minus] )
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    def obj_lbgfs(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
