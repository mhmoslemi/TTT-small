import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial guess
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    # Generate permutation
    h_perm = np.random.permutation(initial_h_values)
    h_perm = np.clip(h_perm, 0, 1)
    sum_h = np.sum(h_perm)
    if sum_h > 0:
        h_perm = h_perm * target_sum / sum_h
    best_c5 = np.min([best_c5, np.max(np.correlate(h_perm, 1 - h_perm, mode='full')) * dx])
    best_h = h_perm.copy()
    
    # Generate mirrored
    h_mirror = initial_h_values[::-1]
    h_mirror = np.clip(h_mirror, 0, 1)
    sum_h = np.sum(h_mirror)
    if sum_h > 0:
        h_mirror = h_mirror * target_sum / sum_h
    best_c5 = np.min([best_c5, np.max(np.correlate(h_mirror, 1 - h_mirror, mode='full')) * dx])
    best_h = h_mirror.copy()
    
    start_time = time.time()
    
    # Phase 1: Global exploration with momentum
    prev_h = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = best_h + 0.9 * (best_h - prev_h) + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        obj_val = np.max(corr) * dx
        if obj_val < best_c5:
            best_h, best_c5 = new_h, obj_val
        prev_h = best_h.copy()
    
    # Phase 2: Local refinement with SLSQP
