import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        return h * target_sum / s if s != 0 else h

    def mutate(h, scale=0.1):
        h = h.copy()
        for i in range(n_points):
            h[i] += np.random.randn() * scale
            h[i] = np.clip(h[i], 0, 1)
        return normalize(h)

    def crossover(h1, h2):
        split = np.random.randint(1, n_points - 1)
        return np.concatenate([h1[:split], h2[split:]])

    def init_population(size):
        pop = []
        for _ in range(size):
            if np.random.rand() < 0.8:
                base = np.array(initial_h_values, dtype=float)
                base = base * np.random.uniform(0.9, 1.1)
                base = normalize(base)
            else:
                base = np.random.rand(n_points)
                base = normalize(base)
            pop.append(base)
        return np.array(pop)

    pop = init_population(50)
    best = np.argmin([evaluate(p) for p in pop])
    best_h = pop[best]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        elites = pop[:int(len(pop)*0.2)]
        candidates = []
        for _ in range(len(pop) - len(elites)):
            p1, p2 = np.random.choice(pop, 2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            candidates.append(child)
        pop = np.vstack([elites, candidates])
        fitness = [evaluate(p) for p in pop]
        idx = np.argsort(fitness)
        pop = pop[idx[:len(pop)]]
        current_best = pop[0]
        if evaluate(current_best) < best_c5:
            best_h, best_c5 = current_best, evaluate(current_best)

    def obj_lbgfs(x):
        return evaluate(x)

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
