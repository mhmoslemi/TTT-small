import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return np.inf
        x_scaled = x_clipped * target_sum / sum_x
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Differential Evolution for global search
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    best_h = result_de.x
    best_c5 = objective(best_h)
    
    # SQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_sqp = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 100}
    )
    if res_sqp.success and res_sqp.fun < best_c5:
        best_h, best_c5 = res_sqp.x, res_sqp.fun
    
    # Final refinement with L-BFGS-B
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 100:
        res_lbgfgs = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 50}
        )
        if res_lbgfgs.success and res_lbgfgs.fun < best_c5:
            best_h, best_c5 = res_lbgfgs.x, res_lbgfgs.fun
    
    # Final constraint check
    h_final = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_final)
    if sum_h > 0:
        h_final = h_final * target_sum / sum_h
    else:
        h_final = np.zeros(n_points)
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr) * dx
    return (h_final, final_c5, n_points)
