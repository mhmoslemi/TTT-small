import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Simulated Annealing
    T = 5.0
    cooling_rate = 0.995
    sa_iterations = 1000
    num_perturbations = 5
    perturbation = 0.05
    sa_start_time = time.time()

    for _ in range(sa_iterations):
        if time.time() - sa_start_time > budget_s:
            break
        current_h = best_h.copy()
        best_perturb = None
        best_perturb_c5 = np.inf
        for _ in range(num_perturbations):
            delta = np.random.uniform(-perturbation, perturbation, n_points)
            new_h = current_h + delta
            new_h = new_h * target_sum / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_perturb_c5:
                best_perturb_c5 = new_c5
                best_perturb = new_h.copy()
        if best_perturb_c5 < best_c5:
            best_c5 = best_perturb_c5
            best_h = best_perturb.copy()
        else:
            prob = np.exp((best_c5 - best_perturb_c5) / T)
            if np.random.rand() < prob:
                best_c5 = best_perturb_c5
                best_h = best_perturb.copy()
        T *= cooling_rate

    # Local Refinement
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=constraints, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
