import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    
    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x > 0:
            x_scaled = x_clipped * target_sum / sum_x
        else:
            x_scaled = x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx

    guesses = [h.copy()]
    guesses.append(np.random.rand(n_points) * 2 - 1)
    guesses.append(np.sin(np.pi * np.arange(n_points) / n_points) > 0)
    guesses.append(np.zeros(n_points))
    guesses.append(np.random.normal(0.5, 0.2, n_points))

    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()

    for guess in guesses[:4]:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, bounds=[(0,1)]*n_points, 
                                    popsize=15, strategy='best1bin', maxiter=50, tol=1e-6)
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='Nelder-Mead', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-8)
    if result.success and result.fun < best_c5:
        best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
