import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(45, 0.5)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(x):
        x = x * target_sum / np.sum(x)
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Simulated Annealing
    T = 5.0
    cooling = 0.995
    iter_max = 1000
    start = time.time()
    for _ in range(iter_max):
        if time.time() - start > budget_s:
            break
        pert = np.random.uniform(-0.05, 0.05, n_points)
        pert_h = h + pert
        pert_h = pert_h * target_sum / np.sum(pert_h)
        pert_h = np.clip(pert_h, 0, 1)
        c5 = compute_c5(pert_h)
        if c5 < best_c5 or np.random.rand() < np.exp((c5 - best_c5) / T):
            h = pert_h
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
        T *= cooling

    # Local refinement
    def objective(x):
        return compute_c5(x)
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-8)
    if res.success:
        best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
