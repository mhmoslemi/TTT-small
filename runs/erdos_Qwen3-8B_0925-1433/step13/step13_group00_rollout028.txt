import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guesses
    def gen_guesses(h_ref):
        perms = np.random.permutation(h_ref)
        mirr = h_ref[::-1]
        unif = np.zeros(n_points)
        unif[::n_points//2] = 1
        return [perms, mirr, unif]

    guesses = gen_guesses(initial_h_values)
    best_h = np.zeros(n_points)
    best_c5 = 1.0

    def eval_obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    def surr_obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        alpha = 10.0
        return np.log(np.sum(np.exp(alpha * corr[1:-1]))) / alpha * dx

    start_time = time.time()
    for guess in guesses:
        guess = np.clip(guess, 0, 1)
        if np.sum(guess) > 0:
            guess = guess * target_sum / np.sum(guess)
        curr_h = guess.copy()
        curr_c5 = eval_obj(curr_h)

        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(curr_h + perturb, 0, 1)
            if np.sum(new_h) > 0:
                new_h = new_h * target_sum / np.sum(new_h)
            new_c5 = surr_obj(new_h)
            if new_c5 < curr_c5:
                curr_h, curr_c5 = new_h, new_c5

        if curr_c5 < best_c5:
            best_h, best_c5 = curr_h, curr_c5

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)

            new_h = new_h * target_sum / np.sum(new_h)

        if res.success and res.fun < best_c5:
