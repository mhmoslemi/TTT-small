from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h):
        R = np.correlate(h, h, mode='full') * dx
        k_min = np.argmin(R)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            R_plus = np.correlate(h_plus, h_plus, mode='full') * dx
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            R_minus = np.correlate(h_minus, h_minus, mode='full') * dx
            grad[i] = (R_plus[k_min] - R_minus[k_min]) / (2e-4)
        return grad

    def obj(x):
        h = clip_and_normalize(x)
        R = np.correlate(h, h, mode='full') * dx
        return -np.min(R)

    learning_rate = 0.2
    decay_rate = 0.998
    max_iterations = 2000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        h = h + learning_rate * grad
        h = clip_and_normalize(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
