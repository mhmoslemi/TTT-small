import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h_ = np.clip(x, 0, 1)
        s = np.sum(h_)
        if s > 0:
            h_ = h_ * target_sum / s
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_guesses = [initial_h_values.copy()]
    for _ in range(2):
        temp = np.random.permutation(initial_h_values)
        temp = np.clip(temp, 0, 1)
        temp = temp * target_sum / np.sum(temp) if np.sum(temp) > 0 else np.zeros(n_points)
        initial_guesses.append(temp)

    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.01 * (0.95 ** _), n_points)
            new_h = np.clip(guess + perturb, 0, 1)
            s = np.sum(new_h)
            if s > 0:
                new_h = new_h * target_sum / s
            else:
                new_h = guess
            res = minimize(objective, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.0005 * (0.95 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        s = np.sum(new_h)
        if s > 0:
            new_h = new_h * target_sum / s
        else:
            new_h = best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
