from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'COBYLA'
            max_iter = 100
            perturb_std = 0.2 * (0.8 ** phase)
            tol = 1e-6
        elif phase == 1:
            method = 'L-BFGS-B'
            max_iter = 100
            perturb_std = 0.05 * (0.9 ** phase)
            tol = 1e-5
        else:
            method = 'SLSQP'
            max_iter = 50
            perturb_std = 0.001 * (0.9 ** phase)
            tol = 1e-8

        for _ in range(max_iter):
            if time.time() - start_time > budget_s:

            perturb = np.random.normal(0, perturb_std, n_points)
