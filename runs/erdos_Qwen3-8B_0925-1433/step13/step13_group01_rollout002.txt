import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        if n_points % 2:
            h[n_points//2] = 0.5
        h = h * target_sum / np.sum(h)

    def obj(h_):
        h_ = np.clip(h_, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr[1:-1]) * dx

    def compute_grad(h_):
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_idx = np.argmax(corr[1:-1])
        k_max = max_idx  # maps to corr[1:-1] indices
        grad = np.zeros_like(h_)
        for i in range(len(h_)):
            if i + k_max < len(h_):
                grad[i] += (1 - h_[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h_[i - k_max]
        return grad

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for t in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad = compute_grad(best_h)
        step_size = 0.1 * (0.95)**t
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new

        new_c5 = obj(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

    return (best_h, best_c5, n_points)
