import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = compute_c5(h)

    def freq_filter(h, cutoff=0.2):
        fft_h = np.fft.fft(h)
        freqs = np.fft.fftfreq(n_points)
        fft_h = np.where(np.abs(freqs) < cutoff, fft_h, 0)
        return np.clip(np.fft.ifft(fft_h).real * target_sum / np.sum(np.fft.ifft(fft_h).real), 0, 1)

    def apply_freq(h):
        return freq_filter(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        return compute_c5(h_)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)

        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons)
