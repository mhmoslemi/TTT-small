import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    # Generate initial guesses
    initial_guesses = [initial_h_values.copy()]
    # Evenly spaced
    period = n_points // 5
    even = np.zeros(n_points)
    even[::period] = 1
    even_sum = np.sum(even)
    if even_sum > 0:
        even = even * target_sum / even_sum
    initial_guesses.append(even)
    # Sinusoidal
    t = np.linspace(0, 2, n_points)
    sin_h = np.where(0.5 + 0.5 * np.sin(np.pi * t) > 0.5, 1, 0)
    sin_sum = np.sum(sin_h)
    if sin_sum > 0:
        sin_h = sin_h * target_sum / sin_sum
    initial_guesses.append(sin_h)
    # Random binary
    rand = np.random.rand(n_points)
    rand_h = np.where(rand > 0.5, 1, 0)
    rand_sum = np.sum(rand_h)
    if rand_sum > 0:
        rand_h = rand_h * target_sum / rand_sum
    initial_guesses.append(rand_h)
    # Block pattern
    block = np.zeros(n_points)
    block[::(n_points // 3)] = 1
    block_sum = np.sum(block)
    if block_sum > 0:
        block = block * target_sum / block_sum
    initial_guesses.append(block)
    
    start_time = time.time()
    best_h = initial_guesses[0].copy()
    best_c5 = obj(best_h)
    
    for guess in initial_guesses[1:]:
        try:
            res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h = res.x
                    best_c5 = current_c5
        except:
            pass
    
    return (best_h, best_c5, n_points)
