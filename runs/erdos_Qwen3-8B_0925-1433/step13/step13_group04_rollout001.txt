import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        h_scaled = h_clipped * (target_sum / sum_h) if sum_h > 0 else np.zeros_like(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Generate initial candidates
    initial_candidates = [
        initial_h_values.copy(),
        np.random.rand(n_points) * 2 - 1,
        np.sin(np.pi * np.arange(n_points) / n_points) > 0,
        np.tile([1, 0, 0.5], n_points // 3) if n_points % 3 == 0 else np.tile([1, 0, 0.5], n_points // 3 + 1),
        np.random.normal(0.5, 0.2, n_points)
    ]
    for i in range(len(initial_candidates)):
        initial_candidates[i] = np.clip(initial_candidates[i], 0, 1)
        initial_candidates[i] = initial_candidates[i] * target_sum / np.sum(initial_candidates[i])

    best_h = initial_candidates[0]
    best_c5 = compute_c5(best_h)

    time_used = 0
    start = time.time()

    # Phase 1: Differential Evolution
    from scipy.optimize import Bounds
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    for candidate in initial_candidates:
        if time_used >= budget_s - 10:
            break
        try:
            result = differential_evolution(
                compute_c5,
                bounds,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.9,
                tol=1e-6,
                maxiter=100,
                polish=True,
                x0=candidate
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_h = result.x.copy()
                    best_c5 = current_c5
                time_used = time.time() - start
        except:
            pass

    # Phase 2: SLSQP Refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    for _ in range(5):
        if time_used >= budget_s:
            break
        res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
        time_used = time.time() - start

    # Phase 3: Hybrid Optimization
    for _ in range(5):
        if time_used >= budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)

        new_h = new_h * target_sum / np.sum(new_h)
        for method in ['Nelder-Mead', 'L-BFGS-B']:
            res = minimize(compute_c5, new_h, method=method, bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun
        time_used = time.time() - start

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, compute_c5(best_h), n_points)
