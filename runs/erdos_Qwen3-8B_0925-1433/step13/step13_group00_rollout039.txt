import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    sum_h_initial = np.sum(h_initial)
    if sum_h_initial > 0:
        h_initial = h_initial * target_sum / sum_h_initial
    else:
        h_initial = np.zeros(n_points)

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h_initial
    best_c5 = objective(best_h)

    # Differential Evolution
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            objective, bounds,
            strategy='best1bin', popsize=20, mutation=(0.5, 1),
            recombination=0.8, tol=1e-6, maxiter=1, disp=False
        )
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    # Sequential Quadratic Programming
    def obj_sqp(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj_sqp, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          tol=1e-9, options={'maxiter': 1})
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
