import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    # Initial COBYLA phase for rapid convergence
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Differential evolution for global search
    if time.time() - start_time < budget_s * 0.6:
        result = differential_evolution(
            obj, bounds, constraints=cons, 
            popsize=25, tol=1e-6, mutation=(0.5, 1.0), 
            strategy='best1bin', maxiter=100, 
            recombination=0.8, polish=True, 
            seed=seed
        )
        if result.success:
            best_h, best_c5 = result.x, result.fun
    
    # Final refinement with L-BFGS-B
    if time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
