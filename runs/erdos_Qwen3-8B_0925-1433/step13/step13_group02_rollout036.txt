        worst_lag = np.argmax(worst_corr)

        # Gradient descent on worst lag
        eps = 1e-5
        grad = np.zeros_like(worst_h)
        for i in range(n_points):
            h_perturbed = worst_h.copy()
            h_perturbed[i] += eps
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed *= target_sum / np.sum(h_perturbed)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_perturbed[worst_lag] - worst_corr[worst_lag]) / eps

        # Apply gradient descent
        learning_rate = 0.1
        for _ in range(5):
            worst_h -= learning_rate * grad
            worst_h = np.clip(worst_h, 0, 1)
            worst_h *= target_sum / np.sum(worst_h)

        # Update worst candidate
        population[worst_idx] = worst_h
        c5_values[worst_idx] = np.max(np.correlate(worst_h, 1 - worst_h, mode='full')) * dx

        # Differential evolution for other candidates
        for i in range(len(population)):
            if i == worst_idx:
                continue
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[c] + 0.8 * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            mutant *= target_sum / np.sum(mutant)
            corr_mutant = np.correlate(mutant, 1 - mutant, mode='full')
            c5_mutant = np.max(corr_mutant) * dx
            if c5_mutant < c5_values[i]:
                population[i] = mutant
                c5_values[i] = c5_mutant

        # Update best solution
        for i in range(len(population)):
            corr = np.correlate(population[i], 1 - population[i], mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = population[i].copy()
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
