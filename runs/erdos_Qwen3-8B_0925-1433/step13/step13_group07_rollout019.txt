import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def normalize(h):
        return h * target_sum / np.sum(h)

    def mutate(h, std):
        h = h + np.random.normal(0, std, n_points)
        return np.clip(normalize(h), 0, 1)

    def crossover(h1, h2):
        idx = np.random.randint(1, n_points - 1)
        return np.concatenate([h1[:idx], h2[idx:]]), np.concatenate([h2[:idx], h1[idx:]])

    pop_size = 50
    pop = [np.clip(normalize(initial_h_values * (1 + 0.1 * np.random.rand(n_points))), 0, 1) for _ in range(pop_size // 2)]
    pop += [np.clip(normalize(np.random.rand(n_points)), 0, 1) for _ in range(pop_size - pop_size // 2)]
    best_h = pop[np.argmin([eval_fitness(h) for h in pop])]
    best_c5 = eval_fitness(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s * 0.8:
            break
        fitness = [eval_fitness(h) for h in pop]
        elites = np.argsort(fitness)[:int(pop_size * 0.2)]
        new_pop = [pop[i] for i in elites]
        for _ in range(len(pop) - len(elites)):
            p1, p2 = np.random.choice(elites, size=2, replace=False)
            h1, h2 = crossover(pop[p1], pop[p2])
            h = mutate(h1, 0.1)
            new_pop.append(h)
        pop = new_pop
        current_best = pop[np.argmin([eval_fitness(h) for h in pop])]
        if eval_fitness(current_best) < best_c5:
            best_h, best_c5 = current_best, eval_fitness(current_best)

    def obj(x):
        return eval_fitness(x)

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
