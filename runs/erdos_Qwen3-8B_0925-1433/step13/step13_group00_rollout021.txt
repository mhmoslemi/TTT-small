import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

        h = clip_and_rescale(h)

        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

        h = clip_and_rescale(h)

    def generate_initializations():
