import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial guess
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    # Objective function
    def objective(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Differential Evolution
    bounds = [(0, 1)] * n_points
    start_time = time.time()

    try:
        result_de = differential_evolution(
            objective, bounds, 
            popsize=20, maxiter=150, tol=1e-5,
            mutation=(0.5, 1), recombination=0.8, polish=True,
            vectorize=True, disp=False
        )
    except:
        result_de = differential_evolution(
            objective, bounds, 
            popsize=15, maxiter=120, tol=1e-5,
            mutation=(0.5, 1), recombination=0.8, polish=True,
            vectorize=True, disp=False
        )

    elapsed_time = time.time() - start_time
    best_h = result_de.x
    best_c5 = result_de.fun
    remaining_time = budget_s - elapsed_time

    # Local optimization
    if remaining_time > 0:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds_slsqp = [(0, 1)] * n_points
        try:
            result_slsqp = minimize(
                objective, best_h, 
                method='SLSQP', bounds=bounds_slsqp, 
                constraints=cons, tol=1e-6, disp=False
            )
            if result_slsqp.success and result_slsqp.fun < best_c5:
                best_h = result_slsqp.x
                best_c5 = result_slsqp.fun
        except:
            pass

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    else:
        best_h = np.zeros(n_points)

    return (best_h, best_c5, n_points)
