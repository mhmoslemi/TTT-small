from scipy.optimize import minimize
import time

    def obj(x, alpha):

        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return (1/alpha) * np.log(np.sum(np.exp(alpha * corr))) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()

    start_time = time.time()

    alpha = 0.1
    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'SLSQP'
            tol = 1e-6
        elif phase == 1:
            method = 'L-BFGS-B'
            tol = 1e-5

            method = 'COBYLA'
            tol = 1e-6
