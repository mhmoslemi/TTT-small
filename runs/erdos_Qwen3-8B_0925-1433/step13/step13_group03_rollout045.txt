import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()

    # Phase 1: DE for global search
    if time.time() - start_time < budget_s:
        try:
            res_de = differential_evolution(
                objective, bounds, strategy='best1bin',
                popsize=15, maxiter=100, tol=1e-6,
                mutation=(0.4, 1), recombination=0.8
            )
            best_h = res_de.x
            best_c5 = res_de.fun
        except:
            pass

    # Phase 2: SLSQP refinement
    if time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res_slsqp = minimize(
            objective, best_h, method='SLSQP', bounds=bounds,
            constraints=cons, tol=1e-8
        )
        if res_slsqp.success:
            best_h = res_slsqp.x
            best_c5 = res_slsqp.fun

    # Phase 3: COBYLA refinement
    if time.time() - start_time < budget_s:
        res_cobyla = minimize(
            objective, best_h, method='COBYLA', bounds=bounds,
            constraints=cons, tol=1e-8
        )
        if res_cobyla.success:
            best_h = res_cobyla.x
            best_c5 = res_cobyla.fun

    # Phase 4: L-BFGS-B fine-tuning
    if time.time() - start_time < budget_s:
        res_lbfgs = minimize(
            objective, best_h, method='L-BFGS-B', bounds=bounds,
            constraints=cons, tol=1e-6
        )
        if res_lbfgs.success:
            best_h = res_lbfgs.x
            best_c5 = res_lbfgs.fun

    # Final perturbation with L-BFGS
    if time.time() - start_time < budget_s:
        for _ in range(30):
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.normal(0, 0.003, n_points)

                constraints=cons, tol=1e-6

                best_h, best_c5 = res.x, res.fun

    # Ensure sum constraint
    sum_h = np.sum(best_h)

        best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
