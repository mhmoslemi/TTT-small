import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return np.clip(h * target_sum / s, 0, 1)

    h = rescale(initial_h_values)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_ = rescale(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()

    for i in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.002 * (0.95 ** i)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = rescale(new_h)
        try:
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    return (best_h, best_c5, n_points)
