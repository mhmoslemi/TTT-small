import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        return h * target_sum / np.sum(h)

    # Initialize population with initial_h_values and structured candidates
    population = [initial_h_values.copy()]
    for _ in range(10):
        h = 0.5 + 0.2 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = np.clip(h, 0, 1)
        h = rescale(h)
        population.append(h)
    population = np.array(population)
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5s.append(np.max(corr) * dx)
        worst_idx = np.argmax(c5s)
        worst_h = population[worst_idx].copy()
        max_c5 = c5s[worst_idx]

        # Gradient descent on worst_h
        corr = np.correlate(worst_h, 1 - worst_h, mode='full')
        lag_idx = np.argmax(corr)
        lag = lag_idx - (n_points - 1)
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = worst_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = rescale(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            o_plus = corr_plus[lag_idx] * dx
            h_minus = worst_h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = rescale(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            o_minus = corr_minus[lag_idx] * dx
            grad[i] = (o_plus - o_minus) / (2 * eps)
        learning_rate = 0.1
        worst_h = worst_h - learning_rate * grad
        worst_h = np.clip(worst_h, 0, 1)
        worst_h = rescale(worst_h)
        population[worst_idx] = worst_h

        # Differential evolution for other candidates
        for i in range(len(population)):
            if i == worst_idx:
                continue

            mutant = population[c] + 0.8 * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            mutant = rescale(mutant)
            corr_mutant = np.correlate(mutant, 1 - mutant, mode='full')

            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return (best_h, best_c5, n_points)
