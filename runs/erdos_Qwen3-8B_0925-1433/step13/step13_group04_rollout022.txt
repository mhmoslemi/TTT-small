        best_h, best_c5 = res.x, res.fun

    # Perturbative refinement with COBYLA
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.0005 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Local search with Nelder-Mead
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.0001 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 20})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
