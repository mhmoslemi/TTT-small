    start_time = time.time()
    F = 0.8
    crossover_rate = 0.9
    while time.time() - start_time < budget_s:
        new_population = []
        c5_values = [compute_c5(h) for h in population]
        diversity_std = np.std(c5_values)
        if diversity_std < 0.1:
            F = min(F + 0.1, 1.2)
        else:
            F = max(F - 0.05, 0.5)
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + F * (population[a] - population[b])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial *= target_sum / np.sum(trial)
            new_population.append(trial)
        new_population = np.array(new_population)
        for i in range(pop_size):
            trial = new_population[i]
            trial_c5 = compute_c5(trial)
            current_c5 = compute_c5(population[i])
            if trial_c5 < current_c5:
                population[i] = trial
        for h in population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        if np.abs(best_c5 - np.min(c5_values)) < 1e-5:
            population = [h_initial.copy()]
            for _ in range(pop_size - 1):
                h_rand = np.random.rand(n_points)
                h_rand = np.clip(h_rand, 0, 1)
                if np.sum(h_rand) > 0:
                    h_rand *= target_sum / np.sum(h_rand)
                population.append(h_rand)
        if time.time() - start_time >= budget_s:
            break
    return (best_h, best_c5, n_points)
