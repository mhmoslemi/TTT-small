import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_scaled = h_clipped * target_sum / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h > 0:
        h_initial = h_initial * target_sum / sum_h
    best_h = h_initial
    best_c5 = compute_c5(best_h)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    phases = [
        ('COBYLA', 0.01, 100, 1e-6),
        ('SLSQP', 0.001, 200, 1e-8),
        ('L-BFGS-B', 0.0001, 100, 1e-9)
    ]
    for method, perturb_std, num_iter, tol in phases:
        for _ in range(num_iter):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = best_h
            res = minimize(compute_c5, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
