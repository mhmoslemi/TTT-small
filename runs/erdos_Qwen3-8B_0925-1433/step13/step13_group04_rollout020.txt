import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_h = np.clip(best_h + 0.05 * np.random.randn(n_points), 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        curr_c5 = evaluate(new_h)
        if curr_c5 < best_c5:
            best_h, best_c5 = new_h, curr_c5

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(lambda x: -evaluate(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = evaluate(best_h)

    H = np.fft.fft(best_h)
    H *= np.blackman(len(H))
    best_h = np.fft.ifft(H).real
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate(best_h)

    return (best_h, best_c5, n_points)
