from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        n_points = len(h)
        k_lag = k_max - (n_points - 1)
        gradient = np.zeros_like(h)
        for j in range(n_points):
            j_plus = j + k_lag
            j_minus = j - k_lag
            j_plus = np.clip(j_plus, 0, n_points - 1)
            j_minus = np.clip(j_minus, 0, n_points - 1)
            h_plus = h[j_plus] if 0 <= j_plus < n_points else 0.0
            h_minus = h[j_minus] if 0 <= j_minus < n_points else 0.0
            gradient[j] = dx * (1 - h_plus) - dx * h_minus
        return -gradient

    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 200
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        step_size = learning_rate
        trial_h = h + step_size * grad
        trial_h = trial_h * target_sum / np.sum(trial_h)
        trial_h = np.clip(trial_h, 0, 1)
        corr_trial = np.max(np.correlate(trial_h, 1 - trial_h, mode='full')) * dx
        if corr_trial < best_c5:
            h = trial_h
            best_c5 = corr_trial
            best_h = h.copy()
        else:
            step_size *= 0.5
            h = h + step_size * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        learning_rate *= decay_rate
