            h_scaled = h_clipped * target_sum / total
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx
        try:
            result_DE = differential_evolution(objective_DE, bounds, maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True, disp=False, workers=1)
            best_h = result_DE.x
            best_c5 = compute_c5(best_h)
        except:
            pass
        time_left = budget_s - (time.time() - start_time)

    # SLSQP refinement
    if time_left > 0:
        def objective_SLSQP(h):
            h_clipped = np.clip(h, 0, 1)
            total = np.sum(h_clipped)
            if total == 0:
                return float('inf')
            h_scaled = h_clipped * target_sum / total
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        try:
            result_SLSQP = minimize(objective_SLSQP, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
            if result_SLSQP.success:
                best_h = result_SLSQP.x
                best_c5 = compute_c5(best_h)
        except:
            pass
        time_left = budget_s - (time.time() - start_time)

    # L-BFGS-B fine-tuning
    if time_left > 0:
        try:
            result_LBFGSB = minimize(objective_SLSQP, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
            if result_LBFGSB.success:
                best_h = result_LBFGSB.x
                best_c5 = compute_c5(best_h)
        except:
            pass

    return (best_h, best_c5, n_points)
