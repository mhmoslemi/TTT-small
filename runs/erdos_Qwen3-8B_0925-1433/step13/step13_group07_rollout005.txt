from scipy.optimize import minimize
import time

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        eps = 1e-5

            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * target_sum / np.sum(h_plus)
