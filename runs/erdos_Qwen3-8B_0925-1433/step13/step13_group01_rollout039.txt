    else:
        h = np.zeros_like(h) + 0.5
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[1:-1]) * dx
    start_time = time.time()
    max_iterations = 200
    learning_rate = 0.01
    decay_rate = 0.95
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        corr_max = corr[1:-1]
        m = np.argmax(corr_max)
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + m < n_points:
                gradient[i] += (1 - h[i + m])
            if i - m >= 0:
                gradient[i] -= h[i - m]
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            scale = target_sum / np.sum(h)
            h *= scale
        else:
            h = np.zeros_like(h) + 0.5
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr[1:-1]) * dx
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        if t % 50 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturb, 0, 1)
            if np.sum(h) > 0:
                scale = target_sum / np.sum(h)
                h *= scale
            else:
                h = np.zeros_like(h) + 0.5
    return (best_h, best_c5, n_points)
