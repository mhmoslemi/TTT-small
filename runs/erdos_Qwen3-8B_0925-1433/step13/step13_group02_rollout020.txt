import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 50
    population = np.zeros((population_size, n_points))
    population[0] = initial_h_values.copy()
    for i in range(1, population_size):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        population[i] = candidate
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        for idx in range(population_size):
            h = population[idx]
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        # Select worst candidate
        worst_idx = np.argmax([np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx for i in range(population_size)])
        worst_h = population[worst_idx]
        # Compute worst lag
        corr = np.correlate(worst_h, 1 - worst_h, mode='full')
        worst_lag_idx = np.argmax(corr)
        # Gradient descent on worst lag
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            perturbed = worst_h.copy()
            perturbed[i] += eps
            perturbed = np.clip(perturbed, 0, 1)
            if np.sum(perturbed) > 0:
                perturbed *= target_sum / np.sum(perturbed)
            corr_pert = np.correlate(perturbed, 1 - perturbed, mode='full')
            diff = (corr_pert[worst_lag_idx] - corr[worst_lag_idx]) * dx / eps
            grad[i] = diff
        step_size = 0.1
        new_h = worst_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h *= target_sum / np.sum(new_h)
        population[worst_idx] = new_h
        # Differential evolution
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[c] + 0.8 * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) > 0:
                mutant *= target_sum / np.sum(mutant)
            new_population.append(mutant)
        new_population = np.array(new_population)
        for i in range(population_size):
            trial = new_population[i]
            corr_trial = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr_trial) * dx

            c5_pop = np.max(corr_pop) * dx
            if c5_trial < c5_pop:
                population[i] = trial
    return (best_h, best_c5, n_points)
