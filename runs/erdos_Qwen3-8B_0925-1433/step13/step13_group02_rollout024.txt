import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 20
    population = np.zeros((pop_size, n_points))
    population[0] = initial_h_values.copy()
    for i in range(1, pop_size):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h *= target_sum / np.sum(h)
        population[i] = h
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx
    start_time = time.time()
    F = 0.8
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + F * (population[a] - population[b])
            trial = np.copy(population[np.random.randint(pop_size)])
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial *= target_sum / np.sum(trial)
            new_population.append(trial)
        new_population = np.array(new_population)
        for i in range(pop_size):
            trial = new_population[i]
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr) * dx
            current = population[i]
            corr_current = np.correlate(current, 1 - current, mode='full')
            c5_current = np.max(corr_current) * dx
            if c5_trial < c5_current:
                population[i] = trial
        corr_list = [np.max(np.correlate(h, 1-h, mode='full')) * dx for h in population]
        std_c5 = np.std(corr_list)
        if std_c5 < 0.01:
            F = 0.5 if F < 1 else 1.2
        else:
            F = 0.8
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return (best_h, best_c5, n_points)
