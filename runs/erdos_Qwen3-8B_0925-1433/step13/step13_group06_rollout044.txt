            grad[i] = dx * (1 - h[i_plus]) - dx * h[i_minus]
        vel = momentum * vel + grad
        h += lr * vel
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h *= target_sum / np.sum(h)
        curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if curr_c5 < best_c5:
            best_h, best_c5 = h.copy(), curr_c5
        lr *= decay

    def obj_lbgfs(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) > 0:
            h *= target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(3):
        if time.time() - t0 > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
