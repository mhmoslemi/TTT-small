import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, maxiter=50,
        tol=1e-8, polish=True, mutation=(0.5, 1), recombination=0.9, seed=seed)
    best_h = result_de.x
    best_c5 = objective(best_h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(objective, best_h, method='SLSQP',
                            bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result_slsqp.x
    best_c5 = objective(best_h)

    result_lbgf = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons, tol=1e-8)

    best_c5 = objective(best_h)

    elapsed = time.time() - start_time
    if elapsed < budget_s:

        best_c5 = objective(best_h)
