import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize with initial_h_values and ensure sum constraint
    h = np.array(initial_h_values)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        # Adaptive perturbation
        perturb_std = 0.001 * (0.95 ** iter_count)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to meet sum constraint
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            scaling = target_sum / new_h_sum
            new_h = np.clip(new_h * scaling, 0, 1)
        else:
            new_h = np.zeros(n_points)

        # Optimization
        res = minimize(
            lambda x: compute_c5(x),
            new_h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5
        # Early stopping if no improvement
        if iter_count > 100 and np.abs(best_c5 - compute_c5(best_h)) < 1e-7:
            break

    return (best_h, best_c5, n_points)
