import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return 1e9
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Phase 1: Differential Evolution
    result = differential_evolution(obj, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, maxiter=100)
    if result.success:
        best_h, best_c5 = result.x, result.fun

    if time.time() - start_time >= budget_s:
        return (best_h, best_c5, n_points)

    # Phase 2: SLSQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    if time.time() - start_time >= budget_s:
        return (best_h, best_c5, n_points)

    # Phase 3: L-BFGS-B refinement
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
