import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def eval_c5(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, delta=0.05):
        idx = np.random.randint(n_points)
        h = h.copy()
        h[idx] += np.random.choice([-delta, delta])
        h = np.clip(h * target_sum / np.sum(h), 0, 1)
        return h

    T = 5.0
    cooling_rate = 0.995
    sa_iter = 1000
    sa_time = 0
    start_time = time.time()

    for i in range(sa_iter):
        if time.time() - start_time > budget_s - sa_time:
            break
        h_new = perturb(best_h)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        else:
            prob = np.exp((c5_new - best_c5) / T)
            if np.random.rand() < prob:
                best_h, best_c5 = h_new, c5_new
        T *= cooling_rate
        sa_time += 1

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
