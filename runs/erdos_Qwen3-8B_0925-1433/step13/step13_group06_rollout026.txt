import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def compute_max_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = compute_max_overlap(h, dx)

    learning_rate = 0.2
    decay_rate = 0.998
    max_iterations = 2000
    start_time = time.time()

    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        current_corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(current_corr)
        k_lag = k_max - (n_points - 1)

        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k_lag < n_points:
                term1 = 1.0 - h[m + k_lag]
            else:
                term1 = 0.0
            if m - k_lag >= 0:
                term2 = h[m - k_lag]
            else:
                term2 = 0.0
            grad[m] = dx * (term1 - term2)

        step = learning_rate
        for _ in range(20):
            trial_h = h + step * grad
            trial_h = clip_and_normalize(trial_h)
            corr_trial = compute_max_overlap(trial_h, dx)
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * grad
        current_c5 = compute_max_overlap(h, dx)
        if current_c5 < best_c5:
