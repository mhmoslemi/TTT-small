def run(seed=42, budget_s=600):

    best_h = h.copy()

    start_time = time.time()

    for _ in range(10000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        worst_lag = np.argmax(overlaps)
        current_c5 = overlaps[worst_lag]
        if current_c5 >= best_c5:
