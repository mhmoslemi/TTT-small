import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h) if np.sum(h) > 0 else 1
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    learning_rate = 0.05
    max_iterations = 1000
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(cross) * dx
        if current_c5 == best_c5:
            break
        max_index = np.argmax(cross)
        m = max_index
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + m
            i_minus = i - m
            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            term2 = -h[i_minus] if 0 <= i_minus < n_points else 0.0
            gradient[i] = term1 + term2
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h) if np.sum(h) > 0 else 1
        current_cross = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_cross) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= 0.99
    return (best_h, best_c5, n_points)
