import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            return h * target_sum / np.sum(h)
        return np.ones_like(h) * 0.5

    def obj(x):
        h_ = rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(corr[1:-1])
        return worst * dx

    h = rescale(initial_h_values)
    best_h = h.copy()
    best_c5 = obj(h)

    def find_worst_lag(h):
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        return worst_idx

    def get_overlap_indices(m):
        overlap_start = 0
        overlap_end = int(n_points - m - 1)
        return slice(overlap_start, overlap_end + 1)

    def perturb_focused(h, m, std):
        indices = get_overlap_indices(m)
        perturb = np.zeros_like(h)
        perturb[indices] = np.random.normal(0, std, len(perturb[indices]))
        return h + perturb

    start_time = time.time()
    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            perturb_std = 0.02
            method = 'L-BFGS-B'
            tolerance = 1e-6
        elif phase == 1:
            perturb_std = 0.005
            method = 'COBYLA'
            tolerance = 1e-5
        else:
            perturb_std = 0.001
            method = 'SLSQP'
            tolerance = 1e-8

        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            m = find_worst_lag(best_h)
            perturbed = perturb_focused(best_h, m, perturb_std)
            res = minimize(obj, perturbed, method=method, bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=tolerance)
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
    return (best_h, best_c5, n_points)
