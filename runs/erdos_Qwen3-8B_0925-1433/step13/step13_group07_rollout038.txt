import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        lag = m - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + lag
            i_minus_k = i - lag
            term1 = 1.0 - h[i_plus_k] if 0 <= i_plus_k < n_points else 0.0
            term2 = h[i_minus_k] if 0 <= i_minus_k < n_points else 0.0
            grad[i] = term1 - term2
        return -grad  # Negative for descent direction

    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 200
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        direction = grad
        step = learning_rate
        for _ in range(20):
            trial_h = h + step * direction
            trial_h = np.clip(trial_h, 0, 1)
            trial_h_norm = trial_h * target_sum / np.sum(trial_h)
            corr_trial = np.max(np.correlate(trial_h_norm, 1 - trial_h_norm, mode='full')) * dx
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * direction
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate

    def obj_lbgfs(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
