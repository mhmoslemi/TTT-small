import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    pop_size = 25
    sigma = 0.1
    mean = h.copy()
    cov = np.eye(n_points) * sigma ** 2

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        candidates = []
        for _ in range(pop_size):
            perturb = np.random.normal(0, sigma, n_points)
            cand = np.clip(mean + perturb, 0, 1)
            sum_cand = np.sum(cand)
            if sum_cand > 0:
                cand = cand * target_sum / sum_cand
            candidates.append(cand)

        c5s = []
        for cand in candidates:
            corr = np.correlate(cand, 1 - cand, mode="full")
            c5s.append(np.max(corr) * dx)

        best_idx = np.argmin(c5s)
        best_cand = candidates[best_idx]
        best_c5_val = c5s[best_idx]

        if best_c5_val < best_c5:
