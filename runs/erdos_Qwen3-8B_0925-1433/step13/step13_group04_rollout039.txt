            new_pop.append(child)
        population = new_pop
        new_fitness = [np.max(np.correlate(ind, 1 - ind, mode='full')) * dx for ind in population]
        for i in range(pop_size):
            if new_fitness[i] < best_c5:
                best_c5 = new_fitness[i]
                best_h = population[i]
    return (best_h, best_c5, n_points)
