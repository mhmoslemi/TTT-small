                h[j] = min(1, h[j] + learning_rate * contrib)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points) + 0.5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        if iter % 10 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points) + 0.5
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr[1:-1]) * dx
