import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = 0.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return np.inf
        x_norm = x_clipped * target_sum / sum_clipped
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    result = differential_evolution(obj, bounds, strategy='best1bin', maxiter=200, popsize=20, tol=1e-5, mutation=(0.5, 1), recombination=0.8, seed=seed)
    best_h = result.x
    best_c5 = result.fun

    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(obj_slsqp, best_h, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-8)
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    return (best_h, best_c5, n_points)
