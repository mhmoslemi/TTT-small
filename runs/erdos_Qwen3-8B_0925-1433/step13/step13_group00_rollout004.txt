from scipy.optimize import minimize
import time

    def obj(x):
        h = rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    initial_guesses = [initial_h_values.copy()]
    permuted = np.random.permutation(initial_h_values)

    mirrored = np.flip(initial_h_values)
    initial_guesses.append(mirrored)

    best_h = rescale(initial_guesses[0])
    best_c5 = obj(best_h)
    best_idx = 0

    start_time = time.time()

    for gi in range(len(initial_guesses)):

        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8)
