            trial_h = h + step * direction
            trial_h = trial_h * target_sum / np.sum(trial_h)
            corr_trial = np.max(np.correlate(trial_h, 1 - trial_h, mode='full')) * dx
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * direction
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate

    def obj_lbgfs(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
