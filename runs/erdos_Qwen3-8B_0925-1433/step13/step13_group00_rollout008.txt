import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    def constrain(x):
        return np.sum(x) - target_sum

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()

    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'COBYLA'
            perturb_std = 0.01
            iters = 100
        elif phase == 1:
            method = 'SLSQP'
            perturb_std = 0.001
            iters = 150
        else:
            method = 'L-BFGS-B'
            perturb_std = 0.0005
            iters = 50

        for _ in range(iters):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = best_h
            res = minimize(obj, new_h, method=method, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constrain}, tol=1e-6 if phase == 0 else 1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x.copy(), res.fun

    return (best_h, best_c5, n_points)
