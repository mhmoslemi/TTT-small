from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    tournament_size = 5

    crossover_rate = 0.8

        if np.random.rand() < 0.8:

            h = np.random.rand(n_points)

        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        # Select elites
        elite_indices = np.argsort(fitness)[:int(pop_size * 0.2)]
        elites = [population[i] for i in elite_indices]

            tournament = np.random.choice(pop_size, tournament_size)
            parents = [population[i] for i in tournament]

                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            # Mutation

        # Replace population
        population = elites + offspring

        for h in population:

    # Final L-BFGS refinement
