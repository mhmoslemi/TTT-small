from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)

            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()

    best_c5 = obj(h)
    
    # PSO initialization
    particles = [np.clip(h + 0.1 * np.random.randn(n_points), 0, 1) * target_sum / np.sum(np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)) for _ in range(20)]
    vel = np.random.randn(20, n_points) * 0.05
    pbest = particles.copy()
    pbest_vals = np.array([obj(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)
    
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break

                particles[i] = particles[i] * target_sum / sum_p
            curr_val = obj(particles[i])
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
        gbest = pbest[np.argmin(pbest_vals)]
        gbest_val = np.min(pbest_vals)
    
    best_h = gbest
    best_c5 = gbest_val
    
    # Frequency filtering
    def freq_filter(h, cutoff=0.25):
        fft_h = np.fft.fft(h)
        freqs = np.fft.fftfreq(n_points)
        mask = np.abs(freqs) < cutoff
        fft_h[mask] *= 0.5
        return np.fft.ifft(fft_h).real
    
    best_h = freq_filter(best_h)
    sum_h = np.sum(best_h)
    if sum_h > 0:
