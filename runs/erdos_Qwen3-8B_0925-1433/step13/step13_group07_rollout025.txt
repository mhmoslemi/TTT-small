import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def evaluate(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with perturbations from initial_h_values
    n_pop = 50
    pop = np.random.normal(0, 0.05, (n_pop, n_points))
    for i in range(n_pop):
        pop[i] = initial_h_values + pop[i]
        pop[i] = np.clip(pop[i], 0, 1)
        pop[i] = pop[i] * target_sum / np.sum(pop[i])
    fitness = np.array([evaluate(p) for p in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Mutation
        for i in range(n_pop):
            a, b, c = np.random.choice(n_pop, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            # Crossover
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.7:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)
            # Evaluate
            corr = evaluate(trial)
            if corr < fitness[i]:
                pop[i] = trial
                fitness[i] = corr
                if corr < best_c5:
                    best_h = trial.copy()
                    best_c5 = corr

    # L-BFGS refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate, best_h, method='L-BFGS-B',
                       bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
            best_h = np.clip(best_h, 0, 1)

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
