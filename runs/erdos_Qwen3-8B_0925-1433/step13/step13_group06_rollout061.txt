import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if k_max < n_points:
                # Lag corresponds to shifting by k_max
                # Overlap for i and i + k_max
                if (i + k_max) < n_points:
                    grad[i] += (1 - h[i + k_max]) * dx
                if (i - k_max) >= 0:
                    grad[i] -= h[i - k_max] * dx
                else:
                    grad[i] -= h[(i - k_max) % n_points] * dx
            else:
                # Handle negative lags
                k = k_max - n_points
                if (i - k) >= 0:
                    grad[i] += (1 - h[i - k]) * dx
                if (i + k) < n_points:
                    grad[i] -= h[i + k] * dx
                else:
                    grad[i] -= h[(i + k) % n_points] * dx
        return grad

    learning_rate = 0.1
