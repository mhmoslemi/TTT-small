import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n_points

    def obj(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2.0 / n_points

    def compute_gradient(h, lag_m):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + lag_m
            if 0 <= j < n_points:
                grad[i] = h[j]
            else:
                grad[i] = 0.0
        return grad

    decay_rate = 0.999
    learning_rate = 0.1
    max_iterations = 6000
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, h, mode='full')
        min_index = np.argmin(corr)
        lag_m = min_index - (n_points - 1)
        grad = compute_gradient(h, lag_m)
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        curr_corr = np.correlate(h, h, mode='full')
