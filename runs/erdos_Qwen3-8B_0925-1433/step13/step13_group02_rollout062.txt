    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h_clipped = np.clip(initial_h_values, 0, 1)
    initial_h_rescaled = initial_h_clipped * target_sum / np.sum(initial_h_clipped)
    best_h = initial_h_rescaled.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.998
    max_iterations = 3000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        current_c5 = np.max(corr)
        worst_lag = np.argmax(corr)
        
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus *= target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[worst_lag] * dx
            
            h_minus = best_h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus *= target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[worst_lag] * dx
            
            grad[i] = (corr_plus - corr_minus) / (2 * epsilon)
        
        best_h -= learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        best_h *= target_sum / np.sum(best_h)
        
        new_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if new_c5 < current_c5:
            best_c5 = new_c5
        else:
            perturb = np.random.normal(0, 0.05, n_points)
            best_h = np.clip(best_h + perturb, 0, 1)
            best_h *= target_sum / np.sum(best_h)
            best_c5 = min(best_c5, np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx)
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
