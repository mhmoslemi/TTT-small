import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 200
    alpha = 0.01
    eta = 2.0
    learning_rate = 0.5
    learning_rate_decay = 0.999
    m = 20

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max_overlap = np.max(corr) * dx
        weights = np.zeros_like(corr)
        for k in range(len(corr)):
            overlap = corr[k] * dx
            weights[k] = eta * np.exp(-alpha * np.abs(overlap - current_max_overlap))
        sorted_indices = np.argsort(weights)[::-1][:m]
        grad = np.zeros(n_points)
        for k in sorted_indices:
            k_adj = k % n_points
            grad += 1 - h[(np.arange(n_points) + k_adj) % n_points]
        lr = learning_rate * (1 + np.cos(t / max_iter)) / 2
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= learning_rate_decay

    return (np.clip(best_h, 0, 1), best_c5, n_points)
