from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    max_iterations = 100
    start_time = time.time()
    learning_rates = [0.2, 0.1, 0.05]
    decay_factors = [0.999, 0.998, 0.99]

    for lr_idx in range(len(learning_rates)):
        if time.time() - start_time > budget_s:
            break
        lr = learning_rates[lr_idx]
        decay = decay_factors[lr_idx]
        h = best_h.copy()
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (n_points - 1)
            grad = np.zeros(n_points)

                grad[i] = (1 - h[j]) - h[(i - k) % n_points]
            h += lr * grad

            h = h * target_sum / np.sum(h)

            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = h.copy()

            lr *= decay

    # Perturbation phase
