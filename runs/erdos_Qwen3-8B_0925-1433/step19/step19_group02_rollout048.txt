import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 1000
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)

        gradient = 1.0 - np.roll(h, lag)

        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * gradient ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        h = h * target_sum / np.sum(h)
