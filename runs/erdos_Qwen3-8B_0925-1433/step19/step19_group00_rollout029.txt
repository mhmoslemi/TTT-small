import time

    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iter = 200
    alpha = 0.01
    eta = 2.0
    decay_rate = 0.999
    lr = 0.5
    m = 30
    C2 = best_c5
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.argsort(np.abs(corr))[-m:]
        grad = np.zeros(n_points)
