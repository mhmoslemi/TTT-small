    best_c5 = min(best_c5, 0.381)

    max_iterations = 10000
    m = 6
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1
    t = 0
    first_moment = np.zeros(n_points)
    second_moment = np.zeros(n_points)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:m]

            grad += 1 - np.roll(h, k)
        # Adam update
        t += 1
        first_moment = beta1 * first_moment + (1 - beta1) * grad
        second_moment = beta2 * second_moment + (1 - beta2) * (grad ** 2)
        bias_corrected_first = first_moment / (1 - beta1 ** t)
        bias_corrected_second = second_moment / (1 - beta2 ** t)
        h -= lr * bias_corrected_first / (np.sqrt(bias_corrected_second) + epsilon)
        h = np.clip(h, 0, 1)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
