import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    temp = 1.0
    temp_decay = 0.995
    max_iterations = 600
    perturb_std = 0.05
    time_start = time.time()

    for iteration in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        if iteration < 150:
            perturb_std = 0.05
        else:
            perturb_std = 0.01
        perturb = np.random.normal(0, perturb_std, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        new_c5 = np.max(np.correlate(h_perturb, 1 - h_perturb, mode='full')) * dx
        delta = new_c5 - current_c5

            best_h = h.copy()

        temp *= temp_decay
