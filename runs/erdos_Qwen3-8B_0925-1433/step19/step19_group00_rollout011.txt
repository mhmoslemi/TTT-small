                pbest[i] = h.copy()
        for i in range(num_particles):
            if pbest_scores[i] < gbest_score:
                gbest_score = pbest_scores[i]
                gbest = pbest[i].copy()
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * omega + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            vel = np.clip(vel, vel_bounds[0], vel_bounds[1])
            velocities[i] = vel
            new_h = particles[i] + velocities[i]
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            particles[i] = new_h
        if gbest_score < best_c5:
            best_c5 = gbest_score
            best_h = gbest.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
