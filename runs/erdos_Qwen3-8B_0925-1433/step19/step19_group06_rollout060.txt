import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 50
    max_gen = 500
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 5

    # Initialize population
    population = []
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population.append(h_initial)
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    # Evaluate individual
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial best
    population_fitness = [evaluate(h) for h in population]
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]

    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        # Selection
        parents = []
        for _ in range(2):
            tournament = np.random.choice(population, tournament_size, replace=False)
            selected = min(tournament, key=evaluate)
            parents.append(selected)
        parent1, parent2 = parents
        # Crossover
        if np.random.rand() < crossover_rate:
            offspring = (parent1 + parent2) / 2.0
        else:
            offspring = parent1.copy()
        # Mutation
        std = 0.05 * (1 - gen / max_gen)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                offspring[i] += np.random.normal(0, std)
        offspring = np.clip(offspring, 0, 1)
        offspring = offspring * target_sum / np.sum(offspring)
        # Evaluate offspring
        offspring_c5 = evaluate(offspring)
        # Replace worst
        worst_idx = np.argmax(population_fitness)
        if offspring_c5 < population_fitness[worst_idx]:
            population[worst_idx] = offspring.copy()
            population_fitness[worst_idx] = offspring_c5
            if offspring_c5 < best_c5:
                best_c5 = offspring_c5
                best_h = offspring.copy()
        # Early stopping
        if gen % 50 == 0 and best_c5 < 0.381:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
