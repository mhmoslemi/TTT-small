        curr_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(curr_corr)
        lag = max_idx - (n_points - 1)
        
        i_plus = (np.arange(n_points) + lag) % n_points
        i_minus = (np.arange(n_points) - lag) % n_points
        grad = (1.0 - h[i_plus]) - h[i_minus]
        
        t += 1
        grad_moment = beta1 * grad_moment + (1 - beta1) * grad
        grad_squares = beta2 * grad_squares + (1 - beta2) * grad ** 2
        m_hat = grad_moment / (1 - beta1**t)
        v_hat = grad_squares / (1 - beta2**t)
        h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
