import numpy as np
import math
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population with 100 diverse candidates
    population = []
    for _ in range(100):
        candidate = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.05, n_points)
        candidate += noise
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)
    
    best_h = population[0].copy()
    best_c5 = np.max(np.correlate(population[0], 1 - population[0], mode='full')) * dx
    
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current population fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c = np.max(corr) * dx
            fitness.append(c)
        
        # Select top 10% as elites
        elite_indices = np.argsort(fitness)[:int(len(population) * 0.1)]
        elites = [population[i] for i in elite_indices]
        
        # Generate new population via crossover and mutation
        new_population = elites.copy()
        while len(new_population) < 100:
            p1, p2 = np.random.choice(elites, size=2, replace=False)
            child = (p1 + p2) / 2.0
            mask = np.random.rand(n_points) < 0.1
            noise = np.random.normal(0, 0.05, n_points)
            child[mask] += noise[mask]
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Apply local refinement to top 5 candidates
        top_fitness = np.argsort(fitness)
        refined = []
        for i in top_fitness[:5]:
            h = population[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            lag_idx = np.argmax(corr)
            lag = lag_idx - (n_points - 1)
            shifted = (np.arange(n_points) + lag) % n_points
            grad = 1 - h[shifted]
            for _ in range(10):
                h = h - 0.05 * grad
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
            refined.append(h)
        new_population += refined
        
        # Final selection: sort and keep top 100
        new_fitness = []
        for h in new_population:
            corr = np.correlate(h, 1 - h, mode='full')
            new_fitness.append(np.max(corr) * dx)
        sorted_indices = np.argsort(new_fitness)
        population = [new_population[i] for i in sorted_indices[:100]]
        
        # Update best solution
        current_c5 = new_fitness[sorted_indices[0]]
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
