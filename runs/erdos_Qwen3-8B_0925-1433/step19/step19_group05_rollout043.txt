import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    def gen_pop(parent):
        pop = []
        mean = 0.1 * np.sin(np.linspace(0, np.pi, n_points))
        noise = np.random.normal(0, 0.02, n_points)
        base = parent + mean + noise
        base = np.clip(base, 0, 1)
        base = base * target_sum / np.sum(base)
        return base

    population = [gen_pop(initial_h_values) for _ in range(50)]
    best_h = min(population, key=compute_c5)
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Genetic Algorithm
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        scores = [compute_c5(h) for h in population]
        top_indices = np.argsort(scores)[:int(len(population)*0.1)]
        parents = [population[i] for i in top_indices]
        new_population = []
        for _ in range(len(population)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            mutation = np.random.normal(0, 0.02, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Simplex refinement
    def simplex_obj(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        h_norm = h * target_sum / sum_h
        h_norm = np.clip(h_norm, 0, 1)
        return compute_c5(h_norm)

    result = minimize(simplex_obj, best_h.ravel(), method='Nelder-Mead', options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100})
    best_h = result.x.reshape(n_points)
    best_h = best_h * target_sum / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)

    # Final perturbations
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.04, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        c5_pert = compute_c5(perturbed)
        if c5_pert < best_c5:
            best_h = perturbed
            best_c5 = c5_pert

    return (best_h, best_c5, n_points)
