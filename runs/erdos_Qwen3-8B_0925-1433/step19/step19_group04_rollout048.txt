from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.1
    decay_rate = 0.9985
    momentum = 0.9
    max_iterations = 500
    start_time = time.time()
    v = np.zeros(n_points)

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            if 0 <= i_plus_k < n_points:
                h_i_plus_k = h[i_plus_k]
            else:
                h_i_plus_k = 0.0
            i_minus_k = i - k
            if 0 <= i_minus_k < n_points:
                h_i_minus_k = h[i_minus_k]
            else:
                h_i_minus_k = 0.0
            gradient[i] = (1 - h_i_plus_k) - h_i_minus_k
        v = momentum * v + learning_rate * gradient

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
