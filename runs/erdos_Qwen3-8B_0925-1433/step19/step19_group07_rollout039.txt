import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    T = 1.0
    alpha = 0.995
    perturbation_scale = 0.1
    max_iter = int(budget_s / 0.1)
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr_perturbed) * dx
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_current) * dx
        
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_perturbed.copy()
        
        if new_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = new_c5
        
        T *= alpha
        if _ % 10 == 0:
            perturbation_scale *= 0.95
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
