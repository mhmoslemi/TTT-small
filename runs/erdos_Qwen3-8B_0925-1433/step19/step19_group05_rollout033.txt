import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial guess: rescale the parent to meet the sum constraint
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[n_points - 3:n_points + 2]) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        valid_indices = np.arange(n_points - 3, n_points + 2)
        valid_indices = np.clip(valid_indices, 0, len(corr) - 1)
        corr_valid = corr[valid_indices]
        return np.max(corr_valid) * dx

    # Nelder-Mead optimization
    start_time = time.time()
    result = minimize(compute_c5, h, method='Nelder-Mead', bounds=[(0, 1)] * n_points)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    if time.time() - start_time > budget_s * 0.7:
        pass  # Early exit if most time is used
    else:
        # Perturbation refinement
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.03, n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            c5 = compute_c5(perturbed)
            if c5 < best_c5:
                best_h = perturbed
                best_c5 = c5

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
