import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    pop_size = 100
    pop = [h.copy() for _ in range(pop_size)]
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in pop]
        elites = np.argsort(fitness)[-int(pop_size * 0.1):]
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (pop[p1] + pop[p2]) / 2.0
            mut_mask = np.random.rand(n_points) < 0.1
            child[mut_mask] += np.random.normal(0, 0.05, sum(mut_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
        top_idx = np.argsort(fitness)[-5:]
        for i in top_idx:
            h = pop[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            lag = np.argmax(corr) - (n_points - 1)
            shifted = (np.arange(n_points) + lag) % n_points
            grad = 1 - h[shifted]
            h = h - 0.05 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_h, best_c5 = h.copy(), curr_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
