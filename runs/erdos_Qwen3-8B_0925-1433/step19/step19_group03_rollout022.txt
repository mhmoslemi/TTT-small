import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population from initial_h_values with perturbations
    population = [initial_h_values.copy() + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) for h in population]
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    max_generations = 100

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:int(len(population)/10)]]
        new_population = elites.copy()

        # Crossover and mutation
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (p1 + p2) / 2
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)

        population = new_population
        top_fitness_indices = np.argsort(fitness)[-5:]
        for i in top_fitness_indices:
            h = population[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr_idx = np.argmax(corr)
            k = max_corr_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for j in range(n_points):

            h = h - 0.05 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            population[i] = h

        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_h = population[current_best].copy()
            best_c5 = fitness[current_best]
