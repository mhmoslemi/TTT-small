    personal_best = np.copy(particles)
    personal_best_fitness = np.inf * np.ones(P)
    global_best = np.copy(particles[0])
    global_best_fitness = np.inf

    for iteration in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        r1 = np.random.rand(P)
        r2 = np.random.rand(P)
        velocities = omega * velocities + c1 * r1 * (personal_best - particles) + c2 * r2 * (global_best - particles)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))

        for i in range(P):
            h_i = particles[i]
            corr = np.correlate(h_i, 1 - h_i, mode='full')
            current_c5 = np.max(corr) * dx
            penalty = 100 * (np.sum(h_i) - target_sum)**2
            fitness = current_c5 + penalty
            if fitness < personal_best_fitness[i]:
                personal_best[i] = h_i.copy()
                personal_best_fitness[i] = fitness
            if fitness < global_best_fitness:
                global_best = h_i.copy()
                global_best_fitness = fitness

    def nelder_mead_objective(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        penalty = 100 * (np.sum(h) - target_sum)**2
        return current_c5 + penalty

    h_nelder = global_best.copy()
    result = minimize(nelder_mead_objective, h_nelder, method='Nelder-Mead', options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 50})
    best_h = np.clip(result.x, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr_final) * dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
