import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    temp = 1.5
    cooling_rate = 0.995
    max_iter = 2000
    start = time.time()

    for _ in range(max_iter):
        if time.time() - start > budget_s:
            break
        sigma = 0.05 * temp
        sigma = max(sigma, 0.005)
        perturb = np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        new_c5 = evaluate_erdos_solution(h_perturbed)

        if new_c5 < best_c5:
            h = h_perturbed.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = h_perturbed.copy()

        temp *= cooling_rate

    best_c5 = evaluate_erdos_solution(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
