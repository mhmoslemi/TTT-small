import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 64

    start_time = time.time()
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    learning_rate = 0.01
    t = 0
    max_iterations = 10000
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation and find top lags
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:10]
        true_lags = [(i - (n_points - 1)) for i in top_indices]

        grad = np.zeros_like(h)
        for k in true_lags:
            if k < 0:
                k += n_points
            grad += 1 - np.roll(h, k)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Evaluate and track best
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        
        # Early stop if gradient is small
        if np.linalg.norm(grad) < 1e-5:
            break
        
        # Decay learning rate
        if t % 100 == 0:
            learning_rate *= 0.99
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
