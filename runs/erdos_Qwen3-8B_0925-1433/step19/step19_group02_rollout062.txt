import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 1000
    T = max_iter
    lr0 = 0.05
    top_lags = 8
    adaptive = True

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        top_indices = np.argsort(overlaps)[::-1][:top_lags]
        grad = np.zeros(n_points)
        for m in top_indices:
            k = m - (n_points - 1)
            shifts = (np.arange(n_points) + k) % n_points
            grad += 1 - h[shifts]
        if adaptive:
            lr = lr0 * (1 + np.cos(t / T)) / 2
            norm = np.linalg.norm(grad)
            if norm > 1e-3:
                lr *= 0.1 / norm
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
