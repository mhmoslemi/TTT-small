            h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
            h = h * (n_points / 2) / np.sum(h)
            h_pop.append(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    max_iter = 10000
    lr = 0.05
    decay_rate = 0.999
    min_lr = 0.0001
    m = 5
    start_time = time.time()
    no_improve = 0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        current_c5 = []
        for h in h_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            current_c5.append(c5)
        best_idx = np.argmin(current_c5)
        best_c5_current = current_c5[best_idx]
        if best_c5_current < best_c5:
            best_h = h_pop[best_idx].copy()
            best_c5 = best_c5_current
        else:
            no_improve += 1
        if no_improve >= 50:
            worst_idx = np.argmax(current_c5)
            h_pop[worst_idx] = np.random.rand(n_points)
            h_pop[worst_idx] = h_pop[worst_idx] * target_sum / np.sum(h_pop[worst_idx])
            no_improve = 0
        for i in range(len(h_pop)):
            h = h_pop[i]
            corr = np.correlate(h, 1 - h, mode='full')
            top_lags = np.argsort(corr)[::-1][:m]
            true_lags = [j - (n_points - 1) for j in top_lags]
            grad = np.zeros_like(h)
            for k in true_lags:
                if k < 0:
                    k += n_points
                grad += 1 - np.roll(h, k)
            lr = lr * decay_rate
            if lr < min_lr:
                lr = min_lr
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            h_pop[i] = h
    return (np.clip(best_h, 0, 1), best_c5, n_points)
