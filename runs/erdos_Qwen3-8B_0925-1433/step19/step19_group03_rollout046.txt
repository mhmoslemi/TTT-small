def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    max_gen = 100
    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    for h in pop:
        h = h * target_sum / np.sum(h)
        np.clip(h, 0, 1, out=h)
    for _ in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fit = []
        for h in pop:
            corr = np.correlate(h, 1-h, mode='full')
            fit.append(np.max(corr) * dx)
        fit = np.array(fit)
        idx = np.argsort(fit)
        elites = pop[idx[:int(pop_size * 0.2)]]
        new_pop = elites.copy()
        for _ in range(pop_size - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = np.copy(p1)
            mask = np.random.rand(n_points) > 0.5
            child[mask] = p2[mask]
            mask = np.random.rand(n_points) < 0.1
            child[mask] += np.random.normal(0, 0.1, np.sum(mask))
            np.clip(child, 0, 1, out=child)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        for h in new_pop:
            corr = np.correlate(h, 1-h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        pop = new_pop
    return (np.clip(best_h, 0, 1), best_c5, n_points)
