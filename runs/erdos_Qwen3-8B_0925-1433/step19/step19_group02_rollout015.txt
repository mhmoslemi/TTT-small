import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 1000
    lr0 = 0.05

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        grad = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            lag_index = k + (n_points - 1)
            overlap_k = corr[lag_index]
            shifted_h1 = np.roll(h, shift=-k)
            shifted_h2 = np.roll(h, shift=k)
            term = (1 - shifted_h1) - shifted_h2
            grad += term * overlap_k
        current_lr = lr0 * (1 + np.cos(t / max_iter)) / 2
        h = h - current_lr * grad

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
