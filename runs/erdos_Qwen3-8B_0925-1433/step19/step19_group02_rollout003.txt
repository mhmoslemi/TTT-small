            lr = lr0 * (1 + np.cos(t / T)) / 2
            h -= lr * m_hat / (np.sqrt(v_hat) + eps)
            
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n_points
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
