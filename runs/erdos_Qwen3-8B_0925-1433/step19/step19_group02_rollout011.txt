import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = initial_h_values.copy()
    best_c5 = 0.381

    # Generate 8 diverse perturbations
    def perturb(h):
        h = h.copy()
        if np.random.rand() < 0.5:
            noise = np.random.normal(0, 0.1, size=n_points)
            h += noise
        if np.random.rand() < 0.5:
            h = 1 - h
        if np.random.rand() < 0.5:
            np.random.shuffle(h)
        if np.random.rand() < 0.5:
            h = np.where(np.random.rand(n_points) < 0.3, 0, h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Multi-start optimization
    for _ in range(8):
        h = perturb(initial_h_values)
        h_current = h.copy()
        best_c5_current = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
        start_time = time.time()
        max_iter = 1000
        lr0 = 0.05
        top_lags = 5

        for t in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            top_lags_indices = np.argsort(corr)[::-1][:top_lags]
            grad = np.zeros(n_points)
            for m in top_lags_indices:
                k = m - (n_points - 1)
                indices = (np.arange(n_points) + k) % n_points
                grad += 1 - h_current[indices]
            current_lr = lr0 * (1 + np.cos(t / max_iter)) / 2
            h_current -= current_lr * grad
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * target_sum / np.sum(h_current)
            current_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                best_h = h_current.copy()

        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
