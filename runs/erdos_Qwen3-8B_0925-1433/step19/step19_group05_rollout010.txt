            if c < best_c5:
                best_h = h.copy()
                best_c5 = c

    def simplex_step(x):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, x, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        return res.x

    best_h = simplex_step(best_h)
    best_c5 = compute_c5(best_h)

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        c = compute_c5(h)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c

    final_h = np.clip(best_h, 0, 1)
    return (final_h, best_c5, n_points)
