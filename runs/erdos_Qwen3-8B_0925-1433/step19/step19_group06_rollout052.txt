import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population_size = 50
    pop = []
    h = initial_h * target_sum / np.sum(initial_h)
    pop.append(h)
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        pop.append(h)
    pop = np.array(pop)

    best_h = pop[0].copy()
    best_c5 = np.max(np.correlate(pop[0], 1 - pop[0], mode='full')) * dx

    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.15
    max_gen = 500
    sigma = 0.05
    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fitness = np.zeros(pop.shape[0])
        for i in range(pop.shape[0]):
            corr = np.correlate(pop[i], 1 - pop[i], mode='full')
            fitness[i] = np.max(corr) * dx
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx].copy()

        parents = []
        for _ in range(2):
            tour = np.random.choice(pop.shape[0], tournament_size, replace=False)
            winner = tour[np.argmin(fitness[tour])]
            parents.append(pop[winner].copy())

        if np.random.rand() < crossover_rate:
            child = (parents[0] + parents[1]) / 2
        else:
            child = parents[0].copy()

        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, sigma * (1 - gen / max_gen), n_points)
            child += noise

        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        worst_idx = np.argmax(fitness)
        pop[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(final_corr) * dx
    return (np.clip(best_h, 0, 1), best_c5, n_points)
