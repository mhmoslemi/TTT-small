import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 300
    alpha = 0.5
    lr0 = 0.05
    top_lags = 10

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        weights = np.exp(-alpha * lags ** 2)
        weighted_corr = corr * weights
        top_lags_indices = np.argsort(weighted_corr)[::-1][:top_lags]
        grad = np.zeros(n_points)
        
        for m in top_lags_indices:
            k = m - (n_points - 1)
            indices = (np.arange(n_points) + k) % n_points
            grad += 1 - h[indices]
        
        current_lr = lr0 / (1 + t / max_iter)
        h = h - current_lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
