import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraints = ({
        'type': 'eq',
        'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - target_sum
    },)
    
    bounds = [(0, 1) for _ in range(n_points)]
    time_start = time.time()
    
    # Initial optimization
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-7, options={'maxiter': 100})
    if res.success:
        best_h = res.x
        best_c5 = objective(best_h)
    
    # Perturbation refinement
    for _ in range(3):
        if time.time() - time_start > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        cur_c5 = np.max(np.correlate(h_perturb, 1 - h_perturb, mode='full')) * dx
        if cur_c5 < best_c5:
            best_h = h_perturb
            best_c5 = cur_c5
    
    # Final clip and scale
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
