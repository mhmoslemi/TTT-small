import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    M = 20
    particles = [np.array(initial_h_values, dtype=float) + np.random.normal(0, 0.05, n_points) for _ in range(M)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(p) for p in particles]
    velocities = [np.zeros(n_points) for _ in range(M)]
    best_h = particles[0]
    best_c5 = compute_c5(best_h)
    gbest = best_h

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(M):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] *= 0.8
            velocities[i] += 1.5 * r1 * (particles[i] - best_h)
            velocities[i] += 1.5 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.05, 0.05)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        current_c5 = compute_c5(particles[0])
        if current_c5 < best_c5:
            best_h = particles[0].copy()
            best_c5 = current_c5
            gbest = best_h
        for i in range(M):
            ci = compute_c5(particles[i])
            if ci < compute_c5(best_h):
                best_h = particles[i].copy()
                best_c5 = ci
                gbest = best_h
    return (np.clip(best_h, 0, 1), best_c5, n_points)
