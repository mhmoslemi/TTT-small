```python
import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    n_particles = 40
    # Initialize population
    particles = []
    base = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    particles.append(base)
    for _ in range(n_particles - 1):
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        p = base + perturb
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        particles.append(p)
    particles = np.array(particles)
    
    # Initialize velocities
    velocities = np.random.uniform(-0.5, 0.5, (n_particles, n_points))
    
    # Initialize personal bests and global best
    pbest = np.copy(particles)
    pbest_c5 = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    gbest = np.copy(pbest[np.argmin(pbest_c5)])
    gbest_c5 = np.min(pbest_c5)
    
    start_time = time.time()
    max_iterations = 400
    omega = 0.9
    c1 = 1.5
    c2 = 1.5
    no_improvement = 0
    best_c5 = gbest_c5
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        current_c5 = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        # Evaluate new fitness
        new_c5 = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
        # Update personal and global best
        for i in range(n_particles):
            if new_c5[i] < pbest_c5[i]:
                pbest[i] = np.copy(particles[i])
                pbest_c5[i] = new_c5[i]
                if new_c5[i] < gbest_c5:
                    gbest = np.copy(particles[i])
                    gbest_c5 = new_c5[i]
        # Early stopping
        if gbest_c5 == best_c5:
            no_improvement += 1
            if no_improvement > 10:
                break
        else:
            best_c5 = gbest_c5
            no_improvement = 0
    
    return (np.clip(gbest, 0, 1), gbest_c5, n_points)
```
