    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 1000
    lr0 = 0.05
    alpha = 0.01

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        weights = np.exp(-alpha * lags**2)
        current_max = np.max(corr) * dx

        grad = np.zeros(n_points)
        for idx in range(len(corr)):
            lag = lags[idx]
            shift = (np.arange(n_points) + lag) % n_points
            contrib = (1 - h[shift])
            grad += contrib * weights[idx] * (1 - 2 * np.random.rand())
            grad += (h - np.mean(h)) * np.random.rand()

        current_lr = lr0 * (1 + np.cos(t / max_iter)) / 2
        h = h - current_lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
