def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        sum_h = np.sum(h_vals)
        h_vals = h_vals * target_sum / sum_h
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    # Wavelet-based initialization
    try:
        coeffs = pywt.wavedec(h, 'db4', level=1)
        coeffs[1] *= 1.2  # Increase high-frequency component
        h = pywt.waverec(coeffs, 'db4')
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    except:
        pass

    # Gradient descent with momentum
    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 300
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        gradient = np.zeros(n_points)

        h -= learning_rate * gradient
