import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if hasattr(globals(), 'initial_h_values'):
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.sin(np.pi * (x - 1)) * 0.5 + 0.5
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points-1:2*n_points-1]
        return np.max(valid_corr) * dx

    result = minimize(objective, h, method='Nelder-Mead', 
                      options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 150})

    best_h = result.x.copy()
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if best_sum != 0:
        best_h = best_h * target_sum / best_sum

    start_time = time.time()
    for _ in range(10):

            break

        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = perturbed * target_sum / perturbed_sum
        corr_p = np.correlate(perturbed, 1 - perturbed, mode='full')
        valid_corr_p = corr_p[n_points-1:2*n_points-1]
        curr_c5 = np.max(valid_corr_p) * dx
        if curr_c5 < objective(best_h):
            best_h = perturbed.copy()

    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if best_sum != 0:
        best_h = best_h * target_sum / best_sum
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    valid_corr_final = corr_final[n_points-1:2*n_points-1]
    final_c5 = np.max(valid_corr_final) * dx
