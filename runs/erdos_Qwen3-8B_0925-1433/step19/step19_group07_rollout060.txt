import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    T = n_points / 2.0
    dx = 2.0 / n_points

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        return h * T / np.sum(h) if np.sum(h) > 0 else h

    # Initialize population
    pop_size = 10
    particles = []
    for _ in range(pop_size):
        perturb = np.random.normal(0, 0.05, n_points)
        h = np.array(initial_h, dtype=float) + perturb
        h = clip_rescale(h)
        particles.append({'pos': h, 'vel': np.random.normal(0, 0.1, n_points), 'pbest': h, 'pbest_c5': float('inf')})

    # Global best
    global_best = {'pos': np.array(initial_h, dtype=float), 'c5': float('inf')}
    for p in particles:
        corr = np.correlate(p['pos'], 1 - p['pos'], mode='full')
        p['pbest_c5'] = np.max(corr) * dx
        if p['pbest_c5'] < global_best['c5']:
            global_best['pos'] = p['pos'].copy()
            global_best['c5'] = p['pbest_c5']

    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = int(budget_s / 0.1)
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Evaluate
        for p in particles:
            h = p['pos']
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < p['pbest_c5']:
                p['pbest'] = h.copy()
                p['pbest_c5'] = c5
                if c5 < global_best['c5']:
                    global_best['pos'] = p['pbest'].copy()
                    global_best['c5'] = c5

        # Update velocities and positions
        for p in particles:
            r1, r2 = np.random.rand(), np.random.rand()
            vel = omega * p['vel'] + c1 * r1 * (p['pbest'] - p['pos']) + c2 * r2 * (global_best['pos'] - p['pos'])
            p['vel'] = vel
            p['pos'] += vel
            p['pos'] = clip_rescale(p['pos'])

    best_h = global_best['pos']
    best_c5 = global_best['c5']
    return (np.clip(best_h, 0, 1), best_c5, n_points)
