import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    P = 12
    max_iter = 5000
    lr = 0.05
    decay_rate = 0.9995
    min_lr = 1e-4
    best_c5 = 1.0
    best_h = np.zeros(n_points)

    if 'initial_h_values' in globals():
        h_pop = [np.array(initial_h_values, dtype=float) for _ in range(P)]
        for i in range(P):
            h_pop[i] = h_pop[i] * target_sum / np.sum(h_pop[i])
    else:
        h_pop = []
        for _ in range(P):
            h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
            h = h * target_sum / np.sum(h)
            h_pop.append(h.copy())

    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(P):
            h = h_pop[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            lag = max_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for j in range(n_points):
                if 0 <= j + lag < n_points:
                    grad[j] += (1 - h[j + lag]) * dx
                if 0 <= j - lag < n_points:
                    grad[j] -= h[j - lag] * dx
            h -= lr * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            h_pop[i] = h
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        lr *= decay_rate
        if lr < min_lr:
            lr = min_lr

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
