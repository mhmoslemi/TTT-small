import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Adam optimization parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    learning_rate = 0.05

    start_time = time.time()
    max_iterations = 1000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation to find maximum lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        # Compute gradient for current lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        bias_m = m / (1 - beta1 ** t)
        bias_v = v / (1 - beta2 ** t)
        step = learning_rate * bias_m / (np.sqrt(bias_v) + epsilon)
        h = h - step
        # Clip and normalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(h) + target_sum / n_points
        # Update best solution
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
