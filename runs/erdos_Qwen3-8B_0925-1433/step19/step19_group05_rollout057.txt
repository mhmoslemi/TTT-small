from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[-3:]) * dx
    best_c5 = min(best_c5, 0.38085)

    # Objective function for optimization
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        # Valid lags correspond to k=0,1,2, indices: n_points-1, n_points, n_points+1
        corr_valid = corr[-3:]
        return np.max(corr_valid) * dx

    # Optimization with Nelder-Mead
    options = {'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 500}
    result = minimize(
        objective, x0=h, method='Nelder-Mead',
        bounds=[(0, 1)] * n_points, tol=1e-5, options=options
    )
    h_opt = result.x
    best_h = np.clip(h_opt, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
