from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.07
    gamma = 0.95
    decay_rate = 0.9995
    max_iterations = 300
    start_time = time.time()

    momentum = 0.0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        momentum = gamma * momentum + (1 - gamma) * gradient

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate
