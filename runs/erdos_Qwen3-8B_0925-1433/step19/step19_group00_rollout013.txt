import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 300
    lr_0 = 0.05
    alpha = 0.01
    m = 20
    T = 100
    lr_decay = 0.999

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < 1e-7:
            break

        top_indices = np.argsort(corr)[-m:]
        grad = np.zeros(n_points)

        for idx in top_indices:
            corr_val = corr[idx]
            weight = np.exp(-alpha * (current_max - corr_val))
            weight = np.clip(weight, 0.1, 10.0)
            k = (idx - (n_points - 1))
            k_adj = k % n_points
            shift = np.arange(n_points) + k_adj
            grad += weight * (1 - h[np.abs(shift) % n_points])

        lr = lr_0 * (1 + np.cos(t / T)) / 2
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.min(np.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= lr_decay

    return (np.clip(best_h, 0, 1), best_c5, n_points)
