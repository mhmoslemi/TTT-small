    max_iter = 200
    lr = 0.05
    top_lags = 5
    momentum = 0.9
    decay = 0.999

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags_indices = np.argsort(corr)[::-1][:top_lags]
        grad = np.zeros(n_points)
        for m in top_lags_indices:
            k = m - (n_points - 1)
            for j in range(n_points):
                j_plus_k = (j + k) % n_points
                j_minus_k = (j - k) % n_points
                grad[j] += (1 - h[j_plus_k] - h[j_minus_k]) * dx
        lr = lr * decay
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
