```python
import numpy as np
import math
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = corr.max() * dx
    best_h = h.copy()
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 1000
    momentum = 0.9
    learning_rate = 0.05
    velocity = np.zeros(n_points)
    no_improvement_count = 0
    improvement_threshold = 1e-5
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        curr_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(curr_corr)
        lag = max_idx - (n_points - 1)
        shifted = (np.arange(n_points) + lag) % n_points
        grad = 1 - h[shifted]
        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = curr_corr.max() * dx
        if current_c5 < best_c5 - improvement_threshold:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count > 50:
            learning_rate *= 0.5
            no_improvement_count = 0
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
```
