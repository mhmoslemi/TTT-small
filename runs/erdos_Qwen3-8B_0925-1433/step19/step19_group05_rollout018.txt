import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    start_time = time.time()
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr) * 2.0 / n_points
        return max_val
    
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(ind) for ind in population]
        top_idx = np.argsort(fitness)[:int(population_size * 0.2)]
        parents = [population[i] for i in top_idx]
        new_pop = []
        for _ in range(population_size):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                mutated = parents[np.random.choice(len(parents))] + noise
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * target_sum / np.sum(mutated)
                new_pop.append(mutated)
            else:
                p1, p2 = np.random.choice(len(parents), 2, replace=False)
                child = (parents[p1] + parents[p2]) / 2
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_pop.append(child)
        population = new_pop
    
    best_h = min(population, key=evaluate_c5)
    def simplex_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * 2.0 / n_points
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(simplex_obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h = result.x

        if time.time() - start_time > budget_s:
            break

            perturbed = best_h + perturb
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            if evaluate_c5(perturbed) < evaluate_c5(best_h):
                best_h = perturbed
        except:
            pass
    
    best_c5 = evaluate_c5(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
