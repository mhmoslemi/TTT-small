import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 500
    tournament_size = 5
    mutation_rate = 0.15
    start_time = time.time()

    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h.copy())
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([compute_c5(h) for h in population])
        # Select parents

            best_idx = indices[np.argmin(fitness[indices])]
            return population[best_idx]
        parent1 = select_parent()
        parent2 = select_parent()

        if np.random.rand() < 0.8:
            child = (parent1 + parent2) / 2.0

            child = parent1.copy()
