import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # PSO parameters
    pop_size = 10
    particles = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(pop_size)]
    velocities = [np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    personal_bests = particles.copy()
    global_best = particles[0]
    global_c5 = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    best_c5 = min(global_c5)
    
    max_iter = int(budget_s / 10)
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    decay_rate = 0.995
    perturb_freq = 50
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update particles
        for i in range(pop_size):
            # Evaluate current particle
            current = particles[i]
            current_c5 = np.max(np.correlate(current, 1 - current, mode='full')) * dx
            # Update personal best
            if current_c5 < global_c5[i]:
                personal_bests[i] = current.copy()
                global_c5[i] = current_c5
            # Update velocity and position
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (personal_bests[i] - current) + 
                            c2 * r2 * (global_best - current))
            particles[i] += velocities[i]
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        # Update global best
        current_global = np.min(global_c5)
        if current_global < best_c5:
            best_c5 = current_global
            best_h = np.array([p for p, c in zip(particles, global_c5) if c == current_global][0])
        # Decay inertia
        inertia *= decay_rate
        # Perturb with noise
