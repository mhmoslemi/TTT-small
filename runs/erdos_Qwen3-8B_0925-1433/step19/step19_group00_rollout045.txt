import numpy as np
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iter = 1000
    lr = 0.3
    decay = 0.99
    alpha = 0.05
    m = 30
    beta = 0.9
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.argsort(corr)[-m:]
        grad = np.zeros(n_points)
        weight = np.zeros(m)
        
        for i, k in enumerate(lags):
            k_adj = k % n_points
            pos = (np.arange(n_points) + k_adj) % n_points
            neg = (np.arange(n_points) - k_adj) % n_points
            term = (1 - h[pos]) - h[neg]
            grad += term
            weight[i] = corr[k] - np.min(corr)
