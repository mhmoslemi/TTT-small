import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    T = 1.0

    perturb_scale = 0.1

    max_iter = 2000
    start_time = time.time()
    stagnation = 0

    best_h = np.zeros(n_points)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 100:
            break
        perturb = np.random.normal(0, perturb_scale, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)

        delta = new_c5 - curr_c5
