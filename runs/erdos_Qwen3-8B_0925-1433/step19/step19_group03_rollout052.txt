import numpy as np
import math
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 1000
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    m_grad = np.zeros(n_points)
    v_grad = np.zeros(n_points)
    learning_rate = 0.01
    t = 0
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        curr_corr = np.correlate(h, 1 - h, mode='full')
        top_lags_indices = np.argsort(curr_corr)[::-1][:3]
        top_lags = [idx - (n_points - 1) for idx in top_lags_indices]
        
        grad = np.zeros(n_points)
        for lag in top_lags:
            shifted = (np.arange(n_points) + lag) % n_points
            shifted_neg = (np.arange(n_points) - lag) % n_points
            grad += (1 - h[shifted]) - h[shifted_neg]
        grad /= len(top_lags)
        
        m_grad = beta1 * m_grad + (1 - beta1) * grad
        v_grad = beta2 * v_grad + (1 - beta2) * (grad ** 2)
        correct_m = m_grad / (1 - beta1**(t + 1))
        correct_v = v_grad / (1 - beta2**(t + 1))
        h -= learning_rate * correct_m / (np.sqrt(correct_v) + epsilon)
        
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        t += 1
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
