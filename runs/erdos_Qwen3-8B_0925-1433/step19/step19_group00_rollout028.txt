import time

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 500
    learning_rate = 0.4
    decay_rate = 0.999
    alpha = 0.01
    m = 30
    C2 = 0.4

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[::-1][:m]
        grad = np.zeros(n_points)
        for lag in top_lags:
            if abs(corr[lag]) < 1e-6:
                continue
            weight = np.exp(-alpha * (corr[lag] - C2))
            weight = np.clip(weight, 0.1, 10.0)
            for i in range(n_points):
                k = (i + lag) % n_points
                grad[i] += weight * (1 - h[k])
        h -= learning_rate * grad

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
