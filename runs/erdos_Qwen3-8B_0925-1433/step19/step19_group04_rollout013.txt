from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with noise
    pop_size = 20
    particles = np.array([initial_h_values + np.random.normal(0, 0.01, n_points) for _ in range(pop_size)])
    particles = np.clip(particles, 0, 1)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)

    # Evaluate initial fitness
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    fitness = np.array([eval_c5(p) for p in particles])
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    gbest = particles[np.argmin(fitness)]
    gbest_fitness = np.min(fitness)
    best_h = gbest.copy()
    best_c5 = gbest_fitness

    max_vel = 0.05
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    max_iter = 300
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        velocities = np.random.rand(pop_size, n_points) * max_vel
        for i in range(pop_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] +
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
        particles += velocities
        particles = np.clip(particles, 0, 1)
        particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)

        # Evaluate new fitness
        new_fitness = np.array([eval_c5(p) for p in particles])
        for i in range(pop_size):
            if new_fitness[i] < pbest_fitness[i]:

                pbest_fitness[i] = new_fitness[i]
