import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 1000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)
    t = 0
    lr = 0.05

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        t += 1

        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        indices = (np.arange(n_points) + lag) % n_points
        grad = 1.0 - h[indices]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)

        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
