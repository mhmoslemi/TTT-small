import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_values = initial_h_values
    n_points = len(initial_values) if initial_values is not None else 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_values, dtype=float) if initial_values is not None else np.random.rand(n_points)

        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[n_points-1:n_points+2]

        return (h * target_sum / sum_h).flatten()
    
    h_initial = rescale_h(h)
    start_time = time.time()
    
    result = minimize(
        objective,
        h_initial,
        method='Nelder-Mead',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 300}

    best_c5 = objective(best_h)
