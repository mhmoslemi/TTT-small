import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iter = 300
    lr = 0.2
    alpha = 0.01
    eta = 2.0
    m = 16
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.argsort(corr)[::-1][:m]
        grad = np.zeros(n_points)
        for k in lags:
            k_adj = k % n_points
            weight = eta * np.exp(-alpha * np.abs(corr[k] - best_c5))
            grad += weight * (1 - h[(np.arange(n_points) + k_adj) % n_points])
        h -= lr * grad

        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        if np.linalg.norm(grad) < 1e-4:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
