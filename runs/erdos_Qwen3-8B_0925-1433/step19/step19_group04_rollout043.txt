import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    T = 1.0
    cooling_rate = 0.995
    perturb_std = 0.05
    max_iterations = 1000
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, perturb_std, n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr_perturbed) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            h = h_perturbed.copy()
            best_h = h.copy()
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                best_c5 = new_c5
                h = h_perturbed.copy()
                best_h = h.copy()
        
        T *= cooling_rate
        if T < 0.5:
            perturb_std = 0.01
        elif T < 0.1:
            perturb_std = 0.005
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
