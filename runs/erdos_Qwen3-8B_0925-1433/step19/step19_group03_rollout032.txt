import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 1000
    lr = 0.5
    beta = 0.9
    velocity = np.zeros(n_points)
    improve_count = 0
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        curr_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(curr_corr)
        max_idx = np.argmax(curr_corr)
        lag = max_idx - (n_points - 1)
        
        # Compute gradient for current lag
        shifted = (np.arange(n_points) + lag) % n_points
        gradient = 1 - h[shifted]
        
        # Update velocity and h
        velocity = beta * velocity + gradient
        h = h - lr * velocity
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
            improve_count = 0
        else:
            improve_count += 1
        
        if improve_count >= 100:
            break
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
