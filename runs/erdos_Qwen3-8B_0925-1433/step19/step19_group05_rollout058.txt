from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h, best_c5 = h.copy(), get_c5(h)

    def evolve(pop):
        return [np.clip((p1 + p2) * 0.5 + np.random.normal(0, 0.02, n_points), 0, 1) * target_sum / np.sum(...) for p1, p2 in zip(pop, pop)]

    for _ in range(150):
        if time.time() > budget_s: break
        if get_c5(evolve([best_h] * 50)) < best_c5:
            best_h, best_c5 = evolve([best_h] * 50)[0], get_c5(best_h)

    def simplex(x):
        x = x * target_sum / np.sum(x)
        x = np.clip(x, 0, 1)
        return get_c5(x)

    res = minimize(simplex, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'ftol': 1e-7, 'maxiter': 100})
