        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        learning_rate *= decay_rate
        
        if _ % perturb_freq == 0:
            perturb = np.random.normal(0, perturb_std, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
            current_c5_perturb = np.max(corr_perturbed) * dx
            if current_c5_perturb < best_c5:
                best_h = perturbed.copy()
                best_c5 = current_c5_perturb
                stagnation_count = 0
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
