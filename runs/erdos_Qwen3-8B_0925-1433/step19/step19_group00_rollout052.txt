import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iter = 200
    lr_0 = 0.5
    decay_rate = 0.999
    m = 20
    alpha = 0.01
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        best_c5 = min(best_c5, current_max)
        lags = np.argsort(corr)[::-1][:m]
        weights = np.exp(-alpha * (current_max - corr[lags]))
        grad = np.zeros(n_points)
        for lag_index, lag in enumerate(lags):
            k_adj = lag % n_points
            contributions = 1 - h[(np.arange(n_points) + k_adj) % n_points]
            grad += weights[lag_index] * contributions
        lr = lr_0 * (1 + np.cos(t / max_iter)) / 2
        h -= lr * grad

        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-4:
            break
        lr_0 *= decay_rate
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
