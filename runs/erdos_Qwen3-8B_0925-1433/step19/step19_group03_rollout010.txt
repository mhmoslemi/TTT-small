        grad = (1 - h[i_plus_k]) - h[i_minus_k]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
