import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(45)
        h[:9] = 1.0
        h[9:18] = 0.5
        h[18:36] = 1.0
        h[36:] = 0.5
        h = h * (45 / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 10000
    lr_0 = 0.05
    T = 100
    lr_decay = 0.999
    min_lr = 0.0001
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        max_lag = max_corr_index - (n_points - 1)
        if max_lag < 0:
            max_lag += n_points
        grad = 1 - np.roll(h, max_lag)
        h -= lr_0 * grad * (1 + np.cos(t / T)) / 2

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= lr_decay
        if lr_0 < min_lr:
            lr_0 = min_lr
