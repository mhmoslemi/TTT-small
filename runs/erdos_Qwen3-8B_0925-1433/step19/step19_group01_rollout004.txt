import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 128

    h = h * target_sum / np.sum(h)

    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iterations = 5000
    m = 20
    lr = 0.01
    beta = 0.999
    decay = 0.9995
    min_lr = 0.0001

    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[-m:]
        true_lags = [(i - (n_points - 1)) for i in top_indices]
        grad = np.zeros(n_points)
        for k in true_lags:
            if k < 0:
                k += n_points
            grad += 1 - np.roll(h, k)
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        lr *= decay
        if lr < min_lr:
            lr = min_lr
        if np.linalg.norm(grad) < 1e-4:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
