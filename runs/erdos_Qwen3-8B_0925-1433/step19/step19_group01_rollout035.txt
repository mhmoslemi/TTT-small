import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(64)
        h[:16] = 1.0
        h[16:48] = 0.5
        h[48:] = 1.0
        h = h * (64 / 2) / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 10000
    m = 6
    lr_0 = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m_t = 0.0
    v_t = 0.0
    m_is = np.zeros(n_points)
    v_is = np.zeros(n_points)
    decay = 0.999
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:m]
        true_lags = [(i - (n_points - 1)) for i in top_indices]
        grad = np.zeros(n_points)
        for k in true_lags:
            if k < 0:
                k += n_points
            grad += 1 - np.roll(h, k)
        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad ** 2)
        m_hat = m_t / (1 - beta1 ** (t + 1))
        v_hat = v_t / (1 - beta2 ** (t + 1))
        h -= lr_0 * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
