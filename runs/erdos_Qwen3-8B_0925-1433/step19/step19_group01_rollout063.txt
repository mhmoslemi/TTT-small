            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= lr_decay
        if lr_0 < min_lr:
            lr_0 = min_lr
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
