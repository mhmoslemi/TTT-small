import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(64)
        h[:32] = 1.0
        h[32:] = 0.5
        h = h * (64 / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 10000
    m = 8
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    
    # Adam variables
    m_ = np.zeros(n_points)
    v_ = np.zeros(n_points)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:m]
        true_lags = [(i - (n_points - 1)) for i in top_indices]
        
        grad = np.zeros(n_points)
        for k in true_lags:
            if k < 0:
                k += n_points
            grad += 1 - np.roll(h, k)
        
        t += 1
        alpha = 0.01 * (1 - beta1**t) / (np.sqrt(1 - beta2**t) + epsilon)
        
        m_ = beta1 * m_ + (1 - beta1) * grad
        v_ = beta2 * v_ + (1 - beta2) * grad * grad
        
        h -= alpha * m_ / (np.sqrt(v_) + epsilon)
        
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        
    return (np.clip(best_h, 0, 1), best_c5, n_points)
