        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 128
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h *= target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 5000
    m = 20
    lr_0 = 0.05
    T = 100
    lr_decay = 0.999
    min_lr = 0.0001
    perturb_freq = 500
    perturb_noise = 0.01
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:m]
        true_lags = [(i - (n_points - 1)) for i in top_indices]
        grad = np.zeros(n_points)
        for k in true_lags:
            rolled = np.roll(h, k)
            grad += 1 - rolled
        h -= lr_0 * grad * (1 + np.cos(t / T)) / 2
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= lr_decay
        if lr_0 < min_lr:
            lr_0 = min_lr
        if t % perturb_freq == 0 and t > 0:
            noise = np.random.normal(0, perturb_noise, h.shape)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
