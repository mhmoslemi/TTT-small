import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 20
    particles = np.array([np.clip(initial_h_values + np.random.normal(0, 0.01, n_points), 0, 1) * target_sum / np.sum(np.clip(initial_h_values + np.random.normal(0, 0.01, n_points), 0, 1)) for _ in range(population_size)])
    velocities = np.zeros((population_size, n_points))
    pbest = particles.copy()
    pbest_fitness = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    best_h = gbest

    max_vel = 0.1
    inertia = 0.8
    c1 = 2.0
    c2 = 2.0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            current_fitness = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
