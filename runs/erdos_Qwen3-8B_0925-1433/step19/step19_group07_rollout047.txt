import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = current_c5

    T = 1.0
    alpha = 0.995
    min_temp = 1e-6
    perturbation_scale = 0.1
    perturbation_decay = 0.999
    start_time = time.time()

    while time.time() - start_time < budget_s:
        if T < min_temp:
            break
        # Generate perturbed h
        perturb = np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        # Compute new C5
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr_perturbed) * dx
        delta = new_c5 - current_c5
        # Acceptance criteria
        if new_c5 < current_c5:
            h = h_perturbed.copy()
            current_c5 = new_c5
        else:
            if np.random.rand() < np.exp(-delta / T):
                h = h_perturbed.copy()
                current_c5 = new_c5
        # Update best_c5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Update temperature and perturbation
        T *= alpha
        perturbation_scale *= perturbation_decay

    return (np.clip(best_h, 0, 1), best_c5, n_points)
