import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial solution
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.array([np.correlate(h, 1 - h, mode='full')[n_points - 1 + k] * dx for k in range(3)]))
    best_c5 = min(best_c5, 0.381)

    # PSO parameters
    P = 50
    particles = np.random.normal(0, 0.04, (P, n_points))
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.uniform(-0.03, 0.03, (P, n_points))
    pbest = particles.copy()
    pbest_c5 = np.full(P, np.inf)
    gbest = particles[0].copy()
    gbest_c5 = np.inf

    # PSO loop
    for _ in range(250):
        if time.time() > budget_s:
            break
        for i in range(P):
            # Evaluate
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max([corr[n_points - 1 + k] * dx for k in range(3)])
            pbest_c5[i] = c5
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
            # Update global best
            if c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = c5
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        
        # Check if best solution improves
        if gbest_c5 < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_c5

    # Nelder-Mead refinement
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max([corr[n_points - 1 + k] * dx for k in range(3)])

    result = minimize(objective, best_h, method='Nelder-Mead', bounds=[(0, 1)]*n_points, options={'maxiter': 50})
