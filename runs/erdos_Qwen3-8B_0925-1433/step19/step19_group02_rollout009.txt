import time

def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 1000
    T = max_iter
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)
    learning_rate = 0.01

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        h_rotated = np.roll(h, shift=-k)
        h_shifted_neg = np.roll(h, shift=k)
        g = 1 - h_rotated - h_shifted_neg
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** (t + 1))
