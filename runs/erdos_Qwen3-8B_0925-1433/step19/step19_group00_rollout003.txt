import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    pop_size = 40
    particles = [h.copy() for _ in range(pop_size)]
    pbest = [p.copy() for p in particles]
    gbest = h.copy()
    velocities = [np.zeros(n_points) for _ in range(pop_size)]
    
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            p = particles[i]
            pcorr = np.correlate(p, 1 - p, mode='full') * dx
            pcorr_max = pcorr.max()
            if pcorr_max < best_c5:
                best_c5 = pcorr_max
                best_h = p.copy()
            if pcorr_max < np.max(np.correlate(pbest[i], 1 - pbest[i], mode='full')) * dx:
                pbest[i] = p.copy()
            if pcorr_max < np.max(np.correlate(gbest, 1 - gbest, mode='full')) * dx:
                gbest = p.copy()
        
        for i in range(pop_size):
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            vel += 0.9 * vel + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            vel = np.clip(vel, -0.5, 0.5)
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            velocities[i] = vel
    
    h = np.clip(gbest, 0, 1)
    h = h * target_sum / np.sum(h)
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
