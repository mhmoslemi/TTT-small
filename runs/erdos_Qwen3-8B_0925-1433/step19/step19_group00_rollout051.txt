import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    h_pop = np.tile(initial_h[np.newaxis, :], (40, 1))
    velocities = np.zeros((40, n_points))
    pbest = h_pop.copy()
    pbest_values = np.full(40, np.inf)
    gbest = initial_h
    gbest_value = np.inf
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    max_iter = 400
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(40):
            h = h_pop[i]
            current_val = evaluate(h)
            if current_val < pbest_values[i]:
                pbest[i] = h.copy()
                pbest_values[i] = current_val
            if current_val < gbest_value:
                gbest = h.copy()
                gbest_value = current_val
        
        for i in range(40):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.9 * velocities[i] + 1.5 * r1 * (pbest[i] - h_pop[i]) + 1.5 * r2 * (gbest - h_pop[i])
            h_pop[i] += velocities[i]

            h_pop[i] = h_pop[i] * target_sum / np.sum(h_pop[i])
    
    h = gbest
    c5_bound = evaluate(h)
    return (np.clip(h, 0, 1), c5_bound, n_points)
