from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.normal(0, 0.05, (P, n_points))

    for _ in range(PSO_epochs):
        if time.time() - time.time() > budget_s:
            break
        for j in range(P):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[j] = inertia * velocities[j] + c1 * r1 * (best_h - particles[j]) + c2 * r2 * (best_h - particles[j])

        best_idx, best_c5_ = np.argmin(c5s), c5s[np.argmin(c5s)]
