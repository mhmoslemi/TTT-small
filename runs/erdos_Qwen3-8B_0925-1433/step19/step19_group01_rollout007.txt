    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 128
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iterations = 10000
    lr_start = 0.05
    lr_decay = 0.999
    min_lr = 0.0001
    for it in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        true_lag = max_corr_idx - (n_points - 1)
        if true_lag < 0:
            true_lag += n_points
        grad = 1 - np.roll(h, true_lag)
        lr = lr_start * (lr_decay ** it)
        h -= lr * grad * (1 + np.cos(it / 1000)) / 2
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3 or lr < min_lr:
            break
    return (np.clip(best_h, 0, 1), best_c5, n_points)
