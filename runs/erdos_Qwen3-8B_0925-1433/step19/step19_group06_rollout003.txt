import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()

    temp = 2.0
    cooling_rate = 0.995
    max_iter = 15000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        sigma = max(0.03 * temp, 0.003)
        perturb = np.random.normal(0, sigma, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr_perturbed) * dx

        if new_c5 < best_c5:
            h = h_perturbed.copy()
            best_c5 = new_c5
