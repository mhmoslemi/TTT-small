import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)

        gradient = np.zeros(n_points)
        for j in range(n_points):
            i_plus_k = (j + k) % n_points

        current_c5 = np.max(corr_current) * dx
        
        if current_c5 < best_c5:
