import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial guesses
    perturbations = []
    for _ in range(8):

        elif np.random.rand() < 0.3:
            pert = 1 - initial_h_values
        elif np.random.rand() < 0.3:
            pert = np.random.permutation(initial_h_values)
        else:
            pert = np.where(np.random.rand(n_points) < 0.3, 0, initial_h_values)
        pert = np.clip(pert, 0, 1)
        pert = pert * target_sum / np.sum(pert)
        perturbations.append(pert)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    max_iter = 1500
    lr0 = 0.07
    top_lags = 10
    for pert in perturbations:
        h = pert.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()

            if time.time() - start_time > budget_s:

            top_lags_indices = np.argsort(corr)[::-1][:top_lags]
            grad = np.zeros(n_points)
            for m in top_lags_indices:
                k = m - (n_points - 1)
                indices = (np.arange(n_points) + k) % n_points
                grad += 1 - h[indices]
            lr = lr0 * (1 + np.cos(t / max_iter)) / 2
            h = h - lr * grad

            h = h * target_sum / np.sum(h)
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
