import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    T = 1.0
    cooling_rate = 0.995
    max_iterations = 600
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)

        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5
            h = perturbed.copy()
        else:
            delta = current_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = perturbed.copy()

        T *= cooling_rate

    return (np.clip(best_h, 0, 1), best_c5, n_points)
