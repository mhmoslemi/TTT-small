import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = np.max(corr) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.1
    momentum = 0.9
    decay = 0.999
    perturb_scale = 0.05
    max_iter = int(budget_s / 10)
    perturb_freq = 20
    stagnation = 0
    velocity = np.zeros(n_points)

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grads = np.zeros(n_points)

        if current_c5 < best_c5:
            best_h = h.copy()
