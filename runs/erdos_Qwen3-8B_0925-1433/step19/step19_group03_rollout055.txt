                y = (i - lag) % n_points
                grad[i] = (1 - h[x]) - h[y]
            for _ in range(local_ref_steps):
                h = h - lr * grad
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2 / n_points)
            if c5 < c5_list[idx]:
                population[idx] = h.copy()
        
        population = new_population
    
    # Final evaluation
    best_c5 = float('inf')
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
