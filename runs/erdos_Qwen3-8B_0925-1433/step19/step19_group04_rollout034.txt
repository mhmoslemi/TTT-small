import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    start_time = time.time()

        if time.time() - start_time > budget_s:
            break

        perturbed = perturbed * target_sum / np.sum(perturbed)
