        i2 = (np.arange(n_points) + k_adj) % n_points
        grad = h[i1] + h[i2]
        # Update h
        h += learning_rate * grad
        # Clip and rescale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        else:
            h = np.full(n_points, 0.5)
        # Evaluate current C5
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Check convergence
        if np.abs(current_c5 - best_c5) < convergence_threshold:
            break
        # Adjust learning rate
        learning_rate *= decay_rate

    return (np.clip(best_h, 0, 1), best_c5, n_points)
