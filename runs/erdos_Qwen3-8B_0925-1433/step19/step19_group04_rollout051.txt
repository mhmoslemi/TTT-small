    cooling_rate = 0.997
    max_iter = 300
    perturb_std = 0.07
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, perturb_std, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        
        corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = current_c5
        else:
            delta = current_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                best_h = h_perturbed.copy()
                best_c5 = current_c5
        
        temp *= cooling_rate
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
