import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    T = 10.0
    cooling_rate = 0.996
    max_iter = 500
    start = time.time()
    
    for _ in range(max_iter):
        if time.time() - start > budget_s:
            break
        noise = np.random.normal(0, 0.015, n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        curr_c5 = np.max(corr) * dx
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = perturbed.copy()
            h = perturbed.copy()
        else:
            delta = curr_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = perturbed.copy()
        T *= cooling_rate
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
