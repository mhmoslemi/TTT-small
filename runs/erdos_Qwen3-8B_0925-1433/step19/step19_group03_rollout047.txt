import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    population_size = 100
    generations = 150
    mut_rate = 0.1
    mut_std = 0.05
    elite_ratio = 0.1
    refine_count = 5
    refine_steps = 10

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    def make_perturbation(h):
        noise = np.random.normal(0, mut_std, h.shape)
        h = np.clip(h + noise, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    population = [make_perturbation(np.array(initial_h_values, dtype=float)) for _ in range(population_size)]
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(h) for h in population])
        elite_indices = np.argsort(fitness)[:int(population_size * elite_ratio)]
        elites = [population[i] for i in elite_indices]

        new_pop = elites.copy()
        while len(new_pop) < population_size:
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (p1 + p2) * 0.5
            mutate_idx = np.random.choice(n_points, int(n_points * mut_rate), replace=False)
            child[mutate_idx] += np.random.normal(0, mut_std, len(mutate_idx))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)

        population = new_pop

        # Refine top candidates
        top_indices = np.argsort(fitness)[:refine_count]
        for idx in top_indices:
            h = population[idx].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag = np.argmax(corr)
            lag = max_lag - (n_points - 1)
            shifted = (np.arange(n_points) + lag) % n_points
            grad = 1 - h[shifted]
            h = h - 0.05 * grad
            h = np.clip(h, 0, 1)
