        best_c5 = max(best_c5, np.max(corr) * dx)
    
    generations_run = 0
    while generations_run < max_gen and time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        
        # Track best in current population
        best_idx = np.argmin(fitness)
        current_best = population[best_idx].copy()
        current_best_c5 = fitness[best_idx]
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Tournament selection
        def select_parent():
            indices = np.random.choice(population_size, tournament_size, replace=False)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            return population[best_idx]
        
        parent1 = select_parent()
        parent2 = select_parent()
        
        # Crossover
        if np.random.rand() < crossover_rate:
            offspring = (parent1 + parent2) / 2.0
        else:
            offspring = parent1.copy()
        
        # Mutation
        mut = np.random.rand() < mutation_rate
        if mut:
            sigma = 0.05 * (1 - (generations_run + 1) / max_gen)
            sigma = max(sigma, 0.01)
            noise = np.random.normal(0, sigma, n_points)
            offspring += noise
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * target_sum / np.sum(offspring)
        
        # Replace worst individual
        worst_idx = np.argmax(fitness)
        population[worst_idx] = offspring.copy()
        generations_run += 1
    
    # Final evaluation
    best_c5 = 0.0
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = max(best_c5, np.max(corr) * dx)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
