def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.1
    decay_rate = 0.998
    max_iterations = 500
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            if i_plus_k < n_points:
                term1 = 1.0 - h[i_plus_k]
            else:
                term1 = 0.0
            i_minus_k = i - k
            if i_minus_k >= 0:
                term2 = h[i_minus_k]
            else:
                term2 = 0.0
            gradient[i] = term1 - term2
        h += learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    for _ in range(20):
        if time.time() - start_time > budget_s:
