    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    m = np.zeros(n_points)
    v = np.zeros(n_points)

    beta2_hat = 1.0
    t = 1
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]
