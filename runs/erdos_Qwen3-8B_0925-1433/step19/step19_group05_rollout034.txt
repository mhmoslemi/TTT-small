            child *= target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            new_pop.append(child)
        population = parents + new_pop
        scores = [compute_c5(h) for h in population]
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = scores[best_idx]

    # Simplex refinement
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h *= target_sum / np.sum(h)
        return compute_c5(h)

    bounds = [(0, 1)] * n_points
    result = minimize(objective, best_h, method='Nelder-Mead', bounds=bounds, options={'xatol': 1e-5, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(best_h)

    # Perturbation refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed *= target_sum / np.sum(perturbed)
        c5 = compute_c5(perturbed)
        if c5 < best_c5:
            best_h = perturbed
            best_c5 = c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
