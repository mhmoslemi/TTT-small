import numpy as np

def run(seed=42, budget_s=600):

    start_time = time.time()
    max_iterations = 10000
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m1 = np.zeros(n_points)
    m2 = np.zeros(n_points)
    decay = 0.999

    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = 1.0 - np.roll(h, shift=-lag)
