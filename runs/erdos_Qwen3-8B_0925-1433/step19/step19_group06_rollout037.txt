import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population = [h_initial]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 5
    sigma = 0.05
    start_time = time.time()
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population])
        new_pop = []
        for _ in range(population.shape[0]):
            a = np.random.choice(population_size, tournament_size, replace=False)
            idx_a = np.argmin(fitness[a])
            p1 = population[a[idx_a]]
            a = np.random.choice(population_size, tournament_size, replace=False)
            idx_b = np.argmin(fitness[a])
            p2 = population[a[idx_b]]
            if np.random.rand() < crossover_rate:
                child = (p1 + p2) / 2
            else:
                child = p1
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, sigma, n_points)
                child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = np.array(new_pop)
        current_fitness = fitness[np.argmin(fitness)]
        if current_fitness < best_c5:
            best_c5 = current_fitness
            best_h = population[np.argmin(fitness)]
        sigma = max(0.01, sigma * 0.998)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (np.clip(best_h, 0, 1), best_c5, n_points)
