import numpy as np

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 2000  # More iterations
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.05
    T = max_iter
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    adam_beta1 = beta1
    adam_beta2 = beta2

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        # Compute the lag k
        k = max_index - (n_points - 1)
        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            shifted_j_plus_k = (j + k) % n_points
            shifted_j_minus_k = (j - k) % n_points
            gradient[j] = (1 - h[shifted_j_plus_k]) - h[shifted_j_minus_k]
        # Adam update
        m = adam_beta1 * m + (1 - adam_beta1) * gradient
        v = adam_beta2 * v + (1 - adam_beta2) * gradient ** 2
        step = learning_rate * m / (np.sqrt(v) + epsilon)
        h = h - step
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iter = 2000
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.05
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            shifted_j_plus_k = (j + k) % n_points
            shifted_j_minus_k = (j - k) % n_points
            grad_val = (1 - h[shifted_j_plus_k]) - h[shifted_j_minus_k]
            gradient[j] = grad_val * dx
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * gradient**2
        step = learning_rate * m / (np.sqrt(v) + epsilon)
        h = h - step
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
