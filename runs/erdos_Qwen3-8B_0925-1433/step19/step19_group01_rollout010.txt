    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 64
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 10000
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)
    t = 0
    lr = 0.1
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = i + lag
            if 0 <= j_plus < n_points:
                grad[i] += 1 - h[j_plus]
            j_minus = i - lag
            if 0 <= j_minus < n_points:
                grad[i] -= h[j_minus]
        grad *= dx
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-5:
            break
        lr = 0.1 * (1 / (1 + 0.1 * t))
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
