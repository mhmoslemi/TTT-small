import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    M = 20
    max_vel = 0.05
    inertia = 0.8
    c1 = 2.0
    c2 = 2.0
    velocities = np.zeros((M, n_points))

    # Initialize population with initial_h_values + noise
    particles = np.tile(initial_h_values, (M, 1)) + np.random.normal(0, 0.01, (M, n_points))
    particles = np.clip(particles, 0, 1)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)

    # Fitness function
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize personal best and global best
    pbest = particles.copy()
    pbest_fitness = np.array([fitness(h) for h in particles])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        for i in range(M):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        # Evaluate new fitness
        new_fitness = np.array([fitness(h) for h in particles])
        for i in range(M):
            if new_fitness[i] < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness[i]
        # Update global best
        gbest = pbest[np.argmin(pbest_fitness)]
        gbest_fitness = np.min(pbest_fitness)

    # Validate and return best solution
    h_best = np.clip(gbest, 0, 1)
    assert np.all(h_best >= 0) and np.all(h_best <= 1)
    assert abs(np.sum(h_best) - target_sum) < 1e-10
    return (h_best, gbest_fitness, n_points)
