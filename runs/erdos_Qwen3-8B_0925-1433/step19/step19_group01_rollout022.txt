def run(seed=42, budget_s=600):

        h[32:64] = 1.0
        h[64:] = 0.5
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iter = 10000
    lr = 0.05
    decay = 0.9995
    min_lr = 0.0001
    m = 10
    noise_scale = 0.05
    pop_size = 8
    pop = [h.copy() for _ in range(pop_size)]
    
    def update_h(h):
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[::-1][:m]
        true_lags = [(i - (n_points - 1)) % n_points for i in top_lags]
        grad = np.zeros(n_points)
        for k in true_lags:
            grad += (1 - np.roll(h, k))
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            h = pop[i].copy()
            h = update_h(h)
            pop[i] = h
        corr_values = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in pop]
        best_idx = np.argmin(corr_values)
        best_c5 = corr_values[best_idx]
        best_h = pop[best_idx].copy()
        lr *= decay
        if lr < min_lr:
