    max_iterations = 5000

    lr = 0.02

    min_lr = 0.0001
    m1, m2 = np.zeros(n_points), np.zeros(n_points)
    t = 0

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:m]
        true_lags = [(i - (n_points - 1)) for i in top_indices]
        grad = np.zeros(n_points)

            if k < 0:
                k += n_points
            grad += 1 - np.roll(h, k)
        m1 = beta1 * m1 + (1 - beta1) * grad
