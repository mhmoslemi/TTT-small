import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def optimize_start(h_init):
        h = h_init.copy()
        h = h * target_sum / np.sum(h)
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.05
        start_time = time.time()
        for t in range(1000):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            top_lags = np.argsort(corr)[::-1][:10]
            grad = np.zeros(n_points)
            for m_val in top_lags:
                k = m_val - (n_points - 1)
                indices = (np.arange(n_points) + k) % n_points
                grad += 1 - h[indices]
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** (t + 1))
            v_hat = v / (1 - beta2 ** (t + 1))
            h -= lr * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                nonlocal best_c5, best_h
                best_c5 = current_c5
                best_h = h.copy()
            if current_c5 < 0.381:
                break
        return best_h

    perturbations = [initial_h]
    for _ in range(7):
        noise = np.random.normal(0, 0.05, n_points)
        p = initial_h + noise

        p = p * target_sum / np.sum(p)
        perturbations.append(p)
    
    for p in perturbations:
        optimize_start(p)
    
    best_h = np.clip(best_h, 0, 1)
