            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            if fitness < pbest_fitness:
                particles[i] = (h, velocity, h, fitness)
            else:
                particles[i] = (h, velocity, pbest, pbest_fitness)
        for p in particles:
            if p[3] < gbest_fitness:
                gbest = p[0]
                gbest_fitness = p[3]

    h_final = gbest
    final_corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5 = np.max(final_corr) * dx
    h_final = np.clip(h_final, 0, 1)
    return (h_final, c5, n_points)
