import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    
    pop_size = 10
    particles = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    velocities = [np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    personal_best = particles[:]
    personal_best_c5 = [np.max(np.correlate(p, 1-p, mode='full')) * (2.0 / n_points) for p in particles]
    
    global_best = min(particles, key=lambda p: np.max(np.correlate(p, 1-p, mode='full')) * (2.0 / n_points))
    global_best_c5 = np.max(np.correlate(global_best, 1-global_best, mode='full')) * (2.0 / n_points)
    
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = int(budget_s / 10)
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            h = particles[i]
            corr = np.correlate(h, 1-h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = h.copy()
                personal_best_c5[i] = current_c5
                if current_c5 < global_best_c5:
                    global_best = h.copy()
                    global_best_c5 = current_c5
        
        for i in range(len(particles)):
            h = particles[i]
            vel = velocities[i]
            pbest = personal_best[i]
            gbest = global_best
            r1, r2 = np.random.rand(2)
            vel_new = omega * vel + c1 * r1 * (pbest - h) + c2 * r2 * (gbest - h)
            h_new = h + vel_new
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            particles[i] = h_new
            velocities[i] = vel_new
        
        if iter >= max_iter - 200:
            indices = np.argsort(personal_best_c5)[:5]
            particles = [particles[i] for i in indices]
            velocities = [velocities[i] for i in indices]
            personal_best = [personal_best[i] for i in indices]
            personal_best_c5 = [personal_best_c5[i] for i in indices]
            omega = 0.4
            c1 = 2.0
            c2 = 2.0
    
    final_corr = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(final_corr) * (2.0 / n_points)
    return (global_best, final_c5, n_points)
