import time

def run(seed=42, budget_s=600):

    pbest = [p.copy() for p in particles]
    gbest = min(pbest, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
    best_c5 = np.max(np.correlate(gbest, 1 - gbest, mode='full')) * dx

                pbest[i] = particles[i].copy()
        gbest = min(pbest, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
