    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 8)
    learning_rate = 0.08
    decay_rate = 0.997
    perturb_std = 0.06
    perturb_freq = 10

    stagnation_count = 0
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)

        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:

            stagnation_count = 0
