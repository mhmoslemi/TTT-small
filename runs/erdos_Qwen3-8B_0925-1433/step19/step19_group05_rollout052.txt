        valid_corr = corr[valid_indices]
        return np.max(valid_corr) * dx
    result = minimize(objective, global_best, method='Nelder-Mead', bounds=[(0,1)]*n_points, options={'xatol': 1e-5, 'fatol': 1e-5})
    best_solution = result.x
    best_solution = np.clip(best_solution, 0, 1)
    best_solution = best_solution * (target_sum / np.sum(best_solution))
    best_c5 = np.max(np.correlate(best_solution, 1 - best_solution, mode='full')[np.array([n_points-1, n_points-2, n_points-3])] * dx
    return (np.clip(best_solution, 0, 1), best_c5, n_points)
