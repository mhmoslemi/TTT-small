import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    # PSO Parameters
    n_particles = 50
    inertia = 0.7
    c1, c2 = 1.5, 1.5
    max_pso_iterations = 250
    particles = np.array([h_initial + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)])
    particles = particles.clip(0, 1) * target_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))

    pbest = particles.copy()
    gbest = particles[0].copy()

    for _ in range(max_pso_iterations):
        if time.time() - start_time > budget_s:
            break
        r1, r2 = np.random.rand(*velocities.shape), np.random.rand(*velocities.shape)
        velocities = inertia * velocities + c1 * r1 * (pbest - particles) + c2 * r2 * (gbest - particles)
        particles += velocities
        particles = np.array([p.clip(0, 1) * target_sum / np.sum(p.clip(0, 1)) for p in particles])
        fitness = np.array([np.max(np.correlate(p, 1-p, mode='full')) * dx for p in particles])
        for i in range(n_particles):
            if fitness[i] < np.min(fitness):
                pbest[i] = particles[i].copy()
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = particles[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        gbest = particles[current_best_idx].copy()

    # Nelder-Mead Refinement
    def obj_func(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return np.max(np.correlate(h, 1-h, mode='full')) * dx

    result = minimize(obj_func, best_h, method='Nelder-Mead', options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 50})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = obj_func(best_h)

    return (best_h, best_c5, n_points)
