    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    elapsed = 0
    start = time.time()

    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.1
    t = 0
    m = np.zeros(n_points)
    v = np.zeros(n_points)

    while elapsed < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        m_idx = np.argmax(corr)
        k = m_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for x in range(n_points):
            next_x = (x + k) % n_points
            prev_x = (x - k) % n_points
            grad[x] = (1 - h[next_x] - h[prev_x]) * dx

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        elapsed = time.time() - start

    return (np.clip(best_h, 0, 1), best_c5, n_points)
