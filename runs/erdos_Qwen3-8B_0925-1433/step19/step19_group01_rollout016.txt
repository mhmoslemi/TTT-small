import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

        n_points = 64

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iterations = 10000
    m = 20  # increased number of lags considered
    lr_0 = 0.05
    T = 100
    lr_decay = 0.999
    min_lr = 0.0001
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:m]
        true_lags = [(i - (n_points - 1)) for i in top_indices]
        grad = np.zeros(n_points)
        for k in true_lags:
            if k < 0:
                k += n_points
            grad += 1 - np.roll(h, k)
        h -= lr_0 * grad * (1 + np.cos(t / T)) / 2
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= lr_decay
        if lr_0 < min_lr:
            lr_0 = min_lr
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
