import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    max_iter = 1000
    lr_0 = 0.05
    T = 100
    lr_decay = 0.999
    m = 1
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        k = max_lag
        grad = np.zeros(n_points)
        for j in range(n_points):
            a = (j + k) % n_points
            b = (j - k) % n_points
            grad[j] = (1.0 - h[a]) - h[b]
        lr = lr_0 * (1 + np.cos(t / T)) / 2
        h -= lr * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        else:
            h = np.full(h.shape, 0.5)
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= lr_decay
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
