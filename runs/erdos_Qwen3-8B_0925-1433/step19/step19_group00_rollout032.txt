        bias_correction1 = 1 / (1 - beta1**(t+1))
        bias_correction2 = 1 / (1 - beta2**(t+1))
        m1_hat = m1 * bias_correction1
        m2_hat = m2 * bias_correction2
        h -= learning_rate * m1_hat / (np.sqrt(m2_hat) + epsilon)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-4:
            break
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
