import numpy as np

def run(seed=42, budget_s=600):

    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    learning_rate = 0.1
    
    start_time = time.time()
    max_iter = 10000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        current_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(current_corr)
        lag = max_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
