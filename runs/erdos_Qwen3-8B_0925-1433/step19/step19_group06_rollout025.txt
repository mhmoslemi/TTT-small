    max_iterations = 250
    start_time = time.time()
    velocity = np.zeros(n_points)

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_indices = np.argsort(corr)[-5:]
        grad_sum = np.zeros(n_points)
        for idx in max_indices:
            k = idx - (n_points - 1)
            for i in range(n_points):
                i_plus_k = (i + k) % n_points
                i_minus_k = (i - k) % n_points
                grad_sum[i] += (1 - h[i_plus_k]) - h[i_minus_k]
        grad_avg = grad_sum / len(max_indices)
        velocity = momentum * velocity + grad_avg
        h -= learning_rate * velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr_perturbed) * dx
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
