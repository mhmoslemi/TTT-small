    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    max_iterations = 1000
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    t = 0
    m = np.zeros(n_points)
    v = np.zeros(n_points)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_indices = np.argsort(corr)[-3:]
        lag_values = [int(max_idx - (n_points - 1)) for max_idx in max_indices]
        gradient = np.zeros(n_points)
        for lag in lag_values:
            for i in range(n_points):
                idx_plus = (i + lag) % n_points
                idx_minus = (i - lag) % n_points
                gradient[i] += (1 - h[idx_plus]) - h[idx_minus]
        gradient /= len(lag_values)
        t += 1
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
