import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with noise
    M = 20
    population = np.array([np.array(initial_h_values) + np.random.normal(0, 0.01, n_points) for _ in range(M)])
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    # Evaluate initial fitness
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = fitness(best_h)
    for h in population[1:]:
        c5 = fitness(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # PSO parameters
    omega = 0.8
    c1, c2 = 2.0, 2.0
    max_vel = 0.05
    vel = np.zeros((M, n_points))
    pbest = population.copy()
    pbest_fitness = np.array([fitness(h) for h in pbest])
    gbest = pbest[np.argmin(pbest_fitness)].copy()
    gbest_fitness = np.min(pbest_fitness)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for i in range(M):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel[i] = omega * vel[i] + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            vel[i] = np.clip(vel[i], -max_vel, max_vel)
            # Update position
            population[i] += vel[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        # Evaluate new fitness
        for i in range(M):
            f = fitness(population[i])
            if f < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = f
        # Update global best
        if np.min(pbest_fitness) < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)].copy()
            gbest_fitness = np.min(pbest_fitness)
    return (np.clip(gbest, 0, 1), gbest_fitness, n_points)
