def run(seed=42, budget_s=600):

    best_c5 = np.max(corr_valid) * dx

    c1 = c2 = 1.5
    max_iter_pso = 250
    max_iter_nelder = 50
    start_time = time.time()

    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
