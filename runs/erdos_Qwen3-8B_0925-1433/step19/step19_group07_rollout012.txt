import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    T = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=np.float64)

    particles.append(h_initial)
    velocities = [0.05 * np.random.randn(n_points) for _ in range(population_size)]
    
    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    personal_best = [np.copy(p) for p in particles]
    personal_best_vals = [calc_c5(p) for p in personal_best]
    global_best = np.copy(personal_best[np.argmin(personal_best_vals)])
    global_best_val = min(personal_best_vals)
    
    omega = 0.8
    c1, c2 = 1.5, 1.5
    max_iter = 1000
    start_time = time.time()

        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            p = particles[i]
            curr = calc_c5(p)
            if curr < personal_best_vals[i]:
                personal_best[i] = np.copy(p)
                personal_best_vals[i] = curr
        curr_global = min(personal_best_vals)
        if curr_global < global_best_val:
            global_best_val = curr_global
            global_best = np.copy(personal_best[np.argmin(personal_best_vals)])
