from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Normalize initial values
    initial_normalized = np.clip(initial_h_values, 0, 1).copy()
    sum_initial = np.sum(initial_normalized)
    if sum_initial > 0:
        initial_normalized = initial_normalized * target_sum / sum_initial
    else:
        initial_normalized = np.zeros_like(initial_normalized)

    # Setup DE parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_gen = 150
    gen = 0
    time_start = time.time()
    population = [initial_normalized.copy() for _ in range(pop_size)]
    best_h = initial_normalized.copy()
    best_c5 = evaluate_c5(best_h)

    while gen < max_gen and time.time() - time_start < budget_s:
        for i in range(pop_size):

            if sum_trial > 0:
                trial_norm = trial_clipped * target_sum / sum_trial
            else:
                trial_norm = np.zeros_like(trial_clipped)
            c5_trial = evaluate_c5(trial_norm)

    # Local refinement with penalty method
