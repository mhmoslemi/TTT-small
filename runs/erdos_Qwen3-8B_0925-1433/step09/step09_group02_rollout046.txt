import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with given initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Use SLSQP with tighter tolerances for refinement
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Targeted perturbations: focus on worst lags
        h_ = np.clip(best_h, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_lag = np.argmax(corr)
        pt_indices = np.arange(n_points)
        if max_lag < n_points - 1:
            pt_indices = np.concatenate([np.arange(0, max_lag + 1), np.arange(n_points - max_lag, n_points)])
        else:
            pt_indices = np.arange(0, n_points)
        pt = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + pt[pt_indices], 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
