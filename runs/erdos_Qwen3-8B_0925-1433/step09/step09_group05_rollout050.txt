import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        initial_h = np.zeros(45)
        n = 45
        initial_h[::5] = 1.0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h /= np.sum(initial_h)
        initial_h *= n / 2
        initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Global optimization
    def objective(x):
        return eval_c5(x)

    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=30, maxiter=150, tol=1e-6,
                                   mutation=(0.8, 1.0), recombination=0.9,
                                   polish=True, disp=False)
    best_x = result.x
    best_c5 = eval_c5(best_x)

    # Local optimization
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(objective, best_x, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 100})
    final_h = result.x
    final_c5 = result.fun

    # Final refinement with perturbations
    for _ in range(10):
        if time.time() - time.time() >= budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        trial = np.clip(final_h + perturb, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * target_sum / trial_sum
        res = minimize(objective, trial, method='L-BFGS-B', bounds=bounds, constraints=cons,
                       tol=1e-8, options={'maxiter': 50})
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
