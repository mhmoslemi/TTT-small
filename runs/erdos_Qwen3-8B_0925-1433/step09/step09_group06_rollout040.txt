import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    initializations = []
    initializations.append(np.array(initial_h_values, dtype=float))
    random_h = np.random.uniform(0, 1, n_points)
    random_h = random_h * target_sum / np.sum(random_h)
    initializations.append(random_h)
    perturbed = np.array(initial_h_values, dtype=float) + np.random.uniform(-0.1, 0.1, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    initializations.append(perturbed)
    periodic = np.zeros(n_points)
    periodic[::5] = 1
    periodic = periodic * target_sum / np.sum(periodic)
    initializations.append(periodic)

    best_h = initializations[0].copy()
    best_c5 = compute_c5(best_h)
    best_c5_val = best_c5

    start_time = time.time()

    for init in initializations:
        h = init.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        learning_rate = 0.5
        t = 0
        max_iter = 1000

        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = corr.argmax()
            k_val = max_index - (n_points - 1)
            G = np.zeros(n_points)
            for i in range(n_points):
                j = (i - k_val) % n_points
                G[i] = (1 - h[j]) * dx
            t += 1
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * G**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = learning_rate * m_hat / (np.sqrt(v_hat) + eps)
            h = h - step
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5_val:
                best_c5_val = current_c5
                best_h = h.copy()

    return (best_h, best_c5_val, n_points)
