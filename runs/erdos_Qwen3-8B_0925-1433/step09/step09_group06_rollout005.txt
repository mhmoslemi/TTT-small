import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        return h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    # Initialize population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        h = np.array(initial_h_values, dtype=float) * np.random.uniform(0.8, 1.2)
        h = np.clip(h, 0, 1)
        h = rescale(h)
        population.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = eval_c5(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        fitness = [ -eval_c5(h) for h in population ]
        # Select elites (top 20%)
        elite_idx = np.argsort(fitness)[-int(pop_size*0.2):]
        elites = [population[i] for i in elite_idx]
        # Tournament selection
        new_pop = elites[:]
        for _ in range(pop_size - len(elites)):
            a, b = np.random.randint(0, pop_size, 2)
            if np.random.rand() < 0.9:
                # Crossover
                pt = np.random.randint(1, n_points - 1)
                c1 = population[a][:pt] + population[b][pt:]
                c2 = population[b][:pt] + population[a][pt:]
            else:
                # Mutation
                c1 = population[a].copy()
                c1[np.random.choice(n_points, size=5, replace=False)] += np.random.uniform(-0.2, 0.2)
                c1 = np.clip(c1, 0, 1)
                c1 = rescale(c1)
                c2 = population[b].copy()
                c2[np.random.choice(n_points, size=5, replace=False)] += np.random.uniform(-0.2, 0.2)
                c2 = np.clip(c2, 0, 1)
                c2 = rescale(c2)
            new_pop.append(c1)
            new_pop.append(c2)
        population = new_pop[:pop_size]
        # Track best
        for h in population:
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return (best_h, best_c5, n_points)
