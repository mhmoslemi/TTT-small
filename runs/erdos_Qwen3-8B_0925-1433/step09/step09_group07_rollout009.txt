import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    starting_points = []
    # Use initial as base and add perturbations
    base = np.clip(initial_h_values, 0, 1)
    base = base * target_sum / np.sum(base) if np.sum(base) > 0 else np.zeros(n_points)
    starting_points.append(base)

    # Generate 2 random perturbations
    for _ in range(2):
        perturb = initial_h_values + 0.1 * np.random.randn(n_points)
        perturb = np.clip(perturb, 0, 1)
        perturb = perturb * target_sum / np.sum(perturb) if np.sum(perturb) > 0 else np.zeros(n_points)
        starting_points.append(perturb)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    max_iter = int(budget_s / 0.1)  # ~6000 iterations per starting point

    for h in starting_points:
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        learning_rate = 0.1
        t = 0
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            t += 1
            # Compute correlation and worst shift
            corr = np.correlate(h, 1 - h, mode='full')
            idx_max = np.argmax(corr)
            k = idx_max - (n_points - 1)

            # Compute gradient for current worst k
            gradient = np.zeros(n_points)
            for x in range(n_points):
                shifted = x + k
                if 0 <= shifted < n_points:
                    gradient[x] = 1 - h[shifted]

            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * (gradient ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

            # Clip and normalize
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points)

            # Track best solution
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

            # Decay learning rate
            learning_rate *= 0.999

    return best_h, best_c5, n_points
