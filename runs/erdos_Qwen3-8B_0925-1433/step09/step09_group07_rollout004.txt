    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    max_iter = int(budget_s / 0.1)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)

        # Compute gradient for this k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx1 = (j + k_max) % n_points
            idx2 = (j - k_max) % n_points
            grad[j] = (1 - h[idx1]) - h[idx2]

        if current_c5 < best_c5:

            best_h = np.copy(h)

    return (best_h, best_c5, n_points)
