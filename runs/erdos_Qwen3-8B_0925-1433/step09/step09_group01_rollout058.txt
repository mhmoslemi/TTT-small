import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun
    start_time = time.time()

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        shift = np.random.randint(1, n_points // 2)
        perturb = np.roll(best_h, shift)
        res = minimize(obj, perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        freq = np.random.uniform(0.1, 0.5)
        amp = np.random.uniform(0.05, 0.15)
        mod_h = best_h + amp * np.sin(2 * np.pi * freq * np.arange(n_points) * dx)
        mod_h = np.clip(mod_h, 0, 1)
