from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

            if (i % 5 == 0) or (i % 5 == 4):
                initial_h[i] = 1.0
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return eval_c5(x)

    def clip_and_rescale(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum > 0:
            x = x * target_sum / x_sum
        return x

    # Hybrid optimization
    pop_size = 20
    pop = [clip_and_rescale(initial_h + np.random.normal(0, 0.05, n_points)) for _ in range(pop_size)]
    best_x = initial_h.copy()
    best_c5 = eval_c5(best_x)
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s - 200:
        if gen >= 100:
            break
        for _ in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.8 * (pop[b] - pop[c])
            trial = pop[np.random.randint(pop_size)]
            for i in range(n_points):
                if np.random.rand() < 0.9:
                    trial[i] = mutant[i]
            trial = clip_and_rescale(trial)
            trial_c5 = eval_c5(trial)
            if trial_c5 < eval_c5(pop[np.random.randint(pop_size)]):
                pop[np.random.randint(pop_size)] = trial
                if trial_c5 < best_c5:
                    best_x, best_c5 = trial, trial_c5
        gen += 1

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
