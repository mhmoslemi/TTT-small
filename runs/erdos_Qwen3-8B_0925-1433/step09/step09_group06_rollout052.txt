def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)

    def gen_symmetric():
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                pattern[i] = 1.0
            else:
                pattern[i] = 0.0
        return pattern

    def gen_periodic():
        pattern = np.zeros(n_points)
        for i in range(n_points):
            seg = (i // (n_points // 5)) % 5
            pattern[i] = 1.0 if seg % 2 == 0 else 0.0
        return pattern

    for _ in range(24):

        pattern = np.clip(pattern, 0, 1)
        pattern = pattern * target_sum / np.sum(pattern) if np.sum(pattern) > 0 else np.zeros(n_points)
        population.append(pattern.copy())

        pattern = gen_periodic()
        pattern = np.clip(pattern, 0, 1)

    population = np.array(population[:50], dtype=float)
