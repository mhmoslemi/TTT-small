from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        h = h * target_sum / np.sum(h)
    else:

        if np.sum(h_) > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        perturbations = []
        for _ in range(num):
            noise = np.random.normal(0, 0.05, n_points)
            sin_amp = 0.1

            perturbations.append(new_h)
        return perturbations

    perturbations = generate_perturbations(h, n_points, num=10)

    best_h = h.copy()
