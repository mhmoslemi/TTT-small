import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        x_norm = x_clipped * target_sum / np.sum(x_clipped) if np.sum(x_clipped) > 0 else np.zeros_like(x_clipped)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', maxiter=100,
        popsize=30, tol=1e-8, mutation=(0.5, 1), recombination=0.8, seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        best_h = result_de.x
    else:
        result_lbgfs = minimize(
            objective, result_de.x, method='L-BFGS-B', bounds=bounds, tol=1e-8
        )
        best_h = result_lbgfs.x

    final_result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8)
    return (final_result.x, final_result.fun, n_points)
