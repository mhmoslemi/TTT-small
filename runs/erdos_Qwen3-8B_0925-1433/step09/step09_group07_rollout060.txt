import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Blend with a structured periodic pattern for better local minima avoidance
    pattern = 0.5 * (1 + np.cos(np.pi * np.arange(n_points) / 3))
    h = initial_h_values * 0.5 + pattern * 0.5
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    learning_rate = 0.1
    decay_rate = 0.996
    max_iter = int(budget_s / 0.05)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)

        # Vectorized gradient computation
        j_plus = (np.arange(n_points) + k_max) % n_points
        j_minus = (np.arange(n_points) - k_max) % n_points
        grad = (1 - h[j_plus]) - h[j_minus]
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
