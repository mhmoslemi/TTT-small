import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return eval_c5(x)

    def de_population(n_pop=30):
        pop = [initial_h.copy()]
        for _ in range(n_pop - 1):
            perturb = np.random.normal(0, 0.08, n_points)
            new_h = initial_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            pop.append(new_h)
        return np.array(pop)

    population = de_population()
    best_x = initial_h.copy()
    best_c5 = eval_c5(best_x)

    start_time = time.time()
    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() - start_time < budget_s - 200:
        if gen >= 100:
            break
        indices = np.random.choice(len(population), 3, replace=False)
        r1, r2, r3 = population[indices]
        F_current = 0.8 + 0.1 * gen / 100
        mutant = r1 + F_current * (r2 - r3)
        target_idx = np.random.randint(len(population))
        target = population[target_idx]
        trial = target.copy()
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * target_sum / trial_sum
        trial_c5 = eval_c5(trial)
        if trial_c5 < eval_c5(target):
            population[target_idx] = trial
            if trial_c5 < best_c5:
                best_x, best_c5 = trial, trial_c5
        gen += 1

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
