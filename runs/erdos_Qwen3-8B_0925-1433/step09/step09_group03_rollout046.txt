        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final refinement
    def final_obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(final_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    return (best_h, best_c5, n_points)
