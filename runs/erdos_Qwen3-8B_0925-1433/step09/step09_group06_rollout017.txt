def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    candidates.append(h_periodic.copy())

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h_init in candidates:
        h = h_init.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        learning_rate = 0.2
        t = 0
        max_iter = 1000
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            t += 1
            corr = np.correlate(h, 1 - h, mode='full')
            lag = corr.argmax()
            G = np.zeros(n_points)
            for i in range(n_points):
                j = (i - lag) % n_points
                G[i] = (1 - h[j]) * dx
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * (G ** 2)
            m_hat = m / (1 - beta1 ** t)

            h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            current_corr = np.correlate(h, 1 - h, mode='full')
