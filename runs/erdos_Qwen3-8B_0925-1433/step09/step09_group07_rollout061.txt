    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.995
    max_iter = int(budget_s / 0.1)  # Estimated iterations per second

    velocity = np.zeros(n_points)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)

        # Compute gradient with modulo for all lags
        j = (np.arange(n_points) + k_max) % n_points
        grad = 1.0 - h[j]

        # Momentum update
        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
