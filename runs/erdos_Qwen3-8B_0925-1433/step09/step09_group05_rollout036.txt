    result = minimize(eval_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = result.x
    final_c5 = result.fun

    for _ in range(5):
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = final_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        res = minimize(eval_c5, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
