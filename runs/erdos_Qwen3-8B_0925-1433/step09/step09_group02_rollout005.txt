    def obj(x):

        sum_h = np.sum(h_)

            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate current best
        curr_c5 = obj(best_h)

        perturb = np.zeros(n_points)
