import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_periodic():
        x = np.linspace(0, 2, n_points)
        h = 1 - np.cos(np.pi * x) / 2
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_binary():
        h = np.random.choice([0, 1], size=n_points)
        count = np.sum(h)
        h = h * target_sum / count if count > 0 else np.zeros(n_points)
        return h

    def generate_clustered():
        x = np.linspace(0, 2, n_points)
        centers = np.array([0.5, 1.0, 1.5])
        std_dev = 0.1
        h = np.zeros(n_points)
        for i in range(n_points):
            dist = np.min(np.abs(x[i] - centers))
            h[i] = np.exp(-dist**2 / (2 * std_dev**2))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_reversed():
        return np.flip(initial_h_values)

    initializations = [
        initial_h_values,
        generate_periodic(),
        generate_binary(),
        generate_clustered(),
        generate_reversed()
    ]

    bounds = [(0, 1) * n_points]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = np.array(initial_h_values)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for init in initializations:
        init = np.array(init)
        init = np.clip(init, 0, 1)
        init_sum = np.sum(init)
        if init_sum > 0:
            init = init * target_sum / init_sum
        else:
            init = np.zeros(n_points)
        result = minimize(evaluate, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x.copy(), result.fun
        if time.time() - start_time > budget_s:
            break

    final_result = minimize(evaluate, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    if final_result.success:
        best_h, best_c5 = final_result.x, final_result.fun

    return (best_h, best_c5, n_points)
