import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Validate and normalize initial solution
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    # Initialize population
    pop_size = 100
    pop = [h.copy()]
    for _ in range(pop_size - 1):
        if np.random.rand() < 0.5:
            perturb = np.random.normal(0, 0.1, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            pop.append(new_h)
        else:
            while True:
                rand_h = np.random.rand(n_points)
                if np.sum(rand_h) > 0:
                    new_h = rand_h * target_sum / np.sum(rand_h)
                    break
            pop.append(new_h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 50:
        # Evaluate fitness
        fitness = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        fitness = np.array(fitness)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_h = pop[best_idx].copy()
            best_c5 = current_c5
        
        # Tournament selection
        K = 50
        tournament = np.random.choice(len(pop), size=K*2, replace=False)
        t_fitness = [np.max(np.correlate(pop[i], 1-pop[i], mode='full')) * dx for i in tournament]
        parent_idx = tournament[np.argsort(t_fitness)[:2*K]]
        
        # Generate offspring
        offspring = []
        for _ in range(len(pop)//2):
            if np.random.rand() < 0.7:  # Crossover
                p1, p2 = np.random.choice(parent_idx, size=2, replace=False)
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * pop[p1] + (1 - alpha) * pop[p2]
            else:  # Mutation
                p = np.random.choice(parent_idx)
                child = pop[p] + np.random.normal(0, 0.1, n_points)
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            child = child * target_sum / sum_c if sum_c > 0 else pop[p]
            offspring.append(child)
        
        # Replace worst half
        idx = np.argsort(fitness)[-len(offspring):]
        for i, _ in enumerate(idx):
            pop[idx[i]] = offspring[i]
        
        # Elitism
        top_idx = np.argsort(fitness)[:int(len(pop)*0.1)]
        pop[top_idx] = pop[top_idx]
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
