def run(seed=42, budget_s=600):

    import time
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    best_c5 = best_c5

    # Generate initial perturbations
    perturbations = [initial_h_values.copy()]
    for _ in range(2):
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_h_values + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 0:

        perturbations.append(perturbed)

    # Adam parameters
    initial_lr = 0.05
    beta1 = 0.9
    beta2 = 0.999

    max_iter = int(budget_s / 0.1)  # 10 iter per second
    iter_count = 0
    start_time = time.time()

    for init in perturbations:
        h = np.clip(init, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            iter_count += 1
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            k = k_max
            # Compute gradient
            term1 = np.zeros(n_points)
            valid = np.arange(n_points) + k < n_points
            term1[valid] = 1.0 - h[np.arange(n_points) + k]

            m = beta1 * m + (1 - beta1) * grad

            m_hat = m / (1 - beta1 ** iter_count)
