import time

        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return eval_c5(x)

    def gen_structured_guesses(h0, n_guesses=3):
        guesses = [h0.copy()]
        for i in range(n_guesses-1):

            h = h[::-1]  # Mirror
            h = h * np.random.uniform(0.95, 1.05)

            guesses.append(h)
        return np.array(guesses)

    CR = 0.9

            trial = target.copy()
            for i in range(n_points):
                if np.random.rand() < CR:
                    trial[i] = mutant[i]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
