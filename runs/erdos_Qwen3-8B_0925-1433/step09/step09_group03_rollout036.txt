import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def generate_patterns():
        patterns = []
        # Initial
        h_initial = initial_h_values.copy()
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
        patterns.append(h_initial)
        # Periodic
        x = np.linspace(0, 2, n_points)
        h_p = 1 - (1 + np.cos(np.pi * x)) / 2
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * target_sum / np.sum(h_p)
        patterns.append(h_p)
        # Binary
        b = np.zeros(n_points)
        np.random.shuffle(np.arange(n_points))
        b[:n_points//2] = 1
        b = b * target_sum / np.sum(b)
        patterns.append(b)
        # Clustered
        c = np.random.choice([0.2, 0.5, 0.8], n_points)
        h_c = np.zeros(n_points)
        for i in range(n_points):
            if c[i] < 0.4:
                h_c[i] = 0.2
            elif c[i] < 0.7:
                h_c[i] = 0.5
            else:
                h_c[i] = 0.8
        h_c = np.clip(h_c, 0, 1)
        h_c = h_c * target_sum / np.sum(h_c)
        patterns.append(h_c)
        return patterns

    patterns = generate_patterns()
    best_h = patterns[0].copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for h in patterns:
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5

    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(best_h + perturb, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        result = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
