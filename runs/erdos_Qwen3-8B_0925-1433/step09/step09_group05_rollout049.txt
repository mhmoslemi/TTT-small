from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def make_symmetric(h):
        mid = n_points // 2
        return np.concatenate([h[:mid], h[mid:mid+1], h[:mid][::-1]])

    def obj(x):
        return eval_c5(x)

    def de_population(n_pop=30):
        pop = [initial_h.copy()]
        for _ in range(n_pop - 1):
            perturb = np.random.normal(0, 0.05, n_points // 2)
            new_h = initial_h.copy()
            new_h[:n_points // 2] += perturb
            new_h = make_symmetric(new_h)
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            pop.append(new_h)
        return np.array(pop)

    population = de_population()
    best_x = initial_h.copy()
    best_c5 = eval_c5(best_x)

    start_time = time.time()
    F_start = 0.8
    F_end = 0.6
    CR_start = 0.9
    CR_end = 0.7
    max_gen = 100
    gen = 0
    while time.time() - start_time < budget_s - 200:
        if gen >= max_gen:
            break
        indices = np.random.choice(len(population), 3, replace=False)
        r1, r2, r3 = population[indices]
        F = F_start - (F_start - F_end) * gen / max_gen
        CR = CR_start - (CR_start - CR_end) * gen / max_gen
        mutant = r1 + F * (r2 - r3)
        mutant = make_symmetric(mutant)
        target_idx = np.random.randint(len(population))
        target = population[target_idx]
        trial = target.copy()
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * target_sum / trial_sum
        trial_c5 = eval_c5(trial)
        if trial_c5 < eval_c5(target):
            population[target_idx] = trial
            if trial_c5 < best_c5:
                best_x, best_c5 = trial, trial_c5
        gen += 1

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_x, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
    final_c5 = res.fun

    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(final_h + perturb, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * target_sum / new_sum
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
