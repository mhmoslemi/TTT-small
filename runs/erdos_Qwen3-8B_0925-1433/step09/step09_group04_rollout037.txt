        delta = c5_perturb - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_perturb.copy()
            best_c5 = c5_perturb
        temp *= cooling_rate

    step_size = 0.1
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        grad = compute_gradient(h, max_corr_idx)
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros_like(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        step_size *= 0.95

    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    best_c5 = compute_c5(h)
    return (h, best_c5, n_points)
