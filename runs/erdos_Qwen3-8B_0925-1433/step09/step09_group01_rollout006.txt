import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    base_h = clip_scale(initial_h_values.copy())
    candidates = [base_h.copy()]

    def perturb_sine(h):
        freq = np.pi / 2
        amp = 0.1
        h += amp * np.sin(freq * np.arange(n_points))
        return clip_scale(h)

    def perturb_swap(h):
        i = np.random.choice(n_points, 2, replace=False)
        h[i[0]], h[i[1]] = h[i[1]], h[i[0]]
        return clip_scale(h)

    for _ in range(4):
        candidates.append(perturb_sine(base_h.copy()))
        candidates.append(perturb_swap(base_h.copy()))

    def obj(x):
        h_ = x.copy()
        h_ = clip_scale(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    best_h = base_h
    best_c5 = obj(best_h)
    start_time = time.time()
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        if iter_count > 200:
            break
        h = candidates.pop(np.random.choice(len(candidates)))
        corr = np.correlate(h, 1 - h, mode='full')
        bad_lags = np.argsort(corr)[-5:]
        perturb_mask = np.zeros(n_points)
        for k in bad_lags:
            if k < len(corr) - n_points:
                indices = np.arange(n_points) + k
                indices = np.mod(indices, n_points)
                perturb_mask[indices] += 1
        h[perturb_mask.astype(bool)] += np.random.normal(0, 0.02, np.sum(perturb_mask))
        h = clip_scale(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        candidates.append(best_h.copy())
        if iter_count % 10 == 0:
            candidates = [best_h.copy()] + candidates[:4]

    return (best_h, best_c5, n_points)
