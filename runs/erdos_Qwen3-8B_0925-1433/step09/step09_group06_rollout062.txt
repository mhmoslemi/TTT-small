import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    
    # Initialize population with initial and perturbed variants
    population_size = 40
    population = [h_initial]
    for _ in range(population_size - 1):
        pert = np.random.normal(0, 0.1, size=n_points)
        p = h_initial + pert
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p) if np.sum(p) > 0 else np.zeros(n_points)
        population.append(p)
    
    # Best solution tracking
    best_h = h_initial
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    # DE parameters
    mutation_factor = 0.7
    crossover_rate = 0.9
    max_iterations = 500
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all members
        fitness = []
        for p in population:
            corr = np.correlate(p, 1 - p, mode='full')
            current_c5 = corr.max() * dx
            fitness.append(-current_c5)
        
        # Generate new population
        new_population = []
        for i in range(population_size):
            # Mutation
            idx1, idx2, idx3 = np.random.choice(population_size, 3, replace=False)
            mutant = population[idx1] + mutation_factor * (population[idx2] - population[idx3])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant) if np.sum(mutant) > 0 else np.zeros(n_points)
            
            # Crossover
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    trial[j] = population[i][j]
            
            # Evaluate trial
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = corr.max() * dx
            trial_fitness = -trial_c5
            
            # Selection
            if trial_fitness > fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Update best solution
        for p in population:
            corr = np.correlate(p, 1 - p, mode='full')
            current_c5 = corr.max() * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = p.copy()
    
    return (best_h, best_c5, n_points)
