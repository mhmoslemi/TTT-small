import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        h_ = h_ * target_sum / sum_h if sum_h > 0 else np.zeros(n_points)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_c5 = obj(h)
    best_h = h.copy()

    start_time = time.time()

    # Perturbations
    perturbations = []
    noise_std = 0.05
    for _ in range(5):
        noise = np.random.normal(0, noise_std, n_points)
        p = np.clip(h + noise, 0, 1)
        p = p * target_sum / np.sum(p) if np.sum(p) > 0 else np.zeros(n_points)
        perturbations.append(p)
    sine_pert = h + 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    sine_pert = np.clip(sine_pert, 0, 1)
    sine_pert = sine_pert * target_sum / np.sum(sine_pert) if np.sum(sine_pert) > 0 else np.zeros(n_points)
    perturbations.append(sine_pert)

    for p in perturbations:
        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(obj, p, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Final refinement
    if time.time() - start_time < budget_s - 60:
        result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 500})
        if result.success:
            best_h = result.x
            best_c5 = result.fun

    # Final clipping and scaling
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else np.zeros(n_points)
    return (best_h, best_c5, n_points)
