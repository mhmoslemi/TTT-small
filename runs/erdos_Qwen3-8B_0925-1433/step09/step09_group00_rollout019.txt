import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return compute_c5(x)

    bounds = [(0, 1)] * n_points

    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h
    best_c5 = compute_c5(h)
    start_time = time.time()

    for _ in range(40):
        if time.time() - start_time > budget_s - 60:
            break
        # Generate structured perturbation
        perturb = np.zeros(n_points)
        if np.random.random() < 0.4:
            perturb = 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        elif np.random.random() < 0.3:
            perturb = np.random.normal(0, 0.05, n_points)
        else:
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            perturb[indices] = np.random.choice([-1, 1], size=len(indices)) * 0.2
        candidate = h + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate = candidate * target_sum / np.sum(candidate)
        result = differential_evolution(obj, bounds, strategy='best1bin', maxiter=15, popsize=10, tol=1e-6)
        res_lbgf = minimize(obj, result.x, method='L-BFGS-B', bounds=bounds, tol=1e-8)
        if res_lbgf.fun < best_c5:
            best_h, best_c5 = res_lbgf.x, res_lbgf.fun

    res_final = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-10)
    return (res_final.x, res_final.fun, n_points)
