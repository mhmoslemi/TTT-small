from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        dx = 2.0 / n_points
        target_sum = n_points / 2.0

        remainder = target_sum - m
        h = np.zeros(n_points)
        h[:m] = 1.0
        h[-1] = remainder
        initial_h = np.clip(h, 0, 1)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(x):
        return evaluate_c5(x)

    # Differential evolution with adaptive parameters
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=20, mutation=(0.5, 1.0), recombination=0.9,
        tol=1e-6, maxiter=100, disp=False,
        callback=lambda x: None
    )

    best_x = result_de.x
    best_c5 = evaluate_c5(best_x)

    # Local optimization with SLSQP
    res = minimize(
        objective, best_x, method='SLSQP',
        bounds=bounds, constraints=cons,
        tol=1e-6, options={'ftol': 1e-8, 'maxiter': 100}
    )

    final_h = res.x
    final_c5 = evaluate_c5(final_h)

    # Refinement with random perturbation
    for _ in range(5):
        if time.time() > time.time() + budget_s - 30:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(final_h + perturb, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        res = minimize(
            objective, new_h, method='SLSQP',
            bounds=bounds, constraints=cons,
            tol=1e-6, options={'ftol': 1e-8, 'maxiter': 20}
        )
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
