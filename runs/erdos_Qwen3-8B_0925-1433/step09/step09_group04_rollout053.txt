import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Initialize solution with initial_h_values and normalize
    h = np.array(initial_h_values)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    # Local refinement with SLSQP
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return float('inf')
        h_norm = h_clipped * target_sum / sum_h
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    best_h = result.x
    best_c5 = evaluate_c5(best_h)

    # Final check
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h_clipped)
    if sum_h > 0:
        best_h = best_h_clipped * target_sum / sum_h
    else:
        best_h = np.zeros_like(best_h_clipped)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
