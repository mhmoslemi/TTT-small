        # Binary pattern
        b = np.random.choice([0, 1], size=n_points)
        s = np.sum(b)
        delta = target_sum - s
        b[:delta] = 1 if delta > 0 else 0
        b = b * target_sum / np.sum(b)
        patterns.append(b)

    best_h = np.array(initial_h_values)
