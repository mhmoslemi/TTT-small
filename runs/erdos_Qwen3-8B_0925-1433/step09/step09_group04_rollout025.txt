import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Generate diverse initial population
    population_size = 20
    population = [np.clip(initial_h_values, 0, 1).copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        new_member = initial_h_values + perturb
        new_member = np.clip(new_member, 0, 1)
        sum_new = np.sum(new_member)
        if sum_new > 0:
            new_member = new_member * target_sum / sum_new
        else:
            new_member = np.zeros_like(new_member)
        population.append(new_member)

    # DE parameters
    F = 0.8
    CR = 0.7
    max_gen = 300
    time_start = time.time()
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            indices = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = a + F * (b - c)
            trial = a.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            # Clip and normalize
            trial_clipped = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial_clipped)
            if sum_trial > 0:
                trial_norm = trial_clipped * target_sum / sum_trial
            else:
                trial_norm = np.zeros_like(trial_clipped)
            # Evaluate
            c5_trial = evaluate_c5(trial_norm)
            if c5_trial < evaluate_c5(population[i]):
                new_population.append(trial_norm)
                if c5_trial < best_c5:
                    best_h = trial_norm
                    best_c5 = c5_trial
            else:
                new_population.append(population[i])
        population = new_population

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
