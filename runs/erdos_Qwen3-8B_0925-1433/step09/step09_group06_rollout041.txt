import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial candidates
    def generate_candidate(base):
        if base is None:
            return np.random.uniform(0, 1, n_points)
        noisy = np.clip(base + np.random.uniform(-0.1, 0.1, n_points), 0, 1)
        return noisy * target_sum / np.sum(noisy) if np.sum(noisy) > 0 else np.zeros(n_points)

    # Initial candidates
    candidates = [
        initial_h_values,
        generate_candidate(None),
        generate_candidate(initial_h_values)
    ]

    best_h = None
    best_c5 = float('inf')

    for h in candidates:
        h = h.copy()
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        # Adam params
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m, v = np.zeros(n_points), np.zeros(n_points)
        lr = 0.5
        max_iter = 1000
        start_time = time.time()
        for t in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            lag = np.argmax(corr)
            # Compute gradient G
            G = np.zeros(n_points)
            for i in range(n_points):
                j = (i - lag) % n_points
                G[i] = (1 - h[j])
            # Adam update
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * (G ** 2)
            m_hat = m / (1 - beta1 ** (t + 1))
            v_hat = v / (1 - beta2 ** (t + 1))
            h -= lr * m_hat / (np.sqrt(v_hat) + eps)
            # Clip and scale
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            # Compute current c5

            current_c5 = corr_current.max() * dx

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        initial_h_values,
        generate_candidate(None),
        generate_candidate(initial_h_values)
    ]
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for h in candidates:
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

        m, v = np.zeros_like(h), np.zeros_like(h)
        lr = 0.5
        max_iter = 1000
        iter_count = 0
        
        while iter_count < max_iter and time.time() - start_time < budget_s:
            corr = np.correlate(h, 1 - h, mode='full')
            lag = np.argmax(corr)
            G = np.zeros(n_points)
            for i in range(n_points):
                j = (i - lag) % n_points
                G[i] = (1 - h[j])
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * (G ** 2)
            m_hat = m / (1 - beta1 ** (iter_count + 1))
            v_hat = v / (1 - beta2 ** (iter_count + 1))

            iter_count += 1
