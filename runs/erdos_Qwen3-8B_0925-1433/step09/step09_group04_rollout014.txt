import numpy as np
import time

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()

    # Local search
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Perturb a random position
        i = np.random.randint(n_points)
        for delta in [0.05, -0.05]:
            new_h = h.copy()
            new_h[i] += delta
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.zeros_like(new_h)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
