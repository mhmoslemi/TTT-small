import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:

            evaluations.append(corr.max() * dx)
        
        elites = []
        for _ in range(len(population) // 5):
            idx = np.argmin(evaluations)
            elites.append(population[idx])
