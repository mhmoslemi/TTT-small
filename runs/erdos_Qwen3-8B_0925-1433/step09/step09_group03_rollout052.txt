import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def obj(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    def generate_patterns():
        x_vals = np.linspace(0, 2, n_points)
        patterns = []
        # Periodic
        shift = np.random.uniform(0, 2)
        p = 1 - np.cos(np.pi * (x_vals - shift)) / 2
        patterns.append(clip_rescale(p))
        # Binary
        b = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        b[indices] = 1
        patterns.append(clip_rescale(b))
        # Clustered
        c = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 3:
                c[i] = 0.3
            elif i < 2 * n_points // 3:
                c[i] = 0.5
            else:
                c[i] = 0.2
        patterns.append(clip_rescale(c))
        # Sparse
        s = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
        s[indices] = 1
        patterns.append(clip_rescale(s))
        # Reversed
        r = initial_h_values[::-1].copy()
        patterns.append(clip_rescale(r))
        return patterns

    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)

    patterns = generate_patterns()
    for init in patterns:
        result = minimize(obj, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break

    final_result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-12)
    if final_result.success:
        best_h = final_result.x
        best_c5 = final_result.fun

    return (best_h, best_c5, n_points)
