import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)

    start_time = time.time()
    learning_rate = 0.5
    momentum = 0.8
    decay_rate = 0.9995
    vel = np.zeros(n_points)
    iter_count = 0
    early_stop = 50
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full')
        l = corr.argmax()
        G = np.zeros(n_points)
        for i in range(n_points):
            j = (i - l) % n_points
            G[i] = (1 - h[j]) * dx
        std_corr = np.std(corr)
        G = G / (std_corr + 1e-9)
        vel = momentum * vel + (1 - momentum) * G
        h += learning_rate * vel
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = current_corr.max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

        if iter_count % early_stop == 0 and learning_rate > 1e-6:
