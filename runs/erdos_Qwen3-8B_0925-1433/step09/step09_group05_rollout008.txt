import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                initial_h[i] = 1.0
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return eval_c5(x)

    start_time = time.time()
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    F = 0.8
    CR = 0.9
    gen = 0
    max_gen = 150

    while gen < max_gen and time.time() - start_time < budget_s - 200:
        for i in range(population_size):
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs]
            mutant = a + F * (b - c)
            trial = np.random.rand(n_points) * (mutant - 1) + mutant
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum > 0:
                trial = trial * target_sum / trial_sum
            if eval_c5(trial) < eval_c5(population[i]):
                population[i] = trial
        gen += 1

    best_h = min(population, key=eval_c5)
    best_c5 = eval_c5(best_h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(10):
            if time.time() - start_time >= budget_s:
                break
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
