    pattern_initializations = []
    # Periodic
    shift = np.random.uniform(0, 2)
    x = np.linspace(0, 2, n_points)
    h_per = 1 - np.cos(np.pi * (x - shift)) / 2
    h_per = np.clip(h_per, 0, 1)
    h_per = h_per * target_sum / np.sum(h_per)
    pattern_initializations.append(h_per)
    # Binary
    h_bin = np.random.choice([0, 1], size=n_points)
    current_sum = np.sum(h_bin)
    if current_sum < target_sum:
        diff = int(target_sum - current_sum)
        indices = np.where(h_bin == 0)[0]
        if len(indices) > diff:
            flip_indices = indices[:diff]
            h_bin[flip_indices] = 1
    elif current_sum > target_sum:
        diff = int(current_sum - target_sum)
        indices = np.where(h_bin == 1)[0]
        if len(indices) > diff:
            flip_indices = indices[:diff]
            h_bin[flip_indices] = 0
    pattern_initializations.append(h_bin)
    # Clustered
    h_clust = np.zeros(n_points)
    for _ in range(3):
        idx = np.random.choice(n_points, size=1)
        h_clust[idx] = np.random.uniform(0.5, 1)
    h_clust = np.clip(h_clust, 0, 1)
    h_clust = h_clust * target_sum / np.sum(h_clust)
    pattern_initializations.append(h_clust)
    # Sparse
    h_sparse = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
    h_sparse[indices] = 1
    current_sum = np.sum(h_sparse)
    if current_sum < target_sum:
        diff = int(target_sum - current_sum)
        indices = np.where(h_sparse == 0)[0]
        if len(indices) > diff:
            flip_indices = indices[:diff]
            h_sparse[flip_indices] = 1
    elif current_sum > target_sum:
        diff = int(current_sum - target_sum)
        indices = np.where(h_sparse == 1)[0]
        if len(indices) > diff:
            flip_indices = indices[:diff]
            h_sparse[flip_indices] = 0
    pattern_initializations.append(h_sparse)
    # Reversed
    h_rev = np.flip(initial_h_values)
    h_rev = np.clip(h_rev, 0, 1)
    h_rev = h_rev * target_sum / np.sum(h_rev)
    pattern_initializations.append(h_rev)

    best_h = np.copy(initial_h_values)
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    # Optimize each patterned initialization
    for h_init in pattern_initializations:
        h = np.copy(h_init)
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break

    # Final refinement
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    best_c5 = result.fun
    best_h = result.x

    return (best_h, best_c5, n_points)
