import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def objective(x):
        h_vals = x
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) / dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        objective, h, method='SLSQP', bounds=bounds, constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )

    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = h
        best_c5 = objective(h)

    return (best_h, best_c5, n_points)
