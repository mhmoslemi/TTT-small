        h_ = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h_)

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def cons(h_vec):
        return np.sum(h_vec) - target_sum
