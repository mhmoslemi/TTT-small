import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    n_particles = 30
    particles = [np.copy(initial_h_values) + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in particles]
    
    best_h = particles.copy()
    best_c5 = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    
    omega = 0.729
    c1 = 1.496
    c2 = 1.496
    max_vel = 0.1
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (best_h[i] - particles[i]) + c2 * r2 * (best_h[0] - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            if np.sum(particles[i]) == 0:
                particles[i] = np.copy(initial_h_values)
            else:
                particles[i] = particles[i] * target_sum / np.sum(particles[i])
            current_c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if current_c5 < best_c5[i]:
                best_h[i] = np.copy(particles[i])
                best_c5[i] = current_c5
        global_best_idx = np.argmin(best_c5)
        best_h[0] = np.copy(best_h[global_best_idx])
        best_c5[0] = best_c5[global_best_idx]
    
    best_h_final = best_h[0]
    best_c5_final = best_c5[0]
    return (best_h_final, best_c5_final, n_points)
