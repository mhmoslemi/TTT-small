import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Generate diverse initial population with noise perturbations
    population_size = 50
    population = []
    base = initial_h_values.copy()
    for _ in range(population_size):
        variant = base + np.random.normal(0, 0.05, n_points)
        variant = np.clip(variant, 0, 1)
        sum_v = np.sum(variant)
        if sum_v > 0:
            variant = variant * target_sum / sum_v
        population.append(variant)

    # DE hyperparameters
    F = 0.7
    CR = 0.8
    max_gen = 300
    time_start = time.time()
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            indices = np.random.choice(len(population), 3, replace=False)
            rand1, rand2, rand3 = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = rand1 + F * (rand2 - rand3)
            trial = rand1.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            # Clip and normalize
            trial_clipped = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial_clipped)
            if sum_trial > 0:
                trial_norm = trial_clipped * target_sum / sum_trial
            else:
                trial_norm = np.zeros_like(trial_clipped)
            # Evaluate and select
            c5_trial = evaluate_c5(trial_norm)
            c5_current = evaluate_c5(population[i])
            if c5_trial < c5_current:
                new_population.append(trial_norm)
            else:
                new_population.append(population[i])
            if c5_trial < best_c5:
                best_h = trial_norm
                best_c5 = c5_trial
        population = new_population

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    initial_guess = best_h
    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
