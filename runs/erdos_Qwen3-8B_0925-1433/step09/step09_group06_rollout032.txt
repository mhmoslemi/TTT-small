        init3 = init3 * target_sum / np.sum(init3) if np.sum(init3) > 0 else np.zeros(n_points)
        init4 = np.zeros(n_points)
        for i in range(n_points):
            if (i % 5) < 2:
                init4[i] = 1.0
            else:
                init4[i] = 0.0
        init4 = np.clip(init4, 0, 1)
        init4 = init4 * target_sum / np.sum(init4) if np.sum(init4) > 0 else np.zeros(n_points)
        return [init1, init2, init3, init4]
    
    initializations = generate_initializations()
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    max_iter = 1000
    for h in initializations:
        h = h.copy()
        m, v = np.zeros(n_points), np.zeros(n_points)
        t = 1
        learning_rate = 0.5
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k = np.argmax(corr)
            grad = 1 - np.roll(h, shift=-k)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            current_corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = current_corr.max() * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            learning_rate *= 0.999
            t += 1
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
    return (best_h, final_c5, n_points)
