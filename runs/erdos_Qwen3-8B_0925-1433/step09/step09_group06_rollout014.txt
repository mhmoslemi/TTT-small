import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.clip(h_initial, 0, 1)

        h_initial = h_initial * target_sum / np.sum(h_initial)
    else:
        h_initial = np.zeros(n_points)
    
    # Initialize population with initial and random variations
    population = [h_initial.copy()]
    for _ in range(39):
        perturbed = h_initial + np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) > 0:
            perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    
    start_time = time.time()
    num_generations = 0
    max_iterations = 100000
    
    while num_generations < max_iterations and time.time() - start_time < budget_s:
        num_generations += 1
        
        # Evaluate and track fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = corr.max()
            fitness_val = -max_corr * dx
            fitness.append(fitness_val)
        
        new_population = []
        for _ in range(len(population)):
            # Tournament selection
            idx1, idx2, idx3 = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idx1], population[idx2], population[idx3]
            
            # Mutation
            mutant = a + 0.7 * (b - c)
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) > 0:
                mutant = mutant * target_sum / np.sum(mutant)
            
            # Crossover
            trial = mutant.copy()
            for i in range(n_points):
                if np.random.rand() < 0.9:
                    trial[i] = a[i]
            
            # Evaluation
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = -np.max(corr) * dx
            
            # Selection
            if trial_fitness > fitness[idx1]:
                new_population.append(trial)
            else:
                new_population.append(population[idx1])
        
        population = new_population
        
        # Update best solution
        current_best = np.inf
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current = corr.max() * dx
            if current < current_best:
                current_best = current
                best_h = h.copy()
        if current_best < best_c5:
            best_c5 = current_best
    
    return (best_h, best_c5, n_points)
