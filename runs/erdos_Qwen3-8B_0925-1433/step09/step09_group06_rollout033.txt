import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_candidates = []
    
    # Normalize initial_h
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros(n_points)
    initial_candidates.append(initial_h)
    
    # Random initialization
    random_h = np.random.rand(n_points)
    random_h = random_h * (target_sum / np.sum(random_h))
    random_h = np.clip(random_h, 0, 1)
    random_h = random_h * target_sum / np.sum(random_h)
    initial_candidates.append(random_h)
    
    # Perturbed initialization
    perturbed_h = initial_h.copy()
    perturbed_h += np.random.uniform(-0.1, 0.1, size=n_points)
    perturbed_h = np.clip(perturbed_h, 0, 1)
    perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
    initial_candidates.append(perturbed_h)
    
    # Periodic initialization
    periodic_h = np.zeros(n_points)
    for i in range(n_points):
        if (i // (n_points // 5)) % 2 == 0:
            periodic_h[i] = 1.0
    periodic_h = periodic_h * target_sum / np.sum(periodic_h)
    periodic_h = np.clip(periodic_h, 0, 1)
    initial_candidates.append(periodic_h)
    
    best_h = initial_candidates[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    for init_h in initial_candidates:
        h = init_h.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.5
        max_iter = 500
        t = 0
        start_time = time.time()
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k = corr.argmax()
            G = np.zeros(n_points)
            for i in range(n_points):
                j = (i - k) % n_points
                G[i] = 1 - h[j]
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * (G ** 2)

            t += 1
    
    return (best_h, best_c5, n_points)
