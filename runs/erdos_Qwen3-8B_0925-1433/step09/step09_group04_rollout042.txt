import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Initialize with parent and normalize
    h = np.array(initial_h_values)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # DE parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_gen = 150
    time_start = time.time()
    population = [h.copy() for _ in range(pop_size)]

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            # Select 3 random parents
            indices = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            # Generate mutant
            mutant = a + F * (b - c)
            # Crossover
            trial = a.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            # Clip and normalize
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            if sum_trial > 0:
                trial = trial * target_sum / sum_trial
            # Evaluate
            c5_trial = compute_c5(trial)
            if c5_trial < compute_c5(population[i]):

            else:
                new_pop.append(population[i])
            if c5_trial < best_c5:
                best_h = trial.copy()
                best_c5 = c5_trial
        population = new_pop

    # Local search with greedy hill climbing
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        else:
            x = np.zeros_like(x)
        return compute_c5(x)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    if sum_h > 0:

    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
