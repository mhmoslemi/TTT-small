import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Build structured initial guess using sinusoidal components
    x = np.linspace(0, 2, n_points)
    # Frequencies for harmonic modulation
    freqs = [n_points / 4, n_points / 6, n_points / 8]
    amps = [0.1, 0.05, 0.03]
    sine_components = sum(amp * np.sin(2 * np.pi * freq * x) for amp, freq in zip(amps, freqs))
    h = initial_h_values + sine_components
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    # Multi-phase optimization strategy
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Add small perturbation to current best
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        
        # Run SLSQP with tighter tolerances
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
