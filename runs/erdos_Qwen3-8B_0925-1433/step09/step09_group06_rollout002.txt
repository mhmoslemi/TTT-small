import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Assume initial_h_values is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    # Normalize initial h
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    # Parameters
    learning_rate = 0.5
    beta = 0.8
    decay_rate = 0.995
    max_iterations = 100000
    velocity = np.zeros(n_points)
    no_improvement_count = 0
    improvement_threshold = 50
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        l = corr.argmax()
        # Compute gradient G for worst lag
        G = np.zeros(n_points)
        for i in range(n_points):
            j = (i - l) % n_points
            G[i] = (1 - h[j]) * dx
        # Scale G by standard deviation of overlaps
        overlap_std = np.std(corr)
        G = G * overlap_std
        # Update velocity and h
        velocity = beta * velocity + (1 - beta) * G
        h -= learning_rate * velocity
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        # Compute current C5
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = current_corr.max() * dx
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        # Adjust learning rate and beta
        if no_improvement_count >= improvement_threshold:
            learning_rate *= decay_rate
            beta *= 0.99
            no_improvement_count = 0
        # Optional: Early stopping

    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
