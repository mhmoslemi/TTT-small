import numpy as np
import time

def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return initial_h_values.copy()
        return h * target_sum / np.sum(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    n_particles = 30
    particles = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    particles = [normalize(p) for p in particles]
    pbest = particles.copy()
    gbest = min(particles, key=compute_c5)
    best_c5 = compute_c5(gbest)

    max_vel = 0.1
    w, c1, c2 = 0.729, 1.496, 1.496
