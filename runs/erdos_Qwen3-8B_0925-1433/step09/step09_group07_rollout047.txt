import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial solution
    h_clamped = np.clip(initial_h_values, 0, 1)
    h_initial = h_clamped * target_sum / np.sum(h_clamped) if np.sum(h_clamped) > 0 else np.zeros(n_points)
    best_h = np.copy(h_initial)
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    # PSO parameters
    n_particles = 30
    particles = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) if np.sum(np.clip(p, 0, 1)) > 0 else h_initial for p in particles]
    personal_best = particles.copy()
    global_best = np.copy(particles[0])
    global_c5 = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    best_c5 = min(global_c5)

    start_time = time.time()
    max_iter = 100
    omega = 0.729
    c1, c2 = 1.496, 1.496
    max_vel = 0.1

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Initialize velocities
        vel = [np.random.uniform(-max_vel, max_vel, n_points) for _ in range(n_particles)]
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel[i] = omega * vel[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            # Update position
            particles[i] += vel[i]
            # Clamp values
            particles[i] = np.clip(particles[i], 0, 1)
            # Normalize sum
            if np.sum(particles[i]) > 0:
                particles[i] = particles[i] * target_sum / np.sum(particles[i])
            else:
                particles[i] = h_initial
            # Compute C5
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(corr) * dx
            # Update personal best
            if current_c5 < np.max(np.correlate(personal_best[i], 1 - personal_best[i], mode='full')) * dx:

                if current_c5 < best_c5:

                    best_h = np.copy(particles[i])
                    global_best = np.copy(particles[i])
                    global_c5[i] = current_c5
            else:
