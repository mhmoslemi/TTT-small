import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

def compute_overlaps(h, dx, n):
    overlaps = np.zeros(n)
    for lag in range(n):
        idx = (np.arange(n) + lag) % n
        overlaps[lag] = np.dot(h, 1 - h[idx]) * dx
    return overlaps

def compute_gradient(h, k, dx, n):
    G = np.zeros(n)
    for i in range(n):
        idx_plus = (i + k) % n
        idx_minus = (i - k) % n
        G[i] = (1 - h[idx_plus] - h[idx_minus]) * dx
    return G
