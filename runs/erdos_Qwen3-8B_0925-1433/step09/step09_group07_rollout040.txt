        pert = perturb(initial_h_values, 0.05)
        initial_candidates.append(np.clip(pert, 0, 1))
    
    best_h = np.copy(initial_h_values)
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx
    start_time = time.time()

    max_iter = int(budget_s / 0.1)
    for init in initial_candidates:
        h = np.clip(init, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        lr = 0.05
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            idx_max = np.argmax(corr)
            k_max = idx_max - (n_points - 1)
            j = np.arange(n_points) + k_max
            valid_mask = (j >= 0) & (j < n_points)
            valid_j = j[valid_mask]
            valid_i = valid_j - k_max
            grad = np.zeros(n_points)
            grad[valid_i] = 1.0 - h[valid_j]
            grad = grad / (np.abs(grad) + 1e-8)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
