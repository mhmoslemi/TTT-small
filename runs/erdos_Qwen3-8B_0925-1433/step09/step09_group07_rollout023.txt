import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial setup
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    best_h = np.copy(h_initial)
    best_c5 = compute_c5(h_initial)

    # PSO parameters
    n_particles = 30
    w = 0.729
    c1 = 1.496
    c2 = 1.496
    max_vel = 0.1
    max_iter = int(budget_s / 0.1)

    # Initialize population
    particles = [np.clip(h_initial + np.random.normal(0, 0.05, n_points), 0, 1)]
    for _ in range(n_particles - 1):
        perturbation = np.random.normal(0, 0.05, n_points)
        p = np.clip(h_initial + perturbation, 0, 1)
        p = p * target_sum / np.sum(p) if np.sum(p) > 0 else np.zeros(n_points)
        particles.append(p)
    particles = np.array(particles)

    velocities = np.zeros((n_particles, n_points))
    personal_best = np.copy(particles)
    personal_best_c5 = np.array([compute_c5(p) for p in particles])
    global_best = np.copy(particles[0])
    global_best_c5 = compute_c5(particles[0])

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        r1, r2 = np.random.rand(n_particles), np.random.rand(n_particles)
        velocities = w * velocities + c1 * r1 * (personal_best - particles) + c2 * r2 * (global_best - particles)
        velocities = np.clip(velocities, -max_vel, max_vel)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            p = particles[i]
            if np.sum(p) > 0:
                particles[i] = p * target_sum / np.sum(p)
            else:
                particles[i] = np.zeros(n_points)
        current_c5 = np.array([compute_c5(p) for p in particles])
        improved = current_c5 < personal_best_c5
        personal_best[improved] = particles[improved]
        personal_best_c5[improved] = current_c5[improved]
        if np.min(current_c5) < global_best_c5:
            global_best = np.copy(particles[np.argmin(current_c5)])
            global_best_c5 = np.min(current_c5)
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5
            best_h = np.copy(global_best)

    return (global_best, best_c5, n_points)
