from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        m = n_points // 2
        T = 2.0 / (m - 1) if m > 1 else 2.0
        h = 0.5 + 0.5 * np.cos(np.pi * x / T)
        h = np.round(h).astype(float)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
