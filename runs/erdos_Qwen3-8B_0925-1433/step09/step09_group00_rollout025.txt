import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(x):
        s = np.sum(x)
        if s > 0:
            return x * target_sum / s
        return x

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = normalize(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    def perturb(h, scale):
        pert = np.random.normal(0, scale, n_points)
        new_h = np.clip(h + pert, 0, 1)
        return normalize(new_h)

    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        h_pert = perturb(best_h, 0.08)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(evaluate, h_pert, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for _ in range(10):
