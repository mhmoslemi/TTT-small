import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    base_h = np.clip(initial_h_values, 0, 1)
    if np.sum(base_h) > 0:
        base_h = base_h * target_sum / np.sum(base_h)
    else:
        base_h = np.zeros(n_points)

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    candidates = [base_h.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(base_h + noise, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        candidates.append(new_h.copy())
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        new_h = new_h.copy()
        new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        candidates.append(new_h.copy())

    best_c5 = float('inf')
    best_h = base_h.copy()
    best_idx = 0
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        current_h = candidates[best_idx]
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_val = np.max(corr)
        indices = np.where(corr == max_val)[0]
        worst_lags = indices[:3]
        masks = []
        for lag in worst_lags:
            if lag >= 0:
                start = 0
                end = n_points - 1 - lag
            else:
                m = -lag
                start = m
                end = n_points - 1
            mask = np.zeros(n_points, dtype=bool)
            if end >= 0:
                mask[start:end+1] = True
            masks.append(mask)
        new_candidates = []
        for mask in masks:
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = np.where(mask, current_h + noise, current_h)
            perturbed = np.clip(perturbed, 0, 1)
            if np.sum(perturbed) > 0:
                perturbed = perturbed * target_sum / np.sum(perturbed)
            new_candidates.append(perturbed.copy())
        new_candidates.append(current_h.copy())
        for candidate in new_candidates:
            res = minimize(objective, candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-7)
            if res.success:
                c5 = res.fun
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = res.x
                    best_idx = len(candidates)
                    candidates.append(res.x.copy())

    final_h = best_h.copy()
    final_c5 = objective(final_h)
    return (final_h, final_c5, n_points)
