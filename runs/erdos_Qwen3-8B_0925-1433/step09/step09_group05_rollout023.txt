import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        initial_h = np.zeros(n_points)

            if i % 3 == 0:
                initial_h[i] = 1.0
        initial_h = initial_h * 45 / 2 / np.sum(initial_h)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def de_population(n_pop=30):
        pop = [initial_h.copy()]
        for _ in range(n_pop - 1):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = initial_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            pop.append(new_h)
        return np.array(pop)

    population = de_population()
    best_x = initial_h.copy()
    best_c5 = obj(best_x)

    start_time = time.time()
    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() - start_time < budget_s - 200:
        if gen >= 100:
            break
        indices = np.random.choice(len(population), 3, replace=False)
        r1, r2, r3 = population[indices]
        mutant = r1 + F * (r2 - r3)
        target_idx = np.random.randint(len(population))
        target = population[target_idx]
        trial = target.copy()
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * target_sum / trial_sum
        trial_c5 = obj(trial)
        if trial_c5 < obj(target):
            population[target_idx] = trial
            if trial_c5 < best_c5:
                best_x, best_c5 = trial, trial_c5
        gen += 1

    res = minimize(obj, best_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
    final_c5 = res.fun

    for _ in range(10):
        if time.time() - start_time >= budget_s:

        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
