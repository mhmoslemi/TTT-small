    pop = [initial_h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        pop[i] = h
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -corr.max() * dx
    best_h = pop[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time >= budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop, 3, replace=False)
            mutant = a + 0.7 * (b - c)
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) == 0:
                mutant = np.zeros_like(mutant)
            else:
                mutant = mutant * target_sum / np.sum(mutant)
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = pop[i][j]
            trial_c5 = evaluate(trial)
            original_c5 = evaluate(pop[i])
            if trial_c5 > original_c5:
                pop[i] = trial
                if trial_c5 > best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n_points)
