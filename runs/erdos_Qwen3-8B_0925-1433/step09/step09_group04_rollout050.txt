import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Initial population
    population_size = 50
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(population_size)]
    for i in range(population_size):
        if np.sum(population[i]) > 0:
            population[i] = population[i] * target_sum / np.sum(population[i])
        else:
            population[i] = np.zeros_like(population[i])

    # DE parameters
    F = 0.8
    CR = 0.9
    max_gen = 200
    time_start = time.time()
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []

            rand1, rand2, rand3 = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = rand1 + F * (rand2 - rand3)
            trial = rand1.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            # Clip and normalize
            trial_clipped = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial_clipped)
            if sum_trial > 0:
                trial_norm = trial_clipped * target_sum / sum_trial
            else:
                trial_norm = np.zeros_like(trial_clipped)
            # Evaluate and select
            c5_trial = evaluate_c5(trial_norm)
            c5_current = evaluate_c5(population[i])
            if c5_trial < c5_current:
                new_population.append(trial_norm)
            else:
                new_population.append(population[i])

        population = new_population

    def obj(x):

            h_norm = np.zeros_like(h_clipped)
