import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h == 0:
            return float('inf')
        h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    try:
        result_de = differential_evolution(
            obj, bounds,
            popsize=20, maxiter=100, tol=1e-6,
            mutation=(0.5, 1), recombination=0.8,
            polish=True, callback=lambda x: None)
        best_h, best_c5 = result_de.x, result_de.fun
    except:
        best_h, best_c5 = h, obj(h)

    if time.time() - start_time > budget_s - 100:
        return (best_h, best_c5, n_points)

    result_lbgf = minimize(
        obj, best_h, method='L-BFGS-B',
        bounds=bounds, options={'maxiter': 100, 'ftol': 1e-8})
    best_h, best_c5 = result_lbgf.x, result_lbgf.fun

    return (best_h, best_c5, n_points)
