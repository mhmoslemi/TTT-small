    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    best_h = np.copy(h)
    rolled = np.array([np.roll(h, k) for k in range(n_points)])
    overlaps = h @ (1 - rolled)
    best_c5 = np.max(overlaps) * dx

    def objective(x):
        h = x
        rolled = np.array([np.roll(h, k) for k in range(n_points)])
        overlaps = h @ (1 - rolled)
        return np.max(overlaps) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    if result.success:
        h_opt = result.x
        c5_opt = objective(h_opt)
        if np.all((h_opt >= 0) & (h_opt <= 1)) and abs(np.sum(h_opt) - target_sum) < 1e-6:
            best_h = h_opt
            best_c5 = c5_opt
    else:
        best_h = h
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    return (best_h, best_c5, n_points)
