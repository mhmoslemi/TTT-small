import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_c5 = float('inf')
    best_h = h_initial

    start_time = time.time()
    for _ in range(5):
        noise = np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_initial + noise, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb) if np.sum(h_perturb) > 0 else h_initial
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if time.time() - start_time > budget_s - 30:
            break

    res_final = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res_final.success and res_final.fun < best_c5:
        best_c5 = res_final.fun
        best_h = res_final.x

    return (best_h, best_c5, n_points)
