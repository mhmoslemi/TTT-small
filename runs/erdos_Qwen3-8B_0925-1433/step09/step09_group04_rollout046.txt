            if 0 <= j < n_points:
                contributions[x] = h[x] * (1 - h[j])
            else:
                contributions[x] = 0
        x_idx = np.argmax(contributions)
        x_val = x_idx * dx

        # Perturb
        new_h = h.copy()
        new_h[x_idx] = 0
        delta = dx / 2
        x1 = x_val + delta
        x2 = x_val - delta
        j1, j2 = int(np.round(x1 / dx)), int(np.round(x2 / dx))
        for j in [j1, j2]:
            if 0 <= j < n_points:
                new_h[j] += 1
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.zeros_like(new_h)
        c5_trial = evaluate_c5(new_h)
        if c5_trial < best_c5:
            best_h = new_h
            best_c5 = c5_trial
            h = best_h

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, best_h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
