import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.997
    max_iter = int(budget_s / 0.1)
    prev_grad = np.zeros(n_points)
    perturb = np.random.normal(0, 0.02, n_points)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)

        j = np.arange(n_points) + k_max
        valid_mask = (j >= 0) & (j < n_points)
        valid_j = j[valid_mask]
        valid_i = valid_j - k_max
        grad = np.zeros(n_points)
        grad[valid_i] = 1.0 - h[valid_j]

        grad = momentum * prev_grad + learning_rate * grad
        h = h - grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        prev_grad = grad
        learning_rate *= decay_rate
        h += perturb

    return (best_h, best_c5, n_points)
