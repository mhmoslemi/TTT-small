import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_periodic():
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_binary():
        n = n_points
        cnt = int(n / 2)
        if n % 2:
            cnt += 1
        ones = np.random.choice(n, cnt, replace=False)
        h = np.zeros(n)
        h[ones] = 1
        h = h * target_sum / np.sum(h)
        return h

    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(2 * np.pi * x))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_sparse():
        ones = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
        h = np.zeros(n_points)
        h[ones] = 1
        h = h * target_sum / np.sum(h)
        return h

    def generate_reversed():
        return initial_h_values[::-1]

    initial_patterns = [generate_periodic(), generate_binary(), generate_sinusoidal(), generate_sparse(), generate_reversed()]

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for pattern in initial_patterns:
        h = np.clip(pattern, 0, 1)
        h = h * target_sum / np.sum(h)
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    if result.success:
        best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
