    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)

        j = (np.arange(n_points) + k_max) % n_points
        grad = 1.0 - h[j]

        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
