def run(seed=42, budget_s=600):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    learning_rate = 0.5
    momentum = 0.8
    velocity = np.zeros_like(h)
    max_iterations = 100000
    improvement_count = 0
    no_improvement_threshold = 50
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        l = corr.argmax()
        # Compute gradient G for lag l
        G = np.zeros(n_points)
        for i in range(n_points):
            j = (i - l) % n_points
            m = (i + l) % n_points
            G[i] = ((1 - h[j]) - h[m]) * dx
        # Apply momentum
        velocity = momentum * velocity + G
        h -= learning_rate * velocity
        # Clip and rescale
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 0:
            h = h * target_sum / total
        else:
            h = np.zeros(n_points)
        # Evaluate current C5
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = current_corr.max() * dx
        if current_c5 < best_c5:

            best_h = h.copy()
            improvement_count = 0
        else:
            improvement_count += 1
        if improvement_count >= no_improvement_threshold:
            learning_rate *= 0.9

        # Check for early stopping
        if learning_rate < 1e-6:
            break
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
    return (best_h, final_c5, n_points)
