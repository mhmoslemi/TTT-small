import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.clip(initial_h, 0, 1)
    norm_factor = target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else 1.0
    initial_h = initial_h * norm_factor

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(parent, scale=0.1):
        h = parent + np.random.normal(0, scale, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        return h

    def crossover(parent1, parent2, alpha):
        return alpha * parent1 + (1 - alpha) * parent2

    pop = []
    for _ in range(100):
        if np.random.rand() < 0.5:
            h = initial_h + np.random.normal(0, 0.15, n_points)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        pop.append(h)

    best_h = initial_h
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(500):
        if time.time() - start_time > budget_s - 50:
            break
        scores = np.array([evaluate(h) for h in pop])
        order = np.argsort(scores)
        elites = order[:10]
        parents = order[:50]

        new_pop = []

            parent1, parent2 = pop[idx1], pop[idx2]
