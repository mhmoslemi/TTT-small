            result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    # Final refinement
    try:
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 100})
        if result.success:
            best_h = result.x
            best_c5 = result.fun
    except:
        pass

    # Final feasibility check
    h = np.clip(best_h, 0, 1).copy()
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h
    return (best_h, best_c5, n_points)
