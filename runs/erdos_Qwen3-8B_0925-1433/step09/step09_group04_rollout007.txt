            best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        worst_lag = np.argmax(corr)

        best_new_h = None
        best_new_c5 = current_c5
        for i in range(n_points):
            if h[i] > 0.5:
                for delta in (-2, -1, 1, 2):
                    new_i = i + delta
                    if 0 <= new_i < n_points and h[new_i] == 0:
                        new_h = h.copy()
                        new_h[i] = 0
                        new_h[new_i] = 1
                        new_sum = np.sum(new_h)
                        if new_sum > 0:
                            new_h = new_h * m / new_sum
                        else:
                            new_h = np.zeros_like(new_h)
                        new_c5 = compute_c5(new_h)
                        if new_c5 < best_new_c5:
                            best_new_c5 = new_c5
                            best_new_h = new_h.copy()
        if best_new_h is not None:
            h = best_new_h
        else:
            break
        iteration += 1

    final_sum = np.sum(h)
    if final_sum > 0:
        h = h * m / final_sum
    else:
        h = np.zeros_like(h)
    final_c5 = compute_c5(h)
    return (h, final_c5, n_points)
