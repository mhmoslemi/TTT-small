def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)

    population_size = 20
    population = []

        base = base_h.copy()
        perm = np.random.permutation(n_points)
        base[perm] += np.random.normal(0, 0.05, n_points)
        base = np.clip(base, 0, 1)
        sum_base = np.sum(base)
        if sum_base > 0:
            base = base * target_sum / sum_base
        population.append(base)

    max_gen = 200
    F_start, F_end = 0.8, 0.5
