from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        s = np.sum(h)
        if s == 0:

        return h * target_sum / s

        h = np.clip(h, 0, 1)
        return rescale(h)

        h_ = clip_rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_h = clip_rescale(initial_h_values)
    best_h = initial_h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Initial refinement
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
