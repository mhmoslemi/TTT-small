            phase = np.pi / 4
            h = a * np.sin(2*np.pi*f1*x) + b * np.sin(2*np.pi*f2*(x - phase)) + c * np.sin(2*np.pi*f3*x)
            h = (h - h.min()) / (h.max() - h.min()) * 0.8 + 0.1

            h = h * (n / 2) / np.sum(h)

    n_points = len(h)
