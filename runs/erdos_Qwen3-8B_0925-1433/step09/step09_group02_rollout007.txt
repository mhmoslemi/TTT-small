import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(np.pi * x / (n_points / 2 - 1))
        h = np.round(h).astype(int)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    target_sum = n_points / 2.0
    h = h * target_sum / sum_h if sum_h > 0 else h
    dx = 2.0 / n_points

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * len(h)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        if _ % 5 == 0:
            x = np.linspace(0, 2, len(h))
            h_struc = 0.5 + 0.5 * np.cos(np.pi * x / (len(h)/2 - 1))
            h_struc = np.round(h_struc).astype(int)

            h_struc = np.clip(h_struc, 0, 1)
            best_h = h_struc + 0.05 * np.random.randn(len(h))
            best_h = np.clip(best_h, 0, 1)

    for _ in range(100):
        if time.time() - start_time > budget_s:

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
