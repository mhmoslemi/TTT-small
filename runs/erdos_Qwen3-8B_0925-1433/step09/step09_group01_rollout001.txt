import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h0 = np.clip(initial_h_values, 0, 1)
    h0 = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else np.zeros(n_points)
    best_h = h0.copy()
    best_c5 = obj(h0)

    start_time = time.time()
    iterations = 0

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
