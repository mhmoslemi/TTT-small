from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def gen_binary():
        positions = np.sort(np.random.choice(n_points, size=int(n_points/3), replace=False))
        h = np.zeros(n_points)
        h[positions] = 1
        h = h * target_sum / np.sum(h)
        return h
    
    patterns = [initial_h_values]
    for _ in range(4):
        shift = np.random.uniform(0, 2)
        patterns.append(gen_sine(shift))
    patterns.append(gen_binary())
    
    best_h = initial_h_values
    best_c5 = float('inf')
    start_time = time.time()
    
    for p in patterns:
        h = np.clip(p, 0, 1)
        h = h * target_sum / np.sum(h)

            sum_h = np.sum(hx)
            if sum_h > 0:
                hx = hx * target_sum / sum_h
            corr = np.correlate(hx, 1 - hx, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        
        if time.time() - start_time > budget_s:

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:

            best_h = res.x

    cons_final = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_final = [(0, 1)] * n_points
