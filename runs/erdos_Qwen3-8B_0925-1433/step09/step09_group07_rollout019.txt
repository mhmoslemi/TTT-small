import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_sum = np.sum(h)

    else:
        h = np.zeros(n_points)

        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]

    x0 = h.copy()

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-8})

    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h
        best_c5 = objective(h)
