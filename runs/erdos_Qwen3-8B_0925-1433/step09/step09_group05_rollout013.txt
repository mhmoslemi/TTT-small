        if res_perturb.fun < best_c5_local:
            best_h_local = res_perturb.x
            best_c5_local = res_perturb.fun

    final_sum = np.sum(best_h_local)
    if not np.isclose(final_sum, target_sum):
        best_h_local = best_h_local * target_sum / final_sum

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 0:
        return (best_h_local, best_c5_local, n_points)

    return (best_h_local, best_c5_local, n_points)
