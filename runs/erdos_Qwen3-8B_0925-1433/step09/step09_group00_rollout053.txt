import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=np.float64)
    if np.sum(h) != 0:
        h *= target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return float('inf')
        x_normalized = x_clipped * target_sum / sum_x
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)

    result_de = None
    if remaining_time > 0:
        result_de = differential_evolution(
            objective, bounds,
            strategy='best1bin', popsize=30, tol=1e-8,
            maxiter=100, mutation=(0.5, 1), recombination=0.8,
            seed=seed
        )

    best_h = h
    best_c5 = float('inf')
    if result_de is not None:
        best_h = result_de.x
        best_c5 = result_de.fun

    def objective_lbgf(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return float('inf')
        x_normalized = x_clipped * target_sum / sum_x
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')

    bounds_lbgf = [(0, 1)] * n_points
