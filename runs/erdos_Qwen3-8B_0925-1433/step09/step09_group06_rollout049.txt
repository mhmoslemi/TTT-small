    initial_h1 = np.array(initial_h_values, dtype=float)
    initial_h1 = initial_h1 * target_sum / np.sum(initial_h1) if np.sum(initial_h1) > 0 else np.zeros(n_points)
    
    # Generate perturbed initial
    perturb = np.random.normal(0, 0.1, n_points)
    initial_h2 = initial_h_values + perturb
    initial_h2 = np.clip(initial_h2, 0, 1)
    initial_h2 = initial_h2 * target_sum / np.sum(initial_h2) if np.sum(initial_h2) > 0 else np.zeros(n_points)
    
    # Generate periodic pattern
    initial_h3 = np.zeros(n_points)
    initial_h3[::2] = 1
    initial_h3[-1] = 0.5
    initial_h3_sum = np.sum(initial_h3)
    if initial_h3_sum > 0:
        initial_h3 *= target_sum / initial_h3_sum
    else:
        initial_h3 = np.zeros(n_points)
    
    best_h = initial_h1.copy()
    best_c5 = np.max(np.correlate(initial_h1, 1 - initial_h1, mode='full')) * (2.0 / n_points)
    
    def optimize_h(h_initial):
        h = h_initial.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        t = 0
        max_iter = 1000
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        learning_rate = 0.5
        best_h_opt = h.copy()
        best_c5_opt = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        for _ in range(max_iter):
            t += 1
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = corr.max()
            k = corr.argmax()
            G = np.zeros_like(h)
            for i in range(n_points):
                j = (i - k) % n_points
                G[i] = (1 - h[j])
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * (G ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
            h = np.clip(h, 0, 1)
            if np.sum(h) > 0:
                h = h * target_sum / np.sum(h)
            else:
                h = np.zeros(n_points)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
            if current_c5 < best_c5_opt:
                best_c5_opt = current_c5
                best_h_opt = h.copy()
            if _ % 100 == 0:
                if np.isclose(current_c5, best_c5_opt):
                    break
        return best_h_opt, best_c5_opt
    
    for h_initial in [initial_h1, initial_h2, initial_h3]:
        h_opt, c5_opt = optimize_h(h_initial)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return (best_h, best_c5, n_points)
