import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def perturb(h, method):
        if method == 'shift':
            k = np.random.randint(1, 5)
            shifted = np.roll(h, k)
            return np.clip(shifted, 0, 1) * target_sum / np.sum(shifted) if np.sum(shifted) > 0 else h
        elif method == 'mod':
            freq = np.pi / 2
            amp = 0.1
            return np.clip(h + amp * np.sin(freq * np.arange(n_points)), 0, 1) * target_sum / np.sum(h) if np.sum(h) > 0 else h
        elif method == 'swap':
            i, j = np.random.choice(n_points, 2, replace=False)
            h[i], h[j] = h[j], h[i]
            return np.clip(h, 0, 1) * target_sum / np.sum(h) if np.sum(h) > 0 else h

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        method = np.random.choice(['shift', 'mod', 'swap'])
        new_h = perturb(best_h, method)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
