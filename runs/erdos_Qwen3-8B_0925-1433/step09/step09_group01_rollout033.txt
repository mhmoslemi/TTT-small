import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

            pert = pert * target_sum / np.sum(pert) if np.sum(pert) > 0 else pert
            perts.append(pert)
        return perts
    
    h_initial = np.clip(initial_h_values, 0, 1)
