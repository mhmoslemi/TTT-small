            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0, 1)
        sum_trial = np.sum(trial)
        if sum_trial > 0:
            trial = trial * target_sum / sum_trial
        trial_c5 = eval_c5(trial)
        if trial_c5 < eval_c5(population[indices[0]]):
            population[indices[0]] = trial
            if trial_c5 < best_c5:
                best_x, best_c5 = trial, trial_c5

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_x, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
    final_c5 = res.fun

    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(final_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
