import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize base_h
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        current_corr = np.correlate(best_h, 1 - best_h, mode='full')
        positive_corr = current_corr[n_points:2*n_points - 1]
        if len(positive_corr) == 0:
            break
        top_p_indices = np.argsort(positive_corr)[-3:]
        k_values = [i - (n_points - 1) for i in top_p_indices]
        perturb = np.zeros(n_points)
        for k in k_values:
            if k > 0:
                start = 0
                end = n_points - 1 - k
                if end >= start:
                    noise = np.random.normal(0, 0.05, n_points)
                    mask = np.zeros(n_points)
                    mask[start:end+1] = 1.0
                    perturb += noise * mask

        best_h += perturb
        best_h = np.clip(best_h, 0, 1)
        if np.sum(best_h) > 0:
            best_h = best_h * target_sum / np.sum(best_h)

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
