        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        weighted = np.sum(corr[top_lags] * weighted_corr)
        return weighted * dx

    result = minimize(final_obj, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-10)
    best_h = result.x
    best_c5 = result.fun

    return (best_h, best_c5, n_points)
