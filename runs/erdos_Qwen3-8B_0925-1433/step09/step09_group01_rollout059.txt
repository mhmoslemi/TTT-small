import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    start_time = time.time()

    def perturb_shift(h, k):
        shifted = np.roll(h, k)
        return np.clip((h + shifted) / 2, 0, 1)

    def perturb_freq(h, amp=0.1, freq=0.5):
        freq = np.pi * freq
        return np.clip(h + amp * np.sin(freq * np.linspace(0, 2, n_points)), 0, 1)

    def perturb_swap(h, swaps=3):
        h = h.copy()
        for _ in range(swaps):
            i, j = np.random.choice(n_points, 2, replace=False)
            h[[i, j]] = h[[j, i]]
        return np.clip(h, 0, 1)

    best_h = h
    best_c5 = obj(h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        if np.random.rand() < 0.4:
            h = perturb_shift(best_h, np.random.randint(1, 6))
        elif np.random.rand() < 0.3:
            h = perturb_freq(best_h)
        else:
            h = perturb_swap(best_h)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else best_h
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_k = np.argmax(corr)
        perturb = np.zeros(n_points)
        perturb[worst_k % n_points] += 0.05
        perturb[(worst_k + n_points) % n_points] -= 0.05
        h = np.clip(best_h + perturb, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else best_h
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
