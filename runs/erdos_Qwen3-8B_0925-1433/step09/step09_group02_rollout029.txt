from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(80):
        if time.time() - start_time > budget_s:
            break
        # Focus on high-impact lags
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)

        # Compute directional perturbation
        grad = np.zeros_like(best_h)

            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            grad[i] = (corr_pert[max_idx] - max_val) * 100

        # Directional move

        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
