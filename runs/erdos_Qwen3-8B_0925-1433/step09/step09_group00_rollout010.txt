import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values

    if sum_h > 0:

    else:
        h_initial = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full') * dx
        return np.max(corr)

    perturbations = []

    # Sinusoidal perturbation
    sin_pert = 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    h_sin = h_initial + sin_pert
    h_sin = np.clip(h_sin, 0, 1)
    sum_h = np.sum(h_sin)
    if sum_h > 0:
        h_sin = h_sin * target_sum / sum_h
    perturbations.append(h_sin)

    # Random sparse perturbation
    mask = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
    h_sparse = h_initial.copy()
    h_sparse[mask] += np.random.uniform(-0.1, 0.1, len(mask))
    h_sparse = np.clip(h_sparse, 0, 1)
    sum_h = np.sum(h_sparse)
    if sum_h > 0:
        h_sparse = h_sparse * target_sum / sum_h
    perturbations.append(h_sparse)

    # Random walk perturbation
    h_walk = h_initial.copy()

    if sum_h > 0:

    perturbations.append(h_walk)
