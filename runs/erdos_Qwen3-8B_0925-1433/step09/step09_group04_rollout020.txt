import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_c5(h)

    initial_h = np.clip(initial_h_values, 0, 1).copy()
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * target_sum / sum_initial
    else:
        initial_h = np.zeros_like(initial_h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    # Simulated Annealing
    T = 1.0
    cooling_rate = 0.995
    iterations = 1000
    step_size = 0.1

    for _ in range(iterations):
        if time.time() - time.time() > budget_s:
            break
        h = best_h.copy()
        # Perturb h
        perturb = np.random.normal(0, step_size, size=n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(h)
        c5 = compute_c5(h)
        if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / T):
            best_h = h.copy()
            best_c5 = c5
        T *= cooling_rate

    # Local refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Final check
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
