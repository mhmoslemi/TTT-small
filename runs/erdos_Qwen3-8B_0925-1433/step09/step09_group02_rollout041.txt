import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            return h * target_sum / np.sum(h)
        return h

    h = rescale(initial_h_values.copy())

    def obj(x):
        h_ = rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        if np.random.rand() < 0.5:
            pert = np.random.normal(0, 0.01, n_points)
            new_h = rescale(best_h + pert)
        else:
            new_h = rescale(best_h + 0.02 * np.random.uniform(-0.5, 0.5, n_points))
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
