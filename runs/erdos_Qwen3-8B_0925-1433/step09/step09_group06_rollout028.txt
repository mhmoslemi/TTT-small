    else:
        initial_h = initial_h * target_sum / np.sum(initial_h)

    population_size = 50
    population = [initial_h.copy()]
    for _ in range(24):
        perturbed = initial_h * (1 + np.random.uniform(-0.1, 0.1, n_points))
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) > 0:
            perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    for _ in range(25):
        sym = initial_h[::-1].copy()
        sym = np.clip(sym, 0, 1)
        if np.sum(sym) > 0:
            sym = sym * target_sum / np.sum(sym)
        population.append(sym)
    for _ in range(25):
        periodic = np.zeros(n_points)
        for i in range(5):
            periodic[i*n_points//5:(i+1)*n_points//5] = 0.2
        periodic += np.random.uniform(-0.05, 0.05, n_points)
        periodic = np.clip(periodic, 0, 1)
        if np.sum(periodic) > 0:
            periodic = periodic * target_sum / np.sum(periodic)
        population.append(periodic)
    population = population[:population_size]

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(-corr.max() * dx)
        sorted_idx = np.argsort(fitness)
        elites = population[sorted_idx[:int(0.2 * population_size)]]
        new_pop = elites.copy()
        remaining = population_size - len(elites)
        for _ in range(remaining // 2):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            if np.random.random() < 0.9:
                pt = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:pt], parent2[pt:]])
                child2 = np.concatenate([parent2[:pt], parent1[pt:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            def mutate(child):
                num_mut = np.random.randint(5, 16)
                idx = np.random.choice(n_points, num_mut, replace=False)
                child[idx] += np.random.uniform(-0.1, 0.1, num_mut)
                child = np.clip(child, 0, 1)
                if np.sum(child) > 0:
                    child = child * target_sum / np.sum(child)
                return child
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.append(child1)
            new_pop.append(child2)
        population = new_pop[:population_size]
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = corr.max() * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)
