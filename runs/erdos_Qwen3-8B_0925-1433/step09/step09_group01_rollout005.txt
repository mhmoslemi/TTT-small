import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb_freq(h, amp=0.05):
        return np.clip(h + amp * np.sin(2 * np.pi * np.arange(n_points) / n_points), 0, 1)

    def perturb_swap(h, n=2):
        idx = np.random.choice(n_points, size=n, replace=False)
        h[idx[0]], h[idx[1]] = h[idx[1]], h[idx[0]]
        return np.clip(h, 0, 1)

    def perturb_noise(h, amp=0.05):
        return np.clip(h + amp * np.random.normal(0, amp, n_points), 0, 1)

    def target_lag(h):
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[-5:][::-1]
        return top_lags

    def optimize_candidate(h, max_iter=500):
        h = clip_scale(h)
        res = minimize(evaluate, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8, options={'maxiter': max_iter})
        return clip_scale(res.x)

    def run_phase(n_candidates=5, n_iterations=100):
        candidates = [clip_scale(initial_h_values.copy())]
        candidates.extend([perturb_noise(initial_h_values.copy()), perturb_freq(initial_h_values.copy()), perturb_swap(initial_h_values.copy())])
        candidates = [clip_scale(c) for c in candidates[:n_candidates]]
        best_h = candidates[0]
        best_c5 = evaluate(best_h)
        
        start_time = time.time()
        for _ in range(n_iterations):
            if time.time() - start_time > budget_s:
                break
            new_candidates = []
            for h in candidates:
                top_lags = target_lag(h)
                perturbed = h.copy()
                for lag in top_lags[:3]:
                    start = max(0, lag - 10)
                    end = min(n_points, lag + 10)
                    perturbed[start:end] += 0.02 * np.random.normal(0, 0.1, end - start)
                perturbed = clip_scale(perturbed)
                new_candidates.append(perturbed)
                new_candidates.append(optimize_candidate(perturbed))
            candidates = [optimization for optimization in new_candidates]
            current_values = [evaluate(h) for h in candidates]
            idx = np.argmin(current_values)
            best_h = candidates[idx]
            best_c5 = current_values[idx]
            if abs(best_c5 - evaluate(best_h)) < 1e-5:
                break
        return best_h, best_c5

    h_opt, c5_opt = run_phase(n_candidates=5, n_iterations=100)
    return (h_opt, c5_opt, n_points)
