import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    start_time = time.time()

    def perturb(h, perturb_type):
        if perturb_type == 'shift':
            k = np.random.randint(1, 5)
            shifted = np.roll(h, k)
            return np.clip((h + shifted) / 2, 0, 1)
        elif perturb_type == 'freq':
            f = np.pi / 2
            amp = 0.1
            freq = np.sin(f * np.linspace(0, 2, n_points))
            return np.clip(h + amp * freq, 0, 1)
        elif perturb_type == 'swap':
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            h = h.copy()
            h[i1], h[i2] = h[i2], h[i1]
            return h
        return h

    best_h = h
    best_c5 = obj(h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_type = np.random.choice(['shift', 'freq', 'swap'])
        new_h = perturb(best_h, perturb_type)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-7)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
