import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_vel = 0.1
    n_particles = 30
    omega = 0.729
    c1 = 1.496
    c2 = 1.496
    max_iter = 100
    start_time = time.time()

    # Initialize particles and velocities
    particles = []
    velocities = []
    for _ in range(n_particles):
        perturb = np.random.normal(0, 0.05, n_points)
        h = initial_h_values + perturb
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))

    personal_best = [np.copy(p) for p in particles]
    current_values = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    global_best = np.min(current_values)
    best_idx = np.argmin(current_values)
    best_h = particles[best_idx].copy()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (best_h - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            h_sum = np.sum(particles[i])
            if h_sum > 0:
                particles[i] = particles[i] * target_sum / h_sum
            else:
                particles[i] = np.zeros(n_points)
        
        new_values = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
        for i in range(n_particles):
            if new_values[i] < current_values[i]:
                personal_best[i] = np.copy(particles[i])
                current_values[i] = new_values[i]
        
        if np.min(current_values) < global_best:
            global_best = np.min(current_values)
            best_idx = np.argmin(current_values)
            best_h = np.copy(particles[best_idx])

    return (best_h, global_best, n_points)
