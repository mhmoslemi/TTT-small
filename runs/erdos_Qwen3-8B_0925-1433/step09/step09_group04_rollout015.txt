    time_start = time.time()
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        ones = np.where(best_h > 0)[0]
        if not ones.size:
            continue
        idx = np.random.choice(ones)
        new_h = best_h.copy()
        new_h[idx] = 0
        for move in (-dx, dx):
            new_x = idx * dx + move
            new_idx = int(np.round(new_x / dx))
            new_idx = np.clip(new_idx, 0, n_points - 1)
            new_h[new_idx] = 1
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * m / sum_new
                new_c5 = evaluate_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - m}]
    bounds = [(0, 1)] * n_points
    
    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_norm = h_clipped * m / sum_h
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * m / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
