import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h_current = x
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 100})

    if result.success:
        h_opt = result.x
        c5_opt = objective(h_opt)
        if np.allclose(np.sum(h_opt), target_sum, atol=1e-6) and np.all(h_opt >= 0) and np.all(h_opt <= 1):
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    return (best_h, best_c5, n_points)
