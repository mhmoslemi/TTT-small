import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_scale(arr):
        arr = np.clip(arr, 0, 1)
        if np.sum(arr) > 0:
            arr = arr * target_sum / np.sum(arr)
        return arr

    # Generate patterned initializations
    x = np.linspace(0, 2, n_points)
    h_periodic = (1 + np.cos(np.pi * x)) / 2
    h_binary = np.random.choice([0, 1], size=n_points)
    h_clustered = np.zeros(n_points)
    for _ in range(3):
        idx = np.random.choice(n_points, size=5, replace=False)
        h_clustered[idx] = np.random.uniform(0.5, 1.0)
    h_sparse = np.zeros(n_points)
    np.random.shuffle(h_sparse)
    h_sparse[:int(n_points * 0.3)] = 1
    h_reversed = np.flip(initial_h_values)
    initializations = [
        clip_scale(h_periodic), clip_scale(h_binary), clip_scale(h_clustered),
        clip_scale(h_sparse), clip_scale(h_reversed)
    ]

    best_h = clip_scale(initial_h_values)
    best_c5 = evaluate_erdos_solution(best_h)

    for h_init in initializations:
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            h_ = h_ * target_sum / sum_h if sum_h > 0 else h_
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        result = minimize(obj, h_init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    # Final refinement with Lagrangian weights
    h = clip_scale(best_h)
    corr = np.correlate(h, 1 - h, mode='full')
    top_lags = np.argsort(corr)[-20:]
    weights = np.zeros_like(corr)
    weights[top_lags] = 1.5

    def obj_final(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        h_ = h_ * target_sum / sum_h if sum_h > 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.dot(corr, weights) * dx

    start = time.time()
    result = minimize(obj_final, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    if result.success and time.time() - start < budget_s:
        best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
