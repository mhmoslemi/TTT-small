import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Validate parent
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    pop_size = 40
    mutation_factors = [0.6, 0.7, 0.8]
    crossover_rates = [0.8, 0.9, 1.0]
    pop = [h.copy() for _ in range(pop_size)]
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def rescale(v):
        v = np.clip(v, 0, 1)
        return v * target_sum / np.sum(v) if np.sum(v) > 0 else np.zeros_like(v)
    
    def mutate(parent, factor):
        idxs = np.random.choice(pop_size, 3, replace=False)

        mutant = np.clip(a + factor * (b - c), 0, 1)
        return rescale(mutant)
    
    start_time = time.time()
    best_c5 = evaluate(best_h)
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = np.array([evaluate(ind) for ind in pop])
        best_idx = np.argmax(fitness)
        current_best = pop[best_idx].copy()
        
        # Mutation and crossover
        new_pop = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            factor = np.random.choice(mutation_factors)
            mutant = mutate(a, factor)
            
            # Crossover
            trial = np.copy(mutant)
            for i in range(n_points):
                if np.random.rand() < 0.9:
                    trial[i] = pop[np.random.choice(pop_size)][i]

        # Replace population

    return (best_h, best_c5, n_points)
