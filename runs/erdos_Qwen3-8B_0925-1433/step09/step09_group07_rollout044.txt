import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h with initial_h_values, clip, normalize
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    # Parameters
    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.995
    noise_std = 0.01
    max_iter = int(budget_s / 0.1)  # 6000 iterations

    velocity = np.zeros(n_points)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        # Find worst lag k
        idx_max = np.argmax(corr)
        k_max = idx_max - (n_points - 1)
        # Compute gradient for this k
        j = np.arange(n_points) + k_max
        valid_mask = (j >= 0) & (j < n_points)
        valid_j = j[valid_mask]
        valid_i = valid_j - k_max
        grad = np.zeros(n_points)
        grad[valid_i] = 1.0 - h[valid_j]
        # Update velocity and h
        velocity = momentum * velocity + learning_rate * grad
        h = h - velocity
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        # Add random noise
        noise = noise_std * np.random.normal(size=n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        # Check if current_c5 is better
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        # Decay learning rate
        learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
