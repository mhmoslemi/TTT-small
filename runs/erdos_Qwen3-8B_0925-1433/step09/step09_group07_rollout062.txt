import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize initial solution
    initial_h = np.clip(initial_h_values, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros(n_points)

    # PSO parameters
    n_particles = 30
    max_vel = 0.1
    omega = 0.729
    c1, c2 = 1.496, 1.496
    max_iter = int(budget_s / 6)
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Initialize particles
    particles = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in particles]
    pbest = [p.copy() for p in particles]
    pbest_c5 = [compute_c5(p) for p in particles]
    gbest = particles[np.argmin(pbest_c5)]
    gbest_c5 = min(pbest_c5)

    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = omega * np.random.rand(n_points) + c1*r1*(pbest[i] - particles[i]) + c2*r2*(gbest - particles[i])
            vel = np.clip(vel, -max_vel, max_vel)
            particles[i] += vel
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            if np.sum(particles[i]) > 0:
                particles[i] = particles[i] * target_sum / np.sum(particles[i])
            else:
                particles[i] = initial_h.copy()

        # Evaluate and update
        for i in range(n_particles):
            current_c5 = compute_c5(particles[i])
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
            if current_c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = current_c5

        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
