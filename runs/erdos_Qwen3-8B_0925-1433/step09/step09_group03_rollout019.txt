import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    initial_guesses = [initial_h_values.copy()]

    # Generate diverse patterns
    x = np.linspace(0, 2, n_points)
    initial_guesses.append((1 + np.cos(np.pi * x / 2)) / 2)
    binary = np.random.rand(n_points)
    binary = (binary > 0.5).astype(float)
    binary = binary * target_sum / np.sum(binary)
    initial_guesses.append(binary)
    initial_guesses.append(initial_h_values[::-1])
    initial_guesses.append(np.zeros(n_points))
    initial_guesses.append(np.ones(n_points))
    initial_guesses.append(0.5 * np.ones(n_points))

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess) if np.sum(guess) > 0 else np.zeros(n_points)
        try:
            result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = result.x, current_c5
        except:
            pass
        if time.time() - start_time > budget_s - 10:
            break

    # Final refinement
    try:
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success:
            best_h, best_c5 = result.x, result.fun
    except:
        pass

    return (best_h, best_c5, n_points)
