        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    # Generate initial candidates
    init_candidates = [h.copy()]
    # Random uniform
    rand_init = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    init_candidates.append(rand_init)
    # Perturbed
    perturbed = np.array(initial_h_values, dtype=float)
    perturbed += np.random.uniform(-0.1, 0.1, size=n_points)
    perturbed = np.clip(perturbed, 0, 1)
    if np.sum(perturbed) > 0:
        perturbed = perturbed * target_sum / np.sum(perturbed)
    else:
        perturbed = np.zeros(n_points)
    init_candidates.append(perturbed)
    # Periodic pattern
    periodic = np.zeros(n_points)
    for i in range(5):
        start = i * 9
        end = start + 9
        if i < 2:
            periodic[start:end] = 1.0
    periodic = periodic * target_sum / np.sum(periodic)
    init_candidates.append(periodic)
    
    # Optimize each candidate
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    for candidate in init_candidates:
        h = candidate.copy()
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        learning_rate = 0.5
        max_iter = 1000
        t = 0
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            t += 1
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = corr.max()
            k = corr.argmax()
            # Compute gradient for current lag k
            G = np.zeros_like(h)
            for i in range(n_points):
                j = (i + k) % n_points
                G[i] = (1 - h[j]) * dx
            # ADAM update
            m = beta1 * m + (1 - beta1) * G
            v = beta2 * v + (1 - beta2) * (G ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
            # Clip and scale
            h = np.clip(h, 0, 1)
            if np.sum(h) > 0:
                h = h * target_sum / np.sum(h)
            else:
                h = np.zeros(n_points)
            # Update best
            current_c5 = max_corr * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Decay learning rate
            learning_rate *= 0.995
    
    # Check final best
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    return (best_h, best_c5, n_points)
