from scipy.optimize import minimize, Bounds
import time

        h = h * target_sum / np.sum(h)
        return h

    def generate_binary():
        h = np.random.binomial(1, 0.5, n_points)
        h = h * target_sum / np.sum(h)
        return h

    def generate_clustered(num_clusters=3):
        clusters = np.random.uniform(0, 2, num_clusters)
        widths = np.random.uniform(0.1, 0.5, num_clusters)
        h = np.zeros(n_points)
        for c in range(num_clusters):
            start = clusters[c] - widths[c] / 2
            end = clusters[c] + widths[c] / 2
            mask = (x >= start) & (x <= end)
            h[mask] = np.random.uniform(0.6, 1.0)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_sparse():
        indices = np.random.choice(n_points, int(0.3 * n_points), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        h = h * target_sum / np.sum(h)
        return h

        return np.flip(initial_h_values) * target_sum / np.sum(initial_h_values)

    def objective(x):
