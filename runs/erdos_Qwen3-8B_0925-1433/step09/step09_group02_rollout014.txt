import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    methods = [('L-BFGS-B', 150, 1e-6), ('SLSQP', 100, 1e-8)]
    for method_name, steps, tol in methods:
        for _ in range(steps):
            if time.time() - start_time > budget_s:
                break
            if method_name == 'L-BFGS-B':
                mid = np.random.uniform(0, 1, n_points)
                new_h = np.clip(best_h + 0.02 * (mid - 0.5), 0, 1)
                new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
                res = minimize(obj, new_h, method=method_name, bounds=bounds, constraints=cons, tol=tol)
            else:
                perturb = np.random.normal(0, 0.01, n_points)
                new_h = np.clip(best_h + perturb, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
                res = minimize(obj, new_h, method=method_name, bounds=bounds, constraints=cons, tol=tol)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
