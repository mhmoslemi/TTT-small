    bounds = [(0, 1)] * n_points

    start_time = time.time()

    # Differential Evolution
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=20, mutation=(0.5, 1), recombination=0.9,
        tol=1e-6, maxiter=100, seed=seed
    )
    best_h_de = result_de.x
    best_c5_de = objective(best_h_de)

    # Local optimization with SLSQP
    def objective_local(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds_local = [(0, 1)] * n_points

    result_local = minimize(
        objective_local, best_h_de, method='SLSQP',
        bounds=bounds_local, constraints=cons, tol=1e-8
    )
    best_h_local = result_local.x
    best_c5_local = result_local.fun

    # Perturbations and refinement
    for _ in range(5):
        if time.time() - start_time >= budget_s - 100:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = best_h_local + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            scale = target_sum / sum_new
            new_h = np.clip(new_h * scale, 0, 1)
        result = minimize(
            objective_local, new_h, method='SLSQP',
            bounds=bounds_local, constraints=cons, tol=1e-8
        )
        if result.fun < best_c5_local:
            best_h_local = result.x
            best_c5_local = result.fun

    initial_c5 = objective(initial_h)
    if initial_c5 < best_c5_local:
        best_h_local = initial_h
        best_c5_local = initial_c5

    return (best_h_local, best_c5_local, n_points)
