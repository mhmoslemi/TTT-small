            perturb = np.random.normal(0, 0.01, len(h_vals))
            new_h_vals = h_vals + perturb
            new_h_vals = np.clip(new_h_vals, 0, 1)
            if np.sum(new_h_vals) > 0:
                new_h_vals = new_h_vals * target_sum / np.sum(new_h_vals)
            else:
                new_h_vals = h_vals
            new_best_h = best_h.copy()
            new_best_h[x_vals.astype(int)] = new_h_vals
            res = minimize(obj, new_best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
