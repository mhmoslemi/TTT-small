import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_patterns():
        x = np.linspace(0, 2, n_points)
        yield np.clip(0.5 * (1 + np.cos(np.pi * x)), 0, 1) * target_sum / np.sum(np.clip(0.5 * (1 + np.cos(np.pi * x)), 0, 1))
        yield np.random.choice([0, 1], n_points) * target_sum / np.sum(np.random.choice([0, 1], n_points))
        yield np.random.choice([0, 1, 1, 1, 0, 0, 0, 1, 1, 0], size=n_points) * target_sum / np.sum(np.random.choice([0, 1, 1, 1, 0, 0, 0, 1, 1, 0], size=n_points))
        yield np.zeros(n_points)
        idxs = np.random.choice(n_points, int(0.3 * n_points), replace=False)
        h = np.zeros(n_points)
        h[idxs] = 1
        yield h * target_sum / np.sum(h)
        yield initial_h_values.copy() * target_sum / np.sum(initial_h_values)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h0 in generate_initial_patterns():
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else np.zeros(n_points)
        result = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        if time.time() - start_time > budget_s * 0.5:
            break

    if best_h is None:
        best_h = initial_h_values * target_sum / np.sum(initial_h_values)

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
