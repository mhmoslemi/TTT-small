        if time.time() - start_time > budget_s:
            break
        h_copy = h.copy()
        # Random perturbation
        idx = np.random.randint(0, n_points)
        h_copy[idx] += np.random.uniform(-0.1, 0.1)
        h_copy = np.clip(h_copy, 0, 1)
        sum_h = np.sum(h_copy)
        if sum_h > 0:
            h_copy = h_copy * target_sum / sum_h
        else:
            h_copy = np.zeros_like(h_copy)
        c5_current = evaluate_c5(h)
        c5_new = evaluate_c5(h_copy)
        if c5_new < c5_current or np.random.rand() < np.exp((c5_current - c5_new) / temperature):
            h = h_copy
        temperature *= cooling_rate
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    initial_guess = best_h
    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Final check
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
