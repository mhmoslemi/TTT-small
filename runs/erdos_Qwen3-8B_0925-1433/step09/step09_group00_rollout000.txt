import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            x = x * target_sum / s
        return x

    h = clip_normalize(initial_h_values.copy())
    
    def obj(x):
        h_ = clip_normalize(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h
    best_c5 = obj(best_h)
    start_time = time.time()
    
    for _ in range(30):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = clip_normalize(best_h + perturb)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    return (result.x, result.fun, n_points)
