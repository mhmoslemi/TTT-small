    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        shifted = shifted * target_sum / np.sum(shifted) if np.sum(shifted) > 0 else h
        return shifted
    
    def perturb_freq(h):

        perturb = 0.1 * freq

        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
        return new_h

        h[idxs[0]], h[idxs[1]] = h[idxs[1]], h[idxs[0]]
        h = np.clip(h, 0, 1)

        return h
    
    best_h = h.copy()
