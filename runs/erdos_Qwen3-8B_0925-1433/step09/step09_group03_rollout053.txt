def run(seed=42, budget_s=600):

        h = np.clip(y, 0, 1)

        if s == 0:
            return np.zeros(n_points)
        return h * target_sum / s

        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)

        h = np.clip(clusters, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return h * target_sum / s

    def generate_sparse():
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
        h[idx] = 1.0
        s = np.sum(h)
        if s == 0:
            return np.zeros(n_points)
        return h * target_sum / s

    def generate_reversed():
        return initial_h_values[::-1]

    patterns = [generate_periodic, generate_binary, generate_clustered, generate_sparse, generate_reversed]
    candidates = []

    start_time = time.time()
    for gen in patterns:
        h_initial = gen()
        def obj(x):
            h_ = np.clip(x, 0, 1)
            s = np.sum(h_)
            if s > 0:
                h_ = h_ * target_sum / s
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
