import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    # Objective function
    def objective(x):
        h_current = np.clip(x, 0, 1)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        return np.max(corr) * dx

    # Constraints
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    initial_guess = h.copy()

    # Optimization parameters
    maxiter = 100
    start_time = time.time()

    # Run SLSQP optimization
    result = minimize(objective, initial_guess, bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-6})

    # Fallback to raw initial guess if optimization failed
    if not result.success or time.time() - start_time > budget_s:
        best_h = h
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    else:
        best_h = result.x
        best_c5 = result.fun

    return (best_h, best_c5, n_points)
