import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def fea(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x
    
    def init_patterns():
        x = np.linspace(0, 2, n_points)
        patterns = [initial_h_values.copy()]
        # Periodic
        shift = np.random.uniform(0, 2)
        p = 1 - np.cos(np.pi * (x - shift)) / 2
        p = np.clip(p, 0, 1)
        patterns.append(p * target_sum / np.sum(p))
        # Binary
        m = int(np.round(target_sum))
        spacing = n_points / m
        b = np.zeros(n_points)
        for i in range(m):
            pos = int(np.random.uniform(0, n_points - i * spacing))
            b[pos] = 1
        patterns.append(b * target_sum / np.sum(b))
        # Sinusoidal
        s = 0.5 * (1 + np.cos(2 * np.pi * x))
        s = np.clip(s, 0, 1)
        patterns.append(s * target_sum / np.sum(s))
        # Sparse
        sparse = np.zeros(n_points)
        m_sparse = int(np.ceil(target_sum * 0.3))
        idx = np.random.choice(n_points, m_sparse, replace=False)
        sparse[idx] = 1
        patterns.append(sparse * target_sum / np.sum(sparse))
        # Reversed
        r = np.flip(initial_h_values)
        r = r * target_sum / np.sum(r)
        patterns.append(r)
        return patterns
    
    patterns = init_patterns()
    best_h = None
    best_c5 = float('inf')
    
    start = time.time()
    for h in patterns:
        h_feas = fea(h.copy())
        res = minimize(obj, h_feas, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type':'eq', 'fun':lambda x: np.sum(x) - target_sum}], tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    if best_h is None:
        best_h = fea(initial_h_values)
        best_c5 = obj(best_h)
    
    # final SLSQP
    best_h = fea(best_h)
    res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type':'eq', 'fun':lambda x: np.sum(x) - target_sum}], tol=1e-10)
    best_h = res.x
    best_c5 = obj(best_h)
    
    best_h = fea(best_h)
    return (best_h, best_c5, n_points)
