from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        current_h = best_h.copy()
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        top_indices = np.argsort(corr)[::-1][:3]
        top_lags = [idx - (n_points - 1) for idx in top_indices]

        candidates = []
        noise = np.random.normal(0, 0.01, n_points)
        for lag in top_lags:
            perturbed = current_h.copy()
            if lag >= 0:
                for i in range(n_points - lag):
                    perturbed[i] += noise[i]
                    perturbed[i + lag] += noise[i + lag]
            else:
                for i in range(-lag, n_points):
                    perturbed[i] += noise[i]
                    perturbed[i - lag] += noise[i - lag]
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed) if np.sum(perturbed) > 0 else perturbed
            candidates.append(perturbed)
        random_perturb = current_h + np.random.normal(0, 0.01, n_points)
        random_perturb = np.clip(random_perturb, 0, 1)
        random_perturb = random_perturb * target_sum / np.sum(random_perturb) if np.sum(random_perturb) > 0 else random_perturb
        candidates.append(random_perturb)
