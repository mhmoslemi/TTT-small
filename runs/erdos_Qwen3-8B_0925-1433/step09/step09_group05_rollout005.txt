import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        initial_h = np.random.rand(n_points)
        target_sum = n_points / 2
        initial_h = initial_h * target_sum / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    def optimize(start):
        result = minimize(eval_c5, start, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        return result.x, result.fun

    start_time = time.time()
    best_h, best_c5 = initial_h.copy(), eval_c5(initial_h)
    best_h_scaled = best_h.copy()

    for _ in range(10):
        if time.time() - start_time >= budget_s - 200:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = best_h_scaled + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        new_h, new_c5 = optimize(new_h)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
            best_h_scaled = new_h

    return (best_h, best_c5, n_points)
