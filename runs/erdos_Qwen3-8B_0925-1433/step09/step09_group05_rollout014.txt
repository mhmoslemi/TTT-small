import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 45
        initial_h = np.zeros(n_points)
        initial_h[::5] = 1.0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points/2) / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return eval_c5(x)

    def de_population(pop_size):
        pop = [initial_h.copy() for _ in range(pop_size)]
        for i in range(pop_size):
            pert = np.random.normal(0, 0.05, n_points)
            pop[i] = np.clip(initial_h + pert, 0, 1)
            pop[i] = pop[i] * target_sum / np.sum(pop[i])
        return np.array(pop)

    def repair(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum > 0:
            x = x * target_sum / x_sum
        return x

    pop = de_population(30)
    best_x = initial_h.copy()
    best_c5 = eval_c5(best_x)

    start_time = time.time()
    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() - start_time < budget_s - 300:
        if gen >= 100:
            break
        indices = np.random.choice(len(pop), 3, replace=False)
        r1, r2, r3 = pop[indices]
        mutant = r1 + F * (r2 - r3)
        target_idx = np.random.randint(len(pop))
        trial = pop[target_idx].copy()
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        trial = repair(trial)
        trial_c5 = eval_c5(trial)
        if trial_c5 < eval_c5(pop[target_idx]):
            pop[target_idx] = trial
            if trial_c5 < best_c5:
                best_x, best_c5 = trial, trial_c5
        gen += 1

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(objective, best_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
    final_c5 = res.fun

    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(final_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
