    scale = target_sum / sum_s if sum_s > 0 else 0
    h_sparse = sparse * scale
    initial_candidates.append(h_sparse)

    # Reversed initial pattern
    reversed_h = initial_h_values[::-1]
    sum_rev = np.sum(reversed_h)
    scale_rev = target_sum / sum_rev if sum_rev > 0 else 0
    h_reversed = reversed_h * scale_rev
    initial_candidates.append(h_reversed)

    best_h = np.copy(initial_h_values)
    best_c5 = float('inf')
    start_time = time.time()

    for candidate in initial_candidates:
        result = minimize(obj, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        if time.time() - start_time > budget_s:
            break

    # Final refinement
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    best_h = result.x
    best_c5 = result.fun

    return (best_h, best_c5, n_points)
