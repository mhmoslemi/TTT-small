from scipy.optimize import minimize
import time

        if s > 0:
            return np.clip(h * target_sum / s, 0, 1)
        return np.zeros(n_points)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return clip_scale(h)
