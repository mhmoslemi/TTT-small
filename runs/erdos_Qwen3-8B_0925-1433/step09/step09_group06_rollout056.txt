                if step > 50 and current_c5 >= current_best:
                    break
            learning_rate *= 0.999
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
    return (best_h, final_c5, n_points)
