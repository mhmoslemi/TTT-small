import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def apply_perturbation(h, seed=None, pattern='sine'):
        np.random.seed(seed)
        if pattern == 'sine':
            wave = 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
            pert = np.random.uniform(-0.1, 0.1, size=n_points)
            new_h = h + wave * pert
        elif pattern == 'walk':
            pert = np.random.normal(0, 0.01, size=n_points)
            new_h = h + pert
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    best_h = h
    best_c5 = obj(best_h)
    start_time = time.time()

    # Initial differential evolution
    result_de = differential_evolution(obj, bounds, constraints=cons, tol=1e-8, maxiter=50, popsize=20, seed=seed)
    best_h, best_c5 = result_de.x, result_de.fun

    # Refinement with L-BFGS-B
    result_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    best_h, best_c5 = result_lbgf.x, result_lbgf.fun

    # Perturbation loop
    for _ in range(50):
        if time.time() - start_time > budget_s - 50:
            break
        # Generate multiple patterns
        patterns = ['sine', 'walk', 'random']
        for pattern in patterns:
            new_h = apply_perturbation(best_h, pattern=pattern)
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    # Final refinement
    result_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    return (result_lbgf.x, result_lbgf.fun, n_points)
