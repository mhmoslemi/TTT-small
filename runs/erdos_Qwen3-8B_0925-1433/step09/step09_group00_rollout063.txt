from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        total = np.sum(h)
        if total > 0:
            h *= target_sum / total
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Initialize population

    best_c5 = compute_c5(best_h)
