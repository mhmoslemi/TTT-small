import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    if 'initial_h_values' in locals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * 25 / (n_points // 2)

    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    def optimize_candidate(h):
        def objective(x):
            h_clipped = np.clip(x, 0, 1)
            sum_h = np.sum(h_clipped)
            if sum_h > 0:
                h_clipped = h_clipped * target_sum / sum_h
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        return result.x, result.fun

    def perturb_by_lag(h, lag, sigma=0.05):
        n_points = len(h)
        indices = []
        if lag >= 0:
            indices = np.arange(0, n_points - lag)
        else:
            indices = np.arange(-lag, n_points)
        perturb = np.random.normal(0, sigma, len(indices))
        perturbed = h.copy()
        perturbed[indices] += perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 0:
            perturbed = perturbed * target_sum / perturbed_sum
        return perturbed

    best_h = h
    best_c5 = compute_c5(h)
    start_time = time.time()
    candidates = [best_h]

    for _ in range(200):
        if time.time() - start_time > budget_s:

        h_candidate = candidates[np.random.choice(len(candidates))]
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_h, best_c5 = optimize_candidate(h_candidate)

            perturbed = perturb_by_lag(h_candidate, lag, sigma=0.05)
            candidates.append(perturbed)

    return (best_h, best_c5, n_points)
