import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def generate_periodic():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(np.pi * (x - 1.0))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_binary():
        num_ones = int(target_sum)
        binary = np.zeros(n_points)
        binary[:num_ones] = 1
        np.random.shuffle(binary)
        return binary

    def generate_sparse():
        sparse = np.zeros(n_points)
        num_sparse = int(0.3 * n_points)
        indices = np.random.choice(n_points, size=num_sparse, replace=False)
        sparse[indices] = 1
        sparse = sparse * target_sum / np.sum(sparse)
        return sparse

    def generate_clustered():
        cluster_sizes = [int(n_points * 0.3), int(n_points * 0.4), int(n_points * 0.3)]
        clustered = np.zeros(n_points)
        for i, size in enumerate(cluster_sizes):
            start = int(n_points * (i * 0.33))
            end = start + size
            cluster = np.zeros(n_points)
            cluster[start:end] = 0.2 + 0.6 * i
            clustered += cluster
        clustered = clustered * target_sum / np.sum(clustered)
        return clustered

    patterns = [initial_h_values.copy()]
    patterns.append(generate_periodic())
    patterns.append(generate_binary())
    patterns.append(generate_sparse())
    patterns.append(generate_clustered())

    start_time = time.time()
    best_h = patterns[0].copy()
    best_c5 = float('inf')

    for pattern in patterns:
        h = np.clip(pattern, 0, 1)
        h = h * target_sum / np.sum(h)
        def obj(x):
            h_ = np.clip(x, 0, 1)
            h_ = h_ * target_sum / np.sum(h_) if np.sum(h_) > 0 else h_
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success and result.fun < best_c5:
