            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def get_overlapping_indices(lag, n_points):
        if lag >= 0:
            start = 0
            end = n_points - 1 - lag
        else:
            start = -lag
            end = n_points - 1
        if start > end:
            return np.array([], dtype=int)
        return np.arange(start, end + 1)

    while iterations < 200 and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        lag = m - (n_points - 1)
        indices = get_overlapping_indices(lag, n_points)
        if indices.size == 0:
            indices = np.arange(n_points)
        noise = np.random.normal(0, 0.05, size=indices.size)
        h[indices] += noise
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-7)
        if result.success:
            h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        iterations += 1

    return (best_h, best_c5, n_points)
