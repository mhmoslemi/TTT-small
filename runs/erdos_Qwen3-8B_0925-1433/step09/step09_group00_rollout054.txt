from scipy.optimize import minimize
import time

        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def genetic_algorithm():

            pop.append(h_pert)
        for _ in range(200):
