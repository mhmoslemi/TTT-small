import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        return x

    def obj(x):
        h_ = clip_and_scale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h = np.array(initial_h_values)
    h = clip_and_scale(h)
    best_h = h
    best_c5 = obj(h)

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s - 100:
            break
        # Sinusoidal perturbation
        pturb = 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        new_h = np.clip(best_h + pturb, 0, 1)
        new_h = clip_and_scale(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        # Random sparse
        indices = np.random.choice(n_points, size=int(n_points*0.1), replace=False)
        new_h = best_h.copy()
        new_h[indices] = 1 - new_h[indices]
        new_h = clip_and_scale(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    return (result.x, result.fun, n_points)
