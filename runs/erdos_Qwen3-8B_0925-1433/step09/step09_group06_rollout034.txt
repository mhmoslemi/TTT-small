import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    # Adam optimizer parameters
    learning_rate = 0.5
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    improve_count = 0
    max_improve = 50
    
    for _ in range(100000):
        if time.time() - start_time > budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        l = corr.argmax()
        G = np.zeros(n_points)
        for i in range(n_points):
            j = (i - l) % n_points
            G[i] = (1 - h[j]) * dx
        variance = np.var(corr)
        G_scaled = G / (np.sqrt(variance) + epsilon)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * G_scaled
        v = beta2 * v + (1 - beta2) * (G_scaled ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        grad_step = learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Add small noise for exploration
        noise = np.random.normal(0, 0.01, size=h.shape)
        h -= grad_step + noise
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = current_corr.max() * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improve_count = 0
        else:
            improve_count += 1
        
        if improve_count >= max_improve:
            learning_rate *= 0.995
            improve_count = 0
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
    return (best_h, final_c5, n_points)
