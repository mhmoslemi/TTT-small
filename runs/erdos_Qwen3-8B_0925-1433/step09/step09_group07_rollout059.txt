import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        h_ = x
        max_overlap = 0.0
        n = len(h_)
        for k in range(2 * n - 1):
            shifted = np.roll(h_, k)
            overlap = np.dot(h_, 1 - shifted)
            current = overlap * dx
            if current > max_overlap:
                max_overlap = current
        return max_overlap

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], options={'maxiter': 100})
    
    if result.success:
        best_h = result.x
        final_c5 = objective(best_h)
    else:
        best_h = h
        final_c5 = objective(h)
    
    return (best_h, final_c5, n_points)
