from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

        return np.max(corr) * dx

    result_de = differential_evolution(
        objective, bounds, strategy='rand', popsize=30, maxiter=100,

    best_h = result_de.x
    best_c5 = result_de.fun

        objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8
