import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' not in globals():
        n_points = 45
        initial_h_values = np.random.rand(n_points)
    else:
        n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    patterns = []
    x_vals = np.linspace(0, 2, n_points)
    h = 1 - np.cos(np.pi * (x_vals - 1.0)) / 2
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    patterns.append(h)

    num_ones = int(target_sum + 0.5)
    indices = np.random.choice(n_points, size=num_ones, replace=False)
    h = np.zeros(n_points)
    h[indices] = 1.0
    sum_h = np.sum(h)
    if sum_h != target_sum:
        if sum_h < target_sum:
            h[np.random.randint(n_points)] = 1.0
        else:
            h[np.random.randint(n_points)] = 0.0
    patterns.append(h)

    for _ in range(3):
        centers = np.random.choice(n_points, 3, replace=False)
        h = np.zeros(n_points)
        for c in centers:
            h[c] = 1.0
        sum_h = np.sum(h)
        h = h * target_sum / sum_h
        patterns.append(h)

    for _ in range(2):
        num_ones = int(0.3 * n_points)
        indices = np.random.choice(n_points, num_ones, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        sum_h = np.sum(h)
        h = h * target_sum / sum_h
        patterns.append(h)

    h_rev = initial_h_values[::-1]
    h_rev = np.clip(h_rev, 0, 1)
    sum_h = np.sum(h_rev)
    h_rev = h_rev * target_sum / sum_h
    patterns.append(h_rev)

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if time.time() - start_time > budget_s:
            break

    if best_h is None:
        best_h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(best_h)
        best_h = best_h * target_sum / sum_h

    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    if res.success:
        best_c5, best_h = res.fun, res.x

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    best_h = best_h * target_sum / sum_h

    return (best_h, best_c5, n_points)
