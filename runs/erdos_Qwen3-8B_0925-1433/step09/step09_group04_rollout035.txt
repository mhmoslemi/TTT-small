import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        s = np.sum(h)
        if s > 0:
            return h * target_sum / s
        return np.zeros_like(h)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 100
    population = [normalize(initial_h.copy()) for _ in range(population_size)]
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    F = 0.8
    CR = 0.4
    max_gen = 200
    time_start = time.time()

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial = normalize(trial)
            c5_trial = evaluate_c5(trial)
            if c5_trial < evaluate_c5(population[i]):
                new_population.append(trial)
                if c5_trial < best_c5:
                    best_h = trial
                    best_c5 = c5_trial
            else:
                new_population.append(population[i])
        population = new_population

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    def obj(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, best_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    best_c5 = evaluate_c5(best_h)
