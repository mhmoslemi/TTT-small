            h_norm = np.zeros_like(h_norm)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points

    h = np.clip(initial_h_values, 0.0, 1.0)
    sum_initial = np.sum(h)
    if sum_initial > 0:
        h = h * target_sum / sum_initial
    else:
        h = np.zeros_like(h)

    result = minimize(objective, h, bounds=bounds, constraints=cons, method='L-BFGS-B',
                      tol=1e-8, options={'maxiter': 200})

    best_h = result.x
    best_c5 = result.fun

    best_h_clipped = np.clip(best_h, 0.0, 1.0)
    sum_final = np.sum(best_h_clipped)
    if sum_final > 0:
        best_h = best_h_clipped * target_sum / sum_final
    else:
        best_h = np.zeros_like(best_h_clipped)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
