import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return h

    candidates = [initial_h.copy() for _ in range(5)]
    for i in range(5):
        candidates[i] = clip_and_scale(candidates[i] + np.random.normal(0, 0.05, n_points))
        candidates[i] = clip_and_scale(candidates[i] + 0.1 * np.sin(np.pi/2 * np.arange(n_points)))
        candidates[i] = clip_and_scale(candidates[i][np.random.permutation(n_points)])

    best_h = initial_h.copy()
    best_c5 = obj(best_h)
    start = time.time()

    for _ in range(200):
        if time.time() - start > budget_s:
            break
        new_candidates = []
        for h in candidates:
            corr = np.correlate(clip_and_scale(h), 1 - clip_and_scale(h), mode='full')
            top_lags = np.argsort(corr)[::-1][:3]
            perturbed = h.copy()
            for lag in top_lags:
                if lag < n_points:
                    perturbed[np.arange(n_points - lag)] += np.random.normal(0, 0.05, n_points - lag)
            perturbed = clip_and_scale(perturbed)
            result = minimize(obj, perturbed, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            new_h = result.x
            new_c5 = result.fun
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h
            new_candidates.append(new_h)
        candidates = [new_candidates[i] for i in np.argsort([obj(h) for h in new_candidates])[:5]]

    final_h = clip_and_scale(best_h)
    final_c5 = obj(final_h)
    return (final_h, final_c5, n_points)
