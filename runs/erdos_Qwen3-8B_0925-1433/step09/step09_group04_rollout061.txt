            best_c5 = new_c5
        temp *= cooling_rate

    # Final refinement
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_c5 = compute_c5(h)

    return (h, best_c5, n_points)
