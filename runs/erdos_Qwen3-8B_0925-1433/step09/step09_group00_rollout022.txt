import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    pop_size = 100
    pop = [h]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.15, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate) if np.sum(candidate) > 0 else h
        pop.append(candidate)

    best_h = h
    best_c5 = float('inf')

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(500):
        if time.time() - start_time > budget_s - 50:
            break
        scores = [compute_c5(ind) for ind in pop]
        idxs = np.argsort(scores)
        top_half = idxs[:len(idxs)//2]
        pop = [pop[i] for i in top_half]

        new_pop = []
        while len(new_pop) < pop_size:
            if np.random.rand() < 0.7:
                pa, pb = np.random.choice(top_half, 2, replace=False)
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * pop[pa] + (1 - alpha) * pop[pb]
            else:
                pa = np.random.choice(top_half)
                child = pop[pa] + np.random.normal(0, 0.1, n_points)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child) if np.sum(child) > 0 else pop[pa]
            new_pop.append(child)
        pop = new_pop

        for candidate in pop:
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h
    return (best_h, best_c5, n_points)
