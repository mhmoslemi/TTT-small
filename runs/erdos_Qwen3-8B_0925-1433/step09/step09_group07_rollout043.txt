import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def objective(x):
        h_ = x
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(
        objective, h, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if result.success and np.all(0 <= result.x) and np.all(result.x <= 1):
        best_h = result.x
        best_c5 = objective(result.x)
    else:
        best_h = h
        best_c5 = objective(h)

    return (best_h, best_c5, n_points)
