import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    n_particles = 30
    particles = [initial_h_values.copy() + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) if np.sum(np.clip(p, 0, 1)) > 1e-10 else initial_h_values.copy() for p in particles]
    
    # Evaluate initial solutions
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = np.copy(particles[0])
    best_c5 = compute_c5(particles[0])
    personal_best = [compute_c5(p) for p in particles]
    
    # PSO parameters
    omega = 0.729
    c1, c2 = 1.496, 1.496
    max_vel = 0.1
    max_iter = int(budget_s / 1.5)
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = particles[i] * 0
            for j in range(n_points):
                vel[j] = omega * vel[j] + c1 * r1 * (personal_best[i] - particles[i][j]) + c2 * r2 * (best_c5 - compute_c5(particles[i]))
            # Clip velocity
            vel = np.clip(vel, -max_vel, max_vel)
            # Update position
            particles[i] += vel
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            if np.sum(particles[i]) > 1e-10:
                particles[i] = particles[i] * target_sum / np.sum(particles[i])
            else:
                particles[i] = initial_h_values.copy()
            current_c5 = compute_c5(particles[i])
            if current_c5 < personal_best[i]:
                personal_best[i] = current_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = np.copy(particles[i])
    
    return (best_h, best_c5, n_points)
