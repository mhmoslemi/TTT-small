import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial_h_values
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    
    # Initialize population
    population_size = 50
    pop = []
    
    # Generate from initial with perturbations
    for _ in range(25):
        h_perturb = h_initial + np.random.uniform(-0.1, 0.1, n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb) if np.sum(h_perturb) > 0 else np.zeros(n_points)
        pop.append(h_perturb)
    
    # Generate symmetric patterns
    for _ in range(25):
        h_sym = np.zeros(n_points)
        for i in range(n_points):
            if abs(i - n_points/2) < n_points/4:
                h_sym[i] = 1.0
        h_sym = h_sym * target_sum / np.sum(h_sym)
        pop.append(h_sym)
    
    # Generate periodic patterns
    for _ in range(25):
        h_period = np.zeros(n_points)
        for i in range(n_points):
            segment = int((i / (n_points/5)) % 5)
            h_period[i] = (segment + 1) / 5
        h_period = np.clip(h_period, 0.0, 1.0)
        h_period = h_period * target_sum / np.sum(h_period)
        pop.append(h_period)
    
    pop = np.array(pop)
    best_h = pop[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    # Genetic algorithm loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = corr.max() * dx
            fitness.append(-current_c5)
        
        # Select elites
        elite_idx = np.argsort(fitness)[-int(pop.shape[0]*0.2):]
        elites = pop[elite_idx]
        
        # Select parents with tournament selection
        parents = []
        for _ in range(int(pop.shape[0] * 0.8)):
            t = np.random.choice(pop.shape[0], 4, replace=False)
            winner = t[np.argmax(fitness[t])]
            parents.append(pop[winner])
        
        # Crossover
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            if len(parents) < 2:
                break
            p1, p2 = np.random.choice(parents, 2, replace=False)
            crossover = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:crossover], p2[crossover:]])
            new_pop.append(child)
        
        # Mutate
        for i in range(len(new_pop)):
            h = new_pop[i]
            mutate_idx = np.random.choice(n_points, size=int(np.random.uniform(5, 15)), replace=False)
            for idx in mutate_idx:
                h[idx] += np.random.uniform(-0.1, 0.1)
                h[idx] = np.clip(h[idx], 0.0, 1.0)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            new_pop[i] = h
        
        # Update population
        pop = np.array(new_pop)

        for h in pop:

            current_c5 = corr.max() * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
