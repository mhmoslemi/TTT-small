import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = np.copy(h)

        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    result = opt.minimize(
