        trial = mutant * trial_mask + population[trial_idx] * (1 - trial_mask)
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * target_sum / trial_sum
        trial_c5 = eval_c5(trial)
        target_idx = np.random.randint(len(population))
        target = population[target_idx]
        if trial_c5 < eval_c5(target):
            population[target_idx] = trial.copy()

    best_h = min(population, key=lambda x: eval_c5(x))
    
    def objective(x):
        return eval_c5(x)
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1)] * n_points
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = result.x
    final_c5 = result.fun

    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(final_h + perturb, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * target_sum / new_sum
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
