from scipy.optimize import minimize
import time

        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_sum_ = np.sum(h_)
        if h_sum_ > 0:
            h_ = h_ * target_sum / h_sum_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    best_h, best_c5 = result.x, result.fun
