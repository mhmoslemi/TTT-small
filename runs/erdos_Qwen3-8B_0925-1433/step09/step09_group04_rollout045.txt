    result = minimize(obj, best_h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8, options={'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun

    # Simulated annealing for further refinement
    def simulated_annealing():
        temp = 1.0
        cooling_rate = 0.995
        max_iter = 200
        time_start = time.time()
        current_h = best_h.copy()
        current_c5 = best_c5
        for _ in range(max_iter):
            if time.time() - time_start > budget_s:
                break
            # Perturb current_h
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            sum_perturbed = np.sum(perturbed_h)
            if sum_perturbed > 0:
                perturbed_h = perturbed_h * target_sum / sum_perturbed
            else:
                perturbed_h = np.zeros_like(perturbed_h)
            perturbed_c5 = evaluate_c5(perturbed_h)
            delta_c5 = perturbed_c5 - current_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
                current_h = perturbed_h
                current_c5 = perturbed_c5
            temp *= cooling_rate
        return current_h, current_c5

    best_h, best_c5 = simulated_annealing()

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
