import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial perturbations
    perturbations = []
    for _ in range(3):
        pert = initial_h_values + np.random.normal(0, 0.05, n_points)
        pert = np.clip(pert, 0, 1)
        pert = pert * target_sum / np.sum(pert)
        perturbations.append(pert)
    perturbations.append(initial_h_values)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for h0 in perturbations:
        h = h0.copy()
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m, v = np.zeros(n_points), np.zeros(n_points)
        t = 0
        max_iter = int(budget_s / 0.1)
        start_time = time.time()
        learning_rate = 0.05

        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            t += 1
            corr = np.correlate(h, 1 - h, mode='full')
            idx_max = np.argmax(corr)
            k_max = idx_max - (n_points - 1)
            j = np.arange(n_points) + k_max
            valid_mask = (j >= 0) & (j < n_points)
            valid_j = j[valid_mask]
            valid_i = valid_j - k_max
            grad = np.zeros(n_points)
            grad[valid_i] = 1.0 - h[valid_j]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * target_sum / h_sum
            else:
                h = np.zeros(n_points)

            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            learning_rate *= 0.999

    return (best_h, best_c5, n_points)
