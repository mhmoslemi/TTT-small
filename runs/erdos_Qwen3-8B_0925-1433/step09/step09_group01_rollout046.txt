import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    def scale_array(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return h
        return h * target_sum / sum_h

    # Initial perturbations
    h_candidates = [scale_array(initial_h_values)]
    for _ in range(4):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturb = np.clip(initial_h_values + noise, 0, 1)
        h_candidates.append(scale_array(h_perturb))
        idxs = np.random.choice(n_points, size=2, replace=False)
        h_swap = h_perturb.copy()
        h_swap[idxs[0]], h_swap[idxs[1]] = h_perturb[idxs[1]], h_perturb[idxs[0]]
        h_candidates.append(scale_array(h_swap))
        h_freq = h_perturb * (1 + 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
        h_candidates.append(scale_array(np.clip(h_freq, 0, 1)))

    best_h = scale_array(initial_h_values)
    best_c5 = float('inf')

    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        # Select best candidate
        h = h_candidates[np.argmin([max(np.correlate(c, 1 - c, mode='full')) * 2.0 / n_points for c in h_candidates])]
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        top_lags = np.argsort(corr)[-5:]
        # Targeted perturbation
        perturb = np.zeros_like(h)
        for lag in top_lags:
            if lag < n_points:
                # Positive lag
                pos = lag
                if pos < n_points:
                    for i in range(pos, n_points):
                        perturb[i] += np.random.normal(0, 0.1)
            if lag >= n_points:
                # Negative lag
                pos = 2 * n_points - lag - 1
                if pos > 0:
                    for i in range(pos):
                        perturb[i] += np.random.normal(0, 0.1)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = scale_array(h_perturb)
        # Optimization
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h == 0:
                return max_corr
            h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return max(corr) * 2.0 / n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x
            c5 = max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * 2.0 / n_points
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
                h_candidates.append(h_candidate)
                h_candidates.pop(0)
        else:
            h_candidates.append(h_perturb)
            h_candidates.pop(0)
    return (best_h, best_c5, n_points)
