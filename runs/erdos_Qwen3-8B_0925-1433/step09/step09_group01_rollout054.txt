import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values)
    h = np.clip(initial_h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h_current = np.clip(best_h, 0, 1)
        if np.sum(h_current) > 0:
            h_current = h_current * target_sum / np.sum(h_current)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        worst_lag = np.argmax(corr)
        if worst_lag >= n_points:
            shift = worst_lag - n_points + 1
        else:
            shift = -worst_lag

        perturb = np.zeros(n_points)
        if shift > 0:
            for i in range(n_points - shift):
                perturb[i] += np.random.normal(0, 0.02, 1)
        else:
            for i in range(shift, n_points):
                perturb[i] += np.random.normal(0, 0.02, 1)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_final = np.clip(best_h, 0, 1)
    if np.sum(best_final) > 0:
        best_final = best_final * target_sum / np.sum(best_final)
    final_c5 = obj(best_final)
    return (best_final, final_c5, n_points)
