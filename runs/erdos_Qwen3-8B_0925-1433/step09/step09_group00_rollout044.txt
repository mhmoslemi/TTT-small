import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,

        tol=1e-10,
