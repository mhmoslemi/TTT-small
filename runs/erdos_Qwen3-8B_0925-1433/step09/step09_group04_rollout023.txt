                            trial_h = trial_h * target_sum / sum_trial
                        else:
                            trial_h = np.zeros_like(trial_h)
                        trial_c5 = evaluate_c5(trial_h)
                        if trial_c5 < current_c5:
                            h = trial_h
                            break

    final_h = h.copy()
    final_c5 = evaluate_c5(final_h)
    final_h = np.clip(final_h, 0, 1)
    sum_h = np.sum(final_h)
    if sum_h > 0:
        final_h = final_h * target_sum / sum_h
    final_c5 = evaluate_c5(final_h)
    return (final_h, final_c5, n_points)
