            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            target = population[i]
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)
            trial_c5 = compute_c5(trial)
            if trial_c5 < compute_c5(target):
                new_population.append(trial)
            else:
                new_population.append(target)
        population = new_population

    return (best_h, best_c5, n_points)
