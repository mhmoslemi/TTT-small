import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.clip(initial_h_values, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(x, scale, freq):
        x_pert = np.copy(x)
        noise = np.random.normal(0, scale, n_points)
        x_pert += noise
        x_pert = np.clip(x_pert, 0, 1)
        if np.sum(x_pert) > 0:
            x_pert = x_pert * target_sum / np.sum(x_pert)
        return x_pert

    def generate_start_points(x0, scale, freq, num=5):
        return [perturb(x0, scale, freq) for _ in range(num)]

    start_time = time.time()
    best_h = initial_h
    best_c5 = obj(best_h)

    # Initial diverse starts
    for scale in [0.05, 0.1, 0.15]:
        starts = generate_start_points(initial_h, scale, 1.0)
        for s in starts:
            res = minimize(obj, s, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    # Adaptive perturbation loop
    noise_scale = 0.05
    iterations = 0
    while time.time() - start_time < budget_s - 10:
        iterations += 1
        noise_scale *= 0.995
        pert = np.random.normal(0, noise_scale, n_points)
        new_h = np.clip(best_h + pert, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    best_c5 = res.fun
    return (res.x, best_c5, n_points)
