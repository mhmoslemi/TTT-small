from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        initial_h = np.zeros(45)
        initial_h[::5] = 1.0
        initial_h = initial_h[::-1] + initial_h
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * 45 / 2 / np.sum(initial_h)
    
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return eval_c5(x)

    bounds = [(0, 1)] * n_points

    # Differential Evolution
    result_de = differential_evolution(
        fun=obj,
        bounds=bounds,
        strategy='best1bin',
        mutation=(0.5, 1.0),
        recombination=0.9,
        popsize=20,
        tol=1e-4,
        maxiter=100,
        seed=seed,
        polish=True,
        callback=None,
        disp=False
    )
    best_x = result_de.x
    best_c5 = eval_c5(best_x)

    # Local refinement with SLSQP
    def cons(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': cons}]
    res = minimize(obj, best_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'ftol': 1e-9, 'maxiter': 100})
    final_h = res.x
    final_c5 = res.fun

    # Final refinement
    for _ in range(5):
        if time.time() - time.time() >= budget_s:
            break

        new_h = np.clip(final_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)

        if res.fun < final_c5:
            final_h, final_c5 = res.x, res.fun

    return (final_h, final_c5, n_points)
