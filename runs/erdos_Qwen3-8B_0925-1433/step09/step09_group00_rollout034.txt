        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(x, method='sin'):
        if method == 'sin':
            return x + 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        elif method == 'walk':

        return x

    best_h = h.copy()
    best_c5 = obj(h)
    candidates = [h]

    start_time = time.time()
