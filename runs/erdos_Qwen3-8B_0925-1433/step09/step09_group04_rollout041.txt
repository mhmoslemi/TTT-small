import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h_norm = np.clip(h, 0, 1)
        if np.sum(h_norm) > 0:
            h_norm = h_norm * target_sum / np.sum(h_norm)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Initialize population with diverse variants
    population_size = 20
    population = [initial_h_values.copy() for _ in range(population_size)]
    for i in range(population_size):
        hp = population[i]
        if np.sum(hp) > 0:
            hp = hp * target_sum / np.sum(hp)
        population[i] = hp

    # Add transformed variants
    for i in range(population_size):
        hp = population[i].copy()
        # Shifted version
        shift = np.random.uniform(-0.25, 0.25)
        shifted = np.roll(hp, int(shift * n_points / 2))
        shifted = np.clip(shifted, 0, 1) * target_sum / np.sum(shifted)
        population.append(shifted)
        # Noised version
        noised = hp + np.random.normal(0, 0.05, size=hp.shape)
        noised = np.clip(noised, 0, 1) * target_sum / np.sum(noised)
        population.append(noised)
    population = np.array(population)[:population_size]

    # DE parameters
    F = 0.8
    CR = 0.9
    max_gen = 200
    time_start = time.time()
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            j, k, l = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[j], population[k], population[l]
            mutant = a + F * (b - c)
            trial = a.copy()
            for m in range(n_points):
                if np.random.rand() < CR:
                    trial[m] = mutant[m]
            trial_clipped = np.clip(trial, 0, 1)
            trial_norm = trial_clipped * target_sum / np.sum(trial_clipped) if np.sum(trial_clipped) > 0 else np.zeros_like(trial_clipped)
            c5_trial = evaluate_c5(trial_norm)
            c5_current = evaluate_c5(population[i])
            if c5_trial < c5_current:
                new_population.append(trial_norm)
            else:
                new_population.append(population[i])
            if c5_trial < best_c5:
                best_h = trial_norm
                best_c5 = c5_trial
        population = new_population

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    initial_guess = best_h
    def obj(x):

        if np.sum(h_norm) > 0:
            h_norm = h_norm * target_sum / np.sum(h_norm)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Final check and return
    best_h = np.clip(best_h, 0, 1)

    if sum_h > 0:

    best_c5 = evaluate_c5(best_h)
