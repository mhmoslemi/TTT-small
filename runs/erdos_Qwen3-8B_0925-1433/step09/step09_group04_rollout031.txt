import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Initialize population using initial_h_values
    population_size = 20
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(population_size)]
    for i in range(population_size):
        if np.sum(population[i]) > 0:
            population[i] = population[i] * target_sum / np.sum(population[i])
        else:
            population[i] = np.zeros_like(population[i])

    # DE parameters
    F = 0.8
    CR = 0.9
    max_gen = 200
    time_start = time.time()
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            indices = np.random.choice(len(population), 3, replace=False)
            rand1, rand2, rand3 = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = rand1 + F * (rand2 - rand3)
            trial = rand1.copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            # Clip and normalize
            trial_clipped = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial_clipped)
            if sum_trial > 0:
                trial_norm = trial_clipped * target_sum / sum_trial
            else:
                trial_norm = np.zeros_like(trial_clipped)
            # Evaluate and select
            c5_trial = evaluate_c5(trial_norm)
            c5_current = evaluate_c5(population[i])
            if c5_trial < c5_current:
                new_population.append(trial_norm)
            else:
                new_population.append(population[i])
            if c5_trial < best_c5:
                best_h = trial_norm
                best_c5 = c5_trial
        population = new_population

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    initial_guess = best_h
    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:

        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    result = minimize(obj, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
