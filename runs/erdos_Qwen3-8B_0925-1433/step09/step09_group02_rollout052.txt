import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in locals() else None
    n_points = len(initial_h_values) if initial_h_values is not None else 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    if initial_h_values is None:
        x = np.linspace(0, 2, n_points)
        freqs = [0.25, 0.167, 0.125]
        h = np.zeros(n_points)
        for f in freqs:
            w = 2*np.pi*f*x
            h += np.sin(w)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = h * target_sum / np.sum(h)
    else:
        h = initial_h_values.copy()
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
