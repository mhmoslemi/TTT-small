            corr_test = np.correlate(x_test_norm, 1 - x_test_norm, mode='full')
            grad[i] = (np.max(corr_test) * dx - max_corr * dx) / 1e-5
        return grad

    population_size = 20
    population = [np.clip(initial_h_values, 0, 1).copy() for _ in range(population_size)]
    for i in range(population_size):
        if np.sum(population[i]) > 0:
            population[i] = population[i] * target_sum / np.sum(population[i])
        else:
            population[i] = np.zeros_like(population[i])

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    max_gen = 100
    time_start = time.time()
    for gen in range(max_gen):
        if time.time() - time_start > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            rand_indices = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[rand_indices[0]], population[rand_indices[1]], population[rand_indices[2]]
            mutant = a + 0.8 * (b - c)
            trial = a.copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial_clipped = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial_clipped)
            if sum_trial > 0:
                trial_norm = trial_clipped * target_sum / sum_trial
            else:
                trial_norm = np.zeros_like(trial_clipped)
            c5_trial = evaluate_c5(trial_norm)
            c5_current = evaluate_c5(population[i])
            if c5_trial < c5_current:
                new_population.append(trial_norm)
            else:
                new_population.append(population[i])
            if c5_trial < best_c5:
                best_h = trial_norm
                best_c5 = c5_trial
        population = new_population

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    initial_guess = best_h
    result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
