from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def init_h():
        h = np.clip(initial_h_values, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        return h

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 0:
            h = h * target_sum / total
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = init_h()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def obj(x):
        return compute_c5(clip_and_scale(x))

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

    def optimize(start):
        res = minimize(obj, start, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            return res.x
        return start

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)

        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        improved = optimize(new_h)
