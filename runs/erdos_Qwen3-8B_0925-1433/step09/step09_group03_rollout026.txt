import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def optimize(pattern):
        h = np.clip(pattern, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        return res.x, res.fun

    start_time = time.time()
    best_h, best_c5 = initial_h_values, obj(initial_h_values)
    patterns = [initial_h_values]

    x = np.linspace(0, 2, n_points)
    for _ in range(4):
        if time.time() - start_time > budget_s:

        # Binary

        b = b * target_sum / np.sum(b)
