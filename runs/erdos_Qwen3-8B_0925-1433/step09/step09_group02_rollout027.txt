import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def update_directional(h_current):
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        if max_corr == 0:
            return h_current
        h_dir = h_current.copy()
        for i in range(n_points):
            j = (i + max_k) % n_points
            if h_dir[i] > 0.5 and h_dir[j] > 0.5:
                h_dir[i] -= 0.02
                h_dir[j] -= 0.02
        h_dir = np.clip(h_dir, 0, 1)
        h_dir = h_dir * target_sum / np.sum(h_dir) if np.sum(h_dir) > 0 else h_current
        return h_dir

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_h = update_directional(best_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_h = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
