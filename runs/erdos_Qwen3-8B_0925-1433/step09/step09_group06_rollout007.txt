            if np.sum(mutant) > 0:
                mutant = (mutant * target_sum) / np.sum(mutant)
            # Crossover
            target = population[i]
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            # Evaluate trial
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c = corr.max() * dx
            if trial_c < current_c5[i]:
                new_population.append(trial)
            else:
                new_population.append(target)
        population = new_population
        # Update best
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c = corr.max() * dx
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = final_corr.max() * dx
    return (best_h, final_c5, n_points)
