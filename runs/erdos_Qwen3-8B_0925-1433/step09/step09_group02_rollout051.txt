        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Focus perturbations on high variance regions
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_k = np.argmax(corr)
        k_rel = max_k - (len(corr) // 2)

        # Introduce a spatially correlated perturbation
        perturb = np.random.normal(0, 0.015, n_points)
        perturb = np.roll(perturb, int(np.random.randint(0, n_points // 4)))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
