    target_sum = n_points / 2.0

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        best_h = np.copy(h)
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        start_time = time.time()
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            idx_max = np.argmax(corr)
            k_max = idx_max - (n_points - 1)

            valid_mask = (j >= 0) & (j < n_points)

            grad[~valid_mask] = 0.0
            m = beta1 * m + (1 - beta1) * grad
