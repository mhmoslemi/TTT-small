import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    start_time = time.time()

    def perturb(h_vec, max_corr, idxs):
        h_perturb = h_vec.copy()
        for i in idxs:
            h_perturb[i] += np.random.normal(0, 0.05)
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_p = np.sum(h_perturb)
        if sum_p > 0:
            h_perturb = h_perturb * target_sum / sum_p
        return h_perturb

    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_val = np.max(corr)
        idxs = np.argsort(corr)[-3:]
        # Apply targeted perturbation to top 3 lags
        h_perturb = perturb(best_h, max_val, idxs)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final optimization
    res_final = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    best_c5 = res_final.fun
    return (res_final.x, best_c5, n_points)
