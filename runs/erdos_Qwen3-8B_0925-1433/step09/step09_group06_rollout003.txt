import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    sum_h = np.sum(h)
    if sum_h > 0:
        h *= target_sum / sum_h
    else:

            G[i] = 1.0 - h[j]
        velocity = momentum * velocity - learning_rate * G
        h += velocity

        sum_h = np.sum(h)
        if sum_h > 0:
            h *= target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = current_corr.max() * dx
        if current_c5 < best_c5:

            best_h = h.copy()

        if no_improvement_count >= max_no_improvement:

        if not improved and learning_rate < 1e-6:
