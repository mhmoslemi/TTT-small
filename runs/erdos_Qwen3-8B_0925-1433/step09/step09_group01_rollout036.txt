from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    for _ in range(150):

        worst_k = max_corr_index - (n_points - 1)

            perturbed = h_.copy()
            perturbed[indices] += noise
            perturbed = np.clip(perturbed, 0, 1)
