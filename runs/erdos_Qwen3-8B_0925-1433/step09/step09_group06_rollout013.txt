import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    pop_size = 50
    elite_ratio = 0.2
    mutation_rate = 0.15
    crossover_prob = 0.9
    tournament_size = 4
    generations = 100
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr * dx)
    
    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h) if np.sum(h) > 0 else 1.0
        return h * scale
    
    def generate_symmetric():
        return np.zeros(n_points) + 0.5
    
    def generate_periodic():
        return np.tile([0, 1, 0, 1, 0.5], n_points // 5)
    
    population = [h.copy()]
    for _ in range(pop_size - 1):
        if np.random.rand() < 0.7:
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            candidate = h + perturb
        else:
            candidate = generate_symmetric() if np.random.rand() < 0.5 else generate_periodic()
        population.append(clip_rescale(candidate))
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(ind) for ind in population])
        elite_indices = np.argsort(fitness)[:int(pop_size * elite_ratio)]
        elites = [population[i] for i in elite_indices]
        
        parents = []
        for _ in range(pop_size - len(elites)):
            tournament = np.random.choice(population, tournament_size)
            winner = tournament[np.argmax(fitness[np.argsort([np.random.rand() for _ in range(tournament_size)])])]
            parents.append(winner)
        
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents) and np.random.rand() < crossover_prob:
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = parents[i].copy()
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] += np.random.uniform(-0.1, 0.1, np.sum(mutation_mask))
            child = clip_rescale(child)
            offspring.append(child)
        
        population = elites + offspring
        current_best = np.argmax(fitness)
        if evaluate(population[current_best]) < evaluate(best_h):
            best_h = population[current_best].copy()
            best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
