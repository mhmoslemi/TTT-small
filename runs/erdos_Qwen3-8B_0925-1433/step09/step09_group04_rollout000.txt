    result = minimize(obj, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
