            h = h * target_sum / np.sum(h)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    max_iterations = 100000
    learning_rate = 0.5
    momentum = 0.8
    decay_rate = 0.995
    grad_std_clip = 1.0
    
    start_time = time.time()
    prev_update = np.zeros(n_points)
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        
        G = np.zeros(n_points)
        for i in range(n_points):
            j = (i - lag) % n_points
            G[i] = (1 - h[j]) * dx

        if std_dev > 0:
            G = G * grad_std_clip / std_dev

        h -= update
        prev_update = update

    final_c5 = final_corr.max() * dx
