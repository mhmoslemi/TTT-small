from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        initial_h = initial_h * (n_points / 2)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        return h

    def de_population(n_pop=30):
        pop = [initial_h.copy()]
        for _ in range(n_pop - 1):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = initial_h + perturb
            new_h = clip_and_scale(new_h)
            pop.append(new_h)
        return np.array(pop)

    population = de_population()
    best_x = initial_h.copy()
    best_c5 = eval_c5(best_x)

    start_time = time.time()

        target = population[target_idx]
        trial = target.copy()
        for i in range(n_points):
            if np.random.rand() < CR:

        gen += 1

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    res = minimize(eval_c5, best_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
