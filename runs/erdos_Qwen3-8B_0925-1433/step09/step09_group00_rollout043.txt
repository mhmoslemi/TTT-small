import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = obj(best_h)

    start_time = time.time()

    # Generate structured perturbations
    for _ in range(10):
        if time.time() - start_time > budget_s - 60:
            break
        # Sinusoidal perturbation
        perturb = 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else np.zeros(n_points)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Local refinement
    for _ in range(50):
        if time.time() - start_time > budget_s - 60:
            break
        perturb = 0.05 * np.random.normal(0, 1, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else np.zeros(n_points)
        res = minimize(obj, new_h, method='SLSQP', bounds=[(0,1)]*n_points, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
