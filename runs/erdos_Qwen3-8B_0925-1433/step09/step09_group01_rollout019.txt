import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    start_h = np.clip(initial_h_values, 0, 1)
    start_h = start_h * target_sum / np.sum(start_h) if np.sum(start_h) > 0 else np.zeros(n_points)

    start_time = time.time()

    def optimize(start):
        x0 = np.clip(start, 0, 1)
        if np.sum(x0) > 0:
            x0 = x0 * target_sum / np.sum(x0)
        result = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if result.success:
            return result.x, result.fun
        return np.zeros(n_points), np.inf

    best_h, best_c5 = start_h, obj(start_h)
    found = True

    # Perturbations
    starts = [start_h]
    for _ in range(3):
        shifted = np.roll(start_h, 1)
        starts.append(np.clip(shifted, 0, 1) * target_sum / np.sum(np.clip(shifted, 0, 1)))
        h_swapped = start_h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        h_swapped[[i, j]] = h_swapped[[j, i]]
        starts.append(np.clip(h_swapped, 0, 1) * target_sum / np.sum(np.clip(h_swapped, 0, 1)))
        sin_wave = 0.1 * np.sin(2 * np.pi * np.arange(n_points) * dx - 0.5)
        starts.append(np.clip(start_h + sin_wave, 0, 1) * target_sum / np.sum(np.clip(start_h + sin_wave, 0, 1)))

    for start in starts:
        h_opt, c5 = optimize(start)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        if time.time() - start_time > budget_s:
            found = False
            break

    if found:
        for _ in range(5):
            if time.time() - start_time > budget_s:

            h_opt, c5 = optimize(new_h)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5
