from scipy.optimize import minimize
import time

        return np.max(corr) * dx

            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            pop.append(new_h)
        return np.array(pop)

    population = de_population()
