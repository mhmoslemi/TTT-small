import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Generate diverse initial patterns
    patterns = []
    x = np.linspace(0, 2, n_points)
    patterns.append((1 - np.cos(np.pi * x)) * target_sum / np.sum(1 - np.cos(np.pi * x)))
    h_bin = np.zeros(n_points)
    h_bin[np.random.choice(n_points, size=int(n_points//2), replace=False)] = 1
    patterns.append(h_bin * target_sum / np.sum(h_bin))
    h_clust = np.zeros(n_points)
    h_clust[:n_points//3] = 1.0
    h_clust[n_points//3:2*n_points//3] = 0.5
    h_clust[2*n_points//3:] = 0.2
    patterns.append(h_clust * target_sum / np.sum(h_clust))
    patterns.append(np.zeros(n_points))
    patterns.append(initial_h_values[::-1] * target_sum / np.sum(initial_h_values[::-1]))

    best_h, best_c5 = initial_h_values.copy(), float('inf')

    for h in patterns:
        try:
            result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if result.success and result.fun < best_c5:
                best_c5, best_h = result.fun, result.x
        except:
            pass

    # Final refinement
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    return (result.x, result.fun, n_points)
