import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)

    def generate_initializations():
        x = np.linspace(0, 2, n_points)
        yield 1 - np.cos(np.pi * (x - np.random.uniform(0, 2))) / 2
        h = np.random.choice([0, 1], size=n_points)
        cnt = np.sum(h)
        delta = target_sum - cnt
        if delta > 0:
            h[np.where(h == 0)[:delta]] = 1
        elif delta < 0:
            h[np.where(h == 1)[:int(-delta)]] = 0
        yield h
