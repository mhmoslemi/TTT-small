import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values)

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    best_h = initial_h.copy()

    start_time = time.time()
