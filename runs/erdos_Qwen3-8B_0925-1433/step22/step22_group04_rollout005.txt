import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_block(n):
        return np.array([1.0]*n//2 + [0.0]*(n - n//2))

    def generate_sinusoidal(n):
        x = np.linspace(0, 2, n)
        return 0.5 + 0.5 * np.sin(np.pi * x)

    h_initial = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    h_block = generate_block(n_points) * target_sum / np.sum(generate_block(n_points))
    h_sinusoidal = generate_sinusoidal(n_points) * target_sum / np.sum(generate_sinusoidal(n_points))

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [ (0, 1) for _ in range(n_points) ]

    best_c5 = float('inf')
    best_h = h_initial.copy()

    def optimize_pattern(h):
        h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 200})
        if res.success:
            return res.x
        return h

    start = time.time()
    for h in [h_initial, h_block, h_sinusoidal]:
        h = optimize_pattern(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

    for _ in range(2):
        if time.time() - start > budget_s:
            break
        h = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
