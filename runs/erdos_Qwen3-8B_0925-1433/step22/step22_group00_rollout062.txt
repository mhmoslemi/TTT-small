import numpy as np
import time

def rescale_to_sum(h, target_sum):
    clipped = np.clip(h, 0, 1)
    current_sum = np.sum(clipped)
    if current_sum == 0:
        return np.zeros_like(clipped)
    scale = target_sum / current_sum
    return clipped * scale

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    pop_size = 50
    population = [h.copy()]
    for _ in range(pop_size - 1):
