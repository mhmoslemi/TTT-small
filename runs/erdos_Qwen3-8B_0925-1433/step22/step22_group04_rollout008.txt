import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(x):

        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = (0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()
    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(compute_c5, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
