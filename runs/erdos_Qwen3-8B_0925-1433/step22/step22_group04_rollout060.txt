import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def generate_seed():
        h = np.zeros(n_points)
        h[:int(n_points/2)] = 1.0
        h[int(n_points/2):] = 0.0
        h = h * target_sum / np.sum(h)
        return h
    
    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def generate_jittered():
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    seeds = [generate_seed(), generate_sinusoidal(), generate_jittered()]
    best_h = seeds[0].copy()
    best_c5 = np.max(np.correlate(seeds[0], 1 - seeds[0], mode='full')) * dx
    
    for h in seeds:
        h = h.copy()
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        def obj(x):
            hx = x * target_sum / np.sum(x)
            corr = np.correlate(hx, 1 - hx, mode='full')
            return np.max(corr) * dx
        
        def compute_gradient(x):
            hx = x * target_sum / np.sum(x)
            corr = np.correlate(hx, 1 - hx, mode='full')
            k_max = np.argmax(corr)
            if k_max >= len(corr):
                k_max = len(corr) - 1
            shift = k_max - (n_points - 1)
            grad = np.zeros_like(x)
            for i in range(n_points):
                i_plus = i + shift
                if 0 <= i_plus < n_points:
                    term1 = 1.0 - hx[i_plus]
                else:
                    term1 = 0.0
                i_minus = i - shift
                if 0 <= i_minus < n_points:
                    term2 = hx[i_minus]
                else:
                    term2 = 0.0
                grad[i] = term1 - term2
            return grad
        
        learning_rate = 0.05
        momentum = 0.9
        decay_rate = 0.999
        max_iterations = 500
        start_time = time.time()
        
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = momentum * m[i] + (1 - momentum) * grad[i]
                v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
                biased_m = m[i] / (1 - momentum ** (_ + 1))
                biased_v = v[i] / (1 - decay_rate ** (_ + 1))
                h[i] = h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        
        res = minimize(obj, h, method='SLSQP', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    def perturb_and_refine(h):
