from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def perturb(h, scale=0.1):
        noise = np.random.normal(0, scale, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def shift(h, shift_amount=1):
        h = np.roll(h, shift_amount)
        h = h * target_sum / np.sum(h)
        return h

    def swap_blocks(h, block_size=5):
        h = h.copy()
        if block_size > n_points:
            block_size = n_points // 2
        start = np.random.randint(0, n_points - block_size)
        end = start + block_size
        h[start:end], h[end:end+block_size] = h[end:end+block_size], h[start:end]
        h = h * target_sum / np.sum(h)
        return h

    def evolve(parent):
        children = [parent]
        for _ in range(4):
            children.append(perturb(parent))
            children.append(shift(parent))
            children.append(swap_blocks(parent))
        return children

    start_time = time.time()
