import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def optimize_initial(seed):
        h = seed.copy()
        for _ in range(300):
            if time.time() - start_time > budget_s:
                return h
            grad = compute_gradient(h)
            m = np.zeros_like(grad) if _ == 0 else m
            v = np.zeros_like(grad) if _ == 0 else v
            for i in range(n_points):
                m[i] = 0.9 * m[i] + 0.1 * grad[i]
                v[i] = 0.999 * v[i] + 0.001 * grad[i] ** 2
                h[i] = h[i] + 0.01 * m[i] / (np.sqrt(v[i]) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        return h

    def refine_with_slsqp(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        return best_h

    start_time = time.time()
    seeds = [
        np.array([1 if i % 2 == 0 else 0 for i in range(n_points)]) * target_sum / np.sum([1 if i % 2 == 0 else 0 for i in range(n_points)]),
        np.sin(np.linspace(0, np.pi, n_points) * 0.5) * 0.5 + 0.5,
        np.clip(initial_h_values + np.random.normal(0, 0.05, n_points), 0, 1) * target_sum / np.sum(np.clip(initial_h_values + np.random.normal(0, 0.05, n_points), 0, 1))
    ]
    for seed in seeds:
        seed = seed * target_sum / np.sum(seed)
        h = optimize_initial(seed)
        h = refine_with_slsqp(h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
