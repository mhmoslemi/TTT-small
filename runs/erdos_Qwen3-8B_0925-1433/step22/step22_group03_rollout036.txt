from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    else:
        n_points = 45
        h_pop = [np.random.rand(n_points) * 2 - 1]
    
    # Normalize initial population
    for i in range(len(h_pop)):
        h_pop[i] = h_pop[i] * (n_points / 2) / np.sum(h_pop[i])
        h_pop[i] = np.clip(h_pop[i], 0, 1)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h_pop[0].copy()
    best_c5 = compute_c5(best_h)
    
    for gen in range(200):
        if time.time() - time.time() > budget_s:
            break
        # Evaluate population
        fitness = [compute_c5(h) for h in h_pop]
        # Select top 50 individuals
        idx = np.argsort(fitness)[:50]
        selected = [h_pop[i] for i in idx]
        # Generate new population
        new_pop = []
        for _ in range(len(selected)):
            a, b = np.random.choice(selected, 2, replace=False)
            child = (a + np.random.normal(0, 0.02, n_points) + b) / 2
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_pop.append(child)
        h_pop = new_pop
        # Update best
        for h in h_pop:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
    
    # Local refinement
    def obj(x):
        h = x * (n_points / 2) / np.sum(x)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_c5 = compute_c5(best_h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
