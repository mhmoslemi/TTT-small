import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Symmetrized initial guess
    h_sym = np.array([1.0 - h[n_points - 1 - i] for i in range(n_points)], dtype=float)
    h_sym = h_sym * target_sum / np.sum(h_sym)
    best_c5 = min(best_c5, np.max(np.correlate(h_sym, 1 - h_sym, mode='full')) * dx)

    # Randomized initial guess
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)
    best_c5 = min(best_c5, np.max(np.correlate(h_rand, 1 - h_rand, mode='full')) * dx)

    # Function to evaluate C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Function to optimize with SLSQP
    def optimize_with_slsqp(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(lambda x: compute_c5(x), h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x if res.success else h

    # Function to optimize with gradient descent
    def optimize_with_gradient(h):
        h = h.copy()
        lr = 0.5
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m, v = np.zeros(n_points), np.zeros(n_points)
        for _ in range(300):
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            shift = k_max - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                i_plus = i + shift
                if 0 <= i_plus < n_points:
                    term1 = 1.0 - h[i_plus]
                else:
                    term1 = 0.0
                i_minus = i - shift
                if 0 <= i_minus < n_points:
                    term2 = h[i_minus]
                else:
                    term2 = 0.0
                grad[i] = term1 - term2
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**( _ + 1))
            v_hat = v / (1 - beta2**( _ + 1))
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
