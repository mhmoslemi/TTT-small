        mut = np.clip(mut, 0, 1)
        mut = mut * target_sum / np.sum(mut)
        population.append(mut)
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        idx = np.random.randint(len(population))
        omega = np.random.uniform(0, 2 * np.pi)
        phi = np.random.uniform(0, 2 * np.pi)
        mut = 0.5 + 0.5 * np.cos(omega * np.arange(n_points) + phi)
        mut = np.clip(mut, 0, 1)
        mut = mut * target_sum / np.sum(mut)
        h_new = mut.copy()
        res = minimize(obj, h_new, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': 100})
        if res.success:
            population[idx] = res.x
            current_c5 = obj(res.x)
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
