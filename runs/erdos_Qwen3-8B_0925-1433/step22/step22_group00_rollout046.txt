import numpy as np
import time

initial_h_values = []

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def rescale_to_sum(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0, 1)
        return h

    def evaluate_candidate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = []
    for _ in range(50):
        h = np.random.uniform(0, 1, n_points)
        h = rescale_to_sum(h, target_sum)
        population.append(h)

    best_h = population[0]
    best_c5 = evaluate_candidate(best_h)

    start_time = time.time()
    max_iter = int(budget_s // 1)
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(len(population)):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + 0.8 * (b - c)
            trial = np.copy(a)
            for i in range(n_points):
                if np.random.rand() < 0.9:
                    trial[i] = mutant[i]
            trial = rescale_to_sum(trial, target_sum)
            trial_c5 = evaluate_candidate(trial)
            if trial_c5 < evaluate_candidate(population[0]):
                new_population.append(trial)
            else:
                new_population.append(population[0])
        population = new_population
        curr_c5 = evaluate_candidate(population[0])
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = population[0]

    best_h = np.clip(best_h, 0, 1)
