        if np.sum(h) == 0:
            return 1.0
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    initial_samples = [initial_h.copy()]
    for _ in range(5):
        sample = initial_h.copy()
        sample += np.random.normal(0, 0.05, n_points)
        sample = np.clip(sample, 0, 1)
        sample = sample * target_sum / np.sum(sample)
        initial_samples.append(sample)
    
    X = np.array([s for s in initial_samples])
    y = np.array([-compute_c5(s) for s in initial_samples])
    
    kernel = ConstantKernel(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-3, 1e3))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
    gp.fit(X, y)
    
    best_c5 = -y[0]
    best_h = initial_samples[0].copy()
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        candidate = best_h.copy()
        candidate += perturbation
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        
        current_y = -compute_c5(candidate)
        X_new = np.array([candidate])
        y_new = np.array([current_y])
        X = np.vstack((X, X_new))
        y = np.hstack((y, y_new))
        
        gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
        gp.fit(X, y)
        
        if current_y > best_c5:
            best_c5 = current_y
            best_h = candidate.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, -best_c5, n_points)
