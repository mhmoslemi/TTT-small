import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.9995
    max_iter = int(budget_s / (1.1 * n_points))
    decay = 0.995
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Evaluate current best
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Focus on top 3 worst lags
        worst_lags = np.argsort(corr)[-3:]
        
        # Generate perturbation
        perturbation = np.zeros(n_points)
        for lag in worst_lags:
            for i in range(n_points):
                j = (i + lag) % n_points
                if best_h[i] > 0.6 and best_h[j] > 0.6:
                    perturbation[i] -= 0.015
                    perturbation[j] -= 0.015
        
        # Apply perturbation
        trial_h = best_h + perturbation
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = trial_h * target_sum / np.sum(trial_h)
        
        # Compute new C5
        corr_trial = np.correlate(trial_h, 1 - trial_h, mode='full')
        new_c5 = np.max(corr_trial) * dx
        
        # Acceptance
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            best_h = trial_h.copy()
        
        # Cool down
        T *= decay
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    best_c5 = min(best_c5, final_c5)
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
