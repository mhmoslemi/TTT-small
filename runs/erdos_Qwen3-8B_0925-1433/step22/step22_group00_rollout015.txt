def run(seed=42, budget_s=600):

    population = [h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.uniform(0, 1, n_points)
        perturb = rescale_to_sum(perturb)

        best_idx = np.argmin(evals)
