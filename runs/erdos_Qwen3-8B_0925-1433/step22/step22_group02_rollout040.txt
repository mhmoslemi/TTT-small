import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')
    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    def evolve_guess(guess):
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
        if res.success:
            h = res.x
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            return h, c5
        return guess, float('inf')

    # Generate initial guesses
    guesses = [np.array(initial_h_values), np.flip(initial_h_values)]
    if n_points % 2 == 0:
        mid = n_points // 2
        symmetrical = np.zeros(n_points)
        for i in range(mid):
            symmetrical[i] = initial_h_values[i]
            symmetrical[n_points - 1 - i] = initial_h_values[i]
        guesses.append(symmetrical)
    else:
        mid = n_points // 2
        symmetrical = np.zeros(n_points)
        for i in range(mid):
            symmetrical[i] = initial_h_values[i]
            symmetrical[n_points - 1 - i] = initial_h_values[i]
        symmetrical[mid] = initial_h_values[mid]
        guesses.append(symmetrical)
    guesses.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    for guess in guesses:
        if time.time() - start_time > budget_s:

        h, c5 = evolve_guess(guess * target_sum / np.sum(guess))
