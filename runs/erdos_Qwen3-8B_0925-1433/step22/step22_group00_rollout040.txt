        new_pop = []
        for h in elites:
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr[:n_points])
            lag = max_index
            h_mut = h.copy()
            x_range = np.arange(n_points - lag)
            if len(x_range) > 0:
                selected_x = np.random.choice(x_range, size=5, replace=False)
                for x in selected_x:
                    h_mut[x] += np.random.normal(0, 0.01)
                    h_mut[x + lag] += np.random.normal(0, 0.01)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut_sum = np.sum(h_mut)
            if h_mut_sum == 0:
                h_mut = np.random.rand(n_points)
                h_mut_sum = np.sum(h_mut)
            scale = target_sum / h_mut_sum
            h_mut *= scale
            new_pop.append(h_mut)
        
        combined = population + new_pop
        combined_sorted = sorted(combined, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
        population = combined_sorted[:len(combined) // 2]

        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr[:n_points]) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
