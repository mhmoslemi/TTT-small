import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    population = [h.copy()]
    for _ in range(9):
        pert = np.random.normal(0, 0.05, n_points)
        h = np.clip(initial_h_values + pert, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for i in range(9):
            x0 = population[i].copy()

                          tol=1e-7)

                population.append(new_h)
                population.pop(np.argmin([obj(h) for h in population]))
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:

                best_c5 = curr_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
