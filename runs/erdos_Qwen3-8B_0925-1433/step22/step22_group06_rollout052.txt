    best_h = h.copy()

    start_time = time.time()

    n_iter = 0
    convergence = False

    while time.time() - start_time < budget_s:

        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)

        gradient = np.zeros_like(h)
