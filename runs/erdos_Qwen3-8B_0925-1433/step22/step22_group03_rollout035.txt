import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    F, CR, pop_size, gen_max = 0.8, 0.9, 30, 200
    start_time = time.time()
    
    # Initialize population
    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    pop = [h0.copy() for _ in range(pop_size)]
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h, best_c5 = pop[0].copy(), eval_c5(pop[0])
    
    for _ in range(gen_max):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = v[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum
            curr_c5 = eval_c5(trial)
            if curr_c5 < eval_c5(pop[i]):
                pop[i] = trial
                if curr_c5 < best_c5:
                    best_h, best_c5 = trial.copy(), curr_c5
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
