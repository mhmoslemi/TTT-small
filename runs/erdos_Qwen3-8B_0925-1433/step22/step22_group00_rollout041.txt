import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
    
    def rescale(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return np.zeros_like(x)
        return x * (target_sum / total)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    best_c5 = min(best_c5, 0.381)
    
    pop_size = 20
    population = [h.copy() for _ in range(pop_size)]
    
    for _ in range(500):
        if time.time() - time.time() > budget_s:
            break
        for i in range(pop_size):
            x0 = population[i].copy()
            corr = np.correlate(x0, 1 - x0, mode='full')
            worst_lag = np.argmax(corr)
            k = worst_lag
            start = 0
            end = n_points - k
            if end > 0:
                idxs = np.random.choice(range(start, end), size=5, replace=False)
                for idx in idxs:
                    x0[idx] += np.random.normal(0, 0.01)
                    x0[idx + k] += np.random.normal(0, 0.01)
            x0 = np.clip(x0, 0, 1)
            x0 = rescale(x0)
            bounds = [(0, 1)] * n_points
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                population[i] = res.x
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
    
    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
