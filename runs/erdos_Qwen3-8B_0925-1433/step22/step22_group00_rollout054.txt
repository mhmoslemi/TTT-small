import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if current_sum != 0:
        scale = target_sum / current_sum
    else:
        scale = 1.0
    h_initial = h_initial * scale
    h_initial = np.clip(h_initial, 0, 1)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s != 0:
            h = h * (target_sum / s)
        return h
    
    population_size = 50
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s != 0:
            h = h * (target_sum / s)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_c5 = evaluate(population[0])
    best_h = population[0].copy()
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < 0.9:

            trial = rescale(trial)
            trial_c5 = evaluate(trial)
            if trial_c5 < evaluate(population[i]):

            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
