from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    init = np.array(initial_h_values, dtype=float)
    init = init * target_sum / np.sum(init)
    best_h = init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def objective(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = np.array([(0, 1)] * n_points)
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=25, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, seed=seed, disp=False,
        maxiter=int(budget_s / 1.5)
    )
    best_h = result.x * target_sum / np.sum(result.x)
    best_c5 = result.fun
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
