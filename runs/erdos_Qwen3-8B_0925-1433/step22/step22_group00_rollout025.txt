        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = differential_evolution(obj, bounds, strategy='best1bin',
                                   popsize=50, tol=1e-8, mutation=(0.5, 1),
                                   recombination=0.9, maxiter=200, seed=seed,
                                   polish=True, disp=False)
    
    best_h = result.x
    best_c5 = obj(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if abs(total - target_sum) > 1e-6:
        best_h = best_h / total * target_sum
    
    return (best_h, best_c5, n_points)
