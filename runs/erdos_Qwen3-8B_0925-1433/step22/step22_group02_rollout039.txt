import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and perturbations
    pop_size = 50
    pop = np.array([initial_h_values * (np.random.normal(1, 0.1, n_points) + 1) for _ in range(pop_size)])
    pop = np.clip(pop, 0, 1)
    pop = pop * target_sum / np.sum(pop, axis=1)[:, np.newaxis]

    # Genetic algorithm parameters
    generations = 200
    tournament_size = 5
    crossover_prob = 0.8
    mutation_rate = 0.2
    mutation_std = 0.15

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([obj(ind) for ind in pop])
        # Select parents via tournament
        parents = []
        for _ in range(pop_size):
            candidates = np.random.choice(np.arange(pop_size), tournament_size, replace=False)
            best_idx = candidates[np.argmin(fitness[candidates])]
            parents.append(pop[best_idx])
        parents = np.array(parents)
        # Crossover
        children = []
        for i in range(0, pop_size, 2):
            if np.random.rand() < crossover_prob:
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points-1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
                children.extend([c1, c2])
            else:
                children.append(parents[i])
        children = children[:pop_size]
        # Mutation
        for i in range(pop_size):
            if np.random.rand() < mutation_rate:
                perturb = np.random.normal(0, mutation_std, n_points)
                children[i] += perturb
                children[i] = np.clip(children[i], 0, 1)
                children[i] = children[i] * target_sum / np.sum(children[i])
        # Replace population
        pop = children

    # Local refinement on top candidates
    top_candidates = pop[np.argsort(obj(pop))[:5]]
    best_h = top_candidates[0]
    best_c5 = obj(best_h)
    for cand in top_candidates:
        res = minimize(obj, cand, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Perturbations and final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break

        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
