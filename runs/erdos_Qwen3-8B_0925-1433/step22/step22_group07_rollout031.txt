import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population_size = 50
    # Initialize population with perturbations
    population = []
    for _ in range(population_size):
        h = initial_h.copy()
        h += np.random.normal(0, 0.1, size=h.shape)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = compute_C5(best_h)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [compute_C5(h) for h in population]
        # Select parents via tournament
        parents = []
        for _ in range(population_size // 2):

            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            parents.append(population[best_idx])
        # Mutate parents

        for parent in parents:
            noise = np.random.normal(0, 0.15, parent.shape)
            mut = parent + noise
            mut = np.clip(mut, 0, 1)
            mut = mut * target_sum / np.sum(mut)
            mutated.append(mut)
        # Crossover

        for i in range(0, len(parents), 2):
