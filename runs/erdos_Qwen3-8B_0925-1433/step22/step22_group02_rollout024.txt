import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = np.array(initial_h_values, dtype=float)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Initial guesses: original, reversed, random
    initial_guesses = [np.array(initial_h_values), np.array(initial_h_values[::-1]), np.random.rand(n_points)]
    
    for guess in initial_guesses:
        guess = clip_and_scale(guess)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = Bounds(0, 1)
        result = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = clip_and_scale(best_h + perturb)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
    
    best_h = clip_and_scale(best_h)
    return (best_h, best_c5, n_points)
