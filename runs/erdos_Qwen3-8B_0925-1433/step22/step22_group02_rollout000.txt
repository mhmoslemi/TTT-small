import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        h = np.array(h_vec, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_cons(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': compute_cons}

    # Generate initial guesses
    guesses = [h.copy()]
    sym_guess = np.array([initial_h_values[n_points - 1 - i] for i in range(n_points)])
    sym_guess = sym_guess * target_sum / np.sum(sym_guess)
    guesses.append(sym_guess)
    rand_guess = np.random.rand(n_points)
    rand_guess = rand_guess * target_sum / np.sum(rand_guess)
    guesses.append(rand_guess)

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Perturb and refine
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result = minimize(objective, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
