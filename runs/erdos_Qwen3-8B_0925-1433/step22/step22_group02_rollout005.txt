import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)

    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):

        corr = np.correlate(h, 1 - h, mode='full')

        grad = np.zeros_like(x)

            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    for _ in range(10):
