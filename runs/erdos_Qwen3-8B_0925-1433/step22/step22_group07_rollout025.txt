import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        dx = 2.0 / n_points

        h = h * target_sum / np.sum(h)
    else:
        n_points = 45
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': constraint}]
    
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        x0 = h.copy()
        x0 += np.random.normal(0, 0.02, n_points)
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * target_sum / np.sum(x0)
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
        curr_c5 = objective(h)
        if curr_c5 < best_c5:
            best_h = h.copy()

        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
