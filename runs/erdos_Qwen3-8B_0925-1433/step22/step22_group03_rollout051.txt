import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 20
    F = 0.8
    CR = 0.9
    generations = 150
    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)

    population = [initial_h]
    for _ in range(pop_size - 1):
        vec = np.random.rand(n_points)
        vec = vec * target_sum / np.sum(vec)
        population.append(vec)

    best_h = initial_h
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            mutant = a_vec + F * (b_vec - c_vec)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial = trial * target_sum / np.sum(trial)
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(corr) * dx
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5
            if trial_c5 < np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx:
                population[i] = trial.copy()

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
