import numpy as np
from scipy.optimize import minimize, Bounds, SLSQP
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                term1 = 1 - x[j + k]
            else:
                term1 = 0.0
            if 0 <= j - k < n_points:
                term2 = x[j - k]
            else:
                term2 = 0.0
            grad[j] = term1 - term2
        return grad

    variations = [
        h + np.random.normal(0, 0.05, n_points),
        np.roll(h, 5),
        h[::-1],
        np.concatenate([h[10:], h[:10]]),
    ]
    for var in variations:
        var = np.clip(var, 0, 1)
        var = var * target_sum / np.sum(var)
        h_current = var.copy()
        for _ in range(200):
            if time.time() - time.time() > budget_s:
                break
            grad = compute_grad(h_current)
            eta = 0.1 * (1 - _ / 200)
            h_current = np.clip(h_current - eta * grad, 0, 1)
            h_current = h_current * target_sum / np.sum(h_current)
            current_c5 = obj(h_current)
            if current_c5 < best_c5:
                best_h, best_c5 = h_current.copy(), current_c5
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h_current, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
