import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600):

    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.05)
        h_perturb = h.copy()
        h_perturb[idx] += delta
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        if new_c5 < best_c5:
            best_h, best_c5 = h.copy(), new_c5
