import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[0:5] = 1
        h[10:15] = 1
        h[20:25] = 1
        h = h * (n_points / 2.0) / np.sum(h)
    dx = 2.0 / len(h)
    target_sum = len(h) / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.05
    decay_rate = 0.995
    max_iterations = 300
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (len(h) - 1)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            for k in range(1, len(h)):
                pos = (i + k) % len(h)
                neg = (i - k) % len(h)
                grad[i] += (1 - h[pos]) - h[neg]
        h -= learning_rate * grad
        learning_rate *= decay_rate
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    res = minimize(obj, best_h, method='SLSQP', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_perturb = best_h.copy()
        for _ in range(2):
            idx1 = np.random.randint(len(h_perturb))
