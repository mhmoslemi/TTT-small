import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def generate_variants(base):
        variants = []
        for _ in range(5):
            noise = np.random.normal(0, 0.1, n_points)
            v = base + noise
            v = np.clip(v, 0, 1)
            variants.append(v * target_sum / np.sum(v))
        for _ in range(3):
            shift = np.random.randint(1, n_points - 1)
            v = np.roll(base, shift)
            v = np.clip(v, 0, 1)
            variants.append(v * target_sum / np.sum(v))
        for _ in range(2):
            block_size = np.random.randint(2, 10)
            blocks = np.array([0.5] * block_size + [0.5] * block_size)
            blocks = blocks[:n_points]
            v = np.tile(blocks, n_points // len(blocks) + 1)[:n_points]
            v = np.clip(v, 0, 1)
            variants.append(v * target_sum / np.sum(v))
        for _ in range(2):
            v = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
            v = np.clip(v, 0, 1)
            variants.append(v * target_sum / np.sum(v))
        return variants

    particles = generate_variants(h)
    swarm = np.array(particles)
    velocities = np.zeros_like(swarm)
    pbest = swarm.copy()
    pbest_vals = np.array([obj(p) for p in swarm])
    gbest = swarm[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    start_time = time.time()
    for _ in range(200):

            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * target_sum / np.sum(swarm[i])
            curr_val = obj(swarm[i])
            if curr_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = curr_val
        gbest = swarm[np.argmin(pbest_vals)]
        gbest_val = np.min(pbest_vals)

    best_c5 = gbest_val

            grad = compute_gradient(x)
            x += 1e-3 * grad
