import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        n_points = len(h_initial)

        h_initial = np.random.rand(n_points)

    pop_size = 30
    pop = [h_initial.copy() for _ in range(pop_size)]
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, n_points)
    F = 0.8
    CR = 0.9
    generations = 100
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a_idx, b_idx, c_idx = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[a_idx], pop[b_idx], pop[c_idx]
            mutant = a + F * (b - c)
            trial = pop[i].copy()
