import numpy as np
import time

def rescale_to_sum(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.zeros_like(h)
    scale = target_sum / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.random.rand(45)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = rescale_to_sum(initial_h, target_sum)
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = rescale_to_sum(h, target_sum)
        population.append(h)
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, n_points, dx)
    start_time = time.time()
    mutation_factor = 0.8
    crossover_rate = 0.9
    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(population_size):
            idx_a, idx_b, idx_c = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idx_a], population[idx_b], population[idx_c]
            mutant = a + mutation_factor * (b - c)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    trial[j] = mutant[j]
            trial = rescale_to_sum(trial, target_sum)
            trial_score = evaluate_c5(trial, n_points, dx)
            if trial_score < evaluate_c5(population[i], n_points, dx):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        curr_best = min([evaluate_c5(h, n_points, dx) for h in population])
        if curr_best < best_c5:
            best_h = min(population, key=lambda x: evaluate_c5(x, n_points, dx))
            best_c5 = curr_best
    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
