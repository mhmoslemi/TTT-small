    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for t in range(int(budget_s / 0.15)):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        j = np.argmax(cross_corr)
        k = j - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            term1 = 1 - h[i_plus_k] if 0 <= i_plus_k < n_points else 0.0
            i_minus_k = i - k
            term2 = h[i_minus_k] if 0 <= i_minus_k < n_points else 0.0
            grad[i] = term1 - term2

        alpha = 0.04 * (1 - t / (budget_s / 0.15))
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)

        if np.random.rand() < 0.1:
            perturb = np.random.normal(0, 0.02, n_points)
            h_new += perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)

        current_c5 = np.max(cross_corr) * dx
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
        h = h_new

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
