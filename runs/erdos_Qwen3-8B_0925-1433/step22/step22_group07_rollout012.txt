import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()
    temp = 1.0
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        worst_lags = np.argsort(corr)[-5:]
        
        h_perturbed = h.copy()
        for k in worst_lags:
            indices = np.random.choice(n_points, size=5, replace=False)
            for i in indices:
                h_perturbed[i] += np.random.normal(0, 0.05)

        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)

        temp *= 0.999
