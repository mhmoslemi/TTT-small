import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Adam parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    decay = 0.999
    max_iterations = 500
    start_time = time.time()
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[-5:]
        top_lags = [i - (n_points - 1) for i in top_indices]
        
        grad = np.zeros_like(h)
        for k_j in top_lags:
            for i in range(n_points):
                pos = (i + k_j) % n_points
                neg = (i - k_j) % n_points
                grad[i] += (1 - h[pos]) - h[neg]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        adam_lr = learning_rate * decay
        h = h - adam_lr * m / (np.sqrt(v) + 1e-8)
        learning_rate *= decay
        
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    # Local optimization
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=Bounds(0, 1),
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun
    
    # Perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        noise = np.random.normal(0, 0.01, size=n_points)
        h_perturb = np.clip(best_h + noise, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)

        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
