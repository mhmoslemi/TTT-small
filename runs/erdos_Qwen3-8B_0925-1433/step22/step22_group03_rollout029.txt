import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    if hasattr(initial_h_values, '__iter__'):
        n_points = len(initial_h_values)
        target_sum = n_points / 2.0
        h_initial = np.array(initial_h_values, dtype=float)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        initial_population = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(15)]
        initial_population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) for h in initial_population]
    else:
        n_points = 45
        target_sum = n_points / 2.0
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        initial_population = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(15)]
        initial_population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) for h in initial_population]
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)
    
    current_best_val, current_best_h = min([(compute_c5(h), h) for h in initial_population], key=lambda x: x[0])
    
    # Genetic algorithm
    population = initial_population
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        evals = [(compute_c5(h), h) for h in population]
        top_h = [x[1] for x in sorted(evals)[:int(len(evals)*0.2)]]
        new_population = []
        for h in top_h:
            mutated = h + np.random.normal(0, 0.02, n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * target_sum / np.sum(mutated)

        current_best_val, current_best_h = min(evals, key=lambda x: x[0])

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = np.array([[0.0, 1.0] for _ in range(n_points)])
    res = minimize(obj, current_best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h)
    else:
        best_h = current_best_h
        best_c5 = current_best_val
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
