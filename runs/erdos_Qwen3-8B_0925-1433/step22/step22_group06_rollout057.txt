from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    pop_size = 10
    population = [h.copy() for _ in range(pop_size)]
    
    for i in range(pop_size):
        p = np.roll(h, i) + np.random.normal(0, 0.02, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population[i] = p

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(x):
        freq = np.random.uniform(0.1, 1.0)
        phase = np.random.uniform(0, 2*np.pi)
        amp = np.random.uniform(0.01, 0.05)
        x = x + amp * np.cos(freq * np.arange(n_points) + phase)
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            x0 = population[i].copy()
            x0 = mutate(x0)
            res = minimize(obj, x0, method='SLSQP', bounds=[(0, 1)]*n_points,
