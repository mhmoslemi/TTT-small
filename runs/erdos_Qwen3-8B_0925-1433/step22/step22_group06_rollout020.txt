import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    import numpy as np
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    alpha = 0.02
    decay_rate = 0.999
    max_iter = int(budget_s / 0.5)
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = corr * dx
        max_c5 = np.max(c5)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        k_index = np.argmax(corr)
        k = k_index - n_points + 1
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k) % n_points
            i_minus = (i - k) % n_points
            gradient[i] = -(h[i_plus] + h[i_minus])
        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        alpha *= decay_rate
    
    h_values = np.clip(best_h, 0, 1)
