import time

    decay_rate = 0.995
    momentum = 0.8
    max_iter = 400
    start_time = time.time()

        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[-5:][::-1]  # Top 5 lags
        grad = np.zeros_like(h)
        for k in top_lags:
            if k == 0:
                continue
            for i in range(n_points):
                pos = (i + k) % n_points
                neg = (i - k) % n_points
                grad[i] += (1 - h[pos]) - h[neg]
        grad /= len(top_lags)
        v = momentum * v + (1 - momentum) * grad
        h -= learning_rate * v
        learning_rate *= decay_rate

        return np.max(corr) * dx
