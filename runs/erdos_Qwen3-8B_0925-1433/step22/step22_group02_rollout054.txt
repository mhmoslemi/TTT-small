import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    population_size = 50
    initial_pop = []
    for _ in range(population_size):
        h = np.array(initial_h_values).copy()
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        initial_pop.append(h)

    def tournament_select(fitnesses):
        indices = np.random.choice(range(population_size), size=5, replace=False)
        best_idx = indices[np.argmin(fitnesses[indices])]
        return initial_pop[best_idx].copy()

    best_h = initial_pop[0].copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in initial_pop]
        parents = [tournament_select(fitness) for _ in range(population_size // 2)]

        children = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
                children.extend([c1, c2])
            else:
                children.extend([p1.copy(), p2.copy()])

        for c in children:
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.15, n_points)
                c += noise
                c = np.clip(c, 0, 1)
                c = c * target_sum / np.sum(c)
        children = children[:population_size]

        # Merge and sort
        merged = initial_pop + children
        fitness = [compute_c5(h) for h in merged]
        merged = [h for _, h in sorted(zip(fitness, merged))][:population_size]
        initial_pop = merged

    top_candidates = initial_pop[:5]
    for h_candidate in top_candidates:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = Bounds(0, 1)
        res = minimize(lambda x: compute_c5(x), h_candidate, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
