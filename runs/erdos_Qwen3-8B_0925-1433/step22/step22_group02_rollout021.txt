import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h_vec = x * target_sum / np.sum(x)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=compute_gradient, tol=1e-6)
        h = res.x
        current_c5 = obj(h)
        if current_c5 < best_c5:

    def nelder_mead_refine(x):

        return res.x

    current_c5 = obj(best_h)
