import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

        return np.max(corr) * dx
    
    population = [h.copy()]
    for _ in range(49):
        noise = np.random.normal(0, 0.1, n_points)
        h = np.clip(initial_h_values + noise, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    F, CR = 0.5, 0.8
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            x1, x2, x3 = population[idxs]
            mutant = x1 + F * (x2 - x3)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)

            new_pop.append(trial)
        new_pop = np.array(new_pop)

        inds = np.argsort(c5s)[:50]
        population = new_pop[inds]
