def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    population = [np.clip(initial_pop + np.random.normal(0, 0.15, n_points), 0, 1) * target_sum / np.sum(np.clip(initial_pop + np.random.normal(0, 0.15, n_points), 0, 1)) for _ in range(population_size)]
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    mutation_scale_start = 0.15
    mutation_scale_end = 0.01
    cooling_rate = 0.995
    time_start = time.time()
    while time.time() - time_start < budget_s:
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(0.2 * population_size)]
        top_population = [population[i] for i in top_indices]
        new_population = []
        temp = 1.0 - generations / 1000.0
        for h in top_population:
            mutation_std = mutation_scale_start * temp
            mutation_std = max(mutation_std, mutation_scale_end)
            noise = np.random.normal(0, mutation_std, n_points)
            h_mutated = np.clip(h + noise, 0, 1) * target_sum / np.sum(np.clip(h + noise, 0, 1))
            new_population.append(h_mutated)
        for i in range(0, len(top_population), 2):
            if i + 1 < len(top_population):
                alpha = np.random.rand()
                h_crossover = np.clip(alpha * top_population[i] + (1 - alpha) * top_population[i + 1], 0, 1) * target_sum / np.sum(np.clip(alpha * top_population[i] + (1 - alpha) * top_population[i + 1], 0, 1))
                new_population.append(h_crossover)
        while len(new_population) < population_size:
            perturb = np.random.normal(0, 0.1, n_points)
            h = np.clip(initial_pop + perturb, 0, 1) * target_sum / np.sum(np.clip(initial_pop + perturb, 0, 1))
            new_population.append(h)
        population = new_population
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(np.clip(best_h, 0, 1))
    return best_h, best_c5, n_points
