import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [initial_h_values.copy() for _ in range(population_size)]
    for i in range(population_size):
        perturb = np.random.normal(0, 0.1, n_points)
        population[i] = initial_h_values + perturb
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(h) for h in population]
        selected = np.argsort(fitness)[:int(population_size * 0.8)]
        parents = [population[i] for i in selected]
        new_population = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
            else:
                c1, c2 = p1.copy(), p2.copy()
            for c in [c1, c2]:
                mutate_mask = np.random.rand(n_points) < 0.2
                c[mutate_mask] += np.random.normal(0, 0.15, np.sum(mutate_mask))
                c = np.clip(c, 0, 1)
                c = c * target_sum / np.sum(c)
            new_population.extend([c1, c2])
        population = new_population[:population_size]
        for h in population:
            curr = evaluate(h)
            if curr < best_c5:
                best_h = h.copy()
                best_c5 = curr

    top_candidates = sorted(population, key=evaluate)[:5]
    for h in top_candidates:
        def obj(x):
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq','fun':lambda x: np.sum(x) - target_sum}, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
