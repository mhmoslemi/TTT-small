import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 50
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all individuals
        fitness = [evaluate_erdos_solution(h) for h in population]
        # Select top 25
        top_idx = np.argsort(fitness)[:25]
        new_pop = []
        for i in top_idx:
            h = population[i].copy()
            noise = np.random.normal(0, 0.15, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            new_pop.append(h)
        # Crossover
        for i in range(len(top_idx) // 2):
            a, b = top_idx[i], top_idx[i+1]
            p1, p2 = population[a], population[b]
            child = 0.5 * p1 + 0.5 * p2
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        # Replace population
        population = new_pop
        # Track best solution
        for h in population:
            curr_c5 = evaluate_erdos_solution(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
