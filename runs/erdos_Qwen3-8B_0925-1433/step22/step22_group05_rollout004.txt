import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(h_values):
        h = np.clip(h_values, 0, 1)
        h_norm = h * target_sum / np.sum(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(
        obj, bounds, strategy='best1bin', maxiter=800, popsize=15,
        tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True,
        disp=False, vectorized=True, workers=-1
    )
    elapsed = time.time() - start_time

    best_h = result.x
    best_c5 = obj(best_h)

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)

    return (h_values, best_c5, n_points)
