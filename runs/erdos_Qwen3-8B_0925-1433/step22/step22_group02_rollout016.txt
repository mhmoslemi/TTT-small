import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def obj(x):
        np.clip(x, 0, 1, out=x)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - x[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = x[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad
    
    initial_guesses = [
        initial_h_values,
        np.concatenate([initial_h_values[::-1], initial_h_values[1:]]),
        np.random.rand(n_points)
    ]
    for i in range(len(initial_guesses)):
        initial_guesses[i] = initial_guesses[i] * target_sum / np.sum(initial_guesses[i])
    
    best_h = np.copy(initial_h_values)
    best_c5 = float('inf')
    
    for init in initial_guesses:
        h = np.copy(init)
        h = h * target_sum / np.sum(h)
        for _ in range(200):
            if time.time() - time.time() > budget_s:
                break
            grad = compute_gradient(h)
            h += -0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = obj(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5
    
    def refine(x):
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, x, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-8)
        return res.x
