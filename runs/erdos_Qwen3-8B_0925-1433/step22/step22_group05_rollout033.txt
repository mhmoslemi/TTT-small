import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale_and_clip(vec, target_sum):
        vec_clipped = np.clip(vec, 0, 1)
        current_sum = np.sum(vec_clipped)
        if current_sum == 0:
            return np.zeros_like(vec_clipped)
        scaling_factor = target_sum / current_sum
        return vec_clipped * scaling_factor

    # Initial population
    population = []
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    for _ in range(2):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = h_initial + noise
        perturbed = rescale_and_clip(perturbed, target_sum)
        population.append(perturbed)

    binary_pattern = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
    binary_pattern[indices] = 1
    binary_pattern = rescale_and_clip(binary_pattern, target_sum)
    population.append(binary_pattern)

    block_width = n_points // 5
    block_pattern = np.zeros(n_points)
    for i in range(0, n_points, block_width*2):
        block_pattern[i:i+block_width] = 1
    block_pattern = rescale_and_clip(block_pattern, target_sum)
    population.append(block_pattern)

    while len(population) < 5:
        population.append(np.random.uniform(0, 1, n_points))
        population[-1] = rescale_and_clip(population[-1], target_sum)

    best_h = population[0].copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        idx = np.random.randint(len(population))
        a, b, c = np.random.choice(population, 3, replace=False)
        mutant = a + 0.5 * (b - c)
        mutant = rescale_and_clip(mutant, target_sum)
        trial = np.zeros_like(mutant)
        for i in range(n_points):

                trial[i] = population[idx][i]
