import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evaluate(h):

        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return float('inf')
        scaling = target_sum / sum_clipped
        scaled = h_clipped * scaling
        return compute_c5(scaled)

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)

    pop_size = 10
    pop = [initial_h.copy() for _ in range(pop_size)]
    for _ in range(5):
        idx = np.random.choice(n_points, size=2, replace=False)
        perturb = np.zeros(n_points)
        perturb[idx] = 1.0
        new_h = initial_h + np.random.normal(0, 0.05, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        pop.append(new_h)

    pop = np.array([np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in pop[:pop_size]])

    best_h = pop[0].copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        np.random.shuffle(pop)
        for i in range(pop_size):

            mutant = pop[a] + 0.7 * (pop[b] - pop[c])
            trial = np.copy(mutant)
            for j in range(n_points):

                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
