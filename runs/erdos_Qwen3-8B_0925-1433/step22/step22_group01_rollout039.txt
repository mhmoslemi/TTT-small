        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            pos = (i + k) % n_points
            neg = (i - k) % n_points
            grad[i] = (1 - h[pos]) - h[neg]
        h -= learning_rate * grad
        learning_rate *= decay_rate
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=Bounds(0, 1), 
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.02, size=n_points)
        h_perturb = np.clip(best_h + noise, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1),
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
