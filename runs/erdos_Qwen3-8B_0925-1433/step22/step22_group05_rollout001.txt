import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2 / n_points
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start = time.time()
    for _ in range(5):
        if time.time() - start > budget_s:
            break
        pop = []
        for _ in range(2):
            p = initial_h_values + np.random.normal(0, 0.05, n_points)
            p = np.clip(p, 0, 1)
            p = p * target_sum / np.sum(p)
            pop.append(p)
        for _ in range(2):
            p = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            p = p * target_sum / np.sum(p)
            pop.append(p)
        for p in pop:
            try:
                res = differential_evolution(obj, bounds, popsize=10, tol=1e-6, maxiter=50,
                                            mutation=(0.5, 1), recombination=0.7, seed=seed)
                if res.success:
                    curr = res.x
                    curr_c5 = obj(curr)
                    if curr_c5 < best_c5:
                        best_c5 = curr_c5
                        best_h = curr.copy()
            except:
                pass
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
