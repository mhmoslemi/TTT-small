import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_seeds():
        seeds = []
        # Sinusoidal seed
        x = np.linspace(0, 2, n_points)
        h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
        seeds.append(h_sinusoidal)
        # Jittered parent
        h_jittered = np.array(initial_h_values, dtype=float)
        h_jittered += np.random.normal(0, 0.05, n_points)
        h_jittered = np.clip(h_jittered, 0, 1)
        h_jittered = h_jittered * target_sum / np.sum(h_jittered)
        seeds.append(h_jittered)
        # Parent seed
        seeds.append(np.array(initial_h_values, dtype=float))
        return seeds

    seeds = generate_seeds()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')
    start_time = time.time()

    for seed in seeds:
        h = seed.copy()
        h = h * target_sum / np.sum(h)
        for _ in range(500):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if np.isnan(current_c5) or np.isinf(current_c5):
                break
            k_max_index = np.argmax(corr)
            shift = k_max_index - (n_points - 1)
            grad = np.zeros_like(h)
            for i in range(n_points):
                i_plus = i + shift
                term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
                i_minus = i - shift
                term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
                grad[i] = term1 - term2
            h = h * (1 - 0.01) + 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

    # Final refinement with SLSQP
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (np.clip(best_h, 0, 1), best_c5, n_points)
