import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_init = np.array(initial_h_values, dtype=float)
    h_init = h_init * target_sum / np.sum(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def perturb(h, type='noise'):
        if type == 'noise':
            return np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        elif type == 'block':
            mid = n_points // 2
            return np.concatenate([h[mid:], h[:mid]])
        elif type == 'sine':
            return np.clip(0.5 + 0.5 * np.sin(np.linspace(0, np.pi, n_points)), 0, 1)
        elif type == 'alt':
            return np.concatenate([np.full(n_points//2, 1), np.full(n_points//2, 0)])
        return h

    particles = []
    for _ in range(15):
        p = perturb(h_init, np.random.choice(['noise', 'block', 'sine', 'alt']))
        p = p * target_sum / np.sum(p)
        particles.append(p)

    def pso_update(p, pbest, gbest, inertia=0.8, phi=1.5):
        r1, r2 = np.random.rand(), np.random.rand()
        vel = inertia * p.vel + phi * r1 * (pbest - p.pos) + phi * r2 * (gbest - p.pos)
        new_pos = p.pos + vel
        new_pos = np.clip(new_pos, 0, 1)
        new_pos = new_pos * target_sum / np.sum(new_pos)
        return new_pos

    class Particle:
        def __init__(self, pos):
            self.pos = pos.copy()
            self.vel = np.zeros(n_points)
            self.pbest = pos.copy()
            self.pbest_val = compute_c5(pos)
        def update(self, gbest):
            self.vel = 0.8 * self.vel + 1.5 * np.random.rand(n_points) * (self.pbest - self.pos) + 1.5 * np.random.rand(n_points) * (gbest - self.pos)
            self.pos = np.clip(self.pos + self.vel, 0, 1)
            self.pos = self.pos * target_sum / np.sum(self.pos)
            curr_val = compute_c5(self.pos)
            if curr_val < self.pbest_val:
                self.pbest = self.pos.copy()
                self.pbest_val = curr_val

    swarm = [Particle(p) for p in particles]
    gbest = min(swarm, key=lambda x: x.pbest_val).pbest
    gbest_val = min([x.pbest_val for x in swarm])

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in swarm:
            p.update(gbest)
            if compute_c5(p.pos) < gbest_val:
                gbest = p.pbest
                gbest_val = compute_c5(gbest)
        best_h, best_c5 = gbest, gbest_val

    def grad_descent(h, steps=300, lr=0.05, momentum=0.9):
        for _ in range(steps):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            shift = k_max - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                i_plus = i + shift
                if 0 <= i_plus < n_points:
                    term1 = 1 - h[i_plus]
                else:
                    term1 = 0.0
                i_minus = i - shift
                if 0 <= i_minus < n_points:
                    term2 = h[i_minus]
                else:
                    term2 = 0.0
                grad[i] = term1 - term2
            h = h + lr * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    grad_descent(best_h)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)
    res = minimize(lambda x: compute_c5(x), best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_c5 = compute_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
