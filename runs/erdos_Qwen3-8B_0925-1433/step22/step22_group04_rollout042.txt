            for i in range(n_points):
                m[i] = 0.9 * m[i] + 0.1 * grad[i]
                v[i] = 0.999 * v[i] + 0.001 * grad[i] ** 2
                biased_m = m[i] / (1 - 0.9 ** (_ + 1))
                biased_v = v[i] / (1 - 0.999 ** (_ + 1))
                h[i] = h[i] + 0.05 * biased_m / (np.sqrt(biased_v) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(current_corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    current_corr = np.correlate(best_h, 1 - best_h, mode='full')
    current_c5 = np.max(current_corr) * dx
    return (best_h, current_c5, n_points)
