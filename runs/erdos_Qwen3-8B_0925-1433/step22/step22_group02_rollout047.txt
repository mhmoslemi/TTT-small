import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def normalize(h):
        return h * target_sum / np.sum(h)

    population = [np.array(initial_h, dtype=float) for _ in range(50)]
    for i in range(50):
        perturb = np.random.normal(0, 0.1, n_points)
        population[i] = np.clip(initial_h + perturb, 0, 1)
        population[i] = normalize(population[i])

    best_h = population[0].copy()
    best_c5 = compute_C5(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_C5(ind) for ind in population]
        parents = []
        for _ in range(25):
            tournament = np.random.choice(population, 5)
            winner = tournament[np.argmin([compute_C5(ind) for ind in tournament])]
            parents.append(winner)
        new_population = []
        for i in range(0, 50, 2):
            if np.random.rand() < 0.8:
                parent1 = parents[i//2]
                parent2 = parents[i//2 + 1]
                split = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
            else:
                child1, child2 = parents[i//2].copy(), parents[i//2 + 1].copy()
            for child in [child1, child2]:

                    child += noise
                    child = np.clip(child, 0, 1)
                child = normalize(child)
            new_population.extend([child1, child2])
        population = new_population

    top_candidates = [normalize(population[i]) for i in range(5)]
    for candidate in top_candidates:
        res = minimize(

            candidate,
