            perturb = (perturb < 0.1) * 0 + (perturb >= 0.9) * 1
            new_h = h_values * (1 - perturb) + perturb * 0
            new_h = np.clip(new_h, 0, 1)
            new_h = rescale_to_sum(new_h)
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                stagnation_count = 0
            else:
                step_size *= 0.5
                if step_size < 1e-6:
                    break

        h_values = new_h.copy()
    return (best_h, best_c5, n_points)
