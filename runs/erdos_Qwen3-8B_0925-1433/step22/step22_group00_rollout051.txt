        n_points = 100
        h = np.random.rand(n_points)
        h = rescale_to_sum(h, n_points / 2.0)
    
    h = np.clip(h, 0, 1)
    h = rescale_to_sum(h, len(h)/2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    dx_val = 2.0 / len(h)
    step_size = 0.01
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr[1:]) + 1  # Skip k=0
        if max_index == 0:
            break
        k = max_index
        grad = compute_gradient(h, k, dx_val)
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = rescale_to_sum(h, len(h)/2.0)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (np.clip(best_h, 0, 1), best_c5, len(h))
