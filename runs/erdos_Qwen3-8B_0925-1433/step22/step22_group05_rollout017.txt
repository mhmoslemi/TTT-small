from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        population.append(h)

    population.append(h_alt)
    
    # Random binary
    h_rand = np.random.choice([0, 1], size=n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)
