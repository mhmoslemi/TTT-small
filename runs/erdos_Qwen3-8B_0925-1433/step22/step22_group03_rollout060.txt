import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def eval_c5(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    population = [np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1) * target_sum / np.sum(np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)) for _ in range(pop_size)]

    for _ in range(200):
        if time.time() - time.time() > budget_s:
            break
        fitness = [eval_c5(p) for p in population]
        top = np.argsort(fitness)[:10]
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(top, 2, replace=False)
            child = (population[p1] + population[p2]) * 0.5
            child += np.random.normal(0, 0.05, n_points)
            child = np.clip(child, 0, 1) * target_sum / np.sum(np.clip(child, 0, 1))
            new_pop.append(child)
        population = new_pop
        for p in population:
            c = eval_c5(p)
            if c < best_c5:
                best_h = p.copy()
                best_c5 = c

    def refine(x):
        res = minimize(obj, x, method='SLSQP', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        return res.x

    best_h = refine(best_h)
    best_c5 = eval_c5(best_h)

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1) * target_sum / np.sum(np.clip(best_h + perturb, 0, 1))
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
