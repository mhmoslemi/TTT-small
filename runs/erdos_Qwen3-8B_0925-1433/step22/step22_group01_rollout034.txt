        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    decay = 0.999
    max_iterations = 500
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        sorted_indices = np.argsort(corr)[::-1]
        top_lags_indices = sorted_indices[:5]
        n_points_curr = len(h)
        top_lags = [i - (n_points_curr - 1) for i in top_lags_indices]
        
        gradient = np.zeros_like(h)
        for lag in top_lags:
            for i in range(n_points_curr):
                pos = (i + lag) % n_points_curr
                neg = (i - lag) % n_points_curr
                gradient[i] += (1 - h[pos]) - h[neg]
        
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        learning_rate *= decay
        h = h - learning_rate * m / (np.sqrt(v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = _compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / len(x))
    
    bounds = Bounds(0, 1)
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.01, size=len(best_h))
        perturb = best_h + noise
        perturb = np.clip(perturb, 0, 1)
        perturb = perturb * target_sum / np.sum(perturb)
        res = minimize(obj, perturb, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (np.clip(best_h, 0, 1), best_c5, len(best_h))
