        if time.time() - start_time > budget_s:
            break
        sampled_lags = np.random.choice(n_points, min(10, n_points), replace=False)
        lag_values = []
        for k in sampled_lags:
            corr = np.correlate(h, 1 - h, mode='full')
            rk = corr[k] * (2.0 / n_points)
            lag_values.append((rk, k))
        lag_values.sort(reverse=True)
        top_lags = [k for (rk, k) in lag_values[:2]]
        
        grads = []
        for k in top_lags:
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                j_prev = (i - k) % n_points
                grad[i] = (1 - h[j]) - h[j_prev]
            grads.append(grad)
        if grads:
            avg_grad = np.mean(grads, axis=0)
            h -= learning_rate * avg_grad
            h = np.clip(h, 1e-6, 1 - 1e-6)
            h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
