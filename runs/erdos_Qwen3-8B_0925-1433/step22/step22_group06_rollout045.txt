    start_time = time.time()

        k_idx = np.argmax(cross_corr)
        lag_k = k_idx - (n_points - 1)
        # compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = (j + lag_k) % n_points
            j_minus_k = (j - lag_k) % n_points
            grad[j] = (1 - h[j_plus_k]) - h[j_minus_k]
