        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = evaluate_c5(initial_arr)
    best_h = initial_arr.copy()
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness_list = [(evaluate_c5(h), h) for h in population]
        fitness_list.sort()
        top_num = int(population_size * 0.2)
        top_candidates = [tpl[1] for tpl in fitness_list[:top_num]]
        new_pop = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(top_candidates, 2, replace=False)
            alpha = np.random.uniform(0, 1)
            child = alpha * p1 + (1 - alpha) * p2
            noise = np.random.normal(0, 0.12, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        best_half = population[:population_size // 2]
        population = best_half + new_pop
        current_best = min(evaluate_c5(h) for h in population)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin([evaluate_c5(h) for h in population])]
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
