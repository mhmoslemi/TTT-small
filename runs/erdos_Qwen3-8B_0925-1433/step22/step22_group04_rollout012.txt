            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    h_init = np.array(initial_h_values, dtype=float)
    h_init = h_init * target_sum / np.sum(h_init)
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def evaluate(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def refine(h):
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = Bounds(0, 1)
        result = minimize(evaluate, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x

    # Initial refinement
    best_h = refine(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Perturbation and gradient descent
    learning_rate = 0.05
    momentum = 0.9
    decay_rate = 0.999
    max_iterations = 500
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(best_h)
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
            biased_m = m[i] / (1 - momentum ** (_ + 1))
            biased_v = v[i] / (1 - decay_rate ** (_ + 1))
            best_h[i] = best_h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Final refinement
    best_h = refine(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
