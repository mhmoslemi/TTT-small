import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    remaining_time = budget_s

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    # Perturbations
    perturbations = []
    # Add noise
    perturbations.append(np.random.normal(0, 0.05, n_points))
    # Shift
    perturbations.append(np.roll(np.random.random(n_points), 1))
    # Block swap
    perturbations.append(np.concatenate([np.random.random(n_points//2), np.random.random(n_points//2)]))

    perturbations.append(np.random.permutation(np.random.random(n_points)))
    # Small shift
    perturbations.append(np.roll(np.random.random(n_points), 2))

    for p in perturbations:
        if time.time() - start_time >= budget_s:
            break
        h_perturb = h + p
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:

    # Final refinement
    if time.time() - start_time < budget_s:
