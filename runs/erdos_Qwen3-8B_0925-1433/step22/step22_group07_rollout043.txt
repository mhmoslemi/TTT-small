            new_h[i] = np.clip(new_h[i], 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                h = new_h.copy()
        
        T *= cooling_rate
    
    h_values = np.clip(h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
