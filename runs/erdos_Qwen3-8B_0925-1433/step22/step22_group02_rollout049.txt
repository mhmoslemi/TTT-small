        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
