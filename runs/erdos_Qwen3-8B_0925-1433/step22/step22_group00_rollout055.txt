from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    
    def rescale_to_sum(h, target_sum):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h) + 1.0
        scale = target_sum / np.sum(h)
        return np.clip(h * scale, 0, 1)
    
    def evaluate_candidate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    h = rescale_to_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(x):
        x = rescale_to_sum(x, target_sum)
        return evaluate_candidate(x)
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=50,
        tol=1e-6,

        recombination=0.9,
        callback=lambda x: None,
        disp=False,
        polish=True,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)
    
    if time.time() - start_time > budget_s - 10:
