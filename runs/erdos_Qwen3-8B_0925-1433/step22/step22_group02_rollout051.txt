import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    h_sym = h_initial[::-1].copy() * target_sum / np.sum(h_initial[::-1])
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)

    def compute_c5(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = (i + shift) % n_points
            i_minus = (i - shift) % n_points
            grad[i] = (1.0 - h[i_plus]) - h[i_minus]
        return grad

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    def optimize(h0, method='L-BFGS-B', maxiter=100):
        nonlocal best_c5, best_h
        def obj(x):
            return compute_c5(x)
        def grad(x):
            return compute_gradient(x)
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h0, method=method, jac=grad, bounds=bounds, constraints=cons, options={'maxiter': maxiter})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    for h in [h_initial, h_sym, h_rand]:
        optimize(h)

    for _ in range(3):
        if time.time() - start_time > budget_s: break
        optimize(best_h, 'SLSQP', 100)

    for _ in range(3):
        if time.time() - start_time > budget_s: break
        h = best_h.copy() + np.random.normal(0, 0.05, n_points)
        h = np.clip(h * target_sum / np.sum(h), 0, 1)
        optimize(h, 'L-BFGS-B', 100)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
