import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-6:
        h = h / h_sum * target_sum
    best_h = h.copy()
    best_c5 = np.inf

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        positive_corr = corr[n_points-1:]
        return np.max(positive_corr) * dx

    def mutation(parent):
        noise = np.random.normal(0, 0.03, n_points)
        child = parent + noise
        child = np.clip(child, 0, 1)
        child_sum = np.sum(child)
        child = child / child_sum * target_sum
        return child

    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, n_points-1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        return child

    def mutate(parent):
        max_corr = evaluate(parent)
        corr = np.correlate(parent, 1 - parent, mode='full')
        positive_corr = corr[n_points-1:]
        max_lag_idx = np.argmax(positive_corr)
        max_lag = max_lag_idx
        if max_lag == 0:
            return mutation(parent)
        mask = np.zeros(n_points, dtype=bool)
        for i in range(n_points - max_lag):
            mask[i] = True
            mask[i + max_lag] = True
        noise = np.random.normal(0, 0.05, n_points)
        child = parent.copy()
        child[mask] += noise[mask]
        child = np.clip(child, 0, 1)
        child_sum = np.sum(child)
        child = child / child_sum * target_sum
        return child

    population = [h.copy()]
    for _ in range(50):
        population.append(mutation(h))
    population = [np.clip(p, 0, 1) for p in population]
    population = [p / np.sum(p) * target_sum for p in population]

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(p) for p in population]
        elite_indices = np.argsort(scores)[:int(len(population)*0.2)]
        elites = [population[i] for i in elite_indices]
        offspring = []
        for i in range(0, len(elites), 2):
            if i + 1 < len(elites):
                p1 = elites[i]
                p2 = elites[i+1]
                child = crossover(p1, p2)
                child = mutate(child)
                offspring.append(child)
        population = elites + offspring
        population = [np.clip(p, 0, 1) for p in population]
        population = [p / np.sum(p) * target_sum for p in population]
        for candidate in population:
            curr_c5 = evaluate(candidate)
            if curr_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = curr_c5
    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
