        h -= adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1-x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=Bounds(0,1), constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s: break
        perturb = np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0,1), constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
