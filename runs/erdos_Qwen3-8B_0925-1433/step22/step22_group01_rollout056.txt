    learning_rate = 0.05
    decay_rate = 0.995
    max_iterations = 300
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[-5:]
        grads = np.zeros_like(h)
        for idx in top_indices:
            k = idx - (n_points - 1)

            for i in range(n_points):
                pos = (i + k) % n_points
                neg = (i - k) % n_points
                grad_k[i] = (1 - h[pos]) - h[neg]
            grads += grad_k
        grad = grads / len(top_indices)
        h -= learning_rate * grad

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    def obj(x):
