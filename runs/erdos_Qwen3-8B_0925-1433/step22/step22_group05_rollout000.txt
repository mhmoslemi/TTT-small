import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_c5(x):
        return obj(x)
    
    population = []
    for _ in range(5):
        pert = np.random.normal(0, 0.1, n_points)
        h = h_initial + pert
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    pattern = np.zeros(n_points)
    pattern[::2] = 1.0
    pattern = pattern * target_sum / np.sum(pattern)
    population.append(pattern)
    
    pattern = np.zeros(n_points)
    block_size = n_points // 4
    pattern[:block_size] = 1.0
    pattern = pattern * target_sum / np.sum(pattern)
    population.append(pattern)
    
    pattern = np.random.rand(n_points)
    pattern = np.where(pattern < 0.5, 0, 1)
    pattern = pattern * target_sum / np.sum(pattern)
    population.append(pattern)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(population[0])
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='SLSQP', bounds=[(0, 1)] * n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           tol=1e-6)
            if res.success:
                population[i] = res.x
                curr_c5 = compute_c5(res.x)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = res.x.copy()
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
