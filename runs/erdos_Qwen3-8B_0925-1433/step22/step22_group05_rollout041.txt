import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    
    # Generate diverse initial population
    population = []
    for _ in range(10):
        h = initial_h * target_sum / np.sum(initial_h)
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    # Objective function
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Constraint: sum(x) = target_sum
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, constraints=cons,
        popsize=20, mutation=(0.5, 1), recombination=0.8,
        seed=seed, tol=1e-6, maxiter=300,
        polish=True, disp=False
    )
    
    best_h = np.clip(result.x, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Ensure valid sum and bounds
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
