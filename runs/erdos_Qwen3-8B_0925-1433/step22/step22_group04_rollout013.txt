    def obj(x):

        return np.max(corr) * dx

    def compute_gradient(x):

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)

            if 0 <= i_plus < n_points:

            i_minus = i - shift

                term2 = h[i_minus]
            else:

            grad[i] = term1 - term2
        return grad
