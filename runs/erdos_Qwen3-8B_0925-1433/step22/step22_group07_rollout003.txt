def run(seed=42, budget_s=600):

        indices = np.random.choice(n_points, perturb_size, replace=False)

        new_h = h.copy()
        new_h[indices] += perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - best_c5
        
        if delta_c5 < 0:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            prob = np.exp(-delta_c5 / temperature)
            if np.random.rand() < prob:
                best_h = new_h.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
