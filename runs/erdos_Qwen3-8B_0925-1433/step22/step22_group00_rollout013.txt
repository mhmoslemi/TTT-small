        grad_down = corr_down[critical_lag] - corr[critical_lag]

        grad = (grad_up + grad_down) / 2.0
        adjustment = -np.sign(grad) * 0.01 * np.abs(grad)
        new_h = h + adjustment
        new_h = np.clip(new_h, 0, 1)
        new_h = rescale_to_sum(new_h, target_sum)
        
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= stagnation_threshold:
            perturb = np.random.rand(n_points) * 0.5
            h = np.clip(h + perturb, 0, 1)
            h = rescale_to_sum(h, target_sum)
            no_improvement = 0

        iteration += 1
    
    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
