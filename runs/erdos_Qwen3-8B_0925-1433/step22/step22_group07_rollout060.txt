import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    start_time = time.time()

    cooling_rate = 0.999
    
    while time.time() - start_time < budget_s:
        if T < 1e-5:
            break

        noise = np.random.normal(0, 0.03, n_points)

        new_c5 = np.max(corr) * dx

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = perturbed
            if new_c5 < best_c5:
