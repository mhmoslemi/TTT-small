            current = population[np.random.randint(pop_size)]
            corr_current = np.correlate(current, 1 - current, mode='full')
            current_c5 = np.max(corr_current) * dx
            if mut_c5 < current_c5:
                new_pop.append(mutant)
            else:
                new_pop.append(current)
        population = new_pop
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
