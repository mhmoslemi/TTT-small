import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()

    def optimize_h(init):
        x0 = init.copy()
        x0 = x0 * target_sum / np.sum(x0)
        x0 = np.clip(x0, 0, 1)
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(x0)
            x0 -= 0.5 * grad
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * target_sum / np.sum(x0)
            curr_c5 = obj(x0)
            if curr_c5 < best_c5:
                nonlocal best_c5, best_h
                best_h, best_c5 = x0.copy(), curr_c5

        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)

            best_h, best_c5 = res.x, res.fun

    optimize_h(h)
