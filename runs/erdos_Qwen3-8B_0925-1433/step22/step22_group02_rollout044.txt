from scipy.optimize import minimize, Bounds
import time

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h_val = x * target_sum / np.sum(x)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h_val = x * target_sum / np.sum(x)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad_vals = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h_val[i_minus] if 0 <= i_minus < n_points else 0.0
            grad_vals[i] = term1 - term2
        return grad_vals

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()
