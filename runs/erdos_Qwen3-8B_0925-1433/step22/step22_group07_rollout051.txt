import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    T = 1.0
    beta = 0.995
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        perturb_size = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, perturb_size, replace=False)
        std_h = np.std(h)
        delta = np.random.normal(0, 0.05 * std_h, perturb_size)
        for i, idx in enumerate(indices):
            h[idx] += delta[i]
            h[idx] = np.clip(h[idx], 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        else:
            prob = np.exp( -(new_c5 - best_c5) / T )
            if np.random.rand() < prob:
                best_h = h.copy()
                best_c5 = new_c5
        T *= beta
        if T < 1e-5:
            break
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
