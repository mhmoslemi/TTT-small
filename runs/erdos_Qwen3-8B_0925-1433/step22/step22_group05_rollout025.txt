import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    
    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial population with random perturbations
    initial_population = [initial_h * target_sum / np.sum(initial_h)]
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        initial_population.append(candidate)
    
    # Run differential evolution with time budget
    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                    popsize=15, mutation=(0.5, 1), recombination=0.8,
                                    tol=1e-6, maxiter=200, seed=seed)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Final post-processing
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h)
    
    return (best_h, best_c5, n_points)
