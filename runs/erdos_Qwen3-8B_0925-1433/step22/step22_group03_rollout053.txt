import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)

    start_time = time.time()

    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    for _ in range(200):

        fitness = np.array([evaluate(ind) for ind in population])
        elite = population[np.argsort(fitness)[:10]]
