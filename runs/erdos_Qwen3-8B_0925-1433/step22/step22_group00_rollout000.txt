import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if abs(total - target_sum) > 1e-5:
            x = x / total * target_sum
        return x
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    for _ in range(100):

        scores = [obj(p) for p in population]
        elites = np.argsort(scores)[:population_size//5]
