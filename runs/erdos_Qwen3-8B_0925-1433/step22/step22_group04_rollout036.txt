from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x, k_max):
        h = x * target_sum / np.sum(x)
        N = len(h)
        grad = np.zeros_like(h)
        for i in range(N):
            i_plus = i + k_max
            if 0 <= i_plus < N:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - k_max
            if 0 <= i_minus < N:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def get_max_overlap(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx, np.argmax(corr)

    def adaptive_grad_descent(h, max_iter=500, lr=0.1, decay=0.999):
        best_h = h.copy()
        best_val = get_max_overlap(h)[0]
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            val, k_max = get_max_overlap(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
            grad = compute_gradient(h, k_max)
            m = np.zeros_like(h)
            v = np.zeros_like(h)
            for i in range(len(h)):
                m[i] = decay * m[i] + (1 - decay) * grad[i]
                v[i] = decay * v[i] + (1 - decay) * grad[i] ** 2
                h[i] = h[i] - lr * m[i] / (np.sqrt(v[i]) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        return best_h, best_val

    start_time = time.time()
    h_opt, c5_opt = adaptive_grad_descent(h)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    def nelder_mead_refine(h):
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    best_h = nelder_mead_refine(best_h)
