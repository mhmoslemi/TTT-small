def rescale_to_sum(h, target_sum):

    sum_current = np.sum(h)

    if sum_current == 0:

        return np.zeros_like(h)

    scaling_factor = target_sum / sum_current

    h_scaled = h * scaling_factor

    h_clipped = np.clip(h_scaled, 0, 1)

    sum_clipped = np.sum(h_clipped)

    if sum_clipped == 0:

        return np.zeros_like(h)

    return h_clipped * (target_sum / sum_clipped)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()  # Assume this is the provided initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale_to_sum(h, target_sum):
        sum_current = np.sum(h)
        if sum_current == 0:
            return np.zeros_like(h)
        scaling_factor = target_sum / sum_current
        h_scaled = h * scaling_factor
        h_clipped = np.clip(h_scaled, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        return h_clipped * (target_sum / sum_clipped)

    h_values = np.array(initial_h_values, dtype=float)
    h_values = rescale_to_sum(h_values, target_sum)
    best_h = h_values.copy()
    best_c5 = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx

    max_iterations = 1000
    stagnation_threshold = 50
    iterations_without_improvement = 0

    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        # Find critical lag (exclude lag 0)
        crit_idx = np.argmax(correlation[1:])  # from index 1 to end
        lag = crit_idx - (n_points - 1)

        # Determine overlap indices for positive lag
        if lag > 0:
            overlap_end = n_points - lag
            overlap_indices = np.arange(0, overlap_end)
        else:
            # For simplicity, handle negative lag with full array
            overlap_indices = np.arange(0, n_points)

        # Compute directional gradient
        # Perturb up
        perturb_up = h_values.copy()
        perturb_up[overlap_indices] += 1e-4
        perturb_up = rescale_to_sum(perturb_up, target_sum)
        corr_up = np.correlate(perturb_up, 1 - perturb_up, mode='full')
        grad_up = corr_up[crit_idx] - correlation[crit_idx]

        # Perturb down
        perturb_down = h_values.copy()
        perturb_down[overlap_indices] -= 1e-4
        perturb_down = rescale_to_sum(perturb_down, target_sum)
        corr_down = np.correlate(perturb_down, 1 - perturb_down, mode='full')
        grad_down = corr_down[crit_idx] - correlation[crit_idx]

        grad = (grad_up + grad_down) / 2
        adjustment = -np.sign(grad) * 0.01 * np.abs(grad)

        # Apply adjustment
        new_h = h_values.copy()
        new_h[overlap_indices] += adjustment
        new_h = rescale_to_sum(new_h, target_sum)
        new_h = np.clip(new_h, 0, 1)

        # Compute new_c5
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx

        if new_c5 < best_c5:
            h_values = new_h.copy()
            best_c5 = new_c5
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1

        if iterations_without_improvement >= stagnation_threshold:
            # Random perturbation
            h_values += np.random.normal(0, 0.05, n_points)
            h_values = np.clip(h_values, 0, 1)
            h_values = rescale_to_sum(h_values, target_sum)
            iterations_without_improvement = 0

    # Final check
    final_correlation = np.correlate(h_values, 1 - h_values, mode='full')
    best_c5 = np.max(final_correlation) * dx
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
