from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    initial_h = np.array(initial_h_values, dtype=float)
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    def perturb(x, method='shift'):

            x *= 1 + np.random.normal(0, 0.1, size=n_points)
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)

    def optimize_initial(guess):
        x0 = guess.copy()

        res = minimize(objective, x0, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, jac=compute_gradient, tol=1e-6)
