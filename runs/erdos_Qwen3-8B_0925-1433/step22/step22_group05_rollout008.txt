import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial population with perturbations and structured patterns
    initial_population = []
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        h = initial_h_values + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        initial_population.append(h)
    for _ in range(5):
        pattern = np.zeros(n_points)
        pattern[::3] = 1
        pattern = pattern * target_sum / np.sum(pattern)
        initial_population.append(pattern)

    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.9, 
                                   maxiter=200)

    best_h = result.x
    best_c5 = objective(best_h)

    # Evaluate initial population
    for h in initial_population:
        curr_c5 = objective(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
