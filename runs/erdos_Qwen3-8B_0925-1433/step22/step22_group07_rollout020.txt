import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def perturb_worst_lags(h):
        corr = np.correlate(h, 1 - h, mode='full')
        worst_indices = np.argsort(corr)[-5:]
        h = h.copy()
        for idx in worst_indices:
            if idx < len(h):
                i = np.random.randint(0, len(h) - 1)
                j = (i + idx) % len(h)
                if np.random.rand() < 0.5:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 0, 1)
                else:
                    h[j] = np.clip(h[j] + np.random.normal(0, 0.03), 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    start_time = time.time()
    temp = 1.0
    decay = 0.999
    for _ in range(100000):
        if time.time() - start_time > budget_s:
            break
        h = perturb_worst_lags(h)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        if np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            best_c5 = new_c5
        temp *= decay
        if temp < 1e-5:
            break
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
