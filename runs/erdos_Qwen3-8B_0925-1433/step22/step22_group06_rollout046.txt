import numpy as np
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    for _ in range(int(budget_s / 0.1)):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(cross_corr)
        k = k_max - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += dx * (1 - h[i + k])
            if 0 <= i - k < n_points:
                grad[i] -= dx * h[i - k]
        
        alpha = 0.03 * (1 - _ / (budget_s / 0.1))
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < 0.1:
            h_new += np.random.normal(0, 0.05, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = cross_corr.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
