import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    seeds = [initial_h_values.copy()]
    for i in range(n_points):
        seeds.append(np.array([1 if i % 2 == 0 else 0 for i in range(n_points)], dtype=float))
    for i in range(n_points):
        seeds.append(0.5 + 0.5 * np.sin(2 * np.pi * i / n_points))
    for i in range(n_points):
        seeds.append(initial_h_values.copy() + np.random.normal(0, 0.05, size=n_points))
    for initial_h in seeds:
        if np.sum(initial_h) < 1e-8:
            continue
        h = initial_h * target_sum / np.sum(initial_h)
        h = np.clip(h, 0, 1)
        best_c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if best_c5_current < best_c5:
            best_h, best_c5 = h.copy(), best_c5_current
        def obj(x):
            x = x * target_sum / np.sum(x)
            return np.max(np.correlate(x, 1 - x, mode='full')) * dx
        def grad(x):
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            k_max_index = np.argmax(corr)
            shift = k_max_index - (n_points - 1)
            grad = np.zeros_like(x)
            for i in range(n_points):
                i_plus = i + shift
                term1 = 1.0 if 0 <= i_plus < n_points else 0.0
                i_minus = i - shift
                term2 = x[i_minus] if 0 <= i_minus < n_points else 0.0
                grad[i] = (1.0 - x[i_plus]) - term2
            return grad
        for _ in range(100):
            if time.time() - time.time() > budget_s:
                break
            grad_vals = grad(h)
            h = h - 0.1 * grad_vals
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
            if current_c5 < best_c5:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
