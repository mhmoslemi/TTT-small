```python
import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.1
    decay = 0.999
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        if lag >= 0:
            offset = lag
            valid_i = np.arange(n_points) - offset
            valid_mask = valid_i >= 0
            gradient[valid_mask] = 1 - h[valid_i[valid_mask]]
        else:
            offset = lag
            valid_i = np.arange(n_points) - offset
            valid_mask = valid_i < n_points
            gradient[valid_mask] = 1 - h[valid_i[valid_mask]]
        gradient *= dx
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        curr_corr = np.correlate(h, 1 - h, mode='full')
        curr_c5 = np.max(curr_corr) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        learning_rate *= decay

    h_values = best_h
    return (h_values, best_c5, n_points)
```
