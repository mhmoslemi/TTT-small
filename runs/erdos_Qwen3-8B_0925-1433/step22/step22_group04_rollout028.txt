        k_lags = np.argsort(corr)[-5:]
        grad = np.zeros_like(h)
        for lag_idx in k_lags:
            shift = lag_idx - (n_points - 1)
            for i in range(n_points):
                i_plus = i + shift
                term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
                i_minus = i - shift
                term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
                grad[i] += term1 - term2
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
            biased_m = m[i] / (1 - momentum ** (_ + 1))
            biased_v = v[i] / (1 - decay_rate ** (_ + 1))
            h[i] = h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        learning_rate *= 0.99

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
