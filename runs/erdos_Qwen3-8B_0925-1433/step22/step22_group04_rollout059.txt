            i_plus = i + shift
            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)
    start_time = time.time()

    seeds = [initial_h_values, 
             0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)), 
             np.zeros(n_points), 
             np.ones(n_points)]

    for seed in seeds:
        h = np.array(seed, dtype=float)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h_initial + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
