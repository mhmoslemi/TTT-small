def rescale_to_sum(h, target_sum):
    sum_current = np.sum(h)
    if sum_current == 0:
        return np.zeros_like(h)
    scaling_factor = target_sum / sum_current
    h_scaled = h * scaling_factor
    h_clipped = np.clip(h_scaled, 0, 1)
    sum_clipped = np.sum(h_clipped)
    if sum_clipped == 0:
        return np.zeros_like(h)
    return h_clipped * (target_sum / sum_clipped)
