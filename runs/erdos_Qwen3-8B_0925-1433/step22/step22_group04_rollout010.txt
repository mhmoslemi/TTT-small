import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def optimize(initial_h, budget_s):
        h = initial_h.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        learning_rate = 0.1
        momentum = 0.9
        decay_rate = 0.999
        max_iter = 200
        start_time = time.time()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = momentum * m[i] + (1 - momentum) * grad[i]
                v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
                biased_m = m[i] / (1 - momentum ** (_ + 1))
                biased_v = v[i] / (1 - decay_rate ** (_ + 1))
                h[i] = h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
            h = np.clip(h, 0, 1)

                best_c5 = current_c5
