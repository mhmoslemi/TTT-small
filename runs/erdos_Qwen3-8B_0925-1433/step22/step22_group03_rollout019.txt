import numpy as np
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population_size = 30
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    F = 0.8
    CR = 0.9
    generations = 300
    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            a_vec = population[a]
            b_vec = population[b]
            c_vec = population[c]
            mutant = a_vec + F * (b_vec - c_vec)
            mask = np.random.rand(n_points) < CR
            trial = np.where(mask, mutant, population[i])
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros_like(trial) + 1e-6
            trial = trial * target_sum / trial_sum
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(corr) * dx
            current_corr = np.correlate(population[i], 1 - population[i], mode='full')
            current_c5 = np.max(current_corr) * dx
            if trial_c5 < current_c5:
                population[i] = trial.copy()
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
        current_best_c5 = best_c5

            c5 = np.max(corr) * dx
            if c5 < current_best_c5:
                current_best_c5 = c5
                best_h = h.copy()
        best_c5 = current_best_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
