import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, sigma=0.05):
        perturb = np.random.normal(0, sigma, n_points)
        child = h + perturb
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)
        return child

    def crossover(h1, h2):
        alpha = np.random.rand() * 0.5 + 0.5
        child = alpha * h1 + (1 - alpha) * h2
        return child

    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(49):
        h = np.array(initial_h_values, dtype=float)
        perturb = np.random.normal(0, 0.05, n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(len(population)):
            i1, i2, i3 = np.random.choice(len(population), 3, replace=False)
            h1, h2, h3 = population[i1], population[i2], population[i3]
            h = crossover(h1, h2)
            h = mutate(h)
            new_pop.append(h)
        population = new_pop
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    def obj(h):
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_c5 = compute_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
