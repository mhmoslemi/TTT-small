import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Validate initial_h_values if available
    if not (np.all(0 <= initial_h_values) and np.all(initial_h_values <= 1) and np.isclose(np.sum(initial_h_values), target_sum, atol=1e-6)):
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * target_sum / np.sum(h_initial)
    else:
        h_initial = initial_h_values.copy()

    # Initialize population
    population_size = 50
    population = [np.clip(h_initial + np.random.normal(0, 0.1, n_points), 0, 1) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] * target_sum / np.sum(population[i])

    best_h = h_initial.copy()
    best_c5 = float('inf')

    start_time = time.time()
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(population_size):
            # Select three distinct indices
            indices = np.random.choice([j for j in range(population_size) if j != i], 3, replace=False)
            x1, x2, x3 = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = x1 + 0.5 * (x2 - x3)  # Mutation step
            target = population[i]  # Target candidate
            new_candidate = target.copy()
            # Crossover with CR=0.8
            for j in range(n_points):
                if np.random.rand() < 0.8:
                    new_candidate[j] = mutant[j]
            # Clip and rescale
            new_candidate = np.clip(new_candidate, 0, 1)
            new_candidate = new_candidate * target_sum / np.sum(new_candidate)
            new_population.append(new_candidate)
        population = new_population
        # Evaluate best in population
        best_curr = min(population, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
        curr_c5 = np.max(np.correlate(best_curr, 1 - best_curr, mode='full')) * dx
        if curr_c5 < best_c5:
            best_h = best_curr.copy()
            best_c5 = curr_c5
        # Local refinement every 100 generations
        if gen % 100 == 0:
            sorted_pop = sorted(population, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
            for cand in sorted_pop[:5]:
                res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, x0=cand, method='SLSQP', bounds=[(0,1)]*n_points,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
                if res.success:
                    optimized = res.x
                    optimized = np.clip(optimized, 0, 1)
                    optimized = optimized * target_sum / np.sum(optimized)
                    curr_c5_opt = np.max(np.correlate(optimized, 1 - optimized, mode='full')) * dx

    return (best_h, best_c5, n_points)
