import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_corr = np.argmax(corr)
        shift = k_max_corr - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = Bounds(0, 1)

    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds,
                      constraints=cons, jac=gradient, tol=1e-6,
                      options={'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun
    if time.time() - start_time > budget_s:
        return (np.clip(best_h, 0, 1), best_c5, n_points)

    result_nm = minimize(objective, best_h, method='Nelder-Mead',
                         bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result_nm.x
    best_c5 = result_nm.fun
    if time.time() - start_time > budget_s:
        return (np.clip(best_h, 0, 1), best_c5, n_points)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(objective, h_perturb, method='L-BFGS-B',
                       bounds=bounds, constraints=cons, jac=gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
