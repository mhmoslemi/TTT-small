import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h_scaled = initial_h * target_sum / np.sum(initial_h)
    population_size = 50
    population = [initial_h_scaled + np.random.normal(0, 0.05, n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([eval_c5(p) for p in population])
        top_indices = np.argsort(fitness)[:population_size // 2]
        parents = [population[i] for i in top_indices]
        new_population = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0, 1) * target_sum / np.sum(np.clip(child, 0, 1))
            new_population.append(child)
        for _ in range(population_size // 2):
            p = np.random.choice(parents)
            noise = np.random.normal(0, 0.03, n_points)
            mutated = p + noise
            mutated = np.clip(mutated, 0, 1) * target_sum / np.sum(np.clip(mutated, 0, 1))
            new_population.append(mutated)
        population = new_population
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_h = population[current_best]
            best_c5 = fitness[current_best]
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
