import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(10):
        pert = np.random.normal(0, 0.1, n_points)
        h = initial_h_values + pert
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(50):
            idx1, idx2, idx3 = np.random.choice(len(population), 3, replace=False)
            x1, x2, x3 = population[idx1], population[idx2], population[idx3]
            mutant = x1 + 0.5 * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            mutant = mutant * target_sum / mutant_sum
            new_pop.append(mutant)
        population = new_pop
        for h in population:
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
