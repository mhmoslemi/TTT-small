import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate initial population with structured patterns
    def generate_patterns():
        h_initial = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
        h_alternating = np.zeros(n_points)
        for i in range(n_points):
            h_alternating[i] = 1.0 if i % 2 == 0 else 0.0
        h_alternating = h_alternating * target_sum / np.sum(h_alternating)
        h_random = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        h_staggered = np.zeros(n_points)
        block_size = n_points // 5
        for i in range(n_points):
            if i % (block_size * 2) < block_size:
                h_staggered[i] = 1.0
            else:
                h_staggered[i] = 0.0
        h_staggered = h_staggered * target_sum / np.sum(h_staggered)
        return [h_initial, h_alternating, h_random, h_staggered]
    
    initial_population = generate_patterns()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(x):
        x_proj = np.clip(x, 0.0, 1.0)
        sum_proj = np.sum(x_proj)
        if sum_proj == 0:
            return np.inf
        x_proj *= target_sum / sum_proj
        corr = np.correlate(x_proj, 1 - x_proj, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=10, tol=1e-6, mutation=(0.5, 1), 
        crossover=0.9, maxiter=100, polish=True, 
        seed=seed, disp=False
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Final clipping and projection
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        best_h = best_h * target_sum / sum_best
    
    return (best_h, best_c5, n_points)
