from scipy.optimize import minimize, Bounds
import time

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

            h = h.copy()
            h[start:end] = 1.0 - h[start:end]
            return h
        else:
            return h

    def optimize_candidate(h):
