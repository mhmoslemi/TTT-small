from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    max_iterations = 200
    learning_rate = 0.1
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr)
        max_idx = np.argmax(corr)
        
        gradients = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            obj_plus = np.max(corr_plus) * dx
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
