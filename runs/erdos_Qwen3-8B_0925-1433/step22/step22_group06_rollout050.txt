import numpy as np

import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    # Initialize parameters
    max_iterations = int(budget_s / 0.1)  # ~6000 iterations
    alpha = 0.1
    decay_rate = 0.999
    no_improvement_count = 0
    improvement_threshold = 0.0001

    for _ in range(max_iterations):
        # Compute correlation and find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)  # convert index to actual lag

        # Compute gradient for this lag
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = (i + lag) % n_points
            gradient[i] = dx * (1 - h[j])

        # Update h
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Evaluate new C5
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Adjust learning rate
        if no_improvement_count > 10:
            alpha *= 0.5
            no_improvement_count = 0

        # Early stopping
        if best_c5 < improvement_threshold:
            break

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
