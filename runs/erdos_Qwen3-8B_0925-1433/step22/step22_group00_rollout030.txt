import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale_to_sum(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h_clipped)
        scaling = target_sum / current_sum
        return h_clipped * scaling

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = rescale_to_sum(initial_h, target_sum)
    population = [initial_h]
    for _ in range(49):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = rescale_to_sum(h_rand, target_sum)
        population.append(h_rand.copy())

    def compute_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = compute_objective(best_h)
    start_time = time.time()
    F = 0.8
    CR = 0.9
    pop_size = len(population)

    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            mask = np.random.rand(n_points) < CR
            trial = np.where(mask, mutant, trial)
            trial = rescale_to_sum(trial, target_sum)
            trial_obj = compute_objective(trial)
            current_obj = compute_objective(population[i])
            if trial_obj < current_obj:
                population[i] = trial.copy()
        current_best = np.inf
        best_idx = 0
        for idx, h in enumerate(population):
            obj = compute_objective(h)
            if obj < current_best:
                current_best = obj
                best_idx = idx
        if current_best < best_c5:

            best_h = population[best_idx].copy()
