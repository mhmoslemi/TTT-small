import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        h = x * target_sum / np.sum(x)
        return compute_overlap(h)
    
    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)

            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0

            grad[i] = term1 - term2
        return grad
