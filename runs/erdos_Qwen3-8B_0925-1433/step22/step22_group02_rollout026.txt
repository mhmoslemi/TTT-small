        h1 = h1 * target_sum / np.sum(h1)
        h2 = np.random.rand(n_points)
        h2 = h2 * target_sum / np.sum(h2)
        return h0, h1, h2

    h0, h1, h2 = generate_initial_guesses()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def perturb(h, type='noise'):
        if type == 'noise':
            pert = np.random.normal(0, 0.05, size=n_points)
        elif type == 'shift':
            pert = np.roll(h, 1)
        elif type == 'sinusoidal':

        elif type == 'swap':
            i, j = np.random.choice(n_points, 2, replace=False)
            pert = h.copy()
            pert[[i, j]] = pert[[j, i]]
        else:
            pert = h.copy()
        pert = pert * target_sum / np.sum(pert)
        return np.clip(pert, 0, 1)

    start_time = time.time()

    def optimize_phase(h, method='Adam', budget=200):
        nonlocal best_c5, best_h
        if method == 'Adam':
            learning_rate = 0.5

            epsilon = 1e-8
