import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    h = np.array(initial_h_values, dtype=float)

    scaling_factor = target_sum / current_sum
    h *= scaling_factor
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[n_points-1:2*n_points-1]) * (2.0 / n_points)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    max_iterations = 1000
    step_size = 0.01
    eps = 1e-5
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        positive_lags = corr[n_points-1:2*n_points-1]
        max_index = np.argmax(positive_lags) + n_points - 1
        max_corr = corr[max_index]
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            h_perturb = np.clip(h_perturb, 0, 1)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            corr_perturb_k = corr_perturb[max_index]
            h_orig = h.copy()
            h_orig[i] -= eps
            h_orig = np.clip(h_orig, 0, 1)
            corr_orig = np.correlate(h_orig, 1 - h_orig, mode='full')
