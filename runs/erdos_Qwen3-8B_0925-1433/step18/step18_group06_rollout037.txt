import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Initialize population with initial pattern and diverse candidates
    population = [np.clip(initial_h_values, 0, 1) * target_sum / np.sum(initial_h_values)]
    for _ in range(100):
        r = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        s = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)) * target_sum / np.sum(0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        p = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) * 0.5 + 0.5
        p = p * target_sum / np.sum(p)
        m = np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1)
        m = m * target_sum / np.sum(m)
        a = (np.arange(n_points) % 2) * 0.5 + 0.5
        a = a * target_sum / np.sum(a)
        population.extend([r, s, p, m, a])
    population = np.array([clip_normalize(h) for h in population[:100]])

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fit = [compute_c5(h) for h in population]
        # Tournament selection
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 5, replace=False)
            best = idxs[np.argmin(fit[idxs])]
            parents.append(population[best])
        # Generate offspring
        offspring = []
        for _ in range(100):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = 0.8 * p1 + 0.2 * p2
            if np.random.rand() < 0.2:
                fft1 = np.fft.fft(p1)
                fft2 = np.fft.fft(p2)
                child = np.fft.ifft((np.abs(fft1) + np.abs(fft2)) * np.exp(1j * np.angle(fft1))).real
            child = clip_normalize(child + np.random.normal(0, 0.2, n_points))
            offspring.append(child)
        # Evaluate and select
        fit_off = [compute_c5(h) for h in offspring]
        combined = list(zip(population, fit)) + list(zip(offspring, fit_off))
        combined.sort(key=lambda x: x[1])
        population = [c[0] for c in combined[:100]]
        # Update best
        if compute_c5(population[0]) < best_c5:
            best_h = population[0]
            best_c5 = compute_c5(best_h)

    return (clip_normalize(best_h), best_c5, n_points)
