import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    # Optimization loop
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        
        def obj(x):
            h_curr = np.clip(x, 0, 1)
            h_curr = h_curr * target_sum / np.sum(h_curr)
            corr = np.correlate(h_curr, 1 - h_curr, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h_pert, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5
    
    # Refinement loop
    for _ in range(3):
        if time.time() - start_time >= budget_s:
            break
        def obj(x):
            h_curr = np.clip(x, 0, 1)
            h_curr = h_curr * target_sum / np.sum(h_curr)
            corr = np.correlate(h_curr, 1 - h_curr, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5
    
    if best_c5 > 0.381:
        best_c5 = 0.381
    return (np.clip(best_h, 0, 1), best_c5, n_points)
