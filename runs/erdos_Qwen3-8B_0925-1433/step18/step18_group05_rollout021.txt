        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + lag < n_points:
                grad[j] += 1 - h[j + lag]
            if j - lag >= 0:
                grad[j] -= h[j - lag]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        learning_rate *= 0.998
    return best_h, best_c5, n_points
