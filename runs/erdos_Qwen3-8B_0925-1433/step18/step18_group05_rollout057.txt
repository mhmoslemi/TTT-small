import numpy as np
import time

def run(seed=42, budget_s=600):

    def mutate(h, top_lag, mutation_rate):

        delta = mutation_rate * 0.05
        i = np.argmax(np.abs(h_mut - np.roll(h_mut, top_lag)))
        j = (i + top_lag) % n_points
        h_mut[i] -= delta
        h_mut[j] += delta
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        return h_mut

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population = [initial_h + np.random.uniform(-0.1, 0.1, size=n_points) for _ in range(50)]
    population = [h * target_sum / np.sum(h) for h in population]
    population = [np.clip(h, 0, 1) for h in population]

    best_h = initial_h.copy()

        c5_values = [evaluate(h) for h in population]

        elite_size = int(elite_fraction * len(population))
