import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)

    def objective(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.inf
        h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=15, mutation=0.8, recombination=0.9,
                                   tol=1e-6, maxiter=75, seed=seed, polish=True)

    best_h = result.x
    best_c5 = objective(best_h)

    # Local refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
    if res.success:
        best_h = res.x
        best_c5 = objective(best_h)

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
