        def objective(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        if res.success:
            current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
