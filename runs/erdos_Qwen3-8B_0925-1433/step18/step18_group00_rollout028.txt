from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Base configuration
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Generate structured initial guess with regular spacing
        m = int(np.ceil(n_points / 2))
        positions = np.linspace(0, 2, m + 1)[:-1]
        lengths = np.full(m, 1.0 / m)
        h_initial = np.zeros(n_points)
        for i, (start, length) in enumerate(zip(positions, lengths)):
            end = start + length

            h_initial[idx] = 1.0
        h_initial = h_initial * target_sum / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)

        # Perturb
        pert = np.random.normal(0, 0.05 * dx, n_points)
        h_pert = np.clip(h_initial + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(objective, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:

    # Final refinement
    for _ in range(3):
