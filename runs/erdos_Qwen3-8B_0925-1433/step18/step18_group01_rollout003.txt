import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def calculate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 60
    pop = []

    for i in range(population_size):
        if i < population_size // 3:
            h = np.array(initial_h_values, dtype=float)
            h = h * target_sum / np.sum(h)
        elif i < 2 * population_size // 3:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)
            h[:n_points // 2] = 1.0
            h = h * target_sum / np.sum(h)
        pop.append(h)

    best_h = pop[0]
    best_c5 = calculate_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [calculate_c5(ind) for ind in pop]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = pop[best_idx].copy()
            best_c5 = fitness[best_idx]

        parents = []
        for _ in range(population_size // 2):
            ids = np.random.choice(population_size, 3)
            best_id = ids[np.argmin(fitness[ids])]
            parents.append(pop[best_id])

        offspring = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) * 0.5
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)

        combined = pop + offspring
        fitness_combined = [calculate_c5(ind) for ind in combined]
        combined_idx = np.argsort(fitness_combined)
        pop = [combined[i] for i in combined_idx[:population_size]]

    return (np.clip(best_h, 0, 1), best_c5, n_points)
