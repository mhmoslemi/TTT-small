import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * 45 / 2 / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points

    current_h = np.clip(initial_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = float('inf')
    
    max_iterations = 1000
    step_decay = 0.995
    step_size = 0.1
    perturbation_freq = 100
    perturbation_amp = 0.05
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr[1:-1])
        full_idx = max_idx + 1
        k = full_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += dx * (1 - current_h[i + k])
            if i - k >= 0:
                gradient[i] -= dx * current_h[i - k]
        step_size = 0.1 * (step_decay ** iteration)
        current_h -= step_size * gradient

        current_h = current_h * target_sum / np.sum(current_h)

        current_c5 = np.max(corr_current[1:-1]) * dx
