        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:n_points]) * dx

    population_size = 50
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    gbest = None
    gbest_fitness = float('inf')

    for _ in range(population_size):
        h = initial_h.copy()
        perturb = np.random.normal(0, 0.05, n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        v = np.random.uniform(-0.05, 0.05, n_points)
        particles.append(h)
        velocities.append(v)
        pbest.append(h)
        pbest_fitness.append(compute_c5(h))
    
    for i in range(population_size):
        if pbest_fitness[i] < gbest_fitness:
            gbest_fitness = pbest_fitness[i]
            gbest = particles[i].copy()
    
    best_h = gbest.copy()
    best_c5 = gbest_fitness
    start_time = time.time()

    inertia = 0.9
    inertia_decay = 0.5 / 100
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    improve_thres = 1e-5
    no_improve = 0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        inertia = 0.9 - _ * inertia_decay
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] * inertia + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            new_h = particles[i] + v
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            new_fit = compute_c5(new_h)
            if new_fit < pbest_fitness[i]:
                pbest[i] = new_h.copy()
                pbest_fitness[i] = new_fit
                if new_fit < gbest_fitness:
                    gbest_fitness = new_fit
                    gbest = new_h.copy()
            particles[i] = new_h
            velocities[i] = v
        if abs(gbest_fitness - best_c5) < improve_thres:
            no_improve += 1
        else:
            best_c5 = gbest_fitness
            no_improve = 0
        if no_improve > 100:
            break

    best_h = gbest.copy()
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
