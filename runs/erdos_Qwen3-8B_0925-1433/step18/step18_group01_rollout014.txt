import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_used = 0
    start_time = time.time()

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    # Generate diverse starting points
    starting_points = [h.copy()]
    for _ in range(2):
        rand_h = np.random.rand(n_points)
        rand_h = np.clip(rand_h, 0, 1)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        starting_points.append(rand_h)
    for _ in range(2):
        struct_h = np.zeros(n_points)
        struct_h[::2] = 1.0
        struct_h = struct_h * target_sum / np.sum(struct_h)
        struct_h = np.clip(struct_h, 0, 1)
        starting_points.append(struct_h)

    for h_init in starting_points:
        if time.time() - start_time >= budget_s:
            break
        res = minimize(obj, h_init, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    # Refinement loop
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    final_h = np.clip(best_h, 0, 1)
    return (final_h, best_c5, n_points)
