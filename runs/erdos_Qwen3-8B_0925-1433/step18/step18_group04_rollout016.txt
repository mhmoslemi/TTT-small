    initial_h = initial_h * target_sum / np.sum(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = 0.381

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_c5 = compute_c5(h)
    start_time = time.time()

    max_iterations = int(budget_s / 10)
    learning_rate = 0.01
    decay_rate = 0.99

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = corr[1:-1]
        max_overlap = np.max(overlap)
        k = np.argmax(overlap)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                gradient[i] -= h[i - k] * dx

        h += -learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
