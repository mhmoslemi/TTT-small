    def refine_h(h_initial, tol=1e-5):

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=tol)

        return h_initial

    start_time = time.time()
