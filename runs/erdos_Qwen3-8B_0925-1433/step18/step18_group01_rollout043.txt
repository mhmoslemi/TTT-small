import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    population = []
    
    # Initialize population with diverse samples
    for _ in range(10):
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(10):
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx].copy(), fitness[best_idx]
        
        # Tournament selection
        parents = []
        for _ in range(len(population) // 2):
            idxs = np.random.choice(len(population), 3, replace=False)
            winner = idxs[np.argmin(fitness[idxs])]
            parents.append(population[winner])
        
        # Generate offspring
        offspring = []
        for _ in range(len(parents) // 2):
            p1, p2 = random.sample(parents, 2)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Replace worst half
        sorted_idx = np.argsort(fitness)
        for i in range(len(population) // 2):
            population[sorted_idx[i]] = offspring[i]
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (np.clip(best_h, 0, 1), best_c5, n_points)
