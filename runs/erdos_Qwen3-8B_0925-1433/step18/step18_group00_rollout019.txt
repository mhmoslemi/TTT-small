    h_perturbed = scale_guess(h_perturbed)
    initial_guesses.append(h_perturbed)

    best_h = h_initial.copy()
    best_c5 = obj(h_initial)
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
