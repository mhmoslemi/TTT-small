        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = initial_h * target_sum / np.sum(initial_h)
    h = initial_h.copy()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    step_size = 0.01
    decay_rate = 0.999
    max_iter = 1000
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr[1:-1])
        k = max_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points - k):
            j = i + k
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)

        if iter_count % 500 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (target_sum / sum_h)

        step_size *= decay_rate

    return (h, best_c5, n_points)
