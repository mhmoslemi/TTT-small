import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Initialize population
    pop_size = 50
    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)

    best_h = population[0]
    best_c5 = evaluate(best_h)
    time_start = time.time()

    mutation_step = 0.1
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        c5_values = np.array([evaluate(h) for h in population])
        argsort = np.argsort(c5_values)
        sorted_pop = [population[i] for i in argsort]
        best_idx = argsort[0]
        best_c5 = c5_values[best_idx]
        best_h = sorted_pop[0]

        # Crossover and mutation
        new_pop = sorted_pop[:int(0.2 * pop_size)]
        for _ in range(pop_size - len(new_pop)):
            a, b = np.random.choice(range(pop_size), 2, replace=False)
            pa, pb = sorted_pop[a], sorted_pop[b]
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * pa + (1 - alpha) * pb
            noise = np.random.normal(0, mutation_step, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
        mutation_step += 0.0001 if np.random.rand() < 0.5 else 0

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
