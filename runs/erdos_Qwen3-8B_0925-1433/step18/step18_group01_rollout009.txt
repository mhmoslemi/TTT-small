import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values and scale to meet constraint
    h = np.array(initial_h_values, dtype=float)
    h_scaled = h * target_sum / np.sum(h)
    best_h = h_scaled.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    time_start = time.time()
    tolerance = 1e-6

    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    for _ in range(5):
        if time.time() - time_start > budget_s:
            break
        # Generate diverse perturbation
        pert = np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        res = minimize(obj, h_pert, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance)
        if res.success:
            current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5

    # Refinement
    for _ in range(3):
        if time.time() - time_start > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance)
        if res.success:
            current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
