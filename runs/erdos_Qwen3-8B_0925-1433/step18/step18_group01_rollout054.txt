import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    # Initial candidates
    initial_candidates = []
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    initial_candidates.append(h_initial)

    # Add perturbed variants
    for _ in range(3):
        pert = np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_initial + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        initial_candidates.append(h_pert)

    # Add random samples
    for _ in range(2):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        initial_candidates.append(h_rand)

    # Add structured samples
    h_struct = np.zeros(n_points)
    h_struct[::2] = 1.0
    h_struct[1::2] = 0.5
    h_struct = np.clip(h_struct, 0, 1)
    h_struct = h_struct * target_sum / np.sum(h_struct)
    initial_candidates.append(h_struct)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()
    for candidate in initial_candidates:
        res = minimize(obj, candidate, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = obj(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    for _ in range(5):
        if time.time() - start_time > budget_s:

        pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(best_h + pert, 0, 1)
