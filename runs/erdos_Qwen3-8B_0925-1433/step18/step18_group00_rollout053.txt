        end = start + (1.0 / m)
        end = min(end, 2.0)
        indices = np.where((x >= start) & (x < end))[0]
        regular_h[indices] = 1.0
    regular_h = regular_h * target_sum / np.sum(regular_h)
    pert = np.random.normal(0, 0.05, n_points)
    perturbed = np.clip(regular_h + pert, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Optimize with L-BFGS-B
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Final refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
