import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h, k):
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        return grad

    # Alternative initial patterns
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
    triangular = (2 * (np.arange(n_points) / (n_points-1) - 0.5)**2 + 0.5)
    h_candidates = [h, sinusoidal, triangular]
    best_h = np.min([evaluate(h_candidate) for h_candidate in h_candidates], axis=0)
    best_h = h_candidates[np.argmin([evaluate(h_candidate) for h_candidate in h_candidates])]

    max_iterations = 200
    start_time = time.time()
    learning_rate = 0.1
    momentum = 0.99
    decay = 0.998
    velocity = np.zeros(n_points)
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = gradient(best_h, k)
        velocity = momentum * velocity - learning_rate * grad
        best_h += velocity
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = evaluate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        learning_rate *= decay

    for _ in range(10):

        if current_c5 < best_c5:
            best_h = perturbed
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
