from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    # Diverse initial guesses with varying perturbations
    initial_guesses = [h.copy()]
    for _ in range(9):
        if time.time() - start_time > budget_s * 0.4:
            break
        noise = np.random.normal(0, 0.15 if np.random.rand() < 0.5 else 0.2, n_points)
        pert = initial_h_values + noise
        h_pert = np.clip(pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        initial_guesses.append(h_pert)

    for h_initial in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Refinement with L-BFGS-B
    for _ in range(4):
        if time.time() - start_time > budget_s:

        h = np.clip(best_h, 0, 1)

        if res.success:
