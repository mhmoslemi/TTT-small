import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlaps(h):
        overlaps = []
        for k in range(1, n_points):
            part1 = h[:len(h)-k]
            part2 = 1 - h[k:]
            overlap = np.dot(part1, part2) * dx
            overlaps.append(overlap)
        return overlaps

    def compute_grad(h, k):
        grad = np.zeros(len(h))
        grad[:len(h)-k] = (1 - h[k:]) * dx
        grad[k:] -= h[:len(h)-k] * dx
        return grad

    def rescale_h(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = target_sum / current_sum
        return h * scaling_factor

    def clip_h(h):
        return np.clip(h, 0, 1)

    h = initial_h.copy()
    h = clip_h(h)
    h = rescale_h(h, target_sum)

    overlaps = compute_overlaps(h)
    best_c5 = max(overlaps)
    best_h = h.copy()
    start_time = time.time()
    step_size = 0.01
    step_decay_rate = 0.99
    max_iterations = 1000
    improvement_threshold = 1e-6
    stagnation_counter = 0

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        overlaps = compute_overlaps(h)
        if not overlaps:
            break
        max_overlap = max(overlaps)
        max_k = np.argmax(overlaps) + 1
        grad = compute_grad(h, max_k)
        delta = -step_size * grad
        h += delta

        new_overlaps = compute_overlaps(h)
        new_max_overlap = max(new_overlaps)
        if new_max_overlap < max_overlap:
            best_h = h.copy()
            best_c5 = new_max_overlap
            stagnation_counter = 0
