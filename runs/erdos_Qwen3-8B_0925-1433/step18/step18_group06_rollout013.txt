            perturb = np.random.normal(0, 0.1, n_points)
            h = np.clip(initial_h + perturb, 0, 1)
            h = h * target_sum / np.sum(h)
        else:
            h = (np.arange(n_points) % 2) * 0.5 + 0.5
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [evaluate(h) for h in population]
        
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population, size=tournament_size)
            winner = min(tournament, key=evaluate)
            parents.append(winner)
        
        offspring = []
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i+1]
            beta = np.random.uniform(0.5, 1.5)
            child = p1 * beta + p2 * (1 - beta)
            if np.random.rand() < 0.2:
                fft1 = fft.fft(p1)
                fft2 = fft.fft(p2)
                combined = (np.abs(fft1) + np.abs(fft2)) / 2
                child = fft.ifft(combined).real
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.uniform(-mutation_amp, mutation_amp)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        new_pop = population.copy()
        new_pop.extend(offspring)
        new_pop.sort(key=evaluate)
        population = new_pop[:population_size]
        
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
