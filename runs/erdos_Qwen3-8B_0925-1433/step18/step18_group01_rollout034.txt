    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    time_used = 0
    evals = 0

    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def objective(h):
        nonlocal evals
        evals += 1
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def tournament_select(pop, k=3):

        alpha = np.random.rand()
        return alpha * a + (1 - alpha) * b

    def mutate(h, sigma=0.05):

    start_time = time.time()
