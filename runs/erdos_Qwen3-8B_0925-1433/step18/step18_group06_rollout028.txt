import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1))

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def init_population():
        pop = []
        h = np.clip(initial_h_values, 0, 1)
        h = normalize(h)
        pop.append(h)
        for _ in range(4):
            if _ == 0:
                h = np.random.rand(n_points)
            elif _ == 1:
                h = 0.5 + 0.5 * np.sin(np.linspace(0, 2*np.pi, n_points))
            elif _ == 2:
                h = np.random.choice([0, 1], size=n_points) * 0.5 + 0.5
            elif _ == 3:
                h = np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1)
                h = normalize(h)
            else:
                h = (np.arange(n_points) % 2) * 0.5 + 0.5
            pop.append(normalize(h))
        return pop

    pop = init_population()
    best_h = pop[0]
    best_c5 = evaluate(best_h)

    def crossover(parent1, parent2):
        beta = np.random.uniform(0.5, 1.5)
        offspring = beta * parent1 + (1 - beta) * parent2
        return normalize(offspring)

    def freq_crossover(parent1, parent2):
        ft1 = np.fft.fft(parent1)
        ft2 = np.fft.fft(parent2)
        mag1 = np.abs(ft1)
        mag2 = np.abs(ft2)
        mag = np.random.uniform(mag1, mag2)  # random blend
        offspring = np.fft.ifft(mag * np.exp(1j * np.angle(ft1))).real
        return normalize(offspring)

    def mutate(h):
        h = h.copy()
        mask = np.random.rand(n_points) < 0.05
        h[mask] += np.random.uniform(-0.2, 0.2, np.sum(mask))
        return normalize(h)

    max_generations = 200
    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(h) for h in pop]
        idxs = np.argsort(scores)
        top_idx = idxs[-20:]
        parents = [pop[i] for i in top_idx]

        new_pop = []
        for _ in range(len(pop) // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child1 = crossover(p1, p2)
            child2 = freq_crossover(p1, p2)
            new_pop.extend([mutate(child1), mutate(child2)])
        pop = new_pop + [pop[i] for i in idxs[:len(pop)-len(new_pop)]]
        for h in pop:
            if evaluate(h) < best_c5:
                best_h = h.copy()
                best_c5 = evaluate(h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
