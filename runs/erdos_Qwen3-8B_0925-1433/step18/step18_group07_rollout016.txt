from scipy.optimize import minimize
import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj_func(h_vec):
        h = h_vec.copy()
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_func(h_vec):
        h = h_vec.copy()
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        return grad * target_sum / np.sum(h)

    # Adam optimizer
    learning_rate = 0.5
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    iterations = 0
    max_iter = 300
    start_time = time.time()

    while iterations < max_iter and time.time() - start_time < budget_s:
        grad = grad_func(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (iterations + 1))
        v_hat = v / (1 - beta2 ** (iterations + 1))
        h += learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = obj_func(h)
