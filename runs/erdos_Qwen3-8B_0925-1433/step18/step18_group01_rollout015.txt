import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    time_used = 0

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    T = 1.0
    cooling_rate = 0.995
    iterations = 0
    max_iter = int(budget_s * 0.8)

    for _ in range(max_iter):
        if time_used >= budget_s:
            break
        # Generate perturbation with temperature scaling
        pert = np.random.normal(0, 0.05 * T, n_points)
        h_pert = h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5
        T *= cooling_rate
        time_used += 0.1

    # Refinement phase
    for _ in range(3):
        if time_used >= budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:

                best_h, best_c5 = current_h.copy(), current_c5
        time_used += 0.1

    return (np.clip(best_h, 0, 1), best_c5, n_points)
