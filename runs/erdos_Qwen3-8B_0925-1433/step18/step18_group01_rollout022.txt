    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    max_iterations = int(budget_s / 0.1)

    # Simulated Annealing
    for _ in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        sigma = T * 0.15
        pert = np.random.normal(0, sigma, n_points)
        h_pert = h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        c5_pert = np.max(np.correlate(h_pert, 1 - h_pert, mode='full')) * dx
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        else:
            delta = c5_pert - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_pert.copy()
        T *= cooling_rate

    # Local Optimization
    if time.time() - start_time < budget_s:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        for _ in range(3):
            if time.time() - start_time >= budget_s:
                break
            res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                best_h = res.x
                best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
            # Prevent premature convergence by occasionally perturbing
            if np.random.rand() < 0.05:
                pert = np.random.normal(0, 0.05, n_points)
                best_h = np.clip(best_h + pert, 0, 1)
                best_h = best_h * target_sum / np.sum(best_h)
                c5_pert = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
                if c5_pert < best_c5:
                    best_c5 = c5_pert

    return (np.clip(best_h, 0, 1), best_c5, n_points)
