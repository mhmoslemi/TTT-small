        if time.time() - start_time > budget_s:
            break
        eval_list = []
        for h_candidate in population:
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            eval_list.append((c5, h_candidate))
        eval_list.sort()
        parents = [e[1] for e in eval_list[:int(len(population)*0.2)]]
        offspring = []
        
        while len(offspring) < len(population) - len(parents):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            beta = np.random.uniform(0.5, 1.5)
            child = beta * p1 + (1 - beta) * p2
            if np.random.rand() < 0.2:
                fft_p1 = np.fft.fft(p1)
                fft_p2 = np.fft.fft(p2)
                combined_mag = (np.abs(fft_p1) + np.abs(fft_p2)) / 2
                combined_phase = np.angle(fft_p1)
                combined_fft = combined_mag * np.exp(1j * combined_phase)
                child = np.fft.ifft(combined_fft).real
            mutate_indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
            delta = np.random.uniform(-0.2, 0.2, n_points)
            child[mutate_indices] += delta[mutate_indices]
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        
        combined = population + offspring
        eval_combined = []
        for h_candidate in combined:
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            eval_combined.append((c5, h_candidate))
        eval_combined.sort()
        population = [e[1] for e in eval_combined[:population_size]]
        
        if eval_list[0][0] < best_c5:
            best_c5 = eval_list[0][0]
            best_h = eval_list[0][1]
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
