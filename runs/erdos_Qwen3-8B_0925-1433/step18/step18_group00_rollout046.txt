from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    def optimize(guess):
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 200})

            h = res.x
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
