        perturbation = np.random.normal(0, 0.15, n_points)
        h_pert = np.array(initial_h_values, dtype=float) + perturbation
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        initial_guesses.append(h_pert)

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    learning_rate = 0.05
    decay_rate = 0.999
    max_iter = 300

    for guess in initial_guesses:
        h = np.array(guess, dtype=float)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            k = np.argmax(corr)
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += 1 - h[i + k]
                if i - k >= 0:
                    grad[i] += h[i - k]
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            learning_rate *= decay_rate

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += 1 - best_h[i + k]
            if i - k >= 0:
                grad[i] += best_h[i - k]
        best_h = best_h - 0.001 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
