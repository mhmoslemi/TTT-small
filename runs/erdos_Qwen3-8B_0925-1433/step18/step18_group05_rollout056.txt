import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, top_lag, mutation_rate):
        delta = mutation_rate * 0.03
        h_mut = h.copy()
        for i in range(n_points):
            j = (i + top_lag) % n_points
            h_mut[i] -= delta
            h_mut[j] += delta
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        return h_mut

    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(5):
        h = population[0].copy()
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    max_generations = 200
    mutation_rate = 0.1
    elite_fraction = 0.15
    no_improvement = 0

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        c5_values = [evaluate(h) for h in population]
        current_best_idx = np.argmin(c5_values)
        current_best_c5 = c5_values[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > 50:
            mutation_rate += 0.005
            no_improvement = 0
        elites = np.argsort(c5_values)[:int(elite_fraction * len(population))]
        new_population = [population[i].copy() for i in elites]
        for _ in range(len(population) - len(elites)):
            t1 = np.random.choice(len(population), size=5, replace=False)
            t2 = np.random.choice(len(population), size=5, replace=False)
            p1 = population[t1[np.argmin([c5_values[i] for i in t1])]]
            p2 = population[t2[np.argmin([c5_values[i] for i in t2])]]
            child = (p1 + p2) / 2.0
            corr = np.correlate(child, 1 - child, mode='full')
            max_lag = np.argmax(corr)
            child = mutate(child, max_lag, mutation_rate)
            new_population.append(child)
        population = np.array(new_population)
        mutation_rate *= 0.998

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
