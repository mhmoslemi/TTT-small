def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize h from initial_h_values
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate(h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.98
    max_iterations = 200

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        # Compute gradient
        grad = np.zeros_like(h)
        if k >= 0:
            for j in range(n_points):
                if j + k < n_points:
                    grad[j] += (1 - h[j + k])
        else:
            for j in range(n_points):
                if j - k >= 0:
                    grad[j] -= h[j - k]
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            h = h * target_sum / sum_h
        # Evaluate
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Decay learning rate
        learning_rate *= decay_rate

    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
