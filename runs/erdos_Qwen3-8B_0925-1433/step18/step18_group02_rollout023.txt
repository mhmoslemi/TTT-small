import numpy as np
import time

def run(seed=42, budget_s=600):
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    max_iter = 200
    learning_rate = 0.1
    decay_rate = 0.995
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx_plus = (i + k) % n_points
            idx_minus = (i - k) % n_points
            grad[i] = (1.0 - h[idx_plus]) - h[idx_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate
    
    return (h, best_c5, n_points)
