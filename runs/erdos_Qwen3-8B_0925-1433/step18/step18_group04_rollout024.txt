import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        n_points = 45
        initial_h = np.ones(n_points) * (n_points / 2) / n_points
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    def overlap_k(h, k):
        total = 0.0
        for i in range(n_points - k):
            total += h[i] * (1 - h[i + k])
        return total * dx

    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    eps = 1e-5
    iterations = 0
    max_iterations = 500
    time_start = time.time()

    while iterations < max_iterations and time.time() - time_start < budget_s:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap_idx = np.argmax(corr[1:-1])
        k = max_overlap_idx + 1

        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed_up = h.copy()
            h_perturbed_up[i] += eps
            h_perturbed_up = h_perturbed_up * target_sum / np.sum(h_perturbed_up)
            overlap_plus = overlap_k(h_perturbed_up, k)

            h_perturbed_down = h.copy()
            h_perturbed_down[i] -= eps
            h_perturbed_down = h_perturbed_down * target_sum / np.sum(h_perturbed_down)
            overlap_minus = overlap_k(h_perturbed_down, k)

            gradient[i] = (overlap_plus - overlap_minus) / (2 * eps)

        step_size = 0.03
        adjustment = -step_size * gradient
        h += adjustment

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        iterations += 1

    return (best_h, best_c5, n_points)
