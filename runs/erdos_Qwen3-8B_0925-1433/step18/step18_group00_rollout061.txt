        h_pert = np.clip(h_initial + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        # Binary uniform initial guess
        m = int(n_points // 3)
        x_binary = np.linspace(0, 2, m + 1)
        h_binary = np.zeros(n_points)
        for i in range(m):
            start = np.searchsorted(x_binary, x_binary[i], side='right')
            end = np.searchsorted(x_binary, x_binary[i+1], side='left')
            h_binary[start:end] = 1.0
        h_binary = h_binary * target_sum / np.sum(h_binary)
        h_binary = np.clip(h_binary, 0, 1)

        # Use binary as starting guess
        def binary_obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        res = minimize(binary_obj, h_binary, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 300})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

        # Use perturbed sine as starting guess
        def sine_obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        res = minimize(sine_obj, h_pert, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 300})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Final refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 300})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
