import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial configurations
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = sine_guess * target_sum / np.sum(sine_guess)
    initial_guesses.append(sine_guess)

    binary_guess = np.random.choice([0, 1], size=n_points)
    binary_guess = binary_guess * target_sum / np.sum(binary_guess)
    binary_guess = np.clip(binary_guess, 0, 1)
    initial_guesses.append(binary_guess)

    block_guess = np.zeros(n_points)
    for i in range(n_points):
        block_guess[i] = 1 if i % 2 == 0 else 0
    block_guess = block_guess * target_sum / np.sum(block_guess)
    block_guess = np.clip(block_guess, 0, 1)
    initial_guesses.append(block_guess)

    multiwave_guess = 0.5 + 0.5 * np.sin(np.pi * x) + 0.25 * np.abs(np.sin(2 * np.pi * x))
    multiwave_guess = np.clip(multiwave_guess, 0, 1)
    multiwave_guess = multiwave_guess * target_sum / np.sum(multiwave_guess)
    initial_guesses.append(multiwave_guess)

    best_h = initial_guesses[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for guess in initial_guesses[1:]:
        perturbed = guess + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)

        def objective(x):
            h = x * target_sum / np.sum(x)
            return compute_c5(h)

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 500})
