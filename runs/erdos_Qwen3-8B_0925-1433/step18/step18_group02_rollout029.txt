import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_c5(clip_normalize(x))

    base_h = np.array(initial_h_values, dtype=float)
    base_h = clip_normalize(base_h)

    # Generate initial guesses
    guesses = [base_h]
    A_vals = np.linspace(0.1, 0.3, 5)
    f_vals = np.linspace(0.2, 0.5, 5)
    phi_vals = np.linspace(0, 2 * np.pi, 5)
    for A in A_vals:
        for f in f_vals:
            for phi in phi_vals:
                h_mod = 0.5 + A * np.sin(2 * np.pi * f * np.arange(n_points) + phi)
                h_mod = clip_normalize(h_mod)
                guesses.append(h_mod)
    for _ in range(3):
        perm = np.random.permutation(n_points)
        guess = base_h[perm]
        guess = clip_normalize(guess)
        guesses.append(guess)

    best_h = base_h
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Phase 1: Global optimization
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(objective, guess, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:

        if res.success:
