import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(200) * (200 / 2) / 200
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    # Initial h
    h = np.clip(initial_h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    max_iterations = int(budget_s / (n_points * 2))  # Heuristic
    step_size = 0.01
    decay_rate = 100.0

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap_idx = np.argmax(corr[1:-1])
        worst_lag_idx_in_corr = 1 + max_overlap_idx
        # Compute lag m
        m = worst_lag_idx_in_corr - (n_points - 1)

        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + m < n_points:
                gradient[j] += dx * (1 - h[j + m])
            if 0 <= j - m < n_points:
                gradient[j] -= dx * h[j - m]

        # Step in the direction of the negative gradient
        h -= step_size * gradient

        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Update best_c5 if improved
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Occasional perturbation
        if iteration % 100 == 0:
            noise = 0.05 * np.random.randn(n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

    return (best_h, best_c5, n_points)
