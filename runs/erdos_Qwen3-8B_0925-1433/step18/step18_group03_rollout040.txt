from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Diverse initial guesses
    initial_guesses = [h.copy()]
    for _ in range(10):

            break

        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        initial_guesses.append(h_pert)

    for h_initial in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)

        # COBYLA for global exploration
        res = minimize(obj, h_initial, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-4)
        if res.success:

    # Local refinement with L-BFGS-B
