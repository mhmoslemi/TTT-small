import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Parameters for optimization
    alpha = 0.1  # learning rate
    decay_rate = 0.999  # learning rate decay
    max_iterations = 2000
    perturbation_prob = 0.2  # probability of adding noise

    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute correlation to find max lag
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c5 = np.max(corr)
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)

        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
            else:
                grad[i] = 1 * dx

        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Recompute correlation after update
        new_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(new_corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Stochastic perturbation
        if np.random.rand() < perturbation_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * target_sum / np.sum(h)

        # Decay learning rate
        alpha *= decay_rate

    return (np.clip(best_h, 0, 1), best_c5, n_points)
