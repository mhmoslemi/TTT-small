        for i in range(m):
            s = positions[i]
            e = s + l
            x_vals = np.linspace(0, 2, n_points)
            idx = np.argmin(np.abs(x_vals - s))  # Start index
            jdx = np.argmin(np.abs(x_vals - e))  # End index
            h[idx:jdx+1] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    # Alternate initial guess
    h_regular = generate_regular_guess()
    best_h = h_regular.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    # Optimization loop
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Perturbation with structured variance
        h_pert = best_h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
