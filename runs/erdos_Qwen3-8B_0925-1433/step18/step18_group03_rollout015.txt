def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_h = np.clip(best_h, 0, 1)

    # Generate initial guesses
    initial_guesses = [best_h.copy()]
    for _ in range(3):
        pert = np.random.normal(0, 0.15, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        initial_guesses.append(h_pert)

    start_time = time.time()

    for guess in initial_guesses:
        h = guess.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s * 0.8:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i + k >= 0 and i + k < n_points:

                if i - k >= 0:
