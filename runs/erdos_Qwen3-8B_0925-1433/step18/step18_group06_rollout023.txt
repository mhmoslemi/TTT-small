import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize candidates
    candidates = []
    i = np.arange(n_points)
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    candidates.append(h_initial)

    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * i / n_points)
    h_sinusoidal = np.clip(h_sinusoidal, 0, 1)
    h_sinusoidal = h_sinusoidal * target_sum / np.sum(h_sinusoidal)
    candidates.append(h_sinusoidal)

    h_triangular = (2 * (i % n_points) / (n_points - 1) - 1) ** 2 + 0.5
    h_triangular = np.clip(h_triangular, 0, 1)
    h_triangular = h_triangular * target_sum / np.sum(h_triangular)
    candidates.append(h_triangular)

    best_h = candidates[0]
    best_c5 = compute_c5(best_h)

    for h in candidates[1:]:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    learning_rate = 0.1
    momentum = 0.99
    decay_rate = 0.998
    velocity = np.zeros(n_points)
    start_time = time.time()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
