    learning_rate = 0.5
    max_iterations = 100
    time_start = time.time()

    for _ in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        corr = np.zeros(n_points)
        for k in range(n_points):
            shifted_h = np.roll(h, k)
            corr[k] = np.dot(h, 1 - shifted_h)
        corr *= dx
        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        j = (np.arange(n_points) + k_max) % n_points
        i = (np.arange(n_points) - k_max) % n_points
        grad = (1 - h[j]) - h[i]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= 0.95

    for _ in range(20):
        if time.time() - time_start > budget_s:
            break
        corr = np.zeros(n_points)
        for k in range(n_points):
            shifted_h = np.roll(best_h, k)
            corr[k] = np.dot(best_h, 1 - shifted_h)
        corr *= dx
        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        j = (np.arange(n_points) + k_max) % n_points
        i = (np.arange(n_points) - k_max) % n_points
        grad = (1 - best_h[j]) - best_h[i]
        best_h -= 0.1 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_h = best_h.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
