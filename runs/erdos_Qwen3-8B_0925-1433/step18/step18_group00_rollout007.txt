from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

        h_initial = h_initial * target_sum / np.sum(h_initial)
        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 300})
