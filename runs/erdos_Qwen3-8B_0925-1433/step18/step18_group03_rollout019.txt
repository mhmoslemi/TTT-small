import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 1e-12 else x * target_sum / target_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    # Generate initial population from initial_h_values with noise
    initial_population = [initial_h]
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.5:
            break
        perturb = np.random.normal(0, 0.2, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 1e-12 else new_h
        initial_population.append(new_h)

    # Run differential evolution
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=0.8,
            recombination=0.9,
            tol=1e-6,
            maxiter=100,
            polish=False,
            init=initial_population,
            seed=seed
        )
        best_h = result.x
    except:
        best_h = initial_h

    best_c5 = objective(best_h)

    # Final refinement with L-BFGS-B
    if time.time() - start_time < budget_s * 0.8:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 1e-12 else best_h

        def obj_l_bfgs(x):
            h = x * target_sum / np.sum(x) if np.sum(x) > 1e-12 else x
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds_l_bfgs = [(0.0, 1.0) for _ in range(n_points)]
        res_l_bfgs = minimize(
            obj_l_bfgs,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_l_bfgs,
            constraints=cons,
            tol=1e-8
        )
        best_h = res_l_bfgs.x
        best_c5 = objective(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 1e-12 else best_h
    return (best_h, best_c5, n_points)
