import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h_values):
        h = np.clip(h_values, 0., 1.)
        if np.isclose(np.sum(h), 0.):
            return float('inf')
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    initial_pop = [np.array(initial_h_values, dtype=float)]
    for _ in range(5):
        pert = np.random.normal(0, 0.1, n_points)
        perturbed = initial_h_values + pert
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_pop.append(perturbed)
    initial_pop = np.array(initial_pop)

    try:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=15,
            mutation=(0.5, 1.0), recombination=0.9, tol=1e-6,
            maxiter=100, polish=True, vectorize=True, init=initial_pop, seed=seed)
    except:
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, mutation=0.8, recombination=0.9, tol=1e-6, maxiter=100, polish=True, vectorize=True, seed=seed)

    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.8:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = result.x
        best_c5 = objective(best_h)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if res_l_bfgs.success:
        best_h = res_l_bfgs.x
        best_c5 = objective(best_h)

    best_h = np.clip(best_h, 0., 1.)
    return (best_h, best_c5, n_points)
