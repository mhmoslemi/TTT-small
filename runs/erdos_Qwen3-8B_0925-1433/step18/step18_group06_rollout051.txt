import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Generate additional patterns
    patterns = [
        np.clip(0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points), 0, 1),
        np.random.choice([0, 1], size=n_points),
        (np.arange(n_points) / n_points) + np.random.normal(0, 0.05, n_points)
    ]
    for p in patterns:
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        h = p.copy()
        best_c5 = np.min([best_c5, np.max(np.correlate(h, 1 - h, mode='full')) * dx])
        best_h = h.copy()

    # Optimization loop
    start_time = time.time()
    learning_rate = 0.05
    momentum = 0.8
    decay = 0.999
    vel = np.zeros(n_points)
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = (i + k) % n_points
            i_minus = (i - k) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]
        vel = momentum * vel - learning_rate * grad
        h += vel
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay

    # Random perturbation
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h + perturb, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
        current_c5_pert = np.max(corr_pert) * dx
        if current_c5_pert < best_c5:
            best_c5 = current_c5_pert
            best_h = h_pert.copy()
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
