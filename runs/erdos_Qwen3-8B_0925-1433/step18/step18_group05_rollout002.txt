        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = 0.5 * np.ones(n_points)
        else:
            h = h * target_sum / current_sum
        
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        learning_rate *= decay_rate
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * (2.0 / n_points)
    return (best_h, final_c5, n_points)
