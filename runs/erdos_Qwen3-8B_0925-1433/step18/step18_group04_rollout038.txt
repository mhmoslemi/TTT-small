import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

        return np.max(corr[1:-1]) * dx

    gbest_fitness = np.min(pbest_fitness)
    best_h = gbest.copy()

    start_time = time.time()
