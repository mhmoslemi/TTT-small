            tournament_c5 = [c5_values[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_c5)]
            parent2 = population[winner_idx]
            child = (parent1 + parent2) / 2.0
            corr = np.correlate(child, 1 - child, mode='full')
            max_lag = np.argmax(corr)
            if max_lag < len(child):
                i = np.random.randint(n_points)
                j = (i + max_lag) % n_points
                delta = 0.05 * mutation_rate
                child[i] -= delta
                child[j] += delta
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        mutation_rate *= 0.995

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
