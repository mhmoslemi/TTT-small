import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(200) * (200 / 2) / 200
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    step_size = 0.1
    iteration = 0
    decay_rate = 0.99

    while iteration < 1000 and time.time() - start_time < budget_s:
        iteration += 1
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr[1:-1]) + 1
        gradient = np.zeros(n_points)
        k = k_max
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                gradient[i] -= h[i - k] * dx
        h = h - step_size * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        step_size *= decay_rate

    final_c5 = compute_c5(best_h)
    best_c5 = min(final_c5, best_c5)
    return (best_h, best_c5, n_points)
