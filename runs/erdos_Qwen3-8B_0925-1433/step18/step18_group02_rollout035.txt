from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

        return compute_c5(h)

    base = np.array(initial_h_values, dtype=float)
    guesses = [base]
    freqs = [0.1, 0.2, 0.3]
    phases = [0, np.pi/2, np.pi]
    for f in freqs:

            guess = base + mod * 0.1
            guess = clip_normalize(guess)
            guesses.append(guess)

        guess = base[perm]

        pert = np.random.normal(0, 0.05, n_points)
        guess = base + pert

        guesses.append(guess)
    guesses = np.unique(guesses, axis=0)

    best_h = base.copy()

    start_time = time.time()
