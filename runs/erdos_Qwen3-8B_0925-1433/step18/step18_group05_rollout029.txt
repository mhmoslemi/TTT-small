import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    max_iterations = 500
    learning_rate = 0.1
    decay_rate = 0.99
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                gradient[i] = 1.0 - h[i + k]
        # Update h
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * target_sum / np.sum(h)
        # Evaluate new C5
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        # Decay learning rate
        learning_rate *= decay_rate

    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
