            perturbed = np.clip(base + pert, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            guesses.append(perturbed)
        alt = np.zeros(n_points)
        alt[::2] = 1.0
        alt = alt * target_sum / np.sum(alt)
        guesses.append(alt)
        gauss = np.zeros(n_points)
        gaps = np.random.normal(0, 0.2, 5)
        gaps = gaps / np.sum(gaps)
        for i in range(5):
            idx = int(gaps[i] * n_points)
            gauss[idx] = 1.0
        gauss = gauss * target_sum / np.sum(gauss)
        guesses.append(gauss)
        return guesses

    initial_guesses = get_initial_guesses()

    start_time = time.time()
    learning_rate = 0.05
    iterations = 0

    for guess in initial_guesses:
        h_initial = np.clip(guess, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        current_h = h_initial.copy()
        current_c5 = best_c5
        for _ in range(300):
            if time.time() - start_time > budget_s:
                break
            correlation = np.correlate(current_h, 1 - current_h, mode='full')
            max_idx = np.argmax(correlation)
            k = max_idx - (n_points - 1)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    gradient[i] = 1 - current_h[i + k]
            current_h = current_h - learning_rate * gradient
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            new_c5 = np.max(correlation) * dx
            if new_c5 < current_c5:
                current_c5 = new_c5
                current_h = current_h.copy()
            iterations += 1
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(correlation)
        k = max_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                gradient[i] = 1 - best_h[i + k]
        best_h = best_h - 0.01 * gradient
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        new_c5 = np.max(correlation) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
