from scipy.optimize import minimize, Bounds
import time

        return h

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return compute_c5(h)

        guesses = [base.copy()]

                    h_mod = clip_normalize(h_mod)
                    guesses.append(h_mod)
        for _ in range(5):
            perm = np.random.permutation(n_points)
            guess = clip_normalize(base[perm])
            guesses.append(guess)
        return np.unique(guesses, axis=0)
