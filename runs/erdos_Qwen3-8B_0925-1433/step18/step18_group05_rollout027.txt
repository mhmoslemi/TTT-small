        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus = j + k
            j_minus = j - k
            if 0 <= j_plus < n_points:
                grad[j] += (1 - h[j_plus])
            if 0 <= j_minus < n_points:
                grad[j] -= h[j_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = 0.5 * np.ones(n_points)
        else:
            h = h * (target_sum / sum_h)
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        decay_steps += 1
        if decay_steps % 50 == 0:
            learning_rate *= 0.98
        iteration += 1
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
