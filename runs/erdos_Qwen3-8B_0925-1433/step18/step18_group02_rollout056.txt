            perm = np.random.permutation(n_points)
            guess = base[perm]

            guesses.append(guess)
        return np.unique(guesses, axis=0)

    initial_guesses = generate_initial_guesses(np.array(initial_h_values, dtype=float))

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
