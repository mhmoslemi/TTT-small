    h_gaus = np.zeros(n_points)
    center = n_points // 2
    std = n_points // 4
    for i in range(n_points):
        if abs(i - center) < std:
            h_gaus[i] = 1.0
    h_gaus = h_gaus * target_sum / np.sum(h_gaus)
    initial_guesses.append(h_gaus)

    start_time = time.time()

    for h_initial in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)

        for _ in range(300):
            if time.time() - start_time > budget_s:
                break
            conv = np.correlate(h_initial, 1 - h_initial, mode='full')
            max_conv = np.max(conv)
            idx = np.argmax(conv)
            lag = idx - (n_points - 1)

            valid_indices = np.arange(n_points) + lag
            valid_mask = (valid_indices >= 0) & (valid_indices < n_points)
            grad = np.zeros(n_points)
            grad[valid_mask] = 1.0 - h_initial[valid_indices[valid_mask]]

            step_size = 0.05 / (np.sqrt(1 + _ / 100))
            h_initial -= step_size * grad
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * target_sum / np.sum(h_initial)

            current_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_initial.copy()
                best_c5 = current_c5

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        conv = np.correlate(best_h, 1 - best_h, mode='full')
        max_conv = np.max(conv)
        idx = np.argmax(conv)
        lag = idx - (n_points - 1)

        valid_indices = np.arange(n_points) + lag
        valid_mask = (valid_indices >= 0) & (valid_indices < n_points)
        grad = np.zeros(n_points)
        grad[valid_mask] = 1.0 - best_h[valid_indices[valid_mask]]

        step_size = 0.02 / (np.sqrt(1 + _ / 100))
        best_h -= step_size * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)

        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
