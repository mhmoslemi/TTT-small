import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    h = normalize_h(np.array(initial_h_values, dtype=float))
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    max_iterations = 100
    iterations = 0
    learning_rate = 0.15
    decay_rate = 0.95
    decay_interval = 50
    convergence_threshold = 1e-5

    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] += (1 - h[i + lag])
            if 0 <= i - lag < n_points:
                grad[i] -= h[i - lag]
