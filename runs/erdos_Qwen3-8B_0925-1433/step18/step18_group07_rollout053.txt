import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(5):

        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(population[0])
    population_size = 100
    elitism_rate = 0.05
    cross_blend_prob = 0.7
    time_start = time.time()

    while time.time() - time_start < budget_s:
        fitness = [evaluate_erdos_solution(h) for h in population]
        elite_size = int(elitism_rate * len(population))
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        new_pop = elite.copy()
        
        while len(new_pop) < population_size:
            p1 = np.random.choice(elite + [np.random.choice(population)])
            p2 = np.random.choice(elite + [np.random.choice(population)])
            if np.random.rand() < cross_blend_prob:
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * p1 + (1 - alpha) * p2
            else:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])

            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)

        population = new_pop
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
