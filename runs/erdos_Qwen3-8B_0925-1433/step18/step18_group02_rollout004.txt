import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = clip_normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    initial_guesses = []

    # Generate sinusoidal patterns
    for a in np.linspace(0.1, 0.3, 5):
        for f in np.linspace(0.2, 0.5, 5):
            for phi in np.linspace(0, 2*np.pi, 5):
                h_mod = 0.5 + a * np.sin(2 * np.pi * f * np.arange(n_points) / n_points + phi)
                h_mod = clip_normalize(h_mod)
                initial_guesses.append(h_mod)

    # Add permutations
    for _ in range(5):
        perm = np.random.permutation(n_points)
        h_perm = clip_normalize(initial_h[perm])
        initial_guesses.append(h_perm)

    # Add square wave
    square = np.zeros(n_points)
    square[:n_points//2] = 1
    square = clip_normalize(square)
    initial_guesses.append(square)

    # Remove duplicates
    initial_guesses = np.unique(initial_guesses, axis=0)

    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(
                compute_c5,
                guess,
                method='L-BFGS-B',
                bounds=[(0,1)]*n_points,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                tol=1e-6,
                options={'maxiter': 100}
            )
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5
        except:
            pass

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(
                compute_c5,
                best_h,
                method='Nelder-Mead',
                bounds=[(0,1)]*n_points,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                tol=1e-6,
                options={'maxiter': 100}
            )
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5
        except:
            pass

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        alpha = 0.05
        correction = alpha * np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
        h_corr = best_h + correction
        h_corr = clip_normalize(h_corr)
        current_c5 = compute_c5(h_corr)
        if current_c5 < best_c5:
            best_h, best_c5 = h_corr.copy(), current_c5
        else:
            break

    return (clip_normalize(best_h), best_c5, n_points)
