import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    pop_size = 50
    pop = [initial_h.copy()]
    for _ in range(pop_size-1):
        perturb = np.random.normal(0, 0.05, n_points)
        h = initial_h + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    pop = [np.array(p) for p in pop]
    best_h = pop[0].copy()
    best_c5 = np.inf
    time_start = time.time()
    elite_ratio = 0.2
    max_iter = int(budget_s * 1.2)
    mutation_std = 0.05
    decay = 0.998
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        mat = np.array([np.correlate(p, 1.0 - p, mode='full') for p in pop])
        c5 = np.max(mat, axis=1) * dx
        sort_idx = np.argsort(c5)
        fit = c5[sort_idx]
        pop_sorted = [pop[i] for i in sort_idx]
        best_c5 = fit[0]
        best_h = pop_sorted[0].copy()
        elites = pop_sorted[:int(elite_ratio * len(pop_sorted))]
        new_pop = elites.copy()
        remaining = len(pop_sorted) - len(elites)
        for _ in range(remaining):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(pop_sorted[:len(pop_sorted)//2], size=2, replace=False)
                child = np.random.rand() * p1 + (1 - np.random.rand()) * p2
            else:
                child = np.random.normal(0, mutation_std, n_points)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
        mutation_std *= decay
    return (np.clip(best_h, 0, 1), best_c5, n_points)
