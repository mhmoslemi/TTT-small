import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    current_h = np.clip(initial_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for iteration in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr[1:-1]
        worst_k = np.argmax(overlaps)
        if overlaps[worst_k] < best_c5:
            break
        k = worst_k
        positive_indices = np.arange(n_points) + k
        positive_part = np.zeros_like(current_h)
        positive_part[positive_indices < n_points] = 1 - current_h[positive_indices[positive_indices < n_points]]
        negative_indices = np.arange(n_points) - k
        negative_part = np.zeros_like(current_h)
        negative_part[negative_indices >= 0] = current_h[negative_indices[negative_indices >= 0]]
        gradient = dx * (positive_part - negative_part)
        step_size = 0.01 * np.exp(-iteration / 100)
        current_h -= step_size * gradient
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        if iteration % 100 == 0:
            noise = 0.05 * np.random.randn(n_points)
            current_h += noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
    return (best_h, best_c5, n_points)
