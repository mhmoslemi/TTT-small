import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population_size = 60
    population = []
    for _ in range(population_size // 3):
        indiv = np.array(initial_h_values, dtype=float)
        indiv = np.clip(indiv, 0, 1)
        indiv = indiv * target_sum / np.sum(indiv)
        population.append(indiv)
    for _ in range(population_size // 3):
        indiv = np.random.rand(n_points)
        indiv = indiv * target_sum / np.sum(indiv)
        population.append(indiv)
    for _ in range(population_size // 3):
        indiv = np.zeros(n_points)
        indiv[:n_points//2] = 1.0
        indiv = indiv * target_sum / np.sum(indiv)
        population.append(indiv)
    
    def evaluate(indiv):
        h = np.clip(indiv, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def tournament(pops, size=3):
        return min(random.sample(pops, size), key=evaluate)
    
    def crossover(p1, p2):
        return np.random.uniform(0, 1) * p1 + (1 - np.random.uniform(0, 1)) * p2
    
    def mutate(indiv):
        noise = np.random.normal(0, 0.05, n_points)
        mut = np.clip(indiv * (1 + noise), 0, 1)
        return mut * target_sum / np.sum(mut)
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fits = [evaluate(indiv) for indiv in population]
        breeders = np.argsort(fits)[:int(population_size * 0.2)]
        breeders = [population[i] for i in breeders]
        offspring = []
        for _ in range(len(breeders) // 2):
            p1, p2 = tournament(breeders), tournament(breeders)
            child = crossover(p1, p2)
            child = mutate(child)
            offspring.append(child)
        combined = population + offspring
        combined_fitness = [evaluate(indiv) for indiv in combined]
        combined_sorted = sorted(zip(combined_fitness, combined), key=lambda x: x[0])
        population = [indiv for (f, indiv) in combined_sorted[:population_size]]
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
