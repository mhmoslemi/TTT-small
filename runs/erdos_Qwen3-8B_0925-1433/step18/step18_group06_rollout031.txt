from scipy.optimize import minimize
import time

    best_candidate = None
    best_c5 = float('inf')
    for h in candidates:
        h_clipped = np.clip(h, 0, 1)
        h_norm = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_candidate = h_norm
    h = best_candidate.copy()
    best_h = h.copy()
    best_c5 = best_c5
    
    max_iterations = 150
    learning_rate = 0.1
    momentum = 0.99
    decay_rate = 0.998

        if time.time() - start_time > budget_s:
            break

        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points

            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        
        velocity = momentum * velocity - learning_rate * gradient
        h += velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
