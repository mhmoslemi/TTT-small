import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Structure-based initialization
    x = np.linspace(0, 2, n_points)
    h_base = 0.5 * (1 + np.sin(3 * np.pi * x))
    h_base = np.clip(h_base, 0, 1)
    h_base = h_base * target_sum / np.sum(h_base)
    h = h_base.copy()

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    start_time = time.time()

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def optimize(x0, max_iter=5):
        nonlocal best_c5, best_h
        for _ in range(max_iter):
            if time.time() - start_time >= budget_s:
                break
            res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = res.x.copy(), current_c5
                x0 = res.x
            else:
                break

    optimize(h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
