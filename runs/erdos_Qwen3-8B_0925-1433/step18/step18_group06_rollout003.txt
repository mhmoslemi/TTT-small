import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def optimize_pattern(start_h):
        h = start_h.copy()
        h = h * target_sum / np.sum(h)
        best_h = h.copy()
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        velocity = np.zeros(n_points)
        learning_rate = 0.1
        momentum = 0.9
        decay_rate = 0.999
        start_time = time.time()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (n_points - 1)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                i_plus_k = (i + k) % n_points
                i_minus_k = (i - k) % n_points
                gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
            velocity = momentum * velocity - learning_rate * gradient
            h += velocity
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            learning_rate *= decay_rate
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
            current_c5 = np.max(corr_perturbed) * dx
            if current_c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = current_c5
        return best_h, best_c5

    patterns = []
    patterns.append(np.array(initial_h_values, dtype=float))
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    sine = sine * target_sum / np.sum(sine)
    patterns.append(sine)
    binary = np.random.choice([0, 1], size=n_points)
    binary_sum = np.sum(binary)
    if binary_sum == 0 or binary_sum == n_points:
        binary = np.random.choice([0, 1], size=n_points)
        binary_sum = np.sum(binary)
    binary = binary * target_sum / binary_sum
    patterns.append(binary)
    uniform = np.random.rand(n_points)
    uniform = uniform * target_sum / np.sum(uniform)
    patterns.append(uniform)

    best_h, best_c5 = patterns[0].copy(), optimize_pattern(patterns[0])[1]

        h, c5 = optimize_pattern(p)
        if c5 < best_c5:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
