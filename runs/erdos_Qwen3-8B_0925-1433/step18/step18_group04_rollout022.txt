import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    n_particles = 50
    # Initialize with initial_h and random samples
    positions = np.zeros((n_particles, n_points))
    positions[0] = initial_h.copy()
    positions[0] = positions[0] * target_sum / np.sum(positions[0])
    for i in range(1, n_particles):
        positions[i] = np.random.rand(n_points)
        positions[i] = positions[i] * target_sum / np.sum(positions[i])
    velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
    pbest = positions.copy()
    pbest_fitness = np.array([compute_c5(p) for p in positions])
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    start_time = time.time()
    max_iter = 1000

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        omega = 0.9 - (iter / max_iter) * 0.5
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + 2 * r1 * (pbest[i] - positions[i]) + 2 * r2 * (gbest - positions[i])
            positions[i] = positions[i] + velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            s = np.sum(positions[i])
            if s == 0:
                positions[i] = np.random.rand(n_points)
                positions[i] = positions[i] * target_sum / np.sum(positions[i])
            else:
                positions[i] = positions[i] * target_sum / s
            curr_fitness = compute_c5(positions[i])
            if curr_fitness < pbest_fitness[i]:
                pbest[i] = positions[i].copy()
                pbest_fitness[i] = curr_fitness
        current_gbest_idx = np.argmin(pbest_fitness)
        if pbest_fitness[current_gbest_idx] < gbest_fitness:
            gbest_fitness = pbest_fitness[current_gbest_idx]
            gbest = pbest[current_gbest_idx].copy()

    best_h = gbest
    best_c5 = gbest_fitness
    return (best_h, best_c5, n_points)
