    initial_guesses.append(h_sine)

    # Multi-wavelet pattern
    h_multi = 0.5 + 0.3 * np.sin(np.pi * x) + 0.2 * np.sign(np.sin(2 * np.pi * x))

    # Function to compute C5
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

        if res.success:
