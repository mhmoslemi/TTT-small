import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = clip_normalize(x)
        return compute_c5(h)

    def generate_initial_guesses(base):
        guesses = [base]
        for _ in range(20):
            pert = np.random.normal(0, 0.02, n_points)
            guess = clip_normalize(base + pert)
            guesses.append(guess)
        for _ in range(15):
            perm = np.random.permutation(n_points)
            guess = clip_normalize(base[perm])
            guesses.append(guess)
        return np.unique(guesses, axis=0)

    initial_guesses = generate_initial_guesses(np.array(initial_h_values, dtype=float))

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(objective, guess, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8, maxiter=100)
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5
        except:
            pass

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = clip_normalize(best_h)
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8, maxiter=100)
        if res.success:

            current_c5 = compute_c5(current_h)
