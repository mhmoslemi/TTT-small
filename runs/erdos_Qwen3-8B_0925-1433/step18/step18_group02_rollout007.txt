import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for _ in range(100):
        if time.time() > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        i_max = np.argmax(corr)
        k = i_max - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            m = (i - k) % n_points
            grad[i] = (1 - h[j]) - h[m]
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
