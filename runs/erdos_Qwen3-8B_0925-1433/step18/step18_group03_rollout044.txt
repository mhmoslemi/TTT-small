def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    structured_guesses = []
    # Base patterns
    for i in range(4):
        guess = np.zeros(n_points)
        for j in range(n_points):
            if j % (5 + i) == 0:
                guess[j] = 1.0
        structured_guesses.append(guess)
    # Alternating pattern
    structured_guesses.append(np.zeros(n_points))
    structured_guesses[-1][::2] = 1.0

    # Add random perturbations
    for guess in structured_guesses:

        guess_pert = np.clip(guess + pert, 0, 1)
        guess_pert = guess_pert * target_sum / np.sum(guess_pert)
        structured_guesses.append(guess_pert)

    start_time = time.time()
