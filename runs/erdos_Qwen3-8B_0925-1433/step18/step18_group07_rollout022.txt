from scipy.optimize import minimize
import time

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def gradient(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = (1 - h_vec[i_plus_k]) - h_vec[i_minus_k]
        return grad

    n_iter = 300
    learning_rate = 0.2
    decay = 0.997

    vel = np.zeros(n_points)
    start_time = time.time()

    for _ in range(n_iter):

        current_c5 = objective(h)
        if current_c5 < best_c5:
