    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    # Generate diverse initial guesses with higher perturbation
    initial_guesses = [h.copy()]

        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        initial_guesses.append(h_pert)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Initial optimization with COBYLA

        res = minimize(obj, h_initial, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
