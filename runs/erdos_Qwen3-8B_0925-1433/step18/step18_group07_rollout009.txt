import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    def objective(h_vec):
        h = h_vec.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def grad(h_vec):
        h = h_vec.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        return grad * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    start_time = time.time()
    result = minimize(objective, h, method='L-BFGS-B', jac=grad, bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 120})
    best_h = result.x
    best_c5 = objective(best_h) if result.success else np.inf

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.015, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = perturbed
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
