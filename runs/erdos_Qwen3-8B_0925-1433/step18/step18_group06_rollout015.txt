import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def make_initial(name):
        h = np.array(initial_h_values, dtype=float) if name == "initial" else np.zeros(n_points)
        if name == "sinusoidal":
            h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        elif name == "triangular":
            h = ((2 * (np.arange(n_points) % n_points) / (n_points - 1) - 1) ** 2) + 0.5
        return clip_normalize(h)
    
    candidates = [make_initial("initial"), make_initial("sinusoidal"), make_initial("triangular")]
    best_h = min(candidates, key=evaluate)
    best_c5 = evaluate(best_h)
    
    learning_rate = 0.1
    momentum = 0.99
    decay_rate = 0.998
    max_iterations = 300
    velocity = np.zeros(n_points)
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            grad[i] = (1 - best_h[i_plus_k]) - best_h[i_minus_k]
        
        velocity = momentum * velocity - learning_rate * grad
        best_h += velocity
        best_h = clip_normalize(best_h)
        current_c5 = evaluate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.07, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        current_c5 = evaluate(perturbed)
        if current_c5 < best_c5:
            best_h = perturbed
            best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
