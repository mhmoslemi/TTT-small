import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_guesses = []
    x = np.linspace(0, 2, n_points)
    
    # Sine-based
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = sine_guess * target_sum / np.sum(sine_guess)
    initial_guesses.append(sine_guess)
    
    # Binary pattern
    binary_guess = np.zeros(n_points)
    binary_guess[::2] = 1.0
    binary_guess = binary_guess * target_sum / np.sum(binary_guess)
    initial_guesses.append(binary_guess)
    
    # Block pattern
    block_guess = np.zeros(n_points)
    for i in range(n_points):
        if (i // 5) % 2 == 0:
            block_guess[i] = 1.0
    block_guess = block_guess * target_sum / np.sum(block_guess)
    initial_guesses.append(block_guess)
    
    # Multiwavelet
    multiwavelet_guess = 0.5 * np.sin(np.pi * x) + 0.5 * np.abs(np.sin(np.pi * x))
    multiwavelet_guess = np.clip(multiwavelet_guess, 0, 1)
    multiwavelet_guess = multiwavelet_guess * target_sum / np.sum(multiwavelet_guess)
    initial_guesses.append(multiwavelet_guess)
    
    # Random binary
    random_binary = np.random.binomial(1, 0.5, n_points)
    random_binary = random_binary * target_sum / np.sum(random_binary)
    initial_guesses.append(random_binary)
    
    # Use initial_h_values as a default
    initial_guesses.append(np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values))

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for i, guess in enumerate(initial_guesses):
        # Perturb and optimize
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(guess + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        
        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 500})
        if res.success:
            current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

        if time.time() - start_time > budget_s:
            break

    # Refinement phase
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        if best_h is None:
            continue
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        if res.success:
            current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
