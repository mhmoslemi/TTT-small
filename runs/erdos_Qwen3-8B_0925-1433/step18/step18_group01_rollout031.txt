import time

    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    time_used = 0

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    for _ in range(10):

        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
