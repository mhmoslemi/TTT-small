import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, top_lag, mutation_rate):
        n_points = len(h)
        delta = mutation_rate * 0.01
        h_mut = h.copy()
        for i in range(n_points):
            j = (i + top_lag) % n_points
            h_mut[i] -= delta
            h_mut[j] += delta
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        return h_mut

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population = [initial_h.copy() for _ in range(50)]
    for h in population:
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    max_iters = 200
    mutation_rate = 0.15
    elite_size = 5
    no_improvement = 0

    for _ in range(max_iters):
        if time.time() - start_time > budget_s:
            break
        c5_values = [evaluate(h) for h in population]
        elite_indices = np.argsort(c5_values)[:elite_size]
        current_best = population[elite_indices[0]]
        current_c5 = c5_values[elite_indices[0]]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > 50:
            mutation_rate *= 1.1
            no_improvement = 0
        new_population = [population[i] for i in elite_indices]
        for _ in range(len(population) - elite_size):
            a = np.random.choice(range(len(population)))
            b = np.random.choice(range(len(population)))
            parent1 = population[a]
            parent2 = population[b]
            child = (parent1 + parent2) / 2.0
            corr = np.correlate(child, 1 - child, mode='full')
            top_lag = np.argmax(corr)
            child = mutate(child, top_lag, mutation_rate)
            new_population.append(child)
        population = new_population
        mutation_rate *= 0.995

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
