import time

def run(seed=42, budget_s=600):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 60
    population = []
    initial = np.array(initial_h_values, dtype=float)

            h = 0.5 * (1 + np.random.rand(n_points))
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
        h = repair(h)
        population.append(h)

    best_h = population[0]
    best_c5 = evaluate(best_h)
    time_used = 0

    for _ in range(500):
        if time_used >= budget_s:
            break

        fitness = [evaluate(h) for h in population]

        parents = np.array(population)[np.argsort(fitness)[:int(pop_size*0.2)]]

        offspring = []

            inds = np.random.choice(len(parents), 3)
            parent1, parent2 = parents[inds[0]], parents[inds[1]]
