        h = h * target_sum / np.sum(h)
        population.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    population = np.array(population)
    velocities = np.array(velocities)

    pbest = population.copy()
    pbest_fitness = [compute_c5(h) for h in population]
    gbest = population[np.argmin(pbest_fitness)]
    best_c5 = np.min(pbest_fitness)
    start_time = time.time()
    omega = 0.9
    while time.time() - start_time < budget_s:
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + 2 * r1 * (pbest[i] - population[i]) + 2 * r2 * (gbest - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        fitness = [compute_c5(h) for h in population]
        for i in range(len(population)):
            if fitness[i] < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = fitness[i]
        current_best = np.min(pbest_fitness)
        if current_best < best_c5:
            best_c5 = current_best
            gbest = pbest[np.argmin(pbest_fitness)]
        omega -= 0.005
    best_h = gbest
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
