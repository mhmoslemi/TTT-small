        for j in range(n_points):
            if j + k < n_points:
                gradient[j] += (1 - h[j + k])
            if j - k >= 0:
                gradient[j] -= h[j - k]
        
        # Update h
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate new C5
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(new_corr) * (2.0 / n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        # Update learning rate
        learning_rate *= decay_rate
        h = h_new.copy()
    
    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
