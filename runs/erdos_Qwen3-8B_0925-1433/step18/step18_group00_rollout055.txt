import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)

    def evaluate(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    def optimize(initial):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = minimize(evaluate, initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        return res.x if res.success else initial

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        x_vals = np.linspace(0, 2, n_points)
        phi = np.random.rand() * 2 * np.pi

        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        current_h = optimize(h_initial)
        current_c5 = evaluate(current_h)
        if current_c5 < best_c5:
            best_h, best_c5 = current_h.copy(), current_c5

    for _ in range(3):

        perturbed = np.clip(best_h + perturbation, 0, 1)

        current_h = optimize(perturbed)
        current_c5 = evaluate(current_h)
        if current_c5 < best_c5:
