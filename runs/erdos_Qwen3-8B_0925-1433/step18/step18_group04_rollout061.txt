import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.full(45, 0.5)

    target_sum = n_points / 2.0

        return np.max(corr[1:-1]) * dx

    current_h = np.clip(initial_h, 0.0, 1.0)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)
    best_c5_prev = best_c5
    start_time = time.time()

    max_iterations = 500
    perturb_interval = 100
    step_size_decay = 0.999

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr[1:-1] * dx
        worst_idx = np.argmax(overlaps)
        worst_k = worst_idx + 1

        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + worst_k < n_points:
                grad[i] += dx * (1 - current_h[i + worst_k])
            if i - worst_k >= 0:
                grad[i] -= dx * current_h[i - worst_k]

        step_size = 0.01 * np.exp(-iteration / 100)
        current_h -= step_size * grad

        current_h = np.clip(current_h, 0.0, 1.0)
        current_h = current_h * target_sum / np.sum(current_h)
