import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    max_iterations = 300
    learning_rate = 0.3
    decay_factor = 0.99
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k < n_points:
                grad[j] += 1 - h[j + k]
            if j - k >= 0:
                grad[j] -= h[j - k]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)

        else:

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:

        learning_rate *= decay_factor

    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
