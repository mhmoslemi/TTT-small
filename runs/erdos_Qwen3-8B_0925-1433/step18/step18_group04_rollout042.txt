import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step_size = 0.02
    max_iterations = 1000
    no_improvement = 0
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr[1:-1])
        j = max_lag_idx + 1
        grad = np.zeros(n_points)
        for i in range(n_points - j):
            grad[i] += dx * (1 - h[i + j])
            grad[i + j] -= dx * h[i]
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 100:
                break
    return (best_h, best_c5, n_points)
