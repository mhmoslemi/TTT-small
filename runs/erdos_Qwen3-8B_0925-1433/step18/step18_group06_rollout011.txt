import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial patterns
    def gen_patterns():
        h0 = np.clip(initial_h_values, 0, 1).copy()
        h0 = h0 * target_sum / np.sum(h0)
        yield h0
        
        # Sinusoidal
        h_sin = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h_sin = np.clip(h_sin, 0, 1)
        h_sin = h_sin * target_sum / np.sum(h_sin)
        yield h_sin
        
        # Randomized stepping
        h_rand = 0.5 + 0.5 * np.random.choice([0, 1], size=n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        yield h_rand
        
        # Uniform with noise
        h_uni = np.arange(n_points) / n_points
        h_uni += np.random.normal(0, 0.05, n_points)
        h_uni = np.clip(h_uni, 0, 1)
        h_uni = h_uni * target_sum / np.sum(h_uni)
        yield h_uni

    best_h = None
    best_c5 = np.inf

    for h in gen_patterns():
        h = h.copy()
        velocity = np.zeros(n_points)
        learning_rate = 0.1
        momentum = 0.9
        for _ in range(150):
            if time.time() > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)
            
            # Compute gradient for current lag k
            gradient = np.zeros(n_points)
            for i in range(n_points):
                i_plus_k = i + k
                i_minus_k = i - k

                    gradient[i] = -h[i_minus_k]
                else:
                    gradient[i] = 0
            
            velocity = momentum * velocity - learning_rate * gradient
            h += velocity
            h = np.clip(h, 0, 1)

            if current_c5 < best_c5:
