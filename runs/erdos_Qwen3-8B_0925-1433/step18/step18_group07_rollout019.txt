import numpy as np
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    population = np.array(population)

    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(population[0])

    mutation_strength = 0.05
    mutation_decay = 0.995
    mutation_rate = 0.8
    population_size = 50
    elite_ratio = 0.2

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate_erdos_solution(h) for h in population])
        sorted_indices = np.argsort(fitness)
        sorted_population = population[sorted_indices]

        elites = sorted_population[:elite_count]

        for _ in range(population_size - elite_count):

            child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
