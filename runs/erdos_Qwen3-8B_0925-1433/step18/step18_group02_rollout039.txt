import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = 0.0

    def compute_cross_corrs(h):
        n = len(h)
        cross_corrs = []
        for k in range(n):
            corr = 0.0
            for i in range(n):
                j = (i + k) % n
                corr += h[i] * (1 - h[j])
            cross_corrs.append(corr * dx)
        return cross_corrs

    def compute_gradient(h, k):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            j_plus = (i + k) % n
            j_minus = (i - k) % n
            grad[i] = (1 - h[j_plus]) - h[j_minus]
        return grad

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        cross_corrs = compute_cross_corrs(h)
        max_corr = max(cross_corrs)
        max_k = cross_corrs.index(max_corr)
        grad = compute_gradient(h, max_k)
        h = h - 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
