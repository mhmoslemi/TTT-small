    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = [h.copy()]
    for _ in range(3):
        base = np.zeros(n_points)
        step = n_points // 5
        for i in range(5):
            base[i * step:(i + 1) * step] = 1.0
        pert = np.random.normal(0, 0.15, n_points)
        perturbed = np.clip(base + pert, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)
        base = np.zeros(n_points)
        base[::2] = 1.0
        pert = np.random.normal(0, 0.15, n_points)
        perturbed = np.clip(base + pert, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)

    start_time = time.time()
    for h_initial in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        h = h_initial.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            j = np.argmax(corr)
            k = j - (n_points - 1)
            indices = np.arange(n_points) + k
            valid = (indices >= 0) & (indices < n_points)
            grad = np.where(valid, 1 - h[indices], 0)
            alpha = 0.05 / (1 + _)
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new.copy()

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    best_c5 = min(final_c5, best_c5)

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        j = np.argmax(corr)
        k = j - (n_points - 1)
        indices = np.arange(n_points) + k
        valid = (indices >= 0) & (indices < n_points)
        grad = np.where(valid, 1 - best_h[indices], 0)
        alpha = 0.01
        h_new = best_h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        if compute_c5(h_new) < best_c5:
            best_h = h_new.copy()
            best_c5 = compute_c5(h_new)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
