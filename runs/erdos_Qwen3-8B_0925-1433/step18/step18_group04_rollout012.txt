def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = np.ones(45) * (45 / 2) / 45
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    # Initial steps
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    iteration = 0
    max_iterations = 1000
    step_size = 0.01
    perturbation_freq = 100
    decay_rate = 0.999

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr[1:-1])
        k = m - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                gradient[i] += dx * (1 - h[i + k])
            if i - k >= 0 and i - k < n_points:
                gradient[i] -= dx * h[i - k]
        # Update h
        h -= step_size * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Check for improvement
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()

            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        # Decay step size
        step_size *= decay_rate
    return (best_h, best_c5, n_points)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = np.ones(45) * 45 / 2 / 45
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    # Initial clipping and rescaling
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    iteration = 0
    max_iterations = 1000
    step_size = 0.05
    perturb_freq = 100
    decay_rate = 0.999

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr[1:-1])
        k = m - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                gradient[i] += dx * (1 - h[i + k])
            if i - k >= 0 and i - k < n_points:
                gradient[i] -= dx * h[i - k]

        h -= step_size * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        if iteration % perturb_freq == 0:
            noise = 0.05 * np.random.randn(n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        step_size *= decay_rate

    return (best_h, best_c5, n_points)
