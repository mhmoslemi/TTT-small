        lag_values = conv[n_points -1 : 2*n_points -1]
        max_idx = np.argmax(lag_values)
        grad = compute_grad(h, max_idx)
        alpha = 0.001 * (0.995 ** step)
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
