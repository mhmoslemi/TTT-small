import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 60
    population = []
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population.append(h_initial)
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points) * 0.5 + 0.5
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)
    best_h = population[0]
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        sorted_indices = np.argsort(fitness)
        elite_size = int(len(population) * 0.2)
        breeders = sorted_indices[:elite_size]
        offspring = []
        for _ in range(len(population) - elite_size):
            tournament = np.random.choice(breeders, 3, replace=False)
            parents = [population[i] for i in tournament]
            alpha = np.random.rand()
            child = (1 - alpha) * parents[0] + alpha * parents[1]
            mutate_mask = np.random.rand(n_points) < 0.2
            sigma = 0.05
            child[mutate_mask] *= (1 + np.random.normal(0, sigma))
            child = np.clip(child, 0, 1)
