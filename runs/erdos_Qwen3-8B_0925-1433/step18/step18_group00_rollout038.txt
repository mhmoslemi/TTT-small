from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Base configuration

    h = h * target_sum / np.sum(h)

    best_c5 = min(best_c5, 0.381)

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    start_time = time.time()

    for _ in range(5):

            break

        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_initial + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        res = minimize(objective, h_pert, method='SLSQP', bounds=bounds, constraints=cons,
                       tol=1e-6, options={'maxiter': 500, 'ftol': 1e-8})

    # Final refinement
