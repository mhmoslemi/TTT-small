import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        return np.max(corr) * dx

    def mutate(h, lag, mutation_rate):
        n = len(h)
        delta = mutation_rate * 0.05
        h_mut = h.copy()
        for i in range(n):
            j = (i + lag) % n
            h_mut[i] -= delta
            h_mut[j] += delta
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        return h_mut

    def evolve(population, mutation_rate):
        new_pop = []
        for _ in range(len(population)):
            p1, p2 = np.random.choice(population, size=2, replace=False)
            child = (p1 + p2) / 2.0
            corr = np.correlate(child, 1 - child, mode='full')
            max_lag = np.argmax(corr)
            child = mutate(child, max_lag, mutation_rate)
            new_pop.append(child)
        return new_pop

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    best_h = h_initial
