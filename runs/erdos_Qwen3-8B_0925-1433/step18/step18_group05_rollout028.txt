    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    learning_rate = 0.1
    max_iterations = 600
    no_improvement_threshold = 100
    no_improvement_counter = 0

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        if no_improvement_counter > no_improvement_threshold:
            learning_rate *= 0.5
            no_improvement_counter = 0

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        gradient = np.zeros_like(h)
        for j in range(n_points):
            if j + k < n_points:
                gradient[j] += 1 - h[j + k]
            if j - k >= 0:
                gradient[j] -= h[j - k]

        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
