import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def obj(h):
        h = clip_rescale(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_pop = [np.array(initial_h_values, dtype=float) for _ in range(15)]
    for p in initial_pop:
        p = clip_rescale(p)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(
        obj, bounds, strategy='best1bin', popsize=15, mutation=0.8, recombination=0.9,
        tol=1e-6, maxiter=100, polish=False, seed=seed, vectorized=True,
        args=(), callback=None, constraints=None
    )

    best_h = result.x
    best_c5 = obj(best_h)

    for _ in range(3):
        if time.time() - start_time > budget_s * 0.8:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8)
        if res.success:

            best_c5 = obj(best_h)

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
