            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5

    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    return (h_final, best_c5, n_points)
