import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)
    
    # Initialize population
    n_particles = 50
    population = []
    for _ in range(n_particles):
        pert = np.random.normal(0, 0.1, n_points)
        h = initial_h + pert
        h = rescale(h)
        population.append(h)
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    temperature = 1.0
    cooling_rate = 0.995
    
    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        temperature *= cooling_rate
        for i in range(n_particles):
            current_h = population[i]
            perturbation = np.random.normal(0, temperature * 0.15, n_points)
            new_h = current_h + perturbation
            new_h = rescale(new_h)
            current_c5 = compute_c5(current_h)
            new_c5 = compute_c5(new_h)
            if new_c5 < current_c5:
