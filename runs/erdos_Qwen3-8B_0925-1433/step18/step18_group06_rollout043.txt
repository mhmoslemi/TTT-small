import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
        initial_patterns = [np.array(initial_h_values, dtype=float)]
    except:
        n_points = 45
        initial_patterns = []
    # Generate additional starting patterns
    initial_patterns.append(0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points))
    initial_patterns.append(0.5 + 0.5 * np.random.choice([0,1], size=n_points))
    initial_patterns.append((np.arange(n_points) / n_points) + np.random.normal(0, 0.05, n_points))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for pattern in initial_patterns:
        h = np.array(pattern, dtype=float)
        h = np.clip(h, 0, 1)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
        velocity = np.zeros(n_points)
        learning_rate = 0.1
        momentum = 0.9
        max_iterations = 150
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (n_points - 1)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                i_plus_k = i + k
                i_minus_k = i - k
                if i_plus_k < n_points and i_minus_k >= 0:
                    gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
                else:
                    gradient[i] = 0.0
            velocity = momentum * velocity - learning_rate * gradient
            h += velocity
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            learning_rate *= 0.995
        # Perturb after optimization
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr_perturbed) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5
    
    if best_h is None:
        best_h = np.zeros(n_points)
        best_c5 = 1.0
    
    # Ensure constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    return (np.clip(best_h, 0, 1), final_c5, n_points)
