import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate(h)

    learning_rate = 0.1
    decay_rate = 0.98
    decay_steps = 100
    max_iterations = 1000
    iteration = 0
    start_time = time.time()

    while iteration < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)

        gradient = np.zeros_like(h)
        for j in range(n_points):
            if j + k < n_points:
                gradient[j] += (1 - h[j + k])
            if j - k >= 0:
                gradient[j] -= h[j - k]

        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            h = h * target_sum / sum_h

        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        iteration += 1
        if iteration % decay_steps == 0:
            learning_rate *= decay_rate

    return (best_h, best_c5, n_points)
