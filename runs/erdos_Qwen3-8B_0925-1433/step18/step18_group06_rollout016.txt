import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    # Apply periodic initialization
    h = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_points) / n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    lr = 0.15
    momentum = 0.95
    decay = 0.999
    max_iters = 200
    start = time.time()
    vel = np.zeros(n_points)
    for _ in range(max_iters):
        if time.time() - start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k) % n_points
            i_minus = (i - k) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]
        vel = momentum * vel - lr * grad
        h += vel
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current = np.max(corr) * dx
        if current < best_c5:
            best_h = h.copy()
            best_c5 = current
        lr *= decay

    # Perturbation phase
    for _ in range(10):
        if time.time() - start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k) % n_points
            i_minus = (i - k) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]
        h += 0.02 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        perturb = np.random.normal(0, 0.05, n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current = np.max(corr) * dx
