def run(seed=42, budget_s=600):

        h = h * target_sum / np.sum(h)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    base_h = np.array(initial_h_values, dtype=float)
    base_h = clip_normalize(base_h)

    best_c5 = compute_c5(base_h)
    
    start_time = time.time()
    learning_rate = 0.05
    decay_rate = 0.995
    max_iterations = 100
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        max_idx = np.argmax(corr)

        for i in range(n_points):
            j = (i + lag) % n_points
