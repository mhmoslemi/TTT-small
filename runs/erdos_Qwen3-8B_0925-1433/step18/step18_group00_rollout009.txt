    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * target_sum / np.sum(h_sine)
    initial_guesses.append(h_sine)

    # 2. Binary pattern
    binary = np.random.choice([0, 1], size=n_points)
    sum_binary = np.sum(binary)
    delta = target_sum - sum_binary
    zeros = np.where(binary == 0)[0]
    ones = np.where(binary == 1)[0]
    if delta > 0:
        replace_count = int(delta / 0.5)
        if len(zeros) >= replace_count:
            binary[zeros[:replace_count]] = 0.5
        else:
            binary[zeros] = 0.5
            delta -= 0.5 * len(zeros)
            replace_count = int(delta / 0.5)
            if len(ones) >= replace_count:
                binary[ones[:replace_count]] = 0.5
    elif delta < 0:
        replace_count = int(-delta / 0.5)
        if len(ones) >= replace_count:
            binary[ones[:replace_count]] = 0.5
        else:
            binary[ones] = 0.5
            delta += 0.5 * len(ones)
            replace_count = int(-delta / 0.5)
            if len(zeros) >= replace_count:
                binary[zeros[:replace_count]] = 0.5
    h_binary = np.clip(binary, 0, 1)
    h_binary = h_binary * target_sum / np.sum(h_binary)
    initial_guesses.append(h_binary)

    # 3. Block pattern
    block_size = 4
    num_blocks = n_points // block_size
    remainder = n_points % block_size
    blocks = []
    for i in range(num_blocks):
        if i % 2 == 0:
            blocks.append(np.ones(block_size))
        else:
            blocks.append(np.zeros(block_size))
    block_array = np.concatenate(blocks)
    if remainder > 0:
        if num_blocks % 2 == 0:
            block_array = np.concatenate([block_array, np.zeros(remainder)])
        else:
            block_array = np.concatenate([block_array, np.ones(remainder)])
    sum_block = np.sum(block_array)
    delta = target_sum - sum_block
    if delta > 0:
        zeros = np.where(block_array == 0)[0]
        replace_count = int(delta / 0.5)
        if len(zeros) >= replace_count:
            block_array[zeros[:replace_count]] = 0.5
        else:
            block_array[zeros] = 0.5
            delta -= 0.5 * len(zeros)
            replace_count = int(delta / 0.5)
            if len(np.where(block_array == 1)[0]) >= replace_count:
                block_array[np.where(block_array == 1)[0][:replace_count]] = 0.5
    elif delta < 0:
        replace_count = int(-delta / 0.5)
        ones = np.where(block_array == 1)[0]
        if len(ones) >= replace_count:
            block_array[ones[:replace_count]] = 0.5
        else:
            block_array[ones] = 0.5
            delta += 0.5 * len(ones)
            replace_count = int(-delta / 0.5)
            zeros = np.where(block_array == 0)[0]
            if len(zeros) >= replace_count:
                block_array[zeros[:replace_count]] = 0.5
    h_block = np.clip(block_array, 0, 1)
    h_block = h_block * target_sum / np.sum(h_block)
    initial_guesses.append(h_block)

    # 4. Multiwavelet
    h_square = np.where(np.floor(x * 5) % 2 == 0, 1, 0)
    h_square = np.clip(h_square, 0, 1)
    h_square = h_square * target_sum / np.sum(h_square)
    h_multi = 0.5 * h_sine + 0.5 * h_square
    h_multi = np.clip(h_multi, 0, 1)
    h_multi = h_multi * target_sum / np.sum(h_multi)
    initial_guesses.append(h_multi)

    # Optimization
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(guess + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 300})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
