    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.1
    decay_rate = 0.995
    n_iterations = 200
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0

    start_time = time.time()

    for t in range(n_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        # Compute gradient for lag k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        # Adam update
        t += 1

        # Update best solution
        current_c5 = np.max(corr) * dx
