            # Clamp and rescale
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        # Combine and sort
        combined = pop + new_pop
        fitness_combined = np.array([compute_c5(h) for h in combined])
        indices = np.argsort(fitness_combined)
        new_pop = [combined[i] for i in indices[:population_size]]
        pop = new_pop
        # Update best
        current_c5 = compute_c5(pop[0])
        if current_c5 < best_c5:
            best_h = pop[0].copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
