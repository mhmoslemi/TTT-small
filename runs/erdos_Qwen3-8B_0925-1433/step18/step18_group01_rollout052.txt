import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    result = basinhopping(obj, h, niter=50, T=1.0, stepsize=0.1, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0, 1) for _ in range(n_points)]})
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        return (np.clip(best_h, 0, 1), best_c5, n_points)

    best_h = result.x
    best_c5 = obj(best_h)

    for _ in range(3):
        if elapsed >= budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], tol=1e-6, options={'maxiter': 50})
        if res.success:
            current_h = res.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        elapsed += time.time() - start_time

    return (np.clip(best_h, 0, 1), best_c5, n_points)
