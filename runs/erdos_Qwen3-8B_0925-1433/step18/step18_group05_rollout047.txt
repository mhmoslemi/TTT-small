import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    best_c5 = min(best_c5, 0.381)
    time_start = time.time()
    learning_rate = 0.15
    decay_rate = 0.99
    iterations = 0
    while iterations < 1000 and time.time() - time_start < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)
        gradient = np.zeros_like(h)
        for j in range(n_points):
            if j + k < n_points:
                gradient[j] += 1 - h[j + k]
            if j - k >= 0:
                gradient[j] -= h[j - k]
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
        if iterations % 50 == 0:
            if (best_c5 - current_c5) < 1e-6:
                learning_rate = 0.15
        iterations += 1
    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
