    best_c5 = min(best_c5, 0.381)

    max_iterations = 300
    start_time = time.time()
    decay_rate = 0.998
    learning_rate = 0.1
    momentum = 0.85
    velocity = np.zeros(n_points)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-5:]
        top_ks = [idx - (n_points - 1) for idx in top_k_indices]
        gradient = np.zeros(n_points)
        for k in top_ks:
            for i in range(n_points):
                i_plus_k = (i + k) % n_points
                i_minus_k = (i - k) % n_points
                gradient[i] += (1 - h[i_plus_k]) - h[i_minus_k]
        gradient /= len(top_ks)
        velocity = momentum * velocity - learning_rate * gradient
        h += velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    for _ in range(10):
