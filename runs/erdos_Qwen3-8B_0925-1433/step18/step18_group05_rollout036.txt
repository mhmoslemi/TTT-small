import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate(h)

    start_time = time.time()
    learning_rate = 0.5
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k])
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k]

        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)

        new_c5 = evaluate(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new

        learning_rate *= 0.998

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
