import time

        h = h * target_sum / np.sum(h)
        return h

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        h = clip_normalize(x)
        return compute_c5(h)

            guesses.append(clip_normalize(base[perm]))
        return np.unique(guesses, axis=0)
