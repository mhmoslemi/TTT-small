from scipy.optimize import minimize
import time

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    start_time = time.time()

    def optimize(start_h):
        res = minimize(obj, start_h, method='SLSQP', bounds=bounds, constraints=cons,
                       tol=1e-8, options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8})
        if res.success:

        return start_h
