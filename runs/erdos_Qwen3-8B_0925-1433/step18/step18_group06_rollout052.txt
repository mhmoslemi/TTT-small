import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    population = [h_initial]
    for _ in range(4):
        p = np.random.rand(n_points)
        p = p * target_sum / np.sum(p)
        population.append(p)
        p = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
        p = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) * 0.5 + 0.5
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
        p = h_initial + np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
        p = (np.arange(n_points) % 2) * 0.5 + 0.5
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        fitness = np.array(fitness)
        inds = np.argsort(fitness)
        parents = population[inds[:20]]
        new_population = []

        for _ in range(len(population) // 2):
            if time.time() - start_time > budget_s:
                break
            p1 = np.random.choice(parents)
            p2 = np.random.choice(parents)
            alpha = np.random.uniform(0.5, 1.5)
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        for _ in range(len(population) - len(new_population)):
            if time.time() - start_time > budget_s:
                break
            p = np.random.rand(n_points)
            p = p * target_sum / np.sum(p)
            p = np.clip(p, 0, 1)
            new_population.append(p)
        population = np.array(new_population)
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
