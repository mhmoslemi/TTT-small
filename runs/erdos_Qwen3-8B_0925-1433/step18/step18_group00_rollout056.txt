import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h0.copy()
    best_c5 = obj(best_h)
    
    start_time = time.time()
    
    # Generate diverse initial guesses
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Sine-based
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = h * target_sum / np.sum(h)
        # Perturb and scale
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success and obj(res.x) < best_c5:
            best_h, best_c5 = res.x.copy(), obj(res.x)
        # Binary pattern
        h_bin = np.random.choice([0,1], size=n_points)
        h_bin = h_bin * target_sum / np.sum(h_bin)
        h_bin += np.random.normal(0, 0.05, n_points)
        h_bin = np.clip(h_bin, 0, 1)
        h_bin = h_bin * target_sum / np.sum(h_bin)
        res = minimize(obj, h_bin, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success and obj(res.x) < best_c5:

    # Refinement

        if res.success and obj(res.x) < best_c5:
            best_h, best_c5 = res.x.copy(), obj(res.x)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
