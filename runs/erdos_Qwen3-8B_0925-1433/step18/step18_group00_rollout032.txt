    initial_guesses.append(0.5*(0.5 + 0.5*np.sin(np.pi*x)) + 0.5*((1 + np.sign(np.sin(np.pi*x)))/2))

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        guess_clipped = np.clip(guess, 0, 1)
        try:
            guess_scaled = guess_clipped * target_sum / np.sum(guess_clipped)
        except:
            guess_scaled = guess_clipped
        res = minimize(objective, guess_scaled, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h_opt = res.x
            current_c5 = np.max(np.correlate(current_h_opt, 1 - current_h_opt, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h_opt.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        try:
            perturbed = perturbed * target_sum / np.sum(perturbed)
        except:
            perturbed = perturbed
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
        if res.success:
            current_h_opt = res.x
            current_c5 = np.max(np.correlate(current_h_opt, 1 - current_h_opt, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h_opt.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
