import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_variants(base):
        variants = []
        for a in [0.05, 0.1, 0.15]:
            for f in [0.1, 0.2, 0.3]:
                mod = a * np.sin(2 * np.pi * f * np.arange(n_points) / n_points)
                variants.append(base + mod)
        for p in [2, 3, 4]:
            variant = np.zeros(n_points)
            for i in range(n_points):
                variant[i] = 1.0 if (i % p < 2) else 0.0
            variants.append(variant)
        for perm in [np.random.permutation(n_points) for _ in range(3)]:
            variants.append(base[perm])
        return [clip_normalize(v) for v in variants]

    initial_guesses = generate_variants(np.array(initial_h_values, dtype=float))
    initial_guesses = np.unique(initial_guesses, axis=0)

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(lambda x: compute_c5(clip_normalize(x)), guess, method='L-BFGS-B',
                           bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           tol=1e-6)
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5
        except:
            pass

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = clip_normalize(best_h)
        res = minimize(lambda x: compute_c5(clip_normalize(x)), h, method='L-BFGS-B',
                       bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    for _ in range(4):
        if time.time() - start_time > budget_s:

        corr = np.correlate(best_h, 1 - best_h, mode='full')
