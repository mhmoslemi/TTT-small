import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate structured variations
    variations = []
    for _ in range(30):
        # Modulation
        A = np.random.uniform(0.05, 0.2)
        f = np.random.uniform(0.1, 0.5)
        phi = np.random.uniform(0, 2 * np.pi)
        mod = A * np.sin(2 * np.pi * f * np.arange(n_points) + phi)
        new_h = initial_h + mod
        variations.append(clip_normalize(new_h))
        
        # Permutation
        perm = np.random.permutation(n_points)
        new_h = initial_h[perm]
        variations.append(clip_normalize(new_h))
        
        # Noise addition
        noise = np.random.normal(0, 0.05, n_points)
        new_h = initial_h + noise
        variations.append(clip_normalize(new_h))
    
    variations = np.unique(variations, axis=0)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Phase 1: Global optimization
    for guess in variations:
        if time.time() - start_time > budget_s:
            break
        res = minimize(compute_c5, guess, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
        if res.success:

            current_c5 = compute_c5(current_h)
