import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, top_lag, mutation_rate):
        n_points = len(h)
        delta = mutation_rate * 0.03
        h_mut = h.copy()
        for i in range(n_points):
            j = (i + top_lag) % n_points
            h_mut[i] -= delta
            h_mut[j] += delta
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        return h_mut

    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_generations = 200
    mutation_rate = 0.15
    elite_fraction = 0.2

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all candidates
        c5_values = [evaluate(h) for h in population]
        # Find best in current population
        current_best_idx = np.argmin(c5_values)
        current_best_c5 = c5_values[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = np.copy(population[current_best_idx])
            best_c5 = current_best_c5
        # Select elites
        elite_size = int(elite_fraction * population_size)
        elites = np.argsort(c5_values)[:elite_size]
        elites = [population[i] for i in elites]
        # Create new population
        new_population = elites.copy()
        for _ in range(len(population) - elite_size):
            # Tournament selection
            tournament = np.random.choice(range(population_size), size=5, replace=False)
            tournament_c5 = [c5_values[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_c5)]
            parent1 = population[winner_idx]
            # Another tournament for second parent
            tournament = np.random.choice(range(population_size), size=5, replace=False)
            tournament_c5 = [c5_values[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_c5)]

            child = (parent1 + parent2) / 2.0
