    best_h = h_initial.copy()
    best_c5 = np.inf

    mutation_sigma = 0.05
    mutation_decay = 0.99
    crossover_rate = 0.8
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = []
        for ind in population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            current_c5 = np.max(corr) * dx
            fitness.append(current_c5)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        elites = sorted_population[:int(0.2 * population_size)]
        remaining = sorted_population[int(0.2 * population_size):]

        new_population = elites.copy()
        for _ in range(len(remaining)):
            parent1 = np.random.choice(remaining)
            parent2 = np.random.choice(remaining)
            alpha = np.random.rand()
            offspring = alpha * parent1 + (1 - alpha) * parent2
            noise = np.random.normal(0, mutation_sigma, n_points)
            offspring += noise
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * target_sum / np.sum(offspring)
            new_population.append(offspring)

        # Evaluate new_population
        new_fitness = []
        for ind in new_population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            current_c5 = np.max(corr) * dx
            new_fitness.append(current_c5)
        current_best_c5 = min(new_fitness)
        best_idx = new_fitness.index(current_best_c5)
        best_h = new_population[best_idx].copy()
        best_c5 = current_best_c5

        if best_c5 < 0.3807:
            break

        mutation_sigma *= mutation_decay
        population = new_population

    return (np.clip(best_h, 0, 1), best_c5, n_points)
