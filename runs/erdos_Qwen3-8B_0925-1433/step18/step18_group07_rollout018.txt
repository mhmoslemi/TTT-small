import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_h_values + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    
    best_h = np.array(initial_h_values)
    best_c5 = evaluate_erdos_solution(best_h)
    
    mutation_strength = 0.1
    decay_rate = 0.999
    elitism_rate = 0.2
    crossover_rate = 0.8
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate all in population
        fitness = [(evaluate_erdos_solution(h), h) for h in population]
        # Sort by fitness
        fitness.sort()
        # Select elites
        elite_count = int(elitism_rate * population_size)
        elites = fitness[:elite_count]
        # Generate new population
        new_population = [h for _, h in elites]
        while len(new_population) < population_size:
            # Select parents
            p1, p2 = np.random.choice(fitness, size=2, replace=False)
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * p1[1] + (1 - alpha) * p2[1]
            else:
                noise = np.random.normal(0, mutation_strength, n_points)
                child = p1[1] + noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_population.append(child)
        # Update population
        population = new_population
        # Update best
        current_best = min(fitness, key=lambda x: x[0])
        if current_best[0] < best_c5:
            best_h = current_best[1].copy()
            best_c5 = current_best[0]
        # Decay mutation
        mutation_strength *= decay_rate
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
