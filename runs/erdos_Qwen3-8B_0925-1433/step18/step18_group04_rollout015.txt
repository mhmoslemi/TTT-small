import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = np.array(initial_h_values, dtype=float)
    except NameError:
        initial_h = np.ones(45) * (45 / 2) / 45
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    population = []
    velocities = []
    pbest = []
    pbest_fitness = []
    gbest = initial_h.copy()
    gbest_fitness = compute_c5(gbest)
    population.append(initial_h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(initial_h.copy())
    pbest_fitness.append(gbest_fitness)
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_fitness.append(compute_c5(h))
    population = np.array(population)
    velocities = np.array(velocities)
    pbest = np.array(pbest)
    pbest_fitness = np.array(pbest_fitness)

    best_h = gbest.copy()
    best_c5 = gbest_fitness
    start_time = time.time()
    max_iter = 100
    inertia = 0.9
    c1, c2 = 2.0, 2.0
    decay = (0.9 - 0.4) / max_iter

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        inertia -= decay
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * inertia + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            velocities[i] = np.clip(vel, -0.2, 0.2)
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
            fit = compute_c5(population[i])
            if fit < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = fit
        current_best = np.min(pbest_fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = pbest[np.argmin(pbest_fitness)].copy()
        if current_best < gbest_fitness:
            gbest_fitness = current_best
            gbest = pbest[np.argmin(pbest_fitness)].copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
