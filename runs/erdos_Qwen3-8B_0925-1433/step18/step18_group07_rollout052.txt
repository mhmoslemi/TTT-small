import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population = [initial_h.copy()]
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed.copy())
    population.append(np.full(n_points, 0.5))
    while len(population) < 100:
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(initial_h + noise, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed.copy())
    population = np.array(population)

    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    mutation_sigma = 0.05
    mutation_step = 0.001
    elitism_rate = 0.05

    while time.time() - start_time < budget_s:
        fitness = [evaluate_c5(h) for h in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = population[sorted_indices]
        elites = sorted_pop[:int(len(sorted_pop) * elitism_rate)]
        new_pop = elites.copy()

        for _ in range(len(sorted_pop) - len(elites)):
            a, b = np.random.choice(sorted_pop[:len(sorted_pop)//2], size=2, replace=False)
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * a + (1 - alpha) * b
            noise = np.random.normal(0, mutation_sigma, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)

        population = new_pop
        current_best = sorted_pop[np.argmin(fitness)]
        current_c5 = np.min(fitness)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        mutation_sigma = min(mutation_sigma + mutation_step, 0.1)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
