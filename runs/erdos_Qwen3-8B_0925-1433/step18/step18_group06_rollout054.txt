import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    sin_h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
    sin_h = np.clip(sin_h, 0, 1)
    sin_h = sin_h * target_sum / np.sum(sin_h)

    tri_h = ((2 * np.arange(n_points) / (n_points - 1) - 1) ** 2) + 0.5
    tri_h = np.clip(tri_h, 0, 1)
    tri_h = tri_h * target_sum / np.sum(tri_h)

    candidates = [h_initial, sin_h, tri_h]
    best_h = candidates[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for h in candidates[1:]:
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    h = best_h.copy()
    best_c5 = best_c5
    max_iterations = 200
    velocity = np.zeros(n_points)
    learning_rate = 0.1
    momentum = 0.99
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        i_plus_k = (np.arange(n_points) + k) % n_points
        i_minus_k = (np.arange(n_points) - k) % n_points
        gradient = (1 - h[i_plus_k]) - h[i_minus_k]
        velocity = momentum * velocity - learning_rate * gradient
        h += velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= 0.995

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
