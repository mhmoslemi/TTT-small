    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 1000
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k < n_points:
                part1 = 1 - h[j + k]
            else:
                part1 = 0
            if j - k >= 0:
                part2 = h[j - k]
            else:
                part2 = 0
            grad[j] = part1 - part2
        
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
