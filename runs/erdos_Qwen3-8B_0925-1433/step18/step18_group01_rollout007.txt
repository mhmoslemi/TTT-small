import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = []
    for _ in range(50):
        if np.random.rand() < 0.6:
            h = np.array(initial_h_values, dtype=float)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)
    objectives = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in population])

    best_h = population[0].copy()
    best_c5 = objectives[0]
    T = 1.0
    start_time = time.time()

    for _ in range(1000):
        time_used = time.time() - start_time
        if time_used >= budget_s:
            break
        T *= 0.995
        if _ % 50 == 0:
            dist = np.zeros(len(population))
            for i in range(len(population)):
                for j in range(i + 1, len(population)):
                    dist[i] += np.linalg.norm(population[i] - population[j])
                    dist[j] += np.linalg.norm(population[i] - population[j])
            avg_dist = dist.mean() / (len(population) * (len(population) - 1) / 2)
            if avg_dist < 0.3:
                replace = np.argsort(objectives)[:int(len(population)*0.1)]
                for idx in replace:
                    h = np.random.rand(n_points)
                    h = np.clip(h, 0, 1)
                    h = h * target_sum / np.sum(h)
                    population[idx] = h
                    objectives[idx] = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        for i in range(len(population)):
            h = population[i]
            obj_current = objectives[i]
            delta = T * np.random.normal(0, 0.15, n_points)
            h_candidate = np.clip(h + delta, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            obj_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if obj_candidate < obj_current or np.random.rand() < np.exp(-(obj_candidate - obj_current) / T):
                population[i] = h_candidate.copy()
                objectives[i] = obj_candidate
                if obj_candidate < best_c5:
                    best_c5 = obj_candidate
                    best_h = h_candidate.copy()

    return np.clip(best_h, 0, 1), best_c5, n_points
