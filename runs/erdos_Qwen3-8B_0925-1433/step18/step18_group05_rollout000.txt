import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, top_lag, mutation_rate):
        delta = mutation_rate * 0.05
        h_mut = h.copy()
        for i in range(n_points):
            j = (i + top_lag) % n_points
            h_mut[i] -= delta
            h_mut[j] += delta
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        return h_mut

    # Normalize initial values
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population = [initial_h.copy() for _ in range(50)]
    for h in population:
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_generations = 200
    mutation_rate = 0.15
    elite_fraction = 0.15

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        c5_values = [evaluate(h) for h in population]
        current_best_idx = np.argmin(c5_values)
        current_best_c5 = c5_values[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
        elite_size = int(elite_fraction * len(population))
        elites = [population[i] for i in np.argsort(c5_values)[:elite_size]]
        new_population = elites.copy()
        for _ in range(len(population) - elite_size):
            tournament = np.random.choice(range(len(population)), 5, replace=False)
            winner1 = tournament[np.argmin(c5_values[tournament])]
            parent1 = population[winner1]
            tournament = np.random.choice(range(len(population)), 5, replace=False)
            winner2 = tournament[np.argmin(c5_values[tournament])]
            parent2 = population[winner2]
            child = (parent1 + parent2) / 2.0
            corr = np.correlate(child, 1 - child, mode='full')
            max_lag = np.argmax(corr)
            child = mutate(child, max_lag, mutation_rate)
            new_population.append(child)
        population = new_population
        mutation_rate *= 0.995

    best_h = np.clip(best_h, 0, 1)
