import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_correlation(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        h = np.clip(h, 0, 1)
        corr_value = evaluate_correlation(h)
        sum_h = np.sum(h)
        penalty = 1e4 * (sum_h - target_sum) ** 2
        return corr_value + penalty

    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    # Sine wave
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = sine_guess * target_sum / np.sum(sine_guess)
    structured_guesses.append(sine_guess)

    # Triangle wave
    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    triangle_guess = triangle_guess * target_sum / np.sum(triangle_guess)
    structured_guesses.append(triangle_guess)

    # Square wave
    square_guess = np.where(np.sin(np.pi * x) > 0, 1, 0)
    square_guess = square_guess * target_sum / np.sum(square_guess)
    structured_guesses.append(square_guess)

    # Exponential decay
    exponential_guess = 0.5 * (1 - np.exp(-0.5 * x))
    exponential_guess = exponential_guess * target_sum / np.sum(exponential_guess)
    structured_guesses.append(exponential_guess)

    # Modified initial guess
    initial_guess = np.array(initial_h_values, dtype=float)
    modified_guess = initial_guess.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    modified_guess = modified_guess * target_sum / np.sum(modified_guess)
    structured_guesses.append(modified_guess)

    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=20, tol=1e-6,

        if result.success:

            h = h * target_sum / np.sum(h)

            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
