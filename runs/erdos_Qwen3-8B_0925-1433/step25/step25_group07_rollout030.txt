import numpy as np
from scipy.optimize import minimize
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, Matern

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_h = normalize(np.array(initial_h_values, dtype=float))
    candidates = [initial_h]
    
    # Generate more diverse initial samples
    rands = np.random.rand(n_points)
    candidates.append(normalize(rands))
    alt = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
    candidates.append(normalize(alt))
    shifted = np.roll(initial_h, 1)
    candidates.append(normalize(shifted))
    sine = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))
    candidates.append(normalize(np.clip(sine, 0, 1)))
    sparse = np.zeros(n_points)
    sparse[::5] = 1.0
    candidates.append(normalize(sparse))
    sparse2 = np.zeros(n_points)
    sparse2[::4] = 1.0
    candidates.append(normalize(sparse2))

    best_c5 = np.inf
    best_h = initial_h.copy()
    X = np.array([h for h in candidates])
    y = np.array([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in candidates])

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 100)) + Matern(length_scale=1.0, nu=1.5)

    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10, alpha=1e-5, random_state=seed)
    gp.fit(X, y)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def acquisition(X, gp, kappa):
        y_pred, sigma = gp.predict(X, return_std=True)
        return y_pred - kappa * np.sqrt(sigma ** 2 + 1e-8)

    start_time = time.time()
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < 200:
        iter_count += 1
        # Estimate next point
        X_opt = np.random.rand(n_points)
        X_opt = normalize(X_opt)
        X_opt = np.atleast_2d(X_opt)
        acq = acquisition(X_opt, gp, kappa=2.0)
        best_idx = np.argmin(acq)
        X_new = X_opt[best_idx].copy()
        y_new = evaluate(X_new)

        if y_new < best_c5:
            best_c5 = y_new
            best_h = X_new.copy()
        X = np.vstack([X, X_new])
        y = np.append(y, y_new)
        gp.fit(X, y)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
