import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h) + 0.5  # Avoid division by zero
        scale = target_sum / total
        return h * scale

    def evaluate_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = evaluate_correlation(best_h)
    start_time = time.time()

    # Generate structured initializations
    x = np.linspace(0, 2, n_points)
    structured_guesses = []
    structured_guesses.append(0.5 + 0.5 * np.sin(np.pi * x))       # Sine
    structured_guesses.append(0.5 + 0.5 * np.abs(np.sin(np.pi * x)))  # Triangle
    structured_guesses.append(np.concatenate((np.ones(n_points//2), np.zeros(n_points//2))))  # Square
    structured_guesses.append(0.5 * (1 + np.sign(np.sin(2 * np.pi * x))))  # Modulation
    structured_guesses.append(0.5 + 0.5 * (np.sin(4 * np.pi * x) ** 2))  # Modulated sine

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        def obj(h_vec):
            h = normalize(h_vec)
            return evaluate_correlation(h)
        result = differential_evolution(obj, bounds, maxiter=30, polish=True, tol=1e-6)
        h_candidate = normalize(result.x)
        c = evaluate_correlation(h_candidate)
        if c < best_c5:
            best_h, best_c5 = h_candidate.copy(), c

    # Local refinement with SLSQP
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        def obj(h):
            return evaluate_correlation(h)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_candidate = normalize(result.x)
            c = evaluate_correlation(h_candidate)
            if c < best_c5:
                best_h, best_c5 = h_candidate.copy(), c

    return (np.clip(best_h, 0, 1), best_c5, n_points)
