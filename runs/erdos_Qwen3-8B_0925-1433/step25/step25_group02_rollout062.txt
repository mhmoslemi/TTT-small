import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (target_sum / s)

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    population = [initial_guess.copy()]
    for _ in range(10):
        perturbed = initial_guess + 0.1 * np.random.randn(n_points)
        population.append(normalize(perturbed))
    population = [normalize(np.clip(p, 0, 1)) for p in population]

    pop_size = 100
    generations = 50
    mutation_rate = 0.05
    elite_percent = 0.2

    population = np.array(population)
    best_h = initial_guess
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate_c5(h) for h in population])
        elite_idx = np.argsort(fitness)[:int(elite_percent * pop_size)]
        elites = population[elite_idx]
        new_population = []
        for _ in range(pop_size - len(elite_idx)):
            p1, p2 = np.random.choice(elite_idx, 2, replace=False)
            child = (population[p1] + population[p2]) / 2
            if np.random.rand() < mutation_rate:
                child += 0.1 * np.random.randn(n_points)
            new_population.append(normalize(child))
        population = np.vstack([elites, new_population])
        if np.min(fitness[elite_idx]) < best_c5:
            best_c5 = np.min(fitness[elite_idx])
            best_h = population[np.argmin(fitness[elite_idx])]

    top_candidates = population[np.argsort(evaluate_c5(population))[:20]]
    for h in top_candidates:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            refined_h = normalize(result.x)
            current_c5 = evaluate_c5(refined_h)
            if current_c5 < best_c5:
                best_h, best_c5 = refined_h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
