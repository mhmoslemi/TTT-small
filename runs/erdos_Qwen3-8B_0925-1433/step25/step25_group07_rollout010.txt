import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 45
        h = np.full(n, 0.5)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return x
        return x * (target_sum / s)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    particles = []
    velocities = []
    personal_best = []
    for _ in range(50):
        pert = np.random.rand(n) * 0.2
        h_new = h + pert
        h_new = clip_rescale(h_new)
        particles.append(h_new)
        velocities.append(np.zeros(n))
        personal_best.append(h_new.copy())

    global_best = particles[0].copy()
    global_best_fitness = compute_c5(global_best)
    best_h = global_best
    best_c5 = global_best_fitness
    start = time.time()

    w, c1, c2 = 0.8, 1.5, 1.5
    for _ in range(100):
        if time.time() - start > budget_s:
            break
        for i in range(50):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            new_pos = particles[i] + vel
            new_pos = clip_rescale(new_pos)
            new_fitness = compute_c5(new_pos)

                personal_best[i] = new_pos.copy()

                    global_best_fitness = new_fitness
            velocities[i] = vel
