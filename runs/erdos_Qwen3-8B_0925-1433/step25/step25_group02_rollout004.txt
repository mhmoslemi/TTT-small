import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    npoints = n_points

    def normalize(x):

        x = x * target_sum / np.sum(x)

    best_h = initial_guess.copy()

    harmonic_guess = 0.5 + 0.5 * (np.sin(np.pi * x) + np.cos(np.pi * x)) / np.sqrt(2)

    harmonic_guess = normalize(harmonic_guess)
    structured_guesses = [harmonic_guess, initial_guess.copy()]
