    guesses.append(random_guess)
    # Perturbed initial guess
    perturbed = np.array(initial_h_values)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    perturbed = perturbed * (1 + np.random.normal(0, 0.05, n_points))
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    guesses.append(perturbed)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    # Run DE
    result = differential_evolution(
        objective, bounds, popsize=10, strategy='best1bin',
        tol=1e-6, maxiter=100, mutation=(0.5, 1), recombination=0.8,
        polish=True, seed=seed, x0=guesses[0]
    )
    best_h = result.x
    best_c5 = objective(best_h)

    # Refine with local optimization
    result_local = minimize(
        objective, best_h, method='Nelder-Mead', tol=1e-6, bounds=bounds
    )
    if result_local.success:
        best_h = result_local.x
        best_c5 = objective(best_h)

    # Final clip and normalize
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    best_h = best_h * target_sum / sum_h

    return (best_h, best_c5, n_points)
