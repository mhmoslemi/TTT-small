import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 100
    pop = [normalize(np.random.rand(n_points)) for _ in range(pop_size)]
    pop.append(normalize(np.array(initial_h_values, dtype=float)))
    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        fit = np.array([evaluate_c5(ind) for ind in pop])
        elites = np.argsort(fit)[:pop_size//2]
        selected = [pop[i] for i in elites]
        new_pop = []
        for _ in range(pop_size//2):
            a, b = np.random.choice(selected, 2, replace=False)
            child = (a + np.random.rand(n_points) * b) / 2
            if np.random.rand() < 0.1:
                child += np.random.normal(0, 0.1, n_points)
            new_pop.append(normalize(child))
        pop = selected + new_pop
        for h in pop:
            c = evaluate_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c

    fit = np.array([evaluate_c5(ind) for ind in pop])
    top = np.argsort(fit)[:20]
    for i in top:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(evaluate_c5, pop[i], method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            c = evaluate_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c

    return (np.clip(best_h, 0, 1), best_c5, n_points)
