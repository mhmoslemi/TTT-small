        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr[1:]) + 1
            grad = compute_gradient(h, k_max)
            h = h - 0.1 * grad
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr[1:]) + 1
        grad = compute_gradient(best_h, k_max)
        best_h = best_h - 0.1 * grad
        best_h = normalize(best_h)
        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
