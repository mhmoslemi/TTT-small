import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        factor = target_sum / np.sum(h)
        return h * factor

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def initial_guesses():
        x = np.linspace(0, 2, n_points)
        guesses = [np.array(initial_h_values, dtype=float)]
        sine = 0.5 + 0.5 * np.sin(np.pi * x)
        guesses.append(normalize(sine))
        triangle = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
        guesses.append(normalize(triangle))
        square = np.zeros_like(x)
        square[:int(n_points/2)] = 1.0
        guesses.append(normalize(square))
        perturbed = np.array(initial_h_values, dtype=float)
        perturbed += np.random.normal(0, 0.1, n_points)
        guesses.append(normalize(perturbed))
        random_guess = np.random.rand(n_points)
        guesses.append(normalize(random_guess))
        return guesses

    pop = initial_guesses()
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    F, CR = 0.5, 0.8
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_pop = []
        for i in range(len(pop)):
            a, b, c = np.random.choice(len(pop), 3, replace=False)
            mutant = pop[b] + F * (pop[a] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = normalize(trial)
            if compute_c5(trial) < compute_c5(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        for h in pop:
            if compute_c5(h) < best_c5:
                best_h, best_c5 = h.copy(), compute_c5(h)

    result = minimize(compute_c5, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                      constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    if result.success:
        best_h = np.clip(result.x, 0, 1)
        best_h = normalize(best_h)
        best_c5 = compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize(best_h)
        best_c5 = compute_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
