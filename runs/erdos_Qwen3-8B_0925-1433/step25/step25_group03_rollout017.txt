import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    sin_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sin_guess = normalize(sin_guess)
    structured_guesses.append(sin_guess)

    tri_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    tri_guess = normalize(tri_guess)
    structured_guesses.append(tri_guess)

    exp_guess = 0.5 + 0.5 * np.exp(-0.5 * x)
    exp_guess = normalize(exp_guess)
    structured_guesses.append(exp_guess)

    pc_guess = np.zeros(n_points)
    pc_guess[::5] = 1.0
    pc_guess = normalize(pc_guess)
    structured_guesses.append(pc_guess)

    rand_guess = initial_guess.copy()
    rand_guess += np.random.normal(0, 0.1, n_points)
    rand_guess = normalize(rand_guess)
    structured_guesses.append(rand_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_correlation, guess, method='Nelder-Mead', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_correlation, best_h, method='Nelder-Mead', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
