import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with parent solution
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()

    # Generate diversified initial perturbations
    def generate_perturbation(base):
        perturb = base.copy()
        # Random permutation
        perturb = np.random.permutation(perturb)
        # Add Gaussian noise
        perturb += 0.15 * np.random.randn(n_points)
        # Sinusoidal modulation
        wave = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))
        perturb += 0.2 * (wave - np.mean(wave))
        perturb = np.clip(perturb, 0, 1)
        perturb = perturb * target_sum / np.sum(perturb)
        return perturb

    # Perturbations to avoid local optima
    initial_guesses = [generate_perturbation(h) for _ in range(3)]
    initial_guesses.extend([np.random.rand(n_points) * target_sum / n_points for _ in range(2)])

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    # Adaptive optimization with gradient focusing on maximum overlap
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr[1:])  # Exclude zero lag
        k = max_k
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += 1 - best_h[i + k]
            if i - k >= 0:
                grad[i] -= best_h[i - k]
        
        step_size = 0.15 * (1.0 / (1.0 + np.max(np.abs(grad))))
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = objective(new_h)
        
        if new_c5 < best_c5:
            best_h, best_c5 = new_h.copy(), new_c5

    return np.clip(best_h, 0, 1), best_c5, n_points
