import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def true_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def smooth_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_grad(x, lag):
        h = x
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + lag < n_points:
                grad[i] += (1 - h[i + lag]) * dx
            if i - lag >= 0:
                grad[i] -= h[i - lag] * dx
        return grad

    def laguerre_step(x, lags, step_size):
        grad = np.zeros_like(x)
        for lag in lags:
            grad += compute_grad(x, lag)
        x_new = np.clip(x - step_size * grad, 0, 1)
        x_new = normalize(x_new)
        return x_new

    structured_guesses = [initial_guess.copy()]
    for _ in range(3):
        perturbed = np.random.permutation(initial_guess)
        structured_guesses.append(normalize(perturbed))
        perturbed = initial_guess + 0.15 * np.random.randn(n_points)

        perturbed = np.clip(perturbed, 0, 1)
        structured_guesses.append(normalize(perturbed))

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        h = guess.copy()
        for _ in range(50):
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr[1:]) + 1
            lags = [max_idx]
            if len(lags) < 10:
                lags += np.argsort(corr[1:])[-9:]
            h = laguerre_step(h, lags, 0.1)
            current_c5 = smooth_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
            if np.abs(current_c5 - best_c5) < 1e-6:

        h = guess.copy()
        for _ in range(50):
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr[1:]) + 1
            lags = [max_idx]

        result = minimize(
