    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr[1:])
        grad = np.zeros_like(best_h)
        for i in range(n_points):
            j = i + max_k
            if 0 <= j < n_points:
                grad[i] += (1 - best_h[j]) * dx
            j = i - max_k
            if 0 <= j < n_points:
                grad[i] += best_h[j] * dx
        step_size = 0.1
        h_new = project(best_h - step_size * grad)
        current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full') * dx)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
