import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    x = np.linspace(0, 2, n_points)
    guesses = []

    # Square wave
    sq = np.zeros(n_points)
    for i in range(n_points):
        seg = int(x[i] * 4)
        if seg % 2 == 0:
            sq[i] = 1
    sq = normalize(sq)
    guesses.append(sq)

    # Triangle wave
    tri = np.abs(x - 1) - 0.5
    tri = np.clip(tri, 0, 1)
    tri = normalize(tri)
    guesses.append(tri)

    # Sine wave
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = normalize(sine)
    guesses.append(sine)

    # Exponential decay
    exp = 0.5 + 0.5 * np.exp(-x)
    exp = normalize(exp)
    guesses.append(exp)

    # Random permutated
    perm = np.random.permutation(initial_h_values)
    perm = normalize(perm)
    guesses.append(perm)

    # Polynomial
    poly = 0.5 + 0.5 * (np.sin(np.pi * x) ** 2)
    poly = normalize(poly)
    guesses.append(poly)

    def de_cost(h):
        h = normalize(h)
        return evaluate_correlation(h)

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(de_cost, [(0, 1)] * n_points, strategy='best1bin', popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, maxiter=100)
        if res.success:
            candidate = np.clip(res.x, 0, 1)
            candidate = normalize(candidate)
            current_c5 = evaluate_correlation(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_correlation, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
        if res.success:
            candidate = np.clip(res.x, 0, 1)
            candidate = normalize(candidate)
            current_c5 = evaluate_correlation(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
