import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective_smooth(x, kappa):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))

    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = normalize(initial_guess)

    perturbations = [initial_guess]
    x = np.linspace(0, 2, n_points)
    perturbations.append(initial_guess + np.random.normal(0, 0.1, n_points))
    perturbations.append(0.5 + 0.5 * np.sin(np.pi * x))
    perturbations.append(0.5 + 0.5 * np.cos(np.pi * x))
    perturbations.append(0.5 * (2 * (x / 2) - 1))
    perturbations.append(0.5 + 0.2 * (x / 2) + 0.3 * (x / 2) ** 2)
    perturbations = [normalize(p) for p in perturbations]

    best_h = initial_guess.copy()
    best_c5 = objective_true(best_h)
    start_time = time.time()

    for guess in perturbations:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(objective_smooth, guess, args=(50,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h = normalize(h)
                current_c5 = objective_true(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
        except:
            pass

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(objective_true, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h = normalize(h)
                current_c5 = objective_true(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
        except:
            pass

    return (np.clip(best_h, 0, 1), best_c5, n_points)
