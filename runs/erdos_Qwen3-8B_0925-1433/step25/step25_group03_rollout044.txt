import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = normalize(x)
        return evaluate_correlation(h)

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = evaluate_correlation(best_h)

    # Generate structured guesses
    x = np.linspace(0, 2, n_points)
    initial_guesses = [normalize(initial_h_values)]
    initial_guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
    initial_guesses.append(normalize(0.5 + 0.5 * np.abs(np.sin(np.pi * x))))
    initial_guesses.append(normalize(np.exp(-x) / np.exp(-2) * 0.5 + 0.5))
    initial_guesses.append(normalize(np.random.rand(n_points)))
    initial_guesses.append(normalize(initial_h_values * 0.5 + 0.5))

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, maxiter=30, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        if result.success:
            h = normalize(result.x)
            c5 = evaluate_correlation(h)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_correlation, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
            h = normalize(result.x)
            c5 = evaluate_correlation(h)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
