import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x, lambd):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / lambd) * np.log(np.sum(np.exp(-lambd * corr)) + 1e-6)

    def evaluate_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_true(best_h)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    base = initial_guess.copy()
    structured_guesses.append(base)
    structured_guesses.append(np.clip(base + 0.1 * np.random.randn(n_points), 0, 1))
    structured_guesses.append(np.clip(base - 0.1 * np.random.randn(n_points), 0, 1))
    structured_guesses.append(np.clip(0.5 + 0.5 * np.sin(np.pi * x), 0, 1))
    structured_guesses.append(np.clip(0.5 + 0.5 * np.cos(np.pi * x), 0, 1))
    structured_guesses.append(np.clip(0.5 + 0.5 * (2 * (x / 2) - 1), 0, 1))
    structured_guesses.append(np.clip(0.5 + 0.2 * (x / 2) + 0.3 * (x / 2)**2, 0, 1))

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, args=(50,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_true(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_true, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_true(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
