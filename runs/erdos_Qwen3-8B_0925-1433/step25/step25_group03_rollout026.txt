import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h):
        sum_h = np.sum(h)
        penalty = 1e6 * (sum_h - target_sum)**2
        corr = evaluate_correlation(h)
        return corr + penalty

    structured_guesses = []

    x = np.linspace(0, 2, n_points)
    square = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4:
            square[i] = 1.0
        elif i < n_points // 2:
            square[i] = 0.0
        elif i < 3 * n_points // 4:
            square[i] = 1.0
        else:
            square[i] = 0.0
    square /= square.sum() * dx
    structured_guesses.append(square)

    triangle = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    triangle /= triangle.sum() * dx
    structured_guesses.append(triangle)

    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine /= sine.sum() * dx
    structured_guesses.append(sine)

    exponential = 0.5 + 0.5 * np.exp(-0.5 * x)
    exponential /= exponential.sum() * dx
    structured_guesses.append(exponential)

    mod_initial = np.array(initial_h_values, dtype=float)
    mod_initial[np.random.rand(n_points) < 0.3] = 1 - mod_initial[np.random.rand(n_points) < 0.3]
    mod_initial /= mod_initial.sum() * dx
    structured_guesses.append(mod_initial)

    poly = 0.5 + 0.5 * (x**2) / (x**2 + 1)
    poly /= poly.sum() * dx
    structured_guesses.append(poly)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_correlation(best_h)

    start_time = time.time()
    for guess in structured_guesses[:5]:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=20, maxiter=100, tol=1e-6)
        h_candidate = np.clip(result.x, 0, 1)
        h_candidate /= h_candidate.sum() * dx
        current_c5 = evaluate_correlation(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_correlation, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
            h_candidate = np.clip(result.x, 0, 1)
            h_candidate /= h_candidate.sum() * dx
            current_c5 = evaluate_correlation(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    return np.clip(best_h, 0, 1), best_c5, n_points
