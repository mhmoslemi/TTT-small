import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate initial guesses
    initial_guesses = [
        normalize(np.array(initial_h_values, dtype=float)),
        normalize(np.zeros(n_points) + 0.5),
        normalize(0.5 * np.sin(2 * np.pi * 1.2 * np.linspace(0, 2, n_points)) + 0.5),
        normalize(np.random.rand(n_points)),
    ]

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_guesses[0].copy()
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()

    # Differential Evolution with multiple initial guesses
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            evaluate_c5, bounds, strategy='best1bin', 
            popsize=15, mutation=(0.5, 1), recombination=0.9,
            tol=1e-5, maxiter=50, seed=seed, x0=guess, disp=False
        )
        if result.success:
            h = result.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    # L-BFGS-B refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
