import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return x
        x *= target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_initial_guesses():
        guesses = [np.array(initial_h_values, dtype=float)]
        # Pulse pattern
        num_pulses = 5
        spacing = 2.0 / num_pulses
        pulse_positions = np.arange(num_pulses) * spacing
        pulse_pattern = np.zeros(n_points)
        for pos in pulse_positions:
            idx = int(pos / dx)
            pulse_pattern[idx] = 1.0
        guesses.append(pulse_pattern)
        # Cluster pattern
        cluster_size = 3
        cluster_positions = [0.5, 1.5]
        cluster_pattern = np.zeros(n_points)
        for pos in cluster_positions:
            idx = int(pos / dx)
            cluster_pattern[idx] = 1.0
        guesses.append(cluster_pattern)
        return guesses

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_c5(normalize(best_h))
    start_time = time.time()

    def objective(x):
        h = normalize(x)
        return evaluate_c5(h)

    guesses = generate_initial_guesses()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(
            objective,
            bounds=[(0, 1)] * n_points,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.8,
            maxiter=100,
            polish=True,
            disp=False,
            seed=seed,
            vectorized=True
        )
        h_candidate = res.x
        h_candidate = normalize(h_candidate)
        current_c5 = evaluate_c5(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate
            best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            evaluate_c5,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}],
            tol=1e-7,
            options={'maxiter': 100}
        )
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
