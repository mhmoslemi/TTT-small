import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    x = np.linspace(0, 2, n_points)
    guesses = [
        normalize(initial_h_values),
        normalize(1.0 - initial_h_values),
        normalize(np.roll(initial_h_values, 1)),
        normalize(0.5 + 0.5 * np.sin(np.pi * x)),
        normalize(np.random.rand(n_points)),
        normalize(0.5 + 0.5 * (-1)**np.arange(n_points)),
        normalize(0.5 * (1 + np.cos(np.pi * x * 2))),
        normalize(0.5 * (1 + np.cos(np.pi * x * 3))),
    ]

    best_h = guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, kappa=100):
        corr = np.correlate(x, 1 - x, mode='full')
        exp_vals = np.exp(-kappa * corr)
        return np.log(np.sum(exp_vals)) / kappa

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            smooth_objective, guess, method='SLSQP', bounds=bounds,
            constraints=cons, tol=1e-5, options={'maxiter': 100}
        )
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturbation = 0.05 * np.random.randn(n_points)
        perturbed = best_h + perturbation
        perturbed = np.clip(normalize(perturbed), 0, 1)
        result = minimize(
            smooth_objective, perturbed, method='SLSQP', bounds=bounds,
            constraints=cons, tol=1e-5, options={'maxiter': 100}
        )
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
