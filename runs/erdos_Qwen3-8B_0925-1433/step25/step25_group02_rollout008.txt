        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = normalize(result.x)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    bounds = [(0, 1) for _ in range(n_points)]
    def objective(x):
        h = normalize(x)
        return evaluate_c5(h)
    result_de = differential_evolution(objective, bounds, popsize=10, strategy='best1bin', tol=1e-6, maxiter=50, seed=seed)

    best_h = normalize(result_de.x)
    best_c5 = evaluate_c5(best_h)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    result_slsqp = minimize(evaluate_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if result_slsqp.success:
        best_h = normalize(result_slsqp.x)
        best_c5 = evaluate_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
