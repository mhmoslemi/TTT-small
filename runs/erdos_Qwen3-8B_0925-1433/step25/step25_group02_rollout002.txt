import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_g = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_g.copy()
    best_c5 = evaluate_c5(best_h)

    def ga_step(pop, gen=0):
        if time.time() - start_time > budget_s:
            return pop
        if gen >= 50:
            return pop
        new_pop = []
        for _ in range(len(pop)):
            a, b = np.random.choice(pop, 2, replace=False)
            child = 0.5 * (a + b) + 0.15 * np.random.randn(n_points)
            child = normalize(child)
            new_pop.append(child)
        return ga_step(new_pop, gen + 1)

    structured_guesses = [
        normalize(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))),
        normalize(np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]))
    ]
    structured_guesses.append(initial_g)

    start_time = time.time()
    pop = [normalize(np.random.rand(n_points)) for _ in range(20)]
    pop += structured_guesses
    pop = [normalize(np.clip(x, 0, 1)) for x in pop]
    pop = ga_step(pop)

    for candidate in pop[:10]:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
