import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_h = np.array(initial_h_values, dtype=float)
    h = normalize(initial_h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        def objective(x):
            h_opt = x
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            return corr_opt[k_max] * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_new = np.clip(result.x, 0, 1)
            h_new = normalize(h_new)
            current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), current_c5
                h = h_new
    return (np.clip(best_h, 0, 1), best_c5, n_points)
