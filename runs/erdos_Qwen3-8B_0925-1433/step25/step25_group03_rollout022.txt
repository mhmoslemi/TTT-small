import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_norm(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * (target_sum / total) if total != 0 else np.zeros_like(h)

    def obj(h):
        h = clip_norm(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured = []
    x = np.linspace(0, 2, n_points)
    structured.append(0.5 + 0.5 * np.sin(np.pi * x))
    structured.append(0.5 + 0.5 * np.abs(np.sin(np.pi * x)))
    structured.append(np.zeros_like(x))
    structured.append(clip_norm(np.array(initial_h_values)))
    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        structured.append(clip_norm(np.array(initial_h_values) + noise))

    best_h = clip_norm(np.array(initial_h_values))
    best_c5 = obj(best_h)
    start = time.time()

    for guess in structured:
        if time.time() - start > budget_s:
            break
        bounds = [(0, 1)] * n_points
        res = differential_evolution(obj, bounds, popsize=15, strategy='best1bin', mutation=(0.5, 1), recombination=0.8, tol=1e-6, maxiter=50, seed=seed)
        if res.success:
            h = res.x
            curr = obj(h)
            if curr < best_c5:
                best_h, best_c5 = h, curr

    for _ in range(5):
        if time.time() - start > budget_s:
            break
        def slsqp_obj(h):
            return obj(clip_norm(h))
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
        res = minimize(slsqp_obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            curr = obj(clip_norm(h))
            if curr < best_c5:
                best_h, best_c5 = h, curr

    return (best_h, best_c5, n_points)
