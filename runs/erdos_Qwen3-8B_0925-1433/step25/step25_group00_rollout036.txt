import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_p = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_p = np.max(corr_p) * dx
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_m = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_m = np.max(corr_m) * dx
            grad[i] = (c5_p - c5_m) / (2e-6)
        return grad

    max_iter = 300
    start_time = time.time()
    h = h * target_sum / np.sum(h)
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        h -= 0.2 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    h_fft = np.fft.fft(best_h)
    window = np.blackman(len(h_fft))
    filtered_fft = h_fft * window
    h_filtered = np.fft.ifft(filtered_fft).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered)
    final_c5 = np.max(np.correlate(h_filtered, 1 - h_filtered, mode='full')) * dx
    if final_c5 < best_c5:
        best_h, best_c5 = h_filtered, final_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
