def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate initial guesses
    guesses = []
    base = np.array(initial_h_values, dtype=float)
    guesses.append(normalize(base))
    guesses.append(normalize(1.0 - base))
    guesses.append(normalize(np.roll(base, 1)))
    guesses.append(normalize(np.clip(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)), 0, 1)))
    guesses.append(normalize(np.random.rand(n_points)))
    guesses.append(normalize(np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])))
    # Add more guesses: cosine, binary sparse, etc.
    x = np.linspace(0, 2, n_points)
    guesses.append(normalize(0.5 + 0.5 * np.cos(np.pi * x)))
    # Binary sparse pattern: set every third point to 1
    binary = np.zeros(n_points)
    binary[::3] = 1.0
    guesses.append(normalize(binary))
    # Another binary pattern: every fourth point
    binary2 = np.zeros(n_points)
    binary2[::4] = 1.0
    guesses.append(normalize(binary2))
    # Random sparse: random 1's
    sparse = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(n_points/5), replace=False)
    sparse[indices] = 1.0
    guesses.append(normalize(sparse))

    best_h = guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, kappa=100):  # increased kappa
        corr = np.correlate(x, 1 - x, mode='full')
        exp_vals = np.exp(-kappa * corr)

    start_time = time.time()

        if time.time() - start_time > budget_s:

        if result.success:

    for _ in range(3):
