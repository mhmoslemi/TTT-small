    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_max = np.max(corr) * dx
        if current_max < best_c5:
            best_c5 = current_max
        max_index = np.argmax(corr)
        def objective(x):
            h = x
            corr_val = np.correlate(h, 1 - h, mode='full')[max_index]
            return corr_val * dx
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            h_new = np.clip(result.x, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            current_c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = current_c5_new
    return (np.clip(best_h, 0, 1), best_c5, n_points)
