import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x) if np.sum(x) > 0 else x
        return x

    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        def obj(x):
            h = x
            corr_val = np.correlate(h, 1 - h, mode='full')[max_idx]
            return corr_val * dx
        result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            h_new = np.clip(result.x, 0, 1)
            h_new = normalize(h_new)
            current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_new
                best_c5 = current_c5
        else:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
