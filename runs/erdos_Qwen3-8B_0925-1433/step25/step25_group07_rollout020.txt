import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        return x

    initial_h = np.array(initial_h_values, dtype=float)
    population = [normalize(initial_h), normalize(1.0 - initial_h)]

    population.append(normalize(shifted))
    sine = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))

    binary = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
    population.append(normalize(binary))
    random_guess = np.random.rand(n_points)
    population.append(normalize(random_guess))
    sine2 = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 2, n_points))
    population.append(normalize(np.clip(sine2, 0, 1)))
    noise = np.random.rand(n_points)
    population.append(normalize(np.clip(noise, 0, 1)))
    while len(population) < 50:
        population.append(normalize(0.5 + 0.5 * np.random.rand(n_points)))

    best_h = population[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    gen = 0
    while gen < 100 and time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        idx = np.argmin(fitness)

        best_h = population[idx].copy()
