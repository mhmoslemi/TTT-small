import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    triangular_guess = np.zeros(n_points)
    for i in range(n_points):
        triangular_guess[i] = 1.0 if i < n_points // 2 else 0.0
    triangular_guess = normalize(triangular_guess)
    structured_guesses.append(triangular_guess)
    step_guess = np.zeros(n_points)
    step_guess[:n_points//3] = 1.0
    step_guess = normalize(step_guess)
    structured_guesses.append(step_guess)
    random_guess = np.random.rand(n_points)
    random_guess = normalize(random_guess)
    structured_guesses.append(random_guess)

    start_time = time.time()
    population = [g.copy() for g in structured_guesses]
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate_c5(g) for g in population])
        idx = np.argsort(fitness)
        selected = population[idx[:len(population)//2]]
        new_population = selected.copy()
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = (p1 + p2) / 2
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = normalize(np.clip(child, 0, 1))
            new_population.append(child)
        population = new_population

    for guess in population:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
