import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def generate_structured_guesses(x):
        sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
        sine_guess = normalize(sine_guess)
        triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
        triangle_guess = normalize(triangle_guess)
        exp_guess = 0.5 + 0.5 * np.exp(-x / 1.5)
        exp_guess = normalize(exp_guess)
        piecewise_guess = np.zeros_like(x)
        piecewise_guess[::3] = 1.0
        piecewise_guess = normalize(piecewise_guess)
        random_start = np.random.rand(n_points)
        random_start = normalize(random_start)
        return [sine_guess, triangle_guess, exp_guess, piecewise_guess, random_start]

    x = np.linspace(0, 2, n_points)
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    structured_guesses = generate_structured_guesses(x)
    for guess in structured_guesses:
        if time.time() - time.time() > budget_s:
            break
        result = minimize(evaluate_correlation, guess, method='SLSQP', bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        result = minimize(evaluate_correlation, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
