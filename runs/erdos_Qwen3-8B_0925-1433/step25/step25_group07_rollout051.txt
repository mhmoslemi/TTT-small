import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)

    # PSO parameters
    num_particles = 50
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_iter = 100
    penalty_factor = 1e5

    # Initialize particles
    particles = [initial_h.copy() + np.random.uniform(-0.1, 0.1, n_points) for _ in range(num_particles)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(num_particles)]
    pbest = particles.copy()
    pbest_fitness = [float('inf')] * num_particles
    gbest = initial_h.copy()
    gbest_fitness = float('inf')

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(num_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel = velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            new_pos = particles[i] + vel
            # Clip and rescale
            new_pos_clipped = np.clip(new_pos, 0, 1)
            sum_clipped = np.sum(new_pos_clipped)
            new_pos_rescaled = new_pos_clipped * target_sum / sum_clipped if sum_clipped != 0 else new_pos_clipped
            # Force sum to be exactly target_sum
            particles[i] = new_pos_rescaled
            velocities[i] = vel  # Velocity remains as computed
            # Compute fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(corr) * dx
            # Penalty for sum constraint
            penalty = (np.sum(particles[i]) - target_sum) ** 2 * penalty_factor
            fitness = current_c5 + penalty
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        # Update global best
        current_fitness = pbest_fitness
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < gbest_fitness:
            gbest = pbest[best_idx].copy()
            gbest_fitness = current_fitness[best_idx]

    # Final C5 evaluation
    corr = np.correlate(gbest, 1 - gbest, mode='full')
    best_c5 = np.max(corr) * dx
    return (np.clip(gbest, 0, 1), best_c5, n_points)
