import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        x = np.clip(x, 0, 1)
        x = normalize(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Initial guesses
    guesses = [h_initial]
    x = np.linspace(0, 2, n_points)
    h_pulse = np.zeros(n_points)
    for i in range(5):
        h_pulse[int(0.2 + i * 0.4 * n_points)] = 1.0
    guesses.append(normalize(h_pulse))
    freq = 1.0
    phi = np.random.rand() * 2 * np.pi
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phi)
    guesses.append(normalize(h_sinusoidal))
    guesses.append(normalize(np.random.rand(n_points)))
    population = [np.clip(g, 0, 1) for g in guesses]

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(h) for h in population]
        parents = [h for _, h in sorted(zip(fitness, population), key=lambda x: x[0])[:4]]
        new_pop = []
        for _ in range(10):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.1:
                idxs = np.random.choice(n_points, 5, replace=False)
                child[idxs] += np.random.normal(0, 0.1, 5)
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_pop.append(child)
        population = sorted(population + new_pop, key=lambda h: evaluate_c5(h))[:20]
        for h in population:
