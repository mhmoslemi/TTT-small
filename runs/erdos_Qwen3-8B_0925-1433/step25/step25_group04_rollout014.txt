import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[0:22] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_c5 = np.max(corr) * dx
        if max_c5 < best_c5:
            best_c5 = max_c5
        if max_c5 <= 0.3807:
            break
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(best_h)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] += (1 - best_h[j])
                grad[j] -= best_h[i]
        learning_rate = 0.05
        h_new = best_h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize(h_new)
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
    
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    return (np.clip(best_h, 0, 1), final_c5, n_points)
