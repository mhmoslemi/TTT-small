import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guesses = []
    initial_guesses.append(initial_h_values)

    x = np.linspace(0, 2, n_points)
    freqs = [1.1, 1.3, 1.5]
    phases = np.random.rand(len(freqs)) * 2 * np.pi
    for freq, phase in zip(freqs, phases):
        sin_wave = 0.4 * np.sin(2 * np.pi * freq * x + phase) + 0.5
        initial_guesses.append(normalize(sin_wave))

    cluster_sizes = [6, 8, 10]
    cluster_start = n_points // 3
    for size in cluster_sizes:
        cluster = np.zeros(n_points)
        if cluster_start + size <= n_points:
            cluster[cluster_start:cluster_start + size] = 1.0
        initial_guesses.append(normalize(cluster))
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    best_h = initial_h_values
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        initial_guess = normalize(np.array(guess, dtype=float))
        res = minimize(evaluate_c5, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
