import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def harmonic_guess():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * (np.sin(np.pi * x) + np.cos(np.pi * x)) / np.sqrt(2)
        h = np.clip(h, 0, 1)
        return normalize(h)

    def triangle_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return normalize(h)

    def uniform_guess():
        return normalize(np.random.rand(n_points))

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = normalize(initial_h)
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    structured_guesses = [initial_h, harmonic_guess(), triangle_guess(), uniform_guess()]

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
