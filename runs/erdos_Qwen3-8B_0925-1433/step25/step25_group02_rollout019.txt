import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    def genetic_algorithm():
        population = [normalize(np.random.rand(n_points)) for _ in range(20)]
        for gen in range(50):
            if time.time() - start_time > budget_s:
                break
            fitness = np.array([evaluate_c5(ind) for ind in population])
            elite_idx = np.argsort(fitness)[:5]
            elites = [population[i] for i in elite_idx]
            new_pop = elites
            for _ in range(len(population) - 5):
                parent1, parent2 = np.random.choice(elite_idx, 2, replace=False)
                child = (population[parent1] + population[parent2]) / 2 + np.random.normal(0, 0.15, n_points)
                new_pop.append(normalize(child))
            population = new_pop
            if np.min(fitness) < best_c5:
                best_h = population[np.argmin(fitness)].copy()
                best_c5 = np.min(fitness)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    triangular_guess = np.zeros(n_points)
    for i in range(n_points):
        triangular_guess[i] = 1.0 if i < n_points // 2 else 0.0
    triangular_guess = normalize(triangular_guess)
    structured_guesses.append(triangular_guess)
    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    structured_guesses.append(uniform_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
