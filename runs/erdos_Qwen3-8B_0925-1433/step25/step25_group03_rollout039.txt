import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum == 0:
            return np.full_like(x, 0.5)
        return x * target_sum / x_sum

    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    # Sine wave
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    structured_guesses.append(normalize(sine))
    # Triangle wave
    triangle = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    structured_guesses.append(normalize(triangle))
    # Square wave
    square = np.where(np.sin(np.pi * x) > 0, 1, 0)
    structured_guesses.append(normalize(square))
    # Random guess
    random_guess = np.random.rand(n_points)
    structured_guesses.append(normalize(random_guess))
    # Perturbed parent
    perturbed = initial_h_values.copy()
    perturbed = normalize(perturbed)
    structured_guesses.append(perturbed)

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum == 0:
            return np.inf
        h_norm = h_clipped * target_sum / h_sum
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    result_de = differential_evolution(objective, bounds, strategy='best1bin', 
                                      maxiter=100, popsize=20, tol=1e-6, 
                                      mutation=(0.5, 1), recombination=0.8, 
                                      seed=seed, polish=True, disp=False)
    
    best_h = result_de.x
    best_c5 = objective(best_h)

    def optimize_with_slsqp(h):
        def obj_slsqp(x):
            return objective(x)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = minimize(obj_slsqp, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       tol=1e-6, options={'maxiter': 100})
        if res.success:
            return res.x
        return h

    refined_h = optimize_with_slsqp(best_h)
    refined_c5 = objective(refined_h)

    initial_c5 = objective(initial_h_values)
    best_h_final = refined_h if refined_c5 < initial_c5 else initial_h_values
    best_c5_final = min(refined_c5, initial_c5)

    return (normalize(best_h_final), best_c5_final, n_points)
