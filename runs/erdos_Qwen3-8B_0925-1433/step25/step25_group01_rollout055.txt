import time

        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_population():

        pop.append(normalize(h_sinusoidal))
        pop.append(normalize(initial_h_values))
        pop.append(normalize(np.random.rand(n_points)))
        return pop

    population = generate_population()
    best_h = normalize(np.array(initial_h_values, dtype=float))
