import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    population_size = 100
    generations = 50
    elite_ratio = 0.2
    mutation_rate = 0.05
    mutation_amp = 0.05
    time_left = budget_s

    # GA initialization
    population = np.array([np.random.uniform(0, 1, n_points) for _ in range(population_size)])
    population = normalize(population)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([evaluate_c5(ind) for ind in population])
        # Select elites
        elite_idx = np.argpartition(fitness, int(elite_ratio * population_size))[:int(elite_ratio * population_size)]
        elite = population[elite_idx]
        # Generate new population
        new_pop = []
        for _ in range(population_size - len(elite)):
            # Crossover
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = (p1 + p2) / 2
            # Mutation
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, mutation_amp, n_points)
            child = normalize(child)
            new_pop.append(child)
        population = np.vstack([elite, new_pop])
        # Update best
        for ind in population:
            current_c5 = evaluate_c5(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()

    # Local refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 0:
        start_refine = time.time()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        for _ in range(5):
            if time.time() - start_refine > time_left:
                break
            result = minimize(evaluate_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = normalize(h)
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
