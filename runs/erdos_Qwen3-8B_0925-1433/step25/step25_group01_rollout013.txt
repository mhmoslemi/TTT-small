from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = normalize(np.clip(x, 0, 1))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}

    start_time = time.time()

    def generate_initial_guesses():
        guesses = [initial_guess]
        x = np.linspace(0, 2, n_points)
        for freq in [1.1, 1.3, 1.5]:
            phase = np.random.rand() * 2 * np.pi
            sin_wave = 0.4 * np.sin(2 * np.pi * freq * x + phase) + 0.5
            guesses.append(normalize(sin_wave))
        for m_pulses in [3, 5, 7]:
            h_pulse = np.zeros(n_points)
            spacing = 2.0 / (m_pulses * 2)
            for i in range(m_pulses):
                idx = int(i * spacing * n_points)
                h_pulse[idx] = 1.0
            guesses.append(normalize(h_pulse))
        for _ in range(3):
            perturb = np.random.normal(0, 0.05, n_points)
            guesses.append(normalize(initial_guess + perturb))
        return guesses

    for guess in generate_initial_guesses():
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 200})
            if res.success:
                h = res.x
                h = normalize(np.clip(h, 0, 1))
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
        except:
            pass

    for _ in range(5):
