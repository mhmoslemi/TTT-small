import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def compute_max_lag(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)

    def optimize_with_lag(h, lag):
        def objective(x):
            h_val = x
            corr = np.correlate(h_val, 1 - h_val, mode='full')
            return corr[lag] * dx
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_res = np.clip(result.x, 0, 1)
            h_res = normalize(h_res)
            c5 = np.max(np.correlate(h_res, 1 - h_res, mode='full')) * dx
            return h_res, c5
        return h, best_c5

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)

    flipped_guess = 1.0 - initial_h_values
    flipped_guess = normalize(flipped_guess)
    structured_guesses.append(flipped_guess)

    shifted_guess = np.roll(initial_h_values, 1)
    shifted_guess = normalize(shifted_guess)
    structured_guesses.append(shifted_guess)

    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    structured_guesses.append(uniform_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        lag = compute_max_lag(guess)
        h_res, c5 = optimize_with_lag(guess, lag)
        if c5 < best_c5:
            best_h, best_c5 = h_res, c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        lag = compute_max_lag(best_h)
        h_res, c5 = optimize_with_lag(best_h, lag)
        if c5 < best_c5:
            best_h, best_c5 = h_res, c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
