import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)

    # Generate initial population
    population_size = 50
    particles = [initial_h.copy()]
    for _ in range(population_size - 1):
        guess = np.random.rand(n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        particles.append(guess.copy())
    shifted = np.roll(initial_h, 1)
    shifted = shifted * target_sum / np.sum(shifted)
    shifted = np.clip(shifted, 0, 1)
    particles.append(shifted.copy())
    sine = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))
    sine = np.clip(sine, 0, 1)
