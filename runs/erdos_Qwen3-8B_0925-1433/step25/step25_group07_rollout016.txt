from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        return x

    base_guess = normalize(np.array(initial_h_values, dtype=float))
    guesses = [base_guess]

    # Generate diverse initial guesses
    shifted = np.roll(initial_h_values, 1)

    guesses.append(normalize(np.clip(sine, 0, 1)))
    alt = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
    guesses.append(normalize(alt))
    sparse = np.zeros(n_points)
    sparse[:int(n_points/2)] = 1
    guesses.append(normalize(sparse))

    best_h = base_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, kappa=100):
        corr = np.correlate(x, 1 - x, mode='full')
        exp_vals = np.exp(-kappa * corr)
        return np.log(np.sum(exp_vals)) / kappa

    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(smooth_objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
