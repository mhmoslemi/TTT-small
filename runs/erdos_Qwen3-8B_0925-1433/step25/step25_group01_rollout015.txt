import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    best_h = np.array(initial_h_values, dtype=float)
    best_h = normalize(best_h)
    best_c5 = evaluate_c5(best_h)

    def generate_initial_guesses():
        guesses = [best_h.copy()]
        x = np.linspace(0, 2, n_points)
        for _ in range(3):
            freq = 1.2 + np.random.rand() * 0.4
            phase = np.random.rand() * 2 * np.pi
            sin_wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
            guesses.append(normalize(sin_wave))
        for _ in range(2):
            offset = 0.2 * np.random.rand(n_points)
            guesses.append(normalize(best_h + offset))
        for _ in range(2):
            pulse = np.zeros(n_points)
            m_pulses = 5 + int(np.random.rand() * 3)
            spacing = 2.0 / (m_pulses * 2)
            for i in range(m_pulses):
                idx = int(i * spacing * n_points)
                pulse[idx] = 1.0
            guesses.append(normalize(pulse))
        return guesses

    start_time = time.time()
    guesses = generate_initial_guesses()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 200})
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 200})
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
