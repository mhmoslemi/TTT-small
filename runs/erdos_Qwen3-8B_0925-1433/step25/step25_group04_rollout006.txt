```python
import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_current = np.array(initial_h_values, dtype=float)
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * target_sum / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    learning_rate = 0.01
    learning_rate_decay = 0.995
    max_iterations = 500
    time_start = time.time()

    for iteration in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_index = np.argmax(corr)
        if max_index == n_points - 1:
            possible_indices = np.where(corr != corr[max_index])[0]
            if possible_indices.size > 0:
                max_index = possible_indices[0]
            else:
                max_index = n_points - 1
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] = 1.0 - h_current[j]
        h_new = h_current - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        h_new = h_new * target_sum / h_new_sum
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        h_current = h_new
        learning_rate *= learning_rate_decay

    return (np.clip(best_h, 0, 1), best_c5, n_points)
```
