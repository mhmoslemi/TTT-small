        indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        current_c5_perturbed = compute_c5(h_perturbed)
        if current_c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed.copy(), current_c5_perturbed

    if time.time() - start_time < budget_s:
        try:
            coeffs = pywt.wavedec(best_h, 'db4', level=5)
            threshold = np.sqrt(2 * np.log(n_points)) * np.std(coeffs[0])
            coeffs[0] = np.where(np.abs(coeffs[0]) < threshold, 0, coeffs[0])
            h_filtered = pywt.waverec(coeffs, 'db4')
            h_filtered = np.clip(h_filtered, 0, 1)
            h_filtered = h_filtered * target_sum / np.sum(h_filtered)
            current_c5_filtered = compute_c5(h_filtered)
            if current_c5_filtered < best_c5:
                best_h, best_c5 = h_filtered, current_c5_filtered
        except:
            pass

    return (np.clip(best_h, 0, 1), best_c5, n_points)
