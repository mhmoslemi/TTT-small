import numpy as np
import time

def run(seed=42, budget_s=600):
    import numpy as np
    np.random.seed(seed)
    # Assume initial_h_values is available
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate initial guesses
    guesses = []
    # Guess 1: initial_h normalized
    guess1 = normalize(initial_h.copy())
    guesses.append(guess1)
    # Guess 2: reversed initial_h
    guess2 = normalize(np.flip(initial_h.copy()))
    guesses.append(guess2)
    # Guess 3: binary pattern
    binary_pattern = 0.5 + 0.5 * np.power(-1, np.arange(n_points))
    binary_pattern_sum = np.sum(binary_pattern)
    # Scale to target_sum
    binary_pattern = binary_pattern * (target_sum / binary_pattern_sum)
   binary_pattern = np.clip(binary_pattern, 0, 1)
    guesses.append(binary_pattern)
    # Guess 4: sine wave
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    guess4 = normalize(np.clip(sine, 0, 1))
    guesses.append(guess4)
    # Guess 5: random sparse pattern
    random_sparse = np.zeros(n_points)
    random_indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
    random_sparse[random_indices] = 1.0
    sum_sparse = np.sum(random_sparse)
    random_sparse = random_sparse * (target_sum / sum_sparse)
    random_sparse = np.clip(random_sparse, 0, 1)
    guesses.append(random_sparse)

    # Initialize particles
    n_particles = 50
    particles = np.array(guesses[:n_particles])
    velocities = np.random.rand(n_particles, n_points) * 0.1

    # Personal and global best
    personal_best = particles.copy()
    personal_best_fitness = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    global_best = particles[np.argmin(personal_best_fitness)]
    global_best_fitness = np.min(personal_best_fitness)

    start_time = time.time()
    # Parameters for PSO
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            # Generate random numbers
            r1 = np.random.rand()
            r2 = np.random.rand()
            # Update velocity
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            # Update position
            particles[i] += velocities[i]
            # Clip to [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            # Normalize sum
            particles[i] = normalize(particles[i])
            # Compute fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * dx
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
        # Update global best
        current_best_fitness = np.min(personal_best_fitness)
        if current_best_fitness < global_best_fitness:
            global_best = personal_best[np.argmin(personal_best_fitness)]
            global_best_fitness = current_best_fitness

    # Return the best solution
    best_h = global_best
    return (best_h, global_best_fitness, n_points)

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate diverse initial guesses
    guesses = []
    # Raw initial guess
    guesses.append(normalize(initial_h.copy()))
    # Reversed
    guesses.append(normalize(np.flip(initial_h)))
    # Binary alternating
    bin_pat = 0.5 + 0.5 * np.power(-1, np.arange(n_points))
    bin_pat = bin_pat * target_sum / np.sum(bin_pat)
    guesses.append(np.clip(bin_pat, 0, 1))
    # Sine wave
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    guesses.append(normalize(np.clip(sine, 0, 1)))
    # Random sparse
    sparse = np.zeros(n_points)
    idx = np.random.choice(n_points, size=int(n_points//2), replace=False)
    sparse[idx] = 1.0
    sparse = sparse * target_sum / np.sum(sparse)
    guesses.append(np.clip(sparse, 0, 1))
    # 1 - initial
    guesses.append(normalize(1.0 - initial_h.copy()))
    # Random uniform
    guesses.append(normalize(np.random.rand(n_points)))
    # Alternating 0.5
    alt = 0.5 + 0.5 * np.power(-1, np.arange(n_points))
    guesses.append(normalize(alt))
    # Hyperbolic sine
    sinh_x = np.sinh(x)
    sinh_norm = (sinh_x - np.min(sinh_x)) / (np.max(sinh_x) - np.min(sinh_x))
    guesses.append(normalize(0.5 + 0.5 * sinh_norm))
    # Cubic
    cubic = (np.linspace(0, 2, n_points)**3) / 8
    guesses.append(normalize(cubic))

    # Initialize particles and velocities
    n_particles = 50
    particles = np.array(guesses[:n_particles])
    velocities = np.random.rand(n_particles, n_points) * 0.1

    # Track best solutions
    personal_best = particles.copy()
    personal_best_fitness = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    global_best = particles[np.argmin(personal_best_fitness)]
    global_best_fitness = np.min(personal_best_fitness)

    start_time = time.time()
