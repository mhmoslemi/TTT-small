def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    triangle_guess = normalize(triangle_guess)

    square_guess = np.where(x < 1, 1, 0)
    square_guess = normalize(square_guess)
    structured_guesses.append(square_guess)

    random_guess = np.random.rand(n_points)
    random_guess = normalize(random_guess)
    structured_guesses.append(random_guess)

    bounds = [(0, 1)] * n_points
    best_h = initial_guess
    best_c5 = evaluate_correlation(best_h)
    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:

            bounds,

            popsize=15,

            recombination=0.8,
            maxiter=100,
            polish=True,
            seed=seed

            h = res.x
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:

    bounds_slsqp = [(0, 1)] * n_points
