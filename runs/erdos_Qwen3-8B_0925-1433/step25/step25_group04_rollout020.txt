import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    
    perturbations = []
    x = np.linspace(0, 2, n_points)
    
    # Perturbations
    perturbations.append(np.random.permutation(initial_guess))
    perturbations.append(initial_guess + np.random.normal(0, 0.1, n_points))
    perturbations.append(initial_guess * (1 + np.random.normal(0, 0.1, n_points)))
    perturbations.append(0.5 + 0.5 * np.sin(np.pi * (x / 2)))
    perturbations.append(0.5 + 0.5 * (2 * (x / 2) - 1))
    
    perturbations = [normalize(p) for p in perturbations]
    
    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            p,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            tol=1e-6,
            options={'maxiter': 100}
        )
        if result.success:
            h_candidate = normalize(result.x)
            current_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
