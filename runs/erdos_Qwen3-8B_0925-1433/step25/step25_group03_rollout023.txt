import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h):
        return evaluate_c5(h)

    # Generate structured guesses
    x = np.linspace(0, 2, n_points)
    guesses = []
    guess = normalize(np.array(initial_h_values, dtype=float))
    guesses.append(guess)
    guesses.append(normalize(0.5 + 0.5 * np.sin(2 * np.pi * x)))
    guesses.append(normalize(0.5 + 0.5 * np.abs(np.sin(np.pi * x))))
    guesses.append(normalize(np.where(x < 1, 1, 0)))
    guesses.append(normalize(0.5 + 0.5 * np.cos(2 * np.pi * x)))
    guesses.append(normalize(0.5 + 0.5 * (np.sin(np.pi * x) ** 2)))
    guesses.append(normalize(0.5 + 0.5 * np.exp(-0.5 * x)))
    guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * x) * np.cos(np.pi * x)))
    guesses.append(normalize(np.random.rand(n_points)))
    guesses.append(normalize(0.5 + 0.5 * np.sin(4 * np.pi * x)))

    best_h = guesses[0]
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='SLSQP', bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
