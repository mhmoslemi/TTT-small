from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    iter_limit = 300
    for _ in range(iter_limit):
        if time.time() - start_time > budget_s:
            break
        current_corr = np.correlate(best_h, 1 - best_h, mode='full')

            return corr[k] * dx
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
