import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x, k=10):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / k) * np.log(np.sum(np.exp(k * corr))) * dx

    def true_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = true_objective(best_h)

    structured_guesses = [initial_guess]
    x = np.linspace(0, 2, n_points)

    # Additional structured guesses
    structured_guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
    structured_guesses.append(normalize(0.5 + 0.5 * np.abs(2 * (x / 2) - 1)))
    structured_guesses.append(normalize(np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)))

    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = normalize(np.clip(res.x, 0, 1))
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(true_objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = normalize(np.clip(res.x, 0, 1))
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
