import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial guesses
    initial_guesses = []
    initial_guesses.append(np.array(initial_h_values, dtype=float))
    # Binary pattern
    binary = np.zeros(n_points)
    binary[n_points//3:n_points//3*2] = 1.0
    initial_guesses.append(normalize(binary))
    # Sinusoidal
    freq = 1.2
    phase = np.random.rand() * 2 * np.pi
    x = np.linspace(0, 2, n_points)
    sin_wave = 0.4 * np.sin(2 * np.pi * freq * x + phase) + 0.5
    initial_guesses.append(normalize(sin_wave))
    # Random
    initial_guesses.append(np.random.rand(n_points))

    best_h = initial_guesses[0].copy()
    best_c5 = objective(best_h)
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()

    # Differential Evolution optimization
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, bounds, strategy='best1bin', popsize=10, tol=1e-4, maxiter=50)
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    # Local refinement with L-BFGS-B
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
