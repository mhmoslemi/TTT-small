from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h):
        return evaluate_correlation(h)

    def constraint(x):
        return np.sum(x) - target_sum

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    structured_guesses.append(0.5 + 0.5 * np.sin(np.pi * x))
    structured_guesses.append(0.5 + 0.5 * np.abs(np.sin(np.pi * x)))
    structured_guesses.append(0.5 + 0.5 * np.exp(-0.5 * x))
    structured_guesses.append(0.5 + 0.5 * (np.sin(np.pi * x) ** 2))
    structured_guesses.append(0.5 + 0.5 * (np.sin(np.pi * x) ** 3))
    structured_guesses = [normalize(g) for g in structured_guesses]

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(objective, bounds, args=(), strategy='best1bin', popsize=15, mutation=(0.5, 1), recombination=0.8, tol=1e-5, maxiter=50)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
