import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate structured initial guesses
    def make_sine():
        x = np.linspace(0, 2, n_points)
        sine = 0.5 + 0.5 * np.sin(np.pi * x)
        sine = np.clip(sine, 0, 1)
        sine *= target_sum / np.sum(sine)
        return sine
    def make_triangular():
        arr = np.zeros(n_points)
        for i in range(n_points):
            arr[i] = 1.0 if i < n_points // 2 else 0.0
        arr *= target_sum / np.sum(arr)
        return arr
    initial_population = [initial_h_values.copy(), make_sine(), make_triangular()]

    # Normalize initial population
    for i in range(len(initial_population)):
        initial_population[i] = initial_population[i] * target_sum / np.sum(initial_population[i])

    best_h = initial_population[0].copy()
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    population_size = 10
    mutation_rate = 0.1
    generations = 50

    # Genetic Algorithm
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(ind) for ind in initial_population]
        top_indices = np.argsort(fitness)[:int(len(initial_population)*0.2)]
        top_individuals = [initial_population[i] for i in top_indices]
        new_pop = []
        while len(new_pop) < population_size:
            p1, p2 = np.random.choice(top_individuals, 2, replace=False)
            child = (p1 + p2) / 2.0
            for i in range(len(child)):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            child *= target_sum / np.sum(child)
            new_pop.append(child)
        initial_population = new_pop

        for ind in initial_population:
            current_c5 = evaluate_c5(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()

    # SLSQP refinement
    top_candidates = [initial_population[i] for i in np.argsort([evaluate_c5(ind) for ind in initial_population])[:5]]
    for candidate in top_candidates:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        try:
            res = minimize(evaluate_c5, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h = h * target_sum / np.sum(h)
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h
        except:
            pass

    return (np.clip(best_h, 0, 1), best_c5, n_points)
