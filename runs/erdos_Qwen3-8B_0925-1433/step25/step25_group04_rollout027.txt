            for p in range(n_points):
                j_plus = p + k
                if 0 <= j_plus < n_points:
                    grad[p] += (1 - h[j_plus]) * dx
                j_minus = p - k
                if 0 <= j_minus < n_points:
                    grad[p] -= h[j_minus] * dx
        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize(h_new)
        current_c5 = evaluate(h_new)
        if current_c5 < best_c5:
            best_h, best_c5 = h_new.copy(), current_c5
        return best_c5

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        best_c5 = optimize()
        if np.random.rand() < 0.1:
            # Random perturbation
            h_perturb = best_h + 0.1 * np.random.randn(n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = normalize(h_perturb)
            current_c5 = evaluate(h_perturb)
            if current_c5 < best_c5:
                best_h, best_c5 = h_perturb.copy(), current_c5
        if best_c5 < 0.3807:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
