import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)

    population = [initial_h.copy()]
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + perturb
        perturbed = normalize(perturbed)
        population.append(perturbed)
    for _ in range(5):
        rand_h = np.random.rand(n_points)
        rand_h = normalize(rand_h)
        population.append(rand_h)

    start_time = time.time()
    for gen in range(40):
        if time.time() - start_time > budget_s:
            break
        parents = np.random.choice(len(population), size=30, replace=True)
        offspring = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            child = (population[p1] + population[p2]) / 2.0
            mask = np.random.rand(n_points) < 0.05
            child[mask] += np.random.normal(0, 0.05, np.sum(mask))
            child = normalize(child)
            offspring.append(child)
        population += offspring
        evals = [evaluate_c5(h) for h in population]
        best_idx = np.argmin(evals)
        if evals[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx].copy(), evals[best_idx]

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

        if res.success:
