from scipy.optimize import minimize
import time

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x, alpha):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        softmax_val = np.log(np.sum(np.exp(alpha * corr))) / alpha
        return softmax_val * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)

    flipped_guess = 1.0 - initial_h_values

    structured_guesses.append(flipped_guess)

    shifted_guess = np.roll(initial_h_values, 1)
    shifted_guess = normalize(shifted_guess)
    structured_guesses.append(shifted_guess)

    uniform_guess = np.random.rand(n_points)

    structured_guesses.append(uniform_guess)

    start_time = time.time()
    for guess in structured_guesses:

        result = minimize(objective, guess, args=(15,), method='SLSQP',
