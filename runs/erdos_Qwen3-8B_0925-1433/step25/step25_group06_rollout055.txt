import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Initial guess
    h = normalize(np.array(initial_h_values, dtype=float))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    time_start = time.time()
    while time.time() - time_start < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        lag_index = np.argmax(corr)
        # Define objective function
        def objective(x):
            h_opt = x
            corr_val = np.correlate(h_opt, 1 - h_opt, mode='full')[lag_index]
            return corr_val * dx

        # Define constraints
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        # Run optimization
        result = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
