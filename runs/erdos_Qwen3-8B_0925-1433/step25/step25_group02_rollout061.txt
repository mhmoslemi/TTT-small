import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = x * target_sum / np.sum(x)
        return x

        return np.max(corr) * dx

    x = np.linspace(0, 2, n_points)
    harmonic_guess = 0.5 + 0.5 * (np.sin(np.pi * x) + np.cos(np.pi * x)) / np.sqrt(2)
    harmonic_guess = np.clip(harmonic_guess, 0, 1)
    harmonic_guess = normalize(harmonic_guess)

    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = normalize(initial_guess)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()

    best_c5 = evaluate_c5(best_h)

    def optimize_guess(guess):
        result = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            return h, current_c5
        return best_h, best_c5

    result = optimize_guess(initial_guess)
    best_h, best_c5 = result

    result = optimize_guess(harmonic_guess)
    best_h, best_c5 = result

    for _ in range(3):

        result = optimize_guess(best_h)
        best_h, best_c5 = result

    return (np.clip(best_h, 0, 1), best_c5, n_points)
