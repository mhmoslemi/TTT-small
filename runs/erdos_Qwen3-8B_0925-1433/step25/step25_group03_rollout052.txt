import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return x
        x = x * target_sum / total
        return x

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = np.abs(sum_h - target_sum) * 1000
        return c5 + penalty

    x = np.linspace(0, 2, n_points)
    initial_guesses = [
        normalize(np.array(initial_h_values, dtype=float)),
        normalize(0.5 + 0.5 * np.sin(np.pi * x)),
        normalize(0.5 + 0.5 * np.abs(np.sin(np.pi * x))),
        normalize(0.5 + 0.5 * np.exp(-0.5 * x)),
        normalize(np.random.rand(n_points))
    ]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective, bounds, popsize=20, mutation=(0.5, 1), recombination=0.8,
        tol=1e-4, maxiter=100, polish=True, seed=seed
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)

    def objective_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    result_slsqp = minimize(
        objective_slsqp, best_h, method='SLSQP', bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint}, tol=1e-6
    )

    h_opt = result_slsqp.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = normalize(h_opt)
    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx

    return (h_opt, c5_bound, n_points)
