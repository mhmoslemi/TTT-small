import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective_smooth(x, kappa):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))

    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective_smooth, best_h, args=(100,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, max_iter=100)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective_true(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective_true, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, max_iter=50)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective_true(h)
            if current_c5 < best_c5:
