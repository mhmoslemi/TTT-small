import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute correlation to find worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        idx_max = np.argmax(corr)
        # Define objective to minimize the worst lag
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[idx_max] * dx
        # Run optimization to minimize this objective
        result = minimize(objective, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_new = np.clip(result.x, 0, 1)
            h_new = normalize(h_new)
            current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = current_c5
        else:
            break  # or handle failure

    return (np.clip(best_h, 0, 1), best_c5, n_points)
