import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        h -= 0.2 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:

    h_smooth = np.convolve(h, np.ones(5) / 5, mode='same')
