import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    x = np.linspace(0, 2, n_points)

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        softmax_val = np.log(np.sum(np.exp(15.0 * corr))) / 15.0
        return softmax_val * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    structured_guesses = [initial_guess]
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = normalize(np.clip(sine, 0, 1))
    structured_guesses.append(sine)
    triangular = 0.5 - np.abs(x - 1)
    triangular = normalize(np.clip(triangular, 0, 1))
    structured_guesses.append(triangular)
    flipped = 1.0 - initial_h_values
    flipped = normalize(flipped)
    structured_guesses.append(flipped)
    shifted = np.roll(initial_h_values, 1)
    shifted = normalize(shifted)
    structured_guesses.append(shifted)
    uniform = np.random.rand(n_points)
    uniform = normalize(uniform)
    structured_guesses.append(uniform)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
