from scipy.optimize import minimize

def run(seed=42, budget_s=600):

        perturbations.append(np.random.permutation(h))

        perturbations.append(h + np.random.normal(0, 0.1, h.shape))

        sin_h = np.clip(sin_h, 0, 1)
        sin_h = normalize(sin_h)
        perturbations.append(sin_h)
        # Triangular
        tri_h = 0.5 + 0.5 * (2 * (x / 2) - 1)
        tri_h = np.clip(tri_h, 0, 1)
        tri_h = normalize(tri_h)
        perturbations.append(tri_h)
        return [normalize(p) for p in perturbations]
    
    initial_guess = np.array(initial_h_values, dtype=float)
    h = normalize(initial_guess)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    for _ in range(100):  # Max iterations
        if time.time() - start_time > budget_s:

        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr[1:]) * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Perturb and optimize
        perturbations = generate_perturbations(h)
        for p in perturbations:
            x = p.copy()
            # Gradient step
            for _ in range(50):  # Local steps
                corr = np.correlate(x, 1 - x, mode='full')
                max_corr = np.max(corr[1:]) * dx
                if max_corr >= best_c5:
                    break
                # Identify top lags
                lags = np.argsort(corr[1:])[-15:]
                grad = np.zeros(n_points)
                for lag_idx in lags:
                    shift = (lag_idx - (n_points - 1)) * dx
