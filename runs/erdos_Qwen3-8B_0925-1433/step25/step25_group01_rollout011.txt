from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

        x = x * target_sum / np.sum(x)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_patterns = []

    pulse_mask = np.zeros(n_points)
    for i in range(n_points):
        if np.abs(i * dx - 1.0) < 0.1:
            pulse_mask[i] = 1.0
    initial_patterns.append(normalize(pulse_mask))
    # Cluster pattern
    cluster_mask = np.zeros(n_points)
    cluster_size = n_points // 4
    for i in range(0, n_points, cluster_size):
        cluster_mask[i:i+cluster_size] = 1.0
    initial_patterns.append(normalize(cluster_mask))
    # Random pattern
    initial_patterns.append(normalize(np.random.rand(n_points)))
    # Initial guess
    initial_patterns.append(normalize(np.array(initial_h_values, dtype=float)))
