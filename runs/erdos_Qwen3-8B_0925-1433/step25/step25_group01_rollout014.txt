import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones_like}

    # Generate initial population
    population_size = 20
    population = []
    population.append(initial_guess)
    # Pulse pattern
    m_pulses = 5
    spacing = 0.6
    x = np.linspace(0, 2, n_points)
    pulse = np.zeros(n_points)
    for i in range(m_pulses):
        idx = int(i * spacing * n_points)
        pulse[idx] += 1.0
    pulse = normalize(pulse)
    population.append(pulse)
    # Sinusoidal
    f = n_points / 4.0
    phi = np.random.rand() * 2 * np.pi
    sinus = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + phi)
    sinus = normalize(sinus)
    population.append(sinus)
    # Random
    random_guess = np.random.rand(n_points)
    random_guess = normalize(random_guess)
    population.append(random_guess)

    # Fill remaining with random
    for _ in range(population_size - len(population)):
        random = np.random.rand(n_points)
        random = normalize(random)
        population.append(random)

    # GA parameters
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        fitness = [evaluate_c5(ind) for ind in population]
        # Select better individuals
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)

        parent_indices = sorted_indices[:len(population)//2]
        parents = [population[i] for i in parent_indices]
        # Generate new population
        new_population = []

            parent1 = parents[i]
            parent2 = parents[len(parents)//2 + i]
            # Crossover
