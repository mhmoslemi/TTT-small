import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = []
    initial_guesses.append(initial_h_values)
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = np.clip(sine, 0, 1)
    sine = normalize(sine)
    initial_guesses.append(sine)
    triangular = np.zeros(n_points)
    for i in range(n_points):
        triangular[i] = 1.0 if i < n_points // 2 else 0.0
    triangular = normalize(triangular)
    initial_guesses.append(triangular)

    def random_binary():
        bits = np.random.randint(0, 2, size=n_points)
        total = np.sum(bits)
        if total == 0 or total == n_points:
            return np.zeros(n_points)
        scale = target_sum / total
        return np.clip(bits * scale, 0, 1)

    initial_guesses.append(random_binary())
    perturbed = initial_h_values.copy()
    perturbed += np.random.normal(0, 0.15, size=n_points)
    perturbed = normalize(perturbed)
    initial_guesses.append(perturbed)

    population = initial_guesses.copy()
    best_h = np.array(initial_guesses[0])
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(h) for h in population]
        top_indices = np.argsort(fitness)[:10]
        top_candidates = [population[i] for i in top_indices]
        new_population = []
        for _ in range(len(population)):

            child += noise
            child = normalize(child)
            new_population.append(child)
        population = new_population
        for h in population:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:

                best_c5 = current_c5

        if time.time() - start_time > budget_s:
