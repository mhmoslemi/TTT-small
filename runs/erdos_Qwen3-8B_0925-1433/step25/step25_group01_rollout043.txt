            fun=evaluate_c5,
            x0=best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
        )
        if res.success:
            current_h = res.x
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
