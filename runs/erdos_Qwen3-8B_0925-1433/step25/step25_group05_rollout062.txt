import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    # Structured guesses: flat, symmetric, sparse
    flat_guess = np.full(n_points, 0.5)
    structured_guesses.append(flat_guess)
    symmetric_guess = 0.5 + 0.5 * np.cos(np.pi * x)
    symmetric_guess = np.clip(symmetric_guess, 0, 1)
    symmetric_guess = normalize(symmetric_guess)
    structured_guesses.append(symmetric_guess)
    sparse_guess = np.zeros_like(x)
    sparse_guess[::5] = 1.0
    sparse_guess = normalize(sparse_guess)
    structured_guesses.append(sparse_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective_true, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective_true(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective_true, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective_true(h)
            if current_c5 < best_c5:
