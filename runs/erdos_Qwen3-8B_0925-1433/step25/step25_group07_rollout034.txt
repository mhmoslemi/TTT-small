import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        return x * target_sum / np.sum(x) if np.sum(x) > 1e-10 else x

    initial_h = normalize(np.array(initial_h_values, dtype=float))
    num_particles = 50
    particles = []
    for _ in range(num_particles):
        h = initial_h.copy()
        h += 0.1 * np.random.randn(n_points)
        particles.append(normalize(h))
    alt = normalize(np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)]))
    particles.append(alt)
    sine = normalize(np.clip(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)), 0, 1))
    particles.append(sine)
    random_sparse = normalize(np.random.rand(n_points))
    particles.append(random_sparse)
    particles = np.array(particles)
    velocities = np.zeros((len(particles), n_points))
    personal_best = particles.copy()
    global_best = particles[0].copy()
    best_c5 = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 200
    start = time.time()

    for _ in range(max_iter):
        if time.time() - start > budget_s:
            break
        r1 = np.random.rand(len(particles), n_points)
        r2 = np.random.rand(len(particles), n_points)
        for i in range(len(particles)):
            velocities[i] = w * velocities[i] + c1 * r1[i] * (personal_best[i] - particles[i]) + c2 * r2[i] * (global_best - particles[i])
            particles[i] += velocities[i]

        current_c5 = np.array([compute_c5(p) for p in particles])
