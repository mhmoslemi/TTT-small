import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()

    def generate_patterns():
        patterns = []
        x = np.linspace(0, 2, n_points)
        patterns.append(normalize(0.4 * np.sin(2 * np.pi * 1.2 * x) + 0.5))
        patterns.append(normalize(np.zeros_like(x)))
        patterns.append(normalize(np.concatenate([np.ones(n_points//3), np.zeros(n_points//3), np.ones(n_points//3)])))
        patterns.append(np.random.rand(n_points))
        patterns.append(initial_guess)
        return [normalize(p) for p in patterns]

    candidates = generate_patterns()
    for guess in candidates:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(evaluate_c5, bounds, popsize=15, mutation=(0.5, 1.0), recombination=0.8, tol=1e-6, maxiter=100, polish=True)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)

            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
