import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    x = np.linspace(0, 2, n_points)
    h_harmonic = 0.5 + 0.5 * (np.sin(np.pi * x) + np.cos(np.pi * x)) / np.sqrt(2)
    h_harmonic = np.clip(h_harmonic, 0, 1)
    h_harmonic = normalize(h_harmonic)

    structured_guesses = [initial_guess, h_harmonic]
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
