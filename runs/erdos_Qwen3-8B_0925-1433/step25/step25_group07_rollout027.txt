import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    particles = [initial_h.copy()]
    velocities = [np.random.rand(n_points) * 0.1]
    guesses = [
        1.0 - initial_h.copy(),
        np.roll(initial_h, 1).copy(),
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)),
        np.random.rand(n_points),
        np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)]),
    ]
    for g in guesses:
        norm = np.clip(g, 0, 1) * target_sum / np.sum(np.clip(g, 0, 1))
        particles.append(norm)
        velocities.append(np.random.rand(n_points) * 0.1)
    particles = particles[:50]
    velocities = velocities[:50]
    w, c1, c2 = 0.8, 1.5, 1.5
    best_h = particles[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (particles[i] - best_h) + c2 * r2 * (best_h - particles[i])
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            total = np.sum(new_pos)
            if total == 0:
                new_pos = np.zeros_like(new_pos)
            else:
                new_pos = new_pos * target_sum / total
            particles[i] = new_pos
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = particles[i].copy()
                best_c5 = current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
