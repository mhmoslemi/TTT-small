import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    # Generate structured initial guesses
    square_guess = np.zeros(n_points)
    for i in range(n_points):
        if x[i] < 1.0:
            square_guess[i] = 0.5
        else:
            square_guess[i] = 1.0
    structured_guesses.append(normalize(square_guess))

    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    structured_guesses.append(normalize(triangle_guess))

    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    structured_guesses.append(normalize(sine_guess))

    exp_guess = 0.5 + 0.5 * np.exp(-0.5 * x)
    structured_guesses.append(normalize(exp_guess))

    perm_guess = initial_h_values.copy()
    np.random.shuffle(perm_guess)
    structured_guesses.append(normalize(perm_guess))

    poly_guess = 0.5 + 0.5 * (x**2 - x) / (x**2 + 1)
    structured_guesses.append(normalize(poly_guess))

    best_h = initial_h_values.copy()
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    candidates = []

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0.0, 1.0)] * n_points
        result = differential_evolution(
            lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx + 1e4 * (np.sum(h) - target_sum)**2,
            bounds=bounds, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False
        )
        if result.success:
            h_candidate = np.clip(result.x, 0, 1)
            h_candidate = normalize(h_candidate)
            candidates.append(h_candidate)

    for candidate in candidates:
        if time.time() - start_time > budget_s:

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
