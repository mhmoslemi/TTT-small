import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()

    def generate_initial_guesses():
        guesses = [initial_guess]
        x = np.linspace(0, 2, n_points)
        phase = np.random.rand() * 2 * np.pi
        # Sinusoidal with different frequencies
        freqs = [0.5, 0.75, 1.0, 1.5]
        for f in freqs:
            sin_wave = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + phase)
            guesses.append(normalize(sin_wave))
        # Binary patterns
        binary1 = np.zeros(n_points)
        binary1[n_points//3:2*n_points//3] = 1.0
        guesses.append(normalize(binary1))
        binary2 = np.zeros(n_points)
        binary2[n_points//4:3*n_points//4] = 1.0
        guesses.append(normalize(binary2))
        # Pulse clusters with different numbers of pulses
        for m_pulses in [4, 5, 6]:

            guesses.append(normalize(h_pulse))
        # Random perturbation

        guesses.append(normalize(rand))
        return guesses
