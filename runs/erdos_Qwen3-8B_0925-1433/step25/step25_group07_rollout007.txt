import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def normalize(x):

        x = x * target_sum / np.sum(x)
        return x

    x = np.linspace(0, 2, n_points)
    guesses = []
    guesses.append(normalize(np.array(initial_h_values, dtype=float)))
    guesses.append(normalize(1.0 - np.array(initial_h_values, dtype=float)))
    guesses.append(normalize(np.roll(initial_h_values, 1)))
    guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
    guesses.append(normalize(0.5 + 0.5 * np.cos(np.pi * x)))
    guesses.append(normalize(np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])))
    guesses.append(normalize(np.array([0.8 if i % 3 == 0 else 0.2 for i in range(n_points)])))
    guesses.append(normalize(np.array([0.2 if i % 3 == 0 else 0.8 for i in range(n_points)])))
    guesses.append(normalize(np.random.rand(n_points)))
    guesses.append(normalize(np.array([0.5 + 0.5 * np.sin(2 * np.pi * i / n_points) for i in range(n_points)])))
    guesses.append(normalize(0.5 + 0.5 * np.sin(2 * np.pi * x / 3)))

    best_h = guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, kappa=15):
        corr = np.correlate(x, 1 - x, mode='full')
        exp_vals = np.exp(-kappa * corr)
        return np.log(np.sum(exp_vals)) / kappa

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(smooth_objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
