import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        softmax_val = np.log(np.sum(np.exp(10.0 * corr))) / 10.0
        return softmax_val * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)

    flipped_guess = 1.0 - initial_h_values
    flipped_guess = normalize(flipped_guess)
    structured_guesses.append(flipped_guess)

    shifted_guess = np.roll(initial_h_values, 1)
    shifted_guess = normalize(shifted_guess)
    structured_guesses.append(shifted_guess)

    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    structured_guesses.append(uniform_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
