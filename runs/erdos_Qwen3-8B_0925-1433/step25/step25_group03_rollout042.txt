import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    # Square wave
    square = np.zeros(n_points)
    for i in range(n_points):
        if 0 <= x[i] < 0.5 or 1.5 <= x[i] < 2:
            square[i] = 1
    structured_guesses.append(normalize(square))

    # Triangular wave
    triangle = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    structured_guesses.append(normalize(triangle))

    # Sine with harmonics
    sine = 0.5 + 0.5 * np.sin(np.pi * x) + 0.25 * np.sin(2 * np.pi * x)
    structured_guesses.append(normalize(sine))

    # Exponential decay
    exponential = 0.5 + 0.5 * np.exp(-x)
    structured_guesses.append(normalize(exponential))

    # Random permutation of initial guess
    perm = initial_h_values.copy()
    np.random.shuffle(perm)
    structured_guesses.append(normalize(np.array(perm, dtype=float)))

    # Polynomial
    poly = 0.5 + 0.25 * (x / 2) ** 2
    structured_guesses.append(normalize(poly))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_correlation(best_h)
    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        def objective(h):
            sum_h = np.sum(h)
            penalty = 1e6 * abs(sum_h - target_sum)
            return evaluate_correlation(h) + penalty
        result = differential_evolution(
            objective, bounds=[(0, 1)] * n_points,
