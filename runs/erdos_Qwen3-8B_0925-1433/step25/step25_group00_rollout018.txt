        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    h_fft = np.fft.fft(best_h)
    window = np.blackman(len(h_fft))
    filtered_fft = h_fft * window
    h_filtered = np.fft.ifft(filtered_fft).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered)
    final_c5 = np.max(np.correlate(h_filtered, 1 - h_filtered, mode='full')) * dx
    if final_c5 < best_c5:
        best_h, best_c5 = h_filtered, final_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
