        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        k_idx = np.argmax(corr)
        k = k_idx - (n_points - 1)
        j = (np.arange(n_points) + k) % n_points
        m = (np.arange(n_points) - k) % n_points
        gradient = (1 - h[j]) - h[m]
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros_like(h)
            h[0] = target_sum
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (np.clip(best_h, 0, 1), final_c5, n_points)
