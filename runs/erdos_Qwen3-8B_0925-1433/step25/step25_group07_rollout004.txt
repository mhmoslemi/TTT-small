import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        if np.isclose(np.sum(x), target_sum, atol=1e-6):
            return x
        scale = target_sum / np.sum(x)
        return x * scale

    initial = normalize(np.array(initial_h_values, dtype=float))
    pop = []
    pop.append(initial)
    pop.append(normalize(1.0 - np.array(initial_h_values, dtype=float)))
    shifted = np.roll(initial_h_values, 1)
    pop.append(normalize(shifted))
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    pop.append(normalize(np.clip(sine, 0, 1)))
    sparse = np.zeros(n_points)
    for _ in range(3):
        idx = np.random.choice(n_points, size=n_points//2, replace=False)
        sparse[idx] = 1
    pop.append(normalize(sparse))
    random_guess = np.random.rand(n_points)
    pop.append(normalize(random_guess))
    alt = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
    pop.append(normalize(alt))
    pop = [np.clip(x, 0, 1) for x in pop]

    best_h = initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    population = [normalize(np.clip(x, 0, 1)) for x in pop]
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        idx = np.argsort(fitness)
        best_idx = idx[0]
        current_best = population[best_idx].copy()
        if current_best is None:
            continue
        if fitness[best_idx] < best_c5:
            best_h, best_c5 = current_best, fitness[best_idx]
        new_pop = []
        for _ in range(len(population) // 2):
            a, b = np.random.choice(len(population), size=2, replace=False)
            parent1 = population[a]
            parent2 = population[b]
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            child1 = normalize(np.clip(child1, 0, 1))
            child2 = normalize(np.clip(child2, 0, 1))
            new_pop.append(child1)
            new_pop.append(child2)
        for _ in range(len(population) - len(new_pop)):
            vec = np.random.rand(n_points)
            vec = normalize(np.clip(vec, 0, 1))
            new_pop.append(vec)
        population = new_pop
    return (np.clip(best_h, 0, 1), best_c5, n_points)
