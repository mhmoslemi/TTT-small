from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        return max_corr + penalty

    structured = []
    structured.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
    structured.append(normalize(0.5 + 0.5 * np.abs(np.sin(np.pi * x))))
    structured.append(normalize(0.5 + 0.5 * np.sign(np.sin(np.pi * x))))
    structured.append(normalize(0.5 + 0.5 * np.exp(-x)))
    structured.append(normalize(np.random.permutation(initial_h_values)))
    structured.append(normalize(0.5 + 0.5 * (x**2 - x)))

    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = np.inf

            tol=1e-6, mutation=(0.5, 1), recombination=0.9
