import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def generate_initial_guesses():
        guesses = [initial_h_values]
        # Sine wave
        x = np.linspace(0, 2, n_points)
        sine = 0.5 + 0.4 * np.sin(2 * np.pi * 1.2 * x)
        guesses.append(sine)
        # Pulse pattern
        pulse = np.zeros(n_points)
        step = n_points // 4
        pulse[::step] = 1.0
        guesses.append(pulse)
        # Cluster pattern
        cluster = np.zeros(n_points)
        cluster[0:n_points//3] = 1.0
        cluster[n_points//3*2: n_points//3*3] = 1.0
        guesses.append(cluster)
        # Random pattern
        random_pattern = np.random.rand(n_points)
        guesses.append(random_pattern)
        # Alternating pattern
        alt = np.zeros(n_points)
        alt[::2] = 1.0
        guesses.append(alt)
        # Normalize all guesses
        return [normalize(g) for g in guesses]

    start_time = time.time()
    guesses = generate_initial_guesses()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        # Run SLSQP
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}
        res = minimize(evaluate_c5, guess, method='SLSQP', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Run L-BFGS-B
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=Bounds(0, 1), tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
