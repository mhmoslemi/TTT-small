import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective_smooth(x, kappa):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))

    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_guess(base, noise_scale=0.15, freq=20):
        x = np.linspace(0, 2, n_points)
        noise = np.random.normal(0, noise_scale, n_points)
        if freq == 0:
            guess = base + noise
        else:
            guess = base + np.sin(np.pi * x * freq) * noise_scale * 0.5
        return normalize(guess)

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess
    best_c5 = objective_true(best_h)

    structured_guesses.append(generate_guess(initial_guess, 0.15, 3))   # lower freq
    structured_guesses.append(generate_guess(initial_guess, 0.15, 0))   # pure noise
    structured_guesses.append(generate_guess(initial_guess, 0.15, 20))  # high freq

    start_time = time.time()
