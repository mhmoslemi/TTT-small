            best_h = h_new
            best_c5 = new_c5
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 20:
            break
        learning_rate *= learning_decay

    return (np.clip(best_h, 0, 1), best_c5, n_points)
