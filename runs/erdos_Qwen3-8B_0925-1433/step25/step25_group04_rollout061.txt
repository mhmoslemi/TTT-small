import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    initial_h = normalize(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    perturbations = []
    x = np.linspace(0, 2, n_points)
    
    # Permutation perturbation
    perm_h = np.random.permutation(initial_h)
    perm_h = normalize(perm_h)
    perturbations.append(perm_h)
    
    # Noise perturbation
    noise_h = initial_h + np.random.normal(0, 0.15, n_points)
    noise_h = normalize(np.clip(noise_h, 0, 1))
    perturbations.append(noise_h)
    
    # Sinusoidal perturbation
    sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
    sin_h = normalize(np.clip(sin_h, 0, 1))
    perturbations.append(sin_h)

    start_time = time.time()
    for pert in perturbations:
        h = pert.copy()
        for _ in range(300):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr * dx)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
            
            lags = np.argsort(corr)[::-1][:10]
            grads = np.zeros(n_points)
            for k in lags:
                for i in range(n_points):
                    j = i + k
                    if 0 <= j < n_points:
                        grads[i] += 1.0 - h[j]
            
            sigma = 0.1
            h_new = h - sigma * grads
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize(h_new)
            h = h_new

    return (best_h, best_c5, n_points)
