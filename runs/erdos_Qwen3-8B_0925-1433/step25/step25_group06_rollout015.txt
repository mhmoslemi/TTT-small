import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    h = np.array(initial_h_values, dtype=float)
    h = normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    time_used = 0.0

    while time_used < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        def obj(x):
            corr_val = np.correlate(x, 1 - x, mode='full')[k_max] * dx
            return corr_val
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_opt = result.x
            h_opt = normalize(h_opt)
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5_opt = np.max(corr_opt) * dx
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
            h = h_opt.copy()
        time_used = time.time() - time.time() + budget_s
        time_used = min(time_used, budget_s)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
