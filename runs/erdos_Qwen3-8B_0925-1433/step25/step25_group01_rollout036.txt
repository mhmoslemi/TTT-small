import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = normalize(np.clip(x, 0, 1))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones_like}

    start_time = time.time()
    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = evaluate_c5(best_h)

    def generate_guesses():
        guesses = []
        # Pulse clusters
        for m_pulses in [5, 4, 6]:
            h = np.zeros(n_points)
            for i in range(m_pulses):
                idx = int(i * n_points / m_pulses)
                h[idx] = 1.0
            guesses.append(normalize(h))
        # Sinusoidal
        for freq in [1.5, 2.0, 2.5]:
            phase = np.random.rand() * 2 * np.pi
            h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points) + phase)
            guesses.append(normalize(h))
        # Perturbations
        h = np.array(initial_h_values, dtype=float)
        h += np.random.normal(0, 0.05, n_points)
        guesses.append(normalize(np.clip(h, 0, 1)))
        return guesses

    for guess in generate_guesses():
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = normalize(np.clip(res.x, 0, 1))
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = normalize(np.clip(res.x, 0, 1))
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
