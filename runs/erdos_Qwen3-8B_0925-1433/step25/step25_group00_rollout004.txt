        current_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if i % 20 == 0 and current_c5 >= best_c5:
            lambda_tv *= 1.2

    # Phase 3: Adaptive Sampling
    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        perturb_mask = np.random.rand(n_points) < 0.05
        h_perturb = h.copy()
        h_perturb[perturb_mask] += np.random.uniform(-0.1, 0.1, size=np.sum(perturb_mask))
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        current_c5 = np.max(correlate(h_perturb, 1 - h_perturb, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h_perturb.copy(), current_c5

    # Phase 4: Wavelet Refinement
    if n_points > 2:
        coeffs = dwt(best_h, 'db4', level=5)
        threshold = np.std(coeffs[1]) * 0.5
        coeffs[1] = np.sign(coeffs[1]) * np.maximum(np.abs(coeffs[1]) - threshold, 0)
        h_filtered = idwt(coeffs, 'db4')
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * target_sum / np.sum(h_filtered)
        final_c5 = np.max(correlate(h_filtered, 1 - h_filtered, mode='full')) * dx
        if final_c5 < best_c5:
            best_h, best_c5 = h_filtered, final_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
