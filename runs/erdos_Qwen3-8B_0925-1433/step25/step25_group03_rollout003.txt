    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_total = np.sum(h_clipped)
        if h_total <= 1e-12:
            return float('inf')
        h_norm = h_clipped * (target_sum / h_total)
        overlap = np.max(np.correlate(h_norm, 1.0 - h_norm, mode='full')) * dx
        return overlap

    x = np.linspace(0, 2, n_points)
    structured_guesses = []
    # Generate structured patterns
    structured_guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
    structured_guesses.append(normalize(0.5 + 0.5 * np.abs(np.sin(np.pi * x))))
    structured_guesses.append(normalize(0.5 + 0.5 * np.exp(-x)))
    structured_guesses.append(normalize(0.5 + 0.5 * np.sqrt(np.clip(x - 1, 0, 1))))
    structured_guesses.append(normalize(np.array(initial_h_values, dtype=float)))
    structured_guesses.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf

    start_time = time.time()
    for idx, guess in enumerate(structured_guesses):
        if time.time() - start_time > budget_s:
            break
        bounds = [(0.0, 1.0)] * n_points
        result = differential_evolution(objective, bounds, strategy='best1bin',
                                       popsize=15, maxiter=100, tol=1e-6,
                                       mutation=(0.6, 0.9), recombination=0.8,
                                       seed=seed + idx)
        if result.success:
            h = result.x
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    # Local refinement with SLSQP
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        def slsqp_obj(h):
            return objective(h)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = minimize(slsqp_obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=cons, tol=1e-8)
        if res.success:
            h = res.x
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
