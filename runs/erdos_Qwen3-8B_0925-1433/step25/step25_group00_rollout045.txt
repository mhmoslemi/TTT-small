from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):

            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    h = h * target_sum / np.sum(h)

        h -= 0.1 * grad(h)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
