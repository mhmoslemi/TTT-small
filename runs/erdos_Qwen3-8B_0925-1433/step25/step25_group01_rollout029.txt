import time

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def evaluate_c5(x):
        x = normalize(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}
    
    start_time = time.time()
    
    def generate_patterns():
        patterns = []
        x = np.linspace(0, 2, n_points)
        # Pulse clusters

            h = np.zeros(n_points)

                idx = int(i * spacing * n_points)
                h[idx] = 1.0
            patterns.append(normalize(h))
        # Sinusoidal
        freqs = [n_points / 3, n_points / 2.5, n_points / 1.8]

        return patterns
    
    for guess in generate_patterns():
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
