import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h):
        return evaluate_correlation(h)

    def constraint(h):
        return np.sum(h) - target_sum

    def bounds():
        return [(0, 1)] * n_points

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    # Generate structured patterns
    x = np.linspace(0, 2, n_points)
    structured_guesses = []
    structured_guesses.append(normalize(0.5 * (1 + np.sin(np.pi * x))))

    start_time = time.time()

        result = differential_evolution(objective, bounds(), strategy='best1bin',

        if result.success:

            current_c5 = evaluate_correlation(h)
