    noise_guess = normalize(noise_guess)
    structured_guesses.append(noise_guess)

    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            smooth_objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            true_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
