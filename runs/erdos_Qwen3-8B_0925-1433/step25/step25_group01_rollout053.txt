    x = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
    h_sinusoidal = normalize(h_sinusoidal)
    initial_patterns.append(h_sinusoidal)
    
    # Perturbed initial
    h_perturb = initial_h_values + np.random.normal(0, 0.05, n_points)
    h_perturb = np.clip(h_perturb, 0, 1)
    h_perturb = normalize(h_perturb)
    initial_patterns.append(h_perturb)
    
    # Binary pulse pattern
    h_binary = np.zeros(n_points)
    h_binary[n_points//3:n_points//3*2] = 1.0
    h_binary = normalize(h_binary)
    initial_patterns.append(h_binary)
    
    # Initial guess
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    initial_patterns.append(initial_guess)
    
    best_h = initial_patterns[0].copy()
    best_c5 = evaluate_c5(best_h)
    
    start_time = time.time()
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    for pattern in initial_patterns:
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, pattern, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            candidate_h = res.x
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = normalize(candidate_h)
            current_c5 = evaluate_c5(candidate_h)
            if current_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            candidate_h = res.x
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = normalize(candidate_h)
            current_c5 = evaluate_c5(candidate_h)
            if current_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
