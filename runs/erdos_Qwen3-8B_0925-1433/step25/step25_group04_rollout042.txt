            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
            alpha = 0.1
            h_new = h - alpha * grad
            h_new = normalize(h_new)
            new_corr = np.correlate(h_new, 1 - h_new, mode='full')
            new_max = np.max(new_corr) * dx
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h_new.copy()
            h = h_new.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
