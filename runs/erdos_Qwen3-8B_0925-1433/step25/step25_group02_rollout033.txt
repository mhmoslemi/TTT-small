import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(ind):
        ind = np.clip(ind, 0, 1)
        if np.sum(ind) == 0:
            return np.zeros_like(ind)
        ind = ind * target_sum / np.sum(ind)
        return ind

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = normalize(sine_guess)
    initial_guesses.append(sine_guess)
    triangular_guess = np.zeros(n_points)
    triangular_guess[:n_points//2] = 1.0
    triangular_guess = normalize(triangular_guess)
    initial_guesses.append(triangular_guess)
    rand_guess = np.random.rand(n_points)
    rand_guess = normalize(rand_guess)
    initial_guesses.append(rand_guess)

    population = initial_guesses.copy()

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break

        best_idx = np.argmin(fits)
