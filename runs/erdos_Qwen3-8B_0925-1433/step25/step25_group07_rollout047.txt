import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate diverse initial guesses
    guesses = []
    base = np.array(initial_h_values, dtype=float)
    guesses.append(normalize(base))
    guesses.append(normalize(1.0 - base))
    
    # Shifted and inverse
    shifted = np.roll(base, 1)
    guesses.append(normalize(shifted))
    guesses.append(normalize(1.0 - shifted))
    
    # Sinusoidal and inverted
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    guesses.append(normalize(np.clip(sine, 0, 1)))
    guesses.append(normalize(1.0 - np.clip(sine, 0, 1)))
    
    # Random sparse patterns
    for _ in range(3):
        sparse = np.random.choice([0, 1], size=n_points, p=[0.8, 0.2])
        guesses.append(normalize(sparse))
    
    # Alternating pattern
    alt = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
    guesses.append(normalize(alt))
    guesses.append(normalize(1.0 - alt))
    
    # Random perturbations
    for _ in range(3):
        perturbed = base + 0.1 * np.random.randn(n_points)
        guesses.append(normalize(np.clip(perturbed, 0, 1)))
    
    best_h = guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, kappa=20):
        corr = np.correlate(x, 1 - x, mode='full')
        exp_vals = np.exp(-kappa * corr)
        return np.log(np.sum(exp_vals)) / kappa

    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(smooth_objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(smooth_objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
