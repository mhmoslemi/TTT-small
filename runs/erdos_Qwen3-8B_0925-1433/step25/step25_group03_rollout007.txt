import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_normalized = h_clipped * target_sum / sum_h
        return evaluate_correlation(h_normalized)

    x = np.linspace(0, 2, n_points)

    # Generate structured initial guesses
    structured_guesses = []
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = sine_guess * target_sum / np.sum(sine_guess)
    structured_guesses.append(sine_guess)
    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    triangle_guess = np.clip(triangle_guess, 0, 1)
    triangle_guess = triangle_guess * target_sum / np.sum(triangle_guess)
    structured_guesses.append(triangle_guess)
    square_guess = np.zeros(n_points)
    square_guess[:int(n_points/2)] = 1.0
    square_guess[int(n_points/2):] = 0.0
    square_guess = square_guess * target_sum / np.sum(square_guess)
    structured_guesses.append(square_guess)
    perturbed_guess = np.array(initial_h_values, dtype=float)
    perturbed_guess += np.random.normal(0, 0.05, n_points)
    perturbed_guess = np.clip(perturbed_guess, 0, 1)
    perturbed_guess = perturbed_guess * target_sum / np.sum(perturbed_guess)
    structured_guesses.append(perturbed_guess)

    # Best solution so far
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_correlation(best_h)

    # Differential evolution
    bounds = [(0, 1)] * n_points
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, maxiter=50, tol=1e-6,

        polish=True, disp=False

    if result.success:
        best_h = result.x
        best_c5 = evaluate_correlation(best_h)

    # SLSQP refinement
    def slsqp_objective(h):
        return evaluate_correlation(h)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_slsqp = [(0, 1)] * n_points
    result_slsqp = minimize(slsqp_objective, best_h, method='SLSQP',

    if result_slsqp.success:

        best_c5 = result_slsqp.fun

    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-5:

    best_c5 = evaluate_correlation(best_h)

    return (best_h, best_c5, n_points)
