import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, std=0.05):
        h = np.clip(h + np.random.normal(0, std, n_points), 0, 1)
        return normalize(h)

    def crossover(h1, h2):
        split = np.random.randint(1, n_points - 1)
        return np.concatenate([h1[:split], h2[split:]])

    def generate_population():
        pop = []
        base = normalize(np.array(initial_h_values, dtype=float))
        pop.append(base)
        pop.append(normalize(1.0 - np.array(initial_h_values, dtype=float)))
        pop.append(normalize(np.roll(initial_h_values, 1)))
        x = np.linspace(0, 2, n_points)
        sine = 0.5 + 0.5 * np.sin(np.pi * x)
        pop.append(normalize(np.clip(sine, 0, 1)))
        pop.append(normalize(np.random.rand(n_points)))
        alt = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
        pop.append(normalize(alt))
        pop.append(normalize(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])))
        return pop

    population = generate_population()
    best_h = population[0]
    best_c5 = eval_c5(best_h)

    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [eval_c5(h) for h in population]
        idx = np.argsort(fitness)
        selected = [population[i] for i in idx[:10]]
        new_pop = []
        while len(new_pop) < len(population):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = crossover(p1, p2)
            new_pop.append(mutate(child))
        population = new_pop
        current_c5 = eval_c5(population[0])
        if current_c5 < best_c5:
            best_h, best_c5 = population[0].copy(), current_c5

    def refine(h):
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return normalize(np.clip(result.x, 0, 1))

    best_h = refine(best_h)
    best_c5 = eval_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
