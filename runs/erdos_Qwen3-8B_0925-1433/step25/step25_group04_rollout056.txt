import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, k=30):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        smooth = (1 / k) * np.log(np.sum(np.exp(-k * corr)))
        return smooth * dx

    def true_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    
    # Parent solution
    structured_guesses.append(initial_guess)
    
    # Square wave guess
    square_guess = np.where(x < 1, 1, 0)
    square_guess = normalize(square_guess)
    structured_guesses.append(square_guess)
    
    # Sinusoidal guess
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    
    # Triangle guess
    triangle_guess = 1 - np.abs(x - 1)
    triangle_guess = np.clip(triangle_guess, 0, 1)
    triangle_guess = normalize(triangle_guess)
    structured_guesses.append(triangle_guess)
    
    # Randomized parent
    randomized_guess = initial_guess + 0.1 * np.random.randn(n_points)
    randomized_guess = np.clip(randomized_guess, 0, 1)
    randomized_guess = normalize(randomized_guess)
    structured_guesses.append(randomized_guess)
    
    start_time = time.time()
    
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            lambda x: smooth_objective(x, 10),
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            true_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
