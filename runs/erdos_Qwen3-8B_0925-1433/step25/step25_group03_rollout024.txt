import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)

    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    # Sine wave
    structured_guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
    # Triangle wave
    structured_guesses.append(normalize(0.5 + 0.5 * np.abs(np.sin(np.pi * x))))
    # Exponential decay
    structured_guesses.append(normalize(0.5 + 0.5 * np.exp(-0.5 * x)))
    # Random perturbation
    structured_guesses.append(normalize(np.random.rand(n_points)))
    # Initial with flips
    modified_guess = initial_guess.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    structured_guesses.append(normalize(modified_guess))
    # Segment-based
    structured_guesses.append(normalize(np.array([1.0 if i % 3 == 0 else 0.0 for i in range(n_points)])))

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_correlation, guess, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
