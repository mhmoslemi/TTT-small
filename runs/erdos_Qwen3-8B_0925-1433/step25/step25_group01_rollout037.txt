import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = []
    # Pulse cluster
    h_p = np.zeros(n_points)
    for i in range(4):  # 4 pulses
        idx = int(i * n_points / 8)
        h_p[idx] = 1
    h_p = h_p * target_sum / np.sum(h_p)
    initial_guesses.append(h_p)

    # Sine wave
    x = np.linspace(0, 2, n_points)
    h_s = 0.5 + 0.5 * np.sin(2 * np.pi * 1.2 * x + np.random.rand() * 2 * np.pi)
    h_s = h_s * target_sum / np.sum(h_s)
    initial_guesses.append(h_s)

    # Perturb initial
    h_r = initial_h_values + np.random.normal(0, 0.05, n_points)
    h_r = h_r * target_sum / np.sum(h_r)
    initial_guesses.append(h_r)

    # Random
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)
    initial_guesses.append(h_rand)

    best_h = initial_h_values.copy()
    best_c5 = evaluate_c5(initial_h_values)
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0, 1)] * n_points
        res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_candidate = res.x
            current_c5 = evaluate_c5(h_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
