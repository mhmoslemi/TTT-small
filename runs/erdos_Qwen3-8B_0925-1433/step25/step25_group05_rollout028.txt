import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective_smooth(x, lambda_val):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx + (1 / lambda_val) * np.log(np.sum(np.exp(-lambda_val * corr)))

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = objective_true(best_h)
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    perturbations = [
        lambda x: x + 0.05 * np.random.randn(len(x)),
        lambda x: 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, len(x))),
        lambda x: 0.5 + 0.5 * np.cos(np.pi * np.linspace(0, 2, len(x))),
        lambda x: 0.5 + 0.2 * (np.linspace(0, 2, len(x)) / 2) + 0.3 * (np.linspace(0, 2, len(x)) / 2)**2,
        lambda x: np.random.dirichlet(np.ones(len(x)), size=1)[0]
    ]

    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        guess = normalize(p(initial_guess))
        result = minimize(objective_smooth, guess, args=(50,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective_true(h)
            if current_c5 < best_c5:
