import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_initial_population():
        pop = [np.array(initial_h_values, dtype=float)]
        x = np.linspace(0, 2, n_points)
        pop.append(normalize(0.5 + 0.5 * np.sin(np.pi * x)))
        tri = np.zeros(n_points)
        tri[:n_points//2] = 1.0
        pop.append(normalize(tri))
        for _ in range(3):
            pop.append(normalize(np.random.rand(n_points)))
        for _ in range(2):
            pert = np.random.normal(0, 0.1, n_points)
            pop.append(normalize(initial_h_values + pert))
        return pop

    population = generate_initial_population()
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        candidates = [(evaluate_c5(h), h) for h in population]
        candidates.sort()
        parents = [c[1] for c in candidates[:len(population)//2]]
        new_pop = []
        for _ in range(len(population)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.15:
                child += np.random.normal(0, 0.1, n_points)
            new_pop.append(normalize(child))
        population = new_pop
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    top_candidates = [(evaluate_c5(h), h) for h in population]
