import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x, scale=50):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / scale) * np.log(np.sum(np.exp(-scale * corr)))

    def evaluate(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    initial = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial.copy()
    best_c5 = evaluate(best_h)

    guesses = [initial]
    x = np.linspace(0, 2, n_points)
    for sig in [0.05, 0.1, 0.15]:
        perturb = np.random.normal(0, sig, n_points)
        guesses.append(normalize(initial_h_values + perturb))
    for sig in [0.2, 0.25]:
        perturb = np.random.uniform(-sig, sig, n_points)
        guesses.append(normalize(initial_h_values + perturb))

    start = time.time()
    for guess in guesses:
        if time.time() - start > budget_s:
            break
        res = minimize(objective, guess, args=(50,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            curr_c5 = evaluate(h)
            if curr_c5 < best_c5:
                best_h, best_c5 = h, curr_c5

    for _ in range(3):
        if time.time() - start > budget_s:
            break
        res = minimize(evaluate, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            curr_c5 = evaluate(h)
            if curr_c5 < best_c5:
                best_h, best_c5 = h, curr_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
