import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    def generate_initial_population():
        x = np.linspace(0, 2, n_points)
        pulse = np.zeros(n_points)
        m_pulses = min(5, n_points // 4)
        spacing = 2.0 / (m_pulses * 2)
        for i in range(m_pulses):
            idx = int(i * spacing * n_points)
            pulse[idx] = 1.0
        pulse = normalize(pulse)
        sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * (n_points / 4.0) * x + np.random.rand() * 2 * np.pi)
        sinusoidal = normalize(sinusoidal)
        random = np.random.rand(n_points)
        random = normalize(random)
        return [initial_guess, pulse, sinusoidal, random]

    population = generate_initial_population()
    population.extend([np.random.rand(n_points) for _ in range(6)])
    population = [normalize(np.clip(p, 0, 1)) for p in population]
    population = np.array(population)

    start_time = time.time()
    population_size = 20
    generations = 50

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate_c5(p) for p in population])
        elite_idx = np.argsort(fitness)[:population_size // 2]
        elites = population[elite_idx]

        new_population = []
        for _ in range(population_size // 2):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(elites, 2, replace=False)
                child = (p1 + np.random.rand() * (p2 - p1)) * 0.9
            else:
                p = np.random.choice(elites)
                child = p * 0.9 + np.random.randn(n_points) * 0.1
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        population = np.vstack([elites, new_population])
        population = np.random.permutation(population)[:population_size]

    best_h = population[np.argmin(evaluate_c5(population))]

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}, tol=1e-6)
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize(best_h)
            best_c5 = evaluate_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
