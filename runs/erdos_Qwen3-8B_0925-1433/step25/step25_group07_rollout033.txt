from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def normalize(x):

        x = x * target_sum / np.sum(x)
        return x

    guesses = [
        normalize(base),
        normalize(1.0 - base),
        normalize(np.roll(base, 1)),
        normalize(0.5 + 0.5 * np.sin(np.linspace(0, np.pi, n_points))),
        normalize(np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])),
        normalize(np.random.rand(n_points)),
        normalize(np.concatenate([np.ones(10), np.zeros(n_points-10)])),
        normalize(np.concatenate([np.zeros(10), np.ones(n_points-10)])),
    ]

    best_h = guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x, kappa):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.log(np.sum(np.exp(-kappa * corr))) / kappa

    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res1 = minimize(lambda x: smooth_objective(x, 100), guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res1.success:

            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

                best_h, best_c5 = h.copy(), current_c5
        res2 = minimize(lambda x: smooth_objective(x, 50), guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res2.success:
            h = normalize(np.clip(res2.x, 0, 1))
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
