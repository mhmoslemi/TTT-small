import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        x_normalized = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr) * dx

    def evaluate(x):
        x_clipped = np.clip(x, 0, 1)
        x_normalized = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr) * dx

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)
    phase = np.random.rand() * 2 * np.pi
    freq = 1.2
    initial_guesses.append(0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase))
    initial_guesses.append(np.random.rand(n_points))

    best_h = initial_guesses[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = differential_evolution(objective, bounds, strategy='best1bin', 
                                        popsize=15, mutation=(0.5, 1), recombination=0.9, 
                                        maxiter=100, tol=1e-6, disp=False)

                best_h = res.x

        except:

        best_c5 = evaluate(best_h)
