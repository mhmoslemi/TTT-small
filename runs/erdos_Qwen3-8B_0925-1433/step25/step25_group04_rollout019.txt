def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # Perturbations
    def perturb(h):
        p = h.copy()
        p = np.random.permutation(p)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        return p
    
    x = np.linspace(0, 2, n_points)
    def sin_mod(h):
        s = 0.5 + 0.5 * np.sin(np.pi * x)
        s = np.clip(s, 0, 1)
        s = s * target_sum / np.sum(s)
        return s
    
    def noise_perturb(h):
        n = h + 0.15 * np.random.randn(n_points)
        n = np.clip(n, 0, 1)
        n = n * target_sum / np.sum(n)
        return n
    
    perturbations = [initial_h, perturb(initial_h), noise_perturb(initial_h), sin_mod(initial_h)]

            corr = np.correlate(h, 1 - h, mode='full')

            grad = np.zeros(n_points)
