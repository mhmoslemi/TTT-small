import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    def generate_initial_population():
        pop = []
        pop.append(initial_guess)
        x = np.linspace(0, 2, n_points)
        sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
        sine_guess = normalize(np.clip(sine_guess, 0, 1))
        pop.append(sine_guess)
        triangular_guess = np.zeros(n_points)
        for i in range(n_points):
            triangular_guess[i] = 1.0 if i < n_points // 2 else 0.0
        triangular_guess = normalize(triangular_guess)
        pop.append(triangular_guess)
        random_guess = normalize(np.random.rand(n_points))
        pop.append(random_guess)
        return pop

    population = generate_initial_population()
    while len(population) < 20:
        random_guess = normalize(np.random.rand(n_points))
        population.append(random_guess)

    start_time = time.time()
    for gen in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(h) for h in population]
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:5]]
        new_population = elites
        while len(new_population) < 20:
            parent1 = np.random.choice(elites)
            parent2 = np.random.choice(elites)
            child = (parent1 + parent2) / 2.0

                    child[i] = np.clip(child[i], 0, 1)
