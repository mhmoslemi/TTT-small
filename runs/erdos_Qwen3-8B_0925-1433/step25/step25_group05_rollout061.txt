import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective_smooth(x, kappa):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))

    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)

    # Sine and cosine variations
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * (x - 1))
    sine_guess = np.clip(sine_guess, 0.2, 0.8)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)

    cosine_guess = 0.5 + 0.5 * np.cos(np.pi * (x - 1))
    cosine_guess = np.clip(cosine_guess, 0.2, 0.8)
    cosine_guess = normalize(cosine_guess)
    structured_guesses.append(cosine_guess)

    # Piecewise linear patterns
    triangular_guess = 0.5 + 0.5 * (2 * (x / 2) - 1)
    triangular_guess = np.clip(triangular_guess, 0, 1)
    triangular_guess = normalize(triangular_guess)
    structured_guesses.append(triangular_guess)

    # Randomized guesses
    for _ in range(3):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        noisy = initial_guess + noise
        noisy = np.clip(noisy, 0, 1)

        structured_guesses.append(noisy)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:

        result = minimize(objective_smooth, guess, args=(50,), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective_true(h)
            if current_c5 < best_c5:
