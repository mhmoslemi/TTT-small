import numpy as np
from scipy.optimize import minimize
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    # Initial samples
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = normalize(initial_h)
    samples = [initial_h]
    for _ in range(5):
        samples.append(normalize(initial_h + np.random.normal(0, 0.05, n_points)))
        samples.append(normalize(np.clip(np.random.choice([0, 1], size=n_points, p=[0.8, 0.2]), 0, 1)))
        samples.append(normalize(0.5 + 0.5 * (-1)**np.arange(n_points)))
        samples.append(normalize(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))))
    X = np.array([s for s in samples])
    y = np.array([np.max(np.correlate(s, 1 - s, mode='full')) * dx for s in samples])

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e3)) + WhiteKernel(noise_level=1e-2)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate new candidate
        new_h = initial_h + np.random.normal(0, 0.05, n_points)
        new_h = normalize(new_h)
        c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if c5 < best_c5:

        X = np.vstack([X, new_h])
        y = np.append(y, c5)
        gp.fit(X, y)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
