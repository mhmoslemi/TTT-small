import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()

    def generate_initial_guesses():
        guesses = [initial_guess]
        x = np.linspace(0, 2, n_points)
        # Sinusoidal base with varied frequencies
        for freq in [n_points/3, n_points/2, n_points/1.5]:
            phase = np.random.rand() * 2 * np.pi
            sin_wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
            guesses.append(normalize(sin_wave))
        # Pulse clusters with varied counts
        for m_pulses in [4, 5, 6]:
            spacing = 2.0 / (m_pulses * 2)
            h_pulse = np.zeros(n_points)
            for i in range(m_pulses):
                idx = int(i * spacing * n_points)
                h_pulse[idx] = 1.0
            guesses.append(normalize(h_pulse))
        # Random perturbation with increased diversity
        for std in [0.05, 0.1, 0.15]:
            rand = np.random.rand(n_points) + np.random.normal(0, std, n_points)
            guesses.append(normalize(rand))
        # Binary pattern with random intervals
        binary = np.zeros(n_points)
        binary[np.random.choice(n_points, size=n_points//2, replace=False)] = 1.0
        guesses.append(normalize(binary))
        return guesses

    for guess in generate_initial_guesses():
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
