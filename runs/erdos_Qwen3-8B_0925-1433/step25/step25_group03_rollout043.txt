def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Generate structured guesses
    x = np.linspace(0, 2, n_points)
    structured_guesses = []
    structured_guesses.append(initial_h_values)
    # Sine wave
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    # Triangular wave
    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    triangle_guess = normalize(triangle_guess)
    structured_guesses.append(triangle_guess)
    # Exponential decay
    exponential_guess = 0.5 + 0.5 * np.exp(-0.5 * x)
    exponential_guess = normalize(exponential_guess)
    structured_guesses.append(exponential_guess)
    # Random permutation of initial guess
    random_guess = np.random.rand(n_points)
    random_guess = normalize(random_guess)
    structured_guesses.append(random_guess)
    # Modified initial guess
    modified_guess = initial_h_values.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    modified_guess = normalize(modified_guess)
    structured_guesses.append(modified_guess)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_correlation(best_h)
    start_time = time.time()

    # Run DE on each structured guess
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            func=lambda h: evaluate_correlation(normalize(h)),
            bounds=bounds,
            strategy='best2bin',
            popsize=20,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8
        )
        if result.success:
            h_candidate = result.x
            h_candidate = normalize(h_candidate)
            current_c5 = evaluate_correlation(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    # Refine with SLSQP
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Prepare for SLSQP
        cons = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        ]
        result_slsqp = minimize(
            fun=lambda h: evaluate_correlation(h),
            x0=best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=cons,
            tol=1e-6
        )
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = evaluate_correlation(best_h)
            best_h = normalize(best_h)
            current_c5 = evaluate_correlation(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)

        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    x = np.linspace(0, 2, n_points)
    structured_guesses = [np.array(initial_h_values, dtype=float)]
    structured_guesses.append(0.5 + 0.5 * np.sin(np.pi * x))

    structured_guesses.append(0.5 + 0.5 * np.exp(-0.5 * x))
