from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = normalize(np.clip(x, 0, 1))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones_like}

        new_pop = []
