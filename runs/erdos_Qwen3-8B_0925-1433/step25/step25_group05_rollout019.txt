import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)

    # Add randomized and structured initializations
    for _ in range(2):
        perturb = np.random.normal(0, 0.08, n_points)
        guess = initial_guesses[0] + perturb
        guess = normalize(np.clip(guess, 0, 1))
        initial_guesses.append(guess)
    
    # Sine-based patterns
    for freq in [2, 3, 4]:
        sin_guess = 0.5 + 0.5 * np.sin(freq * np.pi * x)
        sin_guess = normalize(np.clip(sin_guess, 0, 1))
        initial_guesses.append(sin_guess)
    
    # Cosine with phase shifts
    for phase in [0, np.pi/4, np.pi/2]:
        cos_guess = 0.5 + 0.5 * np.cos(freq * np.pi * x + phase)
        cos_guess = normalize(np.clip(cos_guess, 0, 1))
        initial_guesses.append(cos_guess)

    best_h = initial_guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    # Smoothed objective for optimization
    def objective_smooth(x, kappa):
        h = x

        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))

    # True objective for evaluation
    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    # Initial optimization with diverse guesses

        result = minimize(

    # Final refinement
    for _ in range(3):
