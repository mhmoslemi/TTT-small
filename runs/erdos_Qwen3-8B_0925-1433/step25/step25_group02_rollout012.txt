import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values, dtype=float)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = normalize(h_initial.copy())
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    x = np.linspace(0, 2, n_points)
    harmonic_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    harmonic_guess = np.clip(harmonic_guess, 0, 1)
    harmonic_guess = normalize(harmonic_guess)

    def optimize_guess(guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=100)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            return h, current_c5
        return None, best_c5

    start_time = time.time()
    candidates = [best_h, harmonic_guess]

    for guess in candidates:
        if time.time() - start_time > budget_s:
            break
        h_opt, c_opt = optimize_guess(guess)
        if h_opt is not None and c_opt < best_c5:
            best_h, best_c5 = h_opt, c_opt

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h_opt, c_opt = optimize_guess(best_h)
        if h_opt is not None and c_opt < best_c5:
            best_h, best_c5 = h_opt, c_opt

    return (np.clip(best_h, 0, 1), best_c5, n_points)
