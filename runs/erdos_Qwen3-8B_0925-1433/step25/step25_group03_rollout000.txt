import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    
    # Sine wave with amplitude modulation
    sine_guess = 0.5 + 0.3 * np.sin(np.pi * x)
    structured_guesses.append(normalize(sine_guess))
    
    # Triangular wave with phase shift
    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(2 * np.pi * x))
    structured_guesses.append(normalize(triangle_guess))
    
    # Exponential decay with base modification
    exp_guess = 0.5 + 0.5 * np.exp(-0.6 * x)
    structured_guesses.append(normalize(exp_guess))
    
    # Piecewise constant pattern
    binary_guess = np.zeros(n_points)
    binary_guess[::3] = 1.0
    structured_guesses.append(normalize(binary_guess))
    
    # Random walk initialization
    random_guess = np.random.rand(n_points)
    structured_guesses.append(normalize(random_guess))
    
    # Flip half of initial pattern
    modified_guess = initial_h_values.copy()
    flip_mask = np.random.rand(n_points) < 0.4
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    structured_guesses.append(normalize(modified_guess))

    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:

    # Refinement phase with tighter constraints
