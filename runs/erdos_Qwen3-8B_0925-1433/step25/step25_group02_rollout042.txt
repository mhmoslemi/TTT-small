        population.append(np.random.rand(n_points))
    
    # Normalization function
    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        return evaluate_c5(x)

    start_time = time.time()
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)

    # GA loop
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        evals = [evaluate_c5(h) for h in population]
        best_idx = np.argmin(evals)
        if evals[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx].copy(), evals[best_idx]
        new_pop = [population[best_idx]]
        while len(new_pop) < len(population):
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = (p1 + p2) / 2
            mutate = np.random.normal(0, 0.05, n_points)
            child += mutate
            child = np.clip(child, 0, 1)
            new_pop.append(child)
        population = new_pop

    # SLSQP optimization
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    initial_guess = normalize(best_h)
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success:
        h = result.x
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
