        bounds=bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        tol=1e-6,
        seed=seed
    )
    
    if time.time() - start_time <= budget_s:
        best_h = result.x
        best_c5 = evaluate_correlation(clip_and_normalize(best_h))
    
    # Local refinement with SLSQP
    best_h_refined = np.copy(best_h)
    best_h_refined += np.random.normal(0, 0.01, best_h_refined.shape)
    best_h_refined = clip_and_normalize(best_h_refined)
    
    result_refined = minimize(
        fun=lambda h: evaluate_correlation(clip_and_normalize(h)),
        x0=best_h_refined,
        method='SLSQP',
        bounds=[(0,1)]*n_points,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
        tol=1e-6
    )
    
    if result_refined.success:
        best_h = result_refined.x
        best_c5 = evaluate_correlation(clip_and_normalize(best_h))
    
    best_h = clip_and_normalize(best_h)
    final_c5 = evaluate_correlation(best_h)
    return (best_h, final_c5, n_points)
