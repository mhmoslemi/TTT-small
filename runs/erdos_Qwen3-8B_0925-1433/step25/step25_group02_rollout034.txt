import time

def run(seed=42, budget_s=600):

        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = []

    initial_guesses.append(h_initial)
    # Sine guess
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = normalize(np.clip(sine, 0, 1))
    initial_guesses.append(sine)
    # Triangular guess
    tri = np.zeros(n_points)
    tri[:n_points//2] = 1.0
    tri = normalize(tri)
    initial_guesses.append(tri)
    # Binary guess
    binary = np.random.rand(n_points) < 0.5
    binary = normalize(binary.astype(float))
    initial_guesses.append(binary)
    # Random guess
    rand = np.random.rand(n_points)
    rand = normalize(rand)
    initial_guesses.append(rand)

    # GA setup
    population = initial_guesses
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)
    pop_size = len(population)

    # Run GA for 50 generations
    start_time = time.time()

            break
        # Evaluate all
        fitness = [evaluate_c5(h) for h in population]
