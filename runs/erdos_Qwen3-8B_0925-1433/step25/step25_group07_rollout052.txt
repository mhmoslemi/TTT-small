import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h0 = normalize(np.array(initial_h_values, dtype=float))
    h1 = normalize(1.0 - h0)
    h2 = normalize(np.roll(h0, 1))
    x = np.linspace(0, 2, n_points)
    h3 = normalize(np.clip(0.5 + 0.5 * np.sin(np.pi * x), 0, 1))
    h4 = normalize(0.5 + 0.5 * (-1)**np.arange(n_points))
    h5 = normalize(np.random.rand(n_points))
    population = [h0, h1, h2, h3, h4, h5]
    while len(population) < 50:
        pert = np.random.normal(0, 0.1, n_points)
        h = normalize(h0 + pert)
        population.append(h)

    best_h = h0
    best_c5 = evaluate(h0)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fit = [evaluate(h) for h in population]
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx].copy(), fit[best_idx]
        parents = []
        for _ in range(25):
            t = np.random.choice(len(population), 5, replace=False)
            best_t = t[np.argmin(fit[t])]
            parents.append(population[best_t])
        offspring = []
        for i in range(0, len(parents), 2):
            a, b = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([a[:split], b[split:]])
            offspring.append(child)
        mutated = []
        for child in offspring:
            noise = np.random.normal(0, 0.05, n_points)
            child = np.clip(child + noise, 0, 1)
            child = normalize(child)
            mutated.append(child)
        fit_mut = [evaluate(h) for h in mutated]
        combined = population + mutated
        fit_combined = [evaluate(h) for h in combined]
        new_pop = [combined[i] for i in np.argsort(fit_combined)[:50]]
        population = new_pop

    return (np.clip(best_h, 0, 1), best_c5, n_points)
