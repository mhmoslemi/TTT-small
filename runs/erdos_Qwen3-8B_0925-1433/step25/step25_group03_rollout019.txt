    best_h = structured_guesses[0].copy()
    best_c5 = evaluate_correlation(best_h)
    start_time = time.time()
    
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full') * dx
            max_corr = np.max(corr)
            sum_dev = np.sum(h) - target_sum
            return max_corr + 1e4 * sum_dev ** 2
        
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=20, tol=1e-6)
        h = np.clip(result.x, 0, 1)
        h = normalize(h)
        current_c5 = evaluate_correlation(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    def slsqp_objective(h):
        return evaluate_correlation(h)
    
    slsqp_constr = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    result_slsqp = minimize(slsqp_objective, best_h, method='SLSQP', bounds=bounds_slsqp, constraints=slsqp_constr, tol=1e-6)
    if result_slsqp.success:
        h = np.clip(result_slsqp.x, 0, 1)
        h = normalize(h)
        current_c5 = evaluate_correlation(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
