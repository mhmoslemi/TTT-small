    def penalty(x):
        sum_diff = np.sum(x) - target_sum
        bounds_penalty = np.sum(np.clip(x, 0, 1) - x)
        return 1e4 * (sum_diff**2 + bounds_penalty)

    class PSOParticle:

            self.vel = vel.copy()

            self.pbest = self.pos.copy()
            self.pbest_fitness = self.fitness
