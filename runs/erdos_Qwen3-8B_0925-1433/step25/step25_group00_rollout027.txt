import numpy as np
from scipy.optimize import minimize
from scipy.signal import dwt, idwt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    best_h = res.x
    best_c5 = res.fun

    coeffs = dwt(best_h, 'db4')
    coeffs[1] = np.where(np.abs(coeffs[1]) < 0.05 * np.max(coeffs[1]), 0, coeffs[1])
    h_filtered = idwt(coeffs, 'db4')
    h_filtered = h_filtered * target_sum / np.sum(h_filtered)
    h_filtered = np.clip(h_filtered, 0, 1)
    best_c5 = compute_c5(h_filtered)
    best_h = h_filtered

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h * (1 + 0.1 * np.random.randn(n_points))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        current_c5 = compute_c5(perturbed)
        if current_c5 < best_c5:
            best_h, best_c5 = perturbed, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
