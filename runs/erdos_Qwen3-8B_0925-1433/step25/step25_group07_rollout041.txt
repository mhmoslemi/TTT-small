import time

        guesses.append(normalize(random_guess))
        alt = np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])
        guesses.append(normalize(alt))
        # Additional guesses
        binary = np.random.choice([0, 1], size=n_points)

        return guesses

    def smooth_objective(x, kappa=100):
        corr = np.correlate(x, 1 - x, mode='full')
        exp_vals = np.exp(-kappa * corr)
        return np.log(np.sum(exp_vals)) / kappa

    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:

        result = minimize(smooth_objective, guess, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            corr = np.correlate(h, 1 - h, mode='full')

            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(2):
