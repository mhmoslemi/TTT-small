import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial structured guesses
    x = np.linspace(0, 2, n_points)
    structured_guesses = []

    # Sine guess
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)

    # Triangular guess
    triangular_guess = np.zeros(n_points)
    for i in range(n_points):
        triangular_guess[i] = 1.0 if i < n_points // 2 else 0.0
    triangular_guess = normalize(triangular_guess)
    structured_guesses.append(triangular_guess)

    # Initial guess
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = normalize(initial_h)
    structured_guesses.append(initial_h)

    # Random guess
    random_guess = np.random.rand(n_points)
    random_guess = normalize(random_guess)
    structured_guesses.append(random_guess)

    population = structured_guesses.copy()
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    # Genetic Algorithm phase
    for gen in range(50):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all individuals in population
        fitnesses = [evaluate_c5(h) for h in population]

        best_idx = np.argmin(fitnesses)

            child = normalize(child)
            new_population.append(child)

        population = new_population + [population[best_idx]]

import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    structured_guesses.append(sine_guess)

    # Triangular guess
    triangular_guess = np.zeros(n_points)
    for i in range(n_points):

    structured_guesses.append(triangular_guess)

    # Initial guess
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = normalize(initial_h)
    structured_guesses.append(initial_h)

    # Random guess
    random_guess = np.random.rand(n_points)
    random_guess = normalize(random_guess)
    structured_guesses.append(random_guess)

    population = structured_guesses.copy()
    best_h = population[0].copy()

    start_time = time.time()

    for gen in range(50):
        if time.time() - start_time > budget_s:

                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
