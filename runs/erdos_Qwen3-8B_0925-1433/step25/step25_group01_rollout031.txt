import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x) if np.sum(x) > 0 else x
        return x

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}

    def generate_population():
        pop = []
        h_pulse = np.zeros(n_points)
        for i in range(4):
            idx = int(i * n_points / 6)
            h_pulse[idx] += 1.0
        h_pulse = normalize(h_pulse)
        pop.append(h_pulse)
        x = np.linspace(0, 2, n_points)
        f = n_points / 6.0
        phi = np.random.rand() * 2 * np.pi
        h_sinusoidal = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + phi)
        h_sinusoidal = normalize(h_sinusoidal)
        pop.append(h_sinusoidal)
        pop.append(normalize(np.random.rand(n_points)))
        pop.append(normalize(initial_h_values))
        return pop

    def evolve_population(pop):
        fitness = [evaluate_c5(h) for h in pop]
        best_idx = np.argsort(fitness)[:len(pop)//3]
        new_pop = [pop[i] for i in best_idx]
        while len(new_pop) < len(pop):
            a, b = np.random.choice(best_idx, 2, replace=False)
            child = (pop[a] + pop[b]) / 2
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                child[idx] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = normalize(child)
            new_pop.append(child)
        return new_pop

    pop = generate_population()
    start = time.time()
    for _ in range(10):
        if time.time() - start > budget_s:
            break
        pop = evolve_population(pop)

    best_h = min(pop, key=evaluate_c5)
    res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
