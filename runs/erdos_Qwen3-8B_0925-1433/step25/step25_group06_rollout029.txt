import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = x * target_sum / np.sum(x)
        return x

    best_h = h.copy()

    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        def objective(x):
            h_opt = x
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            return corr_opt[k] * dx
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 50})
