def run(seed=42, budget_s=600):

    random_guess = random_guess * target_sum / np.sum(random_guess)
    population.append(random_guess)
    alt = 0.5 + 0.5 * (-1)**np.arange(n_points)
    alt = np.clip(alt, 0, 1)
    alt = alt * target_sum / np.sum(alt)
    population.append(alt)

    population = [np.clip(p, 0, 1) * target_sum / np.sum(p) for p in population]
