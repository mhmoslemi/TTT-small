from scipy.optimize import minimize
import time

        x = x * target_sum / np.sum(x)
        return x

    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones_like}

    start_time = time.time()

    def gen_pulse_cluster():
        m_pulses = 5
        spacing = 0.6
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(m_pulses):
            idx = int(i * spacing * n_points)
            h[idx] += 1.0
        return normalize(h)

    def gen_sinusoidal():
        x = np.linspace(0, 2, n_points)
        f = n_points / 4.0
        phi = np.random.rand() * 2 * np.pi
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + phi)
        return normalize(h)

    def gen_random():
        return normalize(np.random.rand(n_points))

    def gen_initial_pop():
        return [initial_guess, gen_pulse_cluster(), gen_sinusoidal(), gen_random(), np.random.rand(n_points)]

    pop = gen_initial_pop()
    pop = [normalize(np.clip(p, 0, 1)) for p in pop]
    pop = [p for p in pop if np.all(p >= 0) and np.all(p <= 1)]

    for gen in range(50):
        if time.time() - start_time > budget_s:

        scores = np.array([evaluate_c5(p) for p in pop])

        for _ in range(20 - len(new_pop)):

            child = normalize(np.clip(child, 0, 1))

        pop = new_pop
