from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = normalize(x)
        return evaluate_c5(h)

    # Generate structured initial guesses
    x = np.linspace(0, 2, n_points)
    guess1 = 0.5 + 0.5 * np.sin(np.pi * x)
    guess2 = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    guess3 = np.where(x < 1, 1, 0)
    guess4 = np.where(x < 0.5, 1, np.where(x < 1.5, 0.5, 0))
    guess5 = 0.5 * (1 + np.sign(np.sin(2 * np.pi * x)))
    guesses = [normalize(g) for g in [guess1, guess2, guess3, guess4, guess5]]

    # Get initial guess
    initial = normalize(np.array(initial_h_values, dtype=float))
    guesses.append(initial)
    guesses.append(np.random.rand(n_points))
    guesses.append(normalize(initial * np.random.rand(n_points)))
    guesses = [np.clip(g, 0, 1) for g in guesses]

    # DE bounds and constraints
    bounds = [(0, 1)] * n_points
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, maxiter=100,
        disp=False, seed=seed, polish=True, polish_x0=initial
    )
    best_h = result.x
    best_c5 = evaluate_c5(best_h)

    # Local refinement
    for _ in range(5):
