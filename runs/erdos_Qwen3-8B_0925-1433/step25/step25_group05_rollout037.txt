import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    # Add structured perturbations
    for amp in [0.15, 0.25]:
        sine_guess = 0.5 + amp * np.sin(np.pi * x)
        sine_guess = np.clip(sine_guess, 0, 1)
        sine_guess = normalize(sine_guess)
        structured_guesses.append(sine_guess)
    for amp in [0.1, 0.2]:
        cosine_guess = 0.5 + amp * np.cos(np.pi * x)
        cosine_guess = np.clip(cosine_guess, 0, 1)
        cosine_guess = normalize(cosine_guess)
        structured_guesses.append(cosine_guess)
    for amp in [0.1, 0.2]:
        triangular_guess = 0.5 + amp * (2 * (x / 2) - 1)
        triangular_guess = np.clip(triangular_guess, 0, 1)
        triangular_guess = normalize(triangular_guess)
        structured_guesses.append(triangular_guess)
    for amp in [0.05, 0.1]:
        square_guess = 0.5 + amp * np.sign(np.sin(np.pi * x))
        square_guess = np.clip(square_guess, 0, 1)
        square_guess = normalize(square_guess)
        structured_guesses.append(square_guess)
    for amp in [0.05, 0.1]:
        sawtooth_guess = 0.5 + amp * (2 * (x / 2) - 1)
        sawtooth_guess = np.clip(sawtooth_guess, 0, 1)
        sawtooth_guess = normalize(sawtooth_guess)
        structured_guesses.append(sawtooth_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        # First optimization with trust-constr
        result = minimize(objective, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        # Second optimization with L-BFGS-B
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
