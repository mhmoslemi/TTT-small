import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Initial population: base + perturbations
    population = []
    for _ in range(10):
        perturbed = initial_h_values.copy()
        perturbed += np.random.uniform(-0.1, 0.1, n_points)
        perturbed = normalize(perturbed)
        population.append(perturbed)

    # Evaluate initial population
    fitness = [evaluate_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # Genetic Algorithm
    pop_size = 20
    generations = 40
    mutation_rate = 0.15
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Select parents
        parent_idx = np.argsort(fitness)[:pop_size//2]
        parents = [population[i] for i in parent_idx]
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            for i in range(n_points):
                if np.random.rand() < mutation_rate:

            child = normalize(child)
            new_pop.append(child)
        population = new_pop
        fitness = [evaluate_c5(ind) for ind in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_h, best_c5 = population[current_best].copy(), fitness[current_best]

    # SLSQP Refinement
    remaining_time = budget_s - (time.time() - start_time)
    top_candidates = [population[i] for i in np.argsort(fitness)[:5]]
    for candidate in top_candidates:
