import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    base_guess = np.array(initial_h_values, dtype=float)
    base_guess = normalize(base_guess)

    def objective_smooth(x, kappa):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))
    
    def objective_true(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evaluate(x):
        x = normalize(x)
        return objective_true(x)

    best_h = base_guess.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    guesses = [base_guess]
    for i in range(3):
        noise = np.random.normal(0, 0.1, n_points)
        guesses.append(base_guess + noise)
        noise = np.random.uniform(-0.1, 0.1, n_points)
        guesses.append(base_guess + noise)
        noise = np.sin(np.pi * np.arange(n_points) / 4) * 0.1
        guesses.append(base_guess * (1 + noise))

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective_smooth, guess, args=(50,), 
                          method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},

            current_c5 = evaluate(h)
