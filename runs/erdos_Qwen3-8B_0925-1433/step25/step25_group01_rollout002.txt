import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones_like}

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)

    def generate_guesses():
        x = np.linspace(0, 2, n_points)
        sin_wave = 0.4 * np.sin(2 * np.pi * 1.2 * x + np.random.rand() * 2 * np.pi) + 0.5
        binary = np.zeros(n_points)
        binary[n_points//3:n_points//3*2] = 1.0
        rand = np.random.rand(n_points)
        return [normalize(sin_wave), normalize(binary), normalize(rand)]

    start_time = time.time()

    for guess in [initial_guess] + generate_guesses():
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
