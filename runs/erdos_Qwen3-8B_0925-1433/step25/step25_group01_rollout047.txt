import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial guess and variations
    population = [
        normalize(np.array(initial_h_values, dtype=float)),
        normalize(np.zeros(n_points)),  # baseline
        normalize(np.random.rand(n_points)),
        normalize(np.linspace(0, 2, n_points) / 2),
        normalize(np.sin(np.linspace(0, np.pi, n_points)) / 2 + 0.5)
    ]
    while len(population) < 20:
        population.append(normalize(np.random.rand(n_points)))

    # GA parameters
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.75
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        # Select top 20% as parents
        parent_indices = np.argsort(fitness)[:int(len(population)*0.2)]
        parents = [population[i] for i in parent_indices]
        # Generate new candidates
        new_pop = []
        for _ in range(len(population) - len(parents)):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = (p1 + p2) / 2
            else:
                p = np.random.choice(parents)
                child = p * (1 + np.random.normal(0, 0.05, size=n_points))
            new_pop.append(normalize(child))
        # Combine and select top performers
        combined = population + new_pop
        combined.sort(key=lambda x: evaluate_c5(x))
        population = combined[:20]

    # Local refinement
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)
    bounds = [(0, 1) * n_points]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}
    res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_h = normalize(best_h)
        best_c5 = evaluate_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
