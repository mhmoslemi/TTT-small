import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_correlation(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    structured_guesses.append(normalize(sine_guess))
    triangle_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    structured_guesses.append(normalize(triangle_guess))
    exp_guess = 0.5 + 0.5 * np.exp(-0.5 * x)
    structured_guesses.append(normalize(exp_guess))
    uniform_guess = np.random.rand(n_points)
    structured_guesses.append(normalize(uniform_guess))
    flipped_guess = initial_guess.copy()
    flipped_mask = np.random.rand(n_points) < 0.3
    flipped_guess[flipped_mask] = 1 - flipped_guess[flipped_mask]
    structured_guesses.append(normalize(flipped_guess))
    piecewise_guess = np.zeros_like(x)
    block_len = 0.4
    for i in range(0, 2, block_len):
        start_idx = int(i / dx)
        end_idx = int((i + block_len) / dx)
        if start_idx < n_points:
            piecewise_guess[start_idx:end_idx] = 1.0
    structured_guesses.append(normalize(piecewise_guess))

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_correlation, guess, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-5)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_correlation, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-5)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
