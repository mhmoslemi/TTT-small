import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = x * target_sum / np.sum(x)
        return x

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        return evaluate_correlation(h) + 1e5 * abs(np.sum(h) - target_sum)

    best_h = np.array(initial_h_values, dtype=float)
    best_h = normalize(best_h)
    best_c5 = evaluate_correlation(best_h)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)

    start_time = time.time()
