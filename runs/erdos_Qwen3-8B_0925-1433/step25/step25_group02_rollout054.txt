import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    def normalize(x):

        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)

    structured_guesses.append(normalize(sine_guess))
    triangular_guess = np.zeros(n_points)
    for i in range(n_points):
        triangular_guess[i] = 1.0 if i < n_points // 2 else 0.0
    structured_guesses.append(normalize(triangular_guess))
    uniform_guess = np.random.rand(n_points)

    structured_guesses.append(normalize(square_guess))
    sawtooth_guess = np.linspace(0, 1, n_points)

    structured_guesses.append(normalize(zigzag_guess))

    start_time = time.time()
