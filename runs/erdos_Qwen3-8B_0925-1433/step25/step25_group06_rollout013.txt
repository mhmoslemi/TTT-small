import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

            corr_current = np.correlate(h, 1 - h, mode='full')
            return corr_current[k] * dx
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
