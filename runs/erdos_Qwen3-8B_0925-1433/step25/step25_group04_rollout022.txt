def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.995
    max_iterations = 1000

    for it in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h_current, 1 - h_current, mode='full')
