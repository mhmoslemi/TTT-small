        grad = np.zeros(n)
        for i in range(n):
            h_p = h.copy()
            h_p[i] += delta
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')
            corr_k_p = corr_p[k]
            h_m = h.copy()
            h_m[i] -= delta
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')
            corr_k_m = corr_m[k]
            grad[i] = (corr_k_p - corr_k_m) / (2 * delta)
        return grad

    def update_h(h, alpha):
        k = np.argmax(np.correlate(h, 1 - h, mode='full'))
        grad = compute_gradient(h, k)
        h_new = h - alpha * grad
        return clip_norm(h_new)

    start_time = time.time()
    h = initial_h.copy()
    alpha = 0.1
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        h = update_h(h, alpha)
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.abs(current_c5 - best_c5) < 1e-6:
            alpha *= 0.9
        elif np.abs(current_c5 - best_c5) > 1e-5:
            alpha *= 1.01
        if np.random.rand() < 0.2:
            h = clip_norm(np.random.permutation(h))
        if np.random.rand() < 0.1:
            h = clip_norm(h + 0.15 * np.random.randn(n_points))

    return (clip_norm(best_h), best_c5, n_points)
