        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize(h_new)
            current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), current_c5
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (np.clip(best_h, 0, 1), final_c5, n_points)
