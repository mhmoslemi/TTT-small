from scipy.optimize import differential_evolution, minimize

        return 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    def generate_sine(x):
        return 0.5 + 0.5 * np.sin(np.pi * x)
    def generate_exp(x):
        return 0.5 + 0.5 * np.exp(-0.5 * x)
    def generate_poly(x):
        return 0.5 + 0.5 * (x**2 - 2*x) / (2**2 - 2*0)

    x_vals = np.linspace(0, 2, n_points)
    guesses = []

    guess3 = generate_sine(x_vals)
    guess3 = np.clip(guess3, 0, 1) * target_sum / np.sum(guess3)
    guesses.append(guess3)

    guess4 = generate_exp(x_vals)
