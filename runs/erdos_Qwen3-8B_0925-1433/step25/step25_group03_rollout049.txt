    period = 0.5
    for i in range(n_points):
        if (x[i] % period) < period / 2:
            periodic_guess[i] = 1
        else:
            periodic_guess[i] = 0
    structured_guesses.append(periodic_guess)

    # Initial guess from parent
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    structured_guesses.append(initial_guess)

    best_h = initial_guess.copy()
    best_c5 = evaluate_correlation(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        guess = normalize(guess)
        result = minimize(evaluate_correlation, guess, method='Nelder-Mead', bounds=[(0,1)]*n_points, tol=1e-5)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(evaluate_correlation, best_h, method='Nelder-Mead', bounds=[(0,1)]*n_points, tol=1e-5)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_correlation(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
