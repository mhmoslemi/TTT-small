import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = x * target_sum / np.sum(x)
        return x

    best_h = h.copy()

        lag = np.argmax(corr)
