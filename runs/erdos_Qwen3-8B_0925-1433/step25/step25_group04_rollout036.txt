        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')[n_points:2*n_points-1]) * dx
        for _ in range(300):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            corr_positive = corr[n_points:2*n_points-1]
            if len(corr_positive) == 0:
                break
            max_index = np.argmax(corr_positive)
            k_max = max_index + 1
            grad = np.zeros_like(current_h)
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] += (1 - current_h[i + k_max])
                if i - k_max >= 0:
                    grad[i] -= current_h[i - k_max]
            step_size = 0.1
            current_h_new = current_h - step_size * grad
            current_h_new = clip_and_normalize(current_h_new)
            corr_new = np.correlate(current_h_new, 1 - current_h_new, mode='full')
            current_c5_new = np.max(corr_new[n_points:2*n_points-1]) * dx
            if current_c5_new < current_c5:
                current_h = current_h_new.copy()
                current_c5 = current_c5_new
            else:
                step_size *= 0.5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    return (best_h, best_c5, n_points)
