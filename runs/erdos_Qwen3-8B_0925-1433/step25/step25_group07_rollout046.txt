import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Diverse initial guesses
    guesses = []
    guesses.append(normalize(np.array(initial_h_values, dtype=float)))
    guesses.append(normalize(1.0 - np.array(initial_h_values, dtype=float)))
    guesses.append(normalize(np.roll(initial_h_values, 1)))
    guesses.append(normalize(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points))))
    guesses.append(normalize(np.random.rand(n_points)))
    guesses.append(normalize(np.array([0.5 + 0.5 * (-1)**i for i in range(n_points)])))
    guesses.append(normalize(np.array([0.5 + 0.5 * np.cos(2 * np.pi * i / 3) for i in range(n_points)])))

    best_h = guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = normalize(np.clip(result.x, 0, 1))
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
