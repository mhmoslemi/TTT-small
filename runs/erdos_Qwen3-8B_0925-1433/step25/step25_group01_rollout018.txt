from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    m_pulses = 4
    pulsed = np.zeros(n_points)

        pos = spacing * (i + 1)

    cluster[:3] = 1.0
    cluster = cluster * target_sum / np.sum(cluster)
    initial_guesses.append(cluster)

    initial_guesses.append(sin_wave)
    
    # Random
    rand = np.random.rand(n_points)

    initial_guesses.append(rand)
    
    best_h = initial_guesses[0].copy()
