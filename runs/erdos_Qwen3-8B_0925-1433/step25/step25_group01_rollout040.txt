import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    x_vals = np.linspace(0, 2, n_points)
    initial_guesses = [initial_h_values.copy()]

    # Binary pattern
    binary = np.zeros(n_points)
    binary[::3] = 1.0
    initial_guesses.append(binary)

    # Sine wave
    phase = np.random.rand() * 2 * np.pi
    freq = 1.2
    sin_wave = 0.4 + 0.6 * np.sin(2 * np.pi * freq * x_vals + phase)
    initial_guesses.append(sin_wave)

    # Random pattern
    random_guess = np.random.rand(n_points)
    initial_guesses.append(random_guess)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    def optimize(guess):
        guess = normalize(np.array(guess, dtype=float))
        res = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, tol=1e-6, maxiter=100)
        refined = res.x
        res_lbg = minimize(objective, refined, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        return res_lbg.x, res_lbg.fun

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        refined, current_c5 = optimize(guess)
        if current_c5 < best_c5:
            best_h, best_c5 = refined, current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res_lbg = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        refined = res_lbg.x
        current_c5 = res_lbg.fun
        if current_c5 < best_c5:
            best_h, best_c5 = refined, current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
