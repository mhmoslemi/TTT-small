import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    # Initial structured guesses
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)

    shifted_guess = np.roll(initial_guess, 1)
    shifted_guess = normalize(np.clip(shifted_guess, 0, 1))

    guesses = [initial_guess, sine_guess, shifted_guess, uniform_guess]

    for guess in guesses:
        if time.time() - start_time >= budget_s:
            break
        result = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, guess, method='COBYLA',
                          bounds=bounds, constraints=cons, tol=1e-4)

            if current_c5 < best_c5:
