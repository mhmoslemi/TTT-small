def run(seed=42, budget_s=600):

    # Structured guesses
    structured_guesses = []
    x = np.linspace(0, 2, n_points)

    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = normalize(np.clip(sine_guess, 0, 1))
    structured_guesses.append(sine_guess)
    # Cosine guess
    cosine_guess = 0.5 + 0.5 * np.cos(np.pi * x)
    cosine_guess = normalize(np.clip(cosine_guess, 0, 1))
    structured_guesses.append(cosine_guess)
