        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k = np.argmax(corr)
        def objective(x):
            h_current = x
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            return corr[k] * dx
        try:
            result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if result.success:
                h_new = np.clip(result.x, 0, 1)
                h_new = normalize(h_new)
                current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = current_c5
        except:
            continue
    return (np.clip(best_h, 0, 1), best_c5, n_points)
