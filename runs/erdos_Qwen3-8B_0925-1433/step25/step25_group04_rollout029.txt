import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    # Generate perturbations
    def generate_perturbations(h):
        # 1. Random permutation
        perm = h.copy()
        np.random.shuffle(perm)
        perm = normalize(perm)
        
        # 2. Gaussian noise
        noise = np.random.normal(0, 0.15, size=h.shape)
        noise_h = h + noise
        noise_h = normalize(noise_h)
        
        # 3. Sinusoidal modulation
        x = np.linspace(0, 2, n_points)
        sin_mod = 0.5 + 0.5 * np.sin(np.pi * x)
        sin_h = h * sin_mod
        sin_h = normalize(sin_h)
        
        # 4. Polynomial perturbation
        poly = 0.5 + 0.3 * (x / 2) + 0.2 * (x / 2) ** 2
        poly_h = h * poly
        poly_h = normalize(poly_h)
        
        return [perm, noise_h, sin_h, poly_h]

    perturbations = generate_perturbations(initial_guess)
    perturbations.append(initial_guess)  # include original

    start_time = time.time()
    for h in perturbations:
        if time.time() - start_time > budget_s:
            break
        # Optimize this h
        # Use L-BFGS-B with constraints

            h_curr = x
            corr = np.correlate(h_curr, 1 - h_curr, mode='full')
            return np.max(corr) * dx
