import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective_smooth(x, kappa):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    guesses = [
        initial_guess,
        normalize(initial_guess + np.random.normal(0, 0.1, n_points)),
        normalize(initial_guess + 0.1 * np.sin(np.pi * np.arange(n_points) / 4)),
        normalize(initial_guess + np.random.normal(0, 0.15, n_points)),
        normalize(initial_guess + 0.1 * np.cos(np.pi * np.arange(n_points) / 4))
    ]

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective_smooth, guess, args=(50,), method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
        if res.success:
            h_opt = normalize(res.x)
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            curr_c5 = np.max(corr) * dx
            if curr_c5 < best_c5:
                best_h = h_opt
                best_c5 = curr_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
