import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Differential evolution with structured seeding
    try:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=20, mutation=(0.7, 1), crossover=0.9,
            tol=1e-6, maxiter=200, seed=seed, disp=False
        )
        best_h = result_de.x
        best_c5 = objective(best_h)
    except:
        best_h, best_c5 = h.copy(), objective(h)

    # Local refinement with SLSQP and COBYLA
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_local = [(0, 1)] * n_points

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        result_slsqp = minimize(
            objective, best_h, method='SLSQP',
            bounds=bounds_local, constraints=cons, tol=1e-8, options={'maxiter': 100}
        )
        result_cobyla = minimize(
            objective, result_slsqp.x, method='COBYLA',
            bounds=bounds_local, constraints=cons, tol=1e-8, options={'maxiter': 100}
        )
        if result_cobyla.fun < best_c5:
            best_h, best_c5 = result_cobyla.x, result_cobyla.fun

    # Perturbation refinement with L-BFGS-B
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(
            objective, new_h, method='L-BFGS-B',
            bounds=bounds_local, constraints=cons, tol=1e-10, options={'maxiter': 50}
        )
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final adjustments
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h

    return (best_h, best_c5, n_points)
