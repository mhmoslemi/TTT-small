from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    half = n_points // 2
    half_idx = np.arange(half)

    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        h_sym = np.concatenate([x, 1.0 - x[::-1]])
        corr = np.correlate(h_sym, 1.0 - h_sym, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0)] * half

    result = minimize(objective, h[:half], method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
