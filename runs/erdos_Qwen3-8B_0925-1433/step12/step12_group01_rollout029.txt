import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    n_particles = 20
    particles = np.array([np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1) for _ in range(n_particles - 1)])
    particles = np.concatenate([particles, [initial_h_values]], axis=0)
    
    def scale(h):
        if np.sum(h) > 0:
            return h * target_sum / np.sum(h)
        return h
    
    particles = np.array([scale(h) for h in particles])
    
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    fitness = np.array([evaluate_c5(p) for p in particles])
    
    pbest_pos = particles.copy()
    pbest_fitness = fitness.copy()
    gbest_idx = np.argmin(fitness)
    gbest_pos = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    omega, phi1, phi2 = 0.7, 1.5, 1.5
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        r1, r2 = np.random.rand(n_particles, n_points), np.random.rand(n_particles, n_points)
        velocities = omega * velocities + phi1 * r1 * (pbest_pos - particles) + phi2 * r2 * (gbest_pos - particles)
        particles += velocities
        
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = scale(particles[i])
        
        new_fitness = np.array([evaluate_c5(p) for p in particles])
        
        for i in range(n_particles):
            if new_fitness[i] < pbest_fitness[i]:
                pbest_pos[i], pbest_fitness[i] = particles[i].copy(), new_fitness[i]
        
        best_idx = np.argmin(pbest_fitness)
        if pbest_fitness[best_idx] < gbest_fitness:
            gbest_pos, gbest_fitness = pbest_pos[best_idx].copy(), pbest_fitness[best_idx]
    
    return (gbest_pos, gbest_fitness, n_points)
