import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_guesses = [initial_h_values.copy()]

    def generate_evenPattern(n_points):
        pattern = np.zeros(n_points)
        pattern[:n_points//2] = 1.0
        return pattern

    def generate_fibPattern(n_points):
        phi = (1 + np.sqrt(5)) / 2
        pos = np.sort([i * phi % 2 for i in range(n_points)])
        pattern = np.zeros(n_points)
        pattern[:n_points//2] = 1.0
        return pattern

    initial_guesses.extend([generate_evenPattern(n_points), generate_fibPattern(n_points)])

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1.0 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
