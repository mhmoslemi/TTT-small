    m, v = np.zeros(n_points), np.zeros(n_points)
    for step in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k_max = k_max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_forward = (i + k_max) % n_points
            i_backward = (i - k_max) % n_points
            grad[i] = (1 - h[i_forward]) - h[i_backward]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = clip_and_norm(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= 0.999

    # L-BFGS refinement
    def l_bfgs_obj(x):
        h = clip_and_norm(x)
        return compute_c5(h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(l_bfgs_obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x

    best_h = clip_and_norm(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
