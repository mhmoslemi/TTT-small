from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        sum_h = np.sum(h_)

            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:

            tol = 1e-5
            perturb_std = 0.1
        elif phase == 1:
            method = 'COBYLA'
            tol = 1e-6
            perturb_std = 0.05
