            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

        # Second phase: SLSQP for convergence
        for _ in range(150):
            if time.time() - time.time() > budget_s:
                break
            perturb_std = 0.001 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

        # Third phase: L-BFGS-B for fine-tuning
        for _ in range(100):
            if time.time() - time.time() > budget_s:
                break
            perturb_std = 0.0005 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

    return (best_h, best_c5, n_points)
