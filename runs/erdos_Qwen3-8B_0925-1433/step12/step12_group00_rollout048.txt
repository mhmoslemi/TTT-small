        trial_c5 = np.max(corr) * dx
        if trial_c5 < best_c5:
            h = trial_h
            best_c5 = trial_c5
            res = minimize(evaluate, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                           jac=compute_gradient, tol=1e-8, options={'maxiter': 50})
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        else:
            h = trial_h

    # Final evaluation
    final_h = clip_and_normalize(best_h)
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
    return (final_h, final_c5, n_points)
