import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        cross_corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(cross_corr)
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.01 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        try:
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.005 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        try:
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:

        except:
            pass
