            else:
                child = p1.copy()
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, std_dev, n_points)
            new_pop.append(clip_and_norm(child))
        pop = np.array(new_pop)
        # Update best
        current_fitness = fitness[idxs[0]]
        if current_fitness < best_c5:
            best_h, best_c5 = pop[idxs[0]].copy(), current_fitness

    # Local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Use top 5 individuals
        top_indices = np.argsort(fitness)[:5]
        for i in top_indices:
            h = pop[i].copy()
            def obj(x):
                return np.max(np.correlate(x, 1 - x, mode='full')) * dx
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x.copy(), res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
