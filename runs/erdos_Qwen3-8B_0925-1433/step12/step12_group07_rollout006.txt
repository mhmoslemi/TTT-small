import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        if np.sum(candidate) > 0:
            candidate *= target_sum / np.sum(candidate)
        pop[i] = candidate
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')

    best_h = pop[0]
