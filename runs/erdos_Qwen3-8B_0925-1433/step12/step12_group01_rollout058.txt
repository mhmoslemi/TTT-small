from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_block_pattern(m_blocks):
        block_size = n_points // m_blocks
        h = np.zeros(n_points)
        for i in range(m_blocks // 2):
            h[i * block_size:(i + 1) * block_size] = 1.0
        return h * target_sum / np.sum(h)

    structured_patterns = [generate_block_pattern(m) for m in [3, 5, 7, 9, 11]]
    if initial_h_values is not None:
        structured_patterns.append(initial_h_values)

    bounds = [(0.0, 1.0)] * n_points

    def objective_de(h):
        sum_h = np.sum(h)
        if sum_h < 1e-10:
            return float('inf')
        h_norm = h * (target_sum / sum_h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    result = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=20, tol=1e-6, mutation=(0.8, 1), recombination=0.9, maxiter=200)
    best_h = result.x

    def objective_local(x):
        sum_h = np.sum(x)
        if sum_h < 1e-10:
            return float('inf')
        h_norm = x * (target_sum / sum_h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_local = [(0.0, 1.0)] * n_points
