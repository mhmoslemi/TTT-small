import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        population.append(clip_and_normalize(h))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')

    start_time = time.time()

    # Genetic Algorithm Loop
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        # Select parents
        parents = []
        for _ in range(population_size // 2):
            t = np.random.choice(range(population_size), 5)
            winner = t[np.argmin([fitness[i] for i in t])]
            parents.append(population[winner])

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i + 1]
            if np.random.rand() < 0.5:
                split = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
                offspring.extend([c1, c2])
            else:
                offspring.extend([p1, p2])

        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                offspring[i] = clip_and_normalize(offspring[i] + noise)

        # Replace population
        new_pop = offspring[:population_size]
        for i in range(population_size):
            if np.random.rand() < 0.2:
                new_pop[i] = clip_and_normalize(np.random.rand(n_points) * 2 - 1)
        population = new_pop

    # Local refinement
    top_cands = [population[i] for i in np.argsort(fitness)[:5]]
    for h in top_cands:
        def obj(x):
            return np.max(np.correlate(clip_and_normalize(x), 1 - clip_and_normalize(x), mode='full')) * dx
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = clip_and_normalize(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
