import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(population_size - 1)]
    population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) if np.sum(np.clip(h, 0, 1)) > 0 else h_initial for h in population]
    population = [h_initial] + population
    
    best_h, best_c5 = h_initial.copy(), compute_c5(h_initial)
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(population_size):
            idxs = np.random.choice(population_size, 3, replace=False)
            h1, h2, h3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = h1 + 0.8 * (h2 - h3)
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum > 0:
                trial = trial * target_sum / trial_sum
            else:
                trial = population[i]
            new_population.append(trial)
        for trial in new_population:
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_h, best_c5 = trial.copy(), c5
        population = new_population
    
    return (best_h, best_c5, n_points)
