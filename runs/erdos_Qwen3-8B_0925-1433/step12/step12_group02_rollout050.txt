import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 50
        h = np.random.rand(n_points)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    # Generate initial population
    population_size = 50
    population = [h.copy()]
    for _ in range(population_size):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)

    best_c5 = float('inf')
    best_h = h.copy()
    for candidate in population:
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Perturb current best and optimize
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(best_h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        result = minimize(lambda x: compute_c5(x), candidate, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        else:
            break

    return (best_h, best_c5, n_points)
