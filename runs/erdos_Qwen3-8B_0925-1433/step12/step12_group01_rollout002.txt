import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_scaled = h_clipped * target_sum / sum_h
        else:
            h_scaled = np.zeros(n_points)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            return h_clipped * target_sum / sum_h
        return np.zeros(n_points)

    def create_block_pattern(num_blocks):
        if num_blocks == 0:
            return np.zeros(n_points)
        block_size = n_points // num_blocks
        h = np.zeros(n_points)
        for i in range(num_blocks):
            h[i*block_size:(i+1)*block_size] = 1.0
        return rescale(h)

    def create_checkerboard(block_size):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % (2 * block_size) < block_size:
                h[i] = 1.0
        return rescale(h)

    if initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = rescale(h_start)

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h_start.copy()
    best_c5 = obj(best_h)

    # Generate structured patterns
    patterns = []
    for b in [3, 5, 7, 9, 11, 13]:
        patterns.append(create_block_pattern(b))
    for b in [4, 6, 8, 10, 12]:
        patterns.append(create_checkerboard(b))
    patterns.append(h_start)

    # Differential evolution
    if time.time() - start_time < budget_s * 0.6:
        result = differential_evolution(
            fun=obj,
            bounds=bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=int(budget_s * 0.3),
            polish=True,
            disp=False,
            vectorized=True
        )
        best_h = result.x
        best_c5 = obj(best_h)

    # Local refinement with SLSQP
    if time.time() - start_time < budget_s * 0.8:
        res = minimize(
            obj,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-8
        )
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    # Local refinement with COBYLA
    if time.time() - start_time < budget_s * 0.9:
        res = minimize(
            obj,
            best_h,
            method='COBYLA',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-8
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Final perturbation
    if time.time() - start_time < budget_s:
        perturb_std = 0.05
        for _ in range(10):
            perturb = np.random.normal(0, perturb_std, n_points)

            res = minimize(

                new_h,

                best_h = res.x
                best_c5 = res.fun

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
