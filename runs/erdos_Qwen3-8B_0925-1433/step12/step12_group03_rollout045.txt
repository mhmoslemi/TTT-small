def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    def generate_pattern(n_points):

            if i % 3 == 0 or i % 3 == 1:
                pattern[i] = 1.0
        sum_pattern = np.sum(pattern)
        scale = target_sum / sum_pattern
        return pattern * scale

        corr = np.correlate(h_, 1 - h_, mode='full')
