import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_index = np.argmax(corr)
        k = max_corr_index - (n_points - 1)
        perturb = np.zeros_like(best_h)
        if k > 0:
            indices = np.arange(n_points - k)
            contrib = best_h[indices] * (1 - best_h[indices + k])
            perturb[indices] -= contrib * 0.01
        else:
            k = -k
            indices = np.arange(k, n_points)
            contrib = best_h[indices] * (1 - best_h[indices - k])
            perturb[indices] -= contrib * 0.01
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')

            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
