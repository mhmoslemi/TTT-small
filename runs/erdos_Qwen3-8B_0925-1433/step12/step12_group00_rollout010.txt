import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    # Initial population
    pop_size = 100
    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(pop_size - 1):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        population.append(repair(h))

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(p) for p in population]
        sorted_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        elites = sorted_pop[:int(0.4 * pop_size)]  # Keep top 40%
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            a = np.random.choice(len(elites))
            b = np.random.choice(len(elites))
            if np.random.rand() < 0.5:
                # Crossover
                idx = np.random.randint(1, n_points-1)
                child = np.concatenate([sorted_pop[a][:idx], sorted_pop[b][idx:]])
            else:
                child = sorted_pop[a].copy()
            if np.random.rand() < 0.1:
                # Mutation
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
            new_pop.append(repair(child))
        population = new_pop

    # Local refinement
    best_h = None
    best_c5 = float('inf')
    for ind in population:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(lambda x: compute_c5(repair(x)), ind, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    best_h = repair(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
