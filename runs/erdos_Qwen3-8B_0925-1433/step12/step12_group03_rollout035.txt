import numpy as np

import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_guesses = []
    initial_guesses.append(initial_h_values.copy())

    x_vals = np.linspace(0, 2, n_points)
    sine_h = 0.5 + 0.5 * np.sin(np.pi * x_vals)

    sine_h = sine_h * target_sum / np.sum(sine_h)
    initial_guesses.append(sine_h)

    even_h = np.zeros(n_points)
    even_h[::2] = 1.0
    even_h = even_h * target_sum / np.sum(even_h)
    initial_guesses.append(even_h)

    best_h = initial_guesses[0].copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        res_cobyla = minimize(obj, guess, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 200})
        curr_h = res_cobyla.x if res_cobyla.success else guess.copy()

        res_slsqp = minimize(obj, curr_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 150})

        curr_c5 = res_l_bfgs.fun if res_l_bfgs.success else curr_c5
