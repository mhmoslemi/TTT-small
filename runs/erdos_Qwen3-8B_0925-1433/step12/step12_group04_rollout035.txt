    for i in range(n_points // 2):
        j = n_points - 1 - i
        if j < i:
            continue
        h[j] = 1.0 - h[i]

    # Function to compute C5 from half-values
    def compute_c5(x):
        h_full = np.zeros(n_points)
        for i in range(n_points // 2):
            h_full[i] = x[i]
            j = n_points - 1 - i
            h_full[j] = 1.0 - x[i]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Bounds for the first half of h
    bounds = [(0.0, 1.0)] * (n_points // 2)

    # Initial guess is the first half of h
    x0 = h[:n_points // 2].copy()

    start_time = time.time()
    best_x = x0.copy()
    best_c5 = compute_c5(best_x)

    # Optimization with L-BFGS-B
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(compute_c5, best_x, method='L-BFGS-B', bounds=bounds, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_x = res.x
            best_c5 = res.fun

    # Construct final h
    h_full = np.zeros(n_points)
    for i in range(n_points // 2):
        h_full[i] = best_x[i]
        j = n_points - 1 - i
        h_full[j] = 1.0 - best_x[i]

    return (h_full, best_c5, n_points)
