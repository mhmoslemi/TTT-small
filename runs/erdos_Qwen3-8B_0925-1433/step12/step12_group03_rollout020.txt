import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        patterns = []
        patterns.append(initial_h_values.copy())
        x = np.linspace(0, 2, n_points)
        # Sine pattern
        patterns.append(0.5 + 0.5 * np.sin(np.pi * x))
        # Binary pattern
        ones = np.zeros(n_points)
        ones[:int(target_sum)] = 1
        patterns.append(ones * target_sum / np.sum(ones))
        # Triangular pattern
        t = 1 - np.abs(x - 1)
        patterns.append(t * target_sum / np.sum(t))
        return patterns

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def optimize_pattern(h0):
        h = np.clip(h0, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        best_h = h.copy()
        best_c5 = obj(h)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        for method in ['COBYLA', 'SLSQP', 'L-BFGS-B']:
            if time.time() - start_time > budget_s:
                break
            if method == 'COBYLA':
                res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            elif method == 'SLSQP':
                res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            else:
                res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
            h = res.x

        return best_h, best_c5

    patterns = generate_patterns()
    best_h, best_c5 = patterns[0].copy(), obj(patterns[0])
    start_time = time.time()

    for p in patterns[1:]:
        h, c5 = optimize_pattern(p)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return (best_h, best_c5, n_points)
