import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    def sample_guesses():
        guesses = [initial_h.copy()]
        for k in [3, 5, 7, 9]:
            s = np.zeros(n_points)
            for i in range(n_points):
                if i % k == 0:
                    s[i] = 1
            s = s * target_sum / np.sum(s) if np.sum(s) > 0 else np.zeros(n_points)
            guesses.append(s)
        for _ in range(2):
            r = np.random.rand(n_points)
            r = r * target_sum / np.sum(r) if np.sum(r) > 0 else np.zeros(n_points)
            guesses.append(r)
        return guesses

    guesses = sample_guesses()
    for guess in guesses:
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.05 * (0.95 ** _), n_points)
            new_h = np.clip(guess + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else guess
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01 * (0.95 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
