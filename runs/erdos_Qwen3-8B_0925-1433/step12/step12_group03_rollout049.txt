import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h > 0:
            return h * target_sum / sum_h
        return np.zeros_like(h)
    
    initial_guesses = [initial_h_values.copy()]
    start = np.random.uniform(0, 1)
    seg = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if start <= x < start + 1:
            seg[i] = 1
    seg = scale_h(seg)
    initial_guesses.append(seg)
    
    start1, start2 = np.random.uniform(0, 2 - 0.5, 2)
    seg = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if start1 <= x < start1 + 0.5:
            seg[i] = 1
        elif start2 <= x < start2 + 0.5:
            seg[i] = 1
    seg = scale_h(seg)
    initial_guesses.append(seg)
    
    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    
    for guess in initial_guesses:
        h = scale_h(guess)
        corr = np.correlate(h, 1 - h, mode='full')
        curr_c5 = np.max(corr) * dx
        if curr_c5 < best_c5:
            best_h, best_c5 = h, curr_c5
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
