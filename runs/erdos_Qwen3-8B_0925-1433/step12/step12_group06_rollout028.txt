    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + lag
            i_minus = i - lag
            term1 = 1.0 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            term2 = -h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 + term2
