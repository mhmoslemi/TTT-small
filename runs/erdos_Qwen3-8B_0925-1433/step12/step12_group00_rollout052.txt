        return -grad

    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)
    t = 0
    learning_rate = 0.1
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        t += 1
        grad = compute_gradient(h)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h = clip_and_normalize(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(lambda x: (compute_c5(x) if np.sum(x) < target_sum else np.inf),
                   best_h, method='SLSQP', bounds=[(0, 1)] * n_points,
                   constraints=constraints, tol=1e-8, options={'maxiter': 200})
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
