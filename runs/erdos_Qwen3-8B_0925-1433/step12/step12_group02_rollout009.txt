import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def rescale(x):
        s = np.sum(x)
        return x * target_sum / s if s != 0 else x

    population = [rescale(h)]
    for _ in range(50):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = rescale(candidate)
        population.append(candidate)

    best_c5 = float('inf')
    best_h = h.copy()

    for candidate in population:
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate

    start_time = time.time()
    while time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        result = minimize(lambda x: compute_c5(x), best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        else:
            break

        if time.time() - start_time > budget_s - 10:
            break

    return (best_h, best_c5, n_points)
