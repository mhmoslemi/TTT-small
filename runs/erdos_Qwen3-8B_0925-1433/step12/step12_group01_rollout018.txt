            pattern = pattern * target_sum / pattern_sum
        structured_patterns.append(pattern)

    bounds = [(0, 1)] * n_points
    result = differential_evolution(
        fun=compute_c5,
        bounds=bounds,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        popsize=20,
        tol=1e-6,
        maxiter=200,
        seed=seed
    )
    best_candidate = result.x
    best_candidate = np.clip(best_candidate, 0, 1)
    best_candidate_sum = np.sum(best_candidate)
    if best_candidate_sum > 0:
        best_candidate = best_candidate * target_sum / best_candidate_sum
    current_c5 = compute_c5(best_candidate)
    if current_c5 < best_c5:
        best_h = best_candidate.copy()
        best_c5 = current_c5

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    def obj(x):
        return compute_c5(x)

    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h = result.x.copy()
        best_c5 = compute_c5(best_h)

    result = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h = result.x.copy()
        best_c5 = compute_c5(best_h)

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success:
            best_h = result.x.copy()
            best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
