from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i + k_max < len(h):
                term1 = 1.0 - h[i + k_max]
            else:
                term1 = 0.0
            if 0 <= i - k_max < len(h):
                term2 = h[i - k_max]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return -grad

    learning_rate = 0.5
    decay_rate = 0.998
    max_iterations = 500
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        direction = grad
        step = learning_rate
        for _ in range(5):
            trial_h = h + step * direction
            trial_h = clip_and_normalize(trial_h)
            corr_trial = np.max(np.correlate(trial_h, 1 - trial_h, mode='full')) * dx
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * direction
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:

        learning_rate *= decay_rate

    def obj_lbgfs(x):
