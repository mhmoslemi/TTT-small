            break
        # Use L-BFGS-B with symmetry enforcement
        res = minimize(lambda x: max(np.correlate(x, 1 - x, mode='full')) * dx,
                       best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
        if res.success:
            h_candidate = res.x.copy()
            for i in range(n_points // 2):
                h_candidate[n_points - 1 - i] = 1 - h_candidate[i]
            if n_points % 2 == 1:
                h_candidate[n_points // 2] = 0.5
            sum_h = np.sum(h_candidate)
            if sum_h != 0:
                h_candidate = h_candidate * target_sum / sum_h
            else:
                h_candidate = np.zeros(n_points) + 0.5
            current_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_candidate
                best_c5 = current_c5

    # Final symmetry check
    for i in range(n_points // 2):
        best_h[n_points - 1 - i] = 1 - best_h[i]
    if n_points % 2 == 1:
        best_h[n_points // 2] = 0.5
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
