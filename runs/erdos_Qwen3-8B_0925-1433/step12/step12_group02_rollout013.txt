import numpy as np

def run(seed=42, budget_s=600):
    import time
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 45
        h = np.random.rand(n_points)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    else:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    alpha = 0.2  # initial step size
    decay_factor = 0.998  # decay alpha over iterations
    improved = True
    iterations = 0
    while time.time() - start_time < budget_s and improved:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag_k = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for m in range(n_points):
            k = lag_k
            term1 = 0.0
            if m + k < n_points:
                term1 = dx * (1 - h[m + k])
            term2 = 0.0
            if m - k >= 0:
                term2 = dx * h[m - k]
            grad[m] = term1 - term2
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        current_sum = np.sum(h_new)
        scale_factor = target_sum / current_sum
        h_new = h_new * scale_factor
        # compute new c5
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            improved = True
        else:
            improved = False
        h = h_new.copy()
        alpha *= decay_factor  # decay step size
    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600):

    decay_factor = 0.998
