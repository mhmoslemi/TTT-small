        current_corr = np.correlate(best_h, 1 - best_h, mode='full')

        lag = max_idx - (n_points - 1)
        mask = np.zeros(n_points)
        for x in range(n_points):
            x_val = x * dx
            x_plus_k = x_val + lag
            if 0 <= x_plus_k <= 2:
                y = int(np.round(x_plus_k / dx))
                if 0 <= y < n_points:
                    mask[x] = best_h[x] * (1 - best_h[y])
        perturb = -0.0005 * mask
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
