    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        t += 1
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        k_max = idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = (i + k_max) % n_points
            i_minus = (i - k_max) % n_points
            grad[i] = (1 - h[i_plus]) - h[i_minus]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h = clip_and_normalize(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.abs(current_c5 - best_c5) < 1e-8:
            break

    def obj_lbgfs(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=constraints, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
