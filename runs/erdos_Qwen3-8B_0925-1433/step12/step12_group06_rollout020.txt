import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        current_c5 = corr[k_star] * dx

            corr = np.correlate(h_, 1 - h_, mode='full')

            sum_h = np.sum(h)
            if sum_h > 0:
