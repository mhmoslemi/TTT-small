import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
            h = candidate.copy()
        else:
            h = best_h.copy()

        res = minimize(compute_c5, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success:
            h = res.x
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break

    return (best_h, best_c5, n_points)
