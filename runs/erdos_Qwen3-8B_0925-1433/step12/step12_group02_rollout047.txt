from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    if initial_h_values is None:

        h = initial_h_values.copy()

    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = Bounds(0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Local refinement
        res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            best_h = res.x
