import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate a new candidate with noise
        noise = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h + noise, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        c5_candidate = compute_c5(candidate)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = candidate.copy()
        
        # Optimization with bounds and equality constraint
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use COBYLA for derivative-free optimization
        res = minimize(
            compute_c5,
            best_h,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        
        if res.success:
            best_h = res.x
            best_c5 = compute_c5(best_h)
        else:
            # Fallback to last best if optimization fails
            pass

    return (best_h, best_c5, n_points)
