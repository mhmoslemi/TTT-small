import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_pattern(pattern_type):
        if pattern_type == 'grid':
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1.0 if (i % 5 == 0) else 0.0
            return h
        elif pattern_type == 'fib':
            h = np.zeros(n_points)
            fib = np.array([1, 1])
            while len(fib) < n_points:
                fib = np.append(fib, fib[-1] + fib[-2])
            h = np.sin(fib / n_points * np.pi) * 0.5 + 0.5
            return h
        elif pattern_type == 'sin':
            h = 0.5 * (1 + np.sin(np.linspace(0, np.pi, n_points)))
            return h
        elif pattern_type == 'rand':
            h = np.random.rand(n_points)
            return h
        else:
            return np.zeros(n_points)

    initial_patterns = [
        create_pattern('grid'),
        create_pattern('fib'),
        create_pattern('sin'),
        create_pattern('rand'),
        initial_h_values.copy()
    ]
    initial_patterns = [p * target_sum / np.sum(p) for p in initial_patterns]

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_patterns[0]
    best_c5 = obj(best_h)
    start_time = time.time()

    for i, pattern in enumerate(initial_patterns):
        if time.time() - start_time > budget_s:
            break
        x0 = pattern
        res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.005 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
