import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Preprocess initial_h_values
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)

    POP_SIZE = 50
    MUTATION_FACTOR = 0.8
    CROSSOVER_RATE = 0.9
    MAX_GENERATIONS = 100

    # Initialize population
    pop = [h_initial + np.random.uniform(-0.2, 0.2, n_points) for _ in range(POP_SIZE)]
    pop = [np.clip(h, 0, 1) for h in pop]
    pop = [h * target_sum / np.sum(h) if np.sum(h) > 0 else h_initial for h in pop]

    best_h = h_initial
    best_c5 = np.correlate(h_initial, 1 - h_initial, mode='full').max() * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(MAX_GENERATIONS):
        if time.time() - start_time > budget_s:
            break
        for i in range(POP_SIZE):
            idxs = np.random.choice(POP_SIZE, 3, replace=False)
            h1, h2, h3 = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = h1 + MUTATION_FACTOR * (h2 - h3)
            mutant = np.clip(mutant, 0, 1)
            trial = np.copy(pop[i])
            for j in range(n_points):
                if np.random.rand() < CROSSOVER_RATE:
                    trial[j] = mutant[j]
            trial_sum = np.sum(trial)
            if trial_sum > 0:
                trial = trial * target_sum / trial_sum
            else:
                trial = h_initial
            current_c5 = objective(pop[i])
            trial_c5 = objective(trial)
            if trial_c5 < current_c5:
                pop[i] = trial
                if trial_c5 < best_c5:
                    best_h = trial
                    best_c5 = trial_c5
    return (best_h, best_c5, n_points)
