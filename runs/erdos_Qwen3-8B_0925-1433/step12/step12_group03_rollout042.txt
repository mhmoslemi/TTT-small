def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    if np.sum(h_initial) > 0:
        h_initial = h_initial * target_sum / np.sum(h_initial)
    else:
        h_initial = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h_initial.copy()
    best_c5 = obj(best_h)

    start_time = time.time()

    # Generate structured initial patterns
    def generate_pattern():
        base = np.zeros(n_points)
        base[::3] = 1.0
        base[::7] = 0.5
        base = np.clip(base, 0, 1)
        sum_b = np.sum(base)
        if sum_b > 0:
            base = base * target_sum / sum_b
        return base

    def generate_sine():
        x = np.linspace(0, 2, n_points)
        pattern = 0.5 * (1 + np.sin(np.pi * x))
        pattern = np.clip(pattern, 0, 1)
        sum_p = np.sum(pattern)
        if sum_p > 0:
            pattern = pattern * target_sum / sum_p
        return pattern

    initial_patterns = [
        h_initial,
        generate_pattern(),
        generate_sine()
    ]

    for pattern in initial_patterns:
        x0 = pattern.copy()
        result = minimize(obj, x0, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.005 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h.copy()
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h.copy()
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
