import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def thue_morse(n):
        tm = np.zeros(n, dtype=int)
        for i in range(n):
            tm[i] = (tm[i // 2] + i % 2) % 2
        return tm

    def repair_h(h, target_sum):
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h) if np.sum(h) > 0 else 1.0
        return h * scale

    # Initial guess
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
    else:
        tm = thue_morse(n_points)
        h = tm.astype(float)
        h = repair_h(h, target_sum)

    def obj(x):

        sum_h = np.sum(h_)
        if sum_h > 0:

        cross_corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(cross_corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()

    # Phase 1: Global search with L-BFGS-B
    if time.time() - start_time < budget_s * 0.4:
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Phase 2: Local refinement with SLSQP
    if time.time() - start_time < budget_s * 0.6:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

    # Phase 3: Fine-tuning with COBYLA
    if time.time() - start_time < budget_s * 0.8:
        res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
