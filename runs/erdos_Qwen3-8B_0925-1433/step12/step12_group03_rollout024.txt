import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = [
        np.clip(initial_h_values, 0, 1) * target_sum / np.sum(initial_h_values) if np.sum(initial_h_values) > 0 else np.zeros(n_points),
        np.zeros(n_points),  # baseline
        np.random.rand(n_points) * 2 - 1,  # random
        np.zeros(n_points),  # sparse
        np.zeros(n_points),  # dense
    ]

    best_h = initial_guesses[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for i, guess in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
        h = guess.copy()

        # Local optimization
        def obj(x):
            return evaluate(x)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        # Refinement
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.001, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
