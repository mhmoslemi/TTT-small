import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h_vals):
        sum_h = np.sum(h_vals)
        if abs(sum_h) > 1e-12:
            return np.clip(h_vals * target_sum / sum_h, 0, 1)
        else:
            return np.zeros_like(h_vals)

    h = scale_h(h)

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if abs(sum_x) > 1e-12:
            x_scaled = x_clipped * target_sum / sum_x
        else:
            x_scaled = np.zeros_like(x_clipped)
        cross_corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(cross_corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02 * (0.9 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005 * (0.9 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
