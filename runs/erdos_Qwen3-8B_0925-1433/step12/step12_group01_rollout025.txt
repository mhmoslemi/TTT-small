import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def optimize_phase(method, *args, **kwargs):
        nonlocal best_h, best_c5
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.05 * (0.9 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, **kwargs)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    optimize_phase('SLSQP', tol=1e-6, max_iter=200)
    optimize_phase('COBYLA', tol=1e-8, maxiter=150)
    optimize_phase('L-BFGS-B', tol=1e-8, maxiter=100)

    return (best_h, best_c5, n_points)
