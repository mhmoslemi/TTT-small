from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

            sum_h_ = np.sum(h_)

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        for phase in range(3):

            if phase == 0:
                method, tol = 'L-BFGS-B', 1e-6
            elif phase == 1:
                method, tol = 'SLSQP', 1e-5

                method, tol = 'COBYLA', 1e-6

            perturb = np.random.normal(0, perturb_std, n_points)

            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
