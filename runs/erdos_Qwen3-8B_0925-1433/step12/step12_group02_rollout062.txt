import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()

    population = [h]
    for _ in range(10):
        perturb = np.random.normal(0, 0.2, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)

    best_c5 = float('inf')
    for candidate in population:
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            h = candidate.copy()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.15, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            h = candidate.copy()

            bounds = [(0, 1)] * n_points

            if res.success:
                h = res.x
                best_c5 = compute_c5(h)
            else:
                break

    return (h, best_c5, n_points)
