import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    alpha = 0.1
    start_time = time.time()
    iters = 0
    max_no_improve = 100

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        lag = k_index - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + lag >= 0 and m + lag < n_points:
                grad[m] += dx * (1 - h[m + lag])
            if m - lag >= 0:
                grad[m] -= dx * h[m - lag]
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / h_sum)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        iters += 1
        if iters > max_no_improve:
            break
    return (best_h, best_c5, n_points)
