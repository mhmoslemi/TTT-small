from scipy.optimize import minimize
import time

    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def analytical_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        m = np.argmax(corr)
        k = m - (n_points - 1)

        for j in range(n_points):

        return -grad

    learning_rate = 0.1

            step *= 0.5
        if step < 1e-10:
