        corr = np.correlate(h_current, 1 - h_current, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        h_current = np.clip(best_h, 0, 1)
        if np.sum(h_current) > 0:
            h_current = h_current * target_sum / np.sum(h_current)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_star = np.argmax(corr)
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = best_h.copy()
            h_perturb[i] += eps
            h_perturb = np.clip(h_perturb, 0, 1)
            if np.sum(h_perturb) > 0:
                h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            grad[i] = (np.max(corr_perturb) - np.max(corr)) / eps
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, jac=grad, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    return (best_h, best_c5, n_points)
