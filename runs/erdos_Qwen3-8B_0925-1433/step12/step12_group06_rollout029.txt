import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(250):
        if time.time() - start_time > budget_s:
            break
        # Step 1: Perturb and optimize
        noise = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + noise, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        # Step 2: Coarse-tune with COBYLA
        if _ % 5 == 0:
            res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

        # Step 3: Random restarts
        if _ % 20 == 0:
            noise = np.random.uniform(-0.1, 0.1, n_points)
            new_h = np.clip(best_h + noise, 0, 1)
            if np.sum(new_h) > 0:
                new_h = new_h * target_sum / np.sum(new_h)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
