import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * target_sum / np.sum(h)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_normalize(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            x = x * target_sum / s
        return x

    def tournament_selection(pop, fitness, k=3):
        indices = np.random.choice(len(pop), k)
        best = indices[np.argmin(fitness[indices])]
        return pop[best].copy()

    def crossover(parent1, parent2):
        points = np.random.choice(range(1, n_points), size=np.random.randint(2, 4), replace=False)
        points.sort()
        child1 = np.concatenate([parent1[:points[0]], parent2[points[0]:points[1]], parent1[points[1]:points[2]], parent2[points[2]:]])
        child2 = np.concatenate([parent2[:points[0]], parent1[points[0]:points[1]], parent2[points[1]:points[2]], parent1[points[2]:]])
        return [clip_and_normalize(child1), clip_and_normalize(child2)]

    population = [clip_and_normalize(np.random.choice([0, 1], size=n_points)) for _ in range(20)]
    for _ in range(100):
        if time.time() - time.time() > budget_s:
            break
        fitness = np.array([evaluate_c5(ind) for ind in population])
        parents = [tournament_selection(population, fitness) for _ in range(10)]
        new_population = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            child1, child2 = crossover(p1, p2)
            new_population.extend([child1, child2])
        new_population = [clip_and_normalize(np.random.normal(0, 0.15, n_points)) for _ in range(len(population) - len(new_population))]
        new_population += new_population
        population = new_population
        if np.min(fitness) < evaluate_c5(h):
            h = population[np.argmin(fitness)]

    def obj_func(x):
        return evaluate_c5(x)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1)] * n_points
    res = minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
