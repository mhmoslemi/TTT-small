import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 20
    pop = np.zeros((pop_size, n_points))
    pop[0] = initial_h_values.copy()
    for i in range(1, pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) == 0:
            candidate = np.full(n_points, 0.5)
        else:
            candidate = candidate * target_sum / np.sum(candidate)
        pop[i] = candidate

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            F = 0.8
            mutant = pop[c] + F * (pop[b] - pop[a])
            trial = mutant * np.random.rand(n_points) < 0.9
            trial = np.where(trial, mutant, pop[i])
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) == 0:
                trial = np.full(n_points, 0.5)
            else:
                trial = trial * target_sum / np.sum(trial)
            c5 = compute_c5(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
            if c5 < compute_c5(pop[i]):
                pop[i] = trial.copy()

    return (best_h, best_c5, n_points)
