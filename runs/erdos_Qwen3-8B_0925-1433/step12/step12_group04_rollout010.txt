import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def apply_symmetry(h):
        h = h.copy()
        for i in range(n_points // 2):
            h[n_points - 1 - i] = 1.0 - h[i]
        return h

    def scale_h(h, target_sum):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (target_sum / s)

    h = np.array(initial_h_values, dtype=float)
    h = apply_symmetry(h)
    h = scale_h(h, target_sum)

    def obj(x):
        x = apply_symmetry(x)
        x = scale_h(x, target_sum)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(800):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = apply_symmetry(new_h)
        new_h = scale_h(new_h, target_sum)
        current_c5 = obj(new_h)
        if current_c5 < best_c5:
            best_h, best_c5 = new_h, current_c5

    return (best_h, best_c5, n_points)
