import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        if np.sum(h) > 0:
            return h * target_sum / np.sum(h)
        return np.zeros_like(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = scale_h(initial_h_values.copy())
    best_h = initial_guess
    best_c5 = compute_c5(best_h)

    def optimize(start):
        def obj(x):
            h_ = np.clip(x, 0, 1)
            h_ = scale_h(h_)
            return compute_c5(h_)
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 200})
        if res.success:
            return res.x
        return start

    def generate_structured():
        n_ones = n_points // 2
        idx = np.sort(np.random.choice(n_points, n_ones, replace=False))
        h = np.zeros(n_points)
        h[idx] = 1
        return scale_h(h)

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        guess = generate_structured() if np.random.rand() < 0.5 else initial_guess
        h = optimize(guess)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    for _ in range(100):

        new_h = scale_h(np.clip(best_h + perturb, 0, 1))
        c5 = compute_c5(new_h)
