import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        s = np.sum(h)
        if s == 0:
            return h
        return h * target_sum / s

    h = np.clip(initial_h_values, 0, 1)
    h = rescale(h)

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()

    def evaluate(h_current):
        h_norm = rescale(h_current)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx, corr

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        c5, corr = evaluate(best_h)
        if c5 < best_c5:
            best_c5 = c5
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        def obj(x):
            h_ = np.clip(x, 0, 1)
            h_norm = rescale(h_)
            corr = np.correlate(h_norm, 1 - h_norm, mode='full')
            return corr[max_idx] * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
