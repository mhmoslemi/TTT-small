import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum != 0:
            x = x * target_sum / x_sum
        return x

    # Structured initializations
    def spaced_pattern(m, spacing):
        positions = np.linspace(0, 2, m + 1)
        intervals = np.zeros(n_points)
        for i in range(m):
            start = int(positions[i] / dx)
            end = int(positions[i + 1] / dx)
            intervals[start:end] = 1.0
        return clip_and_scale(intervals)

    def random_pattern():
        h = np.random.rand(n_points)
        h = clip_and_scale(h)
        return h

    # Generate structured and randomized initial guesses
    initial_guesses = [
        h_initial,
        spaced_pattern(10, 1.0),
        spaced_pattern(15, 0.8),
        spaced_pattern(20, 0.6),
        random_pattern(),
    ]

    best_h = initial_guesses[0]
    best_c5 = float('inf')
    start_time = time.time()

    def obj(x):
        x = clip_and_scale(x)
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for init in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        if best_c5 < 0.38:
            break

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = clip_and_scale(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
