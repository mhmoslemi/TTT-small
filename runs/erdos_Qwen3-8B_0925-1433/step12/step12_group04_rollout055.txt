import numpy as np
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    np.random.seed(seed)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = initial_h_values.copy()

    h = np.clip(h, 0, 1)

    best_h = h.copy()

    start_time = time.time()
    learning_rate = 0.1

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break

        k_max_index = np.argmax(corr)
        k_crit = k_max_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k_crit) % n_points
            j_minus = (i - k_crit) % n_points
            grad[i] = (1 - h[j_plus]) - (1 - h[j_minus])

        sum_h = np.sum(h)
        if sum_h > 0:
