def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        else:
            candidate = np.full(n_points, 0.5)
        pop[i] = candidate

    def evaluate_candidate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = np.inf
    best_h = pop[0].copy()
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = c + 0.8 * (b - a)
            trial = np.clip(mutant, 0, 1)
            if np.sum(trial) > 0:
                trial = trial * target_sum / np.sum(trial)
            else:
                trial = np.full(n_points, 0.5)
            new_pop.append(trial)
        new_c5s = [evaluate_candidate(h) for h in new_pop]
