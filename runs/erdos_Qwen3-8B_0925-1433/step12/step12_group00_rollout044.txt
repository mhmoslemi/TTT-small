    h = np.array(initial_h_values, dtype=float)

                gradient[m_idx] += (1 - h[m_plus_k_mod]) - h[m_minus_k_mod]
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * target_sum / sum_h
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= 0.999

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1)] * n_points

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        best_h = best_h * target_sum / sum_h
    return (best_h, best_c5, n_points)
