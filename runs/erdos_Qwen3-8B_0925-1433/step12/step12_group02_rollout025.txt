import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    population = [h]
    for _ in range(20):
        perturb = np.random.normal(0, 0.2, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)

    best_c5 = float('inf')
    best_h = np.copy(h)
    for candidate in population:
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = minimize(lambda x: compute_c5(x), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            current_c5 = compute_c5(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break

    return (best_h, best_c5, n_points)
