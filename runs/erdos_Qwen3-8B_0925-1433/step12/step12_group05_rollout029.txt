import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for step in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            j_plus = (j + lag) % n_points
            j_minus = (j - lag) % n_points
            gradient[j] = (1 - h[j_plus]) - h[j_minus]
        h -= 0.05 * gradient
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = best_h.copy()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if step % 100 == 0:
            step_size = max(0.001, 0.05 * (0.99 ** step))

    return (best_h, best_c5, n_points)
