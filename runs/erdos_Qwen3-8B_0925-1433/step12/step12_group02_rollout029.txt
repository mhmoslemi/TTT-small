import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.inf

    start_time = time.time()
    iteration = 0
    alpha = 0.1

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_index = np.argmax(corr)
        delta = lag_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if i + delta < n_points:
                term1 = (1 - h[i + delta]) * dx
            term2 = 0.0
            if i - delta >= 0:
                term2 = h[i - delta] * dx
            grad[i] = term1 - term2

        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * target_sum / sum_h

        current_c5 = max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
