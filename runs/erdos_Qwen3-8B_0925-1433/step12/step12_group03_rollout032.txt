import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_even_pattern(n_points):
        h = np.zeros(n_points)
        h[::2] = 1
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if delta > 0:
            h[-1] += delta
        elif delta < 0:
            h[-1] += delta
        h = np.clip(h, 0, 1)
        return h

        h_rand = h_rand / np.sum(h_rand) * target_sum

        initial_guesses.append(h_rand)

    best_h = np.copy(initial_h_values)
    best_c5 = np.inf
