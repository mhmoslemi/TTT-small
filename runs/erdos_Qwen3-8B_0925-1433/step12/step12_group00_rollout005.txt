        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
