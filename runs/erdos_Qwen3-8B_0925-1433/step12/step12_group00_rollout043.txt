from scipy.optimize import minimize
import time

            h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

        elite_idx = np.argsort(fitness)[:int(0.4 * pop_size)]
