from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guesses with structured patterns
    def structured_pattern(n_blocks):
        h = np.zeros(n_points)
        block_size = n_points // n_blocks
        rem = n_points % n_blocks
        for i in range(n_blocks):
            start = i * block_size + min(i, rem)
            end = (i + 1) * block_size + min(i + 1, rem)
            h[start:end] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    # Initial guesses
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)

    # Add structured initial guesses
    for n_blocks in [3, 4, 5, 6]:
        h_struct = structured_pattern(n_blocks)
        if np.sum(h_struct) > 0:
            h = h_struct * target_sum / np.sum(h_struct)
            initial_guesses.append(h.copy())
    initial_guesses.append(h.copy())

    # Objective function
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    # Constraints
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
