import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def get_max_lag(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        lag_idx = np.argmax(corr)
        return lag_idx, max_corr

    for stage in range(3):
        if time.time() - start_time > budget_s:
            break
        if stage == 0:
            solver = 'L-BFGS-B'
            perturb_std = 0.01
            iters = 150
        elif stage == 1:
            solver = 'COBYLA'
            perturb_std = 0.002
            iters = 100
        else:
            solver = 'SLSQP'
            perturb_std = 0.0005
            iters = 50

        for _ in range(iters):
            if time.time() - start_time > budget_s:
                break
            lag_idx, _ = get_max_lag(best_h)
            if lag_idx >= len(best_h):
                lag_idx = len(best_h) - 1
            lag = lag_idx - n_points + 1 if lag_idx >= n_points else lag_idx
            perturb = np.zeros(n_points)
            mask = np.zeros(n_points, dtype=bool)
            for i in range(n_points):
                if i + lag >= 0 and i + lag < n_points:
                    mask[i] = True
            perturb[mask] = np.random.normal(0, perturb_std, np.sum(mask))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method=solver, bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
