import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for i in range(300):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.1 * (0.9 ** i) if i < 100 else 0.01 * (0.9 ** (i - 100)) if i < 200 else 0.001 * (0.9 ** (i - 200))
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h

        if i < 100:
            method = 'Nelder-Mead'
        elif i < 200:
            method = 'L-BFGS-B'
        else:
            method = 'COBYLA'

        res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.0005 * (0.9 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
