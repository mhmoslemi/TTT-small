import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Scale initial guess to meet the sum constraint
    h = np.clip(initial_h, 0, 1)
    sum_h = np.sum(h)
    h = h * target_sum / sum_h if sum_h > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'SLSQP'
            tol = 1e-5
            iter_count = 300
        elif phase == 1:
            method = 'COBYLA'
            tol = 1e-5
            iter_count = 200
        else:
            method = 'L-BFGS-B'
            tol = 1e-6
            iter_count = 150

        for _ in range(iter_count):
            if time.time() - start_time > budget_s:
                break
            if _ < 100 or _ % 20 == 0:
                perturb = np.random.normal(0, 0.02 * (0.98 ** _), n_points)
            else:
                perturb = np.random.normal(0, 0.005 * (0.98 ** _), n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            sum_new = np.sum(new_h)
            new_h = new_h * target_sum / sum_new if sum_new > 0 else best_h

            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
