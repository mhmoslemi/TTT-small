        if np.random.rand() < 0.1:
            perturb = np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h + perturb, 0, 1)
            sum_perturb = np.sum(h_perturb)
            if sum_perturb > 0:
                h_perturb = h_perturb * target_sum / sum_perturb
            else:
                h_perturb = h.copy()
            new_c5_perturb = compute_c5(h_perturb, dx)
            if new_c5_perturb < best_c5:
                h = h_perturb
                best_c5 = new_c5_perturb
                best_h = h.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
