import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_guesses = [initial_h_values.copy()]
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if i % 2 else 0
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.cos(2 * np.pi * x))
    h = h * target_sum / (np.sum(h) * dx)
    initial_guesses.append(h)
    positions = np.sort([( (np.sqrt(5)-1)/2 * i ) % 1 for i in range(n_points+1)])
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if (positions[i] % 1) < 0.5 else 0
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    best_h = initial_guesses[0]
    best_c5 = obj(best_h)

    start_time = time.time()
    for guess in initial_guesses:
        res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        if time.time() - start_time > budget_s:

    for _ in range(100):

        if res.success and res.fun < best_c5:
