import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = []
    if 'initial_h_values' in globals():

        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)

    for _ in range(19):

        idxs = np.random.choice(n_points, size=n_points//2, replace=False)
        h[idxs] = 1.0
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Genetic Algorithm
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        parents = np.argsort(fitness)[:10]
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = population[parents[i]], population[parents[i+1]]
            points = np.random.randint(1, n_points-1, size=np.random.randint(2,4))
            points.sort()

            for p in points:
                segments.append(p1[prev:p])
                segments.append(p2[prev:p])
                prev = p
            segments.append(p1[prev:])
            segments.append(p2[prev:])
            np.random.shuffle(segments)
