import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.full(n_points, 0.5)
        return h_clipped * target_sum / sum_clipped

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Global exploration with differential evolution
    de_result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=15, tol=1e-6, mutation=(0.5, 1), 
        recombination=0.9, seed=seed, 
        maxiter=150, dispatch='sequential'
    )

    best_h = de_result.x
    best_c5 = objective(best_h)

    # Local refinement with SLSQP
    def cons(x):
        return np.sum(x) - target_sum

    cons = {'type': 'eq', 'fun': cons}
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Local refinement with COBYLA
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='COBYLA', 
                      bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final perturbations with L-BFGS-B
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = clip_rescale(new_h)
        res = minimize(objective, new_h, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final clipping and rescaling
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
