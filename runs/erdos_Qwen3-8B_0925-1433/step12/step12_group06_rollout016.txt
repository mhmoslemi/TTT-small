from scipy.optimize import minimize
import time

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Stage 1: Nelder-Mead for global exploration
    for i in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.02 * (0.9 ** i)
        perturb = np.random.normal(0, perturb_std, n_points)

        res = minimize(obj, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:

    # Stage 2: L-BFGS-B for refinement
