import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce symmetry and rescale initial_h_values
    h = np.array(initial_h_values, copy=True)
    mid = n_points // 2
    for i in range(mid):
        h[i] = 1 - h[n_points - 1 - i]
    if n_points % 2 == 1:
        h[mid] = 0.5
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h == 0:
            return float('inf')
        h_ = h_ * target_sum / sum_h
        # Enforce symmetry
        for i in range(mid):
            h_[i] = 1 - h_[n_points - 1 - i]
        if n_points % 2 == 1:
            h_[mid] = 0.5
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Adaptive L-BFGS-B optimization with symmetry
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        # Enforce symmetry and rescale
        for i in range(mid):
            new_h[i] = 1 - new_h[n_points - 1 - i]
        if n_points % 2 == 1:
            new_h[mid] = 0.5
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
