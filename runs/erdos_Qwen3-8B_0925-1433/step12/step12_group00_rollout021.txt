        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        lag = idx_max - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + lag
            term1 = 1.0 - h[i_plus_k] if 0 <= i_plus_k < n_points else 0.0
            i_minus_k = i - lag
            term2 = h[i_minus_k] if 0 <= i_minus_k < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        return analytical_gradient(h)

    # Adam optimizer parameters
    learning_rate = 0.1
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    max_iterations = 500
    start_time = time.time()

    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        t += 1
        grad = compute_gradient(h)
        # Update moving averages
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Update h
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        # Clip and normalize
        h = clip_and_normalize(h)
        # Update best
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        # Decay learning rate
        learning_rate *= 0.999

    # L-BFGS refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
