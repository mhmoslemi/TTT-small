from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_initial_guesses():
        guesses = []
        h = initial_h_values.copy()
        guesses.append(h)
        # Alternating pattern
        alt = np.zeros(n_points)
        alt[::2] = 1.0
        alt = alt * target_sum / np.sum(alt)
        guesses.append(alt)
        # Sine wave
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
        sin_h = np.clip(sin_h, 0, 1)
        sin_h = sin_h * target_sum / np.sum(sin_h)
        guesses.append(sin_h)
        # Random binary
        rand = np.random.choice([0, 1], size=n_points)
        rand = rand * target_sum / np.sum(rand)
        guesses.append(rand)
        return guesses

    guesses = make_initial_guesses()
    best_h = np.copy(initial_h_values)
    best_c5 = evaluate_erdos_solution(best_h)

    for guess in guesses:
        h = np.copy(guess)
        h = h * target_sum / np.sum(h)
        def obj(x):
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1-x, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        for _ in range(200):
            if time.time() > budget_s:
                break
            perturb = np.random.normal(0, 0.02 * (0.95**_), size=n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
