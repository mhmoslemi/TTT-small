import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values)
    h = h * target_sum / np.sum(h)

    def perturb(h):
        noise = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h + noise, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        return candidate

    population = [h]
    for _ in range(10):
        population.append(perturb(h))

    best_c5 = float('inf')
    best_h = h.copy()
    for candidate in population:
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(lambda x: compute_c5(x), best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        else:
            break

    assert np.all(0 <= best_h <= 1.0), "H values out of bounds"
    assert np.isclose(np.sum(best_h), target_sum), "Sum constraint violated"
    return (best_h, best_c5, n_points)
