from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_block_pattern(n_blocks):
        h = np.zeros(n_points)
        block_size = 2.0 / n_blocks
        for i in range(n_blocks):
            start = int(i * block_size * n_points / 2)
            end = int((i + 1) * block_size * n_points / 2)
            if i % 2 == 0:
                h[start:end] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    def generate_checkerboard_pattern(period):
        h = np.zeros(n_points)
        for i in range(n_points):
            if (i * 2) % period == 0:
                h[i] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    def obj(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return np.sum(h) - target_sum

    if 'initial_h_values' in globals():
        h_init = np.clip(initial_h_values, 0, 1)
        h_init = h_init * target_sum / np.sum(h_init) if np.sum(h_init) > 0 else np.zeros(n_points)
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init * target_sum / np.sum(h_init)

    bounds = [(0, 1)] * n_points
    candidates = [h_init]

    for blocks in [3, 5, 7, 9, 11]:
        candidates.append(generate_block_pattern(blocks))

    best_h = h_init

    de_bounds = [(0, 1)] * n_points

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-8)
