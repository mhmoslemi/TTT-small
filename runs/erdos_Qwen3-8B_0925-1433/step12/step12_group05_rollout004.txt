    step_size = 0.1
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        i_max = np.argmax(corr)
        k = i_max - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] = dx * (1 - h[j])
        h = h - step_size * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_h, best_c5 = h.copy(), new_c5
        step_size *= 0.995
    return (best_h, best_c5, n_points)
