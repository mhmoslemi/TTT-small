import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate perturbed candidates
        for _ in range(5):
            perturb = np.random.normal(0, 0.05, n_points)
            candidate = np.clip(h + perturb, 0, 1)
            candidate = candidate * target_sum / np.sum(candidate)
            current_c5 = compute_c5(candidate)
            if current_c5 < best_c5:

        if res.success:

            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
