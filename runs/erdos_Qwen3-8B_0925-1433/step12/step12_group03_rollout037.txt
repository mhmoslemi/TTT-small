import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_sinusoidal(n_points):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    patterns = [initial_h_values.copy(), generate_sinusoidal(n_points)]

    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    for pattern in patterns:
        h = np.clip(pattern, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        start_time = time.time()
        current_h = h.copy()
        current_c5 = obj(current_h)

        for _ in range(250):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.005 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_h
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < current_c5:
                current_h, current_c5 = res.x, res.fun

        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.001 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < current_c5:
                current_h, current_c5 = res.x, res.fun

        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.0005 * (0.95 ** _)

            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.0005 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else current_h
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < current_c5:
                current_h, current_c5 = res.x, res.fun

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    return (best_h, best_c5, n_points)
