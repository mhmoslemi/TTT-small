import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points, dx = len(initial_h_values), 2.0 / len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values).copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h, best_c5 = h.copy(), obj(h)
    
    try:
        for _ in range(4):
            if time.time() - time.time() > budget_s:
                break
            perturb = np.random.normal(0, 0.02 if _ < 2 else 0.005, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_c5, best_h = res.fun, res.x
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:

    except: pass

    return (best_h, best_c5, n_points)
