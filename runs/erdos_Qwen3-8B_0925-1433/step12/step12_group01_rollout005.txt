import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def initialize_population(n_particles):
        return [np.clip(np.random.rand(n_points), 0, 1) * target_sum / np.sum(np.clip(np.random.rand(n_points), 0, 1)) for _ in range(n_particles)]

    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def fallback_pattern():
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = clip_rescale(h)
        return h

    n_particles = 20
    particles = initialize_population(n_particles)
    pbest = particles.copy()
    pbest_fitness = [evaluate_fitness(h) for h in pbest]
    gbest = min(zip(pbest_fitness, pbest), key=lambda x: x[0])[1]
    gbest_fitness = min(pbest_fitness)
    best_h, best_c5 = gbest, gbest_fitness
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = clip_rescale(particles[i])
            current_fitness = evaluate_fitness(particles[i])

                pbest_fitness[i] = current_fitness
            if current_fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = current_fitness
                best_h, best_c5 = gbest, gbest_fitness
