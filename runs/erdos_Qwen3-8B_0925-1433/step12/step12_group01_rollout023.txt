import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def fallback_h():
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy() if hasattr(initial_h_values, '__len__') else fallback_h()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = evaluate(best_h)

    def generate_initial_population():
        pop = []
        for b in [3, 5, 7, 9, 11]:
            h = np.zeros(n_points)
            w = n_points // b
            h[:w] = 1.0
            h = h * target_sum / np.sum(h)
            pop.append(h)
        for _ in range(15):
            h = np.random.choice([0, 1], size=n_points)
            h = h * target_sum / np.sum(h)
            pop.append(h)
        return pop

    pop = generate_initial_population()

    for gen in range(100):
        if time.time() - time.time() > budget_s:
            break
        fitness = [evaluate(ind) for ind in pop]
        ind_fitness = sorted(zip(pop, fitness), key=lambda x: x[1])
        top_candidates = ind_fitness[:5]
        selected = [ind for ind, _ in top_candidates]

        new_pop = []
        while len(new_pop) < len(pop):
            p1, p2 = np.random.choice(selected, 2, replace=False)

            child = np.split(p1, cross_points)

    top_candidates = sorted(pop, key=evaluate)[:5]
