import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population: 100 perturbed samples
    pop_size = 100
    initial_h = np.array(initial_h_values, dtype=float)
    population = []
    for _ in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        h = initial_h + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    # GA parameters
    generations = 80
    mutation_rate = 0.1
    crossover_rate = 0.5
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        parents = [population[i] for i in sorted_idx[:int(0.4 * pop_size)]]

        new_pop = []
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                split = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
            else:
                child1, child2 = np.random.choice(parents, 2, replace=False)

            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                        child[i] = np.clip(child[i], 0, 1)
                child = child * target_sum / np.sum(child)
            new_pop.append(child1)
            new_pop.append(child2)

        population = new_pop[:pop_size]
        for h in population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    # Local refinement
    top_candidates = sorted(population, key=compute_c5)[:5]
    for candidate in top_candidates:
        def obj_lbgfs(x):
            h = np.clip(x, 0, 1)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj_lbgfs, candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
