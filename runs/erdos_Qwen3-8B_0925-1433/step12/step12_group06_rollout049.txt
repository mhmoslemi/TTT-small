from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    iteration = 0
    while time.time() - start_time < budget_s:
        if iteration > 150:
            break

        if res.success and res.fun < best_c5:
