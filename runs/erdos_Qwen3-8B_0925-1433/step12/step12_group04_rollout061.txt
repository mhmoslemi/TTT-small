            perturb = np.random.normal(0, 0.01, n_points)
            for i in range(n_points // 2):
                perturb[i] = 1 - perturb[n_points - 1 - i]
            new_h_perturb = best_h + perturb
            new_h_perturb = new_h_perturb * target_sum / np.sum(new_h_perturb)
            new_h_perturb = np.clip(new_h_perturb, 0, 1)
            new_c5_perturb = obj(new_h_perturb)
            if new_c5_perturb < best_c5:
                best_h = new_h_perturb.copy()
                best_c5 = new_c5_perturb

    best_h = np.array(best_h)
    for i in range(n_points // 2):
        best_h[i] = 1 - best_h[n_points - 1 - i]
    best_h = best_h * target_sum / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, n_points)
