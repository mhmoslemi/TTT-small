import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale(h):
        s = np.sum(h)
        if s > 0:
            return h * target_sum / s
        return np.zeros_like(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_ = scale(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_guesses = []
    h_initial = np.clip(initial_h_values, 0, 1)
    initial_guesses.append(scale(h_initial))

    h_struct = np.zeros(n_points)
    for i in range(0, n_points, 2):
        h_struct[i] = 1.0
    initial_guesses.append(scale(h_struct))

    h_rand = np.random.uniform(0, 1, n_points)
    initial_guesses.append(scale(h_rand))

    best_h = initial_guesses[0]
    best_c5 = obj(initial_guesses[0])

    start = time.time()
    for h_guess in initial_guesses:
        res = minimize(obj, h_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if time.time() - start > budget_s:
            break

    return (best_h, best_c5, n_points)
