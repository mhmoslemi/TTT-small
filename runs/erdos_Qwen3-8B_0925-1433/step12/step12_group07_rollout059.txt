import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def thue_morse(n):
        t = np.zeros(n, dtype=int)
        for i in range(n):
            t[i] = (t[i // 2] + i % 2) % 2
        return t

    h = np.copy(initial_h_values) if hasattr(initial_h_values, '__len__') else \
        np.array([1 - (1 - 0.5 * (1 + t)) for t in thue_morse(n_points)])
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        cross_corr = np.correlate(x, 1 - x, mode='full')
        return cross_corr[np.argmax(cross_corr)] * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'SLSQP'
            tol = 1e-8
            perturb_std = 0.01
        elif phase == 1:
            method = 'L-BFGS-B'
            tol = 1e-6
            perturb_std = 0.005
        else:
            method = 'COBYLA'
            tol = 1e-8
            perturb_std = 0.001

        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
