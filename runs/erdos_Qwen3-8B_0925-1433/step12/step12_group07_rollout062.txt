import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population[0] = initial_h_values.copy()

        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population[i] = candidate

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = np.zeros_like(population)
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = c + 0.8 * (b - a)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial = trial * target_sum / np.sum(trial)
            else:
                trial = np.ones(n_points) * 0.5 * target_sum / n_points
            if compute_c5(trial) < compute_c5(population[i]):

                best_c5 = c5
