import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * target_sum / total

    def objective(h):
        return compute_c5(h)

    # Initial population for diversity
    population = [rescale(h + 0.1 * np.random.randn(n_points)) for _ in range(5)]
    population.append(rescale(h - 0.1 * np.random.randn(n_points)))
    best_c5 = float('inf')
    best_h = np.copy(h)

    for candidate in population:
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        # Use adaptive perturbations
        perturb = 0.1 * np.random.randn(n_points)
        candidate = rescale(best_h + perturb)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
        iteration += 1

        # Perform L-BFGS-B optimization on current best
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        else:
            break

    return (best_h, best_c5, n_points)
