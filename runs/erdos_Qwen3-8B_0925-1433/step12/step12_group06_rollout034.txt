        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        overlap_contribution = np.zeros(n_points)
        for x in range(n_points):
            x_plus_k = x + lag
            if 0 <= x_plus_k < n_points:
                overlap_contribution[x] = best_h[x] * (1 - best_h[x_plus_k])
        epsilon = 0.002 * (0.9 ** _)
        perturb = -overlap_contribution * epsilon
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
