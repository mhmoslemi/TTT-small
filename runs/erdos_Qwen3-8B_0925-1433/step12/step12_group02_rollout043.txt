import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.random.rand(n_points)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h
    best_c5 = compute_c5(best_h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = minimize(lambda x: compute_c5(x), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        else:
            break

    return (best_h, best_c5, n_points)
