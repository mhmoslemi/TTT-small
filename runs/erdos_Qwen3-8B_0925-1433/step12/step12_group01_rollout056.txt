import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_pattern(n_blocks):
        h = np.zeros(n_points)
        block_size = n_points // n_blocks
        for i in range(n_blocks):
            start = i * block_size
            end = (i + 1) * block_size
            h[start:end] = 1.0
        h = h * (target_sum / np.sum(h))
        return h

    initial_guesses = [initial_h_values]
    for n_blocks in [2, 3, 4, 5]:
        initial_guesses.append(generate_pattern(n_blocks))

    best_h = initial_h_values
    best_c5 = np.inf
    start_time = time.time()

    for guess in initial_guesses:
        h = np.clip(guess, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        h = h.copy()

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.001 * (0.9 ** _)
        new_h = np.clip(best_h + np.random.normal(0, perturb_std, n_points), 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:

        perturb_std = 0.0005 * (0.9 ** _)
