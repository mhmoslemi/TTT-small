from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)

        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        h = h_vec.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h_vec):
        h = h_vec.copy()

        k_max = np.argmax(corr)
        n = len(h)
        grad = np.zeros(n)
        for m in range(n):
            k = k_max
            term1 = 0.0
            if m - k >= 0 and m - k < n:
                term1 = (1 - h[m - k]) * dx
            term2 = 0.0
            if m + k < n:
                term2 = -h[m + k] * dx
            grad[m] = term1 + term2
        return grad

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    res = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, constraints=constraints, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    for _ in range(3):
        if time.time() - start_time > budget_s:
