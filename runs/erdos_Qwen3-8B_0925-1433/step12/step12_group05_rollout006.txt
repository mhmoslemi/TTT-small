from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Preprocess initial_h_values
    h_initial = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(h_initial)
    if sum_initial > 0:
        h_initial = h_initial * target_sum / sum_initial
    else:
        h_initial = np.full(n_points, 0.5)

    # Objective function to minimize max overlap integral
    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h = h * target_sum / sum_h
        # Efficiently compute all shifts using np.roll
        shifted_1_h = [np.roll(1 - h, k) for k in range(n_points)]
        overlaps = np.dot(h, np.array(shifted_1_h).T)
        return overlaps.max() * dx

    # Set bounds for differential evolution
    bounds = [(0.0, 1.0)] * n_points

    # Run differential evolution
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=50,
        tol=1e-8, mutation=(0.8, 1), recombination=0.9,
        maxiter=200, polish=True, disp=False, x0=h_initial

    sum_feasible = np.sum(h_feasible)
