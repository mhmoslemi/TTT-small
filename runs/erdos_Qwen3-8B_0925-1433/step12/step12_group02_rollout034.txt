import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    noise_scale = 0.1
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        perturb = np.random.normal(0, noise_scale, n_points)
        candidate = h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if candidate_sum == 0:
            continue
        candidate = candidate * target_sum / candidate_sum

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(
            lambda x: compute_c5(x),
            candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100}
        )
        if result.success:
            new_c5 = compute_c5(result.x)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x.copy()
        noise_scale *= 0.99

    return (best_h, best_c5, n_points)
