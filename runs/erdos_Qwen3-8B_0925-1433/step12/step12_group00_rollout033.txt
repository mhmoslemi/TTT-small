import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = clip_and_normalize(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def objective(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_plus = clip_and_normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            h_minus = clip_and_normalize(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1)] * n_points

    start_time = time.time()

    for _ in range(10):  # Multi-start optimization
        if time.time() - start_time > budget_s:

        h = h_initial + 0.1 * np.random.randn(n_points)

        if result.success and result.fun < best_c5:
