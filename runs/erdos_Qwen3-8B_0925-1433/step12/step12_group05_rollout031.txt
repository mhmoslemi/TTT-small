def run(seed=42, budget_s=600):

    pop_size = 50

    h_initial = initial_h_values.copy()

    if sum_h > 0:

    population.append(h_initial)
    for _ in range(39):  # 40 total initial candidates

        h_perturb += np.random.normal(0, 0.2, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        if sum_h > 0:
            h_perturb = h_perturb * target_sum / sum_h
        population.append(h_perturb)
    population = population[:pop_size]

    def obj(h):

        sum_h = np.sum(h_clipped)
        if sum_h == 0:
