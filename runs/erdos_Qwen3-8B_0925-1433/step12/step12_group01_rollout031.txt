        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        return res.x if res.success else h

    best_h = optimize(best_h)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
