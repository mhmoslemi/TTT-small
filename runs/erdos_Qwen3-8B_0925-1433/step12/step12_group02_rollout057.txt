```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def perturb(h, std):
        return np.clip(h + np.random.normal(0, std, n_points), 0, 1)

    population = []
    for _ in range(20):
        p = perturb(h, 0.05)
        p = p * target_sum / np.sum(p)
        population.append(p)

    best_h = min(population, key=compute_c5)
    best_c5 = compute_c5(best_h)

    def objective(x):
        return compute_c5(x)

    def gradient(x):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_p = x.copy()
            x_p[i] += eps
            x_p = x_p * target_sum / np.sum(x_p)
            grad[i] = (compute_c5(x_p) - compute_c5(x)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=gradient)
        if result.success:
            best_h = result.x
            curr_c5 = compute_c5(best_h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
        else:
            break

    return (best_h, best_c5, n_points)
```
