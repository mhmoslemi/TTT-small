    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial symmetry and rescaling
    h = np.array(initial_h_values, dtype=np.float64)
    for i in range(n_points // 2):
        h[i] = 1.0 - h[n_points - 1 - i]
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    for step in range(5000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            term1 = 0.0
            if 0 <= i + k_max < n_points:
                term1 = dx * (1.0 - h[i + k_max])
            term2 = 0.0
            if 0 <= i - k_max < n_points:
                term2 = -dx * h[i - k_max]
            grad[i] = term1 + term2

        h = h - 0.05 * (0.99 ** step) * grad
        for i in range(n_points // 2):
            h[i] = 1.0 - h[n_points - 1 - i]
        h = h * target_sum / np.sum(h)

        if np.random.rand() < 0.05:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            for i in range(n_points // 2):
                h[i] = 1.0 - h[n_points - 1 - i]
            h = h * target_sum / np.sum(h)

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
