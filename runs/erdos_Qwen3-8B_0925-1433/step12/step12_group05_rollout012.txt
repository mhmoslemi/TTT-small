import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    learning_rate = 0.1
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        j = np.argmax(corr)
        k = j - (n_points - 1)

        grad = np.zeros(n_points)
        for m in range(n_points):
            term1 = 1 - h[(m + k) % n_points]
            term2 = h[(m - k) % n_points]
            grad[m] = term1 - term2

        h -= learning_rate * grad

        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = best_h.copy()

        corr_updated = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_updated) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
