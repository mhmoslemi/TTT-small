import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return compute_c5(x)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        current_c5 = compute_c5(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate.copy()
            h = candidate.copy()
        # Local optimization
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = result.x
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break

    return (best_h, best_c5, n_points)
