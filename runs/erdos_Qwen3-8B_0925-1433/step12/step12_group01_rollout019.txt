            h[i*block_size:(i+1)*block_size] = 1.0
        h = h * target_sum / np.sum(h)
        return h

        h_base = initial_h_values

        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

        return obj(x)
