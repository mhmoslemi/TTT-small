import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize population
    population = []
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum == 0:
        h = np.full(n_points, 0.5)
    else:
        h = h * target_sum / h_sum
    population.append(h)

    for _ in range(49):
        perturb = np.random.normal(0, 0.2, n_points)
        new_h = np.clip(initial_h_values + perturb, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum == 0:
            new_h = np.full(n_points, 0.5)
        else:
            new_h = new_h * target_sum / new_h_sum
        population.append(new_h)

    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            h1, h2, h3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = h1 + 0.8 * (h2 - h3)
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            mutant = mutant * target_sum / mutant_sum if mutant_sum != 0 else np.full(n_points, 0.5)
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    trial[j] = population[i][j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            trial = trial * target_sum / trial_sum if trial_sum != 0 else np.full(n_points, 0.5)
            current_c5 = evaluate_erdos_solution(trial)
            if current_c5 < evaluate_erdos_solution(population[i]):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        current_best = min(evaluate_erdos_solution(h) for h in population)
        if current_best < best_c5:
            best_h = [h for h in population if evaluate_erdos_solution(h) == current_best][0]
            best_c5 = current_best

    return (best_h, best_c5, n_points)
