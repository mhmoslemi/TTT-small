import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_norm(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    # Initialize population
    pop_size = 100
    population = []

        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        population.append(clip_and_norm(h))

    # Track best solution
    best_h = np.array(initial_h_values)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Genetic Algorithm settings
    max_gen = 100
    crossover_rate = 0.5
    mutation_rate = 0.1
    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break

        # Evaluate fitness and select
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        fitness = np.array(fitness)

        selected = [population[i] for i in indices[:int(pop_size * 0.4)]]

        # Generate new population

        while len(new_pop) < pop_size:
            if len(selected) < 2:
