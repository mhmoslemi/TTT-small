import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except NameError:
        n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def feasi(x):
        return np.sum(x) - target_sum

    def obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_block_pattern(n, blocks):
        h = np.zeros(n)
        for i in range(blocks):
            h[i * n // blocks : (i + 1) * n // blocks] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    def generate_checkerboard(n):
        h = np.zeros(n)
        for i in range(n):
            if i % 2 == 0:
                h[i] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    if hasattr(initial_h_values, '__len__'):
        init_pop = [np.clip(initial_h_values, 0, 1)]
    else:
        init_pop = []

    for _ in range(5):
        if _ < 5 and len(init_pop) < 20:
            if _ in [0, 1, 2, 3, 4]:
                h = generate_block_pattern(n_points, _ + 3)
                init_pop.append(h)
            else:
                h = generate_checkerboard(n_points)
                init_pop.append(h)

    pop = np.array([np.clip(x, 0, 1) for x in init_pop])
    pop = [x * target_sum / np.sum(x) if np.sum(x) > 0 else x for x in pop]
    best_h = pop[0]
    best_c5 = obj(best_h)

    start_time = time.time()
    generation = 0

    while time.time() - start_time < budget_s:
        gen_pop = []
        for _ in range(20):
            if np.random.rand() < 0.8:
                a, b = np.random.choice(len(pop), 2, replace=False)
                p1, p2 = pop[a], pop[b]
                cross = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:cross], p2[cross:]])
            else:
                child = np.random.rand(n_points)
            child = np.clip(child, 0, 1)
            s = np.sum(child)
            if s > 0:
                child = child * target_sum / s
            gen_pop.append(child)
        pop = np.array(gen_pop)
        for i in range(len(pop)):
            if np.random.rand() < 0.2:
