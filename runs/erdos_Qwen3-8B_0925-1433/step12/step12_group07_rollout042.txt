def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def thue_morse(n):
        t = np.zeros(n, dtype=int)
        for i in range(n):
            t[i] = (t[i // 2] + i % 2) % 2
        return t

    if n_points < 45:
        h = np.zeros(n_points)
    else:
        h = initial_h_values.copy()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        cross_corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(cross_corr)
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def repair(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            x = x * target_sum / s
        return x

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    if time.time() - start_time < budget_s:
        res = differential_evolution(obj, bounds, constraints=cons, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), recombination=0.9,

        if res.success:

    if time.time() - start_time < budget_s:
