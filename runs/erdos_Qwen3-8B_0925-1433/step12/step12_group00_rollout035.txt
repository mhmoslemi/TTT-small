from scipy.optimize import minimize
import time

def clip_and_normalize(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    if np.sum(h_clipped) > 0:
        h_clipped = h_clipped * target_sum / np.sum(h_clipped)
    return h_clipped

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    target_sum = n_points / 2.0

    for i in range(len(population)):
        population[i] = clip_and_normalize(population[i], target_sum)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all individuals
        fitness = [compute_c5(ind, dx) for ind in population]
        # Select top 40%
        sorted_indices = np.argsort(fitness)
        elite_size = int(0.4 * len(population))
