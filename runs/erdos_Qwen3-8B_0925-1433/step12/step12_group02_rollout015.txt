import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 100
        h = np.random.rand(n_points)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    else:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
        target_sum = n_points / 2.0

    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
            h = candidate.copy()

        result = minimize(lambda x: compute_c5(x), best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
            h = best_h.copy()
        else:
            break

    return (best_h, best_c5, n_points)
