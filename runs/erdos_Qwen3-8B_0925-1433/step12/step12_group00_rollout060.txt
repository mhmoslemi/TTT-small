def run(seed=42, budget_s=600):

            x = x * target_sum / np.sum(x)

    h = clip_and_normalize(h)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    decay = 0.999
    max_iter_adam = 1000
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 1
    start_time = time.time()

    for _ in range(max_iter_adam):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m_index = np.argmax(corr)
        k_max = m_index - (n_points - 1)
        grad = np.zeros(n_points)

            term1 = 0.0
            if i + k_max < n_points:

                term2 = h[i - k_max]
            grad[i] = term1 - term2
