import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        s = np.sum(h)
        return h * target_sum / s if s > 0 else np.zeros_like(h)

    h = np.clip(initial_h_values, 0, 1)
    h = scale_h(h)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_ = scale_h(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    seed = np.random.randint(1000)
    np.random.seed(seed)

    # Generate structured starts
    starts = [h.copy()]
    even = np.zeros(n_points)
    for i in range(n_points):
        even[i] = 1.0 if (i % 4 == 0 or i % 4 == 3) else 0.0
    starts.append(scale_h(even))
    tri = np.zeros(n_points)
    for i in range(n_points):
        tri[i] = 1.0 if (i % 3 == 0 or i % 3 == 1) else 0.0
    starts.append(scale_h(tri))
    rand = np.random.rand(n_points)
    starts.append(scale_h(rand))

    for start in starts:
        res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(120):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.003 * (0.95 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    np.random.seed(seed)
    return (best_h, best_c5, n_points)
