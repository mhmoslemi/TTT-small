    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    m = (n_points + 1) // 2

    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i] = 1 - h[n_points - 1 - i]
    h = np.clip(h, 0, 1)
    h_part = h[:m]
    h_full = np.zeros(n_points)
    for i in range(m):
        h_full[i] = h_part[i]
        h_full[n_points - 1 - i] = 1 - h_part[i]
    if n_points % 2 == 1:
        h_full[m] = 0.5

    def obj(x):
        hf = np.zeros(n_points)
        for i in range(m):
            hf[i] = x[i]
            hf[n_points - 1 - i] = 1 - x[i]
        if n_points % 2 == 1:
            hf[m] = 0.5
        corr = np.correlate(hf, 1 - hf, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * m
    best_h_part = h_part.copy()
    best_c5 = obj(best_h_part)
    start_time = time.time()

    res = minimize(obj, best_h_part, method='L-BFGS-B', bounds=bounds, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h_part = res.x
        best_c5 = res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.001, m)
        new_h_part = np.clip(best_h_part + perturb, 0, 1)
        res = minimize(obj, new_h_part, method='L-BFGS-B', bounds=bounds, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h_part = res.x
            best_c5 = res.fun

    h_full_best = np.zeros(n_points)
    for i in range(m):
        h_full_best[i] = best_h_part[i]
        h_full_best[n_points - 1 - i] = 1 - best_h_part[i]
    if n_points % 2 == 1:
        h_full_best[m] = 0.5

    return (h_full_best, best_c5, n_points)
