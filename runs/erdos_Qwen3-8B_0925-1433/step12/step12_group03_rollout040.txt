def run(seed=42, budget_s=600):

    def optimize_from_guess(guess):
        guess = np.clip(guess, 0, 1)

        if sum_current > 0:

        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 200})
