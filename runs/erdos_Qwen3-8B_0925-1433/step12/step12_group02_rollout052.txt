import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy() if initial_h_values is not None else np.random.rand(n_points)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    alpha = 0.1
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        worst_k = np.argmax(corr)
        k_true = worst_k - (n_points - 1)

        m_plus_k = np.arange(n_points) + k_true
        valid_term1 = (m_plus_k >= 0) & (m_plus_k < n_points)
        term1 = np.zeros(n_points)
        term1[valid_term1] = (1 - h[m_plus_k[valid_term1]]) * dx

        m_minus_k = np.arange(n_points) - k_true
        valid_term2 = (m_minus_k >= 0) & (m_minus_k < n_points)
        term2 = np.zeros(n_points)
        term2[valid_term2] = h[m_minus_k[valid_term2]] * dx

        grad = term1 - term2
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = (h * target_sum) / np.sum(h)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if _ % 100 == 0:
            alpha *= 0.95

    return (best_h, best_c5, n_points)
