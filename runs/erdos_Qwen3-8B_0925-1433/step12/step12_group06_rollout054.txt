            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k = m - (n_points - 1)
        worst_val = corr[m] * dx
        if worst_val <= best_c5:
            continue

        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k = m - (n_points - 1)
        worst_val = corr[m] * dx
        if worst_val <= best_c5:
            continue

        perturb_std = 0.0005 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
