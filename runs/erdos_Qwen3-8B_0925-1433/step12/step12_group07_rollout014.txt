        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        pop[i] = candidate

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()

    F = 0.8
    crossover_rate = 0.9
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[c] + F * (pop[b] - pop[a])
            mask = np.random.rand(n_points) < crossover_rate
            trial = np.where(mask, mutant, pop[i])
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial = trial * target_sum / np.sum(trial)
            else:
                trial = np.ones(n_points) * 0.5
            curr_c5 = compute_c5(trial, dx)
            original_c5 = compute_c5(pop[i], dx)
            if curr_c5 < original_c5:
                pop[i] = trial.copy()
        for h in pop:
            curr_c5 = compute_c5(h, dx)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n_points)
