from scipy.optimize import minimize
import time

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints: sum(h) = target_sum, 0 <= h[i] <= 1
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def optimize_candidate(h_initial):
        result = minimize(

            h_initial,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=cons,
            options={'ftol': 1e-8, 'max_iter': 100}
        )
        if result.success:

        return h_initial

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
