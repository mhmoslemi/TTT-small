def run(seed=42, budget_s=600):

    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1

        # Compute current C5 and find critical lag
        corr_full = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx

        lag = max_index - (len(corr_full) - 1)  # lag = index - (len-1)

        constraint_violation = np.sum(h) - target_sum
