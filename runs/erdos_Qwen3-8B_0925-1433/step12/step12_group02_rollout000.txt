import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    if n_points == 0:
        return (np.array([]), 0.0, 0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = compute_c5(h)

    # Perturbation-based exploration
    perturbations = np.random.normal(0, 0.05, (50, n_points))
    for p in perturbations:
        candidate = np.clip(h + p, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        else:
            break
        # Early stopping if no improvement
        if np.abs(compute_c5(best_h) - best_c5) < 1e-8:
            break

    return (best_h, best_c5, n_points)
