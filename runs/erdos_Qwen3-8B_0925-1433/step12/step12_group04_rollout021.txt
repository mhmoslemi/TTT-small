        new_h_sum = np.sum(new_h)
        new_h = new_h * target_sum / new_h_sum if new_h_sum > 0 else np.zeros(n_points)
        current_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = new_h, current_c5
        h = new_h
        learning_rate *= decay_rate
        step += 1

    # Final symmetry refine
    final_h = (best_h + np.fliplr(best_h)) / 2
    final_h_sum = np.sum(final_h)
    final_h = final_h * target_sum / final_h_sum if final_h_sum > 0 else np.zeros(n_points)
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    return (best_h, best_c5, n_points)
