import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 20
    pop = np.zeros((pop_size, n_points))
    pop[0] = initial_h.copy()
    for i in range(1, pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = initial_h + perturb
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate = candidate * target_sum / np.sum(candidate)
        else:
            candidate = np.ones(n_points) * 0.5
        pop[i] = candidate

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    F = 0.8
    crossover_rate = 0.9
    max_generations = 200
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        new_pop = np.zeros_like(pop)
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[c] + F * (pop[b] - pop[a])
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) > 0:
                mutant = mutant * target_sum / np.sum(mutant)
            else:
                mutant = np.ones(n_points) * 0.5
            trial = pop[i].copy()
            mask = np.random.rand(n_points) < crossover_rate
            trial[mask] = mutant[mask]
            if np.sum(trial) > 0:
                trial = trial * target_sum / np.sum(trial)
            else:
                trial = np.ones(n_points) * 0.5
            c5_trial = evaluate(trial)
            c5_original = evaluate(pop[i])
            if c5_trial < c5_original:
                new_pop[i] = trial
            else:
                new_pop[i] = pop[i]
        pop = new_pop
        current_best = np.min([evaluate(p) for p in pop])
        if current_best < best_c5:
            best_h = pop[np.argmin([evaluate(p) for p in pop])]
            best_c5 = current_best

    return (best_h, best_c5, n_points)
