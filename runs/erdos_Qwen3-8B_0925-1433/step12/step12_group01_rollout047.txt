import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_base = initial_h_values.copy()
    h_base = h_base * target_sum / np.sum(h_base)

    def eval_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_norm(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h) if np.sum(h) > 0 else h_base

    population = [h_base.copy()]
    for m in [3,5,7,9,11]:
        p = np.zeros(n_points)
        for i in range(n_points):
            p[i] = 1 if i % m < m//2 else 0
        population.append(p * target_sum / np.sum(p))
    for _ in range(15):
        p = np.random.choice([0,1], size=n_points)
        population.append(p * target_sum / np.sum(p))
    population = population[:20]

    best_h = h_base.copy()
    best_c5 = eval_c5(best_h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [eval_c5(h) for h in population]
        elite = np.argsort(fitness)[:5]
        parents = [population[i] for i in elite]
        new_pop = []
        for _ in range(5):
            a, b = np.random.choice(parents, 2, replace=False)
            pt = np.random.randint(1, n_points-1)
            c = np.concatenate([a[:pt], b[pt:]])
            new_pop.append(clip_norm(c + 0.15*np.random.randn(n_points)))
        population = new_pop + parents[:5]
        for h in population:
            if eval_c5(h) < best_c5:
                best_h, best_c5 = h.copy(), eval_c5(h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h = population[np.argmin(eval_c5(p) for p in population)]
        def obj(x):
            return np.max(np.correlate(x, 1 - x, mode='full')) * dx
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
