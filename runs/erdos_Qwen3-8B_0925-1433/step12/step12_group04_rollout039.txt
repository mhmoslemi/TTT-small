    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    learning_rate = 0.2
    decay_rate = 0.995
    lambda_lag = 0.1
    max_iter = int(budget_s / 10)  # 60 iterations max

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        k_search = np.unique(np.round(np.linspace(-1.0, 1.0, 50))).astype(int)
        valid_lags = [k for k in k_search if 0 <= k < n_points]
        if not valid_lags:
            continue
        lag_idx = np.argmax(corr[valid_lags] * dx)
        k_crit = valid_lags[lag_idx]

        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in [k_crit, -k_crit]:
                j = (i + k + n_points) % n_points
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j]) - (1 - h[i])
        gradient = np.clip(gradient, -1, 1)

        h -= learning_rate * gradient

        h = (h + np.fliplr(h)) / 2.0
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
