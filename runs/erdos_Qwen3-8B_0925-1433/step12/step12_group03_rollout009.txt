import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def optimize_initial_guess(h_start):
        h = np.copy(h_start)
        best_h = h.copy()
        best_c5 = obj(h)
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                h = best_h
        return best_h, best_c5

    best_h = np.copy(initial_h_values)
    best_c5 = obj(best_h)

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        structured = np.zeros(n_points)
        structured[::2] = 1.0
        structured = structured * target_sum / np.sum(structured) if np.sum(structured) > 0 else np.zeros(n_points)
        optimized_h, optimized_c5 = optimize_initial_guess(structured)
        if optimized_c5 < best_c5:
            best_h, best_c5 = optimized_h, optimized_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        random_guess = np.random.rand(n_points)
        random_guess = random_guess * target_sum / np.sum(random_guess) if np.sum(random_guess) > 0 else np.zeros(n_points)
        optimized_h, optimized_c5 = optimize_initial_guess(random_guess)
        if optimized_c5 < best_c5:
            best_h, best_c5 = optimized_h, optimized_c5

    return (best_h, best_c5, n_points)
