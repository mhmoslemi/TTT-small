from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        cross_corr = np.correlate(h_, 1 - h_, mode='full')
        worst_index = np.argmax(cross_corr)
        return cross_corr[worst_index] * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(50):

        if res.success and res.fun < best_c5:
