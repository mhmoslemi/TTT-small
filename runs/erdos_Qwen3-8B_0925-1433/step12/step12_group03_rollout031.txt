        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    def de_step(pop):
        nonlocal best_c5
        new_pop = []
        for i in range(pop_size):
