import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def enforce_symmetry(h):
        h = h.copy()
        for i in range(n_points // 2):
            h[i] = 1.0 - h[n_points - 1 - i]
        return h

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def obj(x):
        x = clip_scale(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values)
    h = clip_scale(h)
    h = enforce_symmetry(h)
    best_h = h.copy()
    best_c5 = obj(h)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start = time.time()
    for _ in range(200):
        if time.time() - start > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_new = res.x
            h_new = clip_scale(h_new)
            h_new = enforce_symmetry(h_new)
            curr = obj(h_new)
            if curr < best_c5:
                best_h, best_c5 = h_new, curr

    return (best_h, best_c5, n_points)
