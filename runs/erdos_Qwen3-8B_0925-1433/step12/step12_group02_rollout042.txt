import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def objective(x):
        return compute_c5(x)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h
    best_c5 = compute_c5(h)
    start_time = time.time()

    def perturb(h, sigma=0.1):
        return np.clip(h + np.random.normal(0, sigma, n_points), 0, 1) * target_sum / np.sum(np.clip(h + np.random.normal(0, sigma, n_points), 0, 1))

    for _ in range(200):
        candidate = perturb(best_h, sigma=0.05)
        result = minimize(objective, candidate, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            candidate = result.x
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_h = candidate
                best_c5 = c5
        if time.time() - start_time > budget_s - 5:
            break

    return best_h, best_c5, n_points
