import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values.copy()
    x = np.linspace(0, 2, n_points)
    sin_h = 0.5 + 0.5 * np.sin(np.pi * x)
    sin_h = np.clip(sin_h, 0, 1)
    sin_h = sin_h * target_sum / np.sum(sin_h)
    sin_h = np.clip(sin_h, 0, 1)
    initial_guesses = [h_initial, sin_h]

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = obj(best_h)

    for guess in initial_guesses:
        h = np.clip(guess, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.02 * (0.95 ** _), n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
