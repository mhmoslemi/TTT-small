import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def calc_worst_lag(h_):
        corr = np.correlate(h_, 1 - h_, mode='full')
        k_star = np.argmax(corr) - n_points + 1
        return k_star

    def directional_update(h_, k_star):
        grad = np.zeros_like(h_)
        for x in range(n_points):
            if 0 <= x + k_star < n_points:
                grad[x] += (1 - h_[x + k_star]) * dx
                grad[x + k_star] -= h_[x] * dx
        return grad

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        k_star = calc_worst_lag(best_h)

        alpha = 0.1
        new_h = best_h - alpha * grad
        new_h = np.clip(new_h, 0, 1)

        if obj(new_h) < best_c5:
            best_h, best_c5 = new_h, obj(new_h)

    for _ in range(100):
