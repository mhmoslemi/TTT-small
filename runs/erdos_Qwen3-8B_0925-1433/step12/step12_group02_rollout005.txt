import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def compute_c5(h):
        cor = np.correlate(h, 1 - h, mode='full')
        max_cor = np.max(cor)
        return max_cor * dx
    
    def compute_gradient(h, lag, n_points, dx):
        grad = np.zeros_like(h)
        for m in range(n_points):
            term1 = 0.0
            if 0 <= m + lag < n_points:
                term1 = 1.0 - h[m + lag]
            term2 = 0.0
            if 0 <= m - lag < n_points:
                term2 = h[m - lag]
            grad[m] = dx * (term1 - term2)
        return grad
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        cor = np.correlate(h, 1 - h, mode='full')
        max_cor_index = np.argmax(cor)
        k = max_cor_index - (n_points - 1)
        
        grad = compute_gradient(h, k, n_points, dx)
        
        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            h_new = h
        else:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
