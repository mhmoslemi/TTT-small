def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
