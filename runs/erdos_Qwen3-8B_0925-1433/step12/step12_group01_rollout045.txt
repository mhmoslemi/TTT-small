import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize particles
    n_particles = 20
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0, 1)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    # Add initial_h_values as one of the particles
    particles[0] = initial_h_values.copy()
    particles[0] = np.clip(particles[0], 0, 1)
    particles[0] = particles[0] * target_sum / np.sum(particles[0])

    # Fitness evaluation
    def eval_fitness(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial fitness
    fitness = np.array([eval_fitness(p) for p in particles])
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    gbest = particles[np.argmin(fitness)]
    gbest_fitness = np.min(fitness)

    best_h = gbest
    best_c5 = gbest_fitness
    start_time = time.time()
    iteration = 0
    no_improvement = 0
    max_iterations = int(budget_s / (n_points * 2))

    velocities = np.random.rand(n_particles, n_points) * 0.1

    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration > max_iterations:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 0:
                particles[i] = particles[i] * target_sum / sum_h
            current_fitness = eval_fitness(particles[i])
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness
            if current_fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = current_fitness
                best_h, best_c5 = gbest, gbest_fitness
        if gbest_fitness == best_c5:
            no_improvement += 1
            if no_improvement >= 20:
                break
        else:
            no_improvement = 0

    return (best_h, best_c5, n_points)
