from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

        return np.stack([sine, square, triangle, initial_h_values]).mean(axis=0)

    # Generate a few structured initial patterns
    initial_patterns = [h, generate_structured_pattern()]

        new_h = p.copy()

        for _ in range(200):

            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else new_h
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:

        for _ in range(150):
