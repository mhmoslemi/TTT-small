            best_h = h.copy()
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            m_plus_lag = m + lag
            m_minus_lag = m - lag
            if 0 <= m_plus_lag < n_points:
                grad[m] += (1 - h[m_plus_lag]) * dx
            if 0 <= m_minus_lag < n_points:
                grad[m] -= h[m_minus_lag] * dx
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
    
    return (best_h, best_c5, n_points)
