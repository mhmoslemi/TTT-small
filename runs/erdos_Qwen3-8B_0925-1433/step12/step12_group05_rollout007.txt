import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / s

    initial_h = np.array(initial_h_values)
    initial_h = clip_and_normalize(initial_h)

    population = [initial_h.copy()]
    for _ in range(30):
        perturb = np.random.normal(0, 0.2, n_points)
        new_h = initial_h + perturb
        new_h = clip_and_normalize(new_h)
        population.append(new_h)

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(len(population), 3, replace=False)
        h1, h2, h3 = [population[i] for i in indices]
        mutant = h1 + 0.8 * (h2 - h3)
        mutant = clip_and_normalize(mutant)
        trial = np.copy(mutant)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                trial[i] = mutant[i]
        trial = clip_and_normalize(trial)
        c5_trial = np.max(np.correlate(trial, 1 - trial, mode='full')) * dx
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = trial
            population[0] = trial

    return (best_h, best_c5, n_points)
