        h_ = np.clip(x, 0.0, 1.0)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1.0 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Phase 1: SLSQP
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.005 * (1 - _ / 100)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0.0, 1.0)
        # Symmetric copy
        sym_new = np.zeros(n_points)
        for i in range(n_points // 2):
            sym_new[i] = new_h[i]
            sym_new[n_points - 1 - i] = 1.0 - new_h[i]
        new_h = sym_new
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Phase 2: L-BFGS-B
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.002 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0.0, 1.0)
        # Symmetric copy
        sym_new = np.zeros(n_points)
        for i in range(n_points // 2):
            sym_new[i] = new_h[i]
            sym_new[n_points - 1 - i] = 1.0 - new_h[i]
        new_h = sym_new
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Phase 3: COBYLA
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.0005 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0.0, 1.0)
        # Symmetric copy
        sym_new = np.zeros(n_points)
        for i in range(n_points // 2):
            sym_new[i] = new_h[i]
            sym_new[n_points - 1 - i] = 1.0 - new_h[i]
        new_h = sym_new
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = best_h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final symmetry check
    final_h = np.zeros(n_points)
    for i in range(n_points // 2):
        final_h[i] = best_h[i]
        final_h[n_points - 1 - i] = 1.0 - best_h[i]
    sum_final = np.sum(final_h)
    if sum_final > 0:
        final_h = final_h * target_sum / sum_final
    else:
        final_h = best_h
    final_c5 = obj(final_h)
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    return (best_h, best_c5, n_points)
