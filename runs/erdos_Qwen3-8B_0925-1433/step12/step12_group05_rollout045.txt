    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return cross_corr.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    step_size = 0.1
    decay = 0.999
    max_iterations = 2000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_val = cross_corr.max()
        i_max = cross_corr.argmax()
        k_max = i_max - (n_points - 1)

        grad = np.zeros(n_points)
        for m in range(n_points):
            i1 = (m + k_max) % n_points
            i2 = (m - k_max) % n_points
            grad[m] = (1 - h[i1]) - h[i2]

        new_h = h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = h.copy()

        current_c5 = compute_c5(new_h)
        if current_c5 < compute_c5(h):
            h = new_h.copy()
            best_c5 = current_c5
            best_h = h.copy()
        step_size *= decay

    return (best_h, best_c5, n_points)
