from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()

        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    best_c5 = float('inf')
    best_h = np.copy(h)
    eval_count = 0

    def optimize_candidate(x):
        nonlocal best_c5, best_h
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        return c5

    # Initial diversity
    for _ in range(15):
        perturb = np.random.normal(0, 0.08, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        optimize_candidate(candidate)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Perturb and optimize
        perturb = np.random.normal(0, 0.05, n_points)
