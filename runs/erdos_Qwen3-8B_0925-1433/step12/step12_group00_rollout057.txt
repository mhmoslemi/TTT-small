        if curr_c5 < best_c5:

            res = minimize(obj, trial_h, method='L-BFGS-B', jac=grad_func, bounds=bounds, constraints=constraints, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
