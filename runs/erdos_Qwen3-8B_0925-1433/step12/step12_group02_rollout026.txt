import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 45
        h_vals = np.random.rand(n_points)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    else:
        n_points = len(initial_h_values)
        h_vals = initial_h_values.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_vals.copy()
    best_c5 = compute_c5(best_h)

    def optimize_candidate(x):
        nonlocal best_h, best_c5
        result = minimize(lambda x: compute_c5(x), x, method='SLSQP', bounds=[(0, 1)] * n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if result.success:
            current_c5 = compute_c5(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        return best_c5

    start_time = time.time()
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = np.clip(h_vals + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        current_c5 = compute_c5(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate.copy()
        optimize_candidate(best_h)
        if time.time() - start_time > budget_s - 10:
            break
    return (best_h, best_c5, n_points)
