    best_c5_cobyla = objective(best_h_cobyla)

    perturb = np.random.normal(0, 0.05, n_points)
    new_h = np.clip(best_h_cobyla + perturb, 0, 1)
    new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h_cobyla
    result_lbfgs = minimize(
        fun=objective,
        x0=new_h,
        method='L-BFGS-B',
        bounds=bounds_slsqp,
        constraints=cons_slsqp,
        tol=1e-10
    )
    best_h_final = result_lbfgs.x
    best_c5_final = objective(best_h_final)

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (best_h_de, best_c5_de, n_points)

    return (best_h_final, best_c5_final, n_points)
