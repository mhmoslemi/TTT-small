import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h, k_max):
        grad = np.zeros(n_points)
        for i in range(n_points):
            if k_max < 0:
                j = (i - k_max) % n_points
                grad[i] = (1 - h[j]) - h[i]
            else:
                j = (i + k_max) % n_points
                grad[i] = (1 - h[j]) - h[i]
        return -grad

    def obj(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def analytical_grad(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        return compute_gradient(h, k_max)

    def obj_lbgfs(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 500
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = analytical_grad(h)
        direction = grad
        step = learning_rate
        for _ in range(20):
            trial_h = h + step * direction
            trial_h = clip_and_normalize(trial_h)
            corr_trial = np.max(np.correlate(trial_h, 1 - trial_h, mode='full')) * dx
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * direction
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
