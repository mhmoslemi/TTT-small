import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        s = np.sum(h)
        if s > 0:
            return h * target_sum / s
        return np.zeros(n_points)

    # Initial guess
    h = np.clip(initial_h_values, 0, 1)
    h = scale_h(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Generate structured initializations
    structs = []
    for m in [22, 23]:
        pos = np.linspace(0, 2, m, endpoint=False)
        inds = np.round(pos * n_points / 2).astype(int)
        unique = np.unique(np.clip(inds, 0, n_points - 1))
        structs.append(np.zeros(n_points))
        structs[-1][unique] = 1.0
    structs.append(np.clip(initial_h_values, 0, 1))

    def obj(x):
        h_ = np.clip(x, 0, 1)
        h_ = scale_h(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

    for init in structs:
        h_ = np.clip(init, 0, 1)
        h_ = scale_h(h_)
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(h_ + perturb, 0, 1)
            new_h = scale_h(new_h)
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
            h_ = res.x

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
