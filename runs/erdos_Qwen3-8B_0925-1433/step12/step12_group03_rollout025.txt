from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            return x * target_sum / np.sum(x)
        return x

    h = scale_h(initial_h_values.copy())

    def obj(x):
        h_ = scale_h(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    initial_guesses = [h.copy()]
    # Generate a structured initial guess: evenly spaced 1s

    for guess in initial_guesses:

            new_h = np.clip(h + perturb, 0, 1)
            new_h = scale_h(new_h)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
