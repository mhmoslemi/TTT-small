    for i in range(n_points // 2):

        h[n_points - 1 - i] = 1 - h[i]
    # If n is odd, handle the middle
    if n_points % 2 == 1:
        mid = n_points // 2
        h[mid] = 0.5
    # Now, h is symmetric and sum is n_points / 2

    # Extract variables: first half (excluding middle if odd)
    variables = h[:n_points//2]

    # Bounds for variables: [0, 1]

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
