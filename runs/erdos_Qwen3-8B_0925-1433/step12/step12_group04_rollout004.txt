import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 50
        h = np.random.rand(n_points)

    def enforce_symmetry(x):
        for i in range(n_points // 2):
            x[i] = 1 - x[n_points - 1 - i]
        return x

    def rescale(x):
        s = np.sum(x)
        if s > 0:
            x *= target_sum / s
        return x

    def obj(x):
        x = np.clip(x, 0, 1)
        x = enforce_symmetry(x)
        x = rescale(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

        new_h = enforce_symmetry(new_h)

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
