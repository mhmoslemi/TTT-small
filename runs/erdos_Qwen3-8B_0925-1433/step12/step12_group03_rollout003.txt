from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():

        h_fib = h_fib * target_sum / np.sum(h_fib)
        patterns.append(h_fib)
        return patterns

    patterns = generate_patterns()
    best_h = initial_h_values.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    for h0 in patterns:

        h0 = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else h0
