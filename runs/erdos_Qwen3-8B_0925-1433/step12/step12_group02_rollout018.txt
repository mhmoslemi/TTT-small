import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 45

        h[:22] = 1.0

        h = initial_h_values.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    population = [h]
    for _ in range(50):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)

    best_c5 = float('inf')
    best_h = np.copy(h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    while time.time() - start_time < budget_s:
        if len(population) < 50:
            candidate = population[-1]
        else:
            candidate = np.random.choice(population)
        candidate = candidate.copy()
        np.random.shuffle(candidate)
        candidate = candidate * target_sum / np.sum(candidate)
        candidate = np.clip(candidate, 0, 1)

        result = minimize(lambda x: compute_c5(x), candidate, 
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 50})
        if result.success:
            current_c5 = compute_c5(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
            population.append(result.x)
        else:
            break
