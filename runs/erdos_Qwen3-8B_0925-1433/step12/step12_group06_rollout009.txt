import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    best_iter = 0

    for stage in range(3):
        if time.time() - start_time > budget_s:
            break
        if stage == 0:
            method = 'Nelder-Mead'
            perturb_std = 0.03
            iter_count = 100
        elif stage == 1:
            method = 'L-BFGS-B'
            perturb_std = 0.005
            iter_count = 150
        else:
            method = 'COBYLA'
            perturb_std = 0.001
            iter_count = 100

        for _ in range(iter_count):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                if res.fun < best_c5:
                    best_h, best_c5 = res.x, res.fun
                    best_iter = _
        perturb_std *= 0.95

    return (best_h, best_c5, n_points)
