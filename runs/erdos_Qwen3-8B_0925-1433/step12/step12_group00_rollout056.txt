import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial guess
    h = np.array(initial_h_values, dtype=float)
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def compute_gradient(h):
        # Compute current maximum overlap k and its gradient
        conv = np.correlate(h, 1 - h, mode='full')
        argmax_convol = np.argmax(conv)
        k = argmax_convol - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if 0 <= i + k < n_points:
                term1 = 1.0 - h[i + k]
            term2 = 0.0
            if 0 <= i - k < n_points:
                term2 = -h[i - k]
            gradient[i] = term1 + term2
        return gradient

    # Adaptive learning rate
    learning_rate = 0.5
    decay_rate = 0.995
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        # Gradient descent update
        step = learning_rate
        direction = -grad  # Negative for descent
        # Backtracking line search
        for _ in range(20):
            trial_h = h + step * direction
            trial_h = clip_and_normalize(trial_h)
            corr_trial = np.max(np.correlate(trial_h, 1 - trial_h, mode='full')) * dx
            if corr_trial < best_c5:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h = h + step * direction
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate

    # L-BFGS refinement
    def obj_lbgfs(x):
        h = clip_and_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_lbgfs, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
