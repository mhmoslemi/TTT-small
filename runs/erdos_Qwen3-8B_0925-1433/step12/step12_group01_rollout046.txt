import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_candidate(h):
        h_clipped = np.clip(h, 0, 1)
        if np.sum(h_clipped) > 0:
            h_clipped = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial pattern and diverse structures
    population = [initial_h_values.copy()]
    for k in [3, 5, 7, 9, 11]:
        block = np.zeros(n_points)
        block[::k] = 1
        block = block * target_sum / np.sum(block)
        population.append(block)
    for _ in range(5):
        rand = np.random.choice([0, 1], size=n_points)
        rand = rand * target_sum / np.sum(rand)
        population.append(rand)
    population = np.array(population)
    best_h = initial_h_values.copy()
    best_c5 = evaluate_candidate(best_h)

    # Genetic algorithm
    pop_size = min(20, len(population))
    generations = 100
    time_remaining = budget_s
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate_candidate(p) for p in population]
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            idxs = np.random.choice(range(pop_size), 3, replace=False)
            best_idx = idxs[np.argmin(fitness[idxs])]
            selected.append(population[best_idx])
        # Crossover
        new_population = []
        for i in range(0, len(selected), 2):
            p1, p2 = selected[i], selected[i+1] if i+1 < len(selected) else selected[i]
            cross = np.random.choice(n_points, 2, replace=False)
            cross.sort()
            c1 = np.concatenate([p1[:cross[0]], p2[cross[0]:cross[1]], p1[cross[1]:]])
            c2 = np.concatenate([p2[:cross[0]], p1[cross[0]:cross[1]], p2[cross[1]:]])
            new_population.extend([c1, c2])
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < 0.2:

            if c5 < best_c5:

        time_remaining = budget_s - (time.time() - start_time)
