import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and structured patterns
    population = [initial_h_values.copy()]
    for m in [3, 5, 7, 9, 11]:
        block_size = n_points // m
        block = np.zeros(n_points)
        for i in range(m):
            if i % 2 == 0:
                block[i*block_size:(i+1)*block_size] = 1.0
        block = block * target_sum / np.sum(block)
        population.append(block)
    for _ in range(10):
        rand = np.random.choice([0, 1], size=n_points)
        rand = rand * target_sum / np.sum(rand)
        population.append(rand)
    population = [np.clip(p, 0, 1) for p in population]
    population = [p * target_sum / np.sum(p) for p in population]

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    def evolve():
        # Evaluate fitness
        fitness = np.array([compute_c5(p) for p in population])
        parents = np.argsort(fitness)[:int(len(population)*0.2)]
        new_population = []
        for _ in range(len(population)):
            if np.random.rand() < 0.5:
                idx1, idx2 = np.random.choice(parents, 2, replace=False)
                p1, p2 = population[idx1], population[idx2]
                cross = np.random.choice(n_points, 2, replace=False)
                child = np.copy(p1)
                child[cross[0]:cross[1]] = p2[cross[0]:cross[1]]
            else:
                child = np.random.rand(n_points)
                child = child * target_sum / np.sum(child)
            child = np.clip(child, 0, 1)
            new_population.append(child)
        population[:] = new_population

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        evolve()
        for p in population:
            if compute_c5(p) < best_c5:
                best_c5 = compute_c5(p)
                best_h = p

    # Local refinement
    for _ in range(20):
        if time.time() - start_time > budget_s:

        fit = np.array([compute_c5(p) for p in population])
        top_idx = np.argsort(fit)[:3]

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0, 1)] * n_points

                best_h = res.x
