import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        cross_corr = np.correlate(h_, 1 - h_, mode='full')
        worst_k = np.argmax(cross_corr)
        return cross_corr[worst_k] * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    def optimize_phase(perturb_std, method, n_iter):
        nonlocal best_c5, best_h
        for _ in range(n_iter):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    optimize_phase(0.005, 'SLSQP', 150)
    optimize_phase(0.002, 'L-BFGS-B', 100)
    optimize_phase(0.0005, 'COBYLA', 50)
    optimize_phase(0.0001, 'Nelder-Mead', 30)

    return (best_h, best_c5, n_points)
