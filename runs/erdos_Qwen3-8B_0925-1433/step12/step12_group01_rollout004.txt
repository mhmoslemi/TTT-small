import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize PSO
    n_particles = 20
    particles = np.clip(np.random.rand(n_particles, n_points), 0, 1)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.zeros_like(particles)
    p_best = particles.copy()
    p_best_fitness = np.array([eval_c5(p) for p in p_best])
    g_best = p_best[np.argmin(p_best_fitness)]
    g_best_fitness = np.min(p_best_fitness)

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        r1, r2 = np.random.rand(n_particles, n_points), np.random.rand(n_particles, n_points)
        velocities = 0.7 * velocities + 1.5 * r1 * (p_best - particles) + 1.5 * r2 * (g_best - particles)
        particles += velocities
        # Clip and rescale
        particles = np.clip(particles, 0, 1)
        particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
        # Evaluate
        current_fitness = np.array([eval_c5(p) for p in particles])
        # Update personal best
        improve = current_fitness < p_best_fitness
        p_best[improve] = particles[improve]
        p_best_fitness[improve] = current_fitness[improve]
        # Update global best
        if np.min(current_fitness) < g_best_fitness:
            g_best = p_best[np.argmin(p_best_fitness)]
            g_best_fitness = np.min(p_best_fitness)
    best_h = g_best
    return (best_h, g_best_fitness, n_points)
