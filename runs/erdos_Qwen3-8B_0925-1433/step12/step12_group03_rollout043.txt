import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_periodic(n_points):
        h = np.zeros(n_points)
        m = n_points // 2
        h[:m] = 1.0
        h[m] = 0.5
        sum_h = m + 0.5
        h *= target_sum / sum_h
        return np.clip(h, 0, 1)

    def generate_fibonacci(n_points):
        h = np.zeros(n_points)
        phi = (1 + np.sqrt(5)) / 2
        for i in range(n_points):
            pos = i / n_points * 2
            h[i] = 1 if abs(np.sin(pos * phi) * 10) > 5 else 0
        sum_h = np.sum(h)
        h *= target_sum / sum_h
        return np.clip(h, 0, 1)

    initial_guesses = [initial_h_values, generate_periodic(n_points), generate_fibonacci(n_points)]
    best_h, best_c5 = None, float('inf')

    for hint in initial_guesses:
        h = np.clip(hint, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        start_time = time.time()
        h_best = h.copy()
        c5_best = obj(h)

        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.005 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h_best + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h_best
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < c5_best:
                h_best, c5_best = res.x, res.fun

        for _ in range(100):
