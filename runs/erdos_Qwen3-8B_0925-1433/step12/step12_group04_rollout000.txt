        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        # Enforce symmetry
        for i in range(n_points // 2):
            new_h[n_points - i - 1] = 1.0 - new_h[i]
        if n_points % 2 == 1:
            new_h[n_points // 2] = 0.5
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final symmetry check and rescale
    final_h = best_h.copy()
    for i in range(n_points // 2):
        final_h[n_points - i - 1] = 1.0 - final_h[i]
    if n_points % 2 == 1:
        final_h[n_points // 2] = 0.5
    final_h = final_h * target_sum / np.sum(final_h) if np.sum(final_h) > 0 else best_h
    final_c5 = obj(final_h)
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    return (best_h, best_c5, n_points)
