import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_population():
        pop = []
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        pop.append(h)
        for b in [3, 5, 7, 9, 11]:
            block = np.zeros(n_points)
            for i in range(b):
                block[i * n_points // b:(i + 1) * n_points // b] = 1.0 if i % 2 == 0 else 0.0
            block = np.clip(block, 0, 1)
            block = block * target_sum / np.sum(block) if np.sum(block) > 0 else np.zeros(n_points)
            pop.append(block)
        for _ in range(10):
            h = np.random.choice([0, 1], n_points)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            pop.append(h)
        return pop[:20]

    population = generate_population()
    best_h = population[0]
    best_c5 = np.inf

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        parents = []
        for _ in range(10):
            idxs = np.random.choice(len(population), 3)
            best = idxs[np.argmin(fitness[idxs])]
            parents.append(population[best])
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            c1 = np.random.choice(n_points, 2, replace=False)
            p1[c1[0]:c1[1]] = p2[c1[0]:c1[1]]
            p2[c1[0]:c1[1]] = p1[c1[0]:c1[1]]
            new_pop.extend([p1, p2])
        for h in new_pop:
            noise = np.random.normal(0, 0.15, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        population = new_pop
        current = [compute_c5(h) for h in population]
        idxs = np.argsort(current)
        population = [population[i] for i in idxs[:20]]
        best_h, best_c5 = population[0], current[0]

    def obj(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        idx = np.argmin([compute_c5(h) for h in population])
        h = population[idx]

        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            population = [res.x] + population[:19]
    return (best_h, best_c5, n_points)
