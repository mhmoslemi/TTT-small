import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0
    
    # Initialize symmetric and scaled h
    h = (np.copy(initial_h_values) + 1.0) / 2.0
    h = h * target_sum_half / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    
    if n_points % 2 == 0:
        h[n_points//2:] = 1.0 - h[:n_points//2][::-1]
    else:
        h[n_points//2+1:] = 1.0 - h[:n_points//2][::-1]
        h[n_points//2] = 0.5
    
    # Objective function
    def obj(x):
        h_ = np.zeros(n_points)
        h_[:len(x)] = x
        if n_points % 2 == 0:
            h_[len(x):] = 1.0 - x[::-1]
        else:
            h_[len(x):] = 1.0 - x[::-1]
            h_[n_points//2] = 0.5
        h_ = np.clip(h_, 0, 1)
        sum_h = np.sum(h_ * dx)
        if abs(sum_h - 1) > 1e-6:
            h_ = h_ * 1.0 / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    # Bounds
    bounds = [(0.0, 1.0)] * n_points//2
    
    # Constraints
    def constraint_sum(x):
        return np.sum(x) - target_sum_half
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial guess
    x0 = (h[:n_points//2] if n_points % 2 == 0 else h[:n_points//2])
    
    # Optimization
    start_time = time.time()
    result = minimize(obj, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    
    # Reconstruct h
    h_best = np.zeros(n_points)
    h_best[:len(best_h)] = best_h
    if n_points % 2 == 0:
        h_best[len(best_h):] = 1.0 - best_h[::-1]
    else:
        h_best[len(best_h):] = 1.0 - best_h[::-1]
        h_best[n_points//2] = 0.5
    h_best = np.clip(h_best, 0, 1)
    sum_h = np.sum(h_best * dx)
    if abs(sum_h - 1) > 1e-6:
        h_best = h_best * 1.0 / sum_h
    
    # Final check
    final_c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
    return (h_best, final_c5, n_points)
