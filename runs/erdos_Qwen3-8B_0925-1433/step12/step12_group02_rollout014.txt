import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 45
        h = np.zeros(n_points)
        h[::2] = 1.0
    else:
        h = initial_h_values.copy()
        n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    population_size = 20
    population = [clip_rescale(h + np.random.normal(0, 0.15, n_points)) for _ in range(population_size)]
    best_c5 = float('inf')
    best_h = np.copy(population[0])
    for h_candidate in population:
        c5 = compute_c5(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        noise_scale = 0.15 * (0.95 ** iteration)
        new_population = []
        for h_candidate in population:
            perturb = np.random.normal(0, noise_scale, n_points)
            candidate = h_candidate + perturb
            candidate = clip_rescale(candidate)
            new_population.append(candidate)
        population = new_population

        for h_candidate in population:
            c5 = compute_c5(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()

        if iteration % 10 == 0:
            result = minimize(lambda x: compute_c5(x), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
            if result.success:
                best_h = result.x

        if iteration % 50 == 0:

                    if cur_c5 < best_c5:

                        best_h = result.x.copy()
