            if time.time() - start_time > budget_s:
                break
            try:
                res = minimize(obj, guess, method=method, bounds=bounds, constraints=cons, tol=1e-6)
                if res.success:
                    current_c5 = res.fun
                    if current_c5 < best_c5:
                        best_h, best_c5 = res.x, current_c5
            except:
                pass

    return (best_h, best_c5, n_points)
