def run(seed=42, budget_s=600):

    initial_points = [h.copy()]

    mirrored = mirrored * target_sum / np.sum(mirrored)
    initial_points.append(mirrored)
    inverted = 1.0 - h
    inverted = inverted * target_sum / np.sum(inverted)
    initial_points.append(inverted)
    shifted = np.roll(h, 1)

        p = np.clip(p, 0, 1)
