import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    def obj(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        g = np.zeros_like(x)
        for i in range(n_points):
            h_p = x.copy()
            h_p[i] += 1e-4
            h_p = normalize(h_p)
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')[k]
            h_m = x.copy()
            h_m[i] -= 1e-4
            h_m = normalize(h_m)
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')[k]
            g[i] = (corr_p - corr_m) / (2e-4)
        return -g

    def init_points(h0):
        points = [h0]
        points.append(h0[::-1])
        points.append(np.roll(h0, 1))
        points.append(0.5 + 0.5 * np.sin(np.pi * (np.linspace(0, 2, n_points) - 1) / 2))
        points.append(np.clip(h0 + np.random.normal(0, 0.1, n_points), 0, 1))
        points.append(1 - h0)
        points.append(0.5 * h0 + np.random.uniform(0, 0.5, n_points))
        return [normalize(p) for p in points]

    h0 = normalize(initial_h_values)
    best_h = h0.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(5):
        pts = init_points(best_h)
        for p in pts:
            try:
                res = minimize(obj, p, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points,

                               tol=1e-7, options={'maxiter': 150, 'ftol': 1e-9})
                if res.success and res.fun < best_c5:
                    best_c5 = res.fun
                    best_h = res.x
            except:

        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
