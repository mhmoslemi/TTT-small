import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 100
        h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = float('inf')
    time_start = time.time()

    def generate_pattern(m):
        if m == 0:
            return None
        L = 1.0 / m
        d = 1.0 / (m - 1)
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + d)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    m_values = [5, 7, 10, 12, 15, 20]
    patterns = []
    for m in m_values:
        p = generate_pattern(m)
        if p is not None:
            patterns.append(p)

    if 'initial_h_values' in globals():
        patterns.append(h.copy())

    for p in patterns:
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def cons(x):
            return np.sum(x) - target_sum

        res = minimize(obj, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': cons}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    def perturb(x):
        h = x * target_sum / np.sum(x)
        noise = np.random.normal(0, 0.02, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    for _ in range(50):
        if time.time() - time_start > budget_s:
            break
        h_perturbed = perturb(best_h)
        corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed

    return (np.clip(best_h, 0, 1), best_c5, n_points)
