import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            ip = i + lag
            im = i - lag
            term1 = (1 - h[ip] if 0 <= ip < n_points else 0.0)
            term2 = (h[im] if 0 <= im < n_points else 0.0)
            grad[i] = term1 - term2
        return grad

    def perturb(h, scale=0.1):
        noise = np.random.normal(0, scale, h.shape)
        return np.clip(h + noise, 0, 1) * target_sum / np.sum(h + noise)

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = obj(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = perturb(best_h, scale=0.15)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=compute_gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='Nelder-Mead', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
