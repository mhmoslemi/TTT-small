import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = (initial_h * target_sum) / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    patterns = []
    # add random binary pattern
    h_rand = np.zeros(n_points)
    h_rand[np.random.choice(n_points, size=n_points//2, replace=False)] = 1
    h_rand = (h_rand * target_sum) / np.sum(h_rand)
    patterns.append(h_rand)
    
    # add periodic spaced pattern
    for period in [5, 6, 7]:
        h_period = np.zeros(n_points)
        for i in range(n_points):
            if (i % period) == 0:
                h_period[i] = 1
        h_period = (h_period * target_sum) / np.sum(h_period)
        patterns.append(h_period)
    
    # add perturbed initial
    for _ in range(2):
        perturb = np.random.normal(0, 0.05, n_points)
        h_pert = initial_h + perturb
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = (h_pert * target_sum) / np.sum(h_pert)
        patterns.append(h_pert)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    for pattern in patterns:
        result = minimize(objective, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
