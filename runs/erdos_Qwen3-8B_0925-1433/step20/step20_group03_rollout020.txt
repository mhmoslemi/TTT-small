import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    max_iter = 500
    learning_rate = 0.1
    decay_rate = 0.995  # decay learning rate over iterations
    noise_amp = 0.05
    perturb_freq = 10

    for iteration in range(max_iter):
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        k_max = np.argmax(corr)
        k_actual = k_max - (n_points - 1)
        
        # Compute gradient for this lag
        grad = np.zeros_like(h_initial)
        for i in range(n_points):
            i_plus = i + k_actual
            term1 = 1 - h_initial[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k_actual
            term2 = h_initial[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        
        # Update h_initial
        h_new = h_initial - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        h_new = h_new * target_sum / h_new_sum
        
        # Add random perturbation
        if iteration % perturb_freq == 0:
            noise = np.random.normal(0, noise_amp, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * target_sum / h_new_sum
        
        # Evaluate new h
        current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
        
        # Decay learning rate
        learning_rate *= decay_rate
        h_initial = h_new

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
