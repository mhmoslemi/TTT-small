import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 60
        initial_h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h.copy()
    best_c5 = 0.381

    def generate_pattern(m):
        L = 1.0 / m
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for si in range(m):
                start = si * L
                if start <= x < start + L:
                    h[i] = 1.0
                    break
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    m_values = [8, 12, 16, 20, 24, 28, 32]
    for m in m_values:
        h = generate_pattern(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': cons}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        perturbation = np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(best_h + perturbation, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_h = perturbed
            best_c5 = c5_perturbed

    return (np.clip(best_h, 0, 1), best_c5, n_points)
