import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    else:
        n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 50
    particles = []
    velocities = []
    pbests = []
    pbest_c5s = []

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
        particles.append(initial_h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                h += 1e-6
            h = h * target_sum / h_sum
            particles.append(h)

    for _ in range(population_size):
        velocities.append(np.zeros(n_points))
        pbests.append(particles[-1].copy())
        pbest_c5s.append(compute_c5(particles[-1]))

    gbest_idx = np.argmin(pbest_c5s)
    gbest = pbests[gbest_idx].copy()
    gbest_c5 = pbest_c5s[gbest_idx]

    start_time = time.time()
    for iteration in range(1000):
        if time.time() - start_time > budget_s:
            break
        omega = 0.9 - 0.005 * iteration
        omega = max(omega, 0.4)
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] * omega + 2.0 * r1 * (pbests[i] - particles[i]) + 2.0 * r2 * (gbest - particles[i])
            v = np.clip(v, -0.1, 0.1)
            velocities[i] = v
            particles[i] += v
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            current_c5 = compute_c5(particles[i])
            if current_c5 < pbest_c5s[i]:
                pbests[i] = particles[i].copy()
                pbest_c5s[i] = current_c5
        if np.min(pbest_c5s) < gbest_c5:
            gbest_idx = np.argmin(pbest_c5s)
            gbest = pbests[gbest_idx].copy()
            gbest_c5 = pbest_c5s[gbest_idx]

    best_h = np.clip(gbest, 0, 1)
    return (best_h, gbest_c5, n_points)
