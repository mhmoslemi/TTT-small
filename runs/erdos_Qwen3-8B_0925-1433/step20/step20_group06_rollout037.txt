import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in locals() else 45
    dx = 2.0 / n_points

            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)
            population.append(h)
    else:
        population = [np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    c5 = np.array([compute_c5(h) for h in population])
    best_idx = np.argmin(c5)
    best_c5 = c5[best_idx]
    best_h = population[best_idx]
    velocities = [np.random.uniform(-max_vel, max_vel, n_points) for _ in range(population_size)]
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + phi1*r1*(population[best_idx] - population[i]) + phi2*r2*(best_h - population[i])

            population[i] += velocities[i]

                if curr_c5 < best_c5:

                    best_h = population[i]

        omega -= 0.005
