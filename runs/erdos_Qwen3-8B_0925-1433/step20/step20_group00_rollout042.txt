import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Generate structured patterns
    patterns = []
    # Spaced
    spaced = np.zeros(n_points)
    spaced[::4] = 1
    spaced = spaced * target_sum / np.sum(spaced)
    patterns.append(spaced)
    # Random binary
    random_bin = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    random_bin[indices] = 1
    random_bin = random_bin * target_sum / np.sum(random_bin)
    patterns.append(random_bin)
    # Shifted spaced
    spaced_shifted = np.zeros(n_points)
    spaced_shifted[1::4] = 1
    spaced_shifted = spaced_shifted * target_sum / np.sum(spaced_shifted)
    patterns.append(spaced_shifted)

    # Optimization
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for pattern in patterns:
        res = minimize(obj, pattern, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    # Refinement loop
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
