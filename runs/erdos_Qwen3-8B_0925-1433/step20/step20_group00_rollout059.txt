import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.array([0.0]*len(initial_h_values))  # Placeholder; actual import handled by parent
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = initial_h_values.copy() * target_sum / np.sum(initial_h_values)
    best_c5 = obj(best_h)

    start_time = time.time()

    # Generate varied initial patterns
    patterns = []
    for _ in range(3):
        m = np.random.randint(5, 10)
        p = np.zeros(n_points)
        p[np.arange(0, n_points, m)] = 1
        p = p * target_sum / np.sum(p) + np.random.uniform(-0.1, 0.1, n_points)
        p = np.clip(p, 0, 1)
        patterns.append(p)
    for _ in range(2):
        idx = np.random.choice(n_points, n_points//2, replace=False)
        p = np.zeros(n_points)
        p[idx] = 1
        p = p * target_sum / np.sum(p)
        patterns.append(p)

    # Evaluate initial patterns
    for p in patterns:
        try:
            res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    # Refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
