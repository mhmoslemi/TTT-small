from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def scale_h(h):
        return h * target_sum / np.sum(h)

    # Initial solution from parent
    best_h = np.array(initial_h_values, dtype=float)
    best_h = scale_h(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(corr)
    best_c5 = min(best_c5, 0.381)

    # Generate additional initial patterns
    initial_patterns = []
    for _ in range(3):
        h = np.array(initial_h_values, dtype=float)
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = scale_h(h)
        initial_patterns.append(h)

    for _ in range(3):
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, size=n_points//2, replace=False)
        h[idxs] = 1
        h = scale_h(h)
        initial_patterns.append(h)

    # Add periodic pattern with spacing
    h_periodic = np.zeros(n_points)
    for i in range(n_points):
        if (i % 5) < 3:
            h_periodic[i] = 1
    h_periodic = scale_h(h_periodic)
    initial_patterns.append(h_periodic)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return np.max(corr)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Initial optimization
    for pattern in initial_patterns:
        res = minimize(obj, pattern, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Refinement loop
    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final adjustment
    best_h = np.clip(best_h, 0, 1)
    best_h = scale_h(best_h)
    return (best_h, best_c5, n_points)
