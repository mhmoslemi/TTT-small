import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_pattern(h):
        return h * target_sum / np.sum(h)

    # Initialize with given pattern
    initial_h = np.array(initial_h_values, dtype=float)
    best_h = scale_pattern(initial_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Generate diverse initial patterns
    def generate_patterns():
        patterns = []
        # Periodic structure with jitter
        period = int(n_points / 5) or 1
        p = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                p[i] = 1
        patterns.append(scale_pattern(p))
        # Random binary
        idx = np.random.choice(n_points, size=n_points//2, replace=False)
        p = np.zeros(n_points)
        p[idx] = 1
        patterns.append(scale_pattern(p))
        # Shifted initial pattern
        patterns.append(scale_pattern(initial_h))
        # Random perturbed periodic
        p = np.zeros(n_points)
        for i in range(n_points):
            if i % (n_points // 8) == 0:
                p[i] = 1
        p = scale_pattern(p) + np.random.uniform(-0.1, 0.1, n_points)
        p = np.clip(p, 0, 1)
        patterns.append(p)
        return patterns

    patterns = generate_patterns()

    # Optimization objective
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints
    def constraint(x):
        return np.sum(x) - target_sum

    start_time = time.time()

    # Initial optimization with structured patterns
    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': constraint})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Refinement with tighter tolerances
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = scale_pattern(best_h)

    return (best_h, best_c5, n_points)
