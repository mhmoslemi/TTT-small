        x = i * dx
        if (0.0 <= x <= 0.5) or (1.0 <= x <= 1.5):
            h_pattern[i] = 1.0
    sum_pattern = np.sum(h_pattern)
    h_pattern = h_pattern * target_sum / sum_pattern

    # Optimize the pattern-based initial guess
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        n = len(h)
        k_actual = k_max - (n - 1)
        grad = np.zeros_like(x)
        for i in range(n):
            i_plus = i + k_actual
            term1 = 1 - h[i_plus] if 0 <= i_plus < n else 0.0

            term2 = h[i_minus] if 0 <= i_minus < n else 0.0
            grad[i] = term1 - term2
        return grad
