import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(ind):
        corr = np.correlate(ind, 1 - ind, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.6:
            h = initial_h_values.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        elif np.random.rand() < 0.3:
            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = np.random.choice([0, 1], size=n_points)
            h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = float('inf')
    start_time = time.time()

    # Evolutionary loop
    for _ in range(100):
        fitness = [evaluate(ind) for ind in population]
        elite = [ind for ind, f in zip(population, fitness) if f < best_c5]
        best_c5 = min(best_c5, min(fitness))
        best_h = elite[0] if elite else best_h

        if time.time() - start_time > budget_s:
            break

        # Tournament selection
        parents = []
        for _ in range(population_size):
            candidates = np.random.choice(population, 5)
            winner = min(candidates, key=evaluate)
            parents.append(winner)

        # Blend crossover and mutate
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else parents[i]
            child = (p1 + p2) / 2
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)

        # Replace population
        population = offspring + [ind for ind in population if evaluate(ind) < best_c5]
        if len(population) > population_size:
            population = np.random.choice(population, size=population_size, replace=False)

    # Local refinement
    top_candidates = sorted(population, key=evaluate)[:10]
    for cand in top_candidates:
        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        res = minimize(objective, cand, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
