import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate structured initial guess with evenly spaced 1s
    def make_even_initial():
        m = n_points // 2
        positions = np.linspace(0, 2, m + 1)
        indices = np.round(positions * n_points / 2).astype(int)
        unique_indices = np.unique(indices)
        initial = np.zeros(n_points)
        for idx in unique_indices:
            if 0 <= idx < n_points:
                initial[idx] = 1
        initial = initial * target_sum / np.sum(initial)
        return initial
    
    h = make_even_initial()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Generate a few shifted initial guesses
    shifts = [0, 1, n_points // 4, n_points // 2, n_points * 3 // 4]
    initial_guesses = [h]
    for shift in shifts:
        shifted = np.roll(h, shift)
        initial_guesses.append(shifted)
    
    # Use initial_guesses as population
    population = initial_guesses.copy()
    for i in range(len(population)):
        population[i] += np.random.normal(0, 0.05, n_points)
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6)
            if res.success:
                population[i] = res.x
        # Keep top 80% and add new candidates
        population = sorted(population, key=lambda x: obj(x))[:len(population) * 4//5]
        for _ in range(len(population) // 4):
            a, b = np.random.choice(len(population), 2, replace=False)
            population.append((population[a] + population[b]) / 2)
        
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
