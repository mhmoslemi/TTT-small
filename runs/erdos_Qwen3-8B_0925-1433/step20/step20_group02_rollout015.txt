import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial solution
    current_h = np.array(initial_h_values, dtype=float)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)

    time_start = time.time()

    # SA parameters
    T = 100.0
    cooling_rate = 0.995
    sa_iterations = 500

    for _ in range(sa_iterations):
        if time.time() - time_start > budget_s:
            break
        perturb_count = np.random.randint(2, 5)
        indices = np.random.choice(n_points, perturb_count, replace=False)
        noise = np.random.normal(0, 0.1, perturb_count)
        h_perturb = current_h.copy()
        h_perturb[indices] += noise
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)

        c5_perturb = compute_c5(h_perturb)
        delta = c5_perturb - best_c5
        if delta < 0:
            best_c5 = c5_perturb
            best_h = h_perturb.copy()
            current_h = h_perturb.copy()
        else:
            if np.random.rand() < np.exp(-delta / T):
                current_h = h_perturb.copy()
        T *= cooling_rate

    # Final refinement
    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(objective, current_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=constraints, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h)
    else:
        best_h = current_h
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
