import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    pop_size = 100
    offspring_size = pop_size // 2
    mutate_prob = 0.3
    mut_scale = 0.01
    tournament_size = 5

    # Initial population
    pop = []
    for _ in range(pop_size):
        if np.random.rand() < 0.6:
            h = np.array(initial_h_values, dtype=float)
        elif np.random.rand() < 0.3:
            h = np.array(initial_h_values, dtype=float)
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    pop = np.array(pop)

    best_h = pop[0]
    best_c5 = np.inf

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        fitness = np.array(fitness)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx]

        # Select parents via tournament
        parents = []
        for _ in range(offspring_size):
            participants = np.random.choice(len(pop), tournament_size, replace=False)
            winner = participants[np.argmin(fitness[participants])]
            parents.append(pop[winner])
        parents = np.array(parents)

        # Generate offspring
        offspring = []
        for i in range(offspring_size):
            if i >= len(parents) // 2:
                break
            p1, p2 = parents[2*i], parents[2*i+1]
            child = (p1 + p2) / 2
            if np.random.rand() < mutate_prob:
                noise = np.random.normal(0, mut_scale, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            offspring.append(child)
        offspring = np.array(offspring)

        # Replace worst
        worst_idx = np.argsort(fitness)[-offspring_size:]
        for i, idx in enumerate(worst_idx):
            pop[idx] = offspring[i]

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
