import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def obj(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        # Perturb and normalize
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h)
            if curr_c5 < best_c5:
                best_h = curr_h.copy()
                best_c5 = curr_c5
        h = best_h

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
