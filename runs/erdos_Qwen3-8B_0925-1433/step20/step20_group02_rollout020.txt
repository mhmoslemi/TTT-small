import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        return h * target_sum / np.sum(h)

    def mutate(h):
        noise = np.random.normal(0, 0.05, n_points)
        h = np.clip(h + noise, 0, 1)
        return rescale(h)

    def crossover(h1, h2):
        if np.random.rand() < 0.5:
            return h1
        return h2

    def generate_variants(base):
        variants = [base]
        for _ in range(4):
            h = base.copy()
            h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
            h = rescale(h)
            variants.append(h)
        return variants

    initial_variants = generate_variants(initial_h_values)
    population = initial_variants
    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([evaluate(h) for h in population])
        elite_idx = np.argsort(scores)[:5]
        elites = [population[i] for i in elite_idx]
        new_population = elites[:5]
        for _ in range(15):
            h1, h2 = np.random.choice(elites, 2, replace=False)
            child = crossover(h1, h2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        curr_best = min(scores)
        if curr_best < best_c5:
            best_c5 = curr_best
            best_h = population[np.argmin(scores)]
        if gen % 5 == 0 and curr_best == best_c5:
            break

    res = minimize(evaluate, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x)-target_sum}, tol=1e-6)
    if res.success:
        best_h = res.x
    best_h = np.clip(best_h, 0, 1)
    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
