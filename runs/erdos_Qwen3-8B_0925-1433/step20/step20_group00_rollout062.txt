from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def generate_patterns():
        patterns = [np.array(initial_h_values, dtype=float)]
        # Even spacing
        for _ in range(2):
            indices = np.round(np.linspace(0, n_points - 1, int(n_points / 2))).astype(int)
            h = np.zeros(n_points)
            h[indices] = 1
            patterns.append(h)
        # Random binary
        for _ in range(2):
            idxs = np.random.permutation(n_points)
            h = np.zeros(n_points)
            h[idxs[:int(n_points / 2)]] = 1
            patterns.append(h)
        return patterns
    
    patterns = generate_patterns()
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    return (np.clip(best_h, 0, 1), best_c5, n_points)
