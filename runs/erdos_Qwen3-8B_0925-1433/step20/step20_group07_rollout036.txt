from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        x0 = h + perturb
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * target_sum / np.sum(x0)
        res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
