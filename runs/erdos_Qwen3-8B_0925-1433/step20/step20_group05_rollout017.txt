    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    for _ in range(100):
        current_time = time.time()
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, jac=objective_grad, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        if time.time() - current_time > budget_s:
            break
        perturbation = np.random.uniform(-0.02, 0.02, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        c5_perturbed = np.max(np.correlate(perturbed_h, 1 - perturbed_h, mode='full')) * dx
        if c5_perturbed < best_c5:
            best_h = perturbed_h
            best_c5 = c5_perturbed

    return (np.clip(best_h, 0, 1), best_c5, n_points)
