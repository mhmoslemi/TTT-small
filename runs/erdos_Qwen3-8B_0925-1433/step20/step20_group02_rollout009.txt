import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        s = np.sum(x)
        if s == 0:
            return float('inf')
        x_scaled = x * target_sum / s
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guesses = [np.array(initial_h_values, dtype=float)]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        initial_guesses.append(h)

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    best_h = result.x
    best_c5 = result.fun

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        def refine_func(x):
            s = np.sum(x)
            if s == 0:
                return float('inf')
            x_scaled = x * target_sum / s
            corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return np.max(corr) * dx
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(
            refine_func,
            best_h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=cons,
            tol=1e-6
        )
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    s = np.sum(best_h)
    scaled_h = best_h * target_sum / s
    final_c5 = np.max(np.correlate(scaled_h, 1 - scaled_h, mode='full')) * dx
    return (scaled_h, final_c5, n_points)
