from scipy.optimize import minimize
import time

        variants.append(np.roll(h, 1))
        variants.append(0.5 + 0.5 * np.sin(np.linspace(0, np.pi, n_points)))
        variants.append(np.clip(h + np.random.normal(0, 0.2, n_points), 0, 1))
        return [v * target_sum / np.sum(v) for v in variants]

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    initial = initial_h_values * target_sum / np.sum(initial_h_values)
    variants = generate_variants(initial)
    best_h = initial.copy()
    best_c5 = np.max(np.correlate(initial, 1 - initial, mode='full')) * dx

    start_time = time.time()
    temp = 100.0

    iterations = 0
