from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    start_time = time.time()
    
    def obj(x):
        h = x * target_sum / np.sum(x)

    def generate_structured_patterns():

        return patterns
    
    patterns = generate_structured_patterns()
