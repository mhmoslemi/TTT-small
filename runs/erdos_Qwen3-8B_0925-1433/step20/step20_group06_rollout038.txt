import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381  # Initial bound

    # Generate structured patterns
    for m in [5, 10, 20]:
        h = np.zeros(n_points)
        starts = np.random.uniform(0, 2 - 1/m, m)
        for i in range(m):
            start = starts[i]
            end = start + 1/m
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        current_sum = np.sum(h)
        if current_sum == 0:
            continue
        h = h * target_sum / current_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Coordinate descent
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += 0.05
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_p = np.max(corr_plus) * dx
            if c5_p < best_c5:
                best_h = h_plus.copy()
                best_c5 = c5_p
            h_minus = best_h.copy()
            h_minus[i] -= 0.05
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_m = np.max(corr_minus) * dx
            if c5_m < best_c5:
                best_h = h_minus.copy()
                best_c5 = c5_m
        # Random perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = c5_perturbed

    return (np.clip(best_h, 0, 1), best_c5, n_points)
