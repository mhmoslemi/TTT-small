import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Enforce symmetry on initial_h_values
    h = np.copy(initial_h_values)
    for i in range(n_points // 2):
        h[n_points - 1 - i] = h[i]
    h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += 1e-4
            x_minus = x.copy()
            x_minus[i] -= 1e-4
            h_plus = x_plus * target_sum / np.sum(x_plus)
            h_minus = x_minus * target_sum / np.sum(x_minus)
            corr_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
            corr_minus = np.max(np.correlate(h_minus, 1 - h_minus, mode='full')) * dx
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    def perturb(h, mag=0.05):
        h = np.clip(h, 0, 1)
        noise = np.random.uniform(-mag, mag, size=n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        for i in range(n_points // 2):
            h[n_points - 1 - i] = h[i]
        return h

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success and compute_c5(res.x) < best_c5:
            best_h = res.x
            best_c5 = compute_c5(best_h)
        perturbed = perturb(best_h)
        if compute_c5(perturbed) < best_c5:
            best_h = perturbed
            best_c5 = compute_c5(best_h)

    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    for i in range(n_points // 2):
        h[n_points - 1 - i] = h[i]
    return (h, best_c5, n_points)
