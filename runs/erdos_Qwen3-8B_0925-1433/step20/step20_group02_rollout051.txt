import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize particles with variations
    particles = [np.array(initial_h_values, dtype=float)]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles.append(h)

    n_particles = len(particles)
    inertia = 0.729
    cognitive = 1.494
    social = 1.494
    max_iter = 100
    time_start = time.time()

    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]

    # Personal best and global best
    personal_best = particles.copy()
    personal_best_values = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    global_best = personal_best[np.argmin(personal_best_values)]
    global_best_value = min(personal_best_values)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            velocities[i] = vel
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            current_c5 = compute_c5(particles[i])
            if current_c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_c5
        current_min = min(personal_best_values)
        if current_min < global_best_value:
            global_best = personal_best[np.argmin(personal_best_values)]
            global_best_value = current_min

    best_h = np.clip(global_best, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, global_best_value, n_points)
