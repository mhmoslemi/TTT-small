import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values.copy()
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        n = len(h)
        k_actual = k_max - (n - 1)
        grad = np.zeros_like(x)
        for i in range(n):
            i_plus = i + k_actual
            term1 = 1 - h[i_plus] if 0 <= i_plus < n else 0.0
            i_minus = i - k_actual
            term2 = h[i_minus] if 0 <= i_minus < n else 0.0
            grad[i] = term1 - term2
        return grad

    starts = [
        h_initial,
        h_initial[::-1],
        np.clip(h_initial + np.random.normal(0, 0.1, n_points), 0, 1) * target_sum / np.sum(np.clip(h_initial + np.random.normal(0, 0.1, n_points), 0, 1))
    ]

    for h_start in starts:
        res = minimize(obj, h_start, method='SLSQP', bounds=[(0, 1)] * n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=compute_gradient, options={'maxiter': 100})
        if res.success:
            current_c5 = obj(res.x)
            if current_c5 < best_c5:

                best_c5 = current_c5
