import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured patterns
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381  # Initial bound

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate structured patterns for initialization
    structured_patterns = []
    for m in [10, 20, 30]:
        h = np.zeros(n_points)
        starts = np.random.uniform(0, 2 - 1.0/m, m)
        ends = starts + 1.0/m
        for i in range(m):
            for j in range(n_points):
                x = j * dx
                if starts[i] <= x <= ends[i]:
                    h[j] = 1.0
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            structured_patterns.append(h)
    if structured_patterns:
        best_h = structured_patterns[0]
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        for h in structured_patterns:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    def optimize(x0, time_left):
        start = time.time()
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6
        )
        if time.time() - start > time_left:
            return x0
        return res.x if res.success else x0

    start_time = time.time()
    best_h = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        best_h = optimize(best_h, budget_s - (time.time() - start_time))
        if np.isnan(best_c5):
            break
        # Perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(best_h + perturbation, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        perturbed_c5 = np.max(np.correlate(perturbed, 1 - perturbed, mode='full')) * dx
        if perturbed_c5 < best_c5:
            best_h = perturbed
            best_c5 = perturbed_c5
        # Early stopping
        if best_c5 < 0.3808:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
