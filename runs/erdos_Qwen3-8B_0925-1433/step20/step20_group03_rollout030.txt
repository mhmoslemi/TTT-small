    target_sum = n_points / 2.0
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    iterations = 500
    learning_rate = 0.01
    t0 = time.time()
    
    for _ in range(iterations):
        if time.time() - t0 > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            i_plus = i + k
            if 0 <= i_plus < n_points:
                grad[i] += 1 - h[i_plus]
            i_minus = i - k
            if 0 <= i_minus < n_points:
                grad[i] -= h[i_minus]
        
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.38075:
            break
    
    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
