import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        g = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            g[i] = term1 - term2
        return g

    def ga_step():
        pop = [h + np.random.normal(0, 0.1, n_points) for _ in range(20)]
        pop = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in pop]
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            scores = [np.max(np.correlate(p, 1-p, mode='full')) * dx for p in pop]
            top = np.argsort(scores)[-8:]
            new_pop = []
            for i in range(10):
                a, b = top[np.random.choice(len(top), 2, replace=False)]
                child = (1 - np.random.rand(n_points)) * pop[a] + np.random.rand(n_points) * pop[b]
                child = np.clip(child, 0, 1) * target_sum / np.sum(np.clip(child, 0, 1))
                new_pop.append(child)
            for p in top:
                if np.random.rand() < 0.3:
                    mut = p + np.random.normal(0, 0.1, n_points)

                    new_pop.append(mut)
            pop = new_pop[:20]

                if np.max(np.correlate(p, 1-p, mode='full')) * dx < best_c5:
                    best_c5 = np.max(np.correlate(p, 1-p, mode='full')) * dx
                    best_h = p.copy()

    start_time = time.time()

    bounds = Bounds(0, 1)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
