from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    pop_size = 100
    population = [h.copy() for _ in range(pop_size)]
    for p in population:
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            x0 = population[i].copy()
            x0 += np.random.normal(0, 0.05, n_points)
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * target_sum / np.sum(x0)
            res = minimize(obj, x0, method='COBYLA', bounds=[(0, 1)]*n_points,
