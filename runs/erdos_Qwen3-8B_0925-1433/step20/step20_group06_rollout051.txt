def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.uniform(0, 1, 45)
        h = h * (45 / 2) / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    elapsed = 0

    # Simulated Annealing
    T_start = 1.0
    T_end = 0.1
    cooling_rate = 0.95
    max_steps = 100
    iter_per_step = 500

    T = T_start
    start_time = time.time()
    for step in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        T = max(T_end, T_start * (cooling_rate ** step))
        for _ in range(iter_per_step):
            if time.time() - start_time > budget_s:
                break
            noise = np.random.normal(0, T / np.sqrt(n_points), n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum == 0:
                continue
            perturbed = perturbed * target_sum / perturbed_sum
            current_c5 = compute_c5(perturbed)
            if current_c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = current_c5
            else:
                delta = current_c5 - best_c5
                if np.random.rand() < np.exp(-delta / T):
                    best_h = perturbed.copy()
                    best_c5 = current_c5

    # Local Optimization
    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - target_sum

    cons = {'type': 'eq', 'fun': constraint}
