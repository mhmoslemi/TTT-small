            h_new = h_new * target_sum / h_new_sum

        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_max = np.max(new_corr) * 2.0 / n_points
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            alpha *= 0.5

        if best_c5 < 0.3807:
            break

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * 2.0 / n_points
    return (best_h, best_c5, n_points)
