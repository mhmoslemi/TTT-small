import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = clip_and_normalize(initial_h)
    best_h = initial_h
    best_c5 = compute_c5(best_h.copy())
    best_c5 = min(best_c5, 0.381)

    initial_patterns = []

    for _ in range(5):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = clip_and_normalize(h)
        initial_patterns.append(h)

    for _ in range(3):
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:int(n_points // 2)]] = 1.0
        h = clip_and_normalize(h)
        initial_patterns.append(h)

    if n_points % 2 == 0:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = clip_and_normalize(h)
        initial_patterns.append(h)

    for _ in range(2):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.2, n_points)
        h += noise
        h = clip_and_normalize(h)
        initial_patterns.append(h)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    start_time = time.time()
    for pattern in initial_patterns:
        try:
            res = minimize(compute_c5, pattern, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(compute_c5, best_h, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass

    best_h = clip_and_normalize(best_h)
    return (best_h, best_c5, n_points)
