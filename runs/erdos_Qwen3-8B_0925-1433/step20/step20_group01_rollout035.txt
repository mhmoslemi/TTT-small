import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    pop_size = 20
    h_pop = []
    for _ in range(pop_size):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        h_pop.append(h)
    best_h = h_pop[0]
    best_c5 = compute_c5(best_h)

    for gen in range(100):
        if time.time() > budget_s:
            break
        # Evaluate
        scores = [compute_c5(h) for h in h_pop]
        idx = np.argsort(scores)
        elite = h_pop[idx[:8]]
        new_pop = []
        for _ in range(pop_size - len(elite)):
            p1, p2 = np.random.choice(elite, 2, replace=False)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            mut_idx = np.random.choice(n_points, int(n_points * 0.2), replace=False)
            child[mut_idx] += np.random.normal(0, 0.1, len(mut_idx))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        h_pop = elite + new_pop
        curr_c5 = compute_c5(h_pop[0])
        if curr_c5 < best_c5:
            best_h, best_c5 = h_pop[0], curr_c5

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        g = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0
            g[i] = term1 - term2
        return g

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1)] * n_points
    if time.time() < budget_s:
        res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            best_h, best_c5 = res.x, res.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
