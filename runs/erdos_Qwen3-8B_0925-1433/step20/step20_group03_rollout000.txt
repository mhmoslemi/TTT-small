import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h_scaled
    best_c5 = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def generate_initial_guesses(h):
        yield h
        yield h[::-1]
        yield 1 - h
        yield np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        yield np.clip(h + np.random.normal(0, 0.2, n_points), 0, 1)
        yield np.clip(h + np.random.normal(0, 0.3, n_points), 0, 1)

    for guess in generate_initial_guesses(initial_h):
        guess_scaled = guess * target_sum / np.sum(guess)
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        def grad(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            n = len(h)
            k_actual = k_max - (n - 1)
            grad = np.zeros_like(x)
            for i in range(n):
