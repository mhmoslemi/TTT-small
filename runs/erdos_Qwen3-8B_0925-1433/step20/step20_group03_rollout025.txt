import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def generate_guesses(h):
        yield h[::-1]
        yield 1.0 - h
        for _ in range(2):
            noise = np.random.normal(0, 0.1, len(h))
            noisy = np.clip(h + noise, 0, 1)
            noisy = noisy * target_sum / np.sum(noisy)
            yield noisy

    for guess in generate_guesses(h_initial):
        guess_scaled = guess * target_sum / np.sum(guess)
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        def grad(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            k_actual = k_max - (n_points - 1)
            g = np.zeros_like(x)
            for i in range(n_points):
                i_plus = i + k_actual
                g[i] = (1 - h[i_plus] if 0 <= i_plus < n_points else 0) - (h[i - k_actual] if 0 <= i - k_actual < n_points else 0)
            return g
        res = minimize(obj, guess_scaled, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=grad, options={'maxiter': 200})
        if res.success:

            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
