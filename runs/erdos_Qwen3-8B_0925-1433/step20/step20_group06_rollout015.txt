            start = starts[i]
            end = start + interval_len

                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        current_sum = np.sum(h)

            continue
        h = h * (target_sum) / current_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    def obj(x):

        return np.max(corr) * dx
