import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 60
        best_h = np.full(n_points, 0.5)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_pattern(m):
        L = 1.0 / m
        S = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + S)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    m_values = [5, 8, 12, 16, 20, 24, 28, 32]
    for m in m_values:
        h = generate_pattern(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx:
                best_h = h.copy()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
