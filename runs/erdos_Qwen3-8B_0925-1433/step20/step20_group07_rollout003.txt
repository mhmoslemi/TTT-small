import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    
    # Spectral initialization: low-pass FFT
    fft_h = np.fft.fft(h)
    fft_freq = np.fft.fftfreq(n_points)
    cutoff = 0.3
    mask = np.abs(fft_freq) <= cutoff
    fft_h[~mask] = 0
    h = np.fft.ifft(fft_h).real
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def compute_c5(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    population = [h.copy() for _ in range(20)]
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            x0 = p.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            res = minimize(compute_c5, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                population[population.index(p)] = res.x
    
    for cand in population:
        curr_c5 = compute_c5(cand)
        if curr_c5 < best_c5:
            best_h = cand.copy()
            best_c5 = curr_c5
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
