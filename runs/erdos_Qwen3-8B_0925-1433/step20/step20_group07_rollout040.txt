import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = obj(best_h)

    population = [h.copy() for _ in range(50)]
    for p in population:
        p += np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)

    for _ in range(200):
        if time.time() > time.time() + budget_s:
            break
        for i in range(len(population)):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='L-BFGS-B',
                           bounds=[(0, 1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           tol=1e-6)
            if res.success:
                population[i] = res.x
        curr_c5 = obj(population[0])
        if curr_c5 < best_c5:
