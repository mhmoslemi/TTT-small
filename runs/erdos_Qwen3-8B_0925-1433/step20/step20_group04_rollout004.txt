            if current_c5 < p['pbest_value']:
                p['pbest'] = p['position'].copy()
                p['pbest_value'] = current_c5
        global_best = min(particles, key=lambda p: p['pbest_value'])
        best_c5 = global_best['pbest_value']
        best_h = global_best['pbest']
        if iter % 10 == 0:
            if np.abs(best_c5 - best_c5_prev) < 1e-5:
                break
            best_c5_prev = best_c5

    def objective(h):
        h = h * target_sum / np.sum(h)
        return compute_c5(h, dx)

    result = minimize(objective, best_h, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}])
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h, dx)
    else:
        best_h = global_best['pbest']
        best_c5 = global_best['pbest_value']

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
