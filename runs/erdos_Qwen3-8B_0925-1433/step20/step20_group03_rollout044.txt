    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        n = len(h)
        k_actual = k_max - (n - 1)
        grad = np.zeros_like(x)
        for i in range(n):

            term1 = 1 - h[i_plus] if 0 <= i_plus < n else 0.0

    def optimize_from_guess(h_start):
