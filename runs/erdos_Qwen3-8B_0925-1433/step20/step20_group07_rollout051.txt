    h = spectral_init(h)

    # Block coordinate descent
    block_size = 20

    population = [h.copy() for _ in range(50)]
    
    for _ in range(300):
        if time.time() - time.time() > budget_s:

            population = population[-50:]

        candidates.sort()
        best_c5, best_h = candidates[0]
        
        # Block optimization
        for i in range(block_count):
            block_start = i * block_size
            block_end = (i+1) * block_size
            block = best_h[block_start:block_end]

            constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum + np.sum(best_h[:block_start]) - np.sum(best_h[block_end:])}
