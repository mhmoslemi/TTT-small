    def generate_structured_patterns(m):
        h = np.zeros(n_points)
        for _ in range(m):
            start = np.random.uniform(0, 2 - 1/m)
            for i in range(n_points):
                x = i * dx
                if start <= x <= start + 1/m:
                    h[i] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    for m in [10, 20, 30]:
        h = generate_structured_patterns(m)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)

            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4

            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
