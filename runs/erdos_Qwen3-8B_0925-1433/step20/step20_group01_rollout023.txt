import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}

    def generate_initial_guesses(num):
        guesses = []
        for _ in range(num):
            perturbed = initial_guess.copy()
            noise = np.random.normal(0, 0.1, n_points)
            perturbed += noise
            perturbed = normalize(perturbed)
            guesses.append(perturbed)
        for _ in range(num // 2):
            flipped = initial_guess.copy()
            flip_mask = np.random.rand(n_points) < 0.3
            flipped[flip_mask] = 1 - flipped[flip_mask]
            flipped = normalize(flipped)
            guesses.append(flipped)
        for _ in range(num // 2):
            uniform = np.random.rand(n_points)
            uniform = normalize(uniform)
            guesses.append(uniform)
        return guesses

    start_time = time.time()
    initial_guesses = generate_initial_guesses(5)
    for i, guess in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    # Local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
