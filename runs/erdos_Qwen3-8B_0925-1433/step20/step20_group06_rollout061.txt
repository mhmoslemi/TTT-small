import time

def run(seed=42, budget_s=600):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

    pop_size = 50
    particles = np.random.uniform(0, 1, (pop_size, n_points))
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = particles.copy()
    pbest_c5 = np.zeros(pop_size)
    gbest = particles[0].copy()
    gbest_c5 = best_c5

    omega = 0.9
    phi1, phi2 = 2.0, 2.0
    decay = 0.005
    iter_limit = int(budget_s / (pop_size * n_points * 2))  # Basic iteration limit
    start_time = time.time()

    for _ in range(iter_limit):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] +
                            phi1 * r1 * (pbest[i] - particles[i]) +
                            phi2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            # Compute C5
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx

                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
        # Update global best

            gbest = pbest[np.argmin(pbest_c5)].copy()
            gbest_c5 = np.min(pbest_c5)
        # Decay omega
        omega = max(0.4, omega - decay)

    # Final evaluation
    corr = np.correlate(gbest, 1 - gbest, mode='full')
    final_c5 = np.max(corr) * dx
