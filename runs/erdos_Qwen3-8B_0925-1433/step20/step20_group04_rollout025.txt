    base = np.array(initial_h_values, dtype=float)

    random_perturb = (random_perturb * target_sum) / np.sum(random_perturb)
    candidates.append(random_perturb)

        h = (h * target_sum) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
