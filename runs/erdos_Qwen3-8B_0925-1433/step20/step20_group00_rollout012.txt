import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    parent = np.array(initial_h_values, dtype=float)
    parent = parent * target_sum / np.sum(parent)
    best_h = parent
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def eval_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def gen_pattern(pattern_type):
        if pattern_type == 'uniform':
            h = np.zeros(n_points)
            h[::int(n_points/2)] = 1
            return h * target_sum / np.sum(h)
        elif pattern_type == 'periodic':
            h = np.zeros(n_points)
            h[::2] = 1
            phase = np.random.rand()
            h = np.roll(h, int(phase * n_points))
            return h * target_sum / np.sum(h)
        elif pattern_type == 'staggered':
            h = np.zeros(n_points)
            indices = np.random.permutation(n_points)
            h[indices[:int(n_points/2)]] = 1
            return h * target_sum / np.sum(h)
        elif pattern_type == 'golomb':
            h = np.zeros(n_points)
            idxs = np.random.permutation(np.arange(n_points))
            for i in range(int(n_points/2)):
                h[idxs[i]] = 1
            return h * target_sum / np.sum(h)
        elif pattern_type == 'random':
            h = np.random.rand(n_points)
            h = np.sort(h)
            h = np.where(h < 0.5, 0, 1)
            return h * target_sum / np.sum(h)
        else:
            return np.zeros(n_points)

    initial_patterns = [
        gen_pattern('uniform'),
        gen_pattern('periodic'),
        gen_pattern('staggered'),
        gen_pattern('golomb'),
        gen_pattern('random')
    ]

    for p in initial_patterns:
        res = minimize(obj, p, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, eval_c5(best_h), n_points)
