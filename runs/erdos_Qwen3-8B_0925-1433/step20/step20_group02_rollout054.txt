import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = np.array(initial_h_values, dtype=float)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)

    T = 100.0
    cooling_rate = 0.995
    max_iter = 500
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=3, replace=False)
        perturbed = current_h.copy()
        perturbed[indices] += np.random.normal(0, 0.1, 3)
        perturbed = np.clip(perturbed, 0, 1)
        scale = target_sum / np.sum(perturbed)
        perturbed *= scale
        new_c5 = compute_c5(perturbed)
        current_c5 = compute_c5(current_h)

        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = perturbed
            if new_c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = new_c5
        T *= cooling_rate

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)

    final_c5 = compute_c5(final_h)
