        total = np.sum(h)
        if total == 0:
            return h
        return h * target_sum / total

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def update_velocity(v, x, pbest, gbest, omega, phi1, phi2, r1, r2):
        v = omega * v + phi1 * r1 * (pbest - x) + phi2 * r2 * (gbest - x)
        return np.clip(v, -0.1, 0.1)

    # Initial population and pbest/gbest
    pop_size = 50
    particles = np.random.uniform(0, 1, (pop_size, n_points))
    particles = np.apply_along_axis(rescale, 1, particles)
    pbest = particles.copy()
    pbest_c5 = np.array([compute_c5(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)

    start_time = time.time()
