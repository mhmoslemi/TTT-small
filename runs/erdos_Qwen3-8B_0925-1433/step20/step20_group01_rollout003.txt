import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    pop_size = 20
    generations = 100
    mutation_rate = 0.3
    decay_rate = 0.99
    mut_std = 0.1
    alpha = 0.5
    elite = 2
    best_pop = [h.copy()]

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h = np.clip(h, 0, 1)
        factor = target_sum / np.sum(h)
        return h * factor

    def mutate(ind):
        noise = np.random.normal(0, mut_std, size=len(ind))
        child = ind + noise
        child = rescale(child)
        return child

    def crossover(parent1, parent2):
        child = alpha * parent1 + (1 - alpha) * parent2
        return rescale(child)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        pop = best_pop + [mutate(best_pop[i]) for i in range(elite)]
        scores = np.array([evaluate(ind) for ind in pop])
        order = np.argsort(scores)
        top = pop[order[:pop_size]]
        next_pop = []
        for i in range(0, pop_size, 2):
            if i+1 < pop_size:
                child = crossover(top[i], top[i+1])
                next_pop.append(child)
            next_pop.append(top[i])
        best_pop = next_pop[:pop_size]
        if gen % 10 == 0:
            current_c5 = evaluate(best_pop[0])
            if current_c5 < best_c5:
                best_h, best_c5 = best_pop[0].copy(), current_c5
        mutation_rate *= decay_rate
        mut_std *= decay_rate

    x0 = rescale(best_pop[0])
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k
            if 0 <= i_plus < n_points:
                grad[i] += 1 - x[i_plus]
            i_minus = i - k
            if 0 <= i_minus < n_points:
                grad[i] -= x[i_minus]
        return grad
    res = minimize(objective, x0, method='trust-constr', bounds=bounds, constraints=cons, jac=grad, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
