import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Symmetric initialization
    sym_h = np.zeros(n_points)
    for i in range(n_points):
        j = n_points - 1 - i
        sym_h[i] = initial_h_values[i]
        sym_h[j] = initial_h_values[i]
    sym_h = sym_h * target_sum / np.sum(sym_h)
    best_h = sym_h.copy()
    best_c5 = eval_c5(best_h)
    start_time = time.time()
    
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        # Gradient-based optimization
        res = minimize(lambda x: eval_c5(x), best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        # Perturbative exploration
        pert = np.random.uniform(-0.05, 0.05, size=n_points)
        pert_h = np.clip(best_h + pert, 0, 1)
        pert_h = pert_h * target_sum / np.sum(pert_h)
        c5_pert = eval_c5(pert_h)
        if c5_pert < best_c5:
            best_h = pert_h
            best_c5 = c5_pert
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
