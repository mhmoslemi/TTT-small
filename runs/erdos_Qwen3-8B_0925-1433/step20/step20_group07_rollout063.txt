def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)  # normalize initial h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Population-based optimization
    pop_size = 10  # reduce population size to save time
    population = [h.copy() for _ in range(pop_size)]
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(200):  # reduce iterations
        if time.time() - start_time > budget_s:
            break
        # Generate new candidates based on current best
        new_population = []
        # Perturb current best
        for i in range(len(population)):
            x0 = population[i].copy()
            # Add small random noise
            x0 += np.random.normal(0, 0.05, n_points)
            x0 = np.clip(x0, 0, 1)
            # Enforce sum constraint
            x0 = x0 * target_sum / np.sum(x0)
            new_population.append(x0)
        # Optimize each new candidate
        for cand in new_population:
            res = minimize(obj, cand, method='SLSQP', 
                          bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6)
            if res.success:
                population.append(res.x)
                # Prune population to keep the size manageable
                if len(population) > pop_size:
                    population.pop()
        # Evaluate all candidates to find the best
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
    
    # Final evaluation
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
