            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            new_c5 = np.max(corr) * dx
            if new_c5 < pbest_c5[i]:
                pbest_c5[i] = new_c5
                pbest_h[i] = particles[i].copy()
            if new_c5 < gbest_c5:
                gbest_c5 = new_c5
                gbest_h = particles[i].copy()
        omega -= 0.005

    return (np.clip(gbest_h, 0, 1), gbest_c5, n_points)
