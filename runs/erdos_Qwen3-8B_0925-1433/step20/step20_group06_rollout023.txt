    for m in [10, 15, 20]:
        h = np.zeros(n_points)
        starts = np.random.uniform(0, 2 - 1/m, m)
        ends = starts + 1/m
        for i in range(m):
            start = starts[i]
            end = ends[i]
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            structured.append(h)
    structured.append(np.array(initial_h_values, dtype=float))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')

    start_time = time.time()

    for h in structured:
        h = h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            # Compute current C5
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

                h1 = h1 * target_sum / np.sum(h1)
                corr1 = np.correlate(h1, 1 - h1, mode='full')
