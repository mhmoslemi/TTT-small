import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = normalize(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    population_size = 20
    generations = 100
    mutation_rate = 0.3
    crossover_rate = 0.8

    population = [h.copy() for _ in range(population_size)]

    for gen in range(generations):
        if time.time() > budget_s:
            break
        individuals = []
        for indiv in population:
            indiv = normalize(indiv)
            c5 = compute_c5(indiv)
            individuals.append((c5, indiv))
        individuals.sort()
        elite = individuals[:8]
        new_pop = [elite[i][1] for i in range(2)]
        for _ in range(population_size - 2):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(elite, 2, replace=False)
                child = 0.5 * p1 + 0.5 * p2
            else:
                child = np.random.rand(n_points)
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, int(n_points * 0.2), replace=False)
                child[indices] += np.random.normal(0, 0.1, size=len(indices))
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_pop.append(child)
        population = new_pop
        if individuals[0][0] < best_c5:
            best_h, best_c5 = individuals[0][1].copy(), individuals[0][0]

    def obj(x):
        h = x.copy()
        h = normalize(h)
        return compute_c5(h)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
