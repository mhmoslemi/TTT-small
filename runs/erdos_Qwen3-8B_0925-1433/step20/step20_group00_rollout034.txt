from scipy.optimize import minimize
import time

        patterns.append(h)
        return patterns

    patterns = generate_patterns()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=[(0, 1)]*n_points,

        if res.success and res.fun < best_c5:

        res = minimize(obj, best_h, method='SLSQP', bounds=[(0, 1)]*n_points,
