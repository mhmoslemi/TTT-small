import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    # Scale initial_h_values and generate structured patterns
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h_scaled = initial_h * target_sum / np.sum(initial_h)
    patterns = [initial_h_scaled]

    # Generate random binary patterns
    for _ in range(3):
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h[indices] = 1
        h = h * target_sum / np.sum(h)
        patterns.append(h)

    # Generate periodic patterns
    for _ in range(2):
        k = np.random.randint(4, 9)
        h = np.zeros(n_points)
        for i in range(n_points):
            if (i % k) < 2:
                h[i] = 1
        h = h * target_sum / np.sum(h)
        patterns.append(h)

    # Objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h_scaled.copy()
    best_c5 = obj(best_h)

    # Initial optimization
    for pattern in patterns:
        res = minimize(obj, pattern, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-7, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    # Refinement phase
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
