import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values.copy() * target_sum / np.sum(initial_h_values)
    
    def objective(x):
        h = x.copy() * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(x):
        h = x.copy() * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_actual = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k_actual
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - k_actual
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad
    
    bounds = [(0, 1)] * n_points
    
    res = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, jac=gradient, 
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                   options={'maxiter': 100, 'disp': False})
    
    best_h = res.x
    best_c5 = res.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
