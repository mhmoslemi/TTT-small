        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = (1 - h[i_plus] if 0 <= i_plus < n_points else 0.0)
            i_minus = i - shift
            term2 = (h[i_minus] if 0 <= i_minus < n_points else 0.0)
            grad[i] = term1 - term2
        return grad

    # Phase 1: Adam optimization
    learning_rate = 0.1
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    h_current = h.copy()
    m, v = np.zeros(n_points), np.zeros(n_points)
    for _ in range(300):
        if elapsed > budget_s:
            break
        grad = compute_gradient(h_current)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (_ + 1))
        v_hat = v / (1 - beta2 ** (_ + 1))
        h_current = h_current - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * target_sum / np.sum(h_current)
        current_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h_current.copy(), current_c5
        elapsed = time.time() - time_start

    # Phase 2: L-BFGS-B refinement
    for _ in range(5):
        if elapsed > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Phase 3: Perturbed restarts
    for _ in range(3):
        if elapsed > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = best_h * (1 + perturbation)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        h_opt = h_perturbed.copy()
        m_opt, v_opt = np.zeros(n_points), np.zeros(n_points)
        for step in range(100):
            if elapsed > budget_s:
                break
            grad = compute_gradient(h_opt)
            m_opt = beta1 * m_opt + (1 - beta1) * grad
            v_opt = beta2 * v_opt + (1 - beta2) * grad ** 2
            m_opt_hat = m_opt / (1 - beta1 ** (step + 1))
            v_opt_hat = v_opt / (1 - beta2 ** (step + 1))
            h_opt = h_opt - learning_rate * m_opt_hat / (np.sqrt(v_opt_hat) + eps)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
            current_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h_opt.copy(), current_c5
        elapsed = time.time() - time_start

    return (np.clip(best_h, 0, 1), best_c5, n_points)
