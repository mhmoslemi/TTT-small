def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_patterns():
        patterns = []
        # Perturb initial pattern
        for _ in range(2):
            h = np.array(initial_h_values, dtype=float)
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Randomly spaced 1s
        for _ in range(2):
            indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
            h = np.zeros(n_points)
            h[indices] = 1
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Alternating pattern
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * target_sum / np.sum(h)
        patterns.append(h)
        # Uniform spacing
        h = np.zeros(n_points)
        spacing = n_points // 2
        for i in range(n_points // 2):
            h[i * spacing] = 1
        h = h * target_sum / np.sum(h)
        patterns.append(h)
        return patterns

    initial_patterns = generate_initial_patterns()
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points

    for pattern in initial_patterns:
