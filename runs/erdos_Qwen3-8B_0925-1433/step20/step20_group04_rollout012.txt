import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Initial variants
    variants = [
        initial_h_values.copy(), 
        np.flip(initial_h_values).copy(), 
        np.roll(initial_h_values, 1).copy(), 
        (initial_h_values.copy() * 0.5 + 0.5).copy(),
        (1 - initial_h_values).copy()
    ]
    for _ in range(3):
        noise = np.random.normal(0, 0.1, n_points)
        h = initial_h_values + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        variants.append(h)

    # PSO phase
    particles = [v * target_sum / np.sum(v) for v in variants]
    best_h = particles[0]
    best_c5 = eval_c5(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s * 0.7:
            break
        for i in range(len(particles)):
            h = particles[i] + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            particles[i] = h
        current_c5s = [eval_c5(p) for p in particles]
        best_idx = np.argmin(current_c5s)
        best_h, best_c5 = particles[best_idx], current_c5s[best_idx]

    # SQP refinement
    def objective(x):
        h = x * target_sum / np.sum(x)
        return eval_c5(h)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result = minimize(objective, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-8)
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, eval_c5(best_h), n_points)
