import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h

    # Generate structured initial patterns
    for m in [10, 20, 30]:
        h = np.zeros(n_points)
        starts = np.linspace(0, 2 - 1/m, m)
        for i in range(m):
            for j in range(n_points):
                x = j * dx
                if starts[i] <= x <= starts[i] + 1/m:
                    h[j] = 1.0
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Optimization
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_jac(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    def constraint(x):
        return np.sum(x) - target_sum

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, jac=compute_jac, tol=1e-6)
