        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    h = best_h.copy()
    h = np.clip(h, 0.0, 1.0)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
