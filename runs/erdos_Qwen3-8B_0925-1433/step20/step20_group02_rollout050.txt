            guess = guess * (target_sum / np.sum(guess))
            guesses.append(guess)
        return guesses

    initial_guesses = generate_initial_guesses()

    best_result = None
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_result = result.x
        if time.time() - start_time > budget_s:
            break

    if best_result is None:
        best_result = np.array(initial_h_values, dtype=float)
        best_c5 = objective(best_result)

    best_result = np.clip(best_result, 0, 1)
    return (best_result, best_c5, n_points)
