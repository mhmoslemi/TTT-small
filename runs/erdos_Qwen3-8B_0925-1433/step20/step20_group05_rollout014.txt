import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

        h = np.array(initial_h_values, dtype=np.float64)
    else:

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = 0.381
    initial_patterns = [h.copy()]

    def generate_pattern(m, jitter=0.3):
        h = np.zeros(n_points)
        intervals = np.linspace(0, 2, m + 1)
        for i in range(m):
            start = intervals[i] + np.random.uniform(-jitter, jitter)
            end = intervals[i + 1] + np.random.uniform(-jitter, jitter)
            start = np.clip(start, 0, 2)
            end = np.clip(end, 0, 2)
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        return h * target_sum / total

    for m in [5, 10, 15, 20, 25]:
        pattern = generate_pattern(m)
        if pattern is not None and np.any(pattern != initial_patterns[0]):
            initial_patterns.append(pattern)

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    start_time = time.time()
