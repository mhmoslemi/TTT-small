import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def perturb(h, method='gaussian', noise=0.1, frac=0.3):
        h = np.array(h, dtype=float)
        if method == 'gaussian':
            noise_vec = np.random.normal(0, noise, h.shape)
            h += noise_vec
        elif method == 'flip':
            flip_mask = np.random.rand(h.shape[0]) < frac
            h[flip_mask] = 1 - h[flip_mask]
        elif method == 'uniform':
            h = np.random.rand(h.shape[0]) * target_sum / np.sum(h)
        return np.clip(h, 0, 1) * target_sum / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - target_sum

    best_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_c5 = compute_c5(best_h)

    initial_guesses = []
    for _ in range(5):
        if time.time() > budget_s:
            break
        guess = perturb(best_h, method='gaussian', noise=0.15, frac=0.2)
        initial_guesses.append(guess)

    for guess in initial_guesses:
        if time.time() > budget_s:
            break
        res = minimize(
            objective,
            guess,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    for _ in range(5):
        if time.time() > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8,
            options={'maxiter': 200}
        )
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
