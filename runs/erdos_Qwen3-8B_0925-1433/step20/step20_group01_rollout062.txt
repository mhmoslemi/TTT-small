import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    original_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    initial_guesses = [original_scaled]

    for _ in range(4):
        perturbed = original_scaled + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)

    for _ in range(2):
        indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        perturbed = original_scaled.copy()
        perturbed[indices] = 1 - perturbed[indices]
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)

    uniform_guess = np.random.rand(n_points)
    uniform_guess = uniform_guess * target_sum / np.sum(uniform_guess)
    initial_guesses.append(uniform_guess)

    best_h = original_scaled.copy()
    best_c5 = np.max(np.correlate(original_scaled, 1 - original_scaled, mode='full')) * dx

    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum,
    }]

    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            x0=guess,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=cons,
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            res = minimize(
                fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
                x0=best_h,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                constraints=cons,
                tol=1e-6,
                options={'maxiter': 50}
            )
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
