import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    # Perturbations
    def gen_perturb():
        pert = h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        pert += noise
        pert = np.clip(pert, 0, 1)
        pert = pert * target_sum / np.sum(pert)
        return pert

    def gen_invert():
        pert = h.copy()
        mask = np.random.rand(n_points) < 0.3
        pert[mask] = 1 - pert[mask]
        pert = pert * target_sum / np.sum(pert)
        return pert

    initial_guesses = [h.copy(), gen_pert(), gen_invert(), np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))]

    for guess in initial_guesses:
        def obj(x):
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def cons(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1)] * n_points
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}], tol=1e-6, options={'maxiter': 100})
        if res.success:
            curr_c5 = res.fun
            if curr_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), curr_c5

    # Gradient-based refinement
    for _ in range(200):
        if time.time() - time.time() >= budget_s:
            break
        grad = np.zeros(n_points)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)

        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = best_h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        learning_rate = 0.3 / (1 + _ / 200)
        best_h = np.clip(best_h + learning_rate * grad, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        curr_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
