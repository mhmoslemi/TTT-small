import numpy as np
import time
from numpy.random import rand, normal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 100
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.6:
            h = initial_h.copy()
            h += normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        elif np.random.rand() < 0.3:
            h = np.random.rand(n_points)
        else:
            h = initial_h.copy()
        h = h * target_sum / np.sum(h)
        population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5, best_h = fitness[best_idx], population[best_idx].copy()

        parents = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(pop_size, 5, replace=False)
            winner = idxs[np.argmin(fitness[idxs])]
            parents.append(population[winner])

        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                child = (p1 + p2) / 2
                if np.random.rand() < 0.3:
                    child += normal(0, 0.05, n_points)
                    child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            else:
                offspring.append(parents[i])

        combined = population + offspring
        combined.sort(key=compute_c5)
        population = combined[:pop_size]

    best_h = np.clip(best_h, 0, 1)
