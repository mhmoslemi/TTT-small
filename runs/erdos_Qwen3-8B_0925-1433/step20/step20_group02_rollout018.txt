import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    num_particles = 20
    particles = []
    velocities = []
    # Start with initial_h_values
    base = np.array(initial_h_values, dtype=float)
    particles.append(base)
    velocities.append(np.zeros(n_points))
    for _ in range(num_particles - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h = base + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal bests
    personal_bests = [p.copy() for p in particles]
    best_h = base.copy()
    best_c5 = compute_c5(best_h)
    
    w = 0.729
    c1, c2 = 1.494, 1.494
    max_iter = 100
    time_start = time.time()
