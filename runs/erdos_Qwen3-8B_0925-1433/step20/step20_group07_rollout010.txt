import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Spectral initialization
    h_fft = np.fft.fft(initial_h_values)
    h_fft[abs(h_fft) < 1e-6] = 0  # Eliminate noise
    h_spectral = np.fft.ifft(h_fft).real
    h_spectral = np.clip(h_spectral, 0, 1)
    h_spectral = h_spectral * target_sum / np.sum(h_spectral)
    best_h = h_spectral.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Population and initial candidates
    population = [initial_h_values.copy(), h_spectral.copy()]
    for _ in range(50):
        p = initial_h_values.copy()
        p += np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        if len(population) > 100:
            population.pop(0)
        new_pop = []
        for cand in population:
            res = minimize(obj, cand, method='SLSQP',
                           bounds=[(0, 1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           tol=1e-6)
            if res.success:
                new_pop.append(res.x)
        population += new_pop[:30]
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
