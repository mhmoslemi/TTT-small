import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return h
        return h * target_sum / s

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def generate_initial_population():
        pop = [np.array(initial_h_values, dtype=float)]
        for _ in range(4):
            h = np.array(initial_h_values, dtype=float)
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = normalize(h)
            pop.append(h)
        return pop

    population = generate_initial_population()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break

        candidates = [(evaluate(h), h) for h in population]
        candidates.sort()
        current_best = candidates[0][0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = candidates[0][1]

        elites = np.array([c[1] for c in candidates[:5]])
        new_pop = elites.tolist()
        for _ in range(len(population) - 5):
            p1 = elites[np.random.choice(5)]
            p2 = elites[np.random.choice(5)]
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            child += np.random.normal(0, 0.05, n_points)
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_pop.append(child)

        population = new_pop

    def obj(x):
        h = normalize(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                   constraints={'type':'eq', 'fun':lambda x: np.sum(x)-target_sum}, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
