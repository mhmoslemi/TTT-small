        if time.time() - start_time > budget_s:

    best_h = np.clip(best_h, 0, 1)

    variants.append(np.random.rand(n_points) * 0.5)

    # Clip and rescale variants
    for i in range(len(variants)):
        variants[i] = np.clip(variants[i], 0, 1)
        variants[i] = variants[i] * target_sum / np.sum(variants[i])

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0.0, 1.0)] * n_points
        res = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            maxiter=50,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        if res.success:
            current_h = res.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
