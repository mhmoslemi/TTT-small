def run(seed=42, budget_s=600):

    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5

    # Define the objective function
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Define the gradient function
    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k_max = k_max_idx - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j1 = i - k_max
            if 0 <= j1 < n_points:
                grad[i] += (1 - h[j1])
            j2 = i + k_max
            if 0 <= j2 < n_points:
                grad[i] -= h[j2]
        return -grad

    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    # Optimization settings
    start_time = time.time()

            break
