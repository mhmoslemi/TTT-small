import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def norm_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def mutate(h, std=0.05):
        noise = np.random.normal(0, std, n_points)
        h = h + noise
        return norm_h(h)

    def crossover(a, b):
        idx = np.random.randint(1, n_points - 1)
        return np.concatenate([a[:idx], b[idx:]])

    def generate_initial_population():
        pop = [np.array(initial_h_values, dtype=float)]
        for _ in range(4):
            h = np.array(initial_h_values, dtype=float)
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = norm_h(h)
            pop.append(h)
        return pop

    population = generate_initial_population()
    best_h = population[0]
    best_c5 = eval_c5(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([eval_c5(h) for h in population])
        elites = np.argsort(fitness)[:5]
        elites = population[elites]
        new_pop = []
        for _ in range(15):
            a, b = np.random.choice(elites, 2, replace=False)
            child = crossover(a, b)
            child = mutate(child)
            new_pop.append(child)
        population = elites.tolist() + new_pop
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)]
        if gen % 10 == 0 and best_c5 < 0.381:
            break

    res = minimize(lambda x: eval_c5(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res.success:
        best_h = res.x
    best_h = norm_h(best_h)
    return (best_h, best_c5, n_points)
