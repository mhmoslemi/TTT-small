import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial = np.array(initial_h_values, dtype=float)
    n_points = len(initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale(h):
        return h * target_sum / np.sum(h)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = scale(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = scale(x)
        return evaluate(h)

    def grad(x):
        h = scale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_p = x.copy()
            h_p[i] += 1e-4
            h_p = scale(h_p)
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')[k]
            h_m = x.copy()
            h_m[i] -= 1e-4
            h_m = scale(h_m)
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')[k]
            grad[i] = (corr_p - corr_m) / (2e-4)
        return -grad

    # Initial candidates
    candidates = [initial]
    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        cand = initial + noise
        cand = np.clip(cand, 0, 1)
        cand = scale(cand)
        candidates.append(cand)

    best_h = initial.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    # Simulated Annealing
    temp = 100.0
    cooling = 0.995
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for cand in candidates:
            perturb = np.random.normal(0, 0.05, n_points)
            trial = cand + perturb
            trial = np.clip(trial, 0, 1)
            trial = scale(trial)
            trial_c5 = evaluate(trial)
            if np.random.rand() < np.exp((best_c5 - trial_c5) / temp):
                best_h = trial.copy()
                best_c5 = trial_c5
        temp *= cooling

    # L-BFGS Refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n_points, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = scale(best_h)
    return (best_h, best_c5, n_points)
