import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def corr_func(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    best_h = h_initial.copy()
    best_c5 = corr_func(best_h)
    
    def objective(x):
        h = x * target_sum / np.sum(x)
        return corr_func(h)
    
    def equality_constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1)] * n_points
    
    patterns = []

    indices = np.random.permutation(n_points)
