        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_h = perturbed_h
            best_c5 = c5_perturbed

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
