        # Reconstruct full h with symmetry
        x = np.zeros(n_points)
        for i in range(half):
            x[i] = x_vars[i]
            x[n_points - 1 - i] = x_vars[i]
        if is_odd:
            x[half] = x_vars[-1]
        # Ensure total sum is correct
        norm = np.sum(x)
        if norm == 0:
            return 1.0
        x = x * (target_sum / norm)
        # Compute correlation and max C5
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def compute_gradient(x_vars):
        # Reconstruct full h with symmetry
        x = np.zeros(n_points)
        for i in range(half):
            x[i] = x_vars[i]
            x[n_points - 1 - i] = x_vars[i]
        if is_odd:
            x[half] = x_vars[-1]
        norm = np.sum(x)
        if norm == 0:
            return np.zeros_like(x_vars)
        x = x * (target_sum / norm)
        # Compute correlation and max C5
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x_vars)
        for i in range(half):
            x_plus = np.zeros(n_points)
            for j in range(half):
                x_plus[j] = x_vars[j]
                x_plus[n_points - 1 - j] = x_vars[j]
            if is_odd and j == half - 1:
                x_plus[half] = x_vars[j] + 1e-4
            else:
                x_plus[i] += 1e-4
            norm_plus = np.sum(x_plus)
            if norm_plus == 0:
                continue
            x_plus = x_plus * (target_sum / norm_plus)
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            c5_plus = corr_plus[k_max] * dx
            x_minus = np.zeros(n_points)
            for j in range(half):
                x_minus[j] = x_vars[j]
                x_minus[n_points - 1 - j] = x_vars[j]
            if is_odd and j == half - 1:
                x_minus[half] = x_vars[j] - 1e-4
            else:
                x_minus[i] -= 1e-4
            norm_minus = np.sum(x_minus)
            if norm_minus == 0:
                continue
            x_minus = x_minus * (target_sum / norm_minus)
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')
            c5_minus = corr_minus[k_max] * dx
            grad[i] = (c5_plus - c5_minus) / (2e-4)
        return -grad

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Local optimization
        res = minimize(obj, x_vars, method='L-BFGS-B', bounds=[(0, 1)]*len(x_vars), tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = np.zeros(n_points)
            for i in range(half):
                best_h[i] = res.x[i]
                best_h[n_points - 1 - i] = res.x[i]
            if is_odd:
                best_h[half] = res.x[-1]
        # Perturbative exploration
        perturbation = np.random.uniform(-0.01, 0.01, size=len(x_vars))
        perturbed = np.clip(res.x + perturbation, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            for i in range(half):
                best_h[i] = perturbed[i]
                best_h[n_points - 1 - i] = perturbed[i]
            if is_odd:
                best_h[half] = perturbed[-1]
    return (np.clip(best_h, 0, 1), best_c5, n_points)
