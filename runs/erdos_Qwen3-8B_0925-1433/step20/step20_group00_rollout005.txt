    h_initial = h_initial * target_sum / np.sum(h_initial)

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Generate additional initial patterns
    initial_patterns = []

    for _ in range(3):

        phase = np.random.rand() * period

            if (i - phase) % period < period / 2:
                pattern[i] = 1
        pattern = pattern * target_sum / np.sum(pattern)
        initial_patterns.append(pattern)
    for _ in range(2):
        # Randomly spaced 1s
        indices = np.random.permutation(n_points)
        pattern = np.zeros(n_points)
        pattern[indices[:int(n_points/2)]] = 1
        pattern = pattern * target_sum / np.sum(pattern)
        initial_patterns.append(pattern)

    # Optimization function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints
    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum
    })

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

    for pattern in initial_patterns:
