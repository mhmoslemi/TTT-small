from scipy.optimize import minimize
import time

    best_c5 = 0.381  # Initial bound

    def generate_pattern(m_vals):
        patterns = []
        for m in m_vals:
            h = np.zeros(n_points)
            interval = 1.0 / m
            spacing = 1.0 / (m - 1) if m > 1 else 0.0
            for i in range(m):
                start = i * (interval + spacing)
                end = start + interval
                for j in range(n_points):
                    x = j * dx
                    if start <= x <= end:
                        h[j] = 1.0
            total = np.sum(h)
            if total > 0:

                patterns.append(h)
        return patterns

        c5 = np.max(corr) * dx
