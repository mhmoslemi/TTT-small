        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + k
            i_minus = i - k
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        h = h_new
    
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * (2.0 / n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, c5, n_points)
