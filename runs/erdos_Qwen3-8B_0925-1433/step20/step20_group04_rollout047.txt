import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate multiple starting points
    perturbations = []
    for _ in range(5):
        noise = np.random.normal(0, 0.07, n_points)
        p = initial_h_values + noise
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        perturbations.append(p)
    for _ in range(3):
        p = np.random.rand(n_points)
        p = p * target_sum / np.sum(p)
        perturbations.append(p)

    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    start_time = time.time()

    for start in perturbations:
        start = start.copy()
        # Define objective
        def objective(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        # Gradient via finite differences
        def compute_gradient(x):
            grad = np.zeros_like(x)
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            for i in range(n_points):
                h_plus = x.copy()
                h_plus[i] += 1e-4
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
                h_minus = x.copy()
                h_minus[i] -= 1e-4
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
                grad[i] = (corr_plus - corr_minus) / (2e-4)
            return -grad

        # Constraints
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        result = minimize(objective, start, method='L-BFGS-B', jac=compute_gradient, bounds=bounds, constraints=constraints, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        if time.time() - start_time > budget_s:
            break

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
