from scipy.optimize import minimize

    def structured_pattern(m):
        L = 1.0 / m
        d = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + d)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    m_values = [5, 10, 15, 20, 25, 30, 35, 40]

    for m in m_values:
        h = structured_pattern(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)

            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
