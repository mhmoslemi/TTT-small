import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    
    # Generate diverse initial guesses
    guesses = [
        h_initial,
        h_initial[::-1],         # Mirror
        1 - h_initial,           # Invert
        h_initial + np.random.normal(0, 0.05, n_points),  # Perturb
        h_initial * 0.8 + np.random.uniform(0.2, 0.5, n_points),  # Hybrid
    ]
    
    for g in guesses:
        g = np.clip(g, 0, 1)
        g = g * target_sum / np.sum(g)
        
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def grad(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            n = len(h)
            k_actual = k_max - (n - 1)
            grad = np.zeros_like(x)
            for i in range(n):
                i_p = i + k_actual
                term1 = 1 - h[i_p] if 0 <= i_p < n else 0.0
                i_m = i - k_actual
                term2 = h[i_m] if 0 <= i_m < n else 0.0
                grad[i] = term1 - term2
            return grad
        
        bounds = [(0,1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        res = minimize(obj, g, method='SLSQP', bounds=bounds, constraints=cons, jac=grad, options={'maxiter': 150})
        
        if res.success:
            current_c5 = obj(res.x)
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
