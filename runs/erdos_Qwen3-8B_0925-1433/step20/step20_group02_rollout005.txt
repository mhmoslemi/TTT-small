import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with the provided start and some perturbations
    population_size = 20
    population = []
    for _ in range(population_size):
        if _ == 0:
            h = initial_h_values.copy()
        else:
            h = initial_h_values.copy()
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        population.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()

    # Multi-phase genetic algorithm with elitism and adaptive mutation
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(h) for h in population]
        elite_idx = np.argsort(scores)[:5]
        current_best_h = population[np.argmin(scores)]
        current_best_c5 = np.min(scores)

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h

        new_population = [population[i] for i in elite_idx]
        while len(new_population) < population_size:
            parent = np.random.choice(population)
            noise = np.random.normal(0, 0.05, n_points)
            child = parent + noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)

        population = new_population

    # Final local optimization
    best_h = np.array(current_best_h)
    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = minimize(obj, best_h, bounds=[(0, 1)] * n_points, constraints=cons, method='L-BFGS-B', tol=1e-6)
    if res.success:
        best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
