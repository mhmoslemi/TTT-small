from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    population = [h.copy() for _ in range(4)]
    for p in population:
        p += np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
