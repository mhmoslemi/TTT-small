import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Generate structured variants
    variants = [initial_h_values, 1 - initial_h_values, np.roll(initial_h_values, 1)]
    for _ in range(2):
        noise = np.random.normal(0, 0.05, n_points)
        p = initial_h_values + noise
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        variants.append(p)

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    # Simulated Annealing
    T = 100.0
    cooling_rate = 0.99
    for _ in range(400):
        if time.time() > budget_s - 10:
            break
        for variant in variants:
            h_curr = variant * target_sum / np.sum(variant)
            noise = np.random.normal(0, 0.08, n_points)
            h_perturb = h_curr + noise
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            c5_perturb = compute_c5(h_perturb)
            if c5_perturb < best_c5 or np.random.rand() < np.exp(-(c5_perturb - best_c5) / T):
                best_c5 = c5_perturb
                best_h = h_perturb.copy()
        T *= cooling_rate

    # Local refinement
    result = minimize(objective, best_h, method='L-BFGS-B', jac=compute_gradient,
                      bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                      tol=1e-6, options={'maxiter': 100})
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
