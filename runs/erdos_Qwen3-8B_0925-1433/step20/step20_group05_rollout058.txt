import numpy as np

import time

def run(seed=42, budget_s=600):

        n_points = len(initial_h_values)

    else:

    tournament_size = 5
    mutation_rate = 0.1
    local_optimization_per_gen = 5
    population = []

        h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = initial_h.copy() if 'initial_h_values' in globals() else np.zeros(n_points)
    best_c5 = float('inf')

        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def tournament_selection():

        best = min(tournament, key=evaluate_fitness)
