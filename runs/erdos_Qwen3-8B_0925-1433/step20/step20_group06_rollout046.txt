import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381

    # Generate diverse patterns
    def generate_pattern(m):
        h = np.zeros(n_points)
        for _ in range(m):
            pos = np.random.uniform(0, 2 - 1/m)
            start = pos
            end = pos + 1/m
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return h, c5

    # Evaluate initial patterns
    candidates = []
    for m in [10, 15, 20]:
        h, c5 = generate_pattern(m)
        candidates.append((h, c5))
    best_h, best_c5 = min(candidates, key=lambda x: x[1])

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # SLSQP optimization
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=compute_gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
