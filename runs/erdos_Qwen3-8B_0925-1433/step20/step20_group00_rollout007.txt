        idx = np.random.choice(n_points, size=int(n_points//2), replace=False)
        h_rand = np.zeros(n_points)
        h_rand[idx] = 1.0
        h_rand = h_rand * target_sum / np.sum(h_rand)
        patterns.append(h_rand)
        idx = np.arange(int(n_points//2))
        idx = idx * 2
        h_equi = np.zeros(n_points)
        h_equi[idx] = 1.0
        h_equi = h_equi * target_sum / np.sum(h_equi)
        patterns.append(h_equi)
        h_per = np.zeros(n_points)
        h_per[::2] = 1.0
        h_per = h_per * target_sum / np.sum(h_per)
        patterns.append(h_per)
        return patterns

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    patterns = get_initial_patterns()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    for p in patterns:
        res = minimize(objective, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
