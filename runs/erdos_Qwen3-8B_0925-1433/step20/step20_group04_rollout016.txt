import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, scale):
        noisy = h + scale * np.random.normal(0, 1, h.shape)
        return np.clip(noisy * target_sum / np.sum(noisy), 0, 1)

    variations = [
        initial_h_values,
        initial_h_values[::-1],
        1 - initial_h_values,
        np.roll(initial_h_values, 1),
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points) / 2)
    ]
    processed = [v * target_sum / np.sum(v) for v in variations]
    best_h = processed[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for var in processed:
        h = var.copy()
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
