    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = initial_guess * target_sum / np.sum(initial_guess)
    
    # Start differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=200,
        seed=seed,
        func_tol=1e-8,
        disp=False
    )

    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
