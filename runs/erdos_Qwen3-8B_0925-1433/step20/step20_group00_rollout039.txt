from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    def generate_patterns():
        patterns = []

            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Even spacing
        h_even = np.zeros(n_points)
        h_even[::2] = 1.0
        h_even = h_even * target_sum / np.sum(h_even)
        patterns.append(h_even)
        # Random
        indices = np.random.permutation(n_points)
        h_rand = np.zeros(n_points)
        h_rand[indices[:int(n_points/2)]] = 1.0
        h_rand = h_rand * target_sum / np.sum(h_rand)
        patterns.append(h_rand)
        # Shifted
        h_shift = np.roll(np.array(initial_h_values, dtype=float), shift=10)

        patterns.append(h_shift)
        return patterns

    initial_patterns = generate_patterns()
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Optimize all patterns
    for pattern in initial_patterns:
