import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    # Phase 1: Adam-like optimization
    learning_rate = 0.3
    momentum = 0.9
    decay_rate = 0.99
    max_iterations = 150
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
            biased_m = m[i] / (1 - momentum ** (_ + 1))
            biased_v = v[i] / (1 - decay_rate ** (_ + 1))
            h[i] = h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    # Phase 2: Nelder-Mead refinement
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    res = minimize(obj, best_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun
    # Phase 3: Perturbed restarts
    num_trials = 5
    for _ in range(num_trials):
        if time.time() - start_time > budget_s:
            break
        perturb = np.array(initial_h_values, dtype=float)
        perturb = perturb * (1 + np.random.normal(0, 0.05, size=n_points))
        perturb = np.clip(perturb, 0, 1) * target_sum / np.sum(perturb)
        h = perturb.copy()
        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for j in range(n_points):
                m[j] = momentum * m[j] + (1 - momentum) * grad[j]
                v[j] = decay_rate * v[j] + (1 - decay_rate) * grad[j] ** 2
                biased_m = m[j] / (1 - momentum ** (i + 1))
                biased_v = v[j] / (1 - decay_rate ** (i + 1))
                h[j] = h[j] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        res = minimize(obj, h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
