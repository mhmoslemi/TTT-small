from scipy.optimize import minimize
import time

    target_sum = n_points / 2.0
    best_h = initial_h.copy()
    best_c5 = 0.381  # Initial bound

    def clip_and_norm(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Initialize population
    population = []
    for _ in range(50):
        h = np.random.rand(n_points)
        h = clip_and_norm(h)
        population.append(h)

    start_time = time.time()

        if time.time() - start_time > budget_s:
            break

        fitness = [evaluate_fitness(ind) for ind in population]

        parents = []

            candidates = np.random.choice(population, 5, replace=False)

            parents.append(best)
        # Crossover

        for i in range(0, 50, 2):
            if i+1 < 50:

                child = (a + b) / 2

        population = [p for p in population if evaluate_fitness(p) > best_c5]
