import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def objective(h_candidate):
        h = h_candidate.copy()
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * (target_sum / sum_h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Generate initial guesses
    initial_guesses = [initial_h_values.copy()]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        initial_guesses.append(h)
    
    start_time = time.time()
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=150,
            seed=seed,
            init=initial_guesses,
            polish=True,
            disp=False
        )
    except Exception as e:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=150,
            seed=seed,
            init=[initial_h_values],
            polish=True,
            disp=False
        )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        best_h = best_h * (target_sum / sum_h)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
