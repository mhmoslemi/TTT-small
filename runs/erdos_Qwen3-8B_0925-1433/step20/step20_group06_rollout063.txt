import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial patterns
    initial_candidates = [np.array(initial_h_values, dtype=float)]
    for m in [15, 25, 30]:
        h = np.zeros(n_points)
        interval_length = 2.0 / m
        offset = np.random.uniform(0, interval_length)
        for i in range(m):
            start = (i + offset) % 2
            end = start + interval_length
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:

        current_sum = np.sum(h)
        h = h * target_sum / current_sum
        initial_candidates.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381  # Initial bound

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()

        if res.success and res.fun < best_c5:

            best_c5 = res.fun

    # Perturbation refinement
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
