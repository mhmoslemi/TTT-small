import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 60
        initial_h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h.copy()
    best_c5 = 0.381
    pop_size = 50
    generations = 200
    elite_fraction = 0.2
    mutation_rate = 0.1
    mutation_std = 0.05

    def ensure_sum(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop):
        tournament = np.random.choice(pop, size=5, replace=False)
        return tournament[np.argmin([evaluate(ind) for ind in tournament])]

    population = [ensure_sum(np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1)) for _ in range(pop_size)]

    for gen in range(generations):
        if time.time() > time.time() + budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        elites = np.argsort(fitness)[:int(elite_fraction * pop_size)]
        parents = [tournament_selection(population) for _ in range(pop_size - len(elites))]
        offspring = []
        for parent1, parent2 in zip(parents[::2], parents[1::2]):
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = ensure_sum(child)
            offspring.append(child)
        for ind in offspring:
            if np.random.rand() < mutation_rate:
                perturb = np.random.normal(0, mutation_std, n_points)
                ind += perturb
                ind = np.clip(ind, 0, 1)
                ind = ensure_sum(ind)
        population = elites + offspring
        for ind in population:
            if evaluate(ind) < best_c5:
                best_c5 = evaluate(ind)
                best_h = ind.copy()
        if gen % 10 == 0:
            for ind in population[:5]:
                res = minimize(lambda x: evaluate(ensure_sum(np.clip(x, 0, 1))),
                               x0=ind, method='L-BFGS-B',
                               bounds=[(0, 1)]*n_points,
                               constraints={'type': 'eq', 'fun': lambda x: np.sum(ensure_sum(np.clip(x, 0, 1))) - target_sum})
                if res.success and evaluate(ensure_sum(res.x)) < best_c5:
                    best_c5 = evaluate(ensure_sum(res.x))
                    best_h = res.x
    return (np.clip(best_h, 0, 1), best_c5, n_points)
