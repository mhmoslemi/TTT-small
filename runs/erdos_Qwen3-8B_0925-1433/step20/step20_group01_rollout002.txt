        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

            if 0 <= i_plus < n_points:
                grad[i] += 1 - x[i_plus]
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                grad[i] -= x[i_minus]
        return grad
