import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    particles = []
    pbest = []
    velocities = []
    for _ in range(20):

        return np.max(corr) * dx

    best_c5 = evaluate(particles[0].copy())

    omega = 0.729
    c1 = 1.494
    c2 = 1.494
    vel_limit = 0.1
    max_iter = 100
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
