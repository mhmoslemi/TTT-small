import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    else:

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h.copy()
    best_c5 = 0.381

    def structured_pattern(m):
        if m == 0:
            return None
        h = np.zeros(n_points)
        total_length = 0.0
        for i in range(m):
            start = np.random.uniform(0, 2 - 1.0 / m)
            end = start + 1.0 / m
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
                if start < x < end:
                    total_length += dx
        if total_length == 0:
            return None
        h = h * target_sum / total_length
        return h

    m_values = [5, 10, 15, 20, 25, 30]
    for m in m_values:
        h = structured_pattern(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)

    start_time = time.time()

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
