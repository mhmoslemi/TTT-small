def run(seed=42, budget_s=600):
    np.random.seed(seed)

    # Initial scaled pattern
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h_initial
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Generate structured initial patterns
    patterns = [best_h]
    h_binary = np.zeros(n_points)
    h_binary[np.random.choice(n_points, size=n_points//2, replace=False)] = 1
    h_binary = h_binary * target_sum / np.sum(h_binary)
    patterns.append(h_binary)
    spacing = n_points // 4
    h_periodic = np.zeros(n_points)
    h_periodic[::spacing] = 1
    h_periodic = h_periodic * target_sum / np.sum(h_periodic)
    patterns.append(h_periodic)

    # Optimize each pattern
    for pattern in patterns:
        res = minimize(obj, pattern, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Refine
    for _ in range(3):
