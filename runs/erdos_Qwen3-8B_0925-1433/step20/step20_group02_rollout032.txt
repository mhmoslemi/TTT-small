                                   recombination=0.9, maxiter=200, 
                                   seed=seed)
    
    if time.time() - start_time > budget_s:
        best_h = initial_h_values.copy()
    else:
        best_h = result.x
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scaling = T / sum_h
    best_h = best_h * scaling
    best_h = np.clip(best_h, 0, 1)
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
