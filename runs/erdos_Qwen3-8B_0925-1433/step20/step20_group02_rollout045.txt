import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        total = np.sum(h)
        if abs(total - target_sum) < 1e-7:
            return h
        return h * target_sum / total

    def mutate(h, std):
        noise = np.random.normal(0, std, n_points)
        h = np.clip(h + noise, 0, 1)
        return rescale(h)

    def crossover(h1, h2):
        idx = np.random.randint(1, n_points - 1)
        return np.concatenate([h1[:idx], h2[idx:]])

    def generate_population():
        base = np.array(initial_h_values, dtype=float)
        variants = [base]
        for _ in range(4):
            noise = np.random.normal(0, 0.1, n_points)
            h = np.clip(base + noise, 0, 1)
            h = rescale(h)
            variants.append(h)
        return variants

    population = generate_population()
    population = [rescale(h) for h in population]
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)
    n_gen = 0
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        n_gen += 1
        scores = [evaluate(h) for h in population]
        elite_indices = np.argsort(scores)[:5]
        elites = [population[i] for i in elite_indices]
        new_pop = elites[:]
        while len(new_pop) < 20:
            p1, p2 = np.random.choice(elite_indices, 2, replace=False)
            child = crossover(population[p1], population[p2])
            if np.random.rand() < 0.3:
                child = mutate(child, 0.05)
            new_pop.append(child)
        population = new_pop
        if scores[elite_indices[0]] < best_c5:
            best_c5 = scores[elite_indices[0]]
            best_h = population[elite_indices[0]]
        if n_gen % 10 == 0 and len(population) > 0:

    res = minimize(evaluate, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x)-target_sum}, tol=1e-6)
