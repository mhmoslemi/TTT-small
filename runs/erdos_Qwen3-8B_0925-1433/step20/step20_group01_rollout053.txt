import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = 0.0 if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def update_h(h, grad, learning_rate=0.2, momentum=0.9, decay=0.99):
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for _ in range(100):
            if time.time() - start_time > budget_s:
                return h
            m = momentum * m + (1 - momentum) * grad
            v = decay * v + (1 - decay) * grad**2
            unbiased_m = m / (1 - momentum**(1 + _))
            unbiased_v = v / (1 - decay**(1 + _))
            h = h + learning_rate * unbiased_m / (np.sqrt(unbiased_v) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        return h

    start_time = time.time()
    h = update_h(h, compute_gradient(h), learning_rate=0.15, momentum=0.85)
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
