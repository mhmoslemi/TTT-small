import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def generate_perturbations(base):
        perturbations = []
        for _ in range(3):
            noise = np.random.normal(0, 0.15, n_points)
            pert = np.clip(base + noise, 0, 1)
            pert = pert * target_sum / np.sum(pert)
            perturbations.append(pert)
        for _ in range(2):
            mask = np.random.choice(n_points, size=10, replace=False)
            pert = np.copy(base)
            pert[mask] = 1 - pert[mask]
            pert = np.clip(pert, 0, 1)
            pert = pert * target_sum / np.sum(pert)
            perturbations.append(pert)
        for _ in range(2):
            rand = np.random.rand(n_points)
            rand = rand * target_sum / np.sum(rand)
            perturbations.append(rand)
        return perturbations

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start = time.time()
    perturbations = generate_perturbations(h)
    for guess in perturbations:
        if time.time() - start > budget_s:
            break
        res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    for _ in range(3):
        if time.time() - start > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
