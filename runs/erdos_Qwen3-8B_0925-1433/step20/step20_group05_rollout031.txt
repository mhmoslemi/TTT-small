import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(initial_h)
    else:
        n_points = 45
        initial_h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = 0.381

    def structured_pattern(m):
        L = 1.0 / m
        d = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + d)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    m_values = [12, 16, 20, 25, 30, 35, 40]
    candidates = []
    for m in m_values:
        h = structured_pattern(m)
        if h is not None:
            candidates.append(h)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
    for candidate in candidates:
        x0 = candidate.copy()
        for _ in range(30):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h = res.x

            c5_perturbed = np.max(corr) * dx

                best_c5 = c5_perturbed
