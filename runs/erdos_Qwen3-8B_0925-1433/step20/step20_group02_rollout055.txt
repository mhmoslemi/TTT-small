import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    T = 100.0
    cooling = 0.995
    iter_max = 500
    start = time.time()

    for _ in range(iter_max):
        if time.time() - start > budget_s:
            break
        perturb = np.random.randint(2, 5)
        indices = np.random.choice(n_points, perturb, replace=False)
        h_perturb = h.copy()
        h_perturb[indices] += np.random.normal(0, 0.1, perturb)
        h_perturb = np.clip(h_perturb, 0, 1)
        scale = target_sum / np.sum(h_perturb)
        h_perturb *= scale

        curr = compute_c5(h)
        neighbor = compute_c5(h_perturb)
        if neighbor < curr:
            h = h_perturb
            best_c5 = neighbor
        else:
            if np.random.rand() < np.exp((curr - neighbor) / T):
                h = h_perturb
                best_c5 = neighbor
        if compute_c5(h) < best_c5:
            best_h = h.copy()
            best_c5 = compute_c5(h)
        T *= cooling

    def obj(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    scale = target_sum / np.sum(best_h)
    best_h *= scale
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
