import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_initial_patterns():
        patterns = []
        # Random binary
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[indices] = 1
        h = (h * target_sum) / np.sum(h)
        patterns.append(h)
        # Periodic spaced
        spacing = n_points // 4
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % spacing == 0:
                h[i] = 1
        h = (h * target_sum) / np.sum(h)
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = (h * target_sum) / np.sum(h)
        patterns.append(h)
        # Clustered
        cluster_spacing = n_points // 6
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % cluster_spacing == 0 or (i + cluster_spacing) % (n_points // 2) == 0:
                h[i] = 1
        h = (h * target_sum) / np.sum(h)
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = (h * target_sum) / np.sum(h)
        patterns.append(h)
        return patterns

    initial_patterns = generate_initial_patterns()
    # Start with initial_h_values
    best_h = np.array(initial_h_values, dtype=float)
    best_h = (best_h * target_sum) / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    # First optimization
    for pattern in initial_patterns:
        res = minimize(obj, pattern, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    # Refinement
    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    # Final clip and rescale to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h * target_sum) / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
