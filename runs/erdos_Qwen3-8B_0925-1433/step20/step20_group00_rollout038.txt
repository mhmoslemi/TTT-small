from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_patterns():
        patterns = []
        # Perturb initial
        for _ in range(3):
            h = np.array(initial_h_values, dtype=float)
            h += np.random.uniform(-0.1, 0.1, n_points)

        # Random binary
        for _ in range(2):

            h[idx] = 1
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Periodic spacing
        for _ in range(2):
            spacing = int(n_points/3) + 1

            h = h * target_sum / np.sum(h)
            patterns.append(h)
        return patterns

    initial_patterns = generate_patterns()

    for pattern in initial_patterns:
