from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial_h = np.array(initial_h_values, dtype=np.float64)

    population = [initial_h.copy() for _ in range(50)]

        h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = initial_h.copy()
    best_c5 = 1.0

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        new_pop = []

                p2 = new_pop[i + 1]
