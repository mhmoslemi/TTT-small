            pop[i] += noise
            pop[i] = np.clip(pop[i], 0, 1)
            pop[i] *= target_sum / np.sum(pop[i])
        return pop

    mutation_rate = 0.3
    crossover_ratio = 0.8
    decay_rate = 0.99

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    while gen < max_gen and time.time() - time_start < budget_s:
        # Evaluate all individuals
        scores = np.array([eval_c5(h) for h in pop])
        best_idx = np.argmin(scores)
        current_best = scores[best_idx]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = pop[best_idx].copy()

        # Select top 8
        top_indices = np.argsort(scores)[:8]
        top = pop[top_indices]

        # Crossover and mutate
        new_pop = []
