import numpy as np
import time
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 100
    init_pop = []
    
    for _ in range(pop_size):
        if np.random.rand() < 0.6:
            h = initial_h_values.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        elif np.random.rand() < 0.3:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        init_pop.append(h)
    
    population = init_pop
    best_h = population[0]
    best_c5 = float('inf')
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        parents = random.choices(population, weights=[1-f for f in fitness], k=pop_size//2)
        offspring = []
        for _ in range(pop_size):
            p1, p2 = random.sample(parents, 2)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.3:
                scale = 0.05 if _ < 100 else 0.005
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, scale)
                        child[i] = np.clip(child[i], 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        population = offspring
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
    top_idx = np.argsort(fitness)[:10]
    top_candidates = [population[i] for i in top_idx]
    
    for h in top_candidates:
        h = h * target_sum / np.sum(h)
        def obj(x):
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
