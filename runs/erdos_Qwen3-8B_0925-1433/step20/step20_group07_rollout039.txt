            h_structured[idx] = 1.0
        else:
            idx = np.argmax(h_structured)
            h_structured[idx] = 0.0
    h0 = h_structured.copy()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h0 += np.random.normal(0, 0.05, n_points)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * target_sum / np.sum(h0)

    res = minimize(obj, h0, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
    best_h = res.x
    best_c5 = obj(best_h)

    if res.success:
        h_perturbed = best_h + np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        res_perturbed = minimize(obj, h_perturbed, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        best_h = res_perturbed.x
        best_c5 = obj(best_h)

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
