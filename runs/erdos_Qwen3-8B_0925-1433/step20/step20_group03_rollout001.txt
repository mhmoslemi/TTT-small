                term1 = 1 - x[i_plus] if 0 <= i_plus < n_points else 0.0
                i_minus = i - k_actual
                term2 = x[i_minus] if 0 <= i_minus < n_points else 0.0
                grad[i] = term1 - term2
            return grad
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=constraints, jac=grad, options={'maxiter': 100, 'ftol': 1e-10})
        
        if res.success:
            current_c5 = obj(res.x)
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
