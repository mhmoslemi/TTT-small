    for _ in range(60):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(30):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(10):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = population[0]
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(ind, dx) for ind in population])
        parents = select_parents(population, fitness)
        offspring = []
        for _ in range(population_size):
            if np.random.rand() < 0.8:
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = (parent1 + parent2) / 2
            else:
                child = np.random.choice(parents)
                mask = np.random.rand(n_points) < 0.1
                child[mask] += np.random.normal(0, 0.05, sum(mask))
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            offspring.append(child)
        population = np.array(offspring)
        replace_indices = np.argsort(fitness)[-int(0.2 * population_size):]
        for i in replace_indices:
            population[i] = np.random.uniform(0, 1, n_points)
            population[i] = population[i] * target_sum / np.sum(population[i])
        for ind in population:
            current_c5 = compute_c5(ind, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind

    # Local refinement
    top_indices = np.argsort(np.array([compute_c5(population[i], dx) for i in range(len(population))]))[:10]
    for idx in top_indices:
        h = population[idx]
        def objective(x):
            x = x * target_sum / np.sum(x)
            return compute_c5(x, dx)
        def gradient(x):
            grad = np.zeros_like(x)
            for i in range(n_points):
                x_plus = x.copy()
                x_plus[i] += 1e-4
                x_plus = x_plus * target_sum / np.sum(x_plus)
                f_plus = compute_c5(x_plus, dx)
                x_minus = x.copy()
                x_minus[i] -= 1e-4
                x_minus = x_minus * target_sum / np.sum(x_minus)
                f_minus = compute_c5(x_minus, dx)
                grad[i] = (f_plus - f_minus) / (2e-4)
            return -grad
        result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
