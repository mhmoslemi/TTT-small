import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Enhanced initialization: even spacing
    h_binary = np.zeros(n_points)
    h_binary[::2] = 1.0
    h_binary = h_binary * target_sum / np.sum(h_binary)
    current_c5 = np.max(np.correlate(h_binary, 1 - h_binary, mode='full')) * dx
    if current_c5 < best_c5:
        best_h = h_binary.copy()
        best_c5 = current_c5
    
    # Mutation and local optimization
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    population_size = 15
    population = [best_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for i in range(population_size):
            x = population[i].copy()
            if np.random.rand() < 0.3:
                x += np.random.normal(0, 0.05, n_points)
                x = np.clip(x, 0, 1)
                x = x * target_sum / np.sum(x)
            else:
                x += np.random.normal(0, 0.03, n_points)
                x = np.clip(x, 0, 1)
                x = x * target_sum / np.sum(x)
            res = minimize(obj, x, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6)
            if res.success:
                new_pop.append(res.x)
            else:
                new_pop.append(x)
        # Selection
        scores = [obj(p) for p in new_pop]
        indices = np.argsort(scores)
        population = [new_pop[i] for i in indices[:10]]
        # Update best
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
