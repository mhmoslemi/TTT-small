import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 60
        h = np.random.uniform(0.2, 0.8, n_points)
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def normalize(h):
        h = h.copy()
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return np.clip(h, 0, 1)

    def tournament_selection(pop, k=5):
        indices = np.random.randint(len(pop), size=k)
        return pop[indices[np.argmin([evaluate(pop[i]) for i in indices])]]

    def crossover(parent1, parent2):
        child = (parent1 + parent2) / 2
        return normalize(child)

    def mutate(ind, std=0.05):
        idx = np.random.choice(len(ind), int(len(ind) * 0.1), replace=False)
        ind[idx] += np.random.normal(0, std, len(idx))
        return normalize(ind)

    pop_size = 50
    pop = [normalize(np.random.uniform(0.2, 0.8, n_points)) for _ in range(pop_size)]
    pop[0] = best_h

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        elites = [pop[i] for i in np.argsort([evaluate(p) for p in pop])[:5]]
        new_pop = elites[:]
        for _ in range(pop_size - len(elites)):
            p1 = tournament_selection(pop)
            p2 = tournament_selection(pop)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        pop = new_pop
        for ind in pop[:5]:
            res = minimize(lambda x: -evaluate(x), ind, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success:
                candidate = res.x
                c5 = evaluate(candidate)
                if c5 < best_c5:
                    best_h = candidate
                    best_c5 = c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
