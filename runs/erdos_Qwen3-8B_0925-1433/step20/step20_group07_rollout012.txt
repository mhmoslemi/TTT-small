import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    pop_size = 10
    population = [h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        population[i] += perturb
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-5)
            if res.success:
                population[i] = res.x
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_h = cand.copy()
                best_c5 = curr_c5
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
