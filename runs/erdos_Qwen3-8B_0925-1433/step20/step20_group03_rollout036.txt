import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def generate_transformed(h):
        yield h
        ff = np.fft.fft(h)
        phase = np.random.uniform(0, 2 * np.pi, len(ff))
        h_phase = np.fft.irfft(ff * np.exp(1j * phase))
        h_phase = h_phase * target_sum / np.sum(h_phase)
        yield h_phase
        h_reversed = h[::-1] * target_sum / np.sum(h[::-1])
        yield h_reversed
        noise = np.random.normal(0, 0.05, len(h))
        h_noise = np.clip(h + noise, 0, 1)
        h_noise = h_noise * target_sum / np.sum(h_noise)
        yield h_noise

    for h_start in generate_transformed(h_initial):
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def compute_gradient(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            n = len(h)
            k_actual = k_max - (n - 1)
            grad = np.zeros_like(x)
            for i in range(n):
                i_plus = i + k_actual
                term1 = 1.0 - h[i_plus] if 0 <= i_plus < n else 0.0
                i_minus = i - k_actual
                term2 = h[i_minus] if 0 <= i_minus < n else 0.0
                grad[i] = term1 - term2
            return grad

        res = minimize(obj, h_start, method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=compute_gradient, options={'maxiter': 200})
        if res.success:
            current_c5 = obj(res.x)
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
