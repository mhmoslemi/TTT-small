import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = np.array(initial_h_values, dtype=float)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, dx)

    temp = 100.0
    cooling_rate = 0.995
    iterations = 500
    start_time = time.time()

    while iterations > 0 and time.time() - start_time < budget_s:
        idx = np.random.choice(n_points, size=3, replace=False)
        noise = np.random.normal(0, 0.1, 3)
        neighbor_h = current_h.copy()
        neighbor_h[idx] += noise
        neighbor_h = np.clip(neighbor_h, 0, 1)
        neighbor_sum = np.sum(neighbor_h)
        if abs(neighbor_sum - target_sum) > 1e-6:
            neighbor_h *= target_sum / neighbor_sum
        neighbor_c5 = compute_c5(neighbor_h, dx)
        current_c5 = compute_c5(current_h, dx)
        delta = neighbor_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = neighbor_h.copy()
            if neighbor_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = neighbor_c5
        temp *= cooling_rate
        iterations -= 1

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1)] * n_points
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)

    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = compute_c5(final_h, dx)
    return (final_h, final_c5, n_points)
