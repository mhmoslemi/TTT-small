import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def spectral_transform(h):
        fft_h = np.fft.fft(h)
        freqs = np.fft.fftfreq(n_points)
        mask = np.abs(freqs) < 0.3
        fft_h[~mask] = 0
        new_h = np.fft.ifft(fft_h).real
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    # Initial candidate
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = spectral_transform(h)
    population = [h.copy()]
    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_candidates = []
        for cand in population:
            new_candidate = spectral_transform(cand)
            new_candidates.append(new_candidate)
            new_candidate = cand + np.random.normal(0, 0.05, n_points)
            new_candidate = np.clip(new_candidate, 0, 1)
            new_candidate = new_candidate * target_sum / np.sum(new_candidate)
            new_candidates.append(new_candidate)
        for cand in new_candidates:
            res = minimize(obj, cand, method='SLSQP', bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
            if res.success:
                curr_h = res.x
                curr_c5 = obj(curr_h)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = curr_h.copy()
                population.append(curr_h)
        # Keep top 10 candidates
        population = sorted(population, key=lambda x: obj(x))[:10]

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
