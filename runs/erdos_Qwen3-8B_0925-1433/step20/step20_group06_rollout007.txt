import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()

    best_c5 = np.max(corr) * dx

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=5, replace=False)
        new_h = best_h.copy()
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            new_h[idx] += delta
            new_h[idx] = np.clip(new_h[idx], 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            continue
        new_h = new_h * target_sum / new_sum
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = c5
        if np.random.rand() < 0.05:
            large_indices = np.random.choice(n_points, size=2, replace=False)
            for idx in large_indices:
                delta = np.random.uniform(-0.2, 0.2)
                temp_h = best_h.copy()
                temp_h[idx] += delta
                temp_h[idx] = np.clip(temp_h[idx], 0, 1)
