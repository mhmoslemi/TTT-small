from scipy.optimize import minimize
import time

            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Structured patterns
        for _ in range(3):
            # Regular spacing with jitter
            spacing = [i / (n_points/2) for i in range(len(initial_h_values))]
            h = np.zeros(n_points)
            for i, val in enumerate(spacing):
                if i < len(spacing)/2:
                    h[int(val*n_points)] = 1
            h = h * target_sum / np.sum(h)
            patterns.append(h)
            # Anti-regular spacing
            h = np.zeros(n_points)
            for i, val in enumerate(spacing):
                if i >= len(spacing)/2:
                    h[int(val*n_points)] = 1
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        return patterns
    
    initial_patterns = generate_initial_patterns()
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):

        return np.max(corr) * dx
    
    for pattern in initial_patterns:
