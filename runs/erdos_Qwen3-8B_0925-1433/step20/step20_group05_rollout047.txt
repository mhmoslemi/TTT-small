import numpy as np
from scipy.optimize import minimize

            return None
        starts = np.random.rand(m) * 2.0
        lengths = np.random.uniform(0.2, 0.8, m)

        for i in range(n_points):
            x = i * dx
            for s, l in zip(starts, lengths):

                    h[i] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    def obj(x):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):

        corr = np.correlate(h, 1 - h, mode='full')
