import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate population
    population_size = 20
    particles = []
    velocities = []

    # Initial particle
    h_initial = np.array(initial_h_values, dtype=float)
    particles.append(h_initial)
    velocities.append(np.zeros(n_points))

    # Generate other particles
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h = h_initial + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    # Personal best and values
    personal_best = [p.copy() for p in particles]
    personal_best_vals = [compute_objective(p) for p in particles]
    global_best = np.array(particles[0])
    global_best_val = personal_best_vals[0]

    start_time = time.time()

    for _ in range(100):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            inertia = 0.729
            cognitive = 1.494
            social = 1.494
            r1, r2 = np.random.rand(), np.random.rand()
            velocity = velocities[i] * inertia + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            velocities[i] = velocity
            new_pos = particles[i] + velocity
            new_pos_clamped = np.clip(new_pos, 0, 1)
            new_pos_norm = new_pos_clamped * target_sum / np.sum(new_pos_clamped) if np.sum(new_pos_clamped) > 0 else new_pos_clamped
            new_val = compute_objective(new_pos_norm)
            if new_val < personal_best_vals[i]:
                personal_best[i] = new_pos_norm.copy()
                personal_best_vals[i] = new_val
            if new_val < global_best_val:
                global_best = new_pos_norm.copy()
                global_best_val = new_val
            particles[i] = new_pos_norm

    final_h = np.clip(global_best, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h) if np.sum(final_h) > 0 else final_h
    final_c5 = compute_objective(final_h)
    return (final_h, final_c5, n_points)
