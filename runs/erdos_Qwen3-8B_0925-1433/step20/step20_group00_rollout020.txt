import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize parent if needed
    parent = np.array(initial_h_values, dtype=float)
    parent = parent * target_sum / np.sum(parent)

    def generate_patterns():
        patterns = []
        # Uniform spacing
        s = n_points // 2
        h = np.zeros(n_points)
        h[::s] = 1
        h = h * target_sum / np.sum(h)
        patterns.append(h)
        # Periodic with phase shift
        for phase in [0, 1):
            h = np.zeros(n_points)
            for i in range(n_points):
                if (i + phase) % 2 == 0:
                    h[i] = 1
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Random with spacing
        for _ in range(2):
            indices = np.random.permutation(n_points)[:int(n_points/2)]
            h = np.zeros(n_points)
            h[indices] = 1
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        # Quasi-periodic
        h = np.sin(np.linspace(0, np.pi, n_points))
        h = np.clip((h + 1) / 2, 0, 1)
        h = h * target_sum / np.sum(h)
        patterns.append(h)
        return patterns

    patterns = generate_patterns()
    best_h = np.copy(parent)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    for p in patterns:
        res = minimize(obj, p, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(10):
        if time.time() - time.time() > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        temp = best_h + perturb
        temp = np.clip(temp, 0, 1)
        temp = temp * target_sum / np.sum(temp)
        res = minimize(obj, temp, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
