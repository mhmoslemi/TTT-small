import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 50
    max_iter = int(budget_s / 10)
    population = []
    velocities = []
    # Initialize with initial_h_values
    h_init = np.array(initial_h_values, dtype=float)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * target_sum / np.sum(h_init)
    population.append(h_init)
    velocities.append(np.zeros(n_points))
    # Generate other random particles
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
        velocities.append(np.zeros(n_points))
    # Track bests
    pbest = [p.copy() for p in population]
    pbest_c5 = [compute_c5(p, dx) for p in population]
    gbest = population[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        omega = 0.9 * (1 - _ / max_iter)
        phi1, phi2 = 2.0, 2.0
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * omega + phi1 * r1 * (pbest[i] - population[i]) + phi2 * r2 * (gbest - population[i])
            vel = np.clip(vel, -0.1, 0.1)
            velocities[i] = vel
            population[i] += vel
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
            curr_c5 = compute_c5(population[i], dx)
            if curr_c5 < pbest_c5[i]:
                pbest[i] = population[i].copy()
                pbest_c5[i] = curr_c5
            if curr_c5 < gbest_c5:
                gbest = population[i].copy()
                gbest_c5 = curr_c5
    return (np.clip(gbest, 0, 1), gbest_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
