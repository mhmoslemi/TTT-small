import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    T = 100.0
    cooling = 0.995

    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb a few indices
        perturb_indices = np.random.choice(n_points, size=3, replace=False)
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 1e-10:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = np.zeros_like(new_h)
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx

            best_c5 = new_c5
        else:

            if np.random.rand() < prob:
