import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initial population
    pop_size = 100
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.6:
            h = initial_h_values.copy()
        elif np.random.rand() < 0.9:
            h = initial_h_values + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_h = initial_h_values.copy()
    best_c5 = calc_c5(initial_h_values)
    
    start = time.time()
    
    for _ in range(200):
        if time.time() - start > budget_s:
            break
        
        # Evaluate fitness
        fitness = [calc_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        
        # Select top 20% as parents
        parent_idx = sorted_idx[:int(len(population)*0.2)]
        parents = [population[i] for i in parent_idx]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size:
            if len(parents) >= 2:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = (p1 + p2) * 0.5
                if np.random.rand() < 0.3:
                    child += np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                offspring.append(child)
            else:
                offspring.append(parents[0])
        
        # Replace population
        elite_idx = sorted_idx[:int(len(population)*0.2)]
        elite = [population[i] for i in elite_idx]
        new_pop = elite + offspring[:pop_size - len(elite)]
        population = new_pop[:pop_size]
        
        # Update best
        for h in population:
            c = calc_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
