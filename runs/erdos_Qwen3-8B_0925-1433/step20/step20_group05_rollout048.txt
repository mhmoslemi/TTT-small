        def objective(x):
            return compute_c5(x)

        def constraint(x):
            return np.sum(x) - target_sum

        res = minimize(objective, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)

        if res.success:
            best_h = res.x
            best_c5 = res.fun

        perturbation = np.random.normal(0, 0.02, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        c5_perturbed = compute_c5(perturbed_h)
        if c5_perturbed < best_c5:
            best_h = perturbed_h
            best_c5 = c5_perturbed

    return np.clip(best_h, 0, 1), best_c5, n_points
