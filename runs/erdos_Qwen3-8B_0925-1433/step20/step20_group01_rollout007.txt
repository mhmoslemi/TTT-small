import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        return np.max(corr) * dx

    num_guesses = 5
    initial_guesses = []

    for _ in range(num_guesses):
        perturbed = h + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        if np.random.rand() < 0.3:
            indices = np.random.choice(n_points, int(n_points * 0.3), replace=False)
            perturbed[indices] = 1 - perturbed[indices]
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)

    start_time = time.time()
    for guess in initial_guesses:
        def objective(x):
            return compute_c5(x)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:

        except:
            pass
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
