import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def perturb(x):
        noise = np.random.uniform(-0.1, 0.1, size=x.shape)
        return np.clip(x + noise, 0, 1) * target_sum / np.sum(x + noise)

    perturbations = [np.array(initial_h_values)]
    for _ in range(3):
        perturbations.append(perturb(perturbations[-1]))

    for h_start in perturbations:
        h = h_start.copy()
        h = h * target_sum / np.sum(h)
        start_time = time.time()
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            h -= 0.5 * grad
            h = np.clip(h, 0, 1) * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                           constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                           tol=1e-6, jac=compute_gradient)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
