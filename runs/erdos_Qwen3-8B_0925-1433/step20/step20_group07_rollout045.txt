import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_structured_binary(n_p):
        m = n_p // 2
        positions = np.linspace(0, 2, m + 1)
        idx = (positions * n_p / 2).astype(int)
        idx = np.unique(idx)
        idx = np.sort(idx)[:m]
        h = np.zeros(n_p)
        for i in idx:
            if 0 <= i < n_p:
                h[i] = 1
        return h

    initial_h = np.array(initial_h_values, dtype=float)
    structured_h = generate_structured_binary(n_points)
    def evaluate(h):
        h_norm = h * target_sum / np.sum(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    initial_c5 = evaluate(initial_h)
    s_c5 = evaluate(structured_h)
    best_h = structured_h.copy() if s_c5 < initial_c5 else initial_h.copy()
    best_c5 = min(initial_c5, s_c5)
    best_h = best_h * target_sum / np.sum(best_h)

    population = [best_h.copy()]
    for _ in range(3):
        p = best_h.copy() + np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        candidates = []
        for p in population:
            x0 = p.copy() + np.random.normal(0, 0.05, n_points)
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * target_sum / np.sum(x0)
            candidates.append(x0)
        for cand in candidates:
            res = minimize(obj, cand, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           tol=1e-6)
            if res.success:
                population.append(res.x)
        new_pop = []
        for cand in population:
            curr_c5 = obj(cand)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = cand.copy()
            new_pop.append(cand)
        population = new_pop[:5]
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
