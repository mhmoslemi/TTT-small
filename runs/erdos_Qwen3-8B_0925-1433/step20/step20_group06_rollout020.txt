def run(seed=42, budget_s=600):

    if 'initial_h_values' in globals():

    dx = 2.0 / n_points

    n_particles = 50
    particles = np.random.rand(n_particles, n_points)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_c5 = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_c5 = np.inf

        h = particles[i]
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        pbest_c5[i] = c5
        if c5 < gbest_c5:
            gbest = h.copy()
            gbest_c5 = c5

    for _ in range(1000):

        velocities = omega * velocities + phi1 * r1 * (pbest - particles) + phi2 * r2 * (gbest - particles)

            h = particles[i]
