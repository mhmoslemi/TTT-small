from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_init = np.array(initial_h_values, dtype=float)
    h = h_init * target_sum / np.sum(h_init)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    for p in population:
        p += np.random.normal(0, 0.08, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='SLSQP',
                          bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6)
            if res.success:

        for cand in population:

            if curr_c5 < best_c5:

                best_c5 = curr_c5

    h_values = np.clip(best_h, 0, 1)
