from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Strategic perturbation of best_h
        perturbed = best_h.copy()

        perturbed = np.clip(perturbed, 0, 1)
