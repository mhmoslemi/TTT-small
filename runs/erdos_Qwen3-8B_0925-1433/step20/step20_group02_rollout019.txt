import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial population
    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    pop_size = 20
    if len(population) < pop_size:
        for _ in range(pop_size - len(population)):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            population.append(h)
    population = [h.copy() for h in population]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(pop_size)]

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pbest = population.copy()
    pbest_fitness = [evaluate(h) for h in pbest]
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx]
    gbest_fitness = pbest_fitness[gbest_idx]

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.729 * velocities[i] + 1.494 * r1 * (pbest[i] - population[i]) + 1.494 * r2 * (gbest - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        # Evaluate new population
        new_fitness = [evaluate(h) for h in population]
        for i in range(pop_size):
            if new_fitness[i] < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = new_fitness[i]
        gbest_idx = np.argmin(pbest_fitness)
        gbest = pbest[gbest_idx]
        gbest_fitness = pbest_fitness[gbest_idx]

    best_h = gbest
    best_c5 = gbest_fitness
    return (best_h, best_c5, n_points)
