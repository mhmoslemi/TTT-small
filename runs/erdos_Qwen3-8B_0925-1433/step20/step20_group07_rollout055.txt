from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)
    
    pop_size = 10
    population = [h.copy() for _ in range(pop_size)]
    for p in population:
        p += np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(pop_size):
            x0 = population[i].copy()
            x0 += np.random.normal(0, 0.03, n_points)
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * target_sum / np.sum(x0)
            res = minimize(obj, x0, method='SLSQP', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6)
            if res.success:
                new_population.append(res.x)
        population = new_population

                best_h = cand.copy()
                best_c5 = curr_c5
