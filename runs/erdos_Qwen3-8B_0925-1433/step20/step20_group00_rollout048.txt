        patterns.append(h)
        h = initial_pattern.copy()
        h += np.random.uniform(-0.2, 0.2, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        patterns.append(h)
        return patterns

    patterns = generate_patterns()

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for p in patterns:
        res = minimize(obj, p, method='SLSQP',
                       bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-8)
        if res.success:
            current_h = np.clip(res.x, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-8)
        if res.success:
            current_h = np.clip(res.x, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
