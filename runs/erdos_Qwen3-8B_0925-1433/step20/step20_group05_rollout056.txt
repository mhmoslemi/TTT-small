import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 60
        initial_h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h.copy()
    best_c5 = 0.381

    def structured_pattern(m):
        L = 1.0 / m
        d = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + d)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    m_values = [8, 12, 16, 20, 24, 28]
    for m in m_values:
        h = structured_pattern(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    def bounds_func():
        return [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds_func(), constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x

        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

        c5_perturbed = np.max(corr) * dx
