import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize with the parent and generate structured patterns
    initial_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    # Generate additional initial patterns for diversity
    def generate_patterns(base_pattern):
        patterns = [base_pattern]
        # Add perturbed versions
        for _ in range(3):
            p = base_pattern + np.random.uniform(-0.1, 0.1, n_points)
            p = np.clip(p, 0, 1)
            p = p * target_sum / np.sum(p)
            patterns.append(p)
        return patterns

    initial_patterns = generate_patterns(initial_h)

    # Add structured patterns
    for _ in range(2):
        # Periodic pattern
        s = 2
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % s == 0:
                h[i] = 1
        h = h * target_sum / np.sum(h)
        initial_patterns.append(h)
        # Random binary pattern
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:int(n_points/2)]] = 1
        h = h * target_sum / np.sum(h)
        initial_patterns.append(h)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Optimization loop
    for pattern in initial_patterns:
        res = minimize(compute_c5, pattern, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        if res.success:
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    # Refinement phase
    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(compute_c5, perturbed, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-10)
        if res.success:
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
