import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Generate initial particles with perturbations
    particles = [initial_h_values.copy()]
    noise = np.random.normal(0, 0.1, n_points)
    for i in range(2):
        p = initial_h_values + noise
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        particles.append(p)
    num_particles = len(particles)
    velocities = [np.zeros(n_points) for _ in range(num_particles)]

    # PSO parameters
    inertia = 0.8
    cognitive = 1.4
    social = 1.4
    max_iter = 200
    start_time = time.time()
    global_best = particles[0].copy()
    global_best_c5 = float('inf')

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (particles[i] - global_best) + social * r2 * (particles[i] - global_best)
            # Update position
            particles[i] += velocities[i]
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Normalize to target sum
            sum_p = np.sum(particles[i])
            if abs(sum_p) > 1e-6:
                particles[i] = particles[i] * target_sum / sum_p
            else:
                particles[i] = np.zeros_like(particles[i])
                particles[i][0] = target_sum
            # Evaluate C5
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            # Update global best
            if current_c5 < global_best_c5:
                global_best = h.copy()
                global_best_c5 = current_c5

    # SQP refinement on the best particle
    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result = minimize(objective, global_best, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-6)
    best_h = np.clip(result.x, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, result.fun, n_points)
