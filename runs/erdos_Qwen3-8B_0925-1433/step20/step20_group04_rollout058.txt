import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    pop_size = 100
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.6:
            h = initial_h + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population])
        indices = np.argsort(fitness)
        parents = population[indices[:int(pop_size * 0.2)]]
        offspring = []
        for _ in range(pop_size):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(len(parents), 2, replace=False)
                child = (parents[p1] + parents[p2]) / 2
            else:
                p = np.random.choice(len(parents))
                child = parents[p].copy()
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            offspring.append(child)
        population = np.vstack([population, offspring])
        population = population[np.argsort(np.array([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]))[:pop_size]]
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0]

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
