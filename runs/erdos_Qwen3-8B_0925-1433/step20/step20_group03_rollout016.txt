def run(seed=42, budget_s=600):

        guesses.append(np.flip(base))
        guesses.append(1 - base)
        noise = np.random.normal(0, 0.1, len(base))

    for guess in generate_guesses(h_initial):
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        if np.sum(guess) == 0:
            continue

        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def compute_gradient(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
