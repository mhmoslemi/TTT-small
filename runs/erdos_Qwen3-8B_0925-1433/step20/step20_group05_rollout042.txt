    population = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    for i in range(len(population)):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_generations = 200
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        fitnesses = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitnesses.append(c5)
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        parents = []
        for _ in range(50):
            tournament = np.random.choice(len(population), 5, replace=False)
            best_t = tournament[np.argmin([fitnesses[i] for i in tournament])]
            parents.append(population[best_t])
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                off = (p1 + p2) / 2
                off = np.clip(off, 0, 1)
                total = np.sum(off)
                if total == 0:
                    off = np.full(n_points, 1.0 / n_points)
                else:
                    off = off * target_sum / total
                offspring.append(off)
        if len(parents) % 2 == 1:
            offspring.append(parents[-1].copy())
        mutated = []
        for off in offspring:
            mut_idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            off_mut = off.copy()
            off_mut[mut_idx] += np.random.normal(0, 0.05, len(mut_idx))
            off_mut = np.clip(off_mut, 0, 1)
            total = np.sum(off_mut)
            if total == 0:
                off_mut = np.full(n_points, 1.0 / n_points)
            else:
                off_mut = off_mut * target_sum / total
            mutated.append(off_mut)
        new_pop = mutated
        fitnesses_new = []
        for h in new_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitnesses_new.append(c5)
        top_idx = np.argsort(fitnesses_new)[:5]
        top_h = [new_pop[i] for i in top_idx]
        for i in range(len(top_h)):
            h = top_h[i]
            def obj(x):
                x_clipped = np.clip(x, 0, 1)
                sum_clipped = np.sum(x_clipped)
                if sum_clipped == 0:
                    return float('inf')
                x_norm = x_clipped * target_sum / sum_clipped
                corr = np.correlate(x_norm, 1 - x_norm, mode='full')
                return np.max(corr) * dx
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds)
            if res.success:
                top_h[i] = res.x
        for i in top_idx:
            new_pop[i] = top_h[np.argmin([fitnesses_new[i], np.max(np.correlate(new_pop[i], 1-new_pop[i], mode='full'))*dx])]
        population = new_pop
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return (np.clip(best_h, 0, 1), c5_bound, n_points)
