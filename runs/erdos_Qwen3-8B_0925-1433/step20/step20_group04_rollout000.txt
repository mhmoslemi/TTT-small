import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = initial_h_values * target_sum / np.sum(initial_h_values)

    for p in [initial_h_values[::-1], 1 - initial_h_values, 0.5 + 0.5 * np.sin(np.pi * (np.linspace(0, 2, n_points) - 1) / 2)]:
        p = p * target_sum / np.sum(p)
        perturbations.append(p)
    perturbations.append(initial_h_values + np.random.normal(0, 0.05, n_points))

    # Simulated Annealing
    best_h = perturbations[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    temp = 100.0
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        for p in perturbations:
            noise = np.random.normal(0, 0.1, n_points)
            h = p + noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

            c5 = np.max(corr) * dx
            if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / temp):
                best_c5 = c5
                best_h = h.copy()

    # L-BFGS Refinement

        corr = np.correlate(h, 1 - h, mode='full')
