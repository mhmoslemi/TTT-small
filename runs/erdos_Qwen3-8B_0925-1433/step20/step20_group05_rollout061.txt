import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_pattern(m):
        if m == 0:
            return np.zeros(n_points)
        l = 1.0 / m
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (2.0 / m) - l / 2.0
            end = start + l
            for j in range(n_points):
                x = j * dx
                if start <= x < end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    candidates = []
    m_values = [8, 10, 12, 14, 16, 18, 20, 22, 24]
    for m in m_values:
        h = generate_pattern(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            candidates.append((c5, h))

    if candidates:
        best_c5, best_h = min(candidates, key=lambda x: x[0])
    else:
        best_h = np.zeros(n_points)
        best_c5 = 0.381

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        # Perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = perturbed_h

    return (np.clip(best_h, 0, 1), best_c5, n_points)
