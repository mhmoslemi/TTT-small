import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    pop_size = 100
    population = []
    # 60% from initial_h_values
    for _ in range(int(0.6 * pop_size)):
        p = np.array(initial_h_values, dtype=float)
        p = p * target_sum / np.sum(p)
        population.append(p)
    # 30% perturbations
    for _ in range(int(0.3 * pop_size)):
        p = np.array(initial_h_values, dtype=float)
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
    # 10% random
    for _ in range(int(0.1 * pop_size)):
        p = np.random.uniform(0, 1, n_points)
        p = p * target_sum / np.sum(p)
        population.append(p)
    population = np.array(population)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([compute_c5(h) for h in population])
        order = np.argsort(scores)
        population = population[order[:int(0.8 * pop_size)]]

        new_population = []
        for _ in range(pop_size):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = (parent1 + parent2) / 2
            if np.random.rand() < 0.3:
                mu = 0.05 if gen < 100 else 0.005
                noise = np.random.normal(0, mu, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = np.array(new_population)
        population = np.array([h * target_sum / np.sum(h) for h in population])
        population = np.clip(population, 0, 1)

        if compute_c5(population[0]) < best_c5:
            best_h = population[0]
            best_c5 = compute_c5(best_h)

    # Local refinement
    remaining_time = budget_s - (time.time() - start_time)
    for _ in range(10):
        if remaining_time < 0:
            break
        h = best_h
        h = h * target_sum / np.sum(h)
        res = minimize(lambda x: compute_c5(x), h, method='L-BFGS-B', bounds=[(0,1)*n_points], tol=1e-6)
        if res.success and compute_c5(res.x) < best_c5:
            best_h = res.x
            best_c5 = compute_c5(best_h)
        remaining_time -= 1

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
