import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial_h = np.array(initial_h_values, dtype=np.float64)

    best_h = initial_h.copy()

    def structured_pattern(m):
        L = 1.0 / m
        d = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + d)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h
