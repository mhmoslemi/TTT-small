from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial variants
    def generate_variants(base):
        variants = []
        variants.append(base)
        variants.append(np.clip(base[::-1], 0, 1))
        variants.append(np.roll(base, 1))
        variants.append(np.clip(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 1, n_points)), 0, 1))
        variants.append(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1))
        variants.append(1.0 - base)
        return variants

    initial_variants = generate_variants(initial_h_values)
    initial_candidates = [v * target_sum / np.sum(v) for v in initial_variants]

    def compute_c5(h):
        h_norm = h * target_sum / np.sum(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    best_h = initial_candidates[0]
    best_c5 = compute_c5(best_h)

    # PSO parameters
    n_particles = 50
    particles = np.array([np.clip(c, 0, 1) for c in initial_candidates])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = particles.copy()
    global_best = particles[0].copy()
    best_fitness = np.array([compute_c5(p) for p in particles])

    for _ in range(400):
        if time.time() > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 1.4 * r1 * (personal_best[i] - particles[i]) + 1.4 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            current_fitness = compute_c5(particles[i])
            if current_fitness < best_fitness[i]:
                personal_best[i] = particles[i].copy()
                best_fitness[i] = current_fitness
        if np.min(best_fitness) < best_c5:
            best_c5 = np.min(best_fitness)
            global_best = particles[np.argmin(best_fitness)].copy()

        h = x * target_sum / np.sum(x)
        return compute_c5(h)

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
