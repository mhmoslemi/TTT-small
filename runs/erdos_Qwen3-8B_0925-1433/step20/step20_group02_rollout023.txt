import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()

    T = 100.0
    cooling_rate = 0.995
    iterations = 500
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5_perturbed = np.max(corr_perturbed) * dx

        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        else:
            prob = np.exp((best_c5 - c5_perturbed) / T)
            if np.random.rand() < prob:
                h = h_perturbed.copy()
        T *= cooling_rate

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
    if res.success:
        h = res.x
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
    return (h, best_c5, n_points)
