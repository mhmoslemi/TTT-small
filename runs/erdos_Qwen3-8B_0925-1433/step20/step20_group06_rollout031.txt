from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

        h_initial = np.array(initial_h_values, dtype=float)

    target_sum = n_points / 2.0

    def generate_structured():
        m = np.random.randint(15, 35)
        starts = np.random.uniform(0, 2 - 1/m, m)
        ends = starts + 1/m
        h = np.zeros(n_points)
        for i in range(m):

            for j in range(n_points):
                if start <= j*dx <= end:
                    h[j] = 1.0
        if np.sum(h) == 0:
            return np.random.uniform(0, 1, n_points)

        return h

    patterns = [generate_structured() for _ in range(5)]
    best_h = h_initial.copy()
    best_c5 = np.inf

    for p in patterns:
        corr = np.correlate(p, 1-p, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:

            best_h = p.copy()

    def obj(x):

        return np.max(corr) * dx

        k = np.argmax(corr)
