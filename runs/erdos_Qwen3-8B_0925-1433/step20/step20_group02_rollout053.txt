from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        return np.max(corr) * dx

            return np.zeros_like(h)

        return h * scale

        noise = np.random.normal(0, 0.1, n_points)

        h = clip_and_rescale(h)
        pop.append(h)

    pop = pop[:pop_size]

    best_h = initial_h_values.copy()
