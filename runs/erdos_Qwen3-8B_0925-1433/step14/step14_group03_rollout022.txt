                best_c5 = current_c5

        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            new_c5 = compute_correlation(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    h_final = np.clip(best_h, 0, 1)
    sum_h_final = np.sum(h_final)
    if sum_h_final > 0:
        h_final = h_final * target_sum / sum_h_final

    final_c5 = compute_correlation(h_final, dx)
    return (h_final, final_c5, n_points)
