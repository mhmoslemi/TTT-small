import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    # Initialize population from initial_h_values with noise
    pop_size = 15
    initial_guess = initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    sum_initial = np.sum(initial_guess)
    if sum_initial > 0:
        initial_guess = initial_guess * target_sum / sum_initial
    init_population = [initial_guess + np.random.normal(0, 0.1, n_points) for _ in range(pop_size)]

    # Convert init_population to numpy arrays
    init_population = [np.clip(arr, 0, 1) for arr in init_population]
    init_population = [arr * target_sum / np.sum(arr) if np.sum(arr) > 0 else np.zeros(n_points) for arr in init_population]
    init_population = np.array(init_population)

    # Run DE
    if time.time() - start_time < budget_s * 0.7:
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=15, mutation=(0.5, 1), recombination=0.9,
            maxiter=500, tol=1e-6, seed=seed, initial_population=init_population
        )
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = initial_guess
        best_c5 = objective(best_h)

    # Local refinement with L-BFGS-B
    if time.time() - start_time < budget_s * 0.8:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if res.success:
            best_h, best_c5 = res.x, res.fun

    # Final feasibility check and return
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    return (best_h, best_c5, n_points)
