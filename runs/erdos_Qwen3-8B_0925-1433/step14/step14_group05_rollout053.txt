import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        sum_h_ = np.sum(h_)
        if sum_h_ > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')

        return worst * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_c5 = obj(h)
    start_time = time.time()
    
    # Nelder-Mead for global exploration
    for i in range(100):
        if time.time() - start_time > budget_s:

        new_h = new_h * target_sum / np.sum(new_h)

        if res.success and res.fun < best_c5:

    # SLSQP for refinement
    for i in range(80):
