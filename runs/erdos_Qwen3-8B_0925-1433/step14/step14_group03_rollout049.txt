import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h != 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.05 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5
    
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.01 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5
