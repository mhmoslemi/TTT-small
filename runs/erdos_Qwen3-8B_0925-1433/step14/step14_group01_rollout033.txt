import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial points
    initial_points = [
        initial_h_values.copy(),
        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)),
        np.zeros(n_points),
        np.random.rand(n_points) * 2 - 1,
        np.block([np.ones(n_points // 2), np.zeros(n_points // 2)]) * target_sum / np.sum(np.block([np.ones(n_points // 2), np.zeros(n_points // 2)])),
    ]

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for h0 in initial_points:
        h = np.clip(h0, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        h_ = h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.001 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h_ + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h_
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                h_ = res.x.copy()
    return (best_h, best_c5, n_points)
