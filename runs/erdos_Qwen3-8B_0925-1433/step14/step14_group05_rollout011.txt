import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def generate_block(n):
        h = np.zeros(n, dtype=float)
        for i in range(n):
            h[i] = 1 if (i % 2 == 0) else 0
        s = np.sum(h)
        diff = s - target_sum
        if diff > 0:
            idx = np.where(h == 1)[0][:int(np.ceil(diff))]
            h[idx] = 0.5
        elif diff < 0:
            idx = np.where(h == 0)[0][:int(np.ceil(-diff))]
            h[idx] = 1
        return h

    def generate_square(n):
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1
        if n % 2:
            h[-1] = 0.5
        return h

    def generate_smoothed(n, h0):
        h = np.copy(h0)
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.full(n, 0.5)
        h = h * target_sum / s
        return h

    patterns = [generate_block(n_points), generate_square(n_points), generate_smoothed(n_points, initial_h_values)]

    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    start = time.time()

    for p in patterns:
        h = p.copy()
        for _ in range(10):
            if time.time() - start > budget_s:
                break
            h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
            s = np.sum(h)
            if s > 0:
                h = h * target_sum / s
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
