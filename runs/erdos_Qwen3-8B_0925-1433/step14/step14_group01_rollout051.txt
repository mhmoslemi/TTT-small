            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = np.array(population[0])
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        scores = [compute_c5(h) for h in population]
        elite_indices = np.argsort(scores)[:int(0.2 * population_size)]
        elites = [population[i] for i in elite_indices]
        remaining = population_size - len(elite_indices)
        new_pop = elites[:]
        for _ in range(remaining // 2):
            parent1 = min(np.random.choice(population, tournament_size), key=compute_c5)
            parent2 = min(np.random.choice(population, tournament_size), key=compute_c5)
            if np.random.rand() < 0.5:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
            else:
                child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
            noise = np.random.normal(0, mutation_std, n_points)
            child1 += noise
            child1 = np.clip(child1, 0, 1)
            sum_child1 = np.sum(child1)
            child1 = child1 * target_sum / sum_child1 if sum_child1 > 0 else np.full(n_points, 0.5)
            noise = np.random.normal(0, mutation_std, n_points)
            child2 += noise
            child2 = np.clip(child2, 0, 1)
            sum_child2 = np.sum(child2)
            child2 = child2 * target_sum / sum_child2 if sum_child2 > 0 else np.full(n_points, 0.5)
            new_pop.append(child1)
            new_pop.append(child2)
        population = new_pop
        current_min = min(scores)
        if current_min < best_c5:
            best_c5 = current_min
            best_h = population[scores.index(current_min)]
        mutation_std *= mutation_decay
        tournament_size = int(tournament_size * 1.05)

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return (best_h, best_c5, n_points)
