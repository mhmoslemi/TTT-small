from scipy.optimize import minimize
import time

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def norm(h):
        return h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    h = norm(h)
    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()
    evals = 0

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = norm(new_h)
        res = minimize(obj, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
            evals += 1
        else:
            evals = 0

        new_h = np.clip(best_h + perturb, 0, 1)

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
