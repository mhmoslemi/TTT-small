    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) > 0:
            x_scaled = x_clipped * target_sum / np.sum(x_clipped)
        else:
            x_scaled = x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h_best = h.copy()

    if np.sum(h_best) > 0:
        h_best = h_best * target_sum / np.sum(h_best)

        h_best = np.zeros(n_points)
    best_c5 = obj(h_best)

    start_time = time.time()

    for _ in range(200):

        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:

    for _ in range(150):
