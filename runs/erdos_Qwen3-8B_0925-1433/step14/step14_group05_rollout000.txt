        h_initial = h_initial * target_sum / np.sum(h_initial)
    starting_points.append(h_initial)

    h_periodic = generate_periodic(n_points)
    h_periodic = np.clip(h_periodic, 0, 1)
    if np.sum(h_periodic) > 0:
        h_periodic = h_periodic * target_sum / np.sum(h_periodic)
    starting_points.append(h_periodic)

    h_random = generate_random(n_points)
    starting_points.append(h_random)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    start_time = time.time()

    for h in starting_points:
        current_h = h.copy()
        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.2 * (0.8 ** i)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum > 0:
                new_h = new_h * target_sum / new_h_sum
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                current_h = res.x

        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.05 * (0.9 ** i)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            new_h_sum = np.sum(new_h)
            if new_h_sum > 0:
                new_h = new_h * target_sum / new_h_sum
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                current_h = res.x

        for i in range(50):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.005 * (0.9 ** i)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)

                current_h = res.x

    return (best_h, best_c5, n_points)
