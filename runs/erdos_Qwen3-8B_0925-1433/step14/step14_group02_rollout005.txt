import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 45
        initial_h = np.ones(n_points) * (n_points / 2) / n_points
        initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) > 0:
            x_normalized = x_clipped * (target_sum / np.sum(x_clipped))
        else:
            x_normalized = np.zeros_like(x_clipped)
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

    # Differential Evolution for global search
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        seed=seed,
        disp=False
    )
    best_h_de = result_de.x

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_lbgfs = minimize(
        objective,
        best_h_de,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-9,
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    best_h = result_lbgfs.x

    # Final normalization and clipping
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    else:

    c5 = objective(best_h)
