            res = differential_evolution(obj, bounds, args=(), strategy='best1bin', popsize=15, mutation=(0.5, 1), recombination=0.7, tol=1e-6, maxiter=100, disp=False)
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass

        try:
            res = minimize(obj, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    else:
        best_h = np.full(n_points, 0.5)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
