import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * (target_sum / np.sum(h_))
        corr = np.correlate(h_, 1 - h_, mode='full') * dx
        max_overlap = (1 / 0.01) * np.log(np.sum(np.exp(0.01 * corr[1:-1])))
        return max_overlap

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr[1:-1])
