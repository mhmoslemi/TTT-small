import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * (n_points / 2) / np.sum(h)  # Normalize to target sum
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return float('inf')
        x_norm = x_clipped * (target_sum / sum_x)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

    start = time.time()
    # Global optimization with differential evolution
    result = differential_evolution(obj, bounds, strategy='best1bin', popsize=15, maxiter=80,
                                   tol=1e-6, seed=seed)
    best = result.x
    elapsed = time.time() - start
    if elapsed < budget_s:
        # Local refinement with L-BFGS-B
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, best, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        best = res.x

    final = np.clip(best, 0, 1)
    if np.sum(final) > 0:
        final = final * target_sum / np.sum(final)
    return (final, obj(best), n_points)
