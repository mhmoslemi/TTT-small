import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    def mutate(ind):
        perturb = np.random.normal(0, 0.1, len(ind))
        mut = np.clip(ind + perturb, 0, 1)
        return mut * target_sum / np.sum(mut)

    def eval_fitness(ind):
        return compute_c5(ind)

    def update_h(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x

    best_h = h.copy()
    best_c5 = compute_c5(h)

    population = [best_h]
    for _ in range(24):
        rand = np.random.rand(n_points)
        h_rand = np.clip(rand, 0, 1)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)

    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([eval_fitness(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
        parents = np.random.choice(range(25), 5, replace=False)
        parent = population[np.argmin(fitness[parents])]
        mut = mutate(parent)
        population = [mut] + [population[i] for i in range(25) if i != np.argmin(fitness)]

    def obj(x):
        h_ = update_h(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    return (best_h, best_c5, n_points)
