import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_c5 = float('inf')

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.full(n_points, 0.5)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        return res.x, res.fun

    # Initialize with initial seeds
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)

    else:

    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
