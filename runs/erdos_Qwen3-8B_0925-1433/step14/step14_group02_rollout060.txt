import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 50
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) > 0:
            x_scaled = x_clipped * target_sum / np.sum(x_clipped)
        else:
            x_scaled = x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    result = differential_evolution(func=obj, bounds=bounds, strategy='best1bin',
                                   popsize=15, mutation=(0.5, 1), recombination=0.9,

    best_c5 = result.fun

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)

    final_h = np.clip(best_h, 0, 1)
    if np.sum(final_h) > 0:
