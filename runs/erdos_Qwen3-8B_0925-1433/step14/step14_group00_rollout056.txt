import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial guesses
    initial_guesses = [
        np.array(initial_h_values, dtype=float) * (target_sum / np.sum(initial_h_values)),  # Original
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points) - 0.5),  # Sine wave

    ]

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_guesses[0]
    best_c5 = obj(best_h)

    for guess in initial_guesses[1:]:
        res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
