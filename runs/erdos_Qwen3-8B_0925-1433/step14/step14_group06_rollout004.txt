    bounds = [(0, 1)] * n_points

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (best_h, best_c5, n_points)
