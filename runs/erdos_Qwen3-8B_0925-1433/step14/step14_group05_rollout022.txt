            h[-1] = 1
        h = h * target_sum / np.sum(h)
        return h
    
    def generate_sinusoidal():
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5 or x > 1.5:
                h[i] = 1.0
        sum_h = np.sum(h)
        h = h * target_sum / sum_h
        return h
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx
    
    def constraint(x):
        sum_h = np.sum(x)
        return sum_h - target_sum
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': constraint}]
    
    patterns = [h, generate_periodic(), generate_sinusoidal()]
    for pattern in patterns:
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(pattern + perturb, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.full(n_points, 0.5)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full(n_points, 0.5)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
