import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured initial guess
    def generate_structured_h(n_points, dx):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5 or (x >= 1.0 and x < 1.5):
                h[i] = 1.0
        return h

    # Initialize with structured pattern or initial_h_values
    h_initial = generate_structured_h(n_points, dx)
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h_initial.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Nelder-Mead for global search
        res = minimize(obj, best_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        # Resampling if needed
        if best_c5 > 0.381 and np.random.rand() < 0.5:
            new_h = np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.full(n_points, 0.5)
            best_h = new_h
            best_c5 = obj(best_h)

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        # SQP for local refinement
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
