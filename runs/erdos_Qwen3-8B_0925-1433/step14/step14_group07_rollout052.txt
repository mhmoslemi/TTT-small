import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.maximum(0, np.correlate(h, 1 - h, mode='full')[1:-1] - 1e-5)) * dx
    start_time = time.time()

    def compute_all_overlaps(h):
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[1:-1]
        return overlaps

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        overlaps = compute_all_overlaps(best_h)
        worst_idx = np.argmax(overlaps)
        k = worst_idx
        if overlaps[worst_idx] < 1e-10:
            continue

        def sub_obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            overlap_k = corr[1 + k]
            return overlap_k * dx

        def grad_sub(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i < n_points - k:
                    grad[i] = (1 - h_[i + k]) * dx
                elif i >= k:
                    grad[i] = -h_[i - k] * dx
            return grad

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(sub_obj, best_h, method='SLSQP', jac=grad_sub, bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
        else:
            perturb_std = 0.04 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            new_c5 = np.max(np.maximum(0, np.correlate(new_h, 1 - new_h, mode='full')[1:-1] - 1e-5)) * dx
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    return (best_h, best_c5, n_points)
