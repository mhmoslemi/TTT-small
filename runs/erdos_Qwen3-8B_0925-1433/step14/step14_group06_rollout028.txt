import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def feasible(h):
        return np.isclose(np.sum(h), target_sum, atol=1e-6) and np.all((h >= 0) & (h <= 1))

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.full_like(h, 0.5)
        return h

    # Initial population
    population = [np.copy(initial_h_values)]
    for _ in range(24):
        h = np.random.rand(n_points)
        h = clip_normalize(h)
        population.append(h)
    population = np.array(population)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    def tournament_selection(pop, size=5):
        idxs = np.random.choice(len(pop), size=size, replace=False)
        return pop[idxs].mean(axis=0)

    def blend_crossover(a, b, alpha=0.5):
        return alpha * a + (1 - alpha) * b

    def gaussian_mutation(h, sigma=0.2, rate=0.1):
        mut = np.random.normal(0, sigma, h.shape)
        return np.where(np.random.rand(*h.shape) < rate, mut, 0)

    best_h = clip_normalize(initial_h_values)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = [tournament_selection(population) for _ in range(2)]
        offspring = blend_crossover(*parents)
        offspring += gaussian_mutation(offspring)
        offspring = clip_normalize(offspring)
        population = np.vstack((population, offspring))
        population = np.vstack((population[:len(population)//2], offspring))
        fitness = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_h, best_c5 = population[best_idx], fitness[best_idx]

    # Local refinement
    def refine(h):
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(evaluate, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
        return res.x if res.success else h
