from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def obj(x):

        h = scale_h(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_h_values.copy()
    best_h = scale_h(best_h)
    best_c5 = obj(best_h)

    start_time = time.time()

        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
