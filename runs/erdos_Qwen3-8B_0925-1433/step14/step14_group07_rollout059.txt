import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_overlap(h, k):
        return np.sum(h[:n_points - k] * (1 - h[k:])) * dx

    def compute_max_overlap(h):
        max_overlap = 0.0
        for k in range(1, n_points - 1):
            o = compute_overlap(h, k)
            if o > max_overlap:
                max_overlap = o
        return max_overlap

    best_h = h.copy()
    best_c5 = compute_max_overlap(best_h)
    start_time = time.time()

    for iter in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr[1:-1])
        k_star = max_idx

        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                grad[i] += (1 - best_h[i + k_star])
                grad[i + k_star] -= best_h[i]
        step_size = 0.01 / (1 + iter // 100)
        h_candidate = best_h - step_size * grad

        h_candidate = np.clip(h_candidate, 0, 1)
        sum_candidate = np.sum(h_candidate)
