import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    start_time = time.time()
    
    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_clipped = h_clipped * target_sum / sum_h
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
    
    initial_h = initial_h_values.copy()
    random_h = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    sparse_h = np.zeros(n_points)
    sparse_h[:n_points // 10] = 1.0
    sparse_h = sparse_h * target_sum / np.sum(sparse_h)
    blocky_h = np.zeros(n_points)
    block_size = n_points // 5
    for i in range(0, n_points, block_size * 2):
        blocky_h[i:i+block_size] = 1.0
        blocky_h[i+block_size:i+2*block_size] = 0.0
    blocky_h = blocky_h * target_sum / np.sum(blocky_h)
    
    initial_points = [initial_h, random_h, sparse_h, blocky_h]
    
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    
    for h in initial_points:
        current_h = h.copy()
        T = 100.0
        for _ in range(500):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, T * 0.05, n_points)
            new_h = np.clip(current_h + perturb, 0, 1)
            sum_h = np.sum(new_h)
            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            obj_new = np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
            corr_current = np.correlate(current_h, 1 - current_h, mode='full')
            obj_current = np.max(np.maximum(0, corr_current[1:-1] - 1e-5)) * dx
