    for _ in range(15):
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        population.append(h)
    population = np.array(population)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx
    
    def select_parent(pop):
        indices = np.random.choice(len(pop), 5, replace=False)
        fitness = [compute_c5(pop[i]) for i in indices]
        best_idx = indices[np.argmin(fitness)]
        return pop[best_idx]
    
    def crossover(p1, p2):
        alpha = np.random.rand()
        return alpha * p1 + (1 - alpha) * p2
    
    def mutate(h, sigma=0.2, rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                h[i] += np.random.normal(0, sigma)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h
    
    max_gen = 200
    stagnation = 0
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        parents = [select_parent(population) for _ in range(population_size)]
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                o1 = crossover(parents[i], parents[i+1])
                o2 = crossover(parents[i], parents[i+1])
                offspring.append(o1)
                offspring.append(o2)
            else:
                offspring.append(parents[i])
        mutated = [mutate(h) for h in offspring]
        population = np.array(mutated)
        idx = np.argmin([compute_c5(h) for h in population])
        if compute_c5(population[idx]) < best_c5:
            best_h = population[idx].copy()
            best_c5 = compute_c5(best_h)
        else:
            stagnation += 1
        if stagnation >= 50:
            break
    
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        return compute_c5(x)
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
    except:
        pass
    
    return (best_h, best_c5, n_points)
