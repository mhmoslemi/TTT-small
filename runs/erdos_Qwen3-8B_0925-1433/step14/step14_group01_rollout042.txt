import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.8:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        else:
            perturb = np.random.normal(0, 0.1, n_points)
            h = initial_h_values + perturb

            h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    generations = 100
    tournament_size = 5
    mut_rate = 0.1
    mut_std = 0.05
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        parents = []
        for _ in range(population_size // 2):
            contestants = np.random.choice(len(population), tournament_size, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parents.append(population[winner])
        offspring = []
