import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    res_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15,
        mutation=(0.5, 1), recombination=0.9, tol=1e-6,
        maxiter=500, seed=seed, polish=False
    )

    best_h = res_de.x
    best_c5 = res_de.fun

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else np.full(n_points, 0.5)
        res = minimize(
            objective, new_h, method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 300}
        )
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    h_values = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_values)
    if sum_h > 0:
        h_values = h_values * target_sum / sum_h
    return (h_values, best_c5, n_points)
