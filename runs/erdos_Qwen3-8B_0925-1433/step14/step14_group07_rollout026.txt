import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate initial sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[1:-1]) * dx
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        worst_k = worst_idx
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + worst_k
            if j < n_points:
                grad[i] += 1 - best_h[j]
                grad[j] -= best_h[i]
        
        learning_rate = 0.01 * (1 / (1 + _ / 100))
        h_new = best_h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * target_sum / sum_h
        else:
            h_new = np.zeros(n_points)
        
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')[1:-1]) * dx
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
    
    return (best_h, best_c5, n_points)
