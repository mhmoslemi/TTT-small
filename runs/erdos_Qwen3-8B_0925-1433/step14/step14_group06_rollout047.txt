import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(h_initial)
    if sum_initial > 0:
        h_initial = h_initial * target_sum / sum_initial
    else:
        h_initial = np.full(n_points, 0.5)
    best_h = h_initial.copy()
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')[1:-1]) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    def update_h(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            return x * target_sum / sum_x
        else:
            return np.full_like(x, 0.5)

    population_size = 25
    population = []
    for _ in range(10):
        pop = h_initial.copy()
        np.random.shuffle(pop)
        population.append(pop)
    for _ in range(population_size - 10):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        sum_h = np.sum(h_rand)
        if sum_h > 0:
            h_rand = h_rand * target_sum / sum_h
        else:
            h_rand = np.full(n_points, 0.5)
        population.append(h_rand)
    population = np.array([update_h(p) for p in population])

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(p) for p in population])
        idxs = np.argsort(fitness)
        top = idxs[:5]
        candidates = population[top]
        offspring = []
        for i in range(0, 5, 2):
            if i + 1 < 5:
                p1, p2 = candidates[i], candidates[i + 1]
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
                child = update_h(child)
                offspring.append(child)
        population = np.array([update_h(p) for p in population])
        population[-len(offspring):] = offspring
        for i in range(len(population)):
            if compute_c5(population[i]) < best_c5:
                best_h, best_c5 = population[i].copy(), compute_c5(population[i])

    def obj(x):
        h_ = update_h(x)
        return compute_c5(h_)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
