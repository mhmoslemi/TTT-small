import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * target_sum / sum_initial
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        corr_values = corr[1:-1]
        smooth_max = (1/0.01) * np.log(np.sum(np.exp(0.01 * corr_values)))
        return smooth_max * dx

    def exact_c5(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        corr_values = corr[1:-1]
        return np.max(corr_values) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = exact_c5(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    final_h = np.clip(best_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final > 0:
        final_h = final_h * target_sum / sum_final
    final_c5 = exact_c5(final_h)
    return (final_h, final_c5, n_points)
