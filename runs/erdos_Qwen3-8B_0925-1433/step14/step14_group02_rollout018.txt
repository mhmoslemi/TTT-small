        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-9, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    final_h = np.clip(best_h, 0, 1)
    if np.sum(final_h) > 0:
        final_h = final_h * target_sum / np.sum(final_h)
    else:
        final_h = best_h
    return (final_h, best_c5, n_points)
