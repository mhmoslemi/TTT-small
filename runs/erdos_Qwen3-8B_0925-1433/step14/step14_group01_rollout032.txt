import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return compute_c5(h)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial points
    def gen_init():
        points = [initial_h_values.copy()]
        rand = np.random.rand(n_points)
        rand = np.clip(rand, 0, 1) * target_sum / np.sum(rand)
        points.append(rand)
        sparse = np.zeros(n_points)
        sparse[:int(0.1 * n_points)] = 1
        sparse = sparse * target_sum / np.sum(sparse)
        points.append(sparse)
        blocky = np.zeros(n_points)
        blocky[::5] = 1
        blocky = blocky * target_sum / np.sum(blocky)
        points.append(blocky)
        return points

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    init_points = gen_init()

    for h0 in init_points:
        h = np.clip(h0, 0, 1) * target_sum / np.sum(h0) if np.sum(h0) > 0 else np.zeros(n_points)
        # Simulated Annealing
        temp = 100.0
        for _ in range(500):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.05 * temp, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            current = compute_c5(h)
            new_c5 = compute_c5(new_h)
            if np.random.rand() < np.exp(-(new_c5 - current) / temp):
                h = new_h
            temp *= 0.85
        # SLSQP refinement
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.0005, n_points)
