import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured patterns
    m_vals = [9, 15, 20]
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf

    for m in m_vals:
        s = 1.0 / (m - 1)
        l = 1.0 / m
        h_candidate = np.zeros(n_points)
        for j in range(m):
            start = j * s
            end = start + l
            for i in range(n_points):
                x_i = i * dx
                if start <= x_i < end:
                    h_candidate[i] = 1.0
        # Normalize
        sum_h = np.sum(h_candidate) * dx
        h_candidate = h_candidate * (1.0 / sum_h)
        # Compute C5
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        current_c5 = np.max(corr) * dx
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()

    # Also consider the initial_h_values as a candidate
    # Compute its C5
    corr_initial = np.correlate(best_h, 1 - best_h, mode='full')
    current_c5_initial = np.max(corr_initial) * dx
    if current_c5_initial < best_c5:
        best_c5 = current_c5_initial
        best_h = best_h.copy()

    # Now perform local optimization
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
    # Initial optimization
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    # Perturb and refine
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Add random noise
        perturb = np.random.uniform(-0.1, 0.1, n_points) * best_h
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        # Optimize this perturbed version

    final_c5 = np.max(corr) * dx
