from scipy.optimize import minimize
import time

            h = np.full(n_points, 0.5)

        corr = np.correlate(h, 1 - h, mode='full')

    # Initialize population
    pop_size = 100

            h = create_valid_h(h)
        else:
            h = initial_h_values.copy()
            h = create_valid_h(h + np.random.normal(0, 0.1, n_points))
        population.append(h)
    best_h = population[0]

    start_time = time.time()
