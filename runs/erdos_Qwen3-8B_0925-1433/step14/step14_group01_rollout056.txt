import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        else:
            x = np.zeros_like(x)
        return x

    def obj(x):
        h_ = clip_rescale(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h0 = clip_rescale(initial_h_values)
    best_h = h0.copy()
    best_c5 = obj(best_h)

    def generate_initial_points():
        pts = [h0]
        # Random
        r = np.random.rand(n_points)
        r = r * target_sum / np.sum(r)
        pts.append(r)
        # Sparse
        s = np.zeros(n_points)
        s[np.random.choice(n_points, size=int(n_points//10), replace=False)] = 1.0
        s = s * target_sum / np.sum(s)
        pts.append(s)
        # Blocky
        b = np.zeros(n_points)
        b[::2] = 1.0
        b = b * target_sum / np.sum(b)
        pts.append(b)
        return pts

    initial_points = generate_initial_points()

    start_time = time.time()

    for init in initial_points:
        res = minimize(obj, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.0003, n_points)
        new_h = clip_rescale(best_h + perturb)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
