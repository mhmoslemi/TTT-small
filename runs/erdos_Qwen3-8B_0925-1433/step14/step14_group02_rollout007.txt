import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 60
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(h_vec):
        h = np.clip(h_vec, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def de_fitness(h_vec):
        return obj(h_vec)

    # Differential Evolution setup
    de_popsize = 30
    de_mutation = 0.5
    de_crossover = 0.9
    de_maxiter = 100

    def de_func(x):
        return de_fitness(x)

    result_de = differential_evolution(de_func, bounds, strategy='current-to-best/2',
                                      mutation=(de_mutation, de_mutation), crossover=de_crossover,
                                      popsize=de_popsize, tol=1e-8, maxiter=de_maxiter,
                                      recombination='blend', mutation_func='binomial', seed=seed)

    best_h = result_de.x
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       tol=1e-9, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    final_h = np.clip(best_h, 0, 1)
    if np.sum(final_h) > 0:
        final_h = final_h * target_sum / np.sum(final_h)
    else:
        final_h = np.random.rand(n_points) * (n_points / 2) / np.sum(final_h)
    return (final_h, best_c5, n_points)
