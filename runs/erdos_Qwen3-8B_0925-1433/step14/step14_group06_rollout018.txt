import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.full(n_points, 0.5)

    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        else:
            x = np.full_like(x, 0.5)
        corr = np.correlate(x, 1 - x, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Simulated Annealing
    t = 1.0
    alpha = 0.995
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.2, n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full_like(new_h, 0.5)
        new_c5 = obj(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / t):
            h = new_h.copy()
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5
        t *= alpha

    # Local refinement
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
