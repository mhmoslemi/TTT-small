import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        h_ = h_ * target_sum / sum_h_ if sum_h_ > 0 else np.zeros_like(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial points
    initial_points = [initial_h_values.copy()]
    initial_points.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))
    sparse_mask = np.random.choice([0, 1], size=n_points, p=[0.9, 0.1])
    initial_points.append(sparse_mask * target_sum / np.sum(sparse_mask))
    block_size = n_points // 5
    blocky = np.zeros(n_points)
    for i in range(5):
        blocky[i*block_size:(i+1)*block_size] = 1
    initial_points.append(blocky * target_sum / np.sum(blocky))

    best_h = initial_points[0].copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Multi-start optimization with diverse methods
    for idx, h0 in enumerate(initial_points):
        try:
            # First refine with SLSQP
            res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
            # Then with COBYLA
            res = minimize(obj, res.x, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    # Final refinement with adaptive perturbations
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.001 * (0.95**_), size=n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
