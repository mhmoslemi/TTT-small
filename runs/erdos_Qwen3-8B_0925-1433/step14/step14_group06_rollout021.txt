import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.full(n_points, 0.5)

    def fitness(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(np.maximum(0, corr[1:-1])) * dx
        return c5

    population_size = 25
    population = []
    for _ in range(15):
        clone = np.copy(h)
        population.append(clone)
    for _ in range(10):
        vec = np.random.rand(n_points)
        vec = np.clip(vec, 0, 1)
        vec = vec * target_sum / np.sum(vec)
        population.append(vec)
    population = np.array(population)

    best_c5 = float('inf')
    best_h = h
    generations = 200
    tournament_size = 5
    crossover_prob = 0.9
    mutation_rate = 0.1
    mutation_sigma = 0.2

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness_values = np.array([fitness(h) for h in population])
        idx_best = np.argmin(fitness_values)
        current_best = population[idx_best]
        current_c5 = fitness_values[idx_best]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best

        parents = []
        for _ in range(population_size // 2):

            winner = tournament[np.argmin(fitness_values[tournament])]
            parents.append(population[winner])
