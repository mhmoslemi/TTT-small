import numpy as np
from scipy.optimize import minimize, least_squares
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = objective(best_h, dx, target_sum, n_points)
    start_time = time.time()

    def objective(x, dx, target_sum, n_points):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Simulated Annealing
    T = 1.0
    alpha = 0.995
    max_iter = 500
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.2, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) == 0:
            new_h = np.zeros(n_points)
        else:
            new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = objective(new_h, dx, target_sum, n_points)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h[:] = new_h
            best_c5 = new_c5
        T *= alpha

    # Trust-region refinement
    result = minimize(objective, best_h, args=(dx, target_sum, n_points), method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return (best_h, best_c5, n_points)
