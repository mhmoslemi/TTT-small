import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    def obj(x):
        return compute_c5(x)

    def generate_population():
        pop = []
        for _ in range(100):
            if np.random.rand() < 0.5:
                h = initial_h_values + np.random.normal(0, 0.05, n_points)
            elif np.random.rand() < 0.3:
                h = np.random.uniform(0, 1, n_points)
            else:
                h = np.zeros(n_points)
                cnt = int(np.random.randint(1, 5))
                idx = np.random.choice(n_points, cnt, replace=False)
                h[idx] = 1
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
            pop.append(h)
        return np.array(pop)

    population = generate_population()
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        c5_values = np.array([compute_c5(h) for h in population])
        elite_idx = np.argsort(c5_values)[:int(100 * 0.2)]
        elites = population[elite_idx]
        offspring = []
        for _ in range(100 - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            idx1, idx2 = np.random.randint(0, n_points), np.random.randint(idx1, n_points)
            child = np.copy(p1)
            child[idx1:idx2] = p2[idx1:idx2]
            perturb = np.random.normal(0, 0.02, n_points)
            child += perturb
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child) if np.sum(child) > 0 else child
            offspring.append(child)
        population = np.vstack((elites, offspring))
        best_idx = np.argmin(c5_values)
        best_h = population[best_idx].copy()
        best_c5 = c5_values[best_idx]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
