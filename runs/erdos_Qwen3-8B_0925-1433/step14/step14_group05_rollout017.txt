import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_block_pattern(n):
        if n % 2 == 0:
            return np.array([1 if i % 2 == 0 else 0 for i in range(n)])
        else:
            b = np.array([1]*(n//2) + [0]*(n - n//2))
            b[-1] = 0.5
            return b

    def generate_random_pattern(n):
        target = n / 2.0
        ones = int(np.round(target))
        zeros = n - ones
        if ones != target:
            vec = np.concatenate([np.ones(ones), np.zeros(zeros)])
            np.random.shuffle(vec)
            idx = np.where(vec == 0)[0][0]
            vec[idx] = 0.5
            return vec
        else:
            vec = np.concatenate([np.ones(ones), np.zeros(zeros)])
            np.random.shuffle(vec)
            return vec

    initial_patterns = [initial_h_values.copy()]
    block = generate_block_pattern(n_points)
    random = generate_random_pattern(n_points)
    initial_patterns.append(block)
    initial_patterns.append(random)

    best_h = None
    best_c5 = 1.0

    start_time = time.time()

    for init in initial_patterns:
        h = np.clip(init, 0, 1).copy()
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.full(n_points, 0.5)
        
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            corr = np.correlate(h_, 1 - h_, mode='full')
            worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
            return worst * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        best_h_current = h.copy()
        best_c5_current = obj(h)

        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.25 * (0.8 ** i)
            new_h = np.clip(best_h_current + np.random.normal(0, perturb_std, n_points), 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.full(n_points, 0.5)
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5_current:
                best_h_current, best_c5_current = res.x, res.fun

        for i in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.05 * (0.95 ** i)
            new_h = np.clip(best_h_current + np.random.normal(0, perturb_std, n_points), 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.full(n_points, 0.5)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5_current:
                best_h_current, best_c5_current = res.x, res.fun

        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.005 * (0.95 ** i)

            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new

            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5_current:
                best_h_current, best_c5_current = res.x, res.fun

        if best_c5_current < best_c5:
            best_h, best_c5 = best_h_current, best_c5_current

    return (best_h, best_c5, n_points)
