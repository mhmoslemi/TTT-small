import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    target_sum_initial = n_points_initial / 2.0

    # Interpolate to higher resolution
    n_points = 240
    dx = 2.0 / n_points
    t = np.linspace(0, 2, n_points_initial)
    h_interpolated = np.interp(np.linspace(0, 2, n_points), t, initial_h_values)
    target_sum = n_points * dx
    h_interpolated = h_interpolated * target_sum / (np.sum(h_interpolated) * dx)

    def obj(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    best_h = h_interpolated.copy()
    best_c5 = float('inf')

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.001, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
