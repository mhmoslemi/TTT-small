import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize population with 20% perturbations of initial and 80% random baseline
    population_size = 100
    population = []
    for _ in range(population_size):

            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.full(n_points, 0.5)
        else:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(300):
        if time.time() - start_time > budget_s:

        c5s = np.array([compute_c5(h) for h in population])
        elite_indices = np.argsort(c5s)[:int(population_size * 0.1)]
        elites = population[elite_indices]

        for _ in range(population_size - len(elites)):
            mean = np.mean(elites, axis=0)
            perturb = np.random.normal(0, step_size, n_points)
            new_h = mean + perturb
            new_h = np.clip(new_h, 0, 1)
