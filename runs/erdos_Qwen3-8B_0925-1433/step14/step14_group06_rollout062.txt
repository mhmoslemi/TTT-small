import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def evaluate(h):
        h_ = np.clip(h, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    population = [initial_h_values.copy() for _ in range(15)]
    for _ in range(100):
        if time.time() - time.time() > budget_s:
            break
        h_candidate = np.random.rand(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        c5 = evaluate(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(evaluate, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
    return (best_h, evaluate(best_h), n_points)
