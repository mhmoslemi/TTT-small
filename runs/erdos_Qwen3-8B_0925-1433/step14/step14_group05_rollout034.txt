from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_h = initial_patterns[0].copy()
    best_c5 = np.inf
    start_time = time.time()
    
    for init in initial_patterns:
        h = np.clip(init, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.full(n_points, 0.5)
        
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            corr = np.correlate(h_, 1 - h_, mode='full')
            worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
            return worst * dx
        
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Random perturbation for COBYLA

                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.full(n_points, 0.5)
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:

                new_h = np.full(n_points, 0.5)
