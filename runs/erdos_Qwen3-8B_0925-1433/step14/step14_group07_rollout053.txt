import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h *= target_sum / np.sum(h) if np.sum(h) > 0 else 0

    def objective(h_vec):
        h_ = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])

        # Optimize for worst lag
        def lag_func(h_vec):
            h_ = np.clip(h_vec, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return corr[1 + worst_idx]

        def jac_func(h_vec):
            h_ = np.clip(h_vec, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            grad = np.zeros_like(h_)
            for i in range(len(h_)):
                if i + worst_idx < len(h_):
                    grad[i] += (1 - h_[i + worst_idx])
                if i > worst_idx:
                    grad[i] -= h_[i - worst_idx]
            return grad

        res = minimize(lag_func, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=jac_func, tol=1e-8)
        if res.success and res.fun < best_c5:
