import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_scale(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h_clipped)
        scale = target_sum / sum_clipped
        return h_clipped * scale

    # Initial population
    population_size = 25
    population = []
    for _ in range(15):
        h = np.copy(initial_h_values)
        h = clip_and_scale(h, target_sum)
        population.append(h)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = clip_and_scale(h, target_sum)
        population.append(h)
    population = np.array(population)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    # GA parameters
    generations = 200
    tournament_size = 5
    crossover_rate = 0.9
    mutation_rate = 0.1

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        # Tournament selection
        indices = np.random.choice(len(population), size=2 * tournament_size, replace=False)
        tournament = population[indices]
        tournament_fitness = np.array([evaluate(h) for h in tournament])
        parents = tournament[np.argmin(tournament_fitness)]

        # Crossover
        offspring = []
        for _ in range(population_size // 2):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                alpha = np.random.rand()

            else:
                child = np.copy(parents[np.random.randint(len(parents))])
            child = clip_and_scale(child, target_sum)
            offspring.append(child)
        # Mutation
