    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) > 0:
            x_scaled = x_clipped * target_sum / np.sum(x_clipped)
        else:
            x_scaled = x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h = initial_h.copy()

    if np.sum(h) > 0:

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()

    # Global exploration with random perturbations and Nelder-Mead
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:

        res = minimize(obj, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:

    # Local refinement with SLSQP
