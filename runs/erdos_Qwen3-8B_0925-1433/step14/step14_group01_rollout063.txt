import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return h

    initial_h = initial_h_values.copy()
    initial_h = clip_normalize(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')[1:-1]) * dx

    population_size = 100
    population = [initial_h.copy() for _ in range(population_size)]
    for _ in range(population_size // 2):
        perturb = np.random.normal(0, 0.05, n_points)
        h = initial_h + perturb
        h = clip_normalize(h)
        population.append(h)
    population = [clip_normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size // 2)]

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    def obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        evals = [evaluate(h) for h in population]
        elites = np.argsort(evals)[:int(population_size * 0.3)]
        new_pop = [population[i] for i in elites]
        while len(new_pop) < population_size:
            a, b = np.random.choice(elites, 2, replace=False)
            cross = np.random.randint(1, n_points - 1)
            child = np.concatenate([population[a][:cross], population[b][cross:]])
            mutate = np.random.normal(0, 0.02, n_points)
            child += mutate
            child = clip_normalize(child)
            new_pop.append(child)
        population = new_pop

        best_idx = np.argmin(evals)
        best = population[best_idx]
        res = minimize(obj, best, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        if res.success:
            best = res.x
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best.copy()
        else:
            best_h = population[best_idx]

    return (best_h, best_c5, n_points)
