import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 45
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return np.inf
        x_normalized = x_clipped * target_sum / sum_x
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = obj(best_h)

    # Phase 1: Differential Evolution
    if time.time() - start_time < budget_s * 0.7:
        result_de = differential_evolution(obj, bounds, strategy='best1bin', 
                                          popsize=10, maxiter=100, mutation=(0.5, 1), 
                                          recombination=0.9, tol=1e-6, 
                                          seed=seed)
        best_h = result_de.x
        best_c5 = result_de.fun

    # Phase 2: Local refinement with SLSQP
    if time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result_slsqp = minimize(obj, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               tol=1e-9, options={'maxiter': 300})
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = result_slsqp.fun

    final_h = np.clip(best_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final > 0:
        final_h = final_h * target_sum / sum_final
    else:
        final_h = best_h
    return (final_h, best_c5, n_points)
