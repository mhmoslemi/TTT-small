import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.round(initial_h_values).astype(int)
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    for _ in range(abs(int(diff))):
        if diff > 0:
            idx = np.random.randint(n_points)
            while h[idx] == 1:
                idx = np.random.randint(n_points)
            h[idx] = 0
        else:
            idx = np.random.randint(n_points)
            while h[idx] == 0:
                idx = np.random.randint(n_points)
            h[idx] = 1

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    T = 100.0
    cooling_rate = 0.998
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        while True:
            i, j = np.random.randint(n_points), np.random.randint(n_points)
            if i != j:
                break
        if h[i] == 0 and h[j] == 1:
            h[i], h[j] = 1, 0
        elif h[i] == 1 and h[j] == 0:
            h[i], h[j] = 0, 1
        else:
            continue
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            delta = current_c5 - best_c5
            if np.random.random() < np.exp(-delta / T):
                best_h = h.copy()
        T *= cooling_rate

    return (best_h, best_c5, n_points)
