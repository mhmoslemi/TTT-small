import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    start_time = time.time()

    for iter in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = corr[1:-1]
        worst_k = np.argmax(overlap)
        grad = np.zeros(n_points)
        for i in range(n_points - worst_k - 1):
            grad[i] += (1 - h[i + worst_k])
            grad[i + worst_k] += -h[i]
        step_size = 1.0 / (1 + iter)
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # Random perturbation search
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_candidate = np.clip(h + perturb, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        current_c5 = compute_c5(h_candidate, dx)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr[1:-1]) * dx
