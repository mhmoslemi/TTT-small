from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def initial_guesses(n=1):
        h = initial_h_values.copy()

        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        guesses = [h]
        for _ in range(n-1):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(new_h)
            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = h.copy()
            guesses.append(new_h)
        return guesses

    guesses = initial_guesses(5)
    best_h = guesses[0].copy()

    def obj(x):

        sum_h_ = np.sum(h_)
        if sum_h_ > 0:

            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
