import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.round(initial_h_values).astype(int)
    sum_h = np.sum(h)
    # Adjust to target_sum
    while sum_h != target_sum:
        if sum_h < target_sum:
            indices_0 = np.where(h == 0)[0]
            if indices_0.size > 0:
                idx = np.random.choice(indices_0)
                h[idx] = 1
                sum_h += 1
        else:
            indices_1 = np.where(h == 1)[0]
            if indices_1.size > 0:
                idx = np.random.choice(indices_1)
                h[idx] = 0
                sum_h -= 1

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    T = 100.0
    cooling_rate = 0.99
    max_iterations = 10000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb h
        current_h = h.copy()
        i = np.random.randint(n_points)
        if current_h[i] == 0:
            j = np.random.choice(np.where(current_h == 1)[0])
            current_h[i] = 1
            current_h[j] = 0
        else:
            j = np.random.choice(np.where(current_h == 0)[0])
            current_h[i] = 0
            current_h[j] = 1
        # Compute current_c5
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        # Accept or reject
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        else:
            delta = current_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = current_h.copy()
                best_c5 = current_c5
        T *= cooling_rate

    return (np.clip(best_h, 0, 1), best_c5, n_points)
