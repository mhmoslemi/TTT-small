from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x

    # Initialize population: clones + random vectors with correct sum
    population = [np.copy(initial_h_values)]
    for _ in range(24):
        vec = np.random.rand(n_points)
        vec = clip_scale(vec)
        population.append(vec)

    # Evaluation function
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    # Tournament selection
    def select_parents(pop):
        idxs = np.random.choice(len(pop), size=5, replace=False)
        candidates = [pop[i] for i in idxs]
        return [min(candidates, key=eval_c5)]

    best_h = clip_scale(initial_h_values)
    best_c5 = eval_c5(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(25):
            parents = select_parents(population)
            if np.random.rand() < 0.8:
                # Blend crossover
                h1, h2 = parents[0], parents[1]
                child = 0.5 * (h1 + h2)
            else:
                # Mutation
                child = np.copy(parents[0])
                mask = np.random.rand(n_points) < 0.2
                child[mask] += np.random.normal(0, 0.2, sum(mask))
            child = clip_scale(child)
            new_pop.append(child)
        population = new_pop
        for candidate in population:
            c = eval_c5(candidate)
            if c < best_c5:
                best_h, best_c5 = candidate, c

    # Final refinement
    res = minimize(
        lambda x: eval_c5(x),
        x0=best_h,
