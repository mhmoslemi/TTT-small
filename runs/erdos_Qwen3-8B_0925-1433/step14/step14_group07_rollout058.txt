import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[n_points - 1:2 * n_points - 1]) * dx
    start_time = time.time()

    for iter in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        corr_lags = corr[n_points - 1:2 * n_points - 1]
        worst_idx = np.argmax(corr_lags)
        k = worst_idx
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            grad[i] += (1 - h[i + k]) * dx**2
            grad[i + k] -= h[i] * dx**2
        step_size = 0.05 * (0.95 ** iter)
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        current_c5 = np.max(corr_lags) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        if iter % 50 == 0:

    return (best_h, best_c5, n_points)
