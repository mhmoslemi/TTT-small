        l = 1.0 / m
        space = 1.0 / (m - 1)
        h = np.zeros(n_points)
        for i in range(m):
            start = i * space
            end = start + l
            for j in range(n_points):
                x_j = j * dx
                if start <= x_j <= end:
                    h[j] = 1.0
        sum_h = np.sum(h) * dx
        scale = 1.0 / sum_h
        return h * scale

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Start with initial_h_values if available
    if hasattr(initial_h_values, '__iter__'):
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        c5_initial = compute_c5(h)
        structured_h = generate_structured_h(9)
        c5_structured = compute_c5(structured_h)
        if c5_initial < c5_structured:
            best_h = h.copy()
            best_c5 = c5_initial
        else:
            best_h = structured_h.copy()
            best_c5 = c5_structured
    else:
        best_h = generate_structured_h(9).copy()
        best_c5 = compute_c5(best_h)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x.copy()
        best_c5 = res.fun

    # Perturb and refine further
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(obj, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
