import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else h_initial

    epsilon = 0.01

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full') * dx
        corr_values = corr[1:-1]
        return (1 / epsilon) * np.log(np.sum(np.exp(epsilon * corr_values)))

    bounds = [(0, 1)] * n_points

    best_h = h_initial
    best_c5 = obj(best_h)
    start_time = time.time()

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-8, 'maxiter': 200})
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    h_final = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_final)
    if sum_h > 0:
        h_final = h_final * target_sum / sum_h
    corr_final = np.correlate(h_final, 1 - h_final, mode='full') * dx
    exact_c5 = np.max(corr_final[1:-1])

    return (best_h, exact_c5, n_points)
