import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def fitness(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    def clip_norm(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x

    # Initial population
    population = [np.copy(initial_h_values) for _ in range(15)]
    for _ in range(10):
        rand = np.random.rand(n_points)
        norm = rand * target_sum / np.sum(rand)
        population.append(norm)

    def tournament_selection(pop, k=5):
        idxs = np.random.choice(len(pop), size=k, replace=False)
        return min(idxs, key=lambda i: fitness(pop[i]))

    best_h = clip_norm(np.copy(initial_h_values))
    best_c5 = fitness(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = [tournament_selection(population) for _ in range(2)]
        p1, p2 = population[parents[0]], population[parents[1]]
        # Blend crossover
        alpha = np.random.rand(n_points)
        child = (1 - alpha) * p1 + alpha * p2
        child = clip_norm(child)
        # Gaussian mutation
        mut = np.random.normal(0, 0.15, n_points)
        child += mut
        child = clip_norm(child)
        population.append(child)
        # Replace worst
        population.pop(np.argmax([fitness(ind) for ind in population]))
        current_best = min(population, key=fitness)
        if fitness(current_best) < best_c5:
            best_h, best_c5 = current_best, fitness(current_best)

    # Local refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(fitness, best_h, method='L-BFGS-B', bounds=[(0,1)*n_points], constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    return (best_h, best_c5, n_points)
