                current_h = res.x

        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.05 * (0.9 ** i)
            new_h = np.clip(current_h + np.random.normal(0, perturb_std, n_points), 0, 1)
            sum_new = np.sum(new_h)
            new_h = new_h * target_sum / sum_new if sum_new > 0 else np.full(n_points, 0.5)
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
                current_h = res.x

        for i in range(50):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.005 * (0.9 ** i)
            new_h = np.clip(current_h + np.random.normal(0, perturb_std, n_points), 0, 1)
            sum_new = np.sum(new_h)
            new_h = new_h * target_sum / sum_new if sum_new > 0 else np.full(n_points, 0.5)
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
                current_h = res.x

    return (best_h, best_c5, n_points)
