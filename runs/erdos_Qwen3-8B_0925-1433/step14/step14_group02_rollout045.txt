import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 1e-10:
            x = x * target_sum / np.sum(x)
        else:
            x = np.zeros_like(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=30,
        mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=100
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time

    if elapsed < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(
            objective, best_h, method='SLSQP',
            bounds=[(0, 1)] * n_points, constraints=cons,
            tol=1e-9, options={'maxiter': 100}
        )
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 1e-10:
        best_h = best_h * target_sum / np.sum(best_h)
    else:
        best_h = np.zeros_like(best_h)
    return (best_h, best_c5, n_points)
