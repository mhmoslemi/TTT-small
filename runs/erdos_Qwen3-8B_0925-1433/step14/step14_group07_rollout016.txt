from scipy.optimize import minimize
import time

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(corr[1:-1])
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for iter in range(150):
        if time.time() - start_time > budget_s:
            break
        # Compute and find worst lag

        worst_idx = np.argmax(corr[1:-1])

                new_h[i] = max(0, new_h[i] - 0.015)
                new_h[i + shift] = min(1, new_h[i + shift] + 0.015)
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                new_c5 = obj(new_h)
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h, new_c5

        # Constrained optimization
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
