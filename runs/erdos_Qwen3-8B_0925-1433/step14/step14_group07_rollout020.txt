from scipy.optimize import minimize
import time

        sum_h = np.sum(h_)
        if sum_h > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    for iter in range(200):
        if time.time() - start_time > budget_s:
            break

            return overlap
