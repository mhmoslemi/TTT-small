            best_h = h.copy()
            best_c5 = new_c5
        
        step_size *= decay_factor
        if iter % 10 == 0:
            perturb = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h)
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
    
    return (best_h, best_c5, n_points)
