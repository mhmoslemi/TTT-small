import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_h = np.copy(initial_h_values)

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    def obj(x):
        h = clip_and_scale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        worst = np.max(corr[1:-1])
        return worst * dx

    def evaluate(h):
        return obj(clip_and_scale(h))

    # Initialize population
    pop_size = 25
    population = [np.copy(initial_h)]
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(clip_and_scale(h))

    best_h = np.copy(initial_h)
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        fit = [(evaluate(h), h) for h in population]
        fit.sort()
        best_fit, best_ind = fit[0]
        if best_fit < best_c5:
            best_c5 = best_fit
            best_h = best_ind

        # Generate new individuals
        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            # Mutation
            if np.random.rand() < 0.1:
                mutate_idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                child[mutate_idx] += np.random.normal(0, 0.2, size=len(mutate_idx))
            new_pop.append(clip_and_scale(child))
        population = new_pop + [best_ind]

    # Local refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (best_h, best_c5, n_points)
