import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial points
    initial_points = [np.array(initial_h_values)]
    initial_points.append(np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)))
    initial_points.append(np.concatenate([np.ones(int(0.1 * n_points)), np.zeros(n_points - int(0.1 * n_points))]))
    initial_points.append(np.kron(np.array([1, 0] * (n_points // 2 + 1)), np.ones(n_points // (n_points // 2 + 1))))
    for i in range(len(initial_points)):
        initial_points[i] = initial_points[i] * target_sum / np.sum(initial_points[i]) if np.sum(initial_points[i]) > 0 else np.zeros(n_points)

    best_h = np.zeros(n_points)
    best_c5 = np.inf
    start_time = time.time()

    # Multi-start optimization
    for h_initial in initial_points:
        h_initial = np.clip(h_initial, 0, 1)
        if np.sum(h_initial) > 0:
            h_initial = h_initial * target_sum / np.sum(h_initial)
        res = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

    # Refinement with adaptive perturbation
    remaining_time = budget_s - (time.time() - start_time)
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.001 * (0.95 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
