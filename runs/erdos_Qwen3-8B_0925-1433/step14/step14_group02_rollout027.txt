import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        n = 45
        h = np.ones(n) * n / 2 / n
        h = np.clip(h, 0, 1)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n

    # Phase 1: Differential Evolution for global exploration
    result = differential_evolution(obj, bounds, strategy='best1bin',
                                   popsize=15, mutation=(0.5, 1),
                                   recombination=0.9, tol=1e-6, maxiter=100,
                                   disp=False)
    best_h, best_c5 = result.x, result.fun

    # Phase 2: SLSQP for local refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                   tol=1e-9, maxiter=100)
    if res.success:
        best_h, best_c5 = res.x, res.fun

    # Final normalization
    h = np.clip(best_h, 0, 1)
    if np.sum(h) > 0:
        h = h * target / np.sum(h)
    return (h, best_c5, n)
