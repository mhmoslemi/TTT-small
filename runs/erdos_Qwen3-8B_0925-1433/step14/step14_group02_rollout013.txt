        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    # Differential Evolution for global search
    result_de = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=30, 
        tol=1e-6, 
        mutation=(0.5, 1), 
        recombination=0.9, 
        maxiter=100, 
        seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        tol=1e-6
    )
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    # Final check and normalization
    final_h = best_h
    final_h = np.clip(final_h, 0, 1)
    final_h_sum = np.sum(final_h)
    if final_h_sum > 0:
        final_h = final_h * target_sum / final_h_sum
    else:
        final_h = best_h

    return (final_h, best_c5, n_points)
