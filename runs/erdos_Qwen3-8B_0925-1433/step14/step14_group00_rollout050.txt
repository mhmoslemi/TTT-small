from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def obj(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def grad(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max_index = np.argmax(corr)
        l = k_max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if i + l >= 0 and i + l < n_points:
                term1 = 1.0 - h_vec[i + l]
            term2 = 0.0
            if i - l >= 0 and i - l < n_points:
                term2 = h_vec[i - l]
            grad[i] = dx * (term1 - term2)
        return -grad

    bounds = [(0, 1)] * n_points

        if res.success and res.fun < best_c5:
