            sum_h = np.sum(new_h)
            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = h.copy()
            guesses.append(new_h)
        return guesses

    guesses = initial_guesses(4)
    best_h = guesses[0].copy()
    best_c5 = float('inf')

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        max_overlap = 0.0
        for m in range(n_points):
            shifted_h = np.roll(h_, m)
            overlap = h_ * (1 - shifted_h)
            valid_overlap = overlap[:n_points - m]
            sum_overlap = np.sum(valid_overlap)
            max_overlap = max(max_overlap, sum_overlap * dx)
        return max_overlap

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for guess in guesses:
        for _ in range(250):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.0005 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(guess + perturb, 0, 1)
            sum_h = np.sum(new_h)
            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = guess.copy()
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.0003 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        else:
            new_h = best_h.copy()
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
