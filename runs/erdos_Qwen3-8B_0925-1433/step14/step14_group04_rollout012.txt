from scipy.optimize import minimize
import time

        else:

        guesses = [h]
        for _ in range(n-1):
            perturb = np.random.normal(0, 0.1, n_points)

            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = h.copy()
            guesses.append(new_h)
        return guesses

    guesses = initial_guesses(5)
    best_h = guesses[0].copy()
    best_c5 = float('inf')

    def obj(x):

        sum_h_ = np.sum(h_)
        if sum_h_ > 0:

            else:

            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
