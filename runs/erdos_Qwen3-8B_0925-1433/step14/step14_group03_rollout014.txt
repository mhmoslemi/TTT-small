import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize
    best_h = np.clip(initial_h_values, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)[1:-1].max()

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        target_k = np.argmax(corr[1:-1])
        target_idx = 1 + target_k  # shift to full array index

        # Objective function for this target
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            corr = np.correlate(h_, 1 - h_, mode='full') * dx
            return corr[target_idx]

        # Constraints: sum(h) = target_sum
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        # Optimize with SLSQP
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full') * dx)[1:-1].max()
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)[1:-1].max()
    return (best_h, final_c5, n_points)
