import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        h = np.full(n_points, 1.0 / n_points)
    else:
        h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[1:-1] * dx)

    start_time = time.time()

    for iter in range(500):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        c_k = corr[1:-1] * dx
        worst_k = np.argmax(c_k)

        grad = np.zeros(n_points)
        k = worst_k
        for j in range(n_points):
            if j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if j - k >= 0:
                grad[j] -= h[j - k] * dx

        eta = 0.01 * (0.95 ** iter)
        delta = -eta * grad

        sum_delta = np.sum(delta)
        if sum_delta != 0:
            lambda_adj = (target_sum - np.sum(h)) / sum_delta
            delta *= lambda_adj

        new_h = h + delta
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_h = np.full(n_points, 1.0 / n_points)
        else:
            new_h = new_h * target_sum / new_sum

        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')[1:-1] * dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            h = new_h.copy()

        if iter % 100 == 0:
            perturb = np.random.normal(0, 0.01, size=n_points)
            new_h_perturb = h + perturb
            new_h_perturb = np.clip(new_h_perturb, 0, 1)
            new_sum_perturb = np.sum(new_h_perturb)
            if new_sum_perturb == 0:
                new_h_perturb = np.full(n_points, 1.0 / n_points)
            else:
                new_h_perturb = new_h_perturb * target_sum / new_sum_perturb
