from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return h

    def evaluate(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    # Initial population
    pop_size = 100
    pop = []
    for _ in range(pop_size):
        if np.random.rand() < 0.2:
            h = initial_h_values.copy()
            h += np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = repair(h)
        pop.append(h)
    pop = np.array(pop)
    best_idx = np.argmin([evaluate(p) for p in pop])
    best_h = pop[best_idx].copy()
    best_c5 = evaluate(best_h)

    # CMA-ES parameters
    sigma = 0.5
    mu = pop_size // 2
    evals = 0

    for gen in range(300):
        if time.time() - time.time() > budget_s:
            break
        evals += pop_size
        if evals > 2000:
            break
        scores = np.array([evaluate(p) for p in pop])
        elite_idx = np.argsort(scores)[:mu]
        elite = pop[elite_idx]
        mu, cov = np.mean(elite, axis=0), np.cov(elite, rowvar=False)
        new_pop = []
        for _ in range(pop_size - mu):
            delta = np.random.multivariate_normal(np.zeros(n_points), cov, 1)
            new_h = best_h + sigma * delta
            new_h = repair(new_h)
            new_pop.append(new_h)
        pop = np.vstack([elite, new_pop])
        scores = np.array([evaluate(p) for p in pop])
        pop = pop[np.argsort(scores)[:pop_size]]
        best_idx = np.argmin(scores)

    # Local optimization
    def obj(x):
        return evaluate(x)
