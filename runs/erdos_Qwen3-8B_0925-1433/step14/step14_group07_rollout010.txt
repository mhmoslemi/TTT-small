    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        k = worst_idx
        grad = compute_gradient(h, k, n_points)
        step_size = 0.05 * (1.0 / (1 + _))
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if _ % 20 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h)

    return (best_h, best_c5, n_points)
