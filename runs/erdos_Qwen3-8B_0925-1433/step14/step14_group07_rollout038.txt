import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time.time()

    for iter in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        worst_k = np.argmax(corr[1:-1])

        grad = np.zeros(n_points)
        for i in range(n_points - worst_k):
            grad[i] += (1 - best_h[i + worst_k]) * dx
            grad[i + worst_k] += -best_h[i] * dx

        def obj(x):
            return -np.dot(grad, x)

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            best_h = res.x

        best_h = best_h * target_sum / np.sum(best_h)

        if iter % 10 == 0:
            perturb_std = 0.02 * (0.95 ** iter)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
