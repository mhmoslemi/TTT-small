import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_values = np.clip(h_values, 0, 1)
    if np.sum(h_values) > 0:
        h_values = h_values * target_sum / np.sum(h_values)
    else:
        h_values = np.zeros(n_points)
    best_h = h_values.copy()
    best_c5 = float('inf')
    time_start = time.time()

    def obj(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for _ in range(200):
        if time.time() - time_start > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h.copy()
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - time_start > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h.copy()
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
