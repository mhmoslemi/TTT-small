import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    def generate_periodic(n):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            seg = int(x / (2.0 / n))
            h[i] = 1 if seg % 2 == 0 else 0
        sum_h = np.sum(h) * dx
        return h * target_sum / sum_h

    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        h = np.where(h >= 0.5, 1.0, 0.0)
        sum_h = np.sum(h) * dx
        return h * target_sum / sum_h

    def generate_modified():
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    patterns = [generate_periodic(5), generate_sinusoidal(), generate_modified()]

    best_h, best_c5 = initial_h_values.copy(), evaluate(initial_h_values)
    start_time = time.time()

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        for p in patterns:

            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
