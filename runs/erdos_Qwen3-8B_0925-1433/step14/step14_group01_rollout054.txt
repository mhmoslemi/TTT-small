import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    # Initial population
    def generate_population(base_h, size=100):
        pop = []
        for _ in range(size):
            if np.random.rand() < 0.5:
                h = base_h.copy()
                h += np.random.normal(0, 0.05, n_points)
            elif np.random.rand() < 0.3:
                h = np.random.uniform(0, 1, n_points)
            else:
                h = np.zeros(n_points)
                h[:int(n_points * 0.1)] = 1
            pop.append(clip_normalize(h))
        return pop

    base_h = clip_normalize(initial_h_values)
    population = generate_population(base_h)

    best_h = base_h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        scores = [compute_c5(h) for h in population]
        elite_idx = np.argmin(scores)
        best_h = population[elite_idx].copy()
        best_c5 = scores[elite_idx]

        # Generate offspring
        new_population = []
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = p1.copy()
            idx = np.random.choice(n_points)
            child[idx] = p2[idx]
            new_population.append(clip_normalize(child))
        for _ in range(len(population) // 2):
            h = population[np.random.randint(len(population))].copy()
            h += np.random.normal(0, 0.02, n_points)
            new_population.append(clip_normalize(h))
        population = new_population[:len(population)]

    # Refinement
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = clip_normalize(best_h + perturb)
        res = minimize(lambda x: compute_c5(x), new_h, method='SLSQP',
                       bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
