import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def interpolate_h(ref_h, to_n):
        t = np.linspace(0, 2, to_n)
        x = np.linspace(0, 2, n_points)
        return np.interp(t, x, ref_h)

    def evaluate_c5(h):
        h_ = np.clip(h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_val = np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
        return max_val

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_val = np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
        return max_val

    def constraint(x):
        h_ = np.clip(x, 0, 1)
        return np.sum(h_) - target_sum

    def perturb(x, std, seed=None):
        if seed is not None:
            np.random.seed(seed)
        return np.clip(x + np.random.normal(0, std, len(x)), 0, 1)

    # Initial interpolation
    refined_n = 60
    h_ref = interpolate_h(initial_h_values, refined_n)
    h_ref = np.clip(h_ref, 0, 1)
    sum_h = np.sum(h_ref)
    h_ref = h_ref * target_sum / sum_h if sum_h > 0 else np.zeros(refined_n)

    best_h = h_ref.copy()
    best_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, evaluate_c5(perturb(best_h, 0.05, seed)))

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        h_pert = perturb(best_h, 0.01, _)
        h_pert = np.clip(h_pert, 0, 1)
        sum_hp = np.sum(h_pert)
        if sum_hp > 0:
            h_pert = h_pert * target_sum / sum_hp
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=[(0,1)]*refined_n, constraints={'type': 'eq', 'fun': constraint}, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Refinement to higher resolution
    refined_n = 240
    h_high = interpolate_h(best_h, refined_n)
    h_high = np.clip(h_high, 0, 1)
    sum_h = np.sum(h_high)
    h_high = h_high * target_sum / sum_h if sum_h > 0 else np.zeros(refined_n)
    best_h = h_high.copy()
    best_c5 = evaluate_c5(best_h)

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        h_pert = perturb(best_h, 0.005, _)
        h_pert = np.clip(h_pert, 0, 1)
        sum_hp = np.sum(h_pert)

            h_pert = h_pert * target_sum / sum_hp

        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
