        h = h * target_sum / sum_h

        h = np.full(n_points, 0.5)
    best_h = h.copy()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(corr[1:-1])
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        sin_pattern = sin_pattern * target_sum / sin_sum

        smooth_sum = np.sum(smooth) * dx
        smooth = smooth * target_sum / smooth_sum
        return [block, sin_pattern, smooth]

    patterns = generate_patterns()

    start_time = time.time()

    for pattern in patterns:

        if time.time() - start_time > budget_s:

        res = differential_evolution(obj, bounds, args=(dx,), strategy='best1bin', popsize=15, tol=1e-5, mutation=(0.5, 1.0), recombination=0.9, polish=True, maxiter=50)
