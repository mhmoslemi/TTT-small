    # Differential Evolution for global exploration
    def de_obj(x):
        return obj(x)

    try:
        result_de = differential_evolution(
            de_obj,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 0.9),
            recombination=0.9,
            seed=seed
        )
        best_h = result_de.x
        best_c5 = obj(best_h)
    except:
        pass

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - target_sum}]
    bounds_slsqp = [(0, 1)] * n_points

    try:
        result_slsqp = minimize(
            obj,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons,
            tol=1e-9,
            options={'maxiter': 300}
        )
        best_h = result_slsqp.x
        best_c5 = obj(best_h)
    except:
        pass

    final_h = np.clip(best_h, 0, 1)
    s = np.sum(final_h)
    if s > 0:
        final_h = final_h * target_sum / s
    else:
        final_h = best_h

    return (final_h, best_c5, n_points)
