import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial_h_values is feasible
    h = np.array(initial_h_values)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time.time()

    # Smooth max approximation parameter
    epsilon = 0.01

    def objective(x):
        h_ = x.copy()
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        corr = corr[1:-1]
        return (1.0 / epsilon) * np.log(np.sum(np.exp(epsilon * corr)))

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Optimization using L-BFGS-B with finite differences
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)

    if res.success:
        best_h = res.x.copy()
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    else:
        best_h = initial_h_values.copy()
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    # Final validation
    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)

    return (best_h, final_c5, n_points)
