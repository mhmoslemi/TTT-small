            if j - k >= 0:
                grad[j] -= h[j - k] * dx
        h -= 0.05 * grad
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)

    return (best_h, best_c5, n_points)
