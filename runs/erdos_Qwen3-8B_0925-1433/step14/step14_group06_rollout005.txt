import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    dx = 2.0 / n_points

        sum_h = np.sum(h)

        return np.zeros_like(h)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    # Initialize from initial_h_values
    h0 = normalize(initial_h_values.copy())

    # Population setup
    pop_size = 25
    pop = [h0.copy()]

        h = np.random.rand(n_points)
        h = normalize(h)
        pop.append(h)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Genetic algorithm loop
    for gen in range(200):
