import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        x_normal = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_normal, 1 - x_normal, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

    # Differential Evolution for global optimization
    try:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=15, tol=1e-6, mutation=(0.5, 1), 
            recombination=0.9, maxiter=100, seed=seed,
            callback=lambda x: None)
    except:
        result_de = None
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    if result_de is not None and time.time() - start_time < budget_s:
        best_h = result_de.x
        best_c5 = result_de.fun

    # SLSQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        result_refine = minimize(
            objective, best_h, method='SLSQP', 
            bounds=bounds, constraints=cons, tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9})
    except:
