import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_clipped = h_clipped * target_sum / sum_h
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr[1:-1]) * dx

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else h_initial
    best_h = h_initial
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        corr_values = corr[1:-1]
        epsilon = 0.01
        return np.log(np.sum(np.exp(epsilon * corr_values))) / epsilon

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    h_final = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_final)
    if sum_h > 0:
        h_final = h_final * target_sum / sum_h
    final_c5 = compute_c5(h_final)
    return (best_h, final_c5, n_points)
