            j = i + lag
            term1 = (1 - h[j]) if 0 <= j < n_points else 0
            j = i - lag
            term2 = (-h[j]) if 0 <= j < n_points else 0
            grad[i] = term1 + term2
        return grad

    def perturb(h):
        noise = np.random.normal(0, 0.1, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5
        best_h = perturb(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
