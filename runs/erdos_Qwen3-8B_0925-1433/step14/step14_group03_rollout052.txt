import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    x0 = initial_h_values.copy()

    if res.success:

        c5_bound = np.max(integrals)
        return (best_h, c5_bound, n_points)
