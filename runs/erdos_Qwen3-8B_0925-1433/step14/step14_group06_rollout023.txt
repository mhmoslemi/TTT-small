import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x > 0:
            x_scaled = x_clipped * target_sum / sum_x
        else:
            x_scaled = np.zeros_like(x_clipped)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, maxiter=150, tol=1e-6, mutation=(0.5, 1),
        recombination=0.8, disp=False
