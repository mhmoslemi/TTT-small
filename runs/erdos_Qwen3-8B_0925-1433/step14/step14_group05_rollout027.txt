            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full(n_points, 0.5)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Fine-tuning with SLSQP
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005 * (0.9 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full(n_points, 0.5)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return best_h, best_c5, n_points
