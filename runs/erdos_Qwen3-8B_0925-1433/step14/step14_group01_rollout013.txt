import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx
    
    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        if np.random.random() < 0.8:
            h = np.random.rand(n_points)
        else:
            h = initial_h_values.copy()
            h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.full(n_points, 0.5)
        population.append(h)
    
    best_h = np.array(initial_h_values)
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate_c5(h) for h in population]
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = scores[best_idx]
        
        elite_count = int(len(population) * 0.1)
        elites = population[:elite_count]
        new_population = elites.copy()
        
        for _ in range(len(population) - elite_count):
            parent = np.random.choice(population)
            mutant = parent + np.random.normal(0, 0.1, n_points)
            mutant = np.clip(mutant, 0, 1)
            sum_mutant = np.sum(mutant)
            if sum_mutant > 0:
                mutant = mutant * target_sum / sum_mutant
            else:
                mutant = np.full(n_points, 0.5)
            new_population.append(mutant)
        
        population = new_population
    
    return (best_h, best_c5, n_points)
