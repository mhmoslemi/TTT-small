import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        return h * target_sum / (s if s > 0 else 1)
    
    # Generate initial population
    population = []
    for _ in range(100):
        if np.random.rand() < 0.5:
            h = initial_h_values.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = clip_normalize(h)
        elif np.random.rand() < 0.8:
            h = np.random.rand(n_points)
            h = clip_normalize(h)
        else:
            h = np.zeros(n_points)
            sparse_mask = np.random.rand(n_points) < 0.1
            h[sparse_mask] = 1.0
            h = clip_normalize(h)
        population.append(h)
    
    best_h = population[0]
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all candidates
        c5_vals = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5_val = np.max(corr[1:-1]) * dx
            c5_vals.append(c5_val)
        # Select elites (top 20%)
        elite_idx = np.argsort(c5_vals)[:int(len(population)*0.2)]
        elites = [population[i] for i in elite_idx]
        # Generate offspring
        offspring = []
        for _ in range(len(population)//2):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            start = np.random.randint(0, n_points)
            end = np.random.randint(start, n_points)
            child = np.copy(p1)
            child[start:end] = p2[start:end]
            child = clip_normalize(child)
            offspring.append(child)
            # Add mutation
            noise = np.random.normal(0, 0.02, n_points)
            mut = np.copy(p1) + noise
            mut = clip_normalize(mut)
            offspring.append(mut)
        # Replace worst half
        population = elites + offspring
        population = sorted(population, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')[1:-1]) * dx)[:len(population)//2]
        # Refine best
        best_idx = np.argmin(c5_vals)
        best_h = population[best_idx]
        best_c5 = c5_vals[best_idx]
    
    # Final refinement with SLSQP
    def obj(x):
        h = clip_normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
    
    return (best_h, best_c5, n_points)
