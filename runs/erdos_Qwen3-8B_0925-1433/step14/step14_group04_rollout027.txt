import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def initial_guesses(n=3):
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        guesses = [h]
        for _ in range(n-1):
            h = np.clip(h + 0.05 * np.random.randn(n_points), 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = guesses[-1].copy()
            guesses.append(h)
        return guesses

    guesses = initial_guesses(5)
    best_h = guesses[0].copy()
    best_c5 = float('inf')

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for guess in guesses:
        for _ in range(150):
            if time.time() - start_time > budget_s:
                break
            perturb = 0.01 * (0.95 ** _)
            new_h = np.clip(guess + perturb * np.random.randn(n_points), 0, 1)
            sum_h = np.sum(new_h)
            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = guess.copy()
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    for _ in range(100):
