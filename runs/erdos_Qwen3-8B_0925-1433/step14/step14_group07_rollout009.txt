    def obj(x):

        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h

        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

                    if new_c5 < best_c5:

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
