import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):

    if 'initial_h_values' in globals():

        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        x_scaled = x_clipped * target_sum / np.sum(x_clipped) if np.sum(x_clipped) > 0 else x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()

    # Global exploration with DE
    def de_objective(x):
        return obj(x)

    result = differential_evolution(
        de_objective,
        bounds,
        constraints=cons,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        maxiter=100,
        disp=False
    )
    best_h = result.x.copy()
    best_c5 = obj(best_h)

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(
        obj,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-9,
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    best_h = res.x.copy()
    best_c5 = obj(best_h)

    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h) if np.sum(final_h) > 0 else best_h
    return (final_h, best_c5, n_points)
