from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_h():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * target_sum / np.sum(h)
        return h

    def generate_sinusoidal_h():
        h = 0.5 + 0.5 * np.sin(np.pi * (np.arange(n_points) / n_points - 0.5))
        h = h * target_sum / np.sum(h)
        return h

    initial_guesses = [initial_h_values, generate_sinusoidal_h(), generate_initial_h()]
    best_h = None
    best_c5 = 1.0

    for guess in initial_guesses:
        h = np.array(guess, dtype=float)
        h = h * target_sum / np.sum(h)
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, h, 
                       method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, 
                       tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    for _ in range(2):
        if time.time() - time.time() > budget_s:
            break
        for i in range(n_points):
            best_h[i] += np.random.normal(0, 0.01)
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, best_h, 
                       method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, 
                       tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
