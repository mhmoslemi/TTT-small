    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h, best_c5 = h.copy(), np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        n = len(x)
        g = np.zeros(n)
        for i in range(n):
            i_plus = i + k_max
            i_minus = i - k_max
            term = 0.0
            if 0 <= i_plus < n:
                term += 1 - x[i_plus]
            if 0 <= i_minus < n:
                term -= x[i_minus]
            g[i] = dx * term
        return g

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x)-target_sum}, tol=1e-6)
        h = res.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        curr_c5 = obj(h)
        if curr_c5 < best_c5:
            best_h, best_c5 = h.copy(), curr_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
