    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_gradient(h_current, k_max, n_points):
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            term1 = 0.0
            if 0 <= i + k_max < n_points:
                term1 = 1.0 - h_current[i + k_max]
            term2 = 0.0
            if 0 <= i - k_max < n_points:
                term2 = -h_current[i - k_max]
            grad[i] = term1 + term2
        return grad

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    max_iterations = 200
    learning_rate = 0.5
    momentum = 0.9
    decay_rate = 0.999
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        grad = compute_gradient(h, k_max, n_points)
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
            biased_corrected_m = m[i] / (1 - momentum ** (_ + 1))
            biased_corrected_v = v[i] / (1 - decay_rate ** (_ + 1))
            h[i] = h[i] + learning_rate * biased_corrected_m / (np.sqrt(biased_corrected_v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
