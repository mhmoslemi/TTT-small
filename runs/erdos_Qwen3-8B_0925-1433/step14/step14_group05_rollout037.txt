        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Phase 2: Local refinement with L-BFGS-B and smaller perturbations
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb_std = max(0.05 * (0.9 ** i), 0.01)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full(n_points, 0.5)
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Final fine-tuning with SLSQP for convergence
    for i in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_std = max(0.01 * (0.95 ** i), 0.005)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full(n_points, 0.5)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
