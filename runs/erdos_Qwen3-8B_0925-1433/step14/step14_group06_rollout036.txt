    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_normalize(h):

        s = np.sum(h_c)

        corr = np.correlate(h_c, 1 - h_c, mode='full')
        max_corr = np.max(corr[1:-1]) * dx

    pop_size = 25
    pop = []
    # Start with clones of initial_h
    for _ in range(15):

    for _ in range(10):

        hrand = clip_normalize(hrand)
        pop.append(hrand)

    start_time = time.time()
    best_h = np.copy(initial_h)
    best_c5 = eval_c5(best_h)

    for gen in range(200):

        fitness = [eval_c5(ind) for ind in pop]
        # Create new population
        new_pop = []
        for _ in range(pop_size):
            # Tournament selection
            t = np.random.choice(pop, size=5, replace=False)

            # Crossover
