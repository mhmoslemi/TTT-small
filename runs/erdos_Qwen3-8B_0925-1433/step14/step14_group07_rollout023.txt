                g[i] -= best_h[i - worst_k] * dx
        step_size = 0.01 * (0.95 ** iter)
        h_new = best_h - step_size * g
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        if sum_h_new > 0:
            h_new = h_new * target_sum / sum_h_new
        else:
            h_new = np.zeros(n_points)
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full') * dx)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        if iter % 10 == 0:
            perturb = np.random.normal(0, 0.02, n_points)
            h_perturb = np.clip(best_h + perturb, 0, 1)
            sum_perturb = np.sum(h_perturb)
            if sum_perturb > 0:
                h_perturb = h_perturb * target_sum / sum_perturb
            else:
                h_perturb = np.zeros(n_points)
            new_c5_perturb = np.max(np.correlate(h_perturb, 1 - h_perturb, mode='full') * dx)
            if new_c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = new_c5_perturb
    return (best_h, best_c5, n_points)
