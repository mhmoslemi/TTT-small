from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) > 0:
            x_scaled = x_clipped * target_sum / np.sum(x_clipped)
        else:
            x_scaled = x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = obj(h)

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:

    max_iter = 200
