import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 45
        initial_h = np.ones(n_points) * (n_points / 2) / n_points
        initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) == 0:
            return np.inf
        x_norm = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, strategy='current-to-best/2',
        mutation=(0.5, 1), crossover=0.9, popsize=30,
        tol=1e-8, maxiter=150, seed=seed,
        polish=False, disp=False
    )
    best_h = result_de.x
    best_c5 = objective(best_h)

    def slsqp_objective(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) == 0:
            return np.inf
        x_norm = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(
        slsqp_objective, best_h,
        method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 100}
    )
    final_h = np.clip(result_slsqp.x, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum != 0:
        final_h = final_h * target_sum / final_sum
    else:
        final_h = result_slsqp.x
    final_c5 = objective(final_h)
    return (final_h, final_c5, n_points)
