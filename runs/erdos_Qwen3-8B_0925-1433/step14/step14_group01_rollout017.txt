import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_points():
        h1 = initial_h_values.copy()
        h2 = np.random.rand(n_points) * 2.0
        h2 = np.clip(h2, 0, 1)
        h2 = h2 * target_sum / np.sum(h2)
        h3 = np.zeros(n_points)
        h3[::10] = 1.0
        h3 = h3 * target_sum / np.sum(h3)
        h4 = np.zeros(n_points)
        h4[::5] = 1.0
        h4 = h4 * target_sum / np.sum(h4)
        return [h1, h2, h3, h4]

    def obj(x):
        h_ = np.clip(x, 0, 1)
        s = np.sum(h_)
        if s > 0:
            h_ = h_ * target_sum / s
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    best_c5 = obj(best_h)
