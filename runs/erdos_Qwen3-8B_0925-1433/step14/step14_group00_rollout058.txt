        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            term1 = (1.0 - h[i + k_max]) * dx if (i + k_max < n_points) else 0
            term2 = h[i - k_max] * dx if (i - k_max >= 0) else 0
            grad[i] = term1 - term2
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(objective, h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints=constraints, tol=1e-6)
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
            perturb = np.random.normal(0, 0.05, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        except:
            pass
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
