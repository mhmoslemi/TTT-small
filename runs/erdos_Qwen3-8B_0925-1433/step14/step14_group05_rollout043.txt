import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial patterns
    h_initial = initial_h_values.copy()
    h_sine = (1 + np.sin(np.pi * np.linspace(0, 2, n_points))) / 2

    candidates = [h_initial, h_sine]

    best_h = h_initial
    best_c5 = objective(best_h)
    start_time = time.time()

    for start in candidates:
        # L-BFGS-B optimization
        res_lbgfs = minimize(objective, start, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res_lbgfs.success:
            res_slsqp = minimize(objective, res_lbgfs.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res_slsqp.success and res_slsqp.fun < best_c5:
                best_h = res_slsqp.x
                best_c5 = res_slsqp.fun

    return (best_h, best_c5, n_points)
