import numpy as np
from scipy.optimize import minimize
import time
from scipy.signal import convolve

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_block_pattern():
        block = np.zeros(n_points)
        num_ones = int(np.round(target_sum))
        block[:num_ones] = 1.0
        return block

    def generate_smoothed_pattern(h_initial):
        kernel = np.hanning(5)
        smoothed = convolve(h_initial, kernel, mode='same') / np.sum(kernel)
        if np.sum(smoothed) > 0:
            smoothed = smoothed * target_sum / np.sum(smoothed)
        else:
            smoothed = np.full(n_points, 0.5)
        return smoothed

    patterns = [np.array(initial_h_values), generate_block_pattern(), generate_smoothed_pattern(initial_h_values)]

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = patterns[0].copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for pattern in patterns:
        h_initial = pattern.copy()
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.full(n_points, 0.5)
        res = minimize(obj, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for i in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else np.full(n_points, 0.5)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
