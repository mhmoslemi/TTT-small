import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 45
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (n_points / 2) / initial_h.sum()
        initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.inf
        h_norm = h * target_sum / np.sum(h)
        corr = np.correlate(h_norm, 1.0 - h_norm, mode='full')
        return np.max(corr[1:-1]) * dx

    pop_size = 30
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_generations = 100
    time_left = budget_s

    population = np.random.rand(pop_size, n_points)
    population = population * (n_points / 2) / population.sum(axis=1)[:, np.newaxis]
    population = np.clip(population, 0, 1)

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > time_left:
            break
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            trial = population[i] * (1 - crossover_prob) + mutant * crossover_prob
            trial_sum = np.sum(trial)
            trial_norm = trial * target_sum / trial_sum if trial_sum else np.zeros_like(trial)
            trial_norm = np.clip(trial_norm, 0, 1)
            new_pop.append(trial_norm)
        population = np.array(new_pop)
        for h in population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        time_left = budget_s - (time.time() - start_time)

    def objective(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        x_norm = x * target_sum / x_sum if x_sum else np.zeros_like(x)
        corr = np.correlate(x_norm, 1.0 - x_norm, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0., 1.) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h) if np.sum(final_h) else best_h
    return (final_h, best_c5, n_points)
