import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h == 0:
            h_ = np.full_like(h_, 0.5)
        else:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    T = 1.0
    alpha = 0.995
    sigma = 0.2
    start_time = time.time()

    # Simulated Annealing
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, sigma, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else np.full_like(new_h, 0.5)
        new_c5 = obj(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            best_c5 = new_c5
        T *= alpha
        sigma *= 0.999

    # Local optimization
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
