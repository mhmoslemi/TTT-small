import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)

    h = np.round(h).astype(float)
    current_sum = np.sum(h)
    while np.abs(current_sum - target_sum) > 1e-6:
        for i in range(n_points):
            if h[i] < 1e-6 and current_sum < target_sum:
                h[i] = 1.0
                current_sum += 1
            elif h[i] > 1 - 1e-6 and current_sum > target_sum:
                h[i] = 0.0
                current_sum -= 1
            if np.abs(current_sum - target_sum) < 1e-6:
                break

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)

            h_minus[i] -= 1e-4

            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
    for _ in range(100):
