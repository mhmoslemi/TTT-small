from scipy.optimize import minimize
from functools import partial
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.clip(initial_h_values, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h

    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr[1:-1])

    best_c5 = compute_correlation(best_h)

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        target_k = np.argmax(corr[1:-1])
        def obj(x):
            corr_val = np.correlate(x, 1 - x, mode='full')
            return corr_val[1:-1][target_k] * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else new_h
            current_c5 = compute_correlation(new_h)
            if current_c5 < best_c5:
                best_h, best_c5 = new_h, current_c5

    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(final_corr[1:-1])
    return (best_h, best_c5, n_points)
