import numpy as np
from scipy.optimize import minimize, least_squares
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    else:

    temp = 1.0
    cooling_rate = 0.995
    sa_iterations = 500
    perturbation_sigma = 0.2
    start_time = time.time()
    
    for _ in range(sa_iterations):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, perturbation_sigma, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:

        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        temp *= cooling_rate
    
    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

def compute_c5(h_vec, dx, n_points, dx_val, target_sum):
    h = np.clip(h_vec, 0, 1)
    if np.sum(h) > 0:

    corr = np.correlate(h, 1 - h, mode='full')
    worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
    return worst * dx_val
