            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        elif np.random.rand() < 0.3:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)
            num_high = int(0.1 * n_points)
            indices = np.random.choice(n_points, num_high, replace=False)
            h[indices] = 1.0
            h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
