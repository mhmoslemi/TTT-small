    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) > 0:
            x_scaled = x_clipped * target_sum / np.sum(x_clipped)
        else:
            x_scaled = x_clipped
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points

    def de_obj(x):
        return obj(x)

    # Differential Evolution Phase
    result_de = differential_evolution(
        de_obj,
        bounds,
        strategy='best1bin',
        popsize=10,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=100,
        disp=False,
        polish=True,
        seed=seed
    )

    best_h = result_de.x
    best_c5 = obj(best_h)

    # Local Refinement Phase
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_local = minimize(

        method='L-BFGS-B',
        bounds=bounds,
