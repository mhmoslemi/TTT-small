import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.full_like(x, 0.5)
        corr = np.correlate(h, 1 - h, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, mutation=(0.5, 1.0), recombination=0.9,
        tol=1e-6, maxiter=500, seed=seed
    )
    best_h = result.x
    best_c5 = result.fun

    if time.time() - start_time < budget_s:
        res_lbgfs = minimize(
            objective, best_h, method='L-BFGS-B', bounds=bounds,
            tol=1e-6, options={'maxiter': 300}
        )
        best_h = res_lbgfs.x
        best_c5 = res_lbgfs.fun

    return (best_h, best_c5, n_points)
