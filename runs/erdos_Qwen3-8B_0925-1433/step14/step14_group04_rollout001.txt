import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def initial_guess(h0):
        h = np.clip(h0, 0, 1).copy()
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    h = initial_guess(initial_h_values)
    best_h = h.copy()
    best_c5 = float('inf')

    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr[1:-1])
        return max_corr * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for i in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05 * (0.95 ** i), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        try:
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005 * (0.95 ** i), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        try:
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass
