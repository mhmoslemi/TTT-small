        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        res2 = minimize(
            objective, h, method='L-BFGS-B', 
            bounds=[(0, 1)] * n_points, 
            constraints={'type': 'eq', 'fun': constraint}, 
            tol=1e-6, options={'maxiter': 100}
        )
        if res2.success and res2.fun < best_c5:
            best_h, best_c5 = res2.x, res2.fun
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
