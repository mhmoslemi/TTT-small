import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[1:-1]) * dx
    start_time = time.time()
    
    max_iter = 1000
    learning_rate = 0.2
    decay_factor = 0.99
    perturb_interval = 100
    perturb_std = 0.05
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        worst_k = worst_idx
        
        gradient = np.zeros(n_points)
        limit = n_points - worst_k
        if limit > 0:
            i_vals = np.arange(limit)
            j_vals = i_vals + worst_k
            gradient[i_vals] += (1 - h[j_vals]) * dx**2
            gradient[j_vals] -= h[i_vals] * dx**2
        
        h = np.clip(h - learning_rate * gradient, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new[1:-1]) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        learning_rate *= decay_factor
        
        if iter % perturb_interval == 0:
            perturb = np.random.normal(0, perturb_std, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points)
    
    return (best_h, best_c5, n_points)
