        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

        shift = worst_idx

        res = minimize(subobj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
