def run(seed=42, budget_s=600):
    np.random.seed(seed)

    h = np.clip(initial_h_values, 0, 1)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial best solution
    best_h = h
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Optimization with L-BFGS-B
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
