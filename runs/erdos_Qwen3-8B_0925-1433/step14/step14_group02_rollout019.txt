import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

    else:

        return np.max(corr[1:-1]) * dx

    start_time = time.time()

    elapsed = time.time() - start_time

    if elapsed > budget_s:
