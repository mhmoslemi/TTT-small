    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    def objective(x):
        return compute_c5(x)
    
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    final_h = res.x
    final_h = np.clip(final_h, 0, 1)
    if np.sum(final_h) == 0:
        final_h = initial_h.copy()
    else:
        final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    return (final_h, final_c5, n_points)
