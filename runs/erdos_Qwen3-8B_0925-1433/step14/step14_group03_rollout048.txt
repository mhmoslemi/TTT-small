import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(compute_c5, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(compute_c5, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5

    for _ in range(100):
