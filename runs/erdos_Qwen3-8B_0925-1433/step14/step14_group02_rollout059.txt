import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 45
        initial_h = np.ones(n_points) * (n_points / 2) / n_points
        initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x_clamped = np.clip(x, 0, 1)
        if np.sum(x_clamped) == 0:
            return float('inf')
        x_normalized = x_clamped * target_sum / np.sum(x_clamped)
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    # Phase 1: Differential Evolution for global search
    try:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=15, mutation=(0.5, 1), recombination=0.9,
            tol=1e-6, maxiter=100, seed=seed, polish=False)
        if result_de.success:
            best_h = result_de.x
            best_c5 = result_de.fun
    except:
        pass

    # Phase 2: Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       tol=1e-6, options={'maxiter': 150, 'ftol': 1e-9})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
    except:
        pass

    # Final normalization
    final_h = np.clip(best_h, 0, 1)
    if np.sum(final_h) > 0:
        final_h = final_h * target_sum / np.sum(final_h)
    else:
        final_h = best_h

    # Check time budget and return best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)
    return (final_h, best_c5, n_points)
