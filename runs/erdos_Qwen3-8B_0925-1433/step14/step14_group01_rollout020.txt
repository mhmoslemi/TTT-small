import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_h(h):
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    # Initialize population
    pop_size = 100
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.2:
            perturb = np.random.normal(0, 0.1, n_points)
            h = initial_h_values + perturb
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = initial_h_values.copy()
    best_c5 = evaluate_h(best_h)
    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        c5_values = [evaluate_h(h) for h in population]
        elite_idx = np.argsort(c5_values)[:int(pop_size * 0.1)]
        elites = [population[i] for i in elite_idx]
        mean = np.mean(elites, axis=0)
        cov = np.cov(elites, rowvar=False)
        new_population = []
        for _ in range(pop_size - len(elites)):
            sample = np.random.multivariate_normal(mean, cov, 1)[0]
            sample = np.clip(sample, 0, 1)
            sample = sample * target_sum / np.sum(sample)
            new_population.append(sample)
        population = elites + new_population
        current_best = population[np.argmin(c5_values)]
        if evaluate_h(current_best) < best_c5:
            best_h, best_c5 = current_best, evaluate_h(current_best)

    # Final SLSQP optimization
    def obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    return (best_h, best_c5, n_points)
