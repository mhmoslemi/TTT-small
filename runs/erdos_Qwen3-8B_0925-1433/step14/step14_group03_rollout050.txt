import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.clip(initial_h_values, 0, 1).copy()
    if np.sum(initial_h) > 0:
        initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_exact_c5(best_h, dx)
    start_time = time.time()

    def compute_exact_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1]) * dx

    epsilon = 0.01

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        smooth_max = (1 / epsilon) * np.log(np.sum(np.exp(epsilon * corr[1:-1])))
        return smooth_max

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
    if result.success:
        current_h = result.x
        current_c5 = compute_exact_c5(current_h, dx)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    # Final check
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = compute_exact_c5(best_h, dx)

    return (best_h, final_c5, n_points)
