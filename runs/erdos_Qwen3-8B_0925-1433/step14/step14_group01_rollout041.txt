import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Generate diverse initial points
    def gen_sparse():
        sparse = np.zeros(n_points)
        idx = np.random.choice(n_points, size=int(n_points*0.1), replace=False)
        sparse[idx] = 1.0
        return sparse * target_sum / np.sum(sparse)
    
    def gen_blocky():
        blocky = np.zeros(n_points)
        for i in range(0, n_points, 5):
            blocky[i:i+4] = 1.0
        return blocky * target_sum / np.sum(blocky)
    
    initial_candidates = [
        initial_h_values,
        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)),
        gen_sparse(),
        gen_blocky()
    ]

    best_h = initial_candidates[0].copy()
    best_c5 = float('inf')

    start_time = time.time()
    for h in initial_candidates:
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped) if np.sum(h_clipped) > 0 else h_clipped
        res = minimize(lambda x: np.max(np.maximum(0, np.correlate(x, 1 - x, mode='full')[1:-1] - 1e-5)) * dx,
                       h_scaled, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type':'eq', 'fun':lambda x: np.sum(x) - target_sum}],
                       tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        if time.time() - start_time > budget_s:
            break

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.001 * (0.95 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(lambda x: np.max(np.maximum(0, np.correlate(x, 1 - x, mode='full')[1:-1] - 1e-5)) * dx,
                       new_h, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type':'eq', 'fun':lambda x: np.sum(x) - target_sum}],
                       tol=1e-9)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
