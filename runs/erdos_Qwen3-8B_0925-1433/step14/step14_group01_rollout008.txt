import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial points
    def gen_sparse(n):
        s = np.zeros(n)
        m = int(0.1 * n)
        idx = np.random.choice(n, m, replace=False)
        s[idx] = 1
        return s * target_sum / np.sum(s)

    def gen_blocky(n):
        b = np.zeros(n)
        sz = int(n / 4)
        for i in range(0, n, 2 * sz):
            b[i:i+sz] = 1
            b[i+sz:i+2*sz] = 0
        return b * target_sum / np.sum(b)

    initial_points = [initial_h_values, np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)), gen_sparse(n_points), gen_blocky(n_points)]

    best_h = initial_points[0].copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for i, h in enumerate(initial_points):
        h = h.copy()
        for _ in range(80):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.normal(0, 0.002 * (0.98 ** _), n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    for _ in range(120):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.0005 * (0.98 ** _), n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h

        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
