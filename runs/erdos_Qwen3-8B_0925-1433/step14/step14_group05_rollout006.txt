import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from scipy.ndimage import gaussian_filter

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_periodic(n_points, m=5):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            seg = int(x[i] * m / 2)
            if seg % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        sum_h = np.sum(h)
        return h * (target_sum / sum_h) if sum_h else np.full(n_points, 0.5)

    def create_sine(n_points):
        x = np.linspace(0, 2, n_points)
        h = (1 + np.cos(np.pi * x)) / 2
        return h

    def create_modified_parent(n_points):
        h = initial_h_values.copy()
        h = gaussian_filter(h, sigma=0.2)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        return h * (target_sum / sum_h) if sum_h else np.full(n_points, 0.5)

    patterns = [create_periodic(n_points, 5), create_sine(n_points), create_modified_parent(n_points)]

    best_h = np.full(n_points, 0.5)
    best_c5 = float('inf')
    start_time = time.time()

    for pattern in patterns:
        h = pattern.copy()
        def obj(x):
            h_ = np.clip(x, 0, 1)
            corr = np.correlate(h_, 1 - h_, mode='full')
            worst = np.max(corr[1:-1])
            return worst * dx

        def cons(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1)] * n_points
        result = differential_evolution(obj, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.5, seed=seed)
        if result.success:
            h = result.x
            c5 = obj(h)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

        result_refine = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-6)
        if result_refine.success:
            c5 = obj(result_refine.x)
            if c5 < best_c5:
                best_h, best_c5 = result_refine.x, c5

        result_slsqp = minimize(obj, result_refine.x, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8)
        if result_slsqp.success:
            c5 = obj(result_slsqp.x)
            if c5 < best_c5:
                best_h, best_c5 = result_slsqp.x, c5
