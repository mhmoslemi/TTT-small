import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        h = h * target_sum / sum_h

        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    # Nelder-Mead for global search
    for i in range(150):
        if time.time() - start_time > budget_s:

        else:

        res = minimize(obj, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
