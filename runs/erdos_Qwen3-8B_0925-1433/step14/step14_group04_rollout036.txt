            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = guess.copy()
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    for _ in range(80):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.0005 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        else:
            new_h = best_h.copy()
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    return (best_h, best_c5, n_points)
