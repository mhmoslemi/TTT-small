from scipy.optimize import differential_evolution, minimize
import time

    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.full(n_points, 0.5)

        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1]))
        return worst * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def block_pattern(m):

        x = np.linspace(0, 2, n_points)

        sum_wave = np.sum(wave)
        return wave * target_sum / sum_wave

        return np.clip(np.convolve(initial_h_values, np.ones(5), mode='same') / 5, 0, 1) * target_sum / np.sum(np.clip(np.convolve(initial_h_values, np.ones(5), mode='same') / 5, 0, 1))
