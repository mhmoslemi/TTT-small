import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for iter in range(250):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        shift = worst_idx

        overlap_start = 0
        overlap_end = n_points - shift - 1
        if overlap_end >= overlap_start:
            max_contrib = -np.inf
            contrib_i = -1
            for i in range(overlap_start, overlap_end + 1):
                contrib = best_h[i] * (1 - best_h[i + shift]) * dx
                if contrib > max_contrib:
                    max_contrib = contrib
                    contrib_i = i
            if contrib_i != -1:
                delta = 0.015 * (1 - iter / 250)
                new_h = best_h.copy()
                new_h[contrib_i] = max(0, new_h[contrib_i] - delta)
                new_h[contrib_i + shift] = min(1, new_h[contrib_i + shift] + delta)
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                new_c5 = obj(new_h)
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h, new_c5

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:

        if iter % 5 == 0:

            if res.success and res.fun < best_c5:
