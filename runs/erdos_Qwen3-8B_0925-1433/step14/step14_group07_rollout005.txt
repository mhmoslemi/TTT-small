from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(corr[1:-1])
        return worst * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        shift = worst_idx
        
        # Optimization for worst lag
        def sub_obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            return np.sum(h_[np.arange(n_points - shift)] * (1 - h_[np.arange(shift, n_points)])) * dx
        
        res = minimize(sub_obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
