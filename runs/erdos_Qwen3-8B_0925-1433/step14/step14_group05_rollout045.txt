import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.full(n_points, 0.5)
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    # Multi-phase optimization
    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'L-BFGS-B'
            tol = 1e-6
        elif phase == 1:
            method = 'SLSQP'
            tol = 1e-8
        else:
            method = 'Nelder-Mead'
            tol = 1e-6
        
        # Adaptive perturbation
        perturb = np.random.normal(0, 0.1 if phase == 0 else 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.full(n_points, 0.5)
        
        res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=tol)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
