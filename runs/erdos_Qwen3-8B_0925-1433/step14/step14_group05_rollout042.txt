    block_h[:n_points//2] = 1.0
    sum_block = np.sum(block_h) * dx
    scale_block = target_sum / sum_block
    patterns.append(block_h * scale_block)
    # Smoothed initial
    smooth_h = gaussian_filter(initial_h_values, sigma=2)
    sum_smooth = np.sum(smooth_h) * dx
    scale_smooth = target_sum / sum_smooth
    patterns.append(smooth_h * scale_smooth)
    # Alternating pattern
    if n_points % 2 == 0:
        alt_h = np.zeros(n_points)
        alt_h[::2] = 1.0
        sum_alt = np.sum(alt_h) * dx
        scale_alt = target_sum / sum_alt
        patterns.append(alt_h * scale_alt)
    
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimize patterns
    for p in patterns:
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_) * dx
            if sum_h_ > 0:
                h_ = h_ * (target_sum / sum_h_)
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(np.maximum(0, corr[1:-1])) * dx
        
        res = minimize(obj, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()
    
    # Local refinements
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_new = np.sum(new_h) * dx
        if sum_new > 0:
            new_h = new_h * (target_sum / sum_new)
        res = minimize(compute_c5, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()
    
    return (best_h, best_c5, n_points)
