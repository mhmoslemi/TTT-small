    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = 0.0
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr[1:-1]) * dx

    T = 1.0
    cooling_rate = 0.995
    sa_iter = 500
    perturb_sigma = 0.2
    start_time = time.time()

    for i in range(sa_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, perturb_sigma, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = np.zeros(n_points)
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr_new[1:-1]) * dx
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h.copy()
            best_c5 = new_c5
        T *= cooling_rate

    def objective(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr[1:-1])

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr[1:-1]) * dx

    return (best_h, best_c5, n_points)
