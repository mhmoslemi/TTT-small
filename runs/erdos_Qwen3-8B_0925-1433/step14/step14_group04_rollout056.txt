import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy()
    
    # Smooth initial guess with a low-pass filter

    # Normalize sum to meet the constraint
    smoothed = smoothed * target_sum / np.sum(smoothed)
    best_h = smoothed.copy()
    best_c5 = float('inf')
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_corr = np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx
        return max_corr
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()
    
    # Initial optimization phase with larger perturbations
    for step in range(200):
        if time.time() - start_time > budget_s:

        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        else:
            new_h = best_h.copy()
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
