            h = np.full(n, 0.5)
        return h
    
    patterns = [
        initial_h_values.copy(),
        generate_block_pattern(n_points),
        generate_random_block(n_points)
    ]
    
    for pattern in patterns:
        h_opt, c5_opt = optimize_pattern(pattern)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    return (best_h, best_c5, n_points)
