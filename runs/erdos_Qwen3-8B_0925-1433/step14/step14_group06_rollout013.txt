import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        if np.isclose(np.sum(h), 0):
            return np.full_like(h, 0.5)
        scale = target_sum / np.sum(h)
        return h * scale

    def objective(x):
        h = clip_and_scale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        worst = np.max(corr[1:-1])
        return worst * dx

    initial_h = clip_and_scale(initial_h_values)
    best_h = np.copy(initial_h)
    best_c5 = objective(best_h)
    best_h = clip_and_scale(best_h)

    pop_size = 25
    pop = [np.copy(initial_h) for _ in range(pop_size)]
    for _ in range(pop_size // 2):
        new_h = np.random.rand(n_points)
        new_h = clip_and_scale(new_h)
        pop.append(new_h)

    def tournament_selection(pop, k=5):
        indices = np.random.choice(len(pop), k, replace=False)
        return min(indices, key=lambda i: objective(pop[i]))

    def blend_crossover(parent1, parent2):
        alpha = np.random.uniform(-1, 1)
        child = (1 - alpha) * parent1 + alpha * parent2
        return clip_and_scale(child)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = [tournament_selection(pop) for _ in range(2)]
        child = blend_crossover(pop[parents[0]], pop[parents[1]])
        child = clip_and_scale(child)
        pop.append(child)
        pop.pop(np.random.randint(len(pop)))
        for i in range(len(pop)):
            if np.random.rand() < 0.1:
                perturb = np.random.normal(0, 0.2, n_points)
                temp = pop[i] + perturb
                temp = clip_and_scale(temp)
                if objective(temp) < objective(pop[i]):
                    pop[i] = temp
        current_best = min(pop, key=objective)
        if objective(current_best) < best_c5:
            best_h, best_c5 = current_best, objective(current_best)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = best_h + perturb
        new_h = clip_and_scale(new_h)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=[(0,1)*n_points], constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:

    return (clip_and_scale(best_h), best_c5, n_points)
