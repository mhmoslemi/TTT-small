import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h) if np.sum(h) > 0 else 1.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1] * dx)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(500):

            break

        corr_values = corr[1:-1] * dx
        worst_k = np.argmax(corr_values)
        worst_value = corr_values[worst_k]

        def objective(x):
            h_ = x
            corr = np.correlate(h_, 1 - h_, mode='full')
            return corr[1:-1][worst_k] * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)

        if res.success:

            if current_c5 < best_c5:
