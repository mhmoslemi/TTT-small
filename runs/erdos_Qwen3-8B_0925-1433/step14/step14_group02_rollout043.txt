import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.ones(n_points) * (n_points / 2) / n_points

        if sum_h == 0:

        h_norm = x * target_sum / sum_h
        h_clipped = np.clip(h_norm, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.inf
        h_final = h_clipped * target_sum / sum_clipped
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        c5 = np.max(corr[1:-1]) * dx
        return c5

    bounds = [(0, 1)] * n_points

    start = time.time()
    result = differential_evolution(obj, bounds, strategy='best1bin', popsize=15, tol=1e-6, maxiter=100, mutation=(0.5, 1), recombination=0.9, seed=seed)

        return obj(x)
