import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(corr_vals[1:-1])

    start_time = time.time()

    for i in range(500):
        if time.time() - start_time > budget_s:
            break
        corr_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
        target_k = np.argmax(corr_vals[1:-1])
        perturb_std = 0.05 * (0.95 ** i)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        else:
            new_h = best_h.copy()
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[1:-1][target_k] * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_corr = np.correlate(res.x, 1 - res.x, mode='full') * dx
            new_max = np.max(curr_corr[1:-1])
            if new_max < best_c5:
                best_h = res.x.copy()
                best_c5 = new_max

    h = np.clip(best_h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    corr_vals = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(corr_vals[1:-1])
    return (h, final_c5, n_points)
