def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    pop_size = 100
    population = []
    for _ in range(pop_size):

            h = initial_h_values.copy()

        population.append(h)

        return np.max(np.correlate(h, 1 - h, mode='full')[1:-1]) * dx
