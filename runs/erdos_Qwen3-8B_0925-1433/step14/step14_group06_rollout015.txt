from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_h = np.copy(initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h) if np.sum(initial_h) > 0 else np.zeros(n_points)
    best_h = initial_h
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')[1:-1]) * dx

    def clip_and_scale(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            return x * target_sum / s
        return np.full_like(x, 0.5)

    def obj(x):
        x = clip_and_scale(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr[1:-1]) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for ind in population:

    population = [clip_and_scale(x) for x in population]
    best_h = population[0]
    best_c5 = obj(best_h)

        if time.time() - time.time() > budget_s:
            break
