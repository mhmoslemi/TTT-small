    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def process_candidate(candidate):
        clipped = np.clip(candidate, 0, 1)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:

        scaled = clipped * target_sum / sum_clipped
        return np.clip(scaled, 0, 1)
    
    def obj(x):
        x_proc = process_candidate(x)
        corr = np.correlate(x_proc, 1 - x_proc, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    initial_population = np.array([process_candidate(np.random.rand(n_points)) for _ in range(30)])
    
    def de_objective(x):
        return obj(x)
    
    result_de = differential_evolution(de_objective, bounds, strategy='current-to-best/2', 
                                      mutation=(0.5, 1), crossover=0.9, popsize=30, 
                                      tol=1e-6, mutation_size=0.5, recombination=0.5, 
                                      maxiter=100, disp=False, polish=True)
    
    best_h = result_de.x
    best_c5 = obj(best_h)
    
    start_time = time.time()
    
    def slsqp_objective(x):
        return obj(x)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(slsqp_objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
