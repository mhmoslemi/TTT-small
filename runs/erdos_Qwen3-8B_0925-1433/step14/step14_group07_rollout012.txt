    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Compute all overlaps to find worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        worst_shift = worst_idx
        
        # Build objective function for worst lag
        def objective(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            # Compute overlap for worst lag
            if worst_shift + 1 > n_points:
                overlap = 0
            else:
                start = 0

                if end >= start:
                    overlap = np.sum(h_[start:end+1] * (1 - h_[start + worst_shift:end + worst_shift + 1])) * dx
                else:
                    overlap = 0
            return overlap
        
        # Bounds and constraints
        bounds = [(0, 1)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Solve subproblem using trust-constr
        res = minimize(objective, best_h, method='trust-constr', bounds=bounds, constraints=constraints, tol=1e-7)
        if res.success:
