import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if sum_h > 0:
        h = h * target_sum / sum_h

        sum_x = np.sum(x_clipped)
        if sum_x == 0:

        else:
            x_scaled = x_clipped * target_sum / sum_x

    start_time = time.time()
    
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1),
        recombination=0.9, maxiter=500, tol=1e-6, disp=False

    elapsed = time.time() - start_time
    if elapsed > budget_s:
