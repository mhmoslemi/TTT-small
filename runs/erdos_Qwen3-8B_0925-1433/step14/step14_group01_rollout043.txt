import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        s = np.sum(h)
        if s <= 0:
            return np.full_like(h, 0.5)
        return h * target_sum / s

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    # Initialize population
    pop_size = 100
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.8:
            h = np.random.rand(n_points)
            h = normalize(h)
        else:
            h = initial_h_values.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize(h)
        population.append(h)

    best_h = population[0]
    best_c5 = eval_c5(best_h)

    def tournament_selection(pop, k=5):
        candidates = np.random.choice(pop, size=k)
        return min(candidates, key=eval_c5)

    for gen in range(100):
        if time.time() - time.time() > budget_s:
            break
        # Evaluate all
        c5s = np.array([eval_c5(h) for h in population])
        current_best = population[np.argmin(c5s)]
        if eval_c5(current_best) < best_c5:
            best_h, best_c5 = current_best, eval_c5(current_best)

        # Select parents
        parents = [tournament_selection(population) for _ in range(pop_size // 2)]

        # Crossover
        offspring = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            if np.random.rand() < 0.5:
                # Single-point
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                # Uniform
                child = np.random.rand(n_points)
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            offspring.append(normalize(child))

        # Mutate
        for i in range(len(offspring)):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05 * (0.95 ** gen), n_points)
                offspring[i] = normalize(np.clip(offspring[i] + noise, 0, 1))

        # Replace population
        population = offspring + population[:pop_size - len(offspring)]

    # Local refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    def obj(x):
        return eval_c5(x)

    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x

    return (best_h, best_c5, n_points)
