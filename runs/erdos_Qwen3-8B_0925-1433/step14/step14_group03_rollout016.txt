import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Initial Differential Evolution
    def de_func(x):
        return obj(x)
    res = differential_evolution(de_func, bounds, constraints=cons, strategy='best1bin', 
                                popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, 
                                seed=seed, maxiter=100)
    if res.success:
        best_h = res.x
        best_c5 = obj(best_h)

    # Refinement with SLSQP
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, 
                       tol=1e-6, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Final refinement
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                       tol=1e-6, options={'maxiter': 20})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
