import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x, worst_lag):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = corr[worst_lag + 1]
        return worst * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = None
    start_time = time.time()

    for iter in range(400):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        worst_lag = worst_idx
        worst_val = corr[worst_idx + 1]

        if best_c5 is None or worst_val * dx < best_c5:
            best_c5 = worst_val * dx

        # Perturb the worst lag's overlap region
        if worst_lag < n_points - 1:
            overlap_start = 0
            overlap_end = n_points - worst_lag - 1
            if overlap_start <= overlap_end:
                for i in range(overlap_start, overlap_end + 1):
                    new_h = best_h.copy()
                    new_h[i] = max(0, new_h[i] - 0.015)
                    new_h[i + worst_lag] = min(1, new_h[i + worst_lag] + 0.015)
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * target_sum / np.sum(new_h)
                    new_c5 = obj(new_h, worst_lag)
                    if new_c5 < best_c5:
                        best_h, best_c5 = new_h, new_c5

        res = minimize(lambda x: obj(x, worst_lag), best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        if iter % 10 == 0:
            perturb_std = 0.03 * (0.95 ** iter)
