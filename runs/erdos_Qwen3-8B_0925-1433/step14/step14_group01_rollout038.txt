from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h = np.clip(h + noise, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        candidates.append(h)

    # Select best initial candidate
    best_h = min(candidates, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.002 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
