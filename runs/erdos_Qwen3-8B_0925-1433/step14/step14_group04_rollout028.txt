        guesses = [h]

            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = h.copy()
            guesses.append(new_h)

    def obj(x):

        if sum_h_ > 0:

            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
