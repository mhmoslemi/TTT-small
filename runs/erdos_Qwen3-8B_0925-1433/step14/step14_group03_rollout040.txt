import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        epsilon = 5e-4
        soft_max = np.log(np.sum(np.exp(-epsilon * corr[1:-1]))) / epsilon
        return soft_max * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = initial_h_values.copy()
    best_c5 = 0.0
    start_time = time.time()

    def evaluate_c5(h):
        h_ = np.clip(h, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr[1:-1]) * dx

    # Initial optimization
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
    if res.success:
        best_h = res.x
        best_c5 = evaluate_c5(best_h)
    else:
        best_c5 = evaluate_c5(initial_h_values)

    # Refinement with perturbations
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.01 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    return (best_h, best_c5, n_points)
