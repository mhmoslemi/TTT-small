import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize population
    pop_size = 50
    population = [initial_h_values.copy() * np.random.normal(0.95, 0.05, n_points) for _ in range(pop_size)]

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1] * dx)

    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
    time_start = time.time()

    for _ in range(200):
        if time.time() - time_start > budget_s:
            break
        scores = [evaluate(h) for h in population]
        elite_idx = np.argsort(scores)[:int(pop_size * 0.2)]
        elites = [population[i] for i in elite_idx]

        offspring = []
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (p1 + p2) / 2
            noise = np.random.normal(0, 0.01, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)

        population = elites + offspring
        for h in population:
            if evaluate(h) < best_c5:
                best_h, best_c5 = h.copy(), evaluate(h)

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[1:-1] * dx)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
