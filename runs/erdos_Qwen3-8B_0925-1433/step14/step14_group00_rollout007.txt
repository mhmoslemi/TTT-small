            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0
            current_sum -= 1
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h.astype(float), 1 - best_h.astype(float), mode='full')) * dx
    
    temperature = 100.0
    cooling_rate = 0.99
    max_iterations = 5000
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        current_h = best_h.copy()
        i = np.random.randint(n_points)
        if current_h[i] == 1:
            current_h[i] = 0
            j = np.random.choice(np.where(current_h == 0)[0])
            current_h[j] = 1
        else:
            current_h[i] = 1
            j = np.random.choice(np.where(current_h == 1)[0])
            current_h[j] = 0
        
        new_c5 = np.max(np.correlate(current_h.astype(float), 1 - current_h.astype(float), mode='full')) * dx
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            best_h = current_h.copy()
            best_c5 = new_c5
        
        temperature *= cooling_rate
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
