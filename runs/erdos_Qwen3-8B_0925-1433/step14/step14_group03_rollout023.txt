import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.clip(initial_h_values, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h
    best_c5 = max(np.correlate(best_h, 1 - best_h, mode='full') * dx)[1:-1].max()

    start_time = time.time()
    max_iterations = 200

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        worst_k = np.argmax(corr[1:-1])
        worst_val = corr[1:-1][worst_k]

        def objective(x):
            h = x
            corr_val = np.correlate(h, 1 - h, mode='full')[1:-1][worst_k] * dx
            return corr_val

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

        if res.success:
            new_h = res.x
            new_corr = np.correlate(new_h, 1 - new_h, mode='full') * dx
            new_max = new_corr[1:-1].max()
            if new_max < best_c5:
                best_h = new_h
                best_c5 = new_max

        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            best_h = np.clip(best_h + perturb, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, best_c5, n_points)
