import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.full(n_points, 0.5)
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        worst = np.max(np.maximum(0, corr[1:-1] - 1e-5))
        return worst * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    def optimize_start(h_start, stage):
        h = h_start.copy()
        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            if stage == 'COBYLA':
                perturb_std = 0.3 * (0.9 ** i)
            elif stage == 'L-BFGS-B':
                perturb_std = 0.05 * (0.95 ** i)
            elif stage == 'SLSQP':
                perturb_std = 0.005 * (0.95 ** i)
            else:
                perturb_std = 0
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            else:
                new_h = np.full(n_points, 0.5)
            res = minimize(obj, new_h, method=stage, bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                h = res.x
    
    # Generate structured initial patterns
    start_points = [initial_h_values.copy()]
    block_pattern = np.zeros(n_points)
    for i in range(n_points):
        block_pattern[i] = 1.0 if i % 2 == 0 else 0.0
    sum_block = np.sum(block_pattern)
    block_pattern = block_pattern * (target_sum / sum_block)
    start_points.append(block_pattern)
    random_pattern = np.random.rand(n_points)
    sum_rand = np.sum(random_pattern)
    random_pattern = random_pattern * (target_sum / sum_rand)
    start_points.append(random_pattern)
    
    for hp in start_points:
        h = hp.copy()
        optimize_start(h, 'COBYLA')
        optimize_start(h, 'L-BFGS-B')
        optimize_start(h, 'SLSQP')
    
    return (best_h, best_c5, n_points)
