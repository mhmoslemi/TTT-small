import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 45
        initial_h = np.random.rand(n_points)
        target_sum = n_points / 2
        initial_h = initial_h * target_sum / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        s = np.sum(x)
        if s == 0:
            return np.inf
        x_scaled = x * target_sum / s
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    bounds = [(0, 1)] * n_points

    best_h = initial_h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Differential Evolution for global search
    try:
        result_de = differential_evolution(
            obj,
            bounds,
            strategy='best1bin',
            popsize=30,
            maxiter=200,
            tol=1e-5,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True,
            disp=False,
            seed=seed
        )
        best_h = result_de.x
        best_c5 = obj(best_h)
    except Exception:
        pass

    # SLSQP refinement
    if time.time() - start_time < budget_s:
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            result_slsqp = minimize(
                obj,
                best_h,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints=cons,
                tol=1e-8,
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            if result_slsqp.success:
                best_h = result_slsqp.x
                best_c5 = obj(best_h)
        except Exception:
            pass

    # Final normalization
    final_h = np.clip(best_h, 0, 1)
    s = np.sum(final_h)
    if s > 0:
        final_h = final_h * target_sum / s
    else:
        final_h = best_h

    return (final_h, best_c5, n_points)
