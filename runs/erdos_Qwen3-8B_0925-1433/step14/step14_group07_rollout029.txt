import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x, k):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return corr[k+1] * dx

    def grad(x, k):
        h_ = np.clip(x, 0, 1)
        h_ = h_ * target_sum / np.sum(h_)
        grad = np.zeros_like(h_)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h_[i + k]) * dx
            if i >= k:
                grad[i] -= h_[i - k] * dx
        return grad

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[1:-1]) * dx
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_k = np.argmax(corr[1:-1])
        if worst_k < 0 or worst_k >= n_points - 1:
            worst_k = 0
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        res = minimize(obj, best_h, args=(worst_k,), method='SLSQP', jac=grad, bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')[1:-1]) * dx
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    return (best_h, best_c5, n_points)
