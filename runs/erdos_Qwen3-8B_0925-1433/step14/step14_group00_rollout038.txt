import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial guess: scale to meet sum constraint
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons,
                      tol=1e-8, options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8})
    if time.time() - start_time > budget_s:
        return (h_initial, 0.381, n_points)
    best_h = result.x
    best_c5 = result.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
