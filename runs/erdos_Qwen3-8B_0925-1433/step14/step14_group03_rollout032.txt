import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Stage 1: Differential Evolution for global search
    result = differential_evolution(obj, bounds, strategy='best1bin',
                                   popsize=15, tol=1e-6, mutation=(0.5, 1),
                                   recombination=0.8, maxiter=100, disp=False)
    best_h = result.x
    best_c5 = obj(best_h)

    # Stage 2: SLSQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h, best_c5 = result.x, result.fun

    # Final check
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
