        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result_de = differential_evolution(obj_de, bounds, strategy='best1bin', popsize=30, tol=1e-6, maxiter=100, mutation=(0.5, 1), recombination=0.9, disp=False)
    best_h = result_de.x
    best_c5 = obj_de(best_h)

    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_slsqp = [(0, 1)] * n_points

    result_slsqp = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-9)
    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = obj_slsqp(best_h_slsqp)

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h_slsqp + perturb, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(obj_slsqp, new_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-9)
        if res.success and res.fun < best_c5_slsqp:
            best_h_slsqp = res.x
            best_c5_slsqp = res.fun

    final_h = np.clip(best_h_slsqp, 0, 1)
    if np.sum(final_h) > 0:
        final_h = final_h * target_sum / np.sum(final_h)
    else:
        final_h = best_h_slsqp

    return (final_h, best_c5_slsqp, n_points)
