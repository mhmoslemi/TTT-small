import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.full_like(h, 0.5)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.maximum(0, corr[1:-1])) * dx

    def evaluate(h):
        return compute_c5(h)

    # Initial population: clones + random
    best_h = clip_scale(initial_h_values)
    best_c5 = compute_c5(best_h)

    # Neuroevolution parameters
    pop_size = min(25, n_points)
    mut_rate = 0.2
    cross_rate = 0.8
    tournament_size = 5
    max_gen = 100

    pop = [clip_scale(initial_h_values) for _ in range(pop_size)]
    for _ in range(max_gen):
        if time.time() > budget_s:
            break
        # Evaluate fitness
        fitness = [compute_c5(p) for p in pop]
        # Select tournament winners
        winners = np.argsort(fitness)[:tournament_size]
        # Mutation
        mutants = []
        for p in winners:
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.2, n_points)
                mutant = pop[p] + noise
                mutants.append(clip_scale(mutant))
        # Crossover
        offsprings = []
        for i in range(tournament_size):
            if np.random.rand() < cross_rate:
                j = np.random.randint(tournament_size)
                parent1 = pop[winners[i]]
                parent2 = pop[winners[j]]
                cross = np.random.rand(n_points)
                child = np.where(cross < 0.5, parent1, parent2)
                offsprings.append(clip_scale(child))
            else:
                offsprings.append(pop[winners[i]])
        # Replace worst
        for p in offsprings:
            if compute_c5(p) < best_c5:
                best_h, best_c5 = p, compute_c5(p)
        # Update population
        pop = [p for p in pop if compute_c5(p) > best_c5]
        pop += offsprings[:max(0, pop_size - len(pop))]

    # Local refinement
    try:
        res = minimize(
            lambda x: compute_c5(x),
            best_h,
            method='L-BFGS-B',

            tol=1e-8,
            options={'maxiter': 1000}
        )
        if res.success:
            best_h, best_c5 = res.x, res.fun
    except:
        pass

    return (best_h, best_c5, n_points)
