        if result.success:
            current_h = result.x
            new_corr_full = np.correlate(current_h, 1 - current_h, mode='full') * dx
            new_best_c5 = np.max(new_corr_full[1:-1])
            if new_best_c5 < best_c5:
                best_h = current_h
                best_c5 = new_best_c5

        if np.random.rand() < 0.1:
            perturb = np.random.normal(0, 0.01, n_points)
            best_h = np.clip(best_h + perturb, 0, 1)
            sum_h = np.sum(best_h)
            if sum_h > 0:
                best_h = best_h * target_sum / sum_h

    final_corr_full = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_best_c5 = np.max(final_corr_full[1:-1])
    return (best_h, final_best_c5, n_points)
