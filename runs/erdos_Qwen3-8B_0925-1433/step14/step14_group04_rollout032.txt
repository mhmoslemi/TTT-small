import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def initial_guesses():
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        return [h]

    guesses = initial_guesses()
    best_h = guesses[0].copy()
    best_c5 = float('inf')

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(np.maximum(0, corr[1:-1] - 1e-5)) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    for guess in guesses:
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            perturb_std = 0.02 * (0.95 ** _)
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(guess + perturb, 0, 1)
            sum_h = np.sum(new_h)
            if sum_h > 0:
                new_h = new_h * target_sum / sum_h
            else:
                new_h = guess.copy()
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.01 * (0.95 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        else:
            new_h = best_h.copy()
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    return (best_h, best_c5, n_points)
