import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.maximum(0, np.correlate(h, 1 - h, mode='full')[1:-1])) * dx

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        worst_idx = np.argmax(corr[1:-1])
        k = worst_idx
        grad = np.zeros_like(h)
        for i in range(len(h) - k):
            grad[i] += (1 - h[i + k]) * dx
            grad[i + k] += -h[i] * dx
        alpha = 0.1 / (1 + _)  # adaptive learning rate decay
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        current_c5 = np.max(np.maximum(0, corr[1:-1])) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    return (best_h, best_c5, n_points)
