import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 47
    target_sum = n_points / 2
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum

    def evaluate(h_vals):
        return evaluate_erdos_solution(h_vals)

    def rescale(h_vals):
        s = np.sum(h_vals)
        if s == 0:
            return np.zeros_like(h_vals)
        return h_vals * (target_sum / s)

    def mutate(parent, rate=0.1):
        h_mut = parent.copy()
        for i in range(len(h_mut)):
            if np.random.rand() < rate:
                h_mut[i] = np.clip(np.random.rand(), 0, 1)
        return rescale(h_mut)

    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = np.clip(parent2[i], 0, 1)
        return child

    population = [h.copy() for _ in range(50)]
    for i in range(50):
        population[i] = mutate(population[i], 0.5)

    def fitness(ind):
        return evaluate(ind)

    max_iter = int(budget_s / 5)
    best_h = h.copy()
    best_c5 = evaluate(h)
    for _ in range(max_iter):
        fit = [fitness(ind) for ind in population]
        best_idx = np.argmin(fit)
        best_h = population[best_idx]
        best_c5 = fit[best_idx]
        elite = np.argsort(fit)[:10]
        new_pop = []
        for _ in range(50):
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = crossover(population[p1], population[p2])
            child = mutate(child, 0.1)
            new_pop.append(child)
        population = new_pop
    return (rescale(best_h), best_c5, n_points)
