    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        pop_fit = [evaluate_c5(h, n_points) for h in population]
        elites = sorted(zip(pop_fit, population), key=lambda x: x[0])[:int(len(population)*0.1)]
        new_pop = []
        for _, h in elites:
            h_mut = h + npr.uniform(-0.05, 0.05, size=n_points)
            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)
            h_mut *= (n_points / 2) / sum_mut
            new_pop.append(h_mut)
        for _ in range(len(population)//2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            idx = npr.randint(1, n_points-1)
            child = np.concatenate([p1[:idx], p2[idx:]])
            sum_child = np.sum(child)
            child *= (n_points / 2) / sum_child
            child = np.clip(child, 0, 1)
            new_pop.append(child)
        combined = population + new_pop
        combined_fit = [evaluate_c5(h, n_points) for h in combined]
        combined_sorted = sorted(zip(combined_fit, combined), key=lambda x: x[0])
        new_pop = [h for _, h in combined_sorted[:len(combined)//2]]
        for i in range(min(3, len(new_pop))):
            h = new_pop[i]
            def obj(x):
                x = x.reshape(n_points)
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)
                x *= (n_points / 2) / sum_x
                return evaluate_c5(x, n_points)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x.reshape(n_points)) - (n_points / 2)}]
            res = minimize(obj, h.flatten(), method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons)
            if res.success:
                new_pop[i] = res.x.reshape(n_points)
        population = new_pop
        for h in population:
            cur_c5 = evaluate_c5(h, n_points)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
