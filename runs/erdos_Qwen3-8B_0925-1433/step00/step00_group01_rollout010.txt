            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            c5_list.append(c5)
        c5_list = np.array(c5_list)
        
        sorted_indices = np.argsort(c5_list)
        top_idx = int(len(sorted_indices) * PERCENTILE / 100)
        top_candidates = population[sorted_indices[:top_idx]]
        
        new_pop = []
        for _ in range(POP_SIZE):
            parents = np.random.choice(len(population), 3, replace=False)
            x1, x2, x3 = population[parents]
            v = x1 + F * (x2 - x3)
            v = np.clip(v, 0, 1)
            sum_v = np.sum(v)
            if sum_v == 0:
                scale = 0
            else:
                scale = (n_points / 2) / sum_v
            v = v * scale
            
            best_candidate = top_candidates[0]
            u = np.copy(v)
            for j in range(n_points):
                if np.random.rand() < CR:
                    u[j] = best_candidate[j]
            u = np.clip(u, 0, 1)
            sum_u = np.sum(u)
            if sum_u == 0:
                scale = 0
            else:
                scale = (n_points / 2) / sum_u
            u = u * scale
            new_pop.append(u)
        
        population = np.vstack([top_candidates, new_pop])
        c5_list = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5_val = np.max(corr) * (2.0 / n_points)
            c5_list.append(c5_val)
        c5_list = np.array(c5_list)
        sorted_indices = np.argsort(c5_list)
        population = population[sorted_indices[:POP_SIZE]]
        
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best == 0:
        scale = 0
    else:
        scale = (n_points / 2) / sum_best
    best_h = best_h * scale
    
    return (best_h, best_c5, n_points)
