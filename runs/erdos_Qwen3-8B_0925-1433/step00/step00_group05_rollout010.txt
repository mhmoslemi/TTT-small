import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values  # Assume initial_h_values is provided
    n_points = len(initial_h_values)
    # Normalize initial_h_values if needed
    current_sum = np.sum(initial_h_values)
    desired_sum = n_points / 2
    scale = desired_sum / current_sum
    h_initial = np.clip(initial_h_values * scale, 0, 1)
    population = [h_initial]
    for _ in range(49):
        h = np.random.rand(n_points) * 0.5
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = desired_sum / current_sum
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    
    def evaluate(h):
        dx = 2.0 / n_points
        c = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(c) * dx
        return c5
    
    best_h = h_initial
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness = [evaluate(h) for h in population]
        elites = np.argsort(fitness)[:10]
        new_pop = [population[i] for i in elites]
        
        for _ in range(len(population) - len(elites)):
            p1, p2 = population[np.random.choice(elites)], population[np.random.choice(elites)]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            if np.random.rand() < 0.2:
                child += np.random.normal(0, 0.05, n_points)
            child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            scale = desired_sum / current_sum
            child = np.clip(child * scale, 0, 1)
            new_pop.append(child)
        
        population = new_pop
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h) * (2.0 / n_points)
    if not np.isclose(sum_h, 1.0):
        scale = 1.0 / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
    return (best_h, best_c5, n_points)
