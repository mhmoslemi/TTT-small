import numpy as np
from time import time

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from time import time
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time()
    while True:
        elapsed = time() - start_time
        if elapsed > budget_s - 1:
            break
        # Generate perturbation indices
        perturb_indices = np.random.choice(n_points, size=10, replace=False)
        # Perturb and clip
        h_pert = h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h_pert[perturb_indices] += noise[perturb_indices]
        h_pert = np.clip(h_pert, 0.0, 1.0)
        # Rescale to ensure integral constraint
        scale_factor = (n_points / 2.0) / np.sum(h_pert)
        h_pert *= scale_factor
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_pert, n_points)
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_pert.copy()
            best_c5 = current_c5
            h = h_pert.copy()
    return best_h.tolist(), best_c5, n_points
