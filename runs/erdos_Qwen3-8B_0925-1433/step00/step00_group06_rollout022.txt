import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    dx = 2.0 / n_points
    perturbation_radius = 0.08
    decay_rate = 0.995
    early_stop_counter = 0
    start_time = time.time()
    
    while (time.time() - start_time) < budget_s:
        # Perturbation phase
        indices = np.random.choice(n_points, 20, replace=False)
        delta = np.zeros(n_points)
        delta[indices] = np.random.normal(0, perturbation_radius, 20)
        h_pert = h + delta
        h_pert = np.clip(h_pert, 0, 1)
        sum_pert = np.sum(h_pert)
        scale_factor = (n_points / 2) / sum_pert
        h_pert *= scale_factor
        c5_pert = evaluate_erdos_solution(h_pert)
        
        # Update best solution
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
            early_stop_counter = 0
        else:
            early_stop_counter += 1
        
        # Local optimization
        if early_stop_counter < 10:
            def objective(x):
                h_opt = x.reshape(n_points)
                return evaluate_erdos_solution(h_opt)
            bounds = [(0, 1) for _ in range(n_points)]
            result = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                                 options={'ftol': 1e-5, 'maxiter': 30})
            if result.success:
                best_h = result.x
                sum_best = np.sum(best_h)
                scale_factor = (n_points / 2) / sum_best
                best_h *= scale_factor
                best_c5 = evaluate_erdos_solution(best_h)
        
        # Decay perturbation radius
        perturbation_radius *= decay_rate
        
        # Early stopping
        if early_stop_counter >= 10:
            break
    
    # Final scale adjustment
    sum_best = np.sum(best_h)
    scale_factor = (n_points / 2) / sum_best
    best_h *= scale_factor
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
