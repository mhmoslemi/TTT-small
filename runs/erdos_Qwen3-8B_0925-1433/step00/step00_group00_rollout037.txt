            gbest = np.copy(particles[np.argmin(pbest_fitness)])

    top_indices = np.argsort(pbest_fitness)[:int(num_particles * 0.1)]
    for idx in top_indices:
        h_candidate = pbest[idx]
        def obj(x):
            x = x.reshape(-1)
            return compute_c5(x)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 25})
        if res.success:
            if res.fun < pbest_fitness[idx]:
                pbest[idx] = res.x
                pbest_fitness[idx] = res.fun

    best_index = np.argmin(pbest_fitness)
    best_h = pbest[best_index]
    final_c5 = pbest_fitness[best_index]
    return (best_h, final_c5, n_points)
