        elites = [population[i] for i in elite_idx]
        new_pop = []
        for h in population:
            mut = h.copy()
            for i in range(n_points):
                mut[i] += np.random.uniform(-0.05, 0.05)
                mut[i] = np.clip(mut[i], 0.0, 1.0)
            sum_mut = np.sum(mut)
            scale = sum_required / sum_mut
            mut *= scale
            new_pop.append(mut)
        top_idx = np.argsort(fitness)[:int(len(population)*0.2)]
        for i in range(0, len(top_idx), 2):
            if i + 1 < len(top_idx):
                parent1 = population[top_idx[i]]
                parent2 = population[top_idx[i+1]]
                child = (parent1 + parent2) / 2
                sum_c = np.sum(child)
                scale = sum_required / sum_c
                child *= scale
                new_pop.append(child)
        combined = elites + new_pop
        combined.sort(key=lambda x: compute_c5(x))
        population = combined[:len(population)]
        top_candidates = population[:int(len(population)*0.03)]
        for h in top_candidates:
            def obj(x):
                hh = x.copy()
                return compute_c5(hh)
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_required}]
            res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP')
            if res.success:
                optimized = res.x
                sum_opt = np.sum(optimized)
                scale = sum_required / sum_opt
                optimized *= scale
                current_c5 = compute_c5(optimized)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = optimized
        elapsed = time.time() - time_start
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n_points)
