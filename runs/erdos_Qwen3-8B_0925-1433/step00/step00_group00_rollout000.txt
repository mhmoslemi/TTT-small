import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    
    def objective(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h, method='SLSQP',

    if elapsed > budget_s:

        h_opt = result.x
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5 = np.max(corr) * dx
        return h_opt, c5, n_points

        corr = np.correlate(h, 1 - h, mode='full')
