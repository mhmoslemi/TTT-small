    except NameError:
        initial_h = np.random.rand(100)
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h)
        initial_h = initial_h * (100 / 2) / sum_h
    n_points = len(initial_h)
    sum_h = np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    if sum_h > 0:
        initial_h = initial_h * (n_points / 2) / sum_h

    num_particles = 50
    particles = []
    velocities = []
    for _ in range(num_particles):
        perturbation = np.random.normal(0, 0.015, n_points)
        h = initial_h + perturbation
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        particles.append(h)
        velocities.append(np.zeros(n_points))

    def compute_fitness(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    pbest = [p.copy() for p in particles]
    pbest_fitness = [compute_fitness(p) for p in particles]
    gbest = particles[np.argmin(pbest_fitness)]
    best_c5 = min(pbest_fitness)
    omega = 0.9
    c1, c2 = 1.5, 1.5
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s and iteration < 300:
        omega *= 0.995
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocity = velocities[i] * omega + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = velocity
            particles[i] += velocity
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 0:
                particles[i] = particles[i] * (n_points / 2) / sum_h
        for i in range(num_particles):
            fit = compute_fitness(particles[i])
            if fit < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fit
        new_gbest_idx = np.argmin(pbest_fitness)
        if pbest_fitness[new_gbest_idx] < best_c5:
            best_c5 = pbest_fitness[new_gbest_idx]
            gbest = particles[new_gbest_idx]
        if iteration % 10 == 0:
            top = np.argsort(pbest_fitness)[:5]
            for idx in top:
                res = opt.minimize(lambda x: -compute_fitness(x), x0=pbest[idx], bounds=[(0,1)]*n_points,
                                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
                if res.success:
                    pbest[idx] = res.x
                    particles[idx] = res.x
        iteration += 1

    final_c5 = compute_fitness(gbest)
    return (gbest, final_c5, n_points)
