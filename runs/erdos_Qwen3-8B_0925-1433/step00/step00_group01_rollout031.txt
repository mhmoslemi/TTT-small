            mutant = np.clip(mutant, 0, 1)
            s = mutant.sum()
            mutant *= (n_points / 2) / s
            best = elites[0]
            trial = np.where(np.random.rand(n_points) < 0.9, mutant, best)
            corr = np.correlate(trial, 1 - trial, mode='full')
            max_corr = np.max(corr)
            dx_val = 2.0 / n_points
            trial_c5 = max_corr * dx_val
            if trial_c5 < new_pop[-1][1]:
                new_pop[-1] = trial
        population = new_pop
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx_val = 2.0 / n_points
            c5 = max_corr * dx_val
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if iter % 50 == 0:
            top_ids = np.argsort(current_values)[:5]
            for i in top_ids:
                h = sorted_pop[i]
                grad = np.zeros(n_points)
                eps = 1e-5
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    corr_p = np.correlate(h_plus, 1 - h_plus, mode='full')
                    max_p = np.max(corr_p)
                    c5_p = max_p * dx_val
                    h_minus = h.copy()
                    h_minus[j] -= eps
                    corr_m = np.correlate(h_minus, 1 - h_minus, mode='full')
                    max_m = np.max(corr_m)
                    c5_m = max_m * dx_val
                    grad[j] = (c5_p - c5_m) / (2 * eps)
                step = 0.1 / (iter + 1)
                h = h - step * grad
                h = np.clip(h, 0, 1)
                s = h.sum()
                h *= (n_points / 2) / s
                sorted_pop[i] = h
        best_h = np.clip(best_h, 0, 1)
        s = best_h.sum()
        best_h *= (n_points / 2) / s
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        dx_val = 2.0 / n_points
        final_c5 = max_corr * dx_val
    return (best_h.tolist(), final_c5, n_points)
