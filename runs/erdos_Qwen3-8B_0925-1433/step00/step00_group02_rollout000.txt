import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        initial_h = initial_h_values
    except NameError:
        n_points = 45
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        target_sum = n_points / 2
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        n_points = len(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        target_sum = n_points / 2
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if np.isclose(current_sum, target_sum, atol=1e-6):
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    population_size = 50
    mutation_rate = 0.1
    tournament_size = 5
    h_pop = [initial_h]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h_pop.append(h)

    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    start_time = time.time()

    for gen in range(1000):
        fitness = [compute_c5(h) for h in h_pop]
        parents = []
        for _ in range(population_size):
            indices = np.random.choice(range(population_size), tournament_size)
            winner = indices[np.argmin(fitness[indices])]
            parents.append(h_pop[winner])
        
        offspring = []
        for i in range(population_size // 2):
            p1, p2 = parents[2*i], parents[2*i + 1]
            child = (p1 + p2) / 2
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.uniform(-0.1, 0.1)
            child = repair_h(child)
            offspring.append(child)
        
        new_pop = h_pop + offspring
        new_fitness = [compute_c5(h) for h in new_pop]
        sorted_indices = np.argsort(new_fitness)
        h_pop = [new_pop[i] for i in sorted_indices[:population_size]]
        
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = h_pop[current_best]
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h, best_c5, n_points)
