import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = np.array(initial_h_values)

    options = {
        'maxiter': 1000,
        'ftol': 1e-5,
        'eps': 1e-8,
        'disp': False
    }

    start_time = time.time()

    result = scipy.optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
