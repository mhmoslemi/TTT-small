    population_size = 25
    population = []
    initial_sum = np.sum(initial_h_values)
    scaled_initial = initial_h_values * (target_sum / initial_sum)
    population.append(scaled_initial.copy())
    for _ in range(population_size - 1):
        random_h = np.random.rand(n_points)
        scaled = random_h * (target_sum / np.sum(random_h))
        population.append(scaled.copy())

    best_h = scaled_initial.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    F = 0.8
    CR = 0.9
    percentile = 80

    while time.time() - start_time < budget_s:
        c5_values = []
        for h in population:
            c5 = compute_c5(h)
            c5_values.append(c5)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        sorted_indices = np.argsort(c5_values)
        top_indices = sorted_indices[:int(percentile / 100 * len(population))]
        new_population = []
        for i in range(len(population)):
            if i in top_indices:
                new_population.append(population[i].copy())
            else:
                parents = np.random.choice(population, 3, replace=False)
                x1, x2, x3 = parents
                mutant = x1 + F * (x2 - x3)
                mutant = np.clip(mutant, 0, 1)
                mutant_sum = np.sum(mutant)
                if mutant_sum == 0:
                    mutant = np.zeros_like(mutant)
                else:
                    mutant = mutant * (target_sum / mutant_sum)
                trial = mutant.copy()
                mask = np.random.rand(n_points) < CR
                trial[mask] = best_h[mask]
                trial_sum = np.sum(trial)
                if trial_sum == 0:
                    trial = np.zeros_like(trial)
                else:
                    trial = trial * (target_sum / trial_sum)
                new_population.append(trial.copy())
        population = new_population
    return (best_h, best_c5, n_points)
