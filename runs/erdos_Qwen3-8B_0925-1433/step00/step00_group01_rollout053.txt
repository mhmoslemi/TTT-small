        population.append(h_scaled)
    
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([adjust_sum(h, target_sum) for h in population])
    
    best_h = population[0]
    best_c5 = evaluate_fitness(best_h)
    
    start_time = time.time()
    for _ in range(MAX_ITER):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            parents = np.random.choice(population, 3, replace=False)
            h_mut = mutate(parents, target_sum)
            new_population.append(h_mut)
        population = np.array(new_population)
        population = np.array([np.clip(h, 0, 1) for h in population])
        population = np.array([adjust_sum(h, target_sum) for h in population])
        fitness = np.array([evaluate_fitness(h) for h in population])
        elite_idx = np.argsort(fitness)[:POP_SIZE//2]
        population = population[elite_idx]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
    
    final_h = best_h
    final_c5 = evaluate_fitness(final_h)
    final_h = np.clip(final_h, 0, 1)
    final_h = adjust_sum(final_h, target_sum)
    final_c5 = evaluate_fitness(final_h)
    
    return (final_h, final_c5, n_points)
