        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        x0 = best_h.copy()
        
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
            if res.success:
                h_candidate = res.x
                current_c5 = evaluate_erdos_solution(h_candidate, n_points)
                if current_c5 < best_c5:
                    best_h = h_candidate
                    best_c5 = current_c5
        except:
            pass
        
        iter_count += 1
        if iter_count % 50 == 0:
            delta = 0.05 * np.random.randn(n_points)
            h_pert = np.clip(best_h + delta, 0.0, 1.0)
            sum_pert = np.sum(h_pert)
            if abs(sum_pert - n_points / 2) > 1e-6:
                scale_pert = (n_points / 2) / sum_pert
                h_pert = h_pert * scale_pert
            c5_pert = evaluate_erdos_solution(h_pert, n_points)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert
        
        elapsed_time = time.time() - start_time
        
    return (best_h.tolist(), best_c5, n_points)
