import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    desired_sum = n_points / 2
    
    def clip_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros(n_points)
        scale = desired_sum / h_sum
        return np.clip(h * scale, 0, 1)
    
    # Initialize population
    population = [np.array(initial_h_values).copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = clip_scale(h)
        population.append(h)
    
    best_h = np.array(initial_h_values).copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate_erdos_solution(h) for h in population])
        top_indices = np.argsort(fitness)[:10]
        elites = [population[i] for i in top_indices]
        
        new_pop = elites[:]
        for _ in range(len(population) - 10):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = np.random.choice([p1, p2], size=n_points)
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = clip_scale(child)
            new_pop.append(child)
        
        population = new_pop
        for h in population:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
