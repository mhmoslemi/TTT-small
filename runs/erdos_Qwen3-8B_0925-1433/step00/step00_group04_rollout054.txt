        h = np.clip(h * scale_factor, 0, 1)
        while abs(np.sum(h) - target_sum) > 1e-6:
            scale_factor = target_sum / np.sum(h)
            h = np.clip(h * scale_factor, 0, 1)
    # Define objective function
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        c_k = corr * dx
        return np.log(np.mean(np.exp(c_k)))
    # Define constraint
    def constraint(x):
        return np.sum(x) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=100)
    elapsed = time.time() - start_time
    best_h = result.x
    # Post-process
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - target_sum) > 1e-6:
        scale_factor = target_sum / sum_best
        best_h = np.clip(best_h * scale_factor, 0, 1)
    # Compute final C5
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c_k = corr * dx
    c5_bound = np.max(c_k)
    return (best_h, c5_bound, n_points)
