        return np.sum(x) - n_points / 2
    # Optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(compute_c5, h, method='L-BFGS-B', bounds=bounds,
                      constraints={'type': 'eq', 'fun': constraint},
                      options={'maxiter': 200, 'ftol': 1e-7})
    h_opt = result.x
    # Final clipping and adjustment
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    c5_bound = compute_c5(h_opt)
    return h_opt, c5_bound, n_points
