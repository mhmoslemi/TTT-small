            h *= required_sum / np.sum(h)

        iteration += 1

    # Final evaluation
    corr = scipy.signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
