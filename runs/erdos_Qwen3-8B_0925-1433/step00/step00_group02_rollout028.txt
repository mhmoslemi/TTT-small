        k = worst_idx - (n_points - 1)
        # Finite difference gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += 1e-4
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
            delta = (np.correlate(h_perturb, 1 - h_perturb, mode='full')[worst_idx] - overlaps[worst_idx])
            grad[i] = delta * 1e4
        step = -step_size * grad
        h_candidate = h + step
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
        # Check if improvement
        c5_candidate = compute_c5(h_candidate, n_points)
        if c5_candidate < current_c5:
            h = h_candidate
        else:
            # Random perturbation
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h_candidate = h + noise
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
            c5_candidate = compute_c5(h_candidate, n_points)
            if c5_candidate < current_c5:
                h = h_candidate
        step_size *= 0.997

    c5_bound = compute_c5(best_h, n_points)
    return (best_h, c5_bound, n_points)

def compute_c5(h, n_points):
    overlaps = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
    return overlaps.max()
