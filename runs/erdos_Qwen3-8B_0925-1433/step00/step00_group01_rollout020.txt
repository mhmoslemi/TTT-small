import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import minimize

    n_points = len(initial_h_values) if 'initial_h_values' in locals() else 47
    h = initial_h_values.copy() if 'initial_h_values' in locals() else np.random.rand(n_points)
    h *= (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    dx = 2.0 / n_points

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return float('inf')
        h_scaled = h_clipped * (n_points / 2) / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        max_corr = np.max(corr * dx)
        return max_corr

    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        best_h = best_h * (n_points / 2) / sum_best
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr * dx)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h, final_c5, n_points)
    return (best_h, final_c5, n_points)
