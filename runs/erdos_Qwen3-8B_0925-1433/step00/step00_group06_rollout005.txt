import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    required_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1.0 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    perturbation_size = 0.05
    num_perturb_per_iter = 10

    while True:
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        current_h = best_h.copy()
        perturb_indices = np.random.choice(n_points, size=num_perturb_per_iter, replace=False)
        noise = np.random.normal(0, perturbation_size, size=n_points)
        h_pert = current_h + noise
        h_pert = np.clip(h_pert, 0.0, 1.0)
        scale_factor = required_sum / np.sum(h_pert)
        h_pert *= scale_factor
        current_c5 = compute_c5(h_pert)
        if current_c5 < best_c5:
            best_h = h_pert.copy()
            best_c5 = current_c5
        if elapsed_time > budget_s * 0.8:
            perturbation_size *= 0.5

    assert np.all(0 <= best_h <= 1.0)
    assert np.isclose(np.sum(best_h), required_sum, atol=1e-6)
