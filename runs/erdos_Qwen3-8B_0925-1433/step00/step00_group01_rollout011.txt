import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47
    initial_h = initial_h_values

        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx

        s = np.sum(arr)
        if s == 0:
            return np.zeros_like(arr)
        return arr * (n_points / 2) / s

    best_h = population[0].copy()
