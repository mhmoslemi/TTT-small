def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    h = initial_h.copy()
    sum_h = np.sum(h)
    scaling = (n_points / 2) / sum_h

    population = [h]

        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h
        h = h * scaling
        h = np.clip(h, 0, 1)
        population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    best_h = population[np.argmin([compute_c5(h) for h in population])]
    best_c5 = compute_c5(best_h)
    elapsed = 0

        if elapsed > budget_s:

        # Evaluate fitness
        fitness = np.array([compute_c5(h) for h in population])
        # Tournament selection
        parents = []
        for _ in range(len(population) // 2):
            idxs = np.random.choice(len(population), size=5, replace=False)
            best_idx = np.argmin(fitness[idxs])
            parents.append(population[idxs[best_idx]])
        # Crossover with adaptive alpha
        new_population = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[(i+1)%len(parents)]
            alpha = 0.7 + 0.3 * np.random.rand()

            # Mutation
            if np.random.rand() < 0.15:

        population = new_population
        # Elitism
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
