            best_h = h.copy()
        
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        lag_index = m - (n_points - 1)
        lag = lag_index
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - lag) % n_points
            grad[i] = h[j]
        
        perturb = np.random.rand(n_points) < perturb_rate
        adjustment = eta * grad * perturb
        h_candidate = h + adjustment
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = rescale_to_sum(h_candidate, n_points / 2)
        
        new_c5 = compute_c5(h_candidate)
        if new_c5 < current_c5:
            h = h_candidate
        else:
            h = np.random.uniform(0, 1, size=n_points)
            h = rescale_to_sum(h, n_points / 2)
        
        if abs(current_c5 - best_c5) < convergence_threshold:
            break
        
        if time.time() - time_start >= budget_s:
            break
    
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return best_h, final_c5, n_points
