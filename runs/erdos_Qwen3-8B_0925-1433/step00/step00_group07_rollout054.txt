import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    h = initial_h.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = h * scale
    h = np.clip(h, 0, 1)
    pop_size = 200
    pop = [h]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        sum_rand = np.sum(h_rand)
        h_rand = h_rand * (n_points / 2) / sum_rand
        h_rand = np.clip(h_rand, 0, 1)
        pop.append(h_rand)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fit = [compute_c5(p) for p in pop]
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5 = fit[best_idx]
            best_h = pop[best_idx].copy()
        
        new_pop = []
        for _ in range(pop_size // 2):
            p1_idx = np.random.randint(len(pop))
            p2_idx = np.random.randint(len(pop))
            p1, p2 = pop[p1_idx], pop[p2_idx]
            alpha = np.random.uniform(0.6, 0.9)
            child = alpha * p1 + (1 - alpha) * p2
            sum_c = np.sum(child)
            child = child * (n_points / 2) / sum_c
            child = np.clip(child, 0, 1)
            
            if np.random.rand() < 0.2:
                for i in range(n_points):
                    if child[i] > 0:
                        pert = np.random.power(1.5) * (1 - 2 * np.random.rand())
                        child[i] += pert / (child[i] ** 1.5)
                        child[i] = np.clip(child[i], 0, 1)
            
            new_pop.append(child)
        
        pop = new_pop
        if np.random.rand() < 0.1:
            idx = np.random.randint(len(pop))
            pop[idx] = np.random.rand(n_points)
            sum_r = np.sum(pop[idx])
            pop[idx] = pop[idx] * (n_points / 2) / sum_r
            pop[idx] = np.clip(pop[idx], 0, 1)
    
    best_h = np.array(best_h)
    return (best_h, best_c5, n_points)
