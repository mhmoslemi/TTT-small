        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        c_values = cc * dx
        return np.max(c_values)

    def constraint_eq(h_flat):
        return np.sum(h_flat.reshape(n_points)) - required_sum

    bounds = [(0, 1)] * n_points
    initial_guess = h.flatten()

    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_eq}],
        options={'maxiter': 200, 'ftol': 1e-7, 'disp': False}
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (h, np.max(correlate(h, 1 - h, mode='full') * dx), n_points)

    best_h = result.x
    best_h = np.clip(best_h.reshape(n_points), 0, 1)
    current_sum = np.sum(best_h)
    best_h *= required_sum / current_sum

    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    return (best_h, c5_bound, n_points)
