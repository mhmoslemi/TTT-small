    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        initial_h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(initial_h)
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total
        initial_h = np.clip(initial_h, 0, 1)
    else:
        n_points = 500
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total
    dx = 2.0 / n_points

    # Ensure sum constraint
    total = np.sum(initial_h)
    if abs(total - n_points / 2) > 1e-3:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total

    population_size = min(50, n_points)
    population = [initial_h]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        population.append(h)

    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    time_remaining = budget_s

    while time_remaining > 0:
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:int(len(population) * 0.05)]]
        new_population = elites[:]

        while len(new_population) < population_size:
            parent1 = elites[np.random.randint(len(elites))]
            parent2 = elites[np.random.randint(len(elites))]
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])

        top_indices = sorted_indices[:int(len(population) * 0.01)]
        for idx in top_indices:
            h = population[idx]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = opt.minimize(lambda x: evaluate_erdos_solution(x), h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
            if res.success:
                new_population.append(res.x)

        population = new_population
        time_remaining = budget_s - (time.time() - start_time)

    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)

def mutate(h):
    h = h.copy()
    delta = np.random.uniform(-0.05, 0.05, size=len(h))
    h += delta
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total <= 0:
        return h
    h *= (len(h) / 2) / total
    return h
