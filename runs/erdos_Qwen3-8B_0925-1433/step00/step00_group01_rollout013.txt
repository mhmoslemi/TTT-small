        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            h[:] = result.x
    
    # Final validation
    final_h = population[np.argmin([compute_c5(p) for p in population])]
    final_c5 = compute_c5(final_h)
    return (final_h, final_c5, n_points)
