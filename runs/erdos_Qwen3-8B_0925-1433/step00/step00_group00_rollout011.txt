import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    
    # Initialize population with initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    def rescale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.random.rand(n_points)
        return h * (n_points / 2) / s
    
    initial_h = rescale_h(initial_h)
    
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = rescale_h(h)
        population.append(h)
    
    best_h = initial_h
    best_c5 = correlate(best_h, 1 - best_h, mode='full').max() * dx
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            fitness.append(corr.max() * dx)
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(population_size * 0.1)]
        elites = [population[i] for i in elite_idx]
        
        # Generate new population
        new_population = []
        # Mutation
        for h in population:
            mut = np.random.uniform(-0.05, 0.05, n_points)
            h_mut = h + mut
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = rescale_h(h_mut)
            new_population.append(h_mut)
        # Crossover
        for _ in range(population_size - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            child = rescale_h(child)
            new_population.append(child)
        
        # Local refinement
        top_idx = np.argsort(fitness)[:int(population_size * 0.03)]
        for i in top_idx:
            h = population[i]
            def obj(x):
                return correlate(x, 1 - x, mode='full').max() * dx
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)*n_points], constraints=cons, options={'maxiter': 25})
            if res.success:
                population[i] = res.x
        
        # Update best
        for h in population:
            curr = correlate(h, 1 - h, mode='full').max() * dx
            if curr < best_c5:
                best_c5 = curr
                best_h = h
    
    return (best_h, best_c5, n_points)
