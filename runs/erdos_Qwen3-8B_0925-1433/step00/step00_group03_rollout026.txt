import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    # Assume initial_h_values is a global variable
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        h = np.full(n_points, n_points / 2 / n_points)
    n_points = len(h)
    # Normalize to satisfy the sum constraint
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h
    h *= scaling_factor
    dx = 2.0 / n_points

    def obj_func(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def cons_func(h_vec):
        return np.sum(h_vec) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': cons_func}]
    result = opt.minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
    best_h = result.x
    c5_bound = obj_func(best_h)
    return best_h, c5_bound, n_points
