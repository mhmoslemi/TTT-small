import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h0 = initial_h_values.copy()
    tolerance = 1e-5
    max_iter = 1000
    early_stop = 100
    seq = [h0]

    def objective(x):
        x = np.array(x)
        corr = correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    start_time = time.time()
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance, options={'maxiter': max_iter})
    h_values = result.x
    c5_bound = objective(h_values)

    current_time = time.time()
    if current_time - start_time < budget_s:
        for _ in range(early_stop):
            perturb = np.random.normal(0, 1e-3, size=n_points)
            h_perturb = np.clip(h_values + perturb, 0, 1)
            res = minimize(objective, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=tolerance*10, options={'maxiter': 50})
            if res.fun < c5_bound:
                c5_bound = res.fun
                h_values = res.x
            if current_time - start_time >= budget_s:
                break

    return (h_values, c5_bound, n_points)
