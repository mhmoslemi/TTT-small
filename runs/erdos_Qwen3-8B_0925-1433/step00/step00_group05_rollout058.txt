def run(seed=42, budget_s=600, **kwargs):

    scaled_initial = initial_h_values.copy()

        scale_factor = target_sum / current_sum

        scaled_initial = np.clip(scaled_initial, 0, 1)

    population.append(scaled_initial)

    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        # scale to target_sum
        scale_factor = target_sum / np.sum(h)
        h = h * scale_factor
        h = np.clip(h, 0, 1)
        population.append(h)

    best_c5 = float('inf')
    start_time = time.time()

            h_sum = np.sum(h)
            if h_sum == 0:

                h = np.zeros(n_points) + 1.0 / n_points
