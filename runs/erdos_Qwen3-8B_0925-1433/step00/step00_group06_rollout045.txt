import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        n_points = 95
        h = np.full(n_points, 0.5)
    n_points = len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0.0

    while elapsed_time < budget_s:
        start = time.time()
        indices = np.random.choice(n_points, size=10, replace=False)
        h_pert = h.copy()
        noise = np.random.normal(0, 0.05, 10)
        h_pert[indices] += noise
        h_pert = np.clip(h_pert, 0.0, 1.0)
        if np.sum(h_pert) == 0:
            h_pert[:] = 0.5
        else:
            scale = (n_points / 2) / np.sum(h_pert)
            h_pert *= scale
        c5_pert = evaluate_erdos_solution(h_pert, n_points)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        elapsed_time += time.time() - start

    return best_h.tolist(), best_c5, n_points
