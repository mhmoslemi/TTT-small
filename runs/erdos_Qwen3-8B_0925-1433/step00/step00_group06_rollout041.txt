import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def obj_func(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx

    def eq_cons(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h_initial.tolist()
    cons = [{'type': 'eq', 'fun': eq_cons}]
    options = {
        'ftol': 1e-8,
        'maxiter': 1000,
        'disp': False
    }

    # Run optimization
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons, options=options)

    # Extract result
    h_opt = result.x
    h_array = np.array(h_opt)
    corr = np.correlate(h_array, 1 - h_array, mode='full')
    c5_bound = np.max(corr) * dx

    # Final check
    if np.any(h_opt < 0) or np.any(h_opt > 1):
        h_opt = np.clip(h_opt, 0, 1)
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_bound = np.max(corr) * dx

    return h_opt, c5_bound, n_points
