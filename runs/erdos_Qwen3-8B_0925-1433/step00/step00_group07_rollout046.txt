import numpy as np
import scipy.signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    n_points = 200  # Use 200 points for sufficient resolution without exceeding time budget
    dx = 2.0 / n_points
    max_generations = 1000
    pop_size = 100
    tournament_size = 5
    mutation_rate = 0.15
    gamma = 1.5
    diversity_threshold = 0.1
    elite_ratio = 0.01
    mutation_decay = 0.05
    generation_interval = 50

    def normalize(h):
        total = np.sum(h)
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)

    def compute_c5(h):
        try:
            corr = scipy.signal.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        except:
            return np.inf

    def levy_flight(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                power = 1.0 / gamma
                sigma = (np.random.rand() ** power) / (np.sqrt(np.random.rand()))
                noise = np.random.randn() * sigma
                h[i] += noise * (h[i] ** (-gamma)) * (1 - 2 * np.random.rand())
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)

    def crossover(parent1, parent2, diversity):
        alpha = 0.7 + 0.3 * (1 - diversity)
        child = alpha * parent1 + (1 - alpha) * parent2
        return child

    def evaluate_population(pop):
        return [compute_c5(ind) for ind in pop]

    if initial_h_values is not None:
        initial_h_values = normalize(initial_h_values)
        pop = [initial_h_values]
    else:
        pop = [np.random.rand(n_points) for _ in range(pop_size)]

    pop = [normalize(ind) for ind in pop]
    best_h = pop[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        if gen % generation_interval == 0:
            current_c5s = evaluate_population(pop)
            diversity = np.std(current_c5s)
            print(f"Gen {gen}: C5={best_c5:.6f}, Diversity={diversity:.4f}")

        if gen > 0:
            current_c5s = evaluate_population(pop)
            crowding = np.argsort(current_c5s)
            candidates = [pop[i] for i in crowding[:int(len(pop)*0.8)]]
            pop = candidates

        new_pop = []
        for _ in range(pop_size):
            if len(pop) < 2:
                parent1, parent2 = pop[0], pop[0]
            else:
                tournament = np.random.choice(pop, size=tournament_size)
                parent1 = tournament[np.argmin([compute_c5(ind) for ind in tournament])]
                parent2 = tournament[np.argmin([compute_c5(ind) for ind in tournament])]
            child = crossover(parent1, parent2, np.std([compute_c5(ind) for ind in pop]))
            child = levy_flight(child)
            new_pop.append(child)

        pop = new_pop
        current_c5s = evaluate_population(pop)
        elite_idx = np.argsort(current_c5s)[:int(len(pop)*elite_ratio)]
        pop = [pop[i] for i in elite_idx]
        while len(pop) < pop_size:
            pop.append(np.random.rand(n_points))
            pop[-1] = normalize(pop[-1])

        if compute_c5(pop[0]) < best_c5:
            best_c5 = compute_c5(pop[0])
            best_h = pop[0]

        if np.std(current_c5s) < diversity_threshold:
            new_solutions = [np.random.rand(n_points) for _ in range(int(len(pop)*0.1))]
            new_solutions = [normalize(sol) for sol in new_solutions]
            pop = new_solutions + pop[:int(len(pop)*0.9)]

    best_h = normalize(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
