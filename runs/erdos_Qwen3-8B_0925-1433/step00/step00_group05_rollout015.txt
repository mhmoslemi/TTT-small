        b_ub = np.zeros(2 * n_points)
        for i in range(n_points):
            A_ub[i, i] = -1.0
            b_ub[i] = current_h[i]
            A_ub[n_points + i, i] = 1.0
            b_ub[n_points + i] = 1.0 - current_h[i]
        A_eq = np.ones((1, n_points))
        b_eq = 0.0

        result = linprog(-grad, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs', bounds=None)
        if not result.success:
            break
        delta_h = result.x

        best_alpha = 0.0
        min_ov = ov
        alphas = np.linspace(0, 1, line_search_steps)
        for alpha in alphas:
            h_candidate = current_h + alpha * delta_h
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c_candidate = np.correlate(h_candidate, 1 - h_candidate, mode="full")
            ov_candidate = c_candidate[k0] * dx
            if ov_candidate < min_ov:
                min_ov = ov_candidate
                best_alpha = alpha
        
        current_h = current_h + best_alpha * delta_h
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        new_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        if abs(new_c5 - best_c5) < tol:
            break
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return (best_h, best_c5, n_points)
