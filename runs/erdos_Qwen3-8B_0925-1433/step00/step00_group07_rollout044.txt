import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    if sum_h != 0:
        h *= (n_points / 2) / sum_h
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    learning_rate = 0.01
    decay_rate = 0.999
    max_iterations = 500
    start_time = np.datetime64('now')
    for _ in range(max_iterations):
        cor = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(cor)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_idx = np.argmax(cor)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            term_plus = (1 - h[i_plus_k]) * dx if 0 <= i_plus_k < n_points else 0.0
            i_minus_k = i - k
            term_minus = h[i_minus_k] * dx if 0 <= i_minus_k < n_points else 0.0
            grad[i] = term_plus - term_minus
        h = np.clip(h - learning_rate * grad, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= (n_points / 2) / sum_h
        learning_rate *= decay_rate
        if np.datetime64('now') - start_time > np.timedelta64(budget_s, 's'):
            break
    return (best_h, best_c5, n_points)
