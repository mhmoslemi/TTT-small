        k_max = np.argmax(O)

        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += eps
            h_perturbed = adjust_sum(h_perturbed, n_points)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            O_perturbed = corr_perturbed * dx
            grad[i] = (O_perturbed[k_max] - O[k_max]) / eps

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = adjust_sum(h, n_points)
        iter_count += 1

    return best_h, best_c5, n_points
