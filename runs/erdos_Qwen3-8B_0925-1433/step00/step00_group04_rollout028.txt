import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import minimize

    # Handle initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 600
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Ensure sum constraint is satisfied
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-7:
        h = h * (n_points / 2) / sum_h

    # Define objective function for L-BFGS-B optimization
    def obj_func(h_flat):
        h = h_flat.reshape(-1)
        cc = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cc) * dx
        return c5

    # Define constraints and bounds
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization settings
    result = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-7, 'gtol': 1e-7})

    # Clip and rescale to ensure [0,1] bounds and sum constraint
    h_opt = np.clip(result.x, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    if abs(sum_h_opt - n_points / 2) > 1e-7:
        h_opt = h_opt * (n_points / 2) / sum_h_opt

    # Final evaluation
    cc = np.correlate(h_opt, 1 - h_opt, mode='full')
