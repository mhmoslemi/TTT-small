import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    if res.success:

    else:
        h_opt = h

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
