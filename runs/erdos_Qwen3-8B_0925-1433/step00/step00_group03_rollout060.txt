def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    best_h = h.copy()

    max_iter = 1000

    learning_rate = 0.01
    improvement_threshold = 1e-6

    def adjust_sum(arr, n_points):
        s = np.sum(arr)
        target = n_points / 2
        if abs(s - target) < 1e-10:
            return arr
        delta = target - s
        arr[-1] += delta
        return np.clip(arr, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        O = corr * dx
        return np.max(O)

    start_time = time.time()
    for _ in range(max_iter):
        current_c5 = compute_c5(h)
