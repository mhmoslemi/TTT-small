import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    current_h = np.array(initial_h_values, dtype=np.float64)
    best_h = current_h.copy()
    best_c5 = np.inf
    max_iter = 200
    line_search = 5
    tol = 1e-4
    time_start = np.datetime64('now')
    
    def update_h(h, k0):
        epsilon = 1e-6
        c = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(c)
        overlap = c[k0] * dx
        derivative = np.zeros_like(h)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            c_perturbed = np.correlate(h_perturbed, 1 - h, mode='full')
            derivative[i] = (c_perturbed[k0] - c[k0]) / epsilon
        delta_h = cp.Variable(n_points)
        constraints = [
            delta_h >= -h,
            delta_h <= 1 - h,
            cp.sum(delta_h) == 0
        ]
        objective = cp.Maximize(cp.sum(derivative * delta_h))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS)
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            return h.copy()
        delta_h_opt = delta_h.value
        best_alpha = 0.0
        best_val = overlap
        for alpha in np.linspace(0, 1, line_search):
            new_h = h + alpha * delta_h_opt
            new_h = np.clip(new_h, 0, 1)
            total = np.sum(new_h)
            new_h = new_h * (n_points / 2) / total
            c_new = np.correlate(new_h, 1 - new_h, mode='full')
            val = c_new[k0] * dx
            if val < best_val:
                best_val = val
                best_alpha = alpha
        new_h = h + best_alpha * delta_h_opt

        total = np.sum(new_h)
        new_h = new_h * (n_points / 2) / total
        return new_h
