import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)

    initial_guess = initial_h.copy()

    start_time = time.time()

    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 500}
    )

    h_opt = result.x
