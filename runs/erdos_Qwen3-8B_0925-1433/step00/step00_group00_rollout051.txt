import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    population = [h.copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        elite_idx = np.argsort(fitness)[:int(len(population)*0.1)]
        elites = [population[i] for i in elite_idx]
        new_pop = elites.copy()
        while len(new_pop) < len(population):
            idx = np.random.choice(len(elites))
            h = elites[idx].copy()
            delta = np.random.uniform(-0.05, 0.05, size=n_points)
            h_mut = np.clip(h + delta, 0, 1)
            total = np.sum(h_mut)
            scale = (n_points / 2) / total
            h_mut *= scale
            new_pop.append(h_mut)
        new_pop = new_pop[:len(population)]
        top_idx = np.argsort(fitness)[:int(len(population)*0.05)]
        for i in top_idx:
            h = population[i]
            def obj(x):
                return compute_c5(x)
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            result = minimize(obj, h, method='SLSQP', bounds=[(0,1)*n_points], constraints=cons)
            if result.success:
                population[i] = result.x
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        population = new_pop
    
    return (best_h, best_c5, n_points)
