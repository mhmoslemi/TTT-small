    perturbation_amplitude = 0.05
    max_iter = 100
    
    while time.time() - start_time < budget_s:
        iteration_count += 1
        
        # Gradient-based step
        def objective_func(x):
            h = x.reshape(n_points)
            return evaluate_erdos_solution(h, n_points)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Constraints
        def eq_constr(x):
            return np.sum(x) - n_points / 2
        
        result = opt.minimize(
            fun=objective_func,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': eq_constr},
            options={'ftol': 1e-5, 'maxiter': max_iter}
        )
        
        if result.success:
            h_candidate = result.x
            current_c5 = evaluate_erdos_solution(h_candidate, n_points)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
        
        # Perturbation
        if iteration_count % perturbation_interval == 0:
            delta = perturbation_amplitude * np.random.randn(n_points)
            h_pert = best_h + delta
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert_sum = np.sum(h_pert)
            if h_pert_sum > 0:
                h_pert = h_pert * (n_points / 2) / h_pert_sum
            else:
                h_pert = np.zeros(n_points) + 0.5
            current_c5_pert = evaluate_erdos_solution(h_pert, n_points)
            if current_c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = current_c5_pert
        
        # Final adjustments
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - n_points / 2) > 1e-6:
        scaling_factor = (n_points / 2) / best_h_sum
        best_h = best_h * scaling_factor
        best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h.tolist(), best_c5, n_points
