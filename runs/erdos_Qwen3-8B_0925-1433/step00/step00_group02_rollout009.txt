    def select_parent(population, fitness, tournament_size):
        candidates = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = candidates[np.argmin(fitness[candidates])]
        return population[best_idx].copy()

    dx = 2.0 / n_points
    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    elapsed_time = 0
    pop_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_prob = 0.8

    while elapsed_time < budget_s:
        fitness = [evaluate_erdos_solution(h) for h in population]
        new_population = []
        for _ in range(pop_size):
            parent1 = select_parent(population, fitness, tournament_size)
            parent2 = select_parent(population, fitness, tournament_size)
            if np.random.rand() < crossover_prob:
                child = (parent1 + parent2) / 2
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.uniform(-0.1, 0.1)
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                scale = (n_points / 2) / sum_child
                child = child * scale
                child = np.clip(child, 0, 1)
            new_population.append(child)
        population = new_population
        for i in range(len(population)):
            current_c5 = evaluate_erdos_solution(population[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[i]
        elapsed_time = time.time() - start_time
    return best_h, best_c5, n_points
