import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy().astype(float)
    
    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    cons = [{
        'type': 'eq',
        'fun': lambda h: np.sum(h) - n_points / 2
    }]
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={
            'ftol': 1e-9,
            'maxiter': 300,
            'disp': False
        }
    )
    
    if result.success:
        h_values = np.clip(result.x, 0.0, 1.0)
        c5_bound = objective(h_values)
    else:
        h_values = np.clip(h_initial, 0.0, 1.0)
        c5_bound = objective(h_values)
    
    return (h_values, c5_bound, n_points)
