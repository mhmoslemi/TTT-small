import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 80
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        desired_sum = n_points / 2
        current_sum = np.sum(h)
        scaling_factor = desired_sum / current_sum
        h = h * scaling_factor
        # Ensure sum is correct
        h = (h * n_points / np.sum(h)).clip(0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    T = 100.0
    cooling_rate = 0.995
    min_temp = 1e-3
    start_time = time.time()
    
    def compute_c5(h, dx):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    while time.time() - start_time < budget_s:
        i, j = np.random.randint(0, n_points, 2)
        while i == j:
            j = np.random.randint(0, n_points)
        delta = 0.05 * (h[i] - 0.5)
        new_i = np.clip(h[i] + delta, 0, 1)
        new_j = np.clip(h[j] - delta, 0, 1)
        h_new = h.copy()
        h_new[i] = new_i
        h_new[j] = new_j
        sum_change = (new_i - h[i]) + (new_j - h[j])
        if abs(sum_change) > 1e-10:
            k = np.random.randint(0, n_points)
            while k == i or k == j:
                k = np.random.randint(0, n_points)
            h_new[k] += -sum_change
            h_new[k] = np.clip(h_new[k], 0, 1)
        neighbor_c5 = compute_c5(h_new, dx)
        if neighbor_c5 < best_c5 or np.random.rand() < np.exp(-(neighbor_c5 - best_c5) / T):
            h = h_new.copy()
            best_c5 = neighbor_c5
            best_h = h.copy()
        T *= cooling_rate
    return (best_h, best_c5, n_points)
