def run(seed=42, budget_s=600, **kwargs):

        population.append(h_scaled)
    population = np.array(population)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    CR = 0.9

        for _ in range(len(population)):
            idx1, idx2, idx3 = np.random.choice(len(population), 3, replace=False)
            x1, x2, x3 = population[idx1], population[idx2], population[idx3]
            mutant = x1 + F * (x2 - x3)

            if sum_clipped == 0:
                mutant_rescaled = np.random.uniform(0, 1, n_points)

            else:
                mutant_rescaled = mutant_clipped * (n_points/2.0)/sum_clipped
            
            target = top_population[0]
            trial = mutant_rescaled.copy()
            for i in range(n_points):
                if np.random.rand() < CR:
