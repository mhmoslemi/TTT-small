from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    initial_h_values = np.load('initial_h_values.npy')  # Assume initial_h_values is available as a numpy array
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial_h_values
    initial_h_values = np.clip(initial_h_values, 0, 1)

        scale = target_sum / initial_sum
        initial_h_values *= scale
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr[n_points-1:-n_points+1])  # Remove edge effects
        return max_corr * dx
    
    def fitness(h):
        return -compute_c5(h)
    
    num_particles = min(50, n_points // 5)  # Reduce particles for larger n_points
    particles = [initial_h_values.copy() + np.random.normal(0, 0.02, n_points) for _ in range(num_particles)]
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    pbest = [fitness(p) for p in particles]
    gbest_idx = np.argmin(pbest)
    best_h = particles[gbest_idx]
    best_c5 = -pbest[gbest_idx]
    best_time = time.time()
    
    def update_velocity(v, p, g, w, c1, c2, r1, r2):
        v = w * v + c1 * r1 * (p - v) + c2 * r2 * (g - v)
        return np.clip(v, -0.1, 0.1)
    
    params = {'w': 0.9, 'c1': 1.5, 'c2': 1.5}
    time_start = time.time()

            pos = pos * target_sum / np.sum(pos)
            particles[i] = pos
            curr_fitness = fitness(pos)
            if curr_fitness < pbest[i]:
                pbest[i] = curr_fitness
                particles[i] = pos
                if curr_fitness < pbest[gbest_idx]:
                    gbest_idx = i
                    best_h = pos
                    best_c5 = -curr_fitness
        if time.time() - time_start >= budget_s:

            def slsqp_objective(x):
                return -compute_c5(x)
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            bounds = Bounds(0, 1)
