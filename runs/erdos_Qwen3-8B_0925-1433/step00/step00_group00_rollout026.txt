from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    desired_sum = n_points / 2
    initial_h = initial_h_values * (desired_sum / np.sum(initial_h_values))
    population = [initial_h.copy()]
    for _ in range(19):
        h = np.random.uniform(0, 1, n_points)
        h = h * (desired_sum / np.sum(h))
        population.append(h)
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    for _ in range(100):
        fitness = [evaluate(h) for h in population]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()
        new_population = []
        for h in population:
            h_mut = h + np.random.normal(0, 0.01, h.shape)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (desired_sum / np.sum(h_mut))
            new_population.append(h_mut)
        population = new_population
        top_idx = np.argsort(fitness)[:int(len(population)*0.03)]
        for idx in top_idx:
            h = population[idx].copy()

                corr = np.correlate(x, 1 - x, mode='full')
