def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    sum_initial = np.sum(initial_h_values)

    population = [scaled_initial.copy()]
    for _ in range(POP_SIZE - 1):
        candidate = np.random.rand(n_points)
        candidate = candidate * (n_points / 2.0) / np.sum(candidate)
        population.append(candidate)
    
    def smooth_max(corr, eps=0.01):
        return np.log(np.sum(np.exp(eps * corr))) / eps

        corr = np.correlate(h, 1 - h, mode='full')
        return smooth_max(corr)
