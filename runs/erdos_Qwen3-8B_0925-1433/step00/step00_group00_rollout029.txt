    start = time.time()
    omega_start, omega_decay, c1, c2 = 0.9, 0.995, 1.5, 1.5
    
    while time.time() - start < budget_s:
        omega = omega_start * (omega_decay ** 100)
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] + c1 * r1 * (particles[i] - particles[pbest.index(pbest[i])]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = v
            new_p = particles[i] + v
            new_p = np.clip(new_p, 0, 1)
            if np.sum(new_p):
                new_p *= (n_points / 2) / np.sum(new_p)
            particles[i] = new_p
        for i in range(num_particles):
            curr = compute_c5(particles[i])
            if curr < pbest[i]:
                pbest[i] = curr
                if curr < best_c5:
                    best_c5 = curr
                    gbest = particles[i]
        if (time.time() - start) % 100 < 1:
            top = sorted(range(num_particles), key=lambda x: pbest[x])[:int(num_particles * 0.1)]
            for idx in top:
                h = particles[idx]
                def obj(x): return compute_c5(x)
                def cons(x): return np.sum(x) - n_points / 2
                res = minimize(obj, h, method='SLSQP', bounds=Bounds(0, 1), constraints=NonlinearConstraint(cons, 0, 0), options={'maxiter': 25})
                if res.success:
                    particles[idx] = res.x
                    pbest[idx] = compute_c5(res.x)
    
    return (gbest, best_c5, n_points)
