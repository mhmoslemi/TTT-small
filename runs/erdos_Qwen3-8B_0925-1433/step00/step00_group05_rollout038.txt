    for _ in range(max_iterations):

        k0 = np.argmax(c)

        result = opt.linprog(c_coeff, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            delta_h_opt = result.x
        else:
            delta_h_opt = np.zeros(n_points)

        best_alpha = 0.0

        for alpha in np.linspace(0.0, 1.0, line_search_steps):
