import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points

    def objective(x):
        h_vals = x
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    elapsed = time.time() - start_time

    if result.success:
        h_opt = result.x
    else:
        h_opt = h

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(corr) * dx
    return h_opt.tolist(), c5, n_points
