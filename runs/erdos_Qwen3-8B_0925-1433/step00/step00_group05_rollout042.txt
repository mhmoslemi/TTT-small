import numpy as np
import time

initial_h_values = None
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    def enforce_constraints(h, target_sum, n_points):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros(n_points)
        scaling = target_sum / current_sum
        h_scaled = np.clip(h * scaling, 0, 1)
        return h_scaled

    def compute_c5(h):
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = [np.copy(initial_h_values)]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = enforce_constraints(h, target_sum, n_points)
        population.append(h)

    best_h = np.copy(population[0])
    best_c5 = compute_c5(best_h)
    time_start = time.time()

    while time.time() - time_start < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        elite_indices = np.argsort(fitness)[:10]
