import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)

    elapsed_time = 0.0
    perturb_rad = 0.08
    decay = 0.995
    no_imp = 0
    max_iter = int(budget_s / 0.1)

    for _ in range(max_iter):
        if elapsed_time >= budget_s:
            break
        idxs = np.random.choice(n_points, 20, replace=False)
        delta = np.zeros(n_points)
        delta[idxs] = np.random.normal(0, perturb_rad, 20)
        h_pert = h + delta
        h_pert = np.clip(h_pert, 0, 1)
        h_pert_sum = np.sum(h_pert)
        h_pert *= (n_points / 2) / h_pert_sum
        c5_pert = evaluate_erdos_solution(h_pert, n_points)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= 10:
            break
        
        def obj(x):
            h_curr = x.reshape(n_points)
            cor = np.correlate(h_curr, 1-h_curr, mode='full')
            return np.max(cor) * dx

            if c5_opt < best_c5:

        elapsed_time += 0.1

    best_h_sum = np.sum(best_h)
