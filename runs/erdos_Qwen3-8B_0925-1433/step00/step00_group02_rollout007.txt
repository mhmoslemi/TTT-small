        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to maintain sum
        current_sum = np.sum(h_candidate)
        if abs(current_sum) < 1e-6:
            h_candidate = np.random.rand(n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
        else:
            h_candidate = h_candidate * (target_sum / current_sum)
        
        # Evaluate new candidate's C5
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        new_c5 = np.max(corr_candidate)
        
        if new_c5 < current_c5:
            h = h_candidate.copy()
        else:
            # Random perturbation
            h = h + np.random.randn(n_points) * 0.01
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        
        iteration += 1
    
    # Final validation
    corr_best = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_best)
    assert abs(np.sum(best_h) - target_sum) < 1e-5, "Sum constraint violated"
    assert np.all(0 <= best_h <= 1), "Value constraint violated"
    
    return (best_h, c5_bound, n_points)
