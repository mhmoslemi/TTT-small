    while time.time() - start_time < budget_s:
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            j = (j + 1) % n_points
        delta = np.random.uniform(-0.1, 0.1)
        h_new = current_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        neighbor_c5 = compute_C5(h_new)
        
        if neighbor_c5 < current_c5:
            current_h = h_new
            current_c5 = neighbor_c5
        else:
            prob = np.exp(-(neighbor_c5 - current_c5) / T)
            if np.random.rand() < prob:
                current_h = h_new
                current_c5 = neighbor_c5
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        T *= cooling_rate
    
    return best_h, best_c5, n_points
