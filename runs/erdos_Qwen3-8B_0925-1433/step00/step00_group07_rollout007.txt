            c5 = compute_c5(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h1 = h
    h2 = 1 - h
    corr = np.correlate(h1, h2, mode='full')
    return corr.max() * dx
