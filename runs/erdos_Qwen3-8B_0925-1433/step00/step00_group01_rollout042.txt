import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = len(initial_h_values) if initial_h_values is not None else 47
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def rescale(h):
        return h * target_sum / np.sum(h)
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        return rescale(h)
    
    def evaluate_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def cost_func(h):
        return evaluate_overlap(h)
    
    if initial_h_values is not None:
        initial_population = [rescale(initial_h_values)]
    else:
        initial_population = [np.random.rand(n_points) for _ in range(20)]
        initial_population = [rescale(p) for p in initial_population]
    
    population = initial_population
    best_h = min(population, key=cost_func)
    best_score = cost_func(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        if len(population) < 10:
            new_h = np.random.rand(n_points)
            new_h = rescale(new_h)
            population.append(new_h)
            continue
        
        pop_sorted = sorted(population, key=cost_func)
        best = pop_sorted[0]
        if cost_func(best) == best_score:
            continue
        best_score = cost_func(best)
        
        candidates = pop_sorted[:int(len(pop_sorted) * 0.8)]
