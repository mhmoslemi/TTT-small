from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    population = [initial_h.copy()]

        population.append(h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        elites = sorted_pop[:int(len(sorted_pop) * 0.1)]
        
        new_pop = []
        for h in population:
            h_mut = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:

            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)

                scale = (n_points / 2) / sum_mut

        top_parents = sorted_pop[:int(len(sorted_pop) * 0.2)]
        for i in range(0, len(top_parents), 2):
            if i + 1 < len(top_parents):
