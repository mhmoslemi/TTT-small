import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Assume initial_h_values is provided as a global variable
    global initial_h_values
    n_points = len(initial_h_values)

    result = minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        constraints=cons,

    )
