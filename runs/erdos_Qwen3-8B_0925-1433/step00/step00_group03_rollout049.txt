import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        n_points = 100  # fallback to default sample count
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
        target_sum = n_points / 2

        initial_h *= scale
    Dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h.copy()

    def objective(x):
