import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 100
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    assert np.isclose(np.sum(h), n_points / 2)

    def c5(h_):
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = c5(best_h)

    pop_size = 50
    gens = 500
    mut_rate = 0.1

    pop = [h.copy() for _ in range(pop_size)]
    for _ in range(pop_size - 1):
        new_h = np.random.rand(n_points)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_h = np.clip(new_h, 0, 1)
        pop.append(new_h)

    start = time.time()
    for gen in range(gens):
        if time.time() - start > budget_s:
            break
        fit = [c5(ind) for ind in pop]
        sel = []
        for _ in range(pop_size):
            idxs = np.random.choice(pop_size, 5, replace=False)
            sel.append(pop[np.argmin(fit[idxs])])
        new_pop = []
        for i in range(0, pop_size, 2):
            a, b = sel[i], sel[i+1]
            alpha = np.random.rand()
            new_pop.append(alpha * a + (1 - alpha) * b)
            new_pop.append((1 - alpha) * a + alpha * b)
        for i in range(len(new_pop)):
            h_ = new_pop[i]
            for j in range(n_points):
                if np.random.rand() < mut_rate:
                    h_[j] += np.random.normal(0, 0.05)
                    h_[j] = np.clip(h_[j], 0, 1)
            h_sum = np.sum(h_)
            h_ = h_ * (n_points / 2) / h_sum
            new_pop[i] = h_
        pop = new_pop
        for ind in pop:
            if c5(ind) < best_c5:
                best_c5 = c5(ind)
                best_h = ind.copy()
    return (best_h, best_c5, n_points)
