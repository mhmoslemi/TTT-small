                if current_sum == 0:
                    child = np.random.rand(n_points)
                else:
                    scaling = desired_sum / current_sum
                    child = child * scaling
                    child = np.clip(child, 0, 1)
            new_pop.append(child)
        population = new_pop
        if evaluations[0][0] < best_c5:
            best_c5 = evaluations[0][0]
            best_h = evaluations[0][1]
    return (best_h, best_c5, n_points)
