    h_initial = np.clip(h_initial * scale, 0.0, 1.0)

    def objective(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        scale = (n_points / 2) / sum_h
        h_scaled = h_clipped * scale
        one_minus_h = 1.0 - h_scaled
        corr = np.correlate(h_scaled, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with optimized parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False
    )

    h_optimal = result.x
    c5_value = objective(h_optimal)

    return tuple(h_optimal), c5_value, n_points
