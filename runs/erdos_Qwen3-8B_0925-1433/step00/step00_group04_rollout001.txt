import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n_points = 500
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0
        if n_points % 2 != 0:
            h_initial[0] = 0.5
    h = h_initial
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient(x):
        h = x.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + lag
            if 0 <= j_plus < n_points:
                grad[i] += (1.0 - h[j_plus])
            j_minus = i - lag
            if 0 <= j_minus < n_points:
                grad[i] -= h[j_minus]
        grad *= dx
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        jac=gradient,
        options={'maxiter': 100, 'ftol': 1e-7, 'disp': False}
    )
    
    optimized_h = result.x
    corr = np.correlate(optimized_h, 1.0 - optimized_h, mode='full')
    c5 = np.max(corr) * dx
