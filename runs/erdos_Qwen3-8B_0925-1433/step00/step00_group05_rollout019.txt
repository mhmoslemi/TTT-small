    desired_sum = n_points / 2
    h_initial_sum = np.sum(h_initial)
    scale_factor = desired_sum / h_initial_sum
    h_initial = np.clip(h_initial * scale_factor, 0, 1)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial_sum = np.sum(h_initial)
    if h_initial_sum < 1e-6:
        h_initial = np.full(n_points, desired_sum / n_points)
    else:
        scale_factor = desired_sum / h_initial_sum
        h_initial = h_initial * scale_factor
        h_initial = np.clip(h_initial, 0, 1)
    population = [h_initial]
    for _ in range(49):
        rand = np.random.rand(n_points)
        sum_rand = np.sum(rand)
        scale = desired_sum / sum_rand
        scaled = rand * scale
        clipped = np.clip(scaled, 0, 1)
        population.append(clipped)
    best_h = h_initial
    best_c5 = 0.0
    np.random.seed(seed)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            conv = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            fitness.append(c5)
        elites_idx = np.argsort(fitness)[:10]
        elites = [population[i] for i in elites_idx]
        new_population = elites.copy()
        for _ in range(40):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            mut = np.random.normal(0, 0.05, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child < 1e-6:
                child = np.full(n_points, desired_sum / n_points)
            else:
                scale = desired_sum / sum_child
                child = child * scale
            child = np.clip(child, 0, 1)
            new_population.append(child)
        population = new_population
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
    conv = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(conv) * dx
    return best_h, final_c5, n_points
