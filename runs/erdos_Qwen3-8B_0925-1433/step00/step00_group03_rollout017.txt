def run(seed=42, budget_s=600, **kwargs):

        curr_c5 = curr_corr.max()
        if curr_c5 < best_c5:

        # Gradient descent
        eps = 1e-5
        grad = np.zeros_like(h)

            h_plus[i] += eps
