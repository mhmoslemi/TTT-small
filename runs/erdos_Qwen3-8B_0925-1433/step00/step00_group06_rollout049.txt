import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    h_initial = initial_h_values

    start_time = time.time()
    result = opt.minimize(

        constraints=[{'type': 'eq', 'fun': constraint_sum}],
