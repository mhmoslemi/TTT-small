        sum_rand = np.sum(rand_h)
        scaling_rand = (n / 2) / sum_rand
        population.append(rand_h * scaling_rand)
    
    best_h = h_initial_scaled.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for iteration in range(200):
        evaluated = []
        for h in population:
            c5_val = evaluate_erdos_solution(h)
            evaluated.append((c5_val, h))
        
        evaluated.sort()
        top_30 = evaluated[:int(0.3 * len(evaluated))]
        
        new_population = []
        for _ in range(POP_SIZE):
            if len(top_30) >= 3:
                parents = random.sample(top_30, 3)
                h1, h2, h3 = [h for (c, h) in parents]
                F = 0.8
                mutant = h1 + F * (h2 - h3)
            else:
                best_h = top_30[0][1] if top_30 else h_initial_scaled.copy()
                mutant = best_h.copy()
                noise = np.random.rand(n) * 0.1
                mutant += noise
            
            mutant = np.clip(mutant, 0, 1)
            sum_mutant = np.sum(mutant)
            scaling_mutant = (n / 2) / sum_mutant
            mutant = mutant * scaling_mutant
            new_population.append(mutant)
        
        population = new_population
        
        current_best = min(evaluated, key=lambda x: x[0])
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1]
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_clipped = np.sum(best_h_clipped)
    scaling_final = (n / 2) / sum_clipped
    best_h_final = best_h_clipped * scaling_final
    
    best_c5 = evaluate_erdos_solution(best_h_final)
    
    return (best_h_final, best_c5, n)
