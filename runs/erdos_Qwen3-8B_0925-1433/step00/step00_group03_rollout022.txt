def run(seed=42, budget_s=600, **kwargs):

    from scipy.optimize import Bounds, minimize

    h = initial_h_values.copy()

        delta = target_sum - total
        h[-1] += delta
        h = np.clip(h, 0, 1)
        return h
    
    def objective(h_vec):
        h = h_vec.copy()
        h = adjust_sum(h)
        return compute_c5(h)

        h = h_vec.copy()
        h = adjust_sum(h)
        return np.sum(h) - target_sum
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    initial_guess = h.copy()
    
    result = minimize(

        tol=1e-6,
