def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = len(h)

    T = 100.0
    cooling_rate = 0.995
    min_temp = 1e-3
    start_time = time.time()
    
    while T > min_temp:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        i, j = np.random.choice(n_points, size=2, replace=False)
        delta = 0.05 * (h[i] - 0.5)
        new_i = h[i] + delta
        new_i = np.clip(new_i, 0.0, 1.0)
        delta_i = new_i - h[i]
        new_j = h[j] - delta_i
        new_j = np.clip(new_j, 0.0, 1.0)
        new_h = h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        new_c5 = compute_c5(new_h, dx)
        
        if new_c5 < current_c5:

            current_c5 = new_c5

            best_h = h.copy()
        
        T *= cooling_rate
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
