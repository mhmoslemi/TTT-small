import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})
    
    if result.success:
        best_h = result.x
        c5_bound = objective(best_h)
    else:
        best_h = initial_h
        c5_bound = objective(best_h)
    
    return best_h, c5_bound, n_points
