import numpy as np
import numpy.random as npr
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0.0
    perturbation_radius = 0.08
    decay_rate = 0.995
    while elapsed_time < budget_s:
        start_time = time()
        perturb_indices = npr.choice(n_points, size=20, replace=False)
        delta = np.zeros(n_points)
        delta[perturb_indices] = npr.normal(0, perturbation_radius, 20)
        h_pert = h + delta
        h_pert = np.clip(h_pert, 0.0, 1.0)
        scale_factor = (n_points / 2.0) / np.sum(h_pert)
        h_pert *= scale_factor
        c5_pert = evaluate_erdos_solution(h_pert, n_points)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        def objective(x):
            h_opt = x.reshape(n_points)
            cor = np.correlate(h_opt, 1.0 - h_opt, mode='full')
            return np.max(cor) * dx
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = so.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-5, 'maxiter': 30})
        if result.success:
            best_h = result.x
            best_c5 = evaluate_erdos_solution(best_h, n_points)
        elapsed_time += time() - start_time
        perturbation_radius *= decay_rate
        if perturbation_radius < 1e-5:
            break
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
