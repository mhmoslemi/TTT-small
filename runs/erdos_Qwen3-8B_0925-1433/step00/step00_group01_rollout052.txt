import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None

    if initial_h_values is None:
        h = np.random.rand(n_points) * (n_points / 2.0) / np.sum(np.random.rand(n_points))
    else:
        h = initial_h_values * (n_points / 2.0) / np.sum(initial_h_values)

    def objective(h_vec):
        h = h_vec.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def smooth_max(x, eps=0.01):
        return np.log(np.sum(np.exp(eps * x))) / eps

    def evaluate(h_vec):
        h = h_vec.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        obj = smooth_max(corr)
        return obj

    def rescale(h):
        s = np.sum(h) / (n_points / 2.0)
        return h / s

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        return rescale(h)

    def perturb(h, F=0.8, CR=0.9):
        idx = np.random.choice(n_points, 3, replace=False)
        v = h[idx[0]] + F * (h[idx[1]] - h[idx[2]])
        v += np.random.normal(0, 0.01, n_points)
        return clip_and_rescale(v)

    population = [clip_and_rescale(np.random.rand(n_points) * (n_points / 2.0) / np.sum(np.random.rand(n_points))) for _ in range(25)]
    best_h = clip_and_rescale(h)
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        elites = sorted(population, key=evaluate)[:5]
        for _ in range(20):
            if time.time() - start_time > budget_s:
                break
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = perturb(a)
            trial = np.copy(elites[0])
            for i in range(n_points):
                if np.random.rand() < CR:
                    trial[i] = mutant[i]
            trial = clip_and_rescale(trial)
            if evaluate(trial) < evaluate(elites[0]):
                elites[0] = trial
        population = elites + [np.copy(population[-1]) for _ in range(25 - len(elites))]
        for h in population:
            if evaluate(h) < best_c5:
                best_c5 = evaluate(h)
                best_h = h

    def grad_step(h, step=0.05):
        h = h.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        grads = np.zeros_like(h)
        for i in range(n_points):
            h[i] += step
            h_new = clip_and_rescale(h)
            grads[i] = (evaluate(h_new) - evaluate(h)) / step
            h[i] -= step
        h += grads * 0.1
        return clip_and_rescale(h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for i in range(25):
            h = population[i]
            h = grad_step(h)
            population[i] = h
            if evaluate(h) < best_c5:
                best_c5 = evaluate(h)
                best_h = h

    best_h = clip_and_rescale(best_h)
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
