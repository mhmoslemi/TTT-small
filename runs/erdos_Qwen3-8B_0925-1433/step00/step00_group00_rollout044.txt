import numpy as np
import numpy.random as rnd
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        n_points = 200
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
        s = np.sum(initial_h) * 2 / n_points
        initial_h *= n_points / (2 * s)
    n_points, dx = len(initial_h), 2.0 / n_points
    sum_h = np.sum(initial_h)
    initial_h *= (n_points / 2) / sum_h if sum_h else 0.5

    num_particles = 50
    particles = [initial_h.copy() for _ in range(num_particles)]
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    pbest = [0.0]*num_particles
    gbest = float('inf')

    for i in range(num_particles):
        pbest[i] = correlate(particles[i], 1 - particles[i], mode='full').max() * dx
    gbest_idx = np.argmin(pbest)

    max_iter = int(budget_s / 5)
    omega, c1, c2 = 0.9, 1.5, 1.5
    for _ in range(max_iter):
        for i in range(num_particles):
            r1, r2 = rnd.rand(), rnd.rand()
            velocities[i] = omega * velocities[i] + c1*r1*(particles[i] - particles[gbest_idx]) + c2*r2*(particles[gbest_idx] - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = np.sum(particles[i])
            if s:
                particles[i] *= (n_points / 2) / s
            else:
                particles[i] = np.full(n_points, 0.5)
            corr = correlate(particles[i], 1 - particles[i], mode='full').max() * dx
            if corr < pbest[i]:
                pbest[i] = corr
                if corr < gbest:
                    gbest = corr
                    gbest_idx = i

        top = sorted(range(num_particles), key=lambda i: pbest[i])[:10]
        for i in top:
            def obj(x):
                return correlate(x, 1 - x, mode='full').max() * dx
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1)]*n_points
            res = minimize(obj, particles[i], method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                particles[i] = res.x

        if all(abs(pbest[i] - gbest) < 1e-6 for i in range(num_particles)):
            break

    best = particles[gbest_idx]
    return (best, gbest, n_points)
