import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    if "initial_h_values" in globals():
        h = initial_h_values.copy()
    else:
        n_points = 600
        h = np.random.rand(n_points)
    n_points = len(h)
    s = np.sum(h)
    scale = (n_points / 2) / s
    h = np.clip(h * scale, 0, 1)
    s = np.sum(h)
    scale = (n_points / 2) / s
    h = np.clip(h * scale, 0, 1)
    dx = 2.0 / n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c = corr * dx
        return np.log(np.mean(np.exp(c)))

    bounds = [(0., 1.) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    start = time.time()
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300})
    elapsed = time.time() - start

    h_opt = res.x
    h_opt = np.clip(h_opt, 0, 1)
    s = np.sum(h_opt)
    scale = (n_points / 2) / s
    h_opt = h_opt * scale
    h_opt = np.clip(h_opt, 0, 1)

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(corr * dx)

    if elapsed > budget_s:
        return (h_opt, c5, n_points)
    return (h_opt, c5, n_points)
