import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        delta = (n_points / 2) - sum_h
        h[-1] += delta
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            raise ValueError("Normalization failed")

    best_c5 = float('inf')
    best_h = h.copy()
    learning_rate = 0.01
    epsilon = 1e-5
    max_iterations = 300
    start_time = time.time()

    for iteration in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        current_max_overlap = np.max(corr) * dx
        if current_max_overlap < best_c5:
            best_c5 = current_max_overlap
            best_h = h.copy()
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturbed = h.copy()
            h_perturbed[i] += epsilon
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            new_overlap = corr_perturbed[k_max] * dx
            grad[i] = (new_overlap - current_max_overlap) / (2 * epsilon)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            delta = (n_points / 2) - sum_h
            h[-1] += delta
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h, final_c5, n_points)
