import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in kwargs:
        initial_h = np.array(kwargs['initial_h_values'])
    else:
        initial_h = np.random.rand(47)
        initial_h = np.clip(initial_h, 0, 1)
        initial_sum = np.sum(initial_h)
        scaling = 47 / 2 / initial_sum
        initial_h = initial_h * scaling
    n_points = len(initial_h)
    initial_sum = np.sum(initial_h)
    scaling = n_points / 2 / initial_sum
    initial_h = initial_h * scaling
    initial_h = np.clip(initial_h, 0, 1)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    population = [initial_h.copy()]
    for _ in range(49):
        candidate = np.random.rand(n_points)
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        population.append(candidate)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        eval_list = []
        for h in population:
            c5 = evaluate(h)
            eval_list.append((c5, h))
        eval_list.sort()
        elites = eval_list[:5]
        best_c5 = elites[0][0]
        best_h = elites[0][1]
        
        new_pop = []
        for _ in range(len(population) - len(elites)):
            h1, h2, h3 = random.sample(elites, 3)
            mutant = h1[0] + 0.8 * (h2[1] - h3[1])

            mutant = mutant * (n_points / 2) / np.sum(mutant)
            new_pop.append(mutant)
        
        population = [h for c5, h in elites] + new_pop
        for h in population:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return best_h, best_c5, n_points
