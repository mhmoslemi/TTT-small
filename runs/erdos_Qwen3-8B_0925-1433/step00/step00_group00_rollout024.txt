```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    if initial_h_values is None:
        raise ValueError("initial_h_values not provided")
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def obj(h):
        cross = np.correlate(h, 1 - h, mode="full")
        return np.max(cross) * dx
    
    def eq_cons(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': eq_cons}]
    
    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = obj(h_initial)
    
    try:
        result = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            curr_h = result.x
            curr_c5 = obj(curr_h)
            if curr_c5 < best_c5:
                best_h = curr_h
                best_c5 = curr_c5
    finally:
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            return (best_h, best_c5, n_points)
        else:
            return (best_h, best_c5, n_points)
```
