            def obj(x):
                return -compute_c5(x, dx)
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - (n_points / 2)}
            try:
                res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 25})
                if res.success:
                    particles[idx] = res.x
            except:
                pass
    best_idx = np.argmin(pbest_fitness)
    return pbest[best_idx], gbest_fitness, n_points
