        lag = max_idx - (n_points - 1)
        
        # Compute gradient for this lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus_lag = i + lag
            if 0 <= i_plus_lag < n_points:
                grad[i] = (1 - h[i_plus_lag]) * dx
        
        # Update h and clip to [0, 1]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain integral constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        
        # Decay learning rate
        learning_rate *= learning_rate_decay
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx
