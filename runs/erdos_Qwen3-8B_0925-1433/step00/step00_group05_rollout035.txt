                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                scale = desired_sum / child_sum
                child = np.clip(child * scale, 0, 1)
            new_pop.append(child)
        population = new_pop
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1]
    return (best_h, best_c5, n_points)
