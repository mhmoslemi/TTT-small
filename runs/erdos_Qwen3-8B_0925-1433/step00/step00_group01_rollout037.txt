import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600):
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2.0) / initial_sum if initial_sum != 0 else 1.0
    h = initial_h_values * scale_factor

    POP_SIZE = 25
    F = 0.8
    CR = 0.9
    epsilon = 0.01
    time_start = time.time()
    population = [h.copy() for _ in range(POP_SIZE)]
    best_h = h.copy()
    best_value = float('inf')

    def evaluate_func(candidate):
        corr = np.correlate(candidate, 1 - candidate, mode='full')
        smooth_max = (1 / epsilon) * np.log(np.sum(np.exp(epsilon * corr)))
        return smooth_max

    for _ in range(300):
        elapsed = time.time() - time_start
        if elapsed >= budget_s:
            break
        values = [evaluate_func(c) for c in population]
        best_idx = np.argmin(values)
        best_candidate = population[best_idx]
        best_val = values[best_idx]

        for _ in range(POP_SIZE):
            idxs = np.random.choice(len(population), 3, replace=False)
            x1, x2, x3 = [population[i] for i in idxs]
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            if np.sum(mutant) == 0:
                mutant = np.zeros_like(mutant)
            else:
                mutant *= (n_points / 2) / np.sum(mutant)
            trial = np.copy(best_candidate)
            for i in range(n_points):
                if np.random.rand() < CR:
                    trial[i] = mutant[i]
            corr = np.correlate(trial, 1 - trial, mode='full')
            smooth = (1 / epsilon) * np.log(np.sum(np.exp(epsilon * corr)))
            if smooth < best_val:
                worst_idx = np.argmax(values)
                population[worst_idx] = trial
                values[worst_idx] = smooth
                if smooth < best_val:
                    best_val = smooth
                    best_h = trial.copy()

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        smooth = (1 / epsilon) * np.log(np.sum(np.exp(epsilon * corr)))
        return smooth

    bounds = Bounds(0, 1)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    refined_h = res.x

    refined_h = np.clip(refined_h, 0, 1)
    refined_sum = np.sum(refined_h)
    if refined_sum != 0:

    corr_val = np.correlate(refined_h, 1 - refined_h, mode='full')

    return (refined_h, c5_bound, n_points)
