            h_plus = np.clip(h_plus, 0, 1)
            sum_plus = np.sum(h_plus)
            if sum_plus != 0:
                h_plus *= (n_points / 2) / sum_plus
            
            h_minus = np.clip(h_minus, 0, 1)
            sum_minus = np.sum(h_minus)
            if sum_minus != 0:
                h_minus *= (n_points / 2) / sum_minus
            
            # Compute overlap for lag m for both perturbed h
            # Using the compute_overlap function
            overlap_plus = compute_overlap(h_plus, m, dx, n_points)
            overlap_minus = compute_overlap(h_minus, m, dx, n_points)
            
            # Estimate gradient
            grad_i = (overlap_plus - overlap_minus) / (2 * perturbation)
            gradient[i] = grad_i
        
        # Update h
        h -= learning_rate * gradient
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= (n_points / 2) / sum_h
        
        # Check time
        if time.time() - time_start > budget_s:
            break
    
    return (best_h, best_c5, n_points)
