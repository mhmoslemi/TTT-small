        initial_h_values_scaled = initial_h_values * (S / sum_initial)
    
    def objective(h):
        sum_h = np.sum(h)
        penalty = 1e6 * (sum_h - S) ** 2
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution with constraints and penalties
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=25,
        maxiter=500,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        x0=initial_h_values_scaled
    )
    
    best_h = result.x
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_clipped = np.sum(best_h_clipped)
    
    if sum_clipped == 0:
        scale_factor = 0
    else:
        scale_factor = S / sum_clipped
    best_h_final = best_h_clipped * scale_factor
    
    # Recompute C5 with final solution
    corr = np.correlate(best_h_final, 1 - best_h_final, mode='full')
    c5_bound = np.max(corr) * dx
    
    return best_h_final, c5_bound, n_points
