import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Predefined initial_h_values (example placeholder, actual input is handled via global)
    initial_h = [0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5]  # Placeholder
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Validate initial sum
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2, atol=1e-7):
        raise ValueError("Initial h_values does not satisfy integral constraint")
    
    # Objective function to minimize max correlation
    def obj(h_flat):
        h = h_flat.reshape(n_points)
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraints
    def eq_cons(h_flat):
        return np.sum(h_flat.reshape(n_points)) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': eq_cons}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization settings
    options = {
        'maxiter': 1000,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False,
        'iprint': -1
    }
    
    start_time = time.time()
    result = minimize(
        fun=obj,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options=options
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 10:
        return (np.array(initial_h), np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx, n_points)
    
    if result.success:
        h_opt = result.x
        corr_final = np.correlate(h_opt.reshape(n_points), 1 - h_opt.reshape(n_points), mode='full')
        c5_final = np.max(corr_final) * dx
        return (h_opt.reshape(n_points), c5_final, n_points)
    else:
        # Fallback to initial values if optimization fails
        corr_initial = np.correlate(initial_h, 1 - initial_h, mode='full')
        c5_initial = np.max(corr_initial) * dx
        return (np.array(initial_h), c5_initial, n_points)
