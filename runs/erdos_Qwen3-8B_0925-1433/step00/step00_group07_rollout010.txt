    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2):
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = (n_points / 2 - h_sum) / 2
        h[i] += delta
        h[j] -= delta
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    T = 100.0
    cooling_rate = 0.995
    start_time = time.time()
    for _ in range(10000):
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = 0.05 * (best_h[i] - 0.5)
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0, 1)
        sum_diff = np.sum(new_h) - n_points / 2
        if sum_diff != 0:
            k = np.random.randint(n_points)
            new_h[k] += sum_diff
            new_h[k] = np.clip(new_h[k], 0, 1)
        new_c5 = compute_c5(new_h, n_points)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
        else:
            prob = np.exp(- (new_c5 - best_c5) / T)
            if np.random.rand() < prob:
                best_h, best_c5 = new_h, new_c5
        T *= cooling_rate
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
