import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values)
    sum_h = np.sum(h)
    h *= (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun

    for _ in range(3):
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            new_h *= (n_points / 2) / new_sum
        result = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'ftol': 1e-8, 'maxiter': 100})
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if final_sum != 0:
        best_h *= (n_points / 2) / final_sum
    return best_h.tolist(), best_c5, n_points
