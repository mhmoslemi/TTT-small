        valid_j = j[mask]
        grad = np.zeros(n_points)
        grad[valid_i] += (1 - h_values[valid_j]) * dx
        grad[valid_j] += -h_values[valid_i] * dx
        
        h_candidate = h_values - step_size * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        total_candidate = np.sum(h_candidate)
        if total_candidate == 0:
            h_candidate = np.zeros(n_points)
        else:
            scale = (n_points / 2) / total_candidate
            h_candidate *= scale
        
        new_overlaps = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        new_c5 = np.max(new_overlaps)
        if new_c5 < current_c5:
            h_values = h_candidate.copy()
        else:
            perturbation = perturbation_range * npr.normal(size=n_points)
            h_candidate = h_candidate + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            if total_candidate == 0:
                h_candidate = np.zeros(n_points)
            else:
                scale = (n_points / 2) / total_candidate
                h_candidate *= scale
            new_overlaps = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
            new_c5 = np.max(new_overlaps)
            if new_c5 < current_c5:
                h_values = h_candidate.copy()
        if abs(new_c5 - current_c5) < convergence_threshold:
            break
    
    final_overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_overlaps)
    return (best_h, final_c5, n_points)
