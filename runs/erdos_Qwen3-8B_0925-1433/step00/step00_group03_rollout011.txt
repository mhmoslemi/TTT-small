import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_initial = initial_h_values.copy()

        sum_initial = np.sum(h_initial)
        scale = (n_points / 2) / sum_initial
        h_initial *= scale
    else:
        n_points = 200

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = opt.minimize(
