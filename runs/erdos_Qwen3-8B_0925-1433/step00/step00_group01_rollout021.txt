import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    n_points = 47
    dx = 2.0 / n_points
    
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    def smooth_max(x, eps=1e-3):
        return np.log(np.sum(np.exp(eps * x))) / eps
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-5:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[1:-1]  # Remove out-of-range lags
        return smooth_max(corr)
    
    def evaluate_c5(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-5:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[1:-1]  # Remove out-of-range lags
        return np.max(corr) * dx
    
    def gradient_step(h_flat, step_size=0.05):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-5:
            return h
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[1:-1]
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + max_idx < n_points:
                grad[i + max_idx] -= h[i]
            if i - max_idx >= 0:
                grad[i - max_idx] += (1 - h[i])
        grad = grad * step_size
        h += grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        return h
    
    def optimize():

        for _ in range(100):
