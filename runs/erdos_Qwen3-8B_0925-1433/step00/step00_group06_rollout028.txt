import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    best_h = h.copy()
    best_c5 = scipy.signal.correlate(h, 1 - h, mode='full')[-1] * dx
    best_c5 = np.max(best_c5)
    start_time = time.time()
    elapsed_time = 0
    iter_count = 0
    for _ in range(10000):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        perturb_indices = np.random.choice(n_points, 10, replace=False)
        noise = np.random.normal(0, 0.05, n_points)
        h_pert = h + noise
        h_pert = np.clip(h_pert, 0, 1)
        sum_pert = np.sum(h_pert)
        if sum_pert == 0:
            h = h_pert.copy()
            continue
        scale_factor = (n_points / 2) / sum_pert
        h_pert *= scale_factor
        c5 = scipy.signal.correlate(h_pert, 1 - h_pert, mode='full')[-1] * dx
        c5 = np.max(c5)
        if c5 < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5
        h = h_pert.copy()
        iter_count += 1
    return (best_h.tolist(), best_c5, n_points)
