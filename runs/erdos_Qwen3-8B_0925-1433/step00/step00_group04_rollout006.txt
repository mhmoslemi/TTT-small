import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        sum_h = np.sum(h)
        while np.abs(sum_h - n_points / 2) > 1e-6:
            h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
            sum_h = np.sum(h)
    else:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        sum_h = np.sum(h)
        while np.abs(sum_h - n_points / 2) > 1e-6:
            h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
            sum_h = np.sum(h)

    def objective(h_vec):
        h_array = h_vec
        cross_corr = np.correlate(h_array, 1 - h_array, mode='full')
        c_k = cross_corr * dx
        return np.log(np.mean(np.exp(c_k)))

    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=200)

    elapsed = time.time() - start_time
    if elapsed > budget_s - 30:
        return (h, np.max(np.correlate(h, 1 - h, mode='full') * dx), n_points)

    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    sum_h_opt = np.sum(h_opt)
    scale_opt = (n_points / 2) / sum_h_opt
    h_opt = np.clip(h_opt * scale_opt, 0, 1)
    h_opt = np.clip(h_opt, 0, 1)
    sum_h_opt = np.sum(h_opt)
    while np.abs(sum_h_opt - n_points / 2) > 1e-6:
        h_opt = np.clip(h_opt * (n_points / 2) / sum_h_opt, 0, 1)
        sum_h_opt = np.sum(h_opt)

    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
    return (h_opt, c5_bound, n_points)
