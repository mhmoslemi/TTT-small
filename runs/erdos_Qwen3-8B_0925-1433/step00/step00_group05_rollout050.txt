import numpy as np
from numpy.random import normal, choice
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Check if initial_h_values is available
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 500 if initial_h is None else len(initial_h)

    def normalize_h(h):
        desired_sum = n_points / 2.0
        current_sum = np.sum(h)
        if np.isclose(current_sum, 0):
            return np.zeros_like(h)
        scaling = desired_sum / current_sum
        scaled = h * scaling
        clipped = np.clip(scaled, 0, 1.0)
        if np.any(clipped != scaled):
            new_sum = np.sum(clipped)
            if np.isclose(new_sum, 0):
                return np.zeros_like(clipped)
            scaling2 = desired_sum / new_sum
            clipped = np.clip(clipped * scaling2, 0, 1.0)
        return clipped
    
    if initial_h is not None:
        initial_h = normalize_h(initial_h)
    else:
        # Generate random initial_h with n_points=500
        initial_h = np.random.rand(n_points)
        initial_h = normalize_h(initial_h)

        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)

        population.append(h)
    population = np.array(population)
