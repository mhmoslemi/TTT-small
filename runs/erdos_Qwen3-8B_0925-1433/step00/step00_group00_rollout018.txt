import numpy as np
import random
import time
import scipy.optimize as opt
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def rescale(h):
        s = np.sum(h)
        return h * (n_points / 2) / s if s != 0 else h

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, rate=0.05):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                h_mut[i] += np.random.uniform(-0.05, 0.05)
                h_mut[i] = np.clip(h_mut[i], 0, 1)
        return rescale(h_mut)

    def crossover(p1, p2):
        p = np.random.randint(1, n_points - 1)
        child = np.concatenate([p1[:p], p2[p:]])
        return rescale(child)

    def local_refine(h, max_iter=50):
        def obj(x):
            return evaluate_c5(x)
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = opt.minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter})
        return result.x

    pop = [rescale(initial_h_values)]
    for _ in range(19):
        h = np.random.uniform(0, 1, n_points)
        pop.append(rescale(h))

    best_h = pop[0]
    best_c5 = evaluate_c5(best_h)
    start = time.time()

    while time.time() - start < budget_s:
        fits = [evaluate_c5(h) for h in pop]
        if np.min(fits) < best_c5:
            best_c5 = np.min(fits)
            best_h = pop[np.argmin(fits)]

        elites = [pop[i] for i in np.argsort(fits)[:int(len(pop)*0.1)]]
        new_pop = elites.copy()
        while len(new_pop) < len(pop):
            p1, p2 = random.sample(elites + pop, 2)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        pop = new_pop

        for i in np.argsort(fits)[:int(len(pop)*0.03)]:
            h = pop[i]
            h = local_refine(h)
            pop[i] = h

        if time.time() - start > budget_s - 5:
            break

    return (rescale(best_h), best_c5, n_points)
