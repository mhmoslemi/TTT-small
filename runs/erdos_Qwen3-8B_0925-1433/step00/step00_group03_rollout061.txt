import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h_values = np.random.rand(n_points)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    initial_h_values /= np.sum(initial_h_values)
    initial_h_values *= sum_h
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    max_iter = 1000
    learning_rate = 0.01
    eps = 1e-5
    
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = corr * dx
        k_max = np.argmax(c5)
        if c5[k_max] < best_c5:
            best_c5 = c5[k_max]
            best_h = h.copy()
        
        # Numerical gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus[-1] = np.clip(sum_h - np.sum(h_plus[:-1]), 0, 1)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus[-1] = np.clip(sum_h - np.sum(h_minus[:-1]), 0, 1)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[k_max] - corr_minus[k_max]) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h[-1] = np.clip(sum_h - np.sum(h[:-1]), 0, 1)
    
    # Final evaluation
    corr_final = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = corr_final * dx
    best_c5 = np.max(final_c5)
    
    return best_h, best_c5, n_points
