import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    
    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    for _ in range(5):
        perturb = np.random.normal(0, 0.04, size=n_points)
        current_h = best_h + perturb
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h / np.sum(current_h) * (n_points / 2)
        
        result = opt.minimize(
            objective, current_h, method='SLSQP',
            bounds=bounds, constraints=cons,
