import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    total = np.sum(h)
    delta = (n_points / 2) - total
    h[-1] += delta
    h = np.clip(h, 0, 1)
    b_h = h.copy()
    b_c5 = compute_c5(h, dx)
    T = 100.0
    cooling_rate = 0.995
    min_temp = 1e-3
    start_time = time.time()
    no_improvement = 0
    while time.time() - start_time < budget_s:
        i, j = random.sample(range(n_points), 2)
        delta_i = 0.05 * (h[i] - 0.5)
        new_i = np.clip(h[i] + delta_i, 0, 1)
        delta_adj = delta_i - (new_i - h[i])
        new_j = np.clip(h[j] - delta_adj, 0, 1)
        h_new = h.copy()
        h_new[i] = new_i
        h_new[j] = new_j
        total_new = np.sum(h_new)
        delta_sum = (n_points / 2) - total_new
        h_new[-1] += delta_sum
        h_new = np.clip(h_new, 0, 1)
        c5_new = compute_c5(h_new, dx)

        if no_improvement > 100:
            break
        T *= cooling_rate
        if T < min_temp:
            break
    return (b_h, b_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
