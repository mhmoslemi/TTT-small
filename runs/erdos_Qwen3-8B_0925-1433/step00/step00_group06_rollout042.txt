import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    scale = (n_points / 2) / np.sum(initial_h_values)
    h = np.clip(initial_h_values * scale, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()
    iter_count = 0
    prev_c5 = np.inf
    
    while (time.time() - start_time) < budget_s:
        def objective(x):
            cor = np.correlate(x, 1 - x, mode='full')
            return np.max(cor) * dx
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, maxiter=100)
        if res.success:
            current_c5 = evaluate_erdos_solution(res.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        if iter_count % 50 == 0:
            delta = 0.05 * np.random.randn(n_points)
            pert = np.clip(best_h + delta, 0.0, 1.0)
            pert_sum = np.sum(pert)
            pert = pert * (n_points / 2) / pert_sum
            pert_c5 = evaluate_erdos_solution(pert, n_points)
            if pert_c5 < best_c5:
                best_c5 = pert_c5
                best_h = pert
        iter_count += 1
        if abs(best_c5 - prev_c5) < 1e-7:
            break
        prev_c5 = best_c5
    
    final_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    return (final_h.tolist(), final_c5, n_points)
