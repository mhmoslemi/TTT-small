                new_pop.append(child2)
            else:
                new_pop.append(top_h[i])

        # Local optimization
        top_candidates = population[np.argsort(fitness)[:int(0.03 * pop_size)]]
        for h_candidate in top_candidates:
            def obj(x):
                h = np.clip(x, 0, 1)
                s = np.sum(h)
                h = h * target_sum / s
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            res = minimize(obj, h_candidate, bounds=[(0,1)]*n_points, constraints=cons, method='SLSQP', tol=1e-5)
            if res.success:
                h_candidate[:] = res.x[:]

        population = new_pop

    h_final = np.clip(best_h, 0, 1)
    s = np.sum(h_final)
    h_final = h_final * target_sum / s
    h_final = np.clip(h_final, 0, 1)
    dx = 2.0 / n_points
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(corr) * dx
    return (h_final, c5_final, n_points)
