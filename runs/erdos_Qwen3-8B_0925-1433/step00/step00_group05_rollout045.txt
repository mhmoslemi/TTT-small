import numpy as np
import random
import math
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Fallback: create a baseline pattern
        n_points = 70
        h = np.zeros(n_points)
        h[::3] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_c5(h_vec):
        c = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_c = np.max(c)
        return max_c * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Simulated annealing parameters
    T = 1.0
    T_min = 0.01
    cooling_rate = 0.995
    max_iter = 1000
    time_budget = budget_s
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > time_budget:
            break
        new_h = h.copy()
        # Select up to 5 indices to perturb
        idx = np.random.choice(n_points, size=min(5, n_points), replace=False)
        for i in idx:
            delta = np.random.uniform(-0.05, 0.05)
            new_h[i] += delta
            new_h[i] = np.clip(new_h[i], 0.0, 1.0)
        # Normalize to maintain constraint
        new_sum = np.sum(new_h)
        scale = (n_points / 2) / new_sum
        new_h *= scale
        # Compute new c5
        new_c5 = compute_c5(new_h)
        # Acceptance probability
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            T = 1.0  # Reset temperature
        else:
            prob = math.exp((best_c5 - new_c5) / T)
            if random.random() < prob:
                h = new_h
                best_c5 = new_c5
        T *= cooling_rate
    
    # Final check
    final_c5 = compute_c5(h)
    return (h, final_c5, n_points)
