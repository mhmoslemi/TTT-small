    time_limit = budget_s
    start_time = time.time()
    for _ in range(iter_limit):
        if time.time() - start_time > time_limit:
            break
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < 0.3808:
            break
        lag_idx = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            hp = h.copy()
            hp[i] += 1e-4
            hp = np.clip(hp, 0, 1)
            hp = hp * (n_points / 2) / np.sum(hp)
            o_perturb = np.correlate(hp, 1 - hp, mode='full') * dx
            grad[i] = (o_perturb[lag_idx] - overlaps[lag_idx]) / 1e-4
        direction = -grad
        h_candidate = h + 0.1 * direction
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        o_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
        new_c5 = np.max(o_candidate)
        if new_c5 < current_c5:
            h = h_candidate
        else:
            rand_perturb = np.random.rand(n_points) * 0.1 - 0.05
            h_candidate = h + rand_perturb
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            o_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full') * dx
            new_c5 = np.max(o_candidate)
            if new_c5 < current_c5:
                h = h_candidate
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5_bound = best_c5
    return best_h, c5_bound, n_points
