import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    # Determine n_points based on initial_h_values if available
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 400
    # Initialize h from initial_h_values or random
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
    # Normalize to meet the sum constraint
    scale = (n_points / 2) / np.sum(h) if 'initial_h_values' in globals() else (n_points / 2) / np.sum(h)
    h *= scale
    # Define dx
    dx = 2.0 / n_points
    # Objective function
    def objective(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(cross_corr)
    # Gradient function
    def gradient(h_vec):
        n = len(h_vec)
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            k = lag
            j_plus_k = j + k
            if 0 <= j_plus_k < n:
                term1 = 1 - h_vec[j_plus_k]
            else:
                term1 = 0.0
            j_minus_k = j - k
            if 0 <= j_minus_k < n:
                term2 = -h_vec[j_minus_k]
            else:
                term2 = 0.0
            grad[j] = term1 + term2
        return grad
    # Constraints and bounds
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Run optimization with L-BFGS-B
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}

    final_h = result.x
