        if time.time() - time_start > budget_s:
            break
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5 - 1e-8:
            best_c5 = current_c5
            best_h = h.copy()
            if current_c5 < 0.3807:
                break
        lag_index = np.argmax(overlaps)
        k = lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        direction = -grad
        h_candidate = h + step_size * direction
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        new_c5 = max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
        if new_c5 < current_c5:
            h = h_candidate
            best_c5 = new_c5
            best_h = h.copy()
        else:
            perturbation = np.random.uniform(-perturb_range, perturb_range, size=n_points)
            h_candidate = h + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            new_c5 = max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if new_c5 < current_c5:
                h = h_candidate
                best_c5 = new_c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)
