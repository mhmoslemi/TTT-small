import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    alpha = 0.1
    iteration = 0

    while iteration < 1000 and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                gradient[j] += (1 - h[j + k])
            if 0 <= j - k < n_points:
                gradient[j] -= h[j - k]

        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (target_sum / current_sum)
        else:
            h = np.zeros(n_points)

        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

        alpha *= 0.9985
        iteration += 1

    return (best_h, best_c5, n_points)
