import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    def mutate(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        for _ in range(2):
            idx = np.random.choice(n_points)
            h[idx] = np.clip(h[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(h1, h2):
        alpha = np.random.uniform(0.4, 0.6)
        return alpha * h1 + (1 - alpha) * h2

    def generate_initial_population(size):
        pop = []
        for _ in range(size):
            if np.random.rand() < 0.3:
                h = np.random.rand(n_points)
                h = h * target_sum / np.sum(h)
            else:
                h = initial_h_values.copy()
                h = h * target_sum / np.sum(h)
            pop.append(h)
        return np.array(pop)

    population = generate_initial_population(20)
    best_h = population[0]
    best_c5 = eval_c5(best_h)
    start_time = time.time()

    for _ in range(400):
        if time.time() - start_time > budget_s - 60:
            break
        fitness = np.array([eval_c5(h) for h in population])
        elite = population[np.argsort(fitness)[:5]]
        new_pop = np.copy(elite)
        for _ in range(15):
            if time.time() - start_time > budget_s - 60:
                break
            p1, p2 = np.random.choice(20, 2, replace=False)
            child = crossover(population[p1], population[p2])
            child = mutate(child)
            new_pop = np.vstack([new_pop, child])
        population = new_pop[:20]
        current_best = population[np.argmin(fitness)]
        if eval_c5(current_best) < best_c5:
            best_h = current_best
            best_c5 = eval_c5(best_h)

    for _ in range(10):
        if time.time() - start_time > budget_s - 60:
            break
        res = minimize(lambda x: eval_c5(x), best_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
