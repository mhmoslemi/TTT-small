import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):

        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    try:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', 
            popsize=50, mutation=(0.5, 1), recombination=0.7, 
            maxiter=200, tol=1e-5, seed=seed, 
            disp=False
        )
    except:
