import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_symmetric(n_points):
        x = np.linspace(0, 1, n_points // 2)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        h = np.concatenate((h, h[::-1]))
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_sawtooth(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[:n_points//2] = x[:n_points//2] / 1.0
        h[n_points//2:] = (2 - x[n_points//2:])/1.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    initial_candidates = [
        initial_h_values,
        generate_symmetric(n_points),
        generate_sawtooth(n_points),
        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    ]

    best_h = np.array(initial_candidates[0], dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(h_candidate):
        clipped = np.clip(h_candidate, 0, 1)
        scaled = clipped * (target_sum / np.sum(clipped))
        corr = np.correlate(scaled, 1 - scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    best_result = None
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=16,
            maxiter=30,
            tol=1e-2,
            mutation=(0.6, 1),
            crossover=0.8
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_result = result

    if best_result is None:
        best_result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-5
        )

    best_h = best_result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    return (best_h, best_c5, n_points)
