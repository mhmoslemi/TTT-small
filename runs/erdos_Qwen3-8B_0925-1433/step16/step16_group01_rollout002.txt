        res = minimize(obj, guess_scaled, method='L-BFGS-B', jac=compute_gradient,
                       bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       options={'maxiter': 50})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    if best_h is not None:
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            res = minimize(obj, perturbed, method='L-BFGS-B', jac=compute_gradient,
                           bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                           options={'maxiter': 20})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
