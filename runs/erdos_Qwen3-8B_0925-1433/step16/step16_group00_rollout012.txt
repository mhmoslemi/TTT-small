import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = np.array(initial_h_values, dtype=float)
    best_h = (best_h * target_sum) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    max_iterations = 200

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        def obj(x):
            h = x
            corr_val = np.correlate(h, 1 - h, mode='full')[k_max] * dx
            return corr_val
        def grad(x):
            grad = np.zeros_like(x)
            for i in range(n_points):
                x_plus = x.copy()
                x_plus[i] += 1e-4
                x_plus = np.clip(x_plus, 0, 1)
                corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')[k_max] * dx
                x_minus = x.copy()
                x_minus[i] -= 1e-4
                x_minus = np.clip(x_minus, 0, 1)
                corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')[k_max] * dx
                grad[i] = (corr_plus - corr_minus) / (2e-4)
            return -grad
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
