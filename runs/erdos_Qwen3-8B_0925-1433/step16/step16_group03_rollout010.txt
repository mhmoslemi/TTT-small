import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial patterns
    initial_patterns = []
    initial_patterns.append(np.array(initial_h_values, dtype=float))
    # Binary pattern
    bin_pat = np.zeros(n_points)
    bin_pat[::2] = 1.0
    initial_patterns.append(bin_pat)
    # Sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    sin_pat = 0.5 + 0.5 * np.cos(np.pi * x)
    sin_pat = np.clip(sin_pat, 0, 1)
    initial_patterns.append(sin_pat)
    # Random binary
    rand_pat = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    rand_pat = (rand_pat * target_sum) / np.sum(rand_pat)
    initial_patterns.append(rand_pat)

    best_h = None
    best_c5 = 1.0
    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    for pattern in initial_patterns:
        pattern_scaled = (pattern * target_sum) / np.sum(pattern)
        res = minimize(obj, pattern_scaled, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_h = res.x
            curr_c5 = np.max(np.correlate(curr_h, 1 - curr_h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_h, best_c5 = curr_h.copy(), curr_c5

    # Local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_h = res.x
            curr_c5 = np.max(np.correlate(curr_h, 1 - curr_h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_h, best_c5 = curr_h.copy(), curr_c5

    # Final check
    if best_h is None:
        best_h = np.array(initial_h_values)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
