import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def adjust_h(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros_like(x)
        return h_clipped * target_sum / sum_h

    def objective(x):
        h = adjust_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = adjust_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    # Initialize candidates
    initial_candidates = [initial_h_values]
    x = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    h_sinusoidal = h_sinusoidal * target_sum / np.sum(h_sinusoidal)
    initial_candidates.append(h_sinusoidal)
    h_triangular = np.zeros(n_points)
    for i in range(n_points):
        if x[i] <= 1:
            h_triangular[i] = x[i] / 1.0
        else:
            h_triangular[i] = (2 - x[i]) / 1.0
    h_triangular = h_triangular * target_sum / np.sum(h_triangular)
    initial_candidates.append(h_triangular)
    h_random = np.random.rand(n_points)
    h_random = h_random * target_sum / np.sum(h_random)
    initial_candidates.append(h_random)

    start_time = time.time()
    best_h = np.array(initial_candidates[0], dtype=float)
    best_c5 = objective(best_h)

    # Differential evolution
    bounds = [(0.0, 1.0)] * n_points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=20,
                mutation=0.8,
                recombination=0.9,
                tol=1e-6,
                maxiter=100,
                seed=seed,
                disp=False,
                polish=True,
                recombination='arithmetic',
                mutation=(0.5, 1.0),
                callback=lambda x: None
            )
            best_h = result.x
            best_c5 = result.fun
        except:
            pass
        remaining_time = budget_s - (time.time() - start_time)

    # L-BFGS refinement
    if remaining_time > 0:
        best_h = adjust_h(best_h)
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            jac=compute_gradient,
            bounds=bounds,
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    best_h = adjust_h(best_h)
    return (best_h, best_c5, n_points)
