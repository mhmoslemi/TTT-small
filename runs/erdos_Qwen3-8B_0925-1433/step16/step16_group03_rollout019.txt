from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    patterns = [h]
    x = np.linspace(0, 2, n_points)

    sin_pat = np.clip(sin_pat, 0, 1)
    sin_pat = sin_pat * target_sum / np.sum(sin_pat)
    patterns.append(sin_pat)

    rand_pat = np.clip(rand_pat, 0, 1)
    rand_pat = rand_pat * target_sum / np.sum(rand_pat)
    patterns.append(rand_pat)

    start_time = time.time()

        h_pert = np.clip(p + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        if res.success:

    # Refinement
    elapsed = time.time() - start_time
