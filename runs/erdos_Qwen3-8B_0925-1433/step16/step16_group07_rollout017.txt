import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate base patterns with shifts and noise
    base_patterns = []
    base_patterns.append(np.array(initial_h_values, dtype=float))
    # Golden ratio rotation
    golden = (1 + np.sqrt(5)) / 2
    shifted = np.roll(initial_h_values, int(golden * n_points) % n_points)
    base_patterns.append(shifted)
    # Noise pattern
    noise = np.array(initial_h_values) + np.random.normal(0, 0.1, n_points)
    noise = np.clip(noise, 0, 1)
    noise = noise * target_sum / np.sum(noise)
    base_patterns.append(noise)

    # Create population with perturbations
    population = []
    for p in base_patterns:
        for _ in range(5):
            perturbed = p + np.random.normal(0, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            population.append(perturbed)
    population = [np.array(p) for p in population]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]
    if not population:
        population.append(np.full(n_points, 0.5))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    start_time = time.time()
    for p in population:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6)
        if res.success:
            optimized_h = project(res.x)
            current_c5 = objective(optimized_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
