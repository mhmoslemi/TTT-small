import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial guess
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_perturbed_guess(base, scale=0.1):
        pert = np.random.normal(0, scale, n_points)
        return np.clip(base + pert, 0, 1) * target_sum / np.sum(np.clip(base + pert, 0, 1))

    start_time = time.time()
    restarts = 20
    for _ in range(restarts):
        if time.time() - start_time > budget_s:
            break
        # Generate multiple initial guesses
        guesses = [initial_h_values]
        guesses.append(np.clip(initial_h_values + np.random.normal(0, 0.2, n_points), 0, 1))
        guesses.append(np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1))
        guesses.append(np.clip(np.flip(initial_h_values) + np.random.normal(0, 0.1, n_points), 0, 1))
        for guess in guesses:
            h_pert = generate_perturbed_guess(np.array(guess, dtype=float))
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            bounds = [(0, 1) for _ in range(n_points)]

            res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_h = res.x
                current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5

    # Final refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
