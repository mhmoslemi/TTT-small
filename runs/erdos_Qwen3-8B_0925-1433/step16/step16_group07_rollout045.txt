import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    base_patterns = [np.array(initial_h_values)]
    for i in range(3):
        pert = np.random.normal(0, 0.05, n_points)
        base_patterns.append(project(np.array(initial_h_values) + pert))
    base_patterns.append(np.random.rand(n_points))
    base_patterns.append(np.zeros(n_points) + 0.5)
    base_patterns = [project(p) for p in base_patterns]

    population = [np.copy(p) for p in base_patterns]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(project(best_h), 1 - project(best_h), mode='full')) * dx

    start_time = time.time()
    for p in population[:5]:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, p, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
        if res.success:
            candidate = project(res.x)
            current_c5 = np.max(np.correlate(candidate, 1 - candidate, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate
    h_values = project(best_h)
    return (h_values, best_c5, n_points)
