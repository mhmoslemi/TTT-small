import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def eval_c5(h):
        return np.max(np.correlate(h, 1-h, mode='full')) * dx

    def rescale(h):
        s = np.sum(h)
        return h * target_sum / s if s != 0 else h

    def clip_rescale(h):
        return rescale(np.clip(h, 0, 1))

    # Initialize population with diverse patterns
    pop_size = 50
    population = [initial_h_values]
    pop = [np.full(n_points, 0.5)]
    binary = np.random.rand(n_points) < 0.5
    pop.append(rescale(binary.astype(float)))
    for _ in range(pop_size - 3):
        h = np.random.rand(n_points)
        pop.append(rescale(h))
    population = np.array(pop)

    best_h = initial_h_values.copy()
    best_c5 = eval_c5(best_h)

    start_time = time.time()
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([eval_c5(p) for p in population])
        # Select top 30
        top_indices = np.argsort(fitness)[:30]
        top_pop = population[top_indices]
        # Generate new population
        new_pop = []
        for _ in range(pop_size):
            # Crossover
            p1, p2 = np.random.choice(top_pop, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            # Mutation
            child += np.random.normal(0, 0.05, n_points)
            new_pop.append(clip_rescale(child))
        population = np.array(new_pop)
        # Update best
        for h in population:
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
