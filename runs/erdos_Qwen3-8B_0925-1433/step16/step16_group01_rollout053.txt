import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        m = int(target_sum)
        # Uniform
        uniform = np.zeros(n_points)
        uniform[::2] = 1
        # Random
        random = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        while np.sum(random) != m:
            random = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        # Sinusoidal
        x = np.linspace(0, 2, n_points)
        sin_vals = np.sin(2 * np.pi * x)
        idx = np.argsort(sin_vals)[::-1]
        sinusoidal = np.zeros(n_points)
        sinusoidal[idx[:m]] = 1
        return [uniform, random, sinusoidal]

    patterns = generate_patterns()
    best_h = None
    best_c5 = float('inf')

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = (1 - h[j_plus]) if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = (h[j_minus]) if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()

    for p in patterns:
        h = p.copy()
        h = h * target_sum / np.sum(h)
        for _ in range(20):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success:
                h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(obj, perturbed, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
