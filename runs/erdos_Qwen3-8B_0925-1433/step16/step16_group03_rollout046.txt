import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Construct structured sine-based initial guess
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 + 0.5 * np.sin(np.pi * x)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        h = x.copy()
        np.clip(h, 0, 1, out=h)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    start_time = time.time()

    # Optimization with structured initial guess
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=500)
    if res.success:
        best_h = res.x
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Iterative refinement with perturbations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + pert, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=300)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx

    # Fallback to baseline if no improvement

        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
