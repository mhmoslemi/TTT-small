import time

    def obj(x):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

        patterns = []
        # Pattern 1: Modulated sine

        sin_h = np.clip(sin_h, 0, 1)
        patterns.append(sin_h)

            idx = np.abs(x - p) < pulse_width
            pulse_h[idx] = 1.0
        pulse_h = pulse_h * target_sum / np.sum(pulse_h)
        patterns.append(pulse_h)
        # Pattern 3: Random binary
        bin_h = np.random.binomial(1, 0.1, n_points)
        bin_h = bin_h * target_sum / np.sum(bin_h)
        patterns.append(bin_h)
        # Pattern 4: Cosine pulses
