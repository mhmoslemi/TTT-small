import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    initial_guesses = []
    initial_guesses.append(initial_h_values)
    initial_guesses.append(np.concatenate([initial_h_values[::-1], initial_h_values[:-1]]))
    initial_guesses.append(np.zeros(n_points))
    initial_guesses.append(np.full(n_points, 0.5))
    for _ in range(4):
        pert = np.random.normal(0, 0.15, n_points)
        guess = np.clip(initial_h_values + pert, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        h_pert = guess * target_sum / np.sum(guess)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Upsampling to 100 points
    n_points_up = 100
    dx_up = 2.0 / n_points_up
    h_up = np.zeros(n_points_up)
    for i in range(n_points_up):
        x = i * dx_up
        idx = np.searchsorted(np.linspace(0, 2, n_points), x)
        if idx == 0:
            val = best_h[0]
        elif idx == n_points:
            val = best_h[-1]
        else:
            val = (best_h[idx-1] * (x - (idx-1)*dx) + best_h[idx] * ((idx)*dx - x)) / dx
        h_up[i] = val
    h_up = h_up * target_sum / np.sum(h_up)
    corr_up = np.correlate(h_up, 1 - h_up, mode='full')
    best_c5_up = np.max(corr_up) * dx_up

    if best_c5_up < best_c5:
        best_h = h_up
        best_c5 = best_c5_up

    return (np.clip(best_h, 0, 1), best_c5, n_points_up)
