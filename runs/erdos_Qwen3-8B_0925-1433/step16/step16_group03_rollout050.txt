        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    # Final refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
