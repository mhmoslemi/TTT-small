import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_symmetric(n_points):
        x = np.linspace(0, 2, n_points)
        t = x - 1.0
        h = 0.5 + 0.5 * np.cos(np.pi * t)
        h = np.clip(h, 0, 1)
        return h

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - x[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = x[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_candidates = [initial_h_values, generate_symmetric(n_points)]
    best_h = np.array(initial_candidates[0], dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    for candidate in initial_candidates:
        h = np.array(candidate, dtype=float)
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
