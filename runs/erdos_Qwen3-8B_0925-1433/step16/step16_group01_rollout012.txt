import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_patterns = []

    # Uniform pattern
    uniform = np.zeros(n_points)
    uniform[::2] = 1.0

    initial_patterns.append(sinusoidal)

    random_pattern = np.random.rand(n_points)
    random_pattern.sort()

    random_pattern[:30] = 1.0
    random_pattern[30:] = 0.0
    initial_patterns.append(random_pattern)
    # Mirror pattern
    mirror_uniform = np.zeros(n_points)
    mirror_uniform[::2] = 1.0
    mirror_uniform = np.flip(mirror_uniform)
    initial_patterns.append(mirror_uniform)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')
