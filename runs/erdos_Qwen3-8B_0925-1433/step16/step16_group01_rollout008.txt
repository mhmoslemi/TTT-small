import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_guesses():
        symmetric = np.zeros(n_points)
        symmetric[:50] = 1.0
        tiling = np.zeros(n_points)
        tiling[::2] = 1.0
        randomized = np.random.randint(0, 2, n_points)
        randomized = randomized * target_sum / np.sum(randomized)
        return [symmetric, tiling, randomized]

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_guesses = generate_initial_guesses()

    best_h = np.zeros(n_points)
    best_c5 = 0.381
    total_time = 0.0

    for guess in initial_guesses:
        h = guess.copy()
        h = h * target_sum / np.sum(h)
        start_time = time.time()
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5
        total_time += time.time() - start_time
        if total_time > budget_s:
            break

    for _ in range(5):
        if total_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        start_time = time.time()
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun
        total_time += time.time() - start_time

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
