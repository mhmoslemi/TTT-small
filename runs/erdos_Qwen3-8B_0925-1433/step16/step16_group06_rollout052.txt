import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    pop_size = 100
    population = [initial_h.copy() for _ in range(pop_size//2)]
    for _ in range(pop_size//2):
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = (h < 0.5) * 1.0
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    def objective(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def evaluate(h_vec):
        return objective(h_vec)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Mutation and crossover
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + 0.5 * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * target_sum / np.sum(mutant)
            new_pop.append(mutant)
        
        # Crossover
        for i in range(pop_size):
            target = population[i]
            mutant = new_pop[i]
            mask = np.random.rand(n_points) < 0.8
            child = np.where(mask, mutant, target)
            child = child * target_sum / np.sum(child)
            new_pop[i] = child
        
        # Evaluate
        scores = np.array([evaluate(h) for h in new_pop])
        new_pop = np.array([new_pop[i] for i in np.argsort(scores)])
        new_pop = new_pop[:pop_size]
        
        # Update best
        for h in new_pop:
            curr_c5 = evaluate(h)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
    
    # Local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:

        h = best_h.copy()
