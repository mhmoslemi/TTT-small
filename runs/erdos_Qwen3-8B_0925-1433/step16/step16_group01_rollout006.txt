import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1.0 if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()

    # Initial optimization with SLSQP
    res = minimize(obj, h, method='SLSQP', jac=compute_gradient,
                   bounds=[(0, 1)]*n_points,
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res.success:
        h = res.x
        current_c5 = res.fun
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    # Perturb and refine
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.03, n_points)
        perturbed_h = h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient,

        if res.success and res.fun < best_c5:

    best_h = np.clip(best_h, 0, 1)
