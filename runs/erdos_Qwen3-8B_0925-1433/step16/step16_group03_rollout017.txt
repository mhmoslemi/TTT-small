import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    restarts = 15
    for r in range(restarts):
        if time.time() - start_time > budget_s:
            break
        if r == 0:
            pert = np.random.normal(0, 0.1, n_points)
        else:
            pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Fallback pattern: periodic 0.4, 0.8, 0.4
    fallback = np.tile([0.4, 0.8], n_points // 2)
    fallback = np.concatenate([fallback[:n_points], np.zeros(n_points - len(fallback))])
    fallback = fallback * target_sum / np.sum(fallback)
    fallback_c5 = np.max(np.correlate(fallback, 1 - fallback, mode='full')) * dx
    if fallback_c5 < best_c5:
        best_h, best_c5 = fallback.copy(), fallback_c5

    # Final refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
