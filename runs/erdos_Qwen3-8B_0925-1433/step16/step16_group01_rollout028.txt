import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        val = np.max(corr) * dx
        return val

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1.0 - x[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = x[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()
    patterns = [h_initial]
    pm = np.random.permutation(np.arange(n_points))
    patterns.append(np.zeros(n_points))
    patterns[-1][pm[:int(target_sum)]] = 1
    patterns[-1] = patterns[-1] * target_sum / np.sum(patterns[-1])

    for _ in range(1):
        if time.time() - start_time > budget_s:
            break
        h = patterns[0].copy()
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = patterns[1].copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
