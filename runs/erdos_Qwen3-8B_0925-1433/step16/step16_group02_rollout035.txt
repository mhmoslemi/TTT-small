import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        return h_clipped * (target_sum / sum_h) if sum_h > 0 else np.zeros_like(h_clipped)

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0
            grad[i] = dx * (term1 - term2)
        return grad

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [initial_h_values]
    population.append(0.5 + 0.5 * np.sin(np.linspace(0, 2, n_points) * np.pi / 2))
    population.append(np.clip(np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1)), 0, 1))
    population.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    def evaluate_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(10):
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = 0.4 * p1 + 0.6 * p2
            idx = np.random.choice(n_points, 5, replace=False)
            child[idx] = 1 - child[idx]
            new_pop.append(rescale(child))
        population = new_pop + population[:10]
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    h_guess = best_h
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1)] * n_points
    result = minimize(obj, h_guess, method='L-BFGS-B', jac=compute_gradient, bounds=bounds, constraints=constraints)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    best_h = rescale(best_h)
