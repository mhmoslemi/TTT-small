import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return float('inf')
        h_scaled = h_clipped * (target_sum / sum_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (target_sum / np.sum(h_p))
            obj_p = objective(h_p)
            h_m = h.copy()
            h_m[i] -= eps
            h_m = np.clip(h_m, 0, 1)
            h_m = h_m * (target_sum / np.sum(h_m))
            obj_m = objective(h_m)
            grad[i] = (obj_p - obj_m) / (2 * eps)
        return grad

    population = [np.clip(initial_h, 0, 1) * (target_sum / np.sum(np.clip(initial_h, 0, 1)))]
    for _ in range(49):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        population.append(h_rand)

    best_h = population[0]
    best_c5 = objective(best_h)
    gbest = best_h
    gbest_c5 = best_c5
    early_stop = 10
    iterations_without_improvement = 0
    max_gen = 300
    pop_size = 50
    mutation_rate = 0.8
    crossover_rate = 0.9
    generations = 0
    start_time = time.time()

    while generations < max_gen and time.time() - start_time < budget_s:
        new_population = []
        for _ in range(pop_size):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = parent1.copy()
            if np.random.rand() < crossover_rate:
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, child, parent2)
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            child = np.clip(child, 0, 1)
            child = child * (target_sum / np.sum(child))
            new_population.append(child)
        population = new_population

        for h_candidate in population:
            curr_c5 = objective(h_candidate)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_candidate
            if curr_c5 < gbest_c5:
                gbest = h_candidate
                gbest_c5 = curr_c5

        top_idx = np.argsort([objective(h) for h in population])[:int(pop_size * 0.1)]
        for idx in top_idx:
            h = population[idx]
            grad = compute_gradient(h)
            step = 0.05
            h_step = h - step * grad
            h_step = np.clip(h_step, 0, 1)
            h_step = h_step * (target_sum / np.sum(h_step))
            population[idx] = h_step

        if best_c5 < 0.3807:
            break
        if np.abs(gbest_c5 - best_c5) < 1e-5:
            iterations_without_improvement += 1
            if iterations_without_improvement >= early_stop:
                break
        else:
            iterations_without_improvement = 0
        generations += 1

    return (np.clip(best_h, 0, 1), best_c5, n_points)
