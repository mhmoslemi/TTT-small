def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    best_h = np.array(initial_h_values, dtype=float)

        s = np.sum(arr)
        return arr * target_sum / s if s != 0 else np.zeros_like(arr)

    # Generate initial population
    initial_norm = initial_h_values * target_sum / np.sum(initial_h_values)
    population = [initial_norm]

        pert = np.random.uniform(-0.1, 0.1, n_points)
        temp = initial_norm + pert
        population.append(normalize(temp))
    for m in [10]:
        spacing = 2.0 / (m + 1)
        intervals = np.zeros(n_points)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
