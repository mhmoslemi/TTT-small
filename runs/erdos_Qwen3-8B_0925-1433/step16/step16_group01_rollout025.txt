import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guesses
    initializations = []

    # Symmetric initialization
    h_sym = np.zeros(n_points)
    for i in range(n_points):
        h_sym[i] = (i <= n_points // 2)
    initializations.append(h_sym)

    # Tiling initialization
    h_tile = np.zeros(n_points)
    for i in range(n_points):
        seg = i // (n_points // 5)
        h_tile[i] = 1 if seg % 2 == 0 else 0
    current_sum = np.sum(h_tile)
    scale = target_sum / current_sum
    h_tile *= scale
    initializations.append(h_tile)

    # Random binary initialization
    h_rand = np.random.choice([0, 1], size=n_points)
    current_sum = np.sum(h_rand)
    scale = target_sum / current_sum
    h_rand *= scale
    initializations.append(h_rand)

    best_h = initializations[0]
    best_c5 = float('inf')

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()

    for init in initializations:
        h = init.copy()
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
