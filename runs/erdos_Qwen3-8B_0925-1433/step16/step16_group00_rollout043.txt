import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(h):
        s = np.sum(h)
        return h * target_sum / s if s != 0 else h * 0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, sigma=0.05):
        h = h.copy()
        noise = np.random.normal(0, sigma, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        return normalize(h)

    def crossover(h1, h2):
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = h1[i]
            else:
                h[i] = h2[i]
        return normalize(h)

    def tournament_select(pop, k=5):
        idx = np.random.choice(len(pop), k, replace=False)
        return min(idx, key=lambda i: evaluate(pop[i]))

    def generate_initial_population():
        pop = []
        h_base = np.array(initial_h_values, dtype=float)
        pop.append(normalize(h_base))
        for _ in range(10):
            h = h_base + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            pop.append(normalize(h))
        for m in range(5, 16):
            h = np.zeros(n_points)
            spacing = 2.0 / (m + 1)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                h[idx_start:idx_end] = 1
            h = np.clip(h, 0, 1)
            pop.append(normalize(h))
        return pop

    pop = generate_initial_population()
    best_h = min(pop, key=evaluate)
    best_c5 = evaluate(best_h)
    start_time = time.time()
    generation = 0
    no_improvement = 0

    while time.time() - start_time < budget_s:
        generation += 1
        if no_improvement > 20:
            pop = generate_initial_population()
            no_improvement = 0
        new_pop = []
        for _ in range(len(pop)//2):
            p1 = tournament_select(pop)
            p2 = tournament_select(pop)
            child = crossover(p1, p2)
            child = mutate(child, sigma=0.1)
            new_pop.append(child)
        pop += new_pop
        new_pop = []
        for _ in range(len(pop)//2):

            print(f"Generation {generation}, C5: {best_c5:.6f}")
