import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_patterns():
        patterns = [np.array(initial_h_values, dtype=float)]
        patterns.append(np.full(n_points, 0.5))
        phi = (1 + np.sqrt(5)) / 2
        positions = [2.0 * (i * phi) % 2.0 for i in range(n_points)]
        h_golden = np.zeros(n_points)
        for pos in positions:
            idx = int(pos * n_points / 2)
            idx = np.clip(idx, 0, n_points - 1)
            h_golden[idx] += 1
        h_golden = h_golden * target_sum / np.sum(h_golden)
        patterns.append(h_golden)
        return [project(p) for p in patterns]

    population = generate_patterns()
    for _ in range(5):
        population.append(project(np.random.rand(n_points) * target_sum))
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]

    best_h = project(population[0])
    best_c5 = float('inf')

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            h = p.copy()
            for _ in range(50):
                neighbor_sum = np.zeros(n_points)
                for i in range(n_points):
                    s = h[(i - 1) % n_points] + h[i] + h[(i + 1) % n_points]
                    neighbor_sum[i] = s
