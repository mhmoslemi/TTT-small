        h = h * target_sum / np.sum(h)
        population.append(h[:n_points].copy())
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([h * target_sum / np.sum(h) for h in population])

    best_h = population[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        tournament = np.random.choice(len(population), size=5, replace=False)
        parent1, parent2 = population[tournament[0]], population[tournament[1]]
        cross_idx = np.random.randint(n_points)
        child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]])

        corr = np.correlate(child, 1 - child, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        relevant_indices = []
        if lag >= 0:
            for x in range(n_points - lag):
                y = x + lag
                relevant_indices.append(x)
                relevant_indices.append(y)
        else:
            m = -lag
            for x in range(m, n_points):
                y = x + lag
                relevant_indices.append(x)
                relevant_indices.append(y)
        relevant_indices = list(set(relevant_indices))

        for i in relevant_indices:
            if np.random.random() < 0.5:
                child[i] += np.random.uniform(-0.15, 0.15) * child[i]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)

        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
