    h_clipped = np.clip(h_mut, 0, 1)
    sum_clip = np.sum(h_clipped)
    if sum_clip == 0:
        scaled = np.zeros_like(h_clipped)
    else:
        scaled = h_clipped * (target_sum / sum_clip)
    scaled = np.clip(scaled, 0, 1)
    return scaled
