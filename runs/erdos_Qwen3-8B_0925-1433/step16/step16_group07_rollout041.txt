import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    base_patterns = [initial_h_values.copy(), np.roll(initial_h_values, int((np.sqrt(5)-1)/2 * n_points))]

    population = []
    for b in base_patterns:
        p = b.copy()

        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)
    for _ in range(45):
        p = np.random.rand(n_points) * 0.5
        p = p * target_sum / np.sum(p)
        population.append(p)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * target_sum / total if total else np.full_like(h, 0.5)

    best_h = population[0].copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(150):
        if time.time()-start_time > budget_s: break
        for p in population:
            res = minimize(obj, p, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}], tol=1e-6)
            if res.success:
                p[:] = project(res.x)
        for p in population:
            c5 = obj(p)
            if c5 < best_c5:
                best_c5 = c5
                best_h = p.copy()

    h_values = project(best_h)
