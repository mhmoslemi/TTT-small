import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-6:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = project(x)
        return evaluate(h)

    def fitness(x):
        return evaluate(x)

    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)
    population = [base + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    population = [project(p) for p in population]
    population = [p for p in population if abs(np.sum(p) - target_sum) < 1e-6]

    best_h = project(population[0])
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(0, len(population), 2):
            if i + 1 < len(population):
                p1 = population[i]
                p2 = population[i+1]
                child = (p1 + p2) * 0.5 + np.random.normal(0, 0.05, n_points)
                child = project(child)
                new_population.append(child)
            else:
                new_population.append(population[i])
        population = new_population
        for p in population:
            res = minimize(objective, p, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum}])
            if res.success:
                p_new = project(res.x)
                if abs(np.sum(p_new) - target_sum) < 1e-6:
                    p[:] = p_new
        current_c5 = evaluate(project(population[0]))
        if current_c5 < best_c5:
            best_h = project(population[0])
            best_c5 = current_c5

    return (project(best_h), best_c5, n_points)
