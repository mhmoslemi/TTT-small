        k_max = np.argmax(corr)
        dx_val = 2.0 / n_points
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = (i + k_max) % n_points
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % n_points
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    patterns = [h]
    for shift in [1, 5, 10]:
        patterns.append(np.roll(h, shift))
    for _ in range(5):
        patterns.append(h + np.random.normal(0, 0.1, n_points))
    patterns = [project(p) for p in patterns if np.abs(np.sum(p) - target_sum) < 1e-6]

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in patterns:
            res = minimize(obj, p, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                p_new = project(res.x)
                if np.abs(np.sum(p_new) - target_sum) < 1e-6:
                    patterns[patterns.index(p)] = p_new
        current_c5 = np.max(np.correlate(project(patterns[0]), 1 - project(patterns[0]), mode='full')) * dx
        if current_c5 < best_c5:
            best_h = project(patterns[0])
            best_c5 = current_c5

    return (project(best_h), best_c5, n_points)
