import numpy as np
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    max_iterations = 200
    population_size = 20
    elite_size = 3
    mutation_rate = 0.1
    tournament_size = 5

    # Initialize population
    population = []
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population.append(initial_h)
    
    # Add random binary
    rand_binary = np.random.choice([0, 1], size=n_points)
    rand_binary = rand_binary * target_sum / np.sum(rand_binary)
    population.append(rand_binary)
    
    # Add uniform
    uniform = np.full(n_points, 0.5)
    population.append(uniform)
    
    # Add alternating
    alternating = np.array([1 if i % 2 == 0 else 0 for i in range(n_points)])
    alternating = alternating * target_sum / np.sum(alternating)
    population.append(alternating)
    
    # Fill to population size
    while len(population) < population_size:
        temp = np.random.choice([0, 1], size=n_points)
        temp = temp * target_sum / np.sum(temp)
        population.append(temp)
    population = population[:population_size]

    best_h = population[0].copy()
    best_c5 = np.max(np.correlate(population[0], 1 - population[0], mode='full')) * (2.0 / n_points)

    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)

        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = np.array(fitness)[sorted_indices]
        
        # Select elites
        elites = sorted_pop[:elite_size]
        new_pop = elites.copy()
        
        # Generate new individuals
        for _ in range(population_size - elite_size):
            # Tournament selection
            tour_indices = np.random.choice(range(population_size), size=tournament_size, replace=False)
            parent1 = sorted_pop[tour_indices[0]]
            parent2 = sorted_pop[tour_indices[1]]

            # Crossover
            child = (parent1 + parent2) / 2.0

                best_c5 = c5
                best_h = h.copy()
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
