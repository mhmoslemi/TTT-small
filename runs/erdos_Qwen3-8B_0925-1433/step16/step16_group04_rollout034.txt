import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n_points

    start_time = time.time()
    step_size = 0.1
    step_decay = 0.9
    step_decay_freq = 50

    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        lag = max_corr_index - (n_points - 1)
        grad = np.zeros(n_points)
        for x in range(n_points):
            if 0 <= x + lag < n_points:
                grad[x] += 1 - h[x + lag]
            if 0 <= x - lag < n_points:
                grad[x] -= h[x - lag]
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * 2.0 / n_points
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if _ % step_decay_freq == 0:
            step_size *= step_decay

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * 2.0 / n_points
    return (best_h, final_c5, n_points)
