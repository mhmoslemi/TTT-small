        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    x0 = best_h.copy()
    res = minimize(obj, x0, method='SLSQP', jac=grad, bounds=[(0,1)]*n_points,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   tol=1e-6)
    h_opt = res.x if res.success else best_h

    h_values = np.clip(h_opt, 0, 1)
    h_values *= target_sum / np.sum(h_values)
    final_c5 = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    return (h_values, final_c5, n_points)
