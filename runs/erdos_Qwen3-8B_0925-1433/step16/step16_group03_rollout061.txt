        linear_interp = np.interp(upscaled_x, np.linspace(0, 2, n_points), best_h)
        linear_interp = np.clip(linear_interp, 0, 1)
        linear_interp = linear_interp * (upscaled_n / 2.0) / np.sum(linear_interp)
        def objective_upscale(x):
            h = x * (upscaled_n / 2.0) / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * upscaled_dx
        bounds_up = [(0, 1) for _ in range(upscaled_n)]
        cons_up = ({'type': 'eq', 'fun': lambda x: np.sum(x) - (upscaled_n / 2.0)})
        res_up = minimize(objective_upscale, linear_interp, method='L-BFGS-B', bounds=bounds_up, constraints=cons_up, tol=1e-6)
        if res_up.success:
            current_h_up = res_up.x
            current_c5_up = compute_c5(current_h_up, upscaled_dx)
            if current_c5_up < best_c5:
                best_h, best_c5 = current_h_up.copy(), current_c5_up
                n_points = upscaled_n
                dx = upscaled_dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
