import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and others
    pop_size = 50
    population = [np.array(initial_h_values, dtype=float)]
    # Binary pattern
    bin_h = np.zeros(n_points)
    bin_h[:int(n_points//2)] = 1.0
    bin_h = bin_h * target_sum / np.sum(bin_h)
    population.append(bin_h)
    # Uniform
    uni_h = np.full(n_points, 0.5)
    uni_h = uni_h * target_sum / np.sum(uni_h)
    population.append(uni_h)
    # Alternating (if even)
    if n_points % 2 == 0:
        alt_h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)])
        alt_h = alt_h * target_sum / np.sum(alt_h)
        population.append(alt_h)
    # Fill remaining
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start = time.time()
    for _ in range(300):  # 300 generations
        if time.time() - start > budget_s:
            break
        # Evaluate all
        scores = [compute_c5(h) for h in population]
        # Select parents
        parents = np.argsort(scores)[:int(0.3 * pop_size)]
        # Generate new pop
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (population[p1] + population[p2]) / 2
            if np.random.rand() < 0.1:
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                child[idx1], child[idx2] = child[idx2], child[idx1]
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        # Combine and select
        combined = population + new_pop
        scores = [compute_c5(h) for h in combined]
        new_pop = [combined[i] for i in np.argsort(scores)[:pop_size]]
        # Update best
        for h in new_pop:
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c
        population = new_pop

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
