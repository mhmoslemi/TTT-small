import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial population with diverse patterns: base, intervals, and binaries
    pop = []
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    pop.append(h)
    for m in [5, 10, 15]:
        spacing = 2.0 / (m + 1)
        intervals = np.zeros(n_points)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                intervals[idx_start:idx_end] = 1
        intervals = intervals * target_sum / np.sum(intervals)
        pop.append(intervals)
    for _ in range(5):
        h = np.random.binomial(1, 0.5, n_points)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    pop = [np.clip(p, 0, 1) for p in pop]
    pop = pop[:50]
    
    best_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    generations = 200
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all individuals
        fitness = [np.max(np.correlate(h, 1-h, mode='full')) * dx for h in pop]
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = pop[best_idx].copy()
            best_c5 = fitness[best_idx]
        # Select parents via tournament
        parents = []
        for _ in range(len(pop)):
            t = np.random.choice(len(pop), 5)
            w = t[np.argmin([fitness[i] for i in t])]
            parents.append(pop[w])
        # Crossover
        children = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                child1, child2 = np.where(np.random.rand(len(p1)) < 0.8, p1, p2), np.where(np.random.rand(len(p2)) < 0.8, p2, p1)
                children.extend([child1, child2])
            else:
                children.append(parents[i])
        # Mutate and normalize
        mutated = []
        for child in children:
            m = child.copy()
            for j in range(len(m)):
                if np.random.rand() < 0.1:
                    m[j] += np.random.normal(0, 0.1)
                    m[j] = np.clip(m[j], 0, 1)
            m = m * target_sum / np.sum(m)
            mutated.append(m)
        pop = mutated
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
