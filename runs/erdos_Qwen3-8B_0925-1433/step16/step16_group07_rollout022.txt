def run(seed=42, budget_s=600):

        # Compute cross-correlation

        if max_c5 < best_c5:
            best_c5 = max_c5

        # Find the lag with maximum overlap
        k_star_index = np.argmax(corr)
        lag = k_star_index - (n_points - 1)  # map to actual lag

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # (i + lag) mod n_points and (i - lag) mod n_points
