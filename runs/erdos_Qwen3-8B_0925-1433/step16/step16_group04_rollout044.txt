    import time

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    alpha = 0.1
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k = m - (n_points - 1)
        grad = np.zeros(n_points)

        if k >= 0:
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += 1 - h[i + k]
        else:
            for i in range(n_points):
                if i - k < n_points:
                    grad[i] -= h[i - k]

        h = h - alpha * grad
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        alpha *= 0.999

    return (best_h, best_c5, n_points)
