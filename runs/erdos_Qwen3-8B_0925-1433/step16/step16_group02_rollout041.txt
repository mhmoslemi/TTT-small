import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_pattern(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1:
                h[i] = 1.0 if i % 3 == 0 else 0.5
            else:
                h[i] = 0.5 if i % 3 == 0 else 0.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return evaluate(h)

    def constraint(h):
        return np.sum(h) - target_sum

    def generate_initial_candidates():
        candidates = [
            initial_h_values.copy(),
            generate_pattern(n_points),
            np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        ]
        for h in candidates:
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
            yield h

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def differential_evolution_step(bounds, pop_size=20, max_iter=50):
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=max_iter,
            popsize=pop_size,
            tol=1e-6,
            mutation=(0.8, 1),
            recombination=0.9,
            seed=seed,
            polish=True,
            disp=False
        )
        return clip_scale(result.x)

    def refine_candidate(h):
        res = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1)]*n_points,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-6
        )
        return clip_scale(res.x) if res.success else h

    candidates = list(generate_initial_candidates())
    best_h = candidates[0]
    best_c5 = evaluate(best_h)
    elapsed = 0

    start_time = time.time()
    for i in range(50):
        if time.time() - start_time > budget_s:
            break
        h = differential_evolution_step([(0, 1)]*n_points)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h = refine_candidate(h)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        elapsed = time.time() - start_time

    return (clip_scale(best_h), best_c5, n_points)
