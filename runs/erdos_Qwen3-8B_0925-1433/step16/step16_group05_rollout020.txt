def run(seed=42, budget_s=600):

    # Parameters
    max_iterations = 200
    early_stop = 10
    iterations_without_improvement = 0

    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Determine worst lag
        mid = (len(corr) - 1) // 2
        k_max = np.argmax(corr)
        if k_max < mid:
            lag = -(mid - k_max)
        else:
            lag = k_max - mid
        # Compute products
        products = []
        for i in range(n):
            j = (i + lag) % n
            products.append(h[i] * h[j])
        sum_product = np.sum(products)
        if sum_product == 0:
            continue
        # Select top 10% indices
        sorted_indices = np.argsort(products)[::-1]
        top_indices = sorted_indices[:int(0.1 * n)]
        # Adjust
        for idx in top_indices:
            i = idx
            j = (i + lag) % n
            impact = products[idx] / float(sum_product)
            adjust = 0.1 * impact
            h[i] = np.clip(h[i] - adjust, 0, 1)
            h[j] = np.clip(h[j] - adjust, 0, 1)
        # Rescale
        h = h * target_sum / np.sum(h)
        # Check for improvement
        if np.abs(best_c5 - current_c5) < 1e-5:
            iterations_without_improvement += 1
            if iterations_without_improvement >= early_stop:
                break
        else:
            iterations_without_improvement = 0
        if best_c5 < 0.3807:
            break

    return (np.clip(best_h, 0, 1), best_c5, n)
