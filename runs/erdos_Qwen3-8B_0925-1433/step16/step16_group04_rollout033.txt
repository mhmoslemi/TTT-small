    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    population.append(h0)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(5):
        h = np.random.rand(n_points)
        h = np.concatenate([h, h[::-1]])
        h = h * target_sum / np.sum(h)
        population.append(h[:n_points])
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([h * target_sum / np.sum(h) for h in population])

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        tournament = np.random.choice(len(population), size=5, replace=False)
        parent1, parent2 = population[tournament[0]], population[tournament[1]]
        cross_idx = np.random.randint(n_points)
        child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]])

        # Targeted mutation
        corr = np.correlate(child, 1 - child, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        if lag >= 0:
            valid_x = np.arange(0, n_points - lag)
        else:
            valid_x = np.arange(-lag, n_points)
        if len(valid_x) > 0:
            x = np.random.choice(valid_x)
            delta = 0.01 * child[x]
            child[x] -= delta
            child[x + lag] += delta
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)

        # Replace worst
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
