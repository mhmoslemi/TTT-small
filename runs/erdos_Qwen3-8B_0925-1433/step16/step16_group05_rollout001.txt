import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vector):
        h_clipped = np.clip(h_vector, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return float('inf')
        h_scaled = h_clipped * (target_sum / sum_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        maxiter=200,
        tol=1e-5,
        seed=seed
    )

    best_h = result.x
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_clipped = np.sum(best_h_clipped)
    if sum_clipped == 0:
        c5_bound = float('inf')
    else:
        best_h_scaled = best_h_clipped * (target_sum / sum_clipped)
