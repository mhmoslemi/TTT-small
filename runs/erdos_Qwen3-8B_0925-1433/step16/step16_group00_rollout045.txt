                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                grad[i] = (corr_plus[k_max] - corr_minus[k_max]) / (2 * pert)
            return grad

        res = minimize(obj, h, method='SLSQP', jac=grad, bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)

        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
            corr_new = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = corr_new[k_max] * dx
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
        iteration += 1

    return (best_h, best_c5, n_points)
