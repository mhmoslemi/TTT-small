            best_c5 = fitness[best_idx]

        # Tournament selection
        tournament = np.random.choice(len(population), size=5, replace=False)
        parent1, parent2 = population[tournament[0]], population[tournament[1]]
        cross_idx = np.random.randint(n_points)
        child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]]) 

        # Targeted mutation
        corr_child = np.correlate(child, 1 - child, mode='full')
        i_max = np.argmax(corr_child)
        k = i_max - (n_points - 1)
        abs_k = abs(k)
        if abs_k == 0:
            target_positions = list(range(n_points))
        else:
            if k > 0:
                x_indices = np.arange(n_points - abs_k)
                target_positions = list(x_indices) + list(x_indices + abs_k)
            else:
                x_indices = np.arange(abs_k, n_points)
                target_positions = list(x_indices) + list(x_indices - abs_k)
        selected = np.random.choice(target_positions, size=min(5, len(target_positions)), replace=False)
        for i in selected:
            child[i] *= 0.95
        current_sum = np.sum(child)
        if current_sum > 0:
            child = child * (target_sum / current_sum)
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)

        # Replace worst
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
