import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

        b = np.random.choice([0, 1], size=n_points)
        s = np.sum(b)
        if s == 0:
            return np.zeros(n_points)
        return b * target_sum / s

    def gen_alt():
        a = np.zeros(n_points)
        a[::2] = 1.0
        s = np.sum(a)
        if s == 0:
            return np.zeros(n_points)
        return a * target_sum / s

    def gen_unif():
        return np.full(n_points, 0.5)

    pop = [np.array(initial_h_values, dtype=float)]
    pop.append(gen_binary())
    pop.append(gen_unif())
    pop.append(gen_alt())
    while len(pop) < 50:
        c = np.random.rand(n_points)
        c = c * target_sum / np.sum(c)
        pop.append(c)

    def eval_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()

    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        fit = [eval_c5(h) for h in pop]
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5, best_h = fit[best_idx], pop[best_idx].copy()

        new_pop = []
        for _ in range(25):
            p1, p2 = np.random.choice(range(len(pop)), size=2, replace=False)

            s = np.sum(c1)
            if s == 0:

            else:
                c1 *= target_sum / s
            new_pop.append(c1)

        pop = new_pop + [pop[best_idx]]

    return (best_h, best_c5, n_points)
