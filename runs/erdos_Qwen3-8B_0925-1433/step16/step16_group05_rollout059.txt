import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros(n_points)
        return h_clipped * (target_sum / sum_clipped)

    h_initial = np.array(initial_h, dtype=float)
    h_initial = clip_scale(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def compute_obj(h):
        h_clipped = clip_scale(h)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = clip_scale(h)
        population.append(h)

    max_gen = 300
    elite = 10
    tourn_size = 5
    mut_rate = 0.8
    cross_rate = 0.9

    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fit = [compute_obj(ind) for ind in population]
        best_idx = np.argsort(fit)[:elite]
        best_pop = [population[i] for i in best_idx]
        offspr = []

        for _ in range(len(population) - elite):
            a = np.random.choice(population)
            b = np.random.choice(population)
            if np.random.rand() < cross_rate:
                child = (a + b) / 2
            else:
                child = a.copy()
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.05, size=n_points)
                child += noise
                child = clip_scale(child)
            offspr.append(child)

        new_pop = best_pop + offspr
        fit_new = [compute_obj(ind) for ind in new_pop]
        new_pop.sort(key=lambda x: fit_new[pop.index(x)])
        population = new_pop[:pop_size]

        top10 = population[:int(len(population)*0.1)]
        for h in top10:
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_perturb = h.copy()
                h_perturb[i] += eps
                obj_perturb = compute_obj(h_perturb)
                h_perturb[i] -= 2 * eps
                obj_neg = compute_obj(h_perturb)
                grad[i] = (obj_perturb - obj_neg) / (2 * eps)
            step = 0.05
            h -= step * grad
            h = clip_scale(h)

        current_best = min(fit)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fit)].copy()

    return (best_h, best_c5, n_points)
