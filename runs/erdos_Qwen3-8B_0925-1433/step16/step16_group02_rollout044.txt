import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_h = initial_h_values.copy()
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h
    best_c5 = compute_c5(best_h)

    population = [initial_h]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        evaluations = [(compute_c5(h), h) for h in population]
        evaluations.sort()
        elites = [h for _, h in evaluations[:5]]
        new_population = elites
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (np.random.uniform(0.4, 0.6) * p1 + (1 - np.random.uniform(0.4, 0.6)) * p2)
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population

    top_candidates = [h for _, h in evaluations[:5]]
    for h in top_candidates:
        res = minimize(compute_c5, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=compute_gradient, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
