from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def rescale(h):
        return np.clip(h * target_sum / np.sum(h), 0, 1)

        pop.append(rescale(0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points) / 2)))
        pop.append(rescale(np.clip((np.linspace(0, 2, n_points) - 1) / 1.0, 0, 1)))
        pop.append(rescale(np.random.rand(n_points)))

    def eval_c5(h):
        return np.max(np.correlate(h, 1-h, mode='full')) * dx

    best_h = base
    best_c5 = eval_c5(best_h)

    start = time.time()
    for gen in range(100):
        if time.time() - start > budget_s - 10:

        fit = np.array([eval_c5(h) for h in pop])

            best_h, best_c5 = pop[0], eval_c5(pop[0])

    def obj(x):
        return eval_c5(x)

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        g = np.zeros(n_points)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:

            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                g[i] += h[j_minus] * dx
        return g

    if time.time() - start < budget_s - 50:
