import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        population = [h_initial]
    else:
        population = [np.full(n_points, 0.5)]
    for _ in range(population_size - 1):
        h_bin = np.random.rand(n_points)
        h_bin = (h_bin < 0.5).astype(float)
        diff = 50 - np.sum(h_bin)
        if diff > 0:
            idx = np.where(h_bin == 0)[0][:diff]
            h_bin[idx] = 1.0
        elif diff < 0:
            idx = np.where(h_bin == 1)[0][:abs(diff)]
            h_bin[idx] = 0.0
        h_bin = h_bin * target_sum / np.sum(h_bin)
        population.append(h_bin)
    population_size = len(population)
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        top_indices = np.argsort(fitness)[:int(0.8 * population_size)]
        top_candidates = [population[i] for i in top_indices]
        new_population = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(top_candidates, 2, replace=False)
            child = (p1 + p2) / 2.0
            if np.random.rand() < 0.1:
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                child[idx1], child[idx2] = child[idx2], child[idx1]
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        combined = top_candidates + new_population
        combined_fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in combined]
        combined_with_fitness = list(zip(combined, combined_fitness))
        combined_with_fitness.sort(key=lambda x: x[1])
        population = [h for h, _ in combined_with_fitness[:population_size]]
        current_best = population[0]

        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
