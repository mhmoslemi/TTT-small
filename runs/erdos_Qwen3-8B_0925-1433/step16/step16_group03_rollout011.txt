    h_interp = interp1d(x_initial, initial_h_values, kind='linear', fill_value='extrapolate')(x_new)
    h_base = np.clip(h_interp, 0, 1) * target_sum / np.sum(h_interp)
    
    initial_particles = [h_base]
    particles = [h_base.copy()]
    velocities = [np.random.uniform(-0.05, 0.05, n_points)]
    
    for _ in range(15):
        new_particles = []
        new_velocities = []
        for i in range(len(particles)):
            v = velocities[i] + np.random.normal(0, 0.05, n_points)
            h = particles[i] + v
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            new_particles.append(h)
            new_velocities.append(v)
        particles = new_particles
        velocities = new_velocities
    
    best_h = particles[0]
    best_c5 = float('inf')
    for h in particles:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        pert = np.random.normal(0, 0.02, n_points)
        h_pert = best_h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        
        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        res = minimize(objective, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_h = res.x
            corr = np.correlate(curr_h, 1 - curr_h, mode='full')
            curr_c5 = np.max(corr) * dx
            if curr_c5 < best_c5:
                best_h = curr_h
                best_c5 = curr_c5
    
    if time.time() - start_time < budget_s - 200:
        n_points = 150
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        x_upscale = np.linspace(0, 2, n_points)
        x_base = np.linspace(0, 2, 100)
        h_upscale = interp1d(x_base, best_h, kind='linear', fill_value='extrapolate')(x_upscale)
        h_upscale = np.clip(h_upscale, 0, 1)
        h_upscale = h_upscale * target_sum / np.sum(h_upscale)
        
        def obj_upscale(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        cons_upscale = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds_upscale = [(0, 1) for _ in range(n_points)]
        
        res_upscale = minimize(obj_upscale, h_upscale, method='L-BFGS-B', bounds=bounds_upscale, constraints=cons_upscale, tol=1e-6)
        if res_upscale.success:
            best_h = res_upscale.x
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            best_c5 = np.max(corr) * dx
    
    best_h = np.clip(best_h, 0, 1)
    sum_check = np.sum(best_h)
    if abs(sum_check - target_sum) > 1e-6:
        best_h = best_h * target_sum / sum_check
    return (best_h, best_c5, n_points)
