import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        return h * (target_sum / np.sum(h))

    # Generate initial patterns
    initial_patterns = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points, endpoint=False)
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x)
    initial_patterns.append(clip_rescale(h_sine))
    h_pulse = np.zeros(n_points)
    h_pulse[:5] = h_pulse[10:15] = 1.0
    initial_patterns.append(clip_rescale(h_pulse))
    h_random = np.random.binomial(1, 0.1, n_points)
    initial_patterns.append(clip_rescale(h_random))
    for _ in range(len(initial_patterns)):
        h = initial_patterns[0] + np.random.normal(0, 0.05, n_points)
        initial_patterns.append(clip_rescale(h))

    population = initial_patterns[:30]
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        fit = [-compute_c5(h) for h in population]
        new_pop = []
        for _ in range(30):
            idxs = np.random.choice(len(population), 5, replace=False)
            best_idx = np.argmax([fit[i] for i in idxs])
            p1 = population[idxs[best_idx]]
            p2 = population[np.random.choice(len(population))]
            child = (p1 + p2) / 2 + np.random.normal(0, 0.05, n_points)
            child = clip_rescale(child)
            new_pop.append(child)
        population = new_pop
        for h in population:
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_h, best_c5 = h, curr_c5

    def obj(x):
        return compute_c5(clip_rescale(x))

    bounds = [(0, 1) for _ in range(n_points)]

    return (np.clip(best_h, 0, 1), best_c5, n_points)
