import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h, target_sum):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    base_patterns = [initial_h_values]
    golden = (1 + np.sqrt(5)) / 2
    rotated_indices = np.sort((np.arange(n_points) * golden) % n_points).astype(int)
    rotated_h = np.zeros(n_points)
    rotated_h[rotated_indices] = initial_h_values[np.argsort(rotated_indices)]
    base_patterns.append(rotated_h)
    base_patterns.append(np.full(n_points, 0.5))

    population = []
    for pattern in base_patterns:
        for _ in range(5):
            perturbed = pattern + np.random.normal(0, 0.1, n_points)
            perturbed = clip_and_rescale(perturbed, target_sum)
            population.append(perturbed)
    population = [np.array(p, dtype=float) for p in population]

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            result = minimize(lambda x: compute_c5(x), p, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=constraints)
            if result.success:
                new_p = clip_and_rescale(result.x, target_sum)
                current_c5 = compute_c5(new_p)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = new_p
                    if best_c5 < 0.3807:
                        break
        else:
            continue
        break

    h_values = clip_and_rescale(best_h, target_sum)
    return (h_values, best_c5, n_points)
