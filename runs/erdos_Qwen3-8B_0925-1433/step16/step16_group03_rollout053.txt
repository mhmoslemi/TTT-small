import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_sine_pattern(n):
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        return h * (n / 2.0) / np.sum(h)

    def create_binary_pattern(n):
        h = np.zeros(n)
        h[:n//2] = 1.0
        return h

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return evaluate_c5(x)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    candidates = [
        np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values),
        create_sine_pattern(n_points),
        create_binary_pattern(n_points),
        initial_h_values + np.random.normal(0, 0.1, n_points),
    ]
    for v in candidates:
        v = np.clip(v, 0, 1)
        v = v * target_sum / np.sum(v)
        candidates.append(v)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_c5(best_h)

    for c in candidates:
        res = minimize(objective, c, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            h_new = res.x
            c_new = evaluate_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            h_new = res.x
            c_new = evaluate_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
