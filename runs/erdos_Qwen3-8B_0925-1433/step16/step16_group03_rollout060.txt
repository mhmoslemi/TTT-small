import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Generate structured patterns
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * target_sum / np.sum(h_sine)

    h_pulses = np.zeros_like(x)
    pos = [0.25, 0.75, 1.25, 1.75]
    for p in pos:
        mask = np.abs(x - p) < 0.25
        h_pulses[mask] = 1.0
    h_pulses = np.clip(h_pulses, 0, 1)
    h_pulses = h_pulses * target_sum / np.sum(h_pulses)

    h_rand = np.random.binomial(1, 0.1, n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    initial_population = [h_sine, h_pulses, h_rand, h_initial]
    pop_size = 30
    population = initial_population + [np.random.rand(n_points) for _ in range(pop_size - len(initial_population))]

    def evaluate(ind):
        h = ind.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx

    start_time = time.time()
    for gen in range(75):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([evaluate(ind) for ind in population])
        elite = np.argsort(scores)[:5]
        new_pop = [population[i] for i in elite]
        while len(new_pop) < pop_size:
            a, b = np.random.choice(pop_size, 2, replace=False)
            p1, p2 = population[a], population[b]
            child = 0.7 * p1 + 0.3 * p2
            if np.random.rand() < 0.1:
                child += np.random.normal(0, 0.05, n_points)
            new_pop.append(child)
        population = new_pop

    best_idx = np.argmax([evaluate(ind) for ind in population])
    best_h = population[best_idx].copy()
    best_c5 = -evaluate(best_h)

    def refine(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(refine, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = -refine(best_h)

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(refine, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            best_h = res.x
            best_c5 = -refine(best_h)

    new_n = int(n_points * 1.5)
    x_new = np.linspace(0, 2, new_n)
    h_up = np.interp(x_new, x, best_h)
    h_up = h_up * (new_n / 2.0) / np.sum(h_up)
    res_up = minimize(refine, h_up, method='L-BFGS-B', bounds=[(0,1)]*new_n, constraints=cons, tol=1e-6)
    if res_up.success:
        best_h = res_up.x
        best_c5 = -refine(best_h)
        n_points = new_n

    return (np.clip(best_h, 0, 1), best_c5, n_points)
