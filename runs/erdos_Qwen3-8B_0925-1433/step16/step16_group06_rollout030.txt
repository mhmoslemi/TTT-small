from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    best_c5 = min(best_c5, 0.381)

    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    for p in population:
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Evaluate best candidate without gradient
    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            grad[i] = (1 - h[j]) * dx_val - h[(i - k_max) % len(h)] * dx_val
        return grad
    
    start_time = time.time()
    for _ in range(300):  # More iterations for better convergence
        if time.time() - start_time > budget_s:
            break
        # Optimize each candidate
        for i in range(pop_size):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points,

            if res.success:
