import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return h * target_sum / total if total != 0 else np.full_like(h, 0.5)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_population(base, num=50):
        pop = []
        for _ in range(num):
            h = base.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = project(h)
            pop.append(h)
        return pop

    base = np.array(initial_h_values, dtype=float)
    base = project(base)
    population = generate_population(base, 50)

    start_time = time.time()
    best_h = base.copy()
    best_c5 = obj(best_h)

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        if np.random.rand() < 0.7:
            new_pop = []
            for p in population:
                p_mut = p + np.random.normal(0, 0.05, n_points)
                p_mut = project(p_mut)
                new_pop.append(p_mut)
            population = new_pop
        else:
            parents = np.random.choice(population, size=2, replace=False)
            child = (parents[0] + parents[1]) / 2
            child = project(child)
            population = [p for p in population if np.random.rand() < 0.8] + [child]

        for p in population:
            res = minimize(obj, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
            if res.success:
                p[:] = res.x

        current_c5 = obj(population[0])
        if current_c5 < best_c5:

            best_c5 = current_c5

    h_values = best_h
    return (h_values, best_c5, n_points)
