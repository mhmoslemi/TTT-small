    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    for pi in patterns:
        h0 = pi.copy()
        h0 = np.clip(h0, 0, 1)
        res = minimize(obj, h0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=grad_func)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        if time.time() - start_time > budget_s:
            break

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-9, jac=grad_func)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
