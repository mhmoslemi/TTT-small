def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5_with_lag(h):

        max_val = np.max(corr)
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        return max_val * dx, abs(lag)

    # Initialize population
    population = []
    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    population.append(h0)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(5):
        h = np.random.rand(n_points)
        h = np.concatenate([h, h[::-1]])
        h = h * target_sum / np.sum(h)
        population.append(h[:n_points])
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([h * target_sum / np.sum(h) for h in population])

    best_h = population[0].copy()
    best_c5 = compute_c5_with_lag(best_h)[0]
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Fitness evaluation
        fitness = np.array([compute_c5_with_lag(h)[0] for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:

            best_c5 = fitness[best_idx]

        # Tournament selection
        tournament = np.random.choice(len(population), size=5, replace=False)
        parent1, parent2 = population[tournament[0]], population[tournament[1]]
        cross_idx = np.random.randint(n_points)

        # Targeted mutation
        max_val, lag = compute_c5_with_lag(child)
