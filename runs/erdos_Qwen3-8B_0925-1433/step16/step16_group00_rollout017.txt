def run(seed=42, budget_s=600):

                end = (i + 1) * spacing

                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
            intervals = intervals * target_sum / np.sum(intervals)
            patterns.append(intervals)
        for _ in range(3):
            h = np.random.binomial(1, 0.5, n_points)
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        for _ in range(3):
            pert = np.random.uniform(-0.1, 0.1, n_points)
            h = h_base + pert
            h = np.clip(h, 0, 1) * target_sum / np.sum(h)
            patterns.append(h)
        return patterns

    start_time = time.time()
