        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = []
    for base in base_patterns:
        for _ in range(5):
            perturbed = base * (1 + 0.2 * np.random.randn(n_points))
            perturbed = project(perturbed)
            population.append(perturbed)
    for _ in range(20):
        new_h = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        population.append(project(new_h))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for h in population:
            res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, tol=1e-6)
            if res.success:
                h[:] = project(res.x)
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_h = project(population[0])
            best_c5 = current_c5

    return (project(best_h), best_c5, n_points)
