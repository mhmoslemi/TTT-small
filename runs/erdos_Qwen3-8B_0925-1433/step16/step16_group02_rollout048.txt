from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

            np.clip(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)), 0, 1),
            np.clip(np.tile([1, 0], (n_points // 2 + 1, 1))[:, :n_points] * target_sum / np.sum(np.tile([1, 0], (n_points // 2 + 1, 1))[:, :n_points]), 0, 1)
        ]

    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

        if c5 < best_c5:

            best_c5 = c5

        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    try:
        result = differential_evolution(

                best_h = best_de_h.copy()

        pass

        h = x * target_sum / np.sum(x)
