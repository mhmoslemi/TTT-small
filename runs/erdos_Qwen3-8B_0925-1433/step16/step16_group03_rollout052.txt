import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = []

    h_sine = h_sine * target_sum / np.sum(h_sine)
    population.append(h_sine)
    
    h_pulse = np.zeros(n_points)
    positions = np.random.choice(n_points, size=10, replace=False)
    h_pulse[positions] = 1.0
    h_pulse = h_pulse * target_sum / np.sum(h_pulse)
    population.append(h_pulse)
    
    h_binary = np.random.binomial(1, 0.1, n_points)

        h_pert = h_pert * target_sum / np.sum(h_pert)
        population.append(h_pert)
    
    population = [np.clip(p, 0, 1) for p in population]
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    # Genetic Algorithm
    for _ in range(75):
        if time.time() - start_time > budget_s:
            break
        fitness = [-compute_c5(h) for h in population]

        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
