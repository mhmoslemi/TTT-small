import numpy as np
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_initial():
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(ind):
        noise = np.random.normal(0, 0.1, ind.shape)
        h = ind + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(p1, p2):
        child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
        child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
        return child1, child2

    population = [create_initial() for _ in range(50)]
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        fit = [compute_c5(ind) for ind in population]
        if min(fit) < best_c5:
            best_c5 = min(fit)
            best_h = population[fit.index(min(fit))]
        parents = [random.choice(population) for _ in range(50)]
        new_pop = []
        for _ in range(25):
            p1, p2 = random.sample(parents, 2)
            c1, c2 = crossover(p1, p2)
            new_pop.append(mutate(c1))
            new_pop.append(mutate(c2))
        population = new_pop
        gen += 1
    return (best_h, best_c5, n_points)
