def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():

        h_initial = h_initial * target_sum / np.sum(h_initial)

        guesses.append(np.zeros(n_points, dtype=float))
        guesses[-1][indices[:int(n_points/2)]] = 1.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0

            grad[i] = dx * (term1 - term2)
        return grad

    best_c5 = float('inf')
    start_time = time.time()
