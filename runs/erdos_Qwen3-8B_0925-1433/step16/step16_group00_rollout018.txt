import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def eval_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def normalize(h):
        s = np.sum(h)
        return h * target_sum / s if s != 0 else h

    # Initialize population
    pop_size = 50
    population = []
    base = np.array(initial_h_values, dtype=float)
    for _ in range(pop_size):
        h = base.copy()
        if np.random.rand() < 0.5:
            h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = normalize(h)
        population.append(h)

    best_h = base.copy()
    best_c5 = eval_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    generation = 0
    no_improvement = 0
    max_generations = 200
    mutation_rate = 0.1

    while generation < max_generations and time.time() - start_time < budget_s:
        generation += 1
        new_population = []

        # Evaluate current population
        fitness = [eval_c5(ind) for ind in population]
        fit_pop = list(zip(population, fitness))
        fit_pop.sort(key=lambda x: x[1])
        fit_pop = fit_pop[:int(pop_size * 0.2)]  # Select top 20%

        # Generate new population
        while len(new_population) < pop_size:
            # Tournament selection
            a, b = np.random.randint(len(fit_pop)), np.random.randint(len(fit_pop))
            parent1, _ = fit_pop[a]
            parent2, _ = fit_pop[b]
            parent1, parent2 = parent1.copy(), parent2.copy()

            # Crossover
            if np.random.rand() < 0.8:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            else:
                child = np.random.rand(n_points)

            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            child = normalize(child)

            new_population.append(child)

        population = new_population

        # Track best
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitness.index(current_best)]
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= 10:
            break

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
