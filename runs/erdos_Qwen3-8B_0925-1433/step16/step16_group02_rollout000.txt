import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_square_wave(n_points):
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        if np.sum(h) < target_sum:
            excess = target_sum - np.sum(h)
            h[-int(np.ceil(excess)):] += 1.0
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_triangular(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h = np.where(x <= 1, x, 2 - x)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_random(n_points):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_symmetric(n_points):
        h = np.ones(n_points)
        h[::2] = 0
        h[1::2] = 1
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    initial_candidates = [
        initial_h_values,
        generate_square_wave(n_points),
        generate_triangular(n_points),
        generate_random(n_points),
        generate_symmetric(n_points)
    ]

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        h_scaled = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    best_h = initial_candidates[0]
    best_c5 = evaluate(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(evaluate, bounds, strategy='best1bin', popsize=16, maxiter=100, tol=1e-6, mutation=(0.8, 1.0), recombination=0.9)
    best_candidate = result.x
    current_c5 = evaluate(best_candidate)
    if current_c5 < best_c5:
        best_h = best_candidate
        best_c5 = current_c5

    def objective(x):
        h = x.copy()
        h_clip = np.clip(h, 0, 1)
        h_scaled = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x.copy()
        h_clip = np.clip(h, 0, 1)
        h_scaled = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h_scaled[j]) * dx
            j = i - k_max
            if 0 <= j < n_points:
                grad[i] += h_scaled[j] * dx
        return grad

    res = minimize(objective, best_h, method='L-BFGS-B', jac=gradient, bounds=[(0, 1)] * n_points, tol=1e-6)
    best_h = res.x
    best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
