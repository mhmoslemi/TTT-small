import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    # Initial patterns
    patterns = [h]
    # Two pulses
    p = np.zeros(n_points)
    p[round(0.4 * n_points)] = 0.8
    p[round(1.6 * n_points)] = 0.8
    p = p * target_sum / np.sum(p)
    patterns.append(p)
    # Mirrored
    m = np.concatenate([h[:n_points//2], h[:n_points//2][::-1]])
    m = m * target_sum / np.sum(m)
    patterns.append(m)

    for pat in patterns:
        if time.time() - start_time > budget_s:
            break
        h_init = np.clip(pat + np.random.normal(0, 0.05, n_points), 0, 1)
        h_init = h_init * target_sum / np.sum(h_init)
        res = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), current_c5

    # Refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_ref = np.clip(best_h, 0, 1)
        h_ref = h_ref * target_sum / np.sum(h_ref)

            if current_c5 < best_c5:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
