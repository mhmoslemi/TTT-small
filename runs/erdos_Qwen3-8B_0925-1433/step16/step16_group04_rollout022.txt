import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    iterations = 0
    max_iterations = 500
    no_improvement = 0

    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                gradient[i] += 1 - h[i + k]
            if 0 <= i - k < n_points:
                gradient[i] -= h[i - k]
        alphas = [0.5, 0.25, 0.1, 0.05, 0.025]
        best_temp = None
        best_c5_new = np.inf

        for alpha in alphas:
            direction = -gradient
            temp = h + alpha * direction
            temp = np.clip(temp, 0, 1)
            temp_sum = np.sum(temp)
            temp = temp * target_sum / temp_sum
            c5 = compute_c5(temp)
            if c5 < best_c5_new:
                best_c5_new = c5
                best_temp = temp.copy()

        if best_c5_new < best_c5:
            h = best_temp
            best_c5 = best_c5_new
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= 20:
            break
        iterations += 1

    best_h = np.clip(h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
