                best_c5 = curr_c5
    
    # Local refinement
    candidates = sorted(population, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)[:5]
    for cand in candidates:
        def obj(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, cand, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6)
        if res.success:
            h_refined = res.x
            curr_c5 = obj(h_refined)
            if curr_c5 < best_c5:
                best_h = h_refined
                best_c5 = curr_c5
    
    # Final clipping and normalization
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    return (final_h, best_c5, n_points)
