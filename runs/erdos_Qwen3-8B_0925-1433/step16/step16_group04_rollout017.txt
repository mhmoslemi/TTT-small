                child[j] += np.random.uniform(-0.15, 0.15) * child[j]

        # Clip and normalize
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)

        # Replace worst
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
