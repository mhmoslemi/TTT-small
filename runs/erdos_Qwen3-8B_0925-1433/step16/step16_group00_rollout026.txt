import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def init_population(size=50):
        pop = []
        base = np.array(initial_h_values, dtype=float)
        base = base * target_sum / np.sum(base)
        pop.append(base)
        for _ in range(size - 1):
            h = np.random.binomial(1, 0.5, n_points)
            h = h * target_sum / np.sum(h)
            pop.append(h)
            m = np.random.randint(5, 15)
            intervals = np.zeros(n_points)
            spacing = 2.0 / (m + 1)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
            intervals = intervals * target_sum / np.sum(intervals)
            pop.append(intervals)
        return pop

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, sigma=0.1):
        h = h.copy()
        noise = np.random.normal(0, sigma, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        mask = np.random.rand(n_points) < 0.5
        child[mask] = parent2[mask]
        return child

    pop = init_population()
    best_h = pop[0]
    best_c5 = evaluate(best_h)
    gen = 0

    start_time = time.time()
    while time.time() - start_time < budget_s:
        gen += 1
        if gen % 50 == 0:
            print(f"Gen {gen}, C5: {best_c5}")
        if gen > 100:
            break
        ranked = sorted(enumerate(pop), key=lambda x: evaluate(x[1]))
        top_half = ranked[:len(ranked)//2]
        new_pop = [x[1] for x in top_half]
        while len(new_pop) < len(pop):
            p1, p2 = np.random.choice(top_half, 2, replace=False)
            child = crossover(p1[1], p2[1])
