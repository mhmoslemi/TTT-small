import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return float('inf')
        h_scaled = h_clipped * (target_sum / sum_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population = [initial_h.copy()]
    for _ in range(49):
        h_rand = np.random.rand(n_points)
        h_scaled = h_rand * (target_sum / np.sum(h_rand))
        population.append(h_scaled)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    max_generations = 300
    early_stop = 10
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
        new_pop = []
        for _ in range(n_points):
            p1, p2, p3 = np.random.choice(population, 3, replace=False)
            mutant = p1 + 0.8 * (p2 - p3)
            child = (mutant + np.random.choice(population)) / 2
            h_clipped = np.clip(child, 0, 1)
            sum_clipped = np.sum(h_clipped)
            h_scaled = h_clipped * (target_sum / sum_clipped) if sum_clipped > 0 else np.zeros_like(h_clipped)
            new_pop.append(h_scaled)
        population = new_pop
        for h in population:
            val = objective(h)
            if val < best_c5:
                best_c5 = val
                best_h = h.copy()

    # Local refinement
    def grad_approx(h):
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    if best_c5 < 0.3807:
        h_refined = best_h.copy()
        for _ in range(10):
            grad = grad_approx(h_refined)
            h_refined += 0.05 * grad
            h_refined = np.clip(h_refined, 0, 1)
            sum_refined = np.sum(h_refined)
            h_refined = h_refined * (target_sum / sum_refined) if sum_refined > 0 else np.zeros_like(h_refined)
            best_c5 = objective(h_refined)
            best_h = h_refined.copy()
    return (np.clip(best_h, 0, 1), best_c5, n_points)
