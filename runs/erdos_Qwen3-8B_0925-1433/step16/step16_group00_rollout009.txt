import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        patterns = []
        h_base = np.array(initial_h_values, dtype=float)
        h_norm = h_base * target_sum / np.sum(h_base)
        patterns.append(h_norm)
        for _ in range(2):
            pert = np.random.uniform(-0.1, 0.1, n_points)
            h = h_base + pert
            h = np.clip(h, 0, 1) * target_sum / np.sum(h)
            patterns.append(h)
        for m in [5, 10, 15]:
            spacing = 2.0 / (m + 1)
            intervals = np.zeros(n_points)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
            intervals = intervals * target_sum / np.sum(intervals)
            patterns.append(intervals)
        for _ in range(3):
            h = np.random.binomial(1, 0.5, n_points)
            h = h * target_sum / np.sum(h)
            patterns.append(h)
        return patterns

    patterns = generate_patterns()
    best_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_func(x):

            h_plus[i] += 1e-4

            h_minus[i] -= 1e-4
