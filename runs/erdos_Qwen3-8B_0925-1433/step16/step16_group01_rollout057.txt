            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            res_perturb = minimize(
                fun=lambda x: compute_c5(x),
                x0=perturbed_h,
                method='L-BFGS-B',
                jac=compute_gradient,
                bounds=[(0, 1)]*n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            )
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h, best_c5 = res_perturb.x, res_perturb.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
