import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n = len(initial_h)
    target = n / 2.0
    dx = 2.0 / n

    def eval_h(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h) + 1.0 / n * target
        return h * target / s

    # Initial population
    pop = []
    h0 = initial_h * target / np.sum(initial_h)
    pop.append(h0)
    for m in [5, 10, 15]:
        intervals = np.zeros(n)
        for i in range(m):
            start = i * 2.0 / (m + 1)
            end = (i + 1) * 2.0 / (m + 1)
            s = int(start / dx + 0.5)
            e = int(end / dx + 0.5)
            intervals[s:e] = 1
        intervals = normalize(intervals)
        pop.append(intervals)
    for _ in range(5):
        h = np.random.binomial(1, 0.5, n) * target / np.sum(np.random.binomial(1, 0.5, n))
        pop.append(h)

    pop = [normalize(np.clip(p, 0, 1)) for p in pop]
    pop = pop[:50]
    random.shuffle(pop)

    best_h = pop[0]
    best_c5 = eval_h(best_h)
    start = time.time()

    for _ in range(200):
        if time.time() - start > budget_s:
            break
        fit = [eval_h(p) for p in pop]
        idx = np.argmin(fit)
        if fit[idx] < best_c5:
            best_h, best_c5 = pop[idx], fit[idx]

        new_pop = []
        for _ in range(len(pop)):
            a, b = random.choices(pop, k=2)
            child = np.zeros(n)
            for i in range(n):
                if random.random() < 0.5:
                    child[i] = a[i]
                else:
                    child[i] = b[i]
            for i in range(n):
                if random.random() < 0.1:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            child = normalize(child)
            new_pop.append(child)

        pop = new_pop

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n)
