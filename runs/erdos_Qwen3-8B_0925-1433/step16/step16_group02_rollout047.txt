        h = np.zeros(n_points)

            if i < n_points // 2:
                h[i] = base_h[i]
                h[n_points - 1 - i] = base_h[i]
            elif i == n_points // 2:
                h[i] = base_h[i]
            else:
                h[i] = h[n_points - 1 - i]
        return h

    h_base = np.array(initial_h_values, dtype=float)

            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
