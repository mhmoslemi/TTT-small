import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial population based on initial_h_values and variations
    population = [np.array(initial_h_values, dtype=float)]

        ind = np.clip(ind, 0, 1)
        ind = ind * target_sum / np.sum(ind)
        population.append(ind)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    # Genetic algorithm loop
    start_time = time.time()
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [compute_c5(ind) for ind in population]
        # Select top 20 individuals for breeding
        top_indices = np.argsort(fitness)[:20]
        top_pop = [population[i] for i in top_indices]
        new_population = []
        for _ in range(50):
            # Crossover
            p1, p2 = np.random.choice(top_pop, size=2, replace=False)
            alpha = np.random.uniform(0.6, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            # Mutation
            mutate_mask = np.random.rand(n_points) < 0.1
            child[mutate_mask] += np.random.normal(0, 0.05, np.sum(mutate_mask))
            child = project(child)
            new_population.append(child)
        population = new_population

        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:

            best_c5 = current_c5

    # Local optimization on best solution
    x0 = best_h.copy()
    def obj(x):
        h = x * target_sum / np.sum(x)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
