import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    base_patterns = [np.array(initial_h_values, dtype=float)]
    for _ in range(3):
        perturbed = np.random.rand(n_points) * 2 - 1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        base_patterns.append(perturbed)

    population = []
    for p in base_patterns:
        perturbed = p + np.random.normal(0, 0.1, n_points)
        perturbed = project(perturbed)
        population.append(perturbed)

    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            res = minimize(obj, p, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                p[:] = project(res.x)
        current_h = population[0]
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = current_h

    h_values = project(best_h)
