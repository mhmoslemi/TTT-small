        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = (1.0 - h[j_plus]) if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()

    for guess in initial_guesses:
        h = guess.copy()
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            if res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun

        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            noise = np.random.normal(0, 0.05, n_points)
            perturbed_h = h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points,
                           constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
