import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def init_population(size=50):
        pop = []
        base = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
        pop.append(np.clip(base, 0, 1))
        for _ in range(size-1):
            h = np.random.exponential(0.5, n_points)
            h = h * target_sum / np.sum(h)
            pop.append(np.clip(h, 0, 1))
            h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            h = h * target_sum / np.sum(h)
            pop.append(np.clip(h, 0, 1))
        return np.array([np.clip(h, 0, 1) for h in pop])

    def fitness(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    pop = init_population(50)
    best_idx = np.argmin([fitness(h) for h in pop])
    best_h = pop[best_idx].copy()
    best_c5 = fitness(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [fitness(h) for h in pop]
        selected = np.argsort(fitnesses)[:int(len(pop)*0.2)]
        parents = np.random.choice(selected, size=len(pop), replace=True)
        new_pop = []
        for i in range(0, len(pop), 2):
            p1, p2 = pop[parents[i]], pop[parents[i+1]]
            child = np.where(np.random.rand(n_points) < 0.8, p1, p2)
            mut = np.random.normal(0, 0.1, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = np.array(new_pop)
        curr_best = np.argmin([fitness(h) for h in pop])
        if fitness(pop[curr_best]) < best_c5:
            best_h = pop[curr_best].copy()
            best_c5 = fitness(best_h)
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
