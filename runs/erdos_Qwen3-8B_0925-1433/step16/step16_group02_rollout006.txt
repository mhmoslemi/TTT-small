            # Enforce symmetry
            for i in range(n_points // 2):
                current_h[n_points - 1 - i] = 1 - current_h[i]
            current_sum = np.sum(current_h)
            scaling = target_sum / current_sum
            current_h = current_h * scaling
            current_h = np.clip(current_h, 0, 1)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
