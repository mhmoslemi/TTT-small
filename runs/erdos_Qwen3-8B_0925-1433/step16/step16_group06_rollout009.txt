import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population = [np.array(initial_h_values, dtype=float) for _ in range(50)]
    for i in range(50):
        noise = np.random.normal(0, 0.1, n_points)
        population[i] += noise
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate and select top candidates
        ranked = sorted(population, key=compute_c5)
        top_half = ranked[:25]
        # Generate new candidates
        new_pop = []
        for _ in range(50):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(top_half, 2, replace=False)
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                parent = np.random.choice(top_half)
                noise = np.random.normal(0, 0.1, n_points)
                child = parent + noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
    
    # Final refinement with SLSQP
    def obj(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)
    
    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        dx_val = 2.0 / len(h)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            grad[i] = (1 - h[j]) * dx_val - h[(i - k_max) % len(h)] * dx_val
        return grad
    
    res = minimize(obj, best_h, method='SLSQP', jac=grad, bounds=[(0, 1)] * n_points,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   tol=1e-6)
    best_h = res.x
    best_c5 = obj(best_h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
