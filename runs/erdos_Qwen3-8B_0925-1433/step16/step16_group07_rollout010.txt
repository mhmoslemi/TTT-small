import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / n_points
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(49):
        h = initial_h_values + np.random.normal(0, 0.05, n_points)
        h = project(h)
        population.append(h)

    best_h = population[0]
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([obj(p) for p in population])
        top_idx = np.argsort(fitness)[:5]
        new_pop = [population[i] for i in top_idx]
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(top_idx, 2, replace=False)
            child = (population[p1] + population[p2]) / 2 + np.random.normal(0, 0.05, n_points)
            child = project(child)
            new_pop.append(child)
        population = new_pop[:50]
        for p in population:
            res = minimize(obj, p, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                population.append(res.x)
                population = [project(p) for p in population]
        current_c5 = obj(population[0])
        if current_c5 < best_c5:
