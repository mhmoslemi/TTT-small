import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    population = [np.copy(h) + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    population = [project(p) for p in population]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]

    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        for i in range(50):
            p = population[i].copy()
            res = minimize(obj, p, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                p_new = project(res.x)
                if np.abs(np.sum(p_new) - target_sum) < 1e-6:

        current_c5 = np.max(np.correlate(project(population[0]), 1 - project(population[0]), mode='full')) * dx
        if current_c5 < best_c5:

            best_c5 = current_c5

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
