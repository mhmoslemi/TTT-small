    m = (n_points + 1) // 2  # number of variables to optimize

    x = np.zeros(m)
    for i in range(m):
        if i < m - 1:
            x[i] = initial_h_values[i]
        else:
            x[i] = 0.5

    def generate_h(x):
        h = np.zeros(n_points)
        for i in range(m):
            if i < m - 1:
                h[i] = x[i]
                h[n_points - 1 - i] = 1.0 - x[i]
            else:
                h[i] = x[i]
        return h

    def obj(x):
        h = generate_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = generate_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(m):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1.0 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    best_x = x.copy()

    start_time = time.time()

        if time.time() - start_time > budget_s:
