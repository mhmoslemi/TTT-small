import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_triangular():
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i] / 1.0
            else:
                h[i] = (2 - x[i]) / 1.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_random():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)

    population = [initial_h.copy()]
    for _ in range(4):
        population.append(generate_sinusoidal())
        population.append(generate_triangular())
        population.append(generate_random())

    pop_size = min(50, len(population))
    best_h = initial_h
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        elite_mask = np.argsort(fitness)[:int(pop_size * 0.2)]
        elites = [population[i] for i in elite_mask]
        new_population = elites[:]
        while len(new_population) < pop_size:
            a, b = np.random.choice(elites, size=2, replace=False)
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * a + (1 - alpha) * b
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        current_best = min(zip(population, fitness), key=lambda x: x[1])
        if current_best[1] < best_c5:
            best_h, best_c5 = current_best

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
