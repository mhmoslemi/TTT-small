from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    population = [np.copy(h) + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    population = [project(p) for p in population]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:

        fitness = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in population]
        top_idx = np.argsort(fitness)[-20:]
        top = [population[i] for i in top_idx]
