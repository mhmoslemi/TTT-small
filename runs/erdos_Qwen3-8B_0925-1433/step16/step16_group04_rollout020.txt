    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    alpha = 0.1
    decay_rate = 0.999
    max_iterations = 500

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k = m - (N - 1)
        gradient = np.zeros(N)
        for i in range(N):
            if 0 <= i + k < N:
                gradient[i] += 1.0 - h[i + k]
            if 0 <= i - k < N:
                gradient[i] -= h[i - k]
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        alpha *= decay_rate

    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, N)
