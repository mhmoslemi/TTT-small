import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_candidate(base):
        h = base.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    population = [generate_candidate(initial_h_values) for _ in range(50)]
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    max_gen = 200

    for _ in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        current_fitness = [compute_c5(h) for h in population]
        top_indices = np.argsort(current_fitness)[:10]
        top_candidates = [population[i] for i in top_indices]
        new_population = []
        for _ in range(50):
            parent1, parent2 = np.random.choice(top_candidates, 2, replace=False)
            split = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:split], parent2[split:]])
            noise = np.random.normal(0, 0.1, n_points)
            child += noise

            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        for h in population:
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5

    def obj(x):
        h = x.copy() * target_sum / np.sum(x)
        return compute_c5(h)

        max_idx = np.argmax(corr)
