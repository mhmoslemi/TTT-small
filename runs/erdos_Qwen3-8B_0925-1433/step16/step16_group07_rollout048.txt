import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return np.full_like(x, 0.5)
        return x * target_sum / total

    def bounds_func(x):
        return np.array([0.0] * len(x), dtype=float)

    base_patterns = [initial_h_values]

    # Generate a rotated pattern using golden ratio
    shift = int(0.618 * n_points)
    rotated = np.roll(initial_h_values, shift)
    base_patterns.append(rotated)

    # Create initial population with diverse patterns
    population = []
    for base in base_patterns:
        # Apply small perturbations
        for _ in range(5):
            perturbed = base + np.random.normal(0, 0.05, n_points)
            perturbed = project(perturbed)
            population.append(perturbed)

    # Optional: add more base patterns if needed
    # For brevity, we'll stick with 2 base patterns

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    start_time = time.time()

    # Multi-scale exploration loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            # Use L-BFGS-B with bounds and equality constraint
            res = minimize(objective, p, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res.success:
                new_p = project(res.x)
                if np.abs(np.sum(new_p) - target_sum) < 1e-6:
                    p[:] = new_p
                    current_c5 = objective(new_p)
                    if current_c5 < best_c5:
                        best_h = new_p.copy()
                        best_c5 = current_c5

    # Final check
    best_h = project(best_h)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
