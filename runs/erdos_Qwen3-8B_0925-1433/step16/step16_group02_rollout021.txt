import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)
    
    def generate_triangular():
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1.0, max(0.0, x[i] / 1.0) if x[i] <= 1 else (2 - x[i]) / 1.0)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)
    
    def generate_random():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)
    
    initial_population = [
        h_initial, generate_sinusoidal(), generate_triangular(), generate_random()
    ]
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    # Run genetic algorithm
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [(evaluate_c5(h), h) for h in initial_population]
