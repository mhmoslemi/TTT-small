        cnt = np.sum(sin_pat)
        sin_pat = sin_pat * target_sum / cnt

    patterns = generate_initial_patterns()

    best_h = None

    start_time = time.time()

    for pattern in patterns:

        pattern = pattern * target_sum / np.sum(pattern)
        h = pattern.copy()

        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def compute_gradient(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(x)
            for i in range(n_points):
                j_plus = i + k_max
                if 0 <= j_plus < n_points:
                    term1 = 1 - h[j_plus]

                j_minus = i - k_max
                if 0 <= j_minus < n_points:

                else:

                grad[i] = dx * (term1 - term2)
