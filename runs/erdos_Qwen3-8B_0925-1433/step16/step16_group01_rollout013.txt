import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial patterns: uniform, sine, random, and mirror
    def create_uniform(n_points, m):
        h = np.zeros(n_points)
        spacing = n_points // m
        for i in range(m):
            h[i * spacing] = 1
        return h

    def create_sine(n_points, m):
        x = np.linspace(0, 2, n_points)
        wave = np.sin(2 * np.pi * x / 0.5)
        h = (wave > 0.5).astype(float)
        count = int(np.sum(h))
        if count > m:
            h -= (count - m) / n_points
        elif count < m:
            h += (m - count) / n_points
        return h

    def create_random(n_points, m):
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, m, replace=False)
        h[idxs] = 1
        return h

    initial_patterns = [np.array(initial_h_values, dtype=float)]
    h_uniform = create_uniform(n_points, int(np.round(target_sum)))
    initial_patterns.append(h_uniform)
    h_sine = create_sine(n_points, int(np.round(target_sum)))
    initial_patterns.append(h_sine)
    h_random = create_random(n_points, int(np.round(target_sum)))
    initial_patterns.append(h_random)
    h_mirror = np.flip(initial_h_values).copy()
    initial_patterns.append(h_mirror)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')
    start_time = time.time()

    for pattern in initial_patterns:
        pattern = pattern * target_sum / np.sum(pattern)
        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def grad(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            g = np.zeros_like(x)
            for i in range(n_points):

                t2 = h[j_minus] if 0 <= j_minus < n_points else 0
                g[i] = dx * (t1 - t2)
            return g

        if res.success:
            if res.fun < best_c5:

    best_h = best_h * target_sum / np.sum(best_h)
