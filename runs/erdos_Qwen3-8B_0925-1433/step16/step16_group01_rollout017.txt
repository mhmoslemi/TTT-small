import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initializations():
        if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
            h = np.array(initial_h_values, dtype=float)
        else:
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * dx
                seg = int((x) * 5)
                if seg % 2 == 0:
                    h[i] = 1.0
            h = h * target_sum / np.sum(h)
        # Symmetric
        h_sym = np.zeros(n_points)
        h_sym[:n_points//2] = h[:n_points//2]
        h_sym[n_points//2:] = h[:n_points//2][::-1]
        h_sym = h_sym * target_sum / np.sum(h_sym)
        # Random tiling
        h_tile = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            seg = int((x) * 5)
            if seg % 2 == 0:
                h_tile[i] = 1.0
        h_tile = h_tile * target_sum / np.sum(h_tile)
        # Random binary
        h_rand = np.random.binomial(1, 0.5, n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        return [h, h_sym, h_tile, h_rand]

    initial_guesses = generate_initializations()
    best_h = None
    best_c5 = float('inf')

    start_time = time.time()

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        h = x * target_sum / np.sum(x)
        return evaluate(h)

    def compute_gradient(x):
        x = np.clip(x, 0, 1)
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    for init in initial_guesses:
        h = init.copy()
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = res.x.copy(), current_c5
                h = res.x.copy()
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            noise = np.random.normal(0, 0.05, n_points)
            perturbed_h = h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
