import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    m = n_points / 2.0

    def compute_c5(h):
        h_scaled = h * m / np.sum(h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_c5(x)

    def gradient(x):
        h = x * m / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * m / np.sum(best_h)
    best_c5 = compute_c5(best_h)

    patterns = [
        np.array(initial_h_values, dtype=float),
        np.linspace(0, 2, n_points, endpoint=False) * m / 2,
        np.random.rand(n_points) * m / np.sum(np.random.rand(n_points))
    ]

    start_time = time.time()
    for pattern in patterns:
        pattern = pattern * m / np.sum(pattern)
        result = differential_evolution(
            objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=50, popsize=20, tol=1e-5, mutation=(0.5, 1), recombination=0.9)
        if result.success:
            h = result.x * m / np.sum(result.x)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

        result = minimize(objective, pattern, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=gradient, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - m})
        if result.success:
            h = result.x * m / np.sum(result.x)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * m / np.sum(best_h)
    return (best_h, best_c5, n_points)
