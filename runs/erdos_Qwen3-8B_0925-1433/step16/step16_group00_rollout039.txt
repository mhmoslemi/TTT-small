import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, sigma):
        h = h.copy()
        for i in range(n_points):
            h[i] += np.random.normal(0, sigma)
        return normalize(h)

    def crossover(a, b):
        c = a.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                c[i] = b[i]
        return c

    def generate_initial_population(size):
        pop = []
        base = np.array(initial_h_values, dtype=float)
        base = normalize(base)
        for _ in range(size):
            h = base.copy()
            if np.random.rand() < 0.2:
                h = np.random.binomial(1, 0.5, n_points)
                h = normalize(h)

                m = np.random.randint(5, 15)
                spacing = 2.0 / (m + 1)
                h = np.zeros(n_points)
                for i in range(m):
                    start = i * spacing
                    end = (i + 1) * spacing
                    idx_start = int(start / dx + 0.5)
                    idx_end = int(end / dx + 0.5)
                    if idx_start < n_points:
                        h[idx_start:idx_end] = 1
                h = normalize(h)
            else:
                pert = np.random.normal(0, 0.05, n_points)
                h = normalize(base + pert)
            pop.append(h)
        return pop

    pop_size = 50
    pop = generate_initial_population(pop_size)
    best_h = pop[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(ind) for ind in pop])
        inds = np.argsort(fitness)
        elites = pop[inds[:5]]
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
