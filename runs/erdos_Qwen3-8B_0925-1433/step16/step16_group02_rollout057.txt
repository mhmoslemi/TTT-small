            h_clipped = np.clip(res.x, 0, 1)
            s = np.sum(h_clipped)
            h_scaled = h_clipped * (target_sum / s) if s > 0 else h_clipped
            c5 = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
            if c5 < best_c5:
                best_h, best_c5 = h_scaled, c5

    h_clipped = np.clip(best_h, 0, 1)
    s = np.sum(h_clipped)
    best_h = h_clipped * (target_sum / s) if s > 0 else h_clipped
    return (best_h, best_c5, n_points)
