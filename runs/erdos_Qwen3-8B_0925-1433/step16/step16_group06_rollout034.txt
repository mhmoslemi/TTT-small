            cand = np.clip(cand, 0, 1)
            cand = cand * target_sum / np.sum(cand)
        
        # Optimize candidates
        for cand in candidates:
            res = minimize(obj, cand, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6)
            if res.success:
                curr_c5 = res.fun
                if curr_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = curr_c5
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
