import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    # Generate structured patterns
    def make_binary_pulses(n_points):
        h = np.zeros(n_points)
        for i in range(4):
            idx = int(0.4 + i * 0.25)
            if 0 <= idx < n_points:
                h[idx] = 1.0
        return h

    def make_sinusoidal(n_points):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(np.pi * x)
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def make_random_pattern(n_points):
        h = np.random.rand(n_points)
        h = h * 0.2
        h = h * target_sum / np.sum(h)
        return h

    patterns = [initial_h_values, make_binary_pulses(n_points), make_sinusoidal(n_points), make_random_pattern(n_points)]

    for p in patterns:
        h = np.array(p, dtype=float)
        h = h * target_sum / np.sum(h)
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Refinement loop
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(initial_h_values + pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(best_h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
