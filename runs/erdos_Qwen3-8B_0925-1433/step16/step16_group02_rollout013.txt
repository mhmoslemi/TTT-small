            else:
                h[i] = (2 - x[i]) / 1.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def evaluate(h):

        return np.max(corr) * dx

        k_max = np.argmax(corr)

        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0

            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    def mutate(h, mutate_rate=0.2):
