import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(arr):
        return np.clip(arr, 0, 1) * target_sum / np.sum(arr)

    def eval_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 50
    population = [normalize(initial_h_values * target_sum / np.sum(initial_h_values))]
    for _ in range(population_size - 1):
        if np.random.rand() < 0.5:
            m = np.random.randint(5, 15)
            spacing = 2.0 / (m + 1)
            intervals = np.zeros(n_points)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
            intervals = normalize(intervals)
            population.append(intervals)
        else:
            h_bin = np.random.binomial(1, 0.5, n_points)
            h_bin = normalize(h_bin)
            population.append(h_bin)

    best_h = population[0]
    best_c5 = eval_fitness(best_h)
    start_time = time.time()
    generations = 0

    while time.time() - start_time < budget_s:
        generations += 1
        fitness = [eval_fitness(h) for h in population]
        next_population = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population, size=5)
            winner = min(tournament, key=eval_fitness)
            next_population.append(winner)
        for _ in range(population_size - len(next_population)):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = parent1.copy()
            mask = np.random.rand(n_points) < 0.8
            child[mask] = parent2[mask]
            mutation = np.random.normal(0, 0.1, n_points)
            child += mutation
            child = normalize(child)
            next_population.append(child)
        population = next_population
        for h in population:
            current_c5 = eval_fitness(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    return (best_h, best_c5, n_points)
