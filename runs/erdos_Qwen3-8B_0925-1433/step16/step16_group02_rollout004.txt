import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_symmetric(h):
        return np.clip(1.0 - h[::-1], 0.0, 1.0)

    def generate_sparse():
        h = np.zeros(n_points)
        positions = np.random.choice(n_points, size=int(n_points//5), replace=False)
        h[positions] = 1.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def evaluate(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_candidates = [
        initial_h_values,
        generate_symmetric(initial_h_values),
        generate_sparse(),
        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    ]
    best_h = initial_candidates[0]
    best_c5 = evaluate(best_h)

    pop_size = 50
    generations = 100
    mutation_rate = 0.8
    crossover_rate = 0.9
    start_time = time.time()

    population = [np.array(c, dtype=float) for c in initial_candidates]
    for _ in range(pop_size - len(initial_candidates)):
        population.append(generate_sparse())

    for gen in range(generations):
        fitness = [evaluate(h) for h in population]
        indices = np.argsort(fitness)
        elite_indices = indices[:int(pop_size * 0.1)]
        elites = [population[i] for i in elite_indices]
        new_population = elites.copy()
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(elites, 2)
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * p1 + (1 - alpha) * p2

            child = np.clip(child, 0, 1) * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population

            break
        current_best = min(fitness)
