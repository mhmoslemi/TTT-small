import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Preprocess initial_h to meet the sum constraint
    initial_h_scaled = initial_h * target_sum / np.sum(initial_h)

    def objective(h_vector):
        h_clipped = np.clip(h_vector, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return float('inf')
        h_scaled = h_clipped * (target_sum / sum_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        maxiter=300,
        tol=1e-6,
        seed=seed
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    best_h = best_h * (target_sum / sum_best) if sum_best > 0 else np.zeros(n_points)
    c5_bound = result.fun

    return (np.clip(best_h, 0, 1), c5_bound, n_points)
