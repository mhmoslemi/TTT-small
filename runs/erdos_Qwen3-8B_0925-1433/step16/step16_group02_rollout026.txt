from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_candidate():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1.0 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    def obj(x):
        h = x * target_sum / np.sum(x)
        return eval_c5(h)

    def mutate(h):
        h = h.copy()
        h[np.random.choice(n_points, size=5, replace=False)] = 1.0 - h[np.random.choice(n_points, size=5, replace=False)]
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def crossover(h1, h2):
        alpha = np.random.uniform(0.4, 0.6)
        return alpha * h1 + (1 - alpha) * h2

    def rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    initial_pop = [np.array(initial_h_values, dtype=float)]

        initial_pop.append(generate_candidate())

    initial_pop = [rescale(h) for h in initial_pop[:20]]
