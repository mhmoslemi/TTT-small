import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial population
    population = []
    # Add initial pattern
    h_initial = np.array(initial_h_values, dtype=np.float64)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population.append(h_initial)
    
    # Generate interval patterns
    for m in [5, 10, 15]:
        spacing = 2.0 / (m + 1)
        intervals = np.zeros(n_points)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                intervals[idx_start:idx_end] = 1.0
        intervals = intervals * target_sum / np.sum(intervals)
        population.append(intervals)
    
    # Generate binary patterns
    for _ in range(3):
        h_bin = np.random.binomial(1, 0.5, n_points)
        h_bin = h_bin * target_sum / np.sum(h_bin)
        population.append(h_bin)
    
    # Limit population size
    population = population[:50]

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    generations = 0
    max_generations = 300
    while generations < max_generations and time.time() - start_time < budget_s:
        new_population = []
        for _ in range(len(population)):
            # Tournament selection
            tournament = np.random.choice(population, size=5, replace=False)
            selected = min(tournament, key=compute_c5)
            new_population.append(selected)
        
        # Crossover and mutation
        for i in range(len(new_population)):
            h = new_population[i].copy()
            # Mutate
            noise = np.random.normal(0, 0.03, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            new_population[i] = h
        
        population = new_population
        for h in population:
            curr_c5 = compute_c5(h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h
        
        generations += 1

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
