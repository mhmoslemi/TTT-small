import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate base patterns
    base_patterns = [initial_h_values]
    phi = (1 + np.sqrt(5)) / 2
    rotated_indices = (np.arange(n_points) * phi) % n_points
    rotated_indices = np.round(rotated_indices).astype(int)
    base_patterns.append(initial_h_values[rotated_indices])
    base_patterns.append(np.full(n_points, 0.5))

    # Initialize population with diversity
    population = []
    for base in base_patterns:
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = base + perturb
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total == 0:
            perturbed = np.full_like(perturbed, 0.5)
        else:
            perturbed *= target_sum / total
        population.append(perturbed)

    # Project to feasible space
    population = [np.clip(p, 0, 1) for p in population]
    for p in population:
        total = np.sum(p)
        if total == 0:
            p[:] = np.full_like(p, 0.5)
        else:
            p[:] *= target_sum / total

    # Best solution tracker
    best_h = np.copy(population[0])
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        x = np.where(total == 0, np.full_like(x, 0.5), x * target_sum / total)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        x = np.where(total == 0, np.full_like(x, 0.5), x * target_sum / total)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(x)):
            j = (i + k_max) % len(x)
            grad[i] = (1 - x[j]) * dx - x[(i - k_max) % len(x)] * dx
        return grad

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Optimize best candidate
        res = minimize(obj, population[0], method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)

    h_values = best_h
    return (h_values, best_c5, n_points)
