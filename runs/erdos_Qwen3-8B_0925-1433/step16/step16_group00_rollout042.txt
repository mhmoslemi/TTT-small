import numpy as np
import random
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_intervals(m):
        spacing = 2.0 / (m + 1)
        intervals = np.zeros(n_points)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                intervals[idx_start:idx_end] = 1
        return intervals * target_sum / np.sum(intervals)

    pop_size = 50
    population = []
    for _ in range(pop_size):
        h = np.array(initial_h_values, dtype=float)
        if random.random() > 0.5:
            h += np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        else:
            h = make_intervals(10)
        population.append(h)
    population = np.array(population)

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = fitness(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    gen = 0
    while gen < 200 and time.time() - start_time < budget_s:
        fitness_vals = [fitness(h) for h in population]
        best_idx = np.argmin(fitness_vals)
        if fitness_vals[best_idx] < best_c5:
            best_c5 = fitness_vals[best_idx]
            best_h = population[best_idx]
        new_pop = []
        for _ in range(pop_size):
            t = np.random.choice(pop_size, 5)
            p1 = population[t[np.argmin([fitness_vals[i] for i in t])]]
            t = np.random.choice(pop_size, 5)
            p2 = population[t[np.argmin([fitness_vals[i] for i in t])]]
            child = np.copy(p1)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p2[i]
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] += np.random.normal(0, 0.05)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = np.array(new_pop)
        gen += 1

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
