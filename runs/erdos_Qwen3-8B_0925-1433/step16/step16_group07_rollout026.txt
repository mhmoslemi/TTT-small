import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    def generate_patterns(base, n=50):
        patterns = [base + np.random.normal(0, 0.05, n_points) for _ in range(n)]
        for i in range(n):
            patterns[i] = project(patterns[i])
        return patterns

    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)

    initial_patterns = [base]
    for _ in range(3):
        shifted = np.roll(base, np.random.randint(0, n_points))
        initial_patterns.append(shifted)
        spaced = np.linspace(0, 2, n_points, endpoint=False)
        spaced = (spaced + np.random.normal(0, 0.1, n_points)) % 2
        spaced = project(spaced)
        initial_patterns.append(spaced)

    population = [project(p) for p in initial_patterns]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]
    population += [project(np.random.uniform(0, 1, n_points)) for _ in range(50 - len(population))]

    best_h = base.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        parents = np.random.choice(population, size=50, replace=True)
        offspring = []
        for i in range(0, 50, 2):
            p1, p2 = parents[i], parents[i+1]
            alpha = np.random.uniform(0.6, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            child = project(child)
            offspring.append(child)
        population = offspring
        top = sorted(population, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)[:20]
        for p in top:
            res = minimize(obj, p, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
