import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_perturb = np.clip(x + 1e-4 * np.eye(n_points)[i], 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            grad[i] = (np.max(corr_perturb) - np.max(corr)) * dx
        return grad

    # Generate initial candidates
    initial_candidates = [initial_h_values.copy()]
    x = np.linspace(0, 2, n_points)
    
    # Sinusoidal pattern
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    h_sinusoidal = h_sinusoidal * target_sum / np.sum(h_sinusoidal)
    initial_candidates.append(h_sinusoidal)
    
    # Triangular pattern
    h_triangular = np.zeros(n_points)
    for i in range(n_points):
        if x[i] <= 1:
            h_triangular[i] = x[i]
        else:
            h_triangular[i] = 2 - x[i]
    h_triangular = h_triangular * target_sum / np.sum(h_triangular)
    initial_candidates.append(h_triangular)
    
    # Random pattern
    h_random = np.random.rand(n_points)
    h_random = h_random * target_sum / np.sum(h_random)
    initial_candidates.append(h_random)
    
    # Evaluate initial candidates
    best_h = initial_candidates[0].copy()
    best_c5 = objective(best_h)
    for cand in initial_candidates:
        c5 = objective(cand)
        if c5 < best_c5:
            best_h = cand.copy()
            best_c5 = c5

    start_time = time.time()

    # Differential Evolution
    bounds = [(0.0, 1.0)] * n_points
    result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed, disp=False)
    best_h = result_de.x
    best_c5 = result_de.fun

    # Local refinement with L-BFGS-B
    result_lbg = minimize(objective, best_h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, options={'maxiter': 50, 'ftol': 1e-7})
    if result_lbg.success:
        best_h = result_lbg.x
        best_c5 = result_lbg.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
