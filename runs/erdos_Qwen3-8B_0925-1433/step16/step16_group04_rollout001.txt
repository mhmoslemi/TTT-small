        population.append(h[:n_points])
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([h * target_sum / np.sum(h) for h in population])

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for gen in range(150):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        tournament = np.random.choice(len(population), size=5, replace=False)
        parent1, parent2 = population[tournament[0]], population[tournament[1]]
        cross_idx = np.random.randint(n_points)
        child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]])

        def apply_mutation(h):
            corr = np.correlate(h, 1 - h, mode='full')
            i = np.argmax(corr)
            k = i - (n_points - 1)
            modified = []
            if k > 0:
                for x in range(n_points - k):
                    modified.append(x)
                    modified.append(x + k)
            elif k < 0:
                abs_k = -k
                for x in range(abs_k, n_points):
                    modified.append(x)
                    modified.append(x + k)
            modified = np.unique(np.array(modified))
            if len(modified) > 0:
                perturb = np.random.uniform(-0.15, 0.15, len(modified))
                for idx in range(len(modified)):
                    h[modified[idx]] += perturb[idx] * h[modified[idx]]
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            return h

        child = apply_mutation(child)

        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
