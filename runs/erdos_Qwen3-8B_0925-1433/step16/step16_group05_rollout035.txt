import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_scale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        return h_clipped * (target_sum / sum_clipped) if sum_clipped else np.zeros_like(h)

    def objective(h):
        h_scaled = clip_scale(h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = clip_scale(h_plus)
            obj_plus = objective(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = clip_scale(h_minus)
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    pop_size = 50
    pop = [clip_scale(h) for _ in range(pop_size)]
    for _ in range(300):
        if time.time() - time.time() > budget_s:
            break
        evals = [objective(ind) for ind in pop]
        best_idx = np.argmin(evals)
        best_c5 = evals[best_idx]
        if best_c5 < best_c5:
            best_h = pop[best_idx].copy()
        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(pop_size, 2, replace=False)
            parent1, parent2 = pop[p1], pop[p2]
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            child = clip_scale(child)
            new_pop.append(child)
        for ind in new_pop:
            if np.random.rand() < 0.1:
                mutation = np.random.rand(n_points) * 0.2 - 0.1
                ind += mutation

        pop = new_pop + [clip_scale(pop[best_idx]) for _ in range(pop_size - len(new_pop))]
        for i in range(len(pop)):
            if np.random.rand() < 0.2:
                grad = compute_gradient(pop[i])
                pop[i] = clip_scale(pop[i] - 0.05 * grad)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
