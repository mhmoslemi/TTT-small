from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        initial_h_values = initial_h_values * n_points / 2.0 / np.sum(initial_h_values)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def de_step(population, n_points, target_sum):
        new_pop = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[c] + 0.5 * (population[a] - population[b])
            mutant = normalize(mutant)
            target = population[i]
            cross = np.random.rand(n_points) < 0.8
            trial = np.where(cross, mutant, target)
            trial = normalize(trial)
            new_pop.append(trial)
        return new_pop

    population = [normalize(h + np.random.normal(0, 0.1, n_points)) for _ in range(100)]

                best_h = cand.copy()
                best_c5 = curr_c5

    def local_optimize(x):
        res = minimize(lambda x: obj(x), x, method='SLSQP', bounds=[(0,1)*n_points], 
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
