import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def rescale(h):
        return h * target_sum / np.sum(h)

    # Initialize population
    population = []
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = rescale(initial_h)
    population.append(initial_h)
    for _ in range(49):
        if np.random.rand() < 0.3:
            binary = np.random.randint(0, 2, n_points)
            while np.sum(binary) != 50:
                binary = np.random.randint(0, 2, n_points)
            h = binary * target_sum / 50
        elif np.random.rand() < 0.5:
            h = np.ones(n_points) * target_sum / n_points
        else:
            h = np.zeros(n_points)
            h[::2] = 1.0
            h = rescale(h)
        population.append(h)

    best_h = initial_h
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    max_gen = int(budget_s / 0.3)
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all
        evals = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            evals.append((c5, h))
        # Sort and select top 40
        evals.sort()
        selected = [h for (c5, h) in evals[:40]]
        # Generate new
        new_pop = []
        for _ in range(len(population)):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = parent1.copy()
            if np.random.rand() < 0.8:
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = rescale(child)
            new_pop.append(child)
        population = selected + new_pop
        population = population[:len(population)]
        # Update best
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    best_h = np.clip(best_h, 0, 1)
    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
