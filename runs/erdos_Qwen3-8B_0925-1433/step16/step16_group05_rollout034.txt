import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return float('inf')
        h = h * target_sum / s
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [np.array(initial_h, dtype=float)]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = np.array(initial_h, dtype=float)
    best_c5 = objective(best_h)
    start_time = time.time()
    pop_size = 50

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        scores = [objective(ind) for ind in population]
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(population, 3, replace=False)
            candidates = [a, b, c]
            best = min(candidates, key=lambda x: objective(x))
            new_pop.append(best)
        population = new_pop
        for i in range(len(population)):
            if objective(population[i]) < best_c5:
                best_c5 = objective(population[i])
                best_h = population[i].copy()

        elites = sorted(population, key=lambda x: objective(x))[:5]
        for h in elites:
            eps = 1e-5
            h_clipped = np.clip(h, 0, 1)
            s = np.sum(h_clipped)
            if s == 0:
                continue
            h_scaled = h_clipped * target_sum / s
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            grad = np.zeros_like(h)
            for idx in range(n_points):
                h_plus = h.copy()
                h_plus[idx] += eps
                h_plus = np.clip(h_plus, 0, 1)
                s_p = np.sum(h_plus)
                h_plus_scaled = h_plus * target_sum / s_p
                corr_p = np.correlate(h_plus_scaled, 1 - h_plus_scaled, mode='full')
                val_p = np.max(corr_p) * dx
                h_minus = h.copy()
                h_minus[idx] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                s_m = np.sum(h_minus)
                h_minus_scaled = h_minus * target_sum / s_m
                corr_m = np.correlate(h_minus_scaled, 1 - h_minus_scaled, mode='full')
                val_m = np.max(corr_m) * dx
                grad[idx] = (val_p - val_m) / (2 * eps)
            step = 0.06 * grad
            h += step
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            h = h * target_sum / s
            population[population.index(h)] = h
        if np.abs(best_c5 - objective(best_h)) < 1e-6:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
