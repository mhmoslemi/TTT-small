from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.clip(h + np.random.normal(0, scale, h.shape), 0, 1) * target_sum / np.sum(h)
