        h_values = np.zeros(n_points)
        for i in range(m):
            a_i = a_vals[i]
            l_i = l_vals[i]
            start = int(a_i / dx - 0.5)
            end = int((a_i + l_i) / dx - 0.5)
            start = np.clip(start, 0, n_points - 1)
            end = np.clip(end, 0, n_points - 1)
            h_values[start:end+1] = 1.0
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx
    
    def grad(x):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            f_plus = obj(x_plus)
            f_minus = obj(x_minus)
            grad[i] = (f_plus - f_minus) / (2 * eps)
        return grad
    
    bounds = [(0.0, 2.0) for _ in range(2 * m)]
    constraints = []
    constraints.append({'type': 'eq', 'fun': lambda x: np.sum(x[1::2]) - 1.0})
    for i in range(m-1):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*(i+1)] - x[2*i] - x[2*i+1]})
    
    result = minimize(obj, np.concatenate((a, l)), method='SLSQP', jac=grad, bounds=bounds, constraints=constraints, tol=1e-6)
    best_params = result.x
    a_vals = best_params[::2]
    l_vals = best_params[1::2]
    
    h_values = np.zeros(n_points)
    for i in range(m):
        a_i = a_vals[i]
        l_i = l_vals[i]
        start = int(a_i / dx - 0.5)
        end = int((a_i + l_i) / dx - 0.5)
        start = np.clip(start, 0, n_points - 1)
        end = np.clip(end, 0, n_points - 1)
        h_values[start:end+1] = 1.0
    
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_values, c5_bound, n_points)
