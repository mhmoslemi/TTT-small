from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad
    
    start_time = time.time()
    
    # Precompute correlation
    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    # Local optimization
    for _ in range(150):  # Reduced iterations for efficiency
        if time.time() - start_time > budget_s:
            break
        # Local search from best_h
        res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-6, maxiter=50)
        if res.success:
            best_h = res.x
            current_c5 = evaluate(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    # Refinement with perturbations
    for _ in range(50):
        if time.time() - start_time > budget_s:
