    def generate_symmetric_triangular():
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i] / 2
            else:
                h[i] = (2 - x[i]) / 2
        return h

    h = generate_symmetric_triangular()
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    def mirror(h):
        for i in range(n_points // 2):
            h[n_points - 1 - i] = h[i]
        return h

    def obj(x):
        h_full = np.zeros(n_points)
        for i in range(n_points):
            if i < len(x):
                h_full[i] = x[i]
            else:
                h_full[i] = h_full[n_points - 1 - i]
        h_full = np.clip(mirror(h_full), 0, 1)
        h_full = h_full * target_sum / np.sum(h_full)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h_full = np.zeros(n_points)
        for i in range(n_points):
            if i < len(x):
                h_full[i] = x[i]
            else:
                h_full[i] = h_full[n_points - 1 - i]
        h_full = np.clip(mirror(h_full), 0, 1)
        h_full = h_full * target_sum / np.sum(h_full)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            if i < len(x):
                j_plus = i + k_max
                if 0 <= j_plus < n_points:
                    term1 = 1 - h_full[j_plus]
                else:

                j_minus = i - k_max
