import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_population(base, size=50):
        pop = [base.copy()]
        for _ in range(size - 1):
            h = base.copy() + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            pop.append(h)
        return pop

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop = generate_population(np.array(initial_h_values, dtype=float), 50)
    best_c5 = np.inf
    best_h = pop[0]

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        scores = [eval_c5(h) for h in pop]
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h = pop[best_idx]
        new_pop = []
        for _ in range(len(pop) // 2):
            a, b = np.random.choice(len(pop), 2, replace=False)
            parent1, parent2 = pop[a], pop[b]
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            mut = np.random.normal(0, 0.05, n_points)
            child = np.clip(child + mut, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop + [pop[i] for i in np.argsort(scores)[:5]]
        pop = np.unique(pop, axis=0).tolist()
        if len(pop) > 50:
            pop = pop[:50]

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
