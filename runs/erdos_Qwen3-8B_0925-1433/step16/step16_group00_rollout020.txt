import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize from initial_h_values, normalized to sum to target_sum
    h = np.array(initial_h_values, dtype=float)
    h_norm = h * target_sum / np.sum(h)
    h = np.clip(h_norm, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for _ in range(200):  # 200 iterations within budget
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        def obj(x):
            corr_val = np.correlate(x, 1 - x, mode='full')
            return corr_val[k_max] * dx
        
        def grad_func(x):
            grad = np.zeros_like(x)
            for i in range(n_points):
                h_plus = x.copy()
                h_plus[i] += 1e-4

                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')

                h_minus[i] -= 1e-4

                val_minus = corr_minus[k_max] * dx
                grad[i] = (val_plus - val_minus) / (2e-4)
            return -grad

        bounds = [(0, 1) for _ in range(n_points)]
