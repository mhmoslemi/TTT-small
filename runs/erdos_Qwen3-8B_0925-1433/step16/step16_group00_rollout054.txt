                start = i * spacing

                idx_start = int(start / dx + 0.5)

                intervals[idx_start:idx_end] = 1
            intervals = intervals * target_sum / np.sum(intervals)
            pop.append(intervals)
        return pop

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, k=5):

        best = min(inds, key=lambda i: evaluate(pop[i]))
        return pop[best].copy()
