import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_base = np.array(initial_h_values, dtype=float)
    h_roll = np.roll(h_base, int(0.618 * n_points))
    base_patterns = [h_base, h_roll]

    population = []
    for p in base_patterns:
        p += np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)

    for _ in range(50):
        p = np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        population.append(p)

    population = [np.clip(p, 0, 1) for p in population]
    population = [p * target_sum / np.sum(p) for p in population]

    best_h = population[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - target_sum

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            res = minimize(objective, p, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
            if res.success:
                p_new = res.x
                p_new = np.clip(p_new, 0, 1)
                p_new = p_new * target_sum / np.sum(p_new)
                p[:] = p_new
        current_c5 = np.max(np.correlate(population[0], 1 - population[0], mode='full')) * dx
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
