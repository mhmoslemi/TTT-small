    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    def evaluate(h):
        h = project(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):

        return grad
