    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iterations = 300
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)
        if lag < 0:
            lag = -lag
            indices = np.arange(lag, n_points)
            indices_lag = indices - lag
        else:
            indices = np.arange(0, n_points - lag)
            indices_lag = indices + lag
        weights = np.abs(h[indices] * h[indices_lag])
        weight_sum = np.sum(weights)
        if weight_sum == 0:
            continue
        impact = (h[indices] * h[indices_lag]) / weight_sum
        reduction = 0.15 * impact
        h[indices] = np.clip(h[indices] - reduction, 0, 1)
        h[indices_lag] = np.clip(h[indices_lag] - reduction, 0, 1)
        h = h * target_sum / np.sum(h)
    
    # Ensure the best_c5 is not worse than the initial
    best_c5 = min(best_c5, 0.381)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
