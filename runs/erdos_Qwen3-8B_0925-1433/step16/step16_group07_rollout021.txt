import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    population = [np.copy(h) + np.random.normal(0, 0.1, n_points) for _ in range(50)]
    population = [project(p) for p in population]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]

    start_time = time.time()
    gamma = 0.2
    eta = 0.05
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            for _ in range(20):
                h_new = p.copy()
                for i in range(n_points):
                    influence = 0.0
                    for j in range(max(0, i-2), min(n_points, i+3)):
                        if j != i:
                            influence += p[j] * abs(i - j)
                    h_new[i] += eta * (target_sum / n_points - p[i]) + gamma * influence
                    h_new[i] = np.clip(h_new[i], 0, 1)
                p = project(h_new)
            res = minimize(obj, p, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                p[:] = res.x
        current_c5 = np.max(np.correlate(project(population[0]), 1 - project(population[0]), mode='full')) * dx
        if current_c5 < best_c5:
            best_h = project(population[0])
            best_c5 = current_c5

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
