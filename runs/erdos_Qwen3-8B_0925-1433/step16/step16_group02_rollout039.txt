            full_h[j] = 1.0 - h[i]
        return np.clip(full_h, 0, 1)

    def mirror_symmetry(h):
        n_points = len(h)
        for i in range(n_points):
            j = n_points - 1 - i
            if i < j:
                h[j] = 1.0 - h[i]
        return h

    def generate_sinusoidal(n_points):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_triangular(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i] / 1.0
            else:
                h[i] = (2 - x[i]) / 1.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_random(n_points):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_regular(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    initial_candidates = [
        initial_h_values,
        generate_sinusoidal(n_points),
        generate_triangular(n_points),
        generate_random(n_points),
        generate_regular(n_points),
        generate_symmetric_initial(n_points)
    ]

    best_h = np.array(initial_candidates[0], dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()

    for candidate in initial_candidates:
        h = np.array(candidate, dtype=float)
        h = h * target_sum / np.sum(h)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        idx = np.random.choice(len(initial_candidates))
        h = np.array(initial_candidates[idx], dtype=float)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_h = res.x.copy()
            current_h = mirror_symmetry(current_h)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        perturbed_h = mirror_symmetry(perturbed_h)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            current_h = res.x.copy()
            current_h = mirror_symmetry(current_h)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    best_h = mirror_symmetry(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
