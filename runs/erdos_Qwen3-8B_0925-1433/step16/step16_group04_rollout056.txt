import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = []
    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    population.append(h0)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(5):
        h = np.random.rand(n_points)
        h = np.concatenate([h, h[::-1]])
        h = h * target_sum / np.sum(h)
        population.append(h[:n_points])
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([h * target_sum / np.sum(h) for h in population])

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        # Tournament selection
        tournament = np.random.choice(len(population), size=5, replace=False)
        parent1, parent2 = population[tournament[0]], population[tournament[1]]
        cross_idx = np.random.randint(n_points)
        child = np.concatenate([parent1[:cross_idx], parent2[cross_idx:]])

        # Mutation targeting overlap maxima
        corr = np.correlate(child, 1 - child, mode='full')
        k_max_idx = np.argmax(corr)
        lag = k_max_idx - (n_points - 1)
        positions = []
        if lag >= 0:
            for x in range(n_points - lag):
                positions.append(x)
                positions.append(x + lag)
        else:
            k = -lag
            for x in range(k, n_points):
                positions.append(x)
                positions.append(x - k)
        mutate_indices = np.random.choice(positions, size=min(40, len(positions)), replace=False)
        for i in mutate_indices:
            child[i] += np.random.uniform(-0.15, 0.15) * child[i]
        child = np.clip(child, 0, 1)
        child = child * target_sum / np.sum(child)

        # Replace worst
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
