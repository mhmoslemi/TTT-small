from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    best_c5 = min(best_c5, 0.381)
    
    # Population-based optimization
    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    for p in population:
        p += np.random.normal(0, 0.05, n_points)
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            x0 = population[i].copy()
            res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points,

                best_h = cand.copy()
                best_c5 = curr_c5
    
    # Final evaluation
    h_values = np.clip(best_h, 0, 1)
