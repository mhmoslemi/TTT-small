import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return np.full_like(x, 0.5)
        return x * target_sum / total

    base_patterns = [h.copy()]
    phi = (np.sqrt(5) - 1) / 2
    for _ in range(3):
        shift = np.random.uniform(0, 2)
        shifted = np.roll(h, int(round(shift / dx)))
        base_patterns.append(shifted)
    base_patterns.append(np.random.permutation(h))

    population = [project(p + np.random.normal(0, 0.05, n_points)) for p in base_patterns]
    population = [project(p) for p in population]
    population = [p for p in population if np.abs(np.sum(p) - target_sum) < 1e-6]

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in population:
            res = minimize(obj, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                p_new = project(res.x)
                if np.abs(np.sum(p_new) - target_sum) < 1e-6:
                    p[:] = p_new
        current_c5 = np.max(np.correlate(project(population[0]), 1 - project(population[0]), mode='full')) * dx
        if current_c5 < best_c5:
            best_h = project(population[0])
            best_c5 = current_c5

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
