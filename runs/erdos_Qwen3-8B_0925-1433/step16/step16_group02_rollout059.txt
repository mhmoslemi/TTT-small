import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h_clipped = np.clip(h, 0, 1)
        return h_clipped * target_sum / np.sum(h_clipped)

    def mutate(h, mut_type='flip'):
        if mut_type == 'flip':
            idx = np.random.choice(n_points, size=5, replace=False)
            h = h.copy()
            h[idx] = 1 - h[idx]
        elif mut_type == 'shift':
            shift = np.random.randint(1, n_points // 2)
            h = np.roll(h, shift)
        elif mut_type == 'noise':
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
        return rescale(h)

    def crossover(h1, h2, alpha=0.5):
        return alpha * h1 + (1 - alpha) * h2

    initial_candidates = [
        initial_h_values,
        np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5,
        np.clip(np.linspace(0, 1, n_points), 0, 1),
        np.random.rand(n_points),
        np.ones(n_points) * 0.5
    ]
    initial_candidates = [rescale(c) for c in initial_candidates]

    population = [rescale(np.random.rand(n_points)) for _ in range(5)]
    for c in initial_candidates:
        population.append(c)

    population = np.array([rescale(c) for c in population])
    best_h = population[np.argmin([eval_c5(p) for p in population])]
    best_c5 = eval_c5(best_h)

    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(20):
            p1, p2 = np.random.choice(population, size=2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < 0.3:
                child = mutate(child, 'flip')
            elif np.random.rand() < 0.3:
                child = mutate(child, 'shift')
            else:
                child = mutate(child, 'noise')
            new_population.append(child)
        population = np.array([rescale(c) for c in new_population + [best_h]])
        evals = np.array([eval_c5(p) for p in population])
        best_idx = np.argmin(evals)
        best_h, best_c5 = population[best_idx], evals[best_idx]

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    def objective(h):
        return eval_c5(h)

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B',
                          jac=compute_gradient, bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if result.success:
            best_h = result.x
            best_c5 = result.fun

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
