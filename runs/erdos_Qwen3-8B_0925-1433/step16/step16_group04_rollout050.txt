import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    h = initial_h.copy()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    alpha = 0.1
    alpha_decay = 0.99
    max_iterations = 1000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        m_shift = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + m_shift
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]
        adjusted_h = h - alpha * gradient
        adjusted_h = np.clip(adjusted_h, 0.0, 1.0)
        current_sum = np.sum(adjusted_h)
        if current_sum == 0:
            adjusted_h = np.zeros_like(adjusted_h)
        else:
            scale = target_sum / current_sum
            adjusted_h *= scale
        h = adjusted_h
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        alpha *= alpha_decay

    final_c5 = compute_c5(h)
    best_c5 = min(best_c5, final_c5)
    return (best_h, best_c5, n_points)
