import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

    # Generate diverse initial patterns
    base = np.array(initial_h_values, dtype=float)
    variations = []
    variations.append(base)
    variations.append(np.roll(base, 1))
    variations.append(np.roll(base, -1))
    variations.append(np.roll(base, 2))
    variations.append(np.roll(base, -2))
    variations.append(np.roll(base, 3))
    variations.append(np.roll(base, -3))
    variations.append(np.roll(base, 6))
    variations.append(np.roll(base, -6))
    variations.append(np.array([x + np.random.normal(0, 0.1) for x in base]))
    variations.append(np.array([x + np.random.normal(0, 0.05) for x in base]))
    variations.append(np.array([x + np.random.normal(0, 0.03) for x in base]))
    for v in variations:
        v = project(v)
        variations.append(v)
    population = variations[:50]

    best_h = project(base)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for p in population[:40]:
            res = minimize(obj, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=grad, tol=1e-6)
            if res.success:
