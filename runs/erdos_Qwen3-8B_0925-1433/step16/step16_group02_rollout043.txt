import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial population
    population = []
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    population.append(h)
    
    x = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    h_sinusoidal = h_sinusoidal * target_sum / np.sum(h_sinusoidal)
    population.append(h_sinusoidal)
    
    x_tri = np.linspace(0, 2, n_points)
    h_tri = np.zeros(n_points)
    for i in range(n_points):
        if x_tri[i] <= 1:
            h_tri[i] = x_tri[i] / 1.0
        else:
            h_tri[i] = (2 - x_tri[i]) / 1.0
    h_tri = h_tri * target_sum / np.sum(h_tri)
    population.append(h_tri)
    
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)
    population.append(h_rand)
    
    h_reg = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_reg[i] = 1.0
        else:
            h_reg[i] = 0.0
    h_reg = h_reg * target_sum / np.sum(h_reg)
    population.append(h_reg)

    # Additional diversity
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    num_generations = 200
    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate(h) for h in population]
        indices = np.argsort(scores)
        parents = [population[i] for i in indices[:len(population)//5]]
        new_population = []
        for _ in range(len(population)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * p1 + (1 - alpha) * p2
            if np.random.rand() < 0.8:
                idx = np.random.choice(n_points, size=n_points//10, replace=False)
                child[idx] = 1 - child[idx]
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    res = minimize(obj, best_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
