from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):

    initial_h = initial_h_values

    dx = 2.0 / n_points

    scaled_initial = np.array(initial_h) * (target_sum / np.sum(initial_h))
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        maxiter=200,
        tol=1e-5,
        x0=scaled_initial,
        polish=False,
        disp=False
