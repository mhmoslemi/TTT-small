import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_initial_population():
        pop = []
        h_base = np.array(initial_h_values, dtype=float)
        pop.append(normalize(h_base))
        for _ in range(10):
            m = np.random.randint(5, 15)
            spacing = 2.0 / (m + 1)
            intervals = np.zeros(n_points)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
            intervals = normalize(intervals)
            pop.append(intervals)
        for _ in range(10):
            h = np.random.binomial(1, 0.5, n_points)
            h = normalize(h)
            pop.append(h)
        return pop

    population = generate_initial_population()
    best_h = population[0]
    best_c5 = evaluate(best_h)
    generation = 0
    mutation_rate = 0.1
    time_start = time.time()

    while generation < 200 and time.time() - time_start < budget_s:
        if time.time() - time_start > budget_s:
            break
        fits = [(evaluate(h), i) for i, h in enumerate(population)]
        fits.sort()
        next_pop = [population[i] for _, i in fits[:int(len(population) * 0.2)]]
        while len(next_pop) < len(population):
            t = np.random.choice(len(population), size=5)
            candidates = [population[i] for i in t]
            candidates.sort(key=evaluate)
            p1, p2 = candidates[0], candidates[1]
            child = np.random.rand(n_points) * (p1 + p2) / 2
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            child = normalize(child)
            next_pop.append(child)
        population = next_pop
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        generation += 1
        if generation > 50 and np.isclose(best_c5, evaluate(best_h), atol=1e-6):
            mutation_rate += 0.01

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
