import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def _clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def _compute_c5(h):
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    def _mutate(h):
        idx = np.random.choice(n_points, size=4, replace=False)
        h = h.copy()
        h[idx] = 1.0 - h[idx]
        return _clip_rescale(h)

    def _crossover(h1, h2):
        alpha = np.random.uniform(0.4, 0.6)
        h = alpha * h1 + (1 - alpha) * h2
        return _clip_rescale(h)

    def _objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def _gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1.0 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = _clip_rescale(initial_h)
    best_h = initial_h.copy()
    best_c5 = _compute_c5(best_h)

    start = time.time()
    pop = [initial_h]

        if time.time() - start > budget_s - 50:

        scored = [(_compute_c5(h), h) for h in pop]

            p1, p2 = np.random.choice(selected, 2, replace=False)
