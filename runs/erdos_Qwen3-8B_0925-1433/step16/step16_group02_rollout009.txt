import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial candidates
    def gen_sinusoidal(x):
        return 0.5 + 0.5 * np.sin(np.pi * x / 2)
    def gen_triangular(x):
        return np.minimum(x, 2 - x)
    def gen_random(x):
        return np.random.rand(len(x))
    h_s = gen_sinusoidal(np.linspace(0, 2, n_points)) * target_sum / np.sum(gen_sinusoidal(np.linspace(0, 2, n_points)))
    h_t = gen_triangular(np.linspace(0, 2, n_points)) * target_sum / np.sum(gen_triangular(np.linspace(0, 2, n_points)))
    h_r = gen_random(np.linspace(0, 2, n_points)) * target_sum / np.sum(gen_random(np.linspace(0, 2, n_points)))
    initial_candidates = [initial_h_values, h_s, h_t, h_r]

    # Initial evaluation
    best_h = np.array(initial_candidates[0], dtype=float)
    def calc_c5(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        h = h * target_sum / (s + 1e-10)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = calc_c5(best_h)

    # Differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start = time.time()
    try:
        result = differential_evolution(
            lambda x: calc_c5(x),
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=16,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.8,
            polish=False
        )
        best_h = result.x
        best_c5 = result.fun
    except:
        pass

    # L-BFGS refinement
    def refine_obj(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        h = h * target_sum / (s + 1e-10)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - target_sum}
    try:
        result = minimize(
            refine_obj,
            best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=cons,
            options={'maxiter': 20, 'ftol': 1e-6}
        )
        best_h = result.x
        best_c5 = result.fun
    except:
        pass

    # Final clipping and scaling
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if s > 0:
        best_h = best_h * target_sum / s

    elapsed = time.time() - start
