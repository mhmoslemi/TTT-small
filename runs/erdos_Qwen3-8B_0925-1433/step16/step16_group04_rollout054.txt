def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

        return np.correlate(h, 1 - h, mode='full')

    def compute_overlap(h, s):

        total = 0.0
        for i in range(n):
            if i + s >= 0 and i + s < n:
                total += h[i] * (1 - h[i + s])
        return total

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

            break
