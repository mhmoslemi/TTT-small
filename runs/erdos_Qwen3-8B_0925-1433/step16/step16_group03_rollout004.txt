import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_base = np.array(initial_h_values, dtype=float)
    h_base = np.clip(h_base, 0, 1)
    h_base = h_base * target_sum / np.sum(h_base)
    best_h = h_base.copy()
    best_c5 = np.max(np.correlate(h_base, 1 - h_base, mode='full')) * dx

    x = np.linspace(0, 2, n_points)
    def generate_patterns():
        p1 = 0.5 * (1 + np.sin(np.pi * x))
        p1 = np.clip(p1, 0, 1)
        p1 = p1 * target_sum / np.sum(p1)
        p2 = np.zeros_like(x)
        for i in range(5):
            idx = int(i * n_points / 5)
            p2[idx] = 0.5
        p2 = p2 * target_sum / np.sum(p2)
        p3 = np.random.binomial(1, 0.1, n_points) * 0.5
        p3 = p3 * target_sum / np.sum(p3)
        p4 = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        p4 = np.clip(p4, 0, 1)
        p4 = p4 * target_sum / np.sum(p4)
        p5 = initial_h_values + np.random.normal(0, 0.1, n_points)
        p5 = np.clip(p5, 0, 1)
        p5 = p5 * target_sum / np.sum(p5)
        return [p1, p2, p3, p4, p5]

    population = [h_base] + generate_patterns()
    population = [np.clip(p, 0, 1) for p in population]

    start_time = time.time()
    n_gen = 75
    elite_count = 5
    mu = 0.7
    sigma = 0.2

    for gen in range(n_gen):
        if time.time() - start_time > budget_s:
            break
        fit = np.array([-np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in population])
        argsort = np.argsort(fit)
        sorted_pop = [population[i] for i in argsort]
        elites = sorted_pop[:elite_count]
        new_pop = elites.copy()
        while len(new_pop) < n_pop:
            a, b = np.random.randint(len(sorted_pop), size=2)
            p1, p2 = sorted_pop[a], sorted_pop[b]
            child = p1 * mu + p2 * (1 - mu)
            noise = np.random.normal(0, sigma, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop

    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
