import numpy as np
import time

def run(seed=42, budget_s=600):
    import numpy as np
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    max_iterations = 100
    iterations = 0
    start_time = time.time()

    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        overlapping_indices = []

        if lag > 0:
            overlap_end = n_points - lag
            overlapping_indices = range(overlap_end)
        else:
            lag_abs = -lag
            overlap_end = n_points - lag_abs
            overlapping_indices = range(overlap_end)

        contributions = []
        for i in overlapping_indices:
            j = i + lag
            contrib = h[i] * (1 - h[j])
            contributions.append( (contrib, i, j) )

        # Sort contributions in descending order
        contribs_sorted = sorted(contributions, key=lambda x: -x[0])

        # Adjust the top few contributions
        for contrib, i, j in contribs_sorted[:10]:
            h[i] -= 0.05
            h[j] -= 0.05

        # Clip values to [0, 1] and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Recompute C5
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * dx
        if current_c5 < best_c5:

    early_stop = 10

    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        overlapping_indices = []

        if lag > 0:
            overlap_end = n_points - lag
            overlapping_indices = range(overlap_end)

            lag_abs = -lag

            overlapping_indices = range(overlap_end)

        contributions = []

            contrib = h[i] * (1 - h[j])
