from scipy.optimize import minimize
import time

        h_opt = x * target_sum / np.sum(x)
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h_opt = x * target_sum / np.sum(x)
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
