import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_rescale(h, ts):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * ts / s

    def compute_c5(h):
        n = len(h)
        dx_ = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_

    def fitness(h):
        return -compute_c5(h)

    patterns = []
    x = np.linspace(0, 2, n_points)
    patterns.append(clip_and_rescale(0.5 + 0.5 * np.sin(np.pi * x), target_sum))
    h_bin = np.random.binomial(1, 0.1, n_points)
    patterns.append(clip_and_rescale(h_bin, target_sum))
    patterns.append(clip_and_rescale(np.array(initial_h_values, dtype=float), target_sum))
    patterns = np.array(patterns)

    population = patterns.copy()
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for _ in range(75):
        if time.time() - start_time > budget_s:
            break
        fitnesses = np.array([fitness(h) for h in population])
        elite = population[np.argpartition(fitnesses, -5)[-5:]]
        new_pop = elite
        for _ in range(len(population) - 5):
            p1 = np.random.choice(population)
            p2 = np.random.choice(population)
            child = (1-0.7)*p1 + 0.7*p2
            noise = np.random.normal(0, 0.2, len(child))
            child += noise
            child = clip_and_rescale(child, target_sum)
            new_pop = np.vstack((new_pop, child))
        population = new_pop

    best_pop = population[np.argmax(fitness(population))]
    best_h = best_pop
    best_c5 = compute_c5(best_h)

    def obj(x):
        h = clip_and_rescale(x, target_sum)
        return compute_c5(h)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    if time.time() - start_time < budget_s:
        n_new = 150

        x_new = np.linspace(0, 2, n_new)
        h_up = np.interp(x_new, x, best_h)
