            if start_idx <= end_idx:
                h_values[start_idx:end_idx+1] = 1.0
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(corr) * dx
    else:
        h_values = initial_h_values.copy()
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(corr) * dx
    return (np.clip(h_values, 0, 1), c5, n_points)
