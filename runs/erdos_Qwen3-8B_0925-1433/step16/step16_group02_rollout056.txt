                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * target_sum / np.sum(h)
        return np.clip(h, 0, 1)

    def generate_base():
        h = np.zeros(n_points)
        for i in range(n_points):

        h = h * target_sum / np.sum(h)

            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    for c in initial_candidates:

        c5 = objective(h)
        if c5 < best_c5:
