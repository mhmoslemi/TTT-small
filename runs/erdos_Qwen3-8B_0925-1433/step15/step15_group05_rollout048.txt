import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def crossover(a, b):
        idx = np.random.randint(1, n_points)
        return np.concatenate([a[:idx], b[idx:]])

    def mutate(h):
        h = h.copy()
        if np.random.rand() < 0.1:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
        return clip_rescale(h)

    initial = np.array(initial_h_values, dtype=float)
    initial = clip_rescale(initial)
    population = [initial]
    for _ in range(49):
        rand = np.random.rand(n_points)
        rand = clip_rescale(rand)
        population.append(rand)
    population = np.array(population)

    def ga_step(pop):
        new_pop = []
        for _ in range(len(pop)//2):
            a = pop[np.random.choice(len(pop))]
            b = pop[np.random.choice(len(pop))]
            child = crossover(a, b)
            new_pop.append(mutate(child))
        return np.vstack([pop, new_pop])

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.6:
            break
        population = ga_step(population)
        population = population[np.argsort([evaluate(h) for h in population])][:50]

    best_h = population[0]
    def obj(x):
        h = clip_rescale(x)
        return evaluate(h)
    
    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = res.x
    best_c5 = evaluate(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
