    h_bin = np.clip(h_bin, 0, 1)
    h_bin = h_bin * target_sum / np.sum(h_bin)
    initializations.append(h_bin)

    h_block = np.zeros(n_points)
    segment = n_points // 3
    h_block[:segment] = 0.4
    h_block[segment:2*segment] = 0.6
    h_block[2*segment:] = 0.8
    h_block = np.clip(h_block, 0, 1)
    h_block = h_block * target_sum / np.sum(h_block)
    initializations.append(h_block)

    h_sparse = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
    h_sparse[indices] = 1.0
    h_sparse = np.clip(h_sparse, 0, 1)
    h_sparse = h_sparse * target_sum / np.sum(h_sparse)
    initializations.append(h_sparse)

    h_perm = np.random.permutation(initial_h_values)
    h_perm = np.clip(h_perm, 0, 1)
    h_perm = h_perm * target_sum / np.sum(h_perm)
    initializations.append(h_perm)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h, best_c5 = initializations[0].copy(), compute_c5(initializations[0])
    start_time = time.time()

    for h_init in initializations[1:]:
        if time.time() - start_time > budget_s:
            break
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(lambda x: compute_c5(x), h_init, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
