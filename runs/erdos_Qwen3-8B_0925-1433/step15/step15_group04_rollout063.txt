from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def generate_variations(h):
        variations = [
            h[::-1],
            np.fliplr([h])[0],
            np.roll(h, 1),
            (h + h[::-1]) / 2
        ]
        return variations

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    variations = generate_variations(initial_h_values)
    for var in variations:
        h = np.array(var, dtype=float)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        for _ in range(200):
            if time.time() - start_time >= budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-4
                h_plus = h_plus * target_sum / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
                h_minus = h.copy()
                h_minus[i] -= 1e-4

                grad[i] = (corr_plus - corr_minus) / (2e-4)
            h = h - 0.5 * grad
            h = np.clip(h, 0, 1)
