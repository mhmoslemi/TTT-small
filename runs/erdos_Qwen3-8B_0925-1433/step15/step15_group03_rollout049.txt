import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_sine(n):
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        h_sum = np.sum(h)
        return h * target_sum / h_sum

    def generate_random(n):
        h = np.random.rand(n)
        h_sum = np.sum(h)
        return h * target_sum / h_sum

    def de_obj(x):
        h = x.copy()
        h_sum = np.sum(h)
        if h_sum == 0:
            return 1e6
        h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(de_obj, bounds, strategy='best1bin', popsize=20, mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=50, seed=seed)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h_values, 0.381, n_points)

    best_de = result.x
    best_c5 = de_obj(best_de)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        g = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - (h[j_plus] if 0 <= j_plus < n_points else 0.0)
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            g[i] = dx * (term1 - term2)
        return g

    res = minimize(obj, best_de, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    best_h = res.x
