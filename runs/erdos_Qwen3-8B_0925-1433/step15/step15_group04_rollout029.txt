                h_plus[i] += 1e-4
                h_plus = h_plus * target_sum / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
                h_minus = x.copy()
                h_minus[i] -= 1e-4
                h_minus = h_minus * target_sum / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
                grad[i] = (corr_plus - corr_minus) / (2e-4)
            return -grad

        iterations = 0
        max_iterations = 100
        learning_rate = 0.5
        h_opt = h.copy()
        while iterations < max_iterations and time.time() - start_time < budget_s:
            grad = compute_gradient(h_opt)
            h_opt = h_opt + learning_rate * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
            current_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h_opt.copy(), current_c5
            iterations += 1

        if time.time() - start_time < budget_s:
            res = minimize(obj, h_opt, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    initial_scaled = np.clip(initial_scaled, 0, 1)
    current_c5 = np.max(np.correlate(initial_scaled, 1 - initial_scaled, mode='full')) * dx
    if current_c5 < best_c5:
        best_h, best_c5 = initial_scaled, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
