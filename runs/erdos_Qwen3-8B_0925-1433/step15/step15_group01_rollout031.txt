    def evaluate(h_p):

        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def gradient(h_p, k_max):
        h = h_p.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)

            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)

            h_minus = h.copy()
            h_minus[i] -= 1e-4

            h_minus = h_minus * target_sum / np.sum(h_minus)
