import numpy as np
from scipy.optimize import minimize
import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns(h_base):
        patterns = [h_base * target_sum / np.sum(h_base)]
        fft_base = fft(h_base)
        theta = np.random.uniform(0, 2*np.pi, n_points)
        fft_shifted = fft_base * np.exp(1j * theta)
        h_shifted = np.real(ifft(fft_shifted))
        h_shifted = h_shifted * target_sum / np.sum(h_shifted)
        patterns.append(h_shifted)
        h_noise = h_base + np.random.normal(0, 0.05, n_points)
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = h_noise * target_sum / np.sum(h_noise)
        patterns.append(h_noise)
        h_perm = h_base.copy()
        np.random.shuffle(h_perm)
        h_perm = h_perm * target_sum / np.sum(h_perm)
        patterns.append(h_perm)
        return patterns

    initial_patterns = generate_patterns(initial_h_values)

    best_c5 = np.inf
    best_h = initial_h_values.copy()

    for h_start in initial_patterns:
        h = h_start.copy()
        h = h * target_sum / np.sum(h)
        best_c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        def obj(x):
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def compute_gradient(x):
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(x)
            for i in range(n_points):
                h_plus = x.copy()
                h_plus[i] += 1e-4
                h_plus = h_plus * target_sum / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
                h_minus = x.copy()
                h_minus[i] -= 1e-4
                h_minus = h_minus * target_sum / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
                grad[i] = (corr_plus - corr_minus) / (2e-4)
            return -grad

        h_opt = h.copy()
        for _ in range(300):
            if time.time() - time.time() > budget_s:
                break
            grad = compute_gradient(h_opt)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = 0.9 * m[i] + 0.1 * grad[i]
                v[i] = 0.999 * v[i] + 0.001 * grad[i] ** 2
                biased_m = m[i] / (1 - 0.9 ** (_ + 1))
                biased_v = v[i] / (1 - 0.999 ** (_ + 1))
                h_opt[i] += 0.01 * biased_m / (np.sqrt(biased_v) + 1e-8)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
            current_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                h_opt = h_opt.copy()

        res = minimize(obj, h_opt, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5_current:
            best_c5_current = res.fun
            best_h = res.x.copy()

        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = h_opt.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
