                    grad[j] = term1 - term2
                return grad

            res = minimize(obj, h, method='SLSQP', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
            if res.success:
                new_population[0] = res.x

        # Update best
        for h in new_population:
            if compute_c5(h) < best_c5:
                best_h = h
                best_c5 = compute_c5(h)

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
