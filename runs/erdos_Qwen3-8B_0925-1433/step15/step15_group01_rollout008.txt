import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def structured_pattern(m):
        L = 1.0 / m
        d = 1.0 / (m - 1) if m > 1 else 0.0
        h = np.zeros(n_points)
        for i in range(m):
            start = i * (L + d)
            end = start + L
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        eps = 1e-5
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += eps
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= eps
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        return -grad

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381
    time_remaining = budget_s

    candidates = []
    for m in [9, 15, 20, 12, 25]:
        h = structured_pattern(m)
        if h is not None:
            candidates.append(h)

    for h in candidates:
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    start_time = time.time()
    while time_remaining > 0:
        time_remaining = budget_s - time.time() + start_time
        for _ in range(5):
            if time_remaining <= 0:
                break
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
            c5_perturbed = np.max(corr) * dx
            if c5_perturbed < best_c5:
                best_h = perturbed_h
                best_c5 = c5_perturbed
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
            if res.success and res.fun < best_c5:

                best_c5 = res.fun

                break
        if time_remaining <= 0:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
