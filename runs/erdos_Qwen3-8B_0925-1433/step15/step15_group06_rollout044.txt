import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    def clip_and_norm(h):
        h = np.clip(h, 0, 1)
        return normalize(h)

    def calc_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values)
    best_h = normalize(initial_h)
    best_c5 = calc_c5(best_h)

    def generate_initial_population():
        pop = [best_h.copy()]
        for _ in range(19):
            h = np.random.rand(n_points)
            h = normalize(h)
            pop.append(h)
            if np.random.rand() < 0.3:
                h = np.zeros(n_points)
                h[:n_points//2] = np.random.rand(n_points//2)
                h[n_points//2:] = h[:n_points//2][::-1]
                h = normalize(h)
                pop.append(h)
            if np.random.rand() < 0.2:
                h = 0.5 * (1 + np.sin(2 * np.pi * (np.arange(n_points) - n_points/2) / n_points))
                h = normalize(h)
                pop.append(h)
        return pop

    population = generate_initial_population()
    population = [clip_and_norm(h) for h in population]

    def tournament_select(pop, k=5):
        return min(np.random.choice(pop, size=k), key=calc_c5)

    for gen in range(200):
        if time.time() > budget_s:
            break
        evaluations = [calc_c5(h) for h in population]
        best_idx = np.argmin(evaluations)
        best_c5_current = evaluations[best_idx]
        if best_c5_current < best_c5:
            best_h, best_c5 = population[best_idx].copy(), best_c5_current

        new_pop = []
        for _ in range(len(population)//2):
            p1 = tournament_select(population)
            p2 = tournament_select(population)
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:split], p2[split:]])
            child = clip_and_norm(child)
            new_pop.append(child)
        population = new_pop + [tournament_select(population)]

    return (np.clip(best_h, 0, 1), best_c5, n_points)
