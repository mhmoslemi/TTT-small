    h3 = np.random.rand(n_points)
    h3 = h3 * target_sum / np.sum(h3)

    # Define objective function for DE
    def objective_de(x):
        h = x.copy()
        sum_h = np.sum(h)
        h_clamped = np.clip(h, 0, 1)
        sum_clamped = np.sum(h_clamped)
        if sum_clamped == 0:
            return np.inf
        h_norm = h_clamped * (target_sum / sum_clamped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    # Define objective function for L-BFGS-B
    def obj_lbgfs(x):
        h = x.copy()
        sum_h = np.sum(h)
        h_clamped = np.clip(h, 0, 1)
        sum_clamped = np.sum(h_clamped)
        if sum_clamped == 0:
            return np.inf
        h_norm = h_clamped * (target_sum / sum_clamped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    # Define gradient for L-BFGS-B
    def grad_lbgfs(x):
        h = x.copy()
        sum_h = np.sum(h)
        h_clamped = np.clip(h, 0, 1)
        sum_clamped = np.sum(h_clamped)
        if sum_clamped == 0:
            return np.zeros_like(x)
        h_norm = h_clamped * (target_sum / sum_clamped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        max_corr_index = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + max_corr_index
            if 0 <= j_plus < n_points:
                term1 = 1 - h_norm[j_plus]
            else:
                term1 = 0.0
            j_minus = i - max_corr_index
            if 0 <= j_minus < n_points:
                term2 = h_norm[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    # Run DE
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    result_de = differential_evolution(objective_de, bounds, maxiter=50, popsize=20, tol=1e-6, mutation=(0.8, 1), recombination=0.9, strategy='best1bin', polish=True)
    best_de = result_de.x
    best_c5 = result_de.fun
    elapsed = time.time() - start_time

    # Local refinement
    h_local = best_de.copy()
    res_local = minimize(obj_lbgfs, h_local, method='L-BFGS-B', jac=grad_lbgfs, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res_local.success:
        best_local = res_local.x
        best_c5 = res_local.fun
    else:
        best_local = best_de

    # Perturbation refinement
    perturbed = best_local + np.random.normal(0, 0.05, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    res_perturbed = minimize(obj_lbgfs, perturbed, method='L-BFGS-B', jac=grad_lbgfs, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    if res_perturbed.success and res_perturbed.fun < best_c5:
        best_local = res_perturbed.x
        best_c5 = res_perturbed.fun

    best_h = np.clip(best_local, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
