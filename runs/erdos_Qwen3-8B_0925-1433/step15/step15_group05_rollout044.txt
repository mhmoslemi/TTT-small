import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    max_iterations = 100
    alpha = 0.05
    learning_rate_decay = 0.99
    min_lr = 0.001

    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        k_max = np.argmax(corr)
        lag_val = k_max - (n_points - 1)
        gradient = np.array([1 - h[(i + lag_val) % n_points] for i in range(n_points)])
        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        alpha *= learning_rate_decay
        if alpha < min_lr:
            alpha = min_lr

    return (np.clip(best_h, 0, 1), best_c5, n_points)
