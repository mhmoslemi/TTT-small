import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial candidates
    candidates = []
    for _ in range(5):
        perturbed = initial_h_values + np.random.normal(0, 0.15, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        candidates.append(perturbed)
    
    # Add structured patterns
    t = np.linspace(0, 2, n_points)
    tri = 0.5 + 0.5 * np.abs(np.sin(np.pi * t / 2))
    tri = tri * target_sum / np.sum(tri)
    candidates.append(tri)
    
    # Add alternating pattern
    alt = np.zeros(n_points)
    alt[::2] = 1.0
    alt = alt * target_sum / np.sum(alt)
    candidates.append(alt)
    
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    for h_start in candidates:
        res = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
