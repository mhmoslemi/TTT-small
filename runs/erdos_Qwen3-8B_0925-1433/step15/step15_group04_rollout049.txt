import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(h):
        return compute_c5(h)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    # Start with original initial guess
    h_initial = initial_h_values.copy()
    h_initial = h_initial * target_sum / np.sum(h_initial)
    res = minimize(obj, h_initial, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    if res.success:
        current_c5 = compute_c5(res.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = res.x

    # Generate multiple restarted guesses
    num_restarts = 15
    for _ in range(num_restarts):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Perturb initial guess
        perturbation = np.random.normal(0, 0.15, n_points)
        h = initial_h_values + perturbation
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    # Normalize final solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
