import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        elif pattern_type == 'periodic':
            m = np.random.choice([5, 10, 15, 20])
            spacing = 2.0 / (m + 1)
            h = np.zeros(n_points)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    h[idx_start:idx_end] = 1
        elif pattern_type == 'binary':
            h = np.random.choice([0, 1], size=n_points)
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    initial_candidates = [
        generate_initial_candidate('perturb') for _ in range(10) +
