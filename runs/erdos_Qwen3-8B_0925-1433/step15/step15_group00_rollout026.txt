    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        m = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus_m = i + m
            i_minus_m = i - m
            term1 = 1 - h[i_plus_m] if 0 <= i_plus_m < n_points else 0
            term2 = h[i_minus_m] if 0 <= i_minus_m < n_points else 0
            grad[i] = term1 - term2
        return grad

    patterns = generate_patterns(initial_h_values, 5)
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx

    start_time = time.time()
    for pattern in patterns:
        pattern = pattern.copy()
        bounds = [(0, 1)] * n_points
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, pattern, method='L-BFGS-B', bounds=bounds, constraints=constraints, jac=gradient, options={'ftol': 1e-6})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
