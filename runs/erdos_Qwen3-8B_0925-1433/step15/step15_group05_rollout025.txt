        return h * target_sum / np.sum(h)

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        return rescale(h)

    if 'initial_h_values' in globals():
