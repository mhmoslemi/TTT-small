import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    particles = []
    velocities = []

        particles.append(h)
        velocities.append(np.zeros(n_points))

    pbest = [p.copy() for p in particles]

    gbest = particles[np.argmin(pbest_values)]
    gbest_value = min(pbest_values)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        for i in range(population_size):
            # PSO update
            omega, c1, c2 = 0.729, 1.494, 1.494

            velocities[i] = vel

            sum_clipped = np.sum(pos_clipped)
            if sum_clipped == 0:

                pos_clipped = pos_clipped * target_sum / sum_clipped
            particles[i] = pos_clipped

            if c5 < pbest_values[i]:

                pbest_values[i] = c5
