                h_plus = current_h.copy()
                h_plus[i] += 1e-4
                h_plus = h_plus * target_sum / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
                h_minus = current_h.copy()
                h_minus[i] -= 1e-4
                h_minus = h_minus * target_sum / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
                grad[i] = (corr_plus - corr_minus) / (2e-4)
            current_h = current_h + 0.5 * grad
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_corr = np.correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(current_corr) * dx
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            if time.time() - start_time > budget_s:
                break

        res = minimize(
            lambda x: np.max(np.correlate(x * target_sum / np.sum(x), 1 - x * target_sum / np.sum(x), mode='full')) * dx,
            current_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            tol=1e-6
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
