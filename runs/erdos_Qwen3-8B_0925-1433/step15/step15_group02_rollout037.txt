import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def create_periodic(m):
        pattern = np.zeros(n_points)
        spacing = 2.0 / (m + 1)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                pattern[idx_start:idx_end] = 1
        pattern = pattern * (target_sum / np.sum(pattern))
        return pattern

    initial_candidates = []
    if initial_h_values is not None:
        for _ in range(3):
            perturbed = initial_h_values + np.random.uniform(-0.1, 0.1, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            scaled = perturbed * (target_sum / np.sum(perturbed))
            initial_candidates.append(scaled)
        for m in [5, 10, 15]:
            initial_candidates.append(create_periodic(m))
        for _ in range(2):
            ones = np.random.choice(n_points, n_points // 2, replace=False)
            binary = np.zeros(n_points)
            binary[ones] = 1
            scaled = binary * (target_sum / np.sum(binary))
            initial_candidates.append(scaled)

    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    start_time = time.time()

    for candidate in initial_candidates:
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (target_sum / np.sum(candidate))
        res = minimize(obj, candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

        if time.time() - start_time > budget_s:

        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))

        if res.success:

            if current_c5 < best_c5:
                best_h = res.x.copy()
