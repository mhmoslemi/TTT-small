        return h * target_sum / np.sum(h)

                h = np.clip(rescale(h), 0, 1)

                h = np.random.rand(n_points)
                h = rescale(h)
                h += np.random.normal(0, 0.1, n_points)
                h = np.clip(h, 0, 1)
            pop.append(h)
        return np.array(pop)
