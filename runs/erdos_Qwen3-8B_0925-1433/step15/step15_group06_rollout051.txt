import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    # Generate initializations
    def generate_initializations():
        # Original
        yield h_initial
        # Noise
        yield np.clip(h_initial + np.random.uniform(-0.1, 0.1, n_points), 0, 1) * target_sum / np.sum(np.clip(h_initial + np.random.uniform(-0.1, 0.1, n_points), 0, 1))

        num = int(0.2 * n_points)
        h_bin = np.zeros(n_points)
        h_bin[:num] = 0
        h_bin[num:2*num] = 1
        h_bin[2*num:] = 0.5
        yield h_bin * target_sum / np.sum(h_bin)
        # Symmetric
        yield h_initial[::-1] * target_sum / np.sum(h_initial[::-1])
        # Sine
        h_sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 1.5)

        h_start = np.clip(h_start, 0, 1)

            corr = np.correlate(x, 1 - x, mode='full')
