import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def evaluate(h):
        h = clip_scale(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return evaluate(x)

    def grad(x):
        h = clip_scale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1.0 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = (term1 - term2) * dx
        return grad

    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)

    patterns = [initial_h_values.copy(), 
                0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points) / 2), 
                np.random.rand(n_points), 
                np.minimum(np.linspace(0, 1, n_points), np.linspace(1, 0, n_points)), 
                [1 if i % 2 == 0 else 0 for i in range(n_points)]]

    for p in patterns:
        p = np.array(p, dtype=float)
        p = clip_scale(p)
        res = minimize(obj, p, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, 
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(3):
        if time.time() - time.time() > budget_s: break
        noise = np.random.normal(0, 0.05, n_points)
        p = best_h + noise
        p = clip_scale(p)
        res = minimize(obj, p, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, 
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:

    best_h = clip_scale(best_h)
    return (best_h, best_c5, n_points)
