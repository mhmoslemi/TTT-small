from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def init_pop():

        h1 = h1 * target_sum / np.sum(h1)
        h2 = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points) / 2)
        h2 = h2 * target_sum / np.sum(h2)
        h3 = np.random.rand(n_points)
        h3 = h3 * target_sum / np.sum(h3)
        return [h1, h2, h3]

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = (1 - h[j_plus] if 0 <= j_plus < n_points else 0.0)
            j_minus = i - k_max
            term2 = (h[j_minus] if 0 <= j_minus < n_points else 0.0)
            grad[i] = dx * (term1 - term2)
        return grad

    best_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    particles = init_pop()
    velocities = [np.zeros(n_points) for _ in particles]
    pbest = particles.copy()
    pbest_fitness = np.array([obj(p) for p in pbest])
