import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_configs = []

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(10):
        p = initial_h_values + np.random.normal(0, 0.1, n_points)
        p = np.clip(p, 0, 1)
        s = np.sum(p)
        if s > 0:
            p = p * target_sum / s
            initial_configs.append(p)

    for _ in range(5):
        b = np.random.choice([0, 1], size=n_points)
        s = np.sum(b)
        if s > 0:
            b = b * target_sum / s
            initial_configs.append(b)

    for shift in [0, 1, 2, 3, 4]:
        s = np.roll(initial_h_values, shift)
        s = np.clip(s, 0, 1)
        t = np.sum(s)
        if t > 0:
            s = s * target_sum / t
            initial_configs.append(s)

    best_h = initial_h_values.copy() * target_sum / np.sum(initial_h_values)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    for conf in initial_configs:
        res = minimize(obj, conf, method='L-BFGS-B',
                       bounds=[(0, 1)] * n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        res = minimize(obj, best_h, method='L-BFGS-B',

                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-9)

            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
