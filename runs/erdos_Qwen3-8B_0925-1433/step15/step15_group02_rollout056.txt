import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def structured_pattern(m, spacing='uniform'):
        intervals = np.zeros(n_points)
        if spacing == 'uniform':
            spacing_val = 2.0 / (m + 1)
            for i in range(m):
                start = i * spacing_val
                end = (i + 1) * spacing_val
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
        elif spacing == 'random':
            for i in range(m):
                l = np.random.uniform(0.1, 0.3)
                r = l + np.random.uniform(0.1, 0.3)
                start = np.random.uniform(0, 2 - r)
                end = start + r
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
        intervals = intervals * target_sum / np.sum(intervals)
        return intervals

    patterns = []
    for m in [5, 10, 15, 20, 25]:
        patterns.append(structured_pattern(m, 'uniform'))
        patterns.append(structured_pattern(m, 'random'))

    for pattern in patterns:
        h_candidate = pattern.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            curr_c5 = res.fun
            if curr_c5 < best_c5:
                best_h, best_c5 = res.x, curr_c5

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
