            idx = np.random.randint(len(top_candidates))
            h = top_candidates[idx].copy()
            noise = np.random.uniform(-0.1, 0.1, size=n_points)
            h += noise
            h = project(h, target_sum)
            new_population.append(h)
        for _ in range(int(len(population)*0.5)):
            idx1, idx2 = np.random.choice(len(new_population), size=2, replace=False)
            h1, h2 = new_population[idx1], new_population[idx2]
            child = (h1 + h2) / 2
            child = project(child, target_sum)
            new_population.append(child)
        c5_values = [obj(h) for h in new_population]
        sorted_indices = np.argsort(c5_values)
        new_population = [new_population[i] for i in sorted_indices[:population_size]]
        for i in range(len(new_population)):
            h = new_population[i].copy()
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=grad, tol=1e-6)
            new_population[i] = res.x
        population = new_population
        for h in new_population:
            current_c5 = obj(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    best_h = project(best_h, target_sum)
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
