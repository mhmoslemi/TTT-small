        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k_max, dx):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            j1 = i + k_max
            if j1 < n:
                grad[i] += (1 - h[j1]) * dx
            j2 = i - k_max
            if j2 >= 0:
                grad[i] -= h[j2] * dx
        return grad

    start_time = time.time()
    elapsed = 0
    while elapsed < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = compute_gradient(h, k_max, dx)
        learning_rate = 0.5 / (1 + 0.1 * np.sqrt(elapsed))
        h = h + learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        elapsed = time.time() - start_time

    def objective(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1)] * n_points
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
