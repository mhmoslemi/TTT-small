    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.full_like(h, 0.5)
        h = h * target_sum / current_sum
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient_step(h, max_iter=5):
        for _ in range(max_iter):
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                j = (i + k_max) % len(h)
                term1 = (1 - h[j]) * dx
                i2 = (i - k_max) % len(h)
                term2 = h[i2] * dx
                grad[i] = term1 - term2
            h -= grad * 0.01
            h = project(h)
        return h

    initial = np.array(initial_h_values, dtype=float)
