import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def generate_fixed(n_points, m, length):
        dx = 2.0 / n_points
        intervals = np.zeros(n_points)
        for _ in range(m):
            start = np.random.rand() * (2 - length)
            end = start + length
            idx_start = int(start / dx)
            idx_end = int(end / dx)
            idx_end = min(idx_end, n_points)
            intervals[idx_start:idx_end] = 1
        current_sum = np.sum(intervals)
        scale = target_sum / current_sum
        return intervals * scale

    def generate_clustered(n_points, m, cluster_size):
        dx = 2.0 / n_points
        intervals = np.zeros(n_points)
        for _ in range(m):
            start = np.random.rand() * (2 - cluster_size)
            end = start + cluster_size
            idx_start = int(start / dx)
            idx_end = int(end / dx)
            idx_end = min(idx_end, n_points)
            intervals[idx_start:idx_end] = 1
        current_sum = np.sum(intervals)
        scale = target_sum / current_sum
        return intervals * scale

    def compute_gradient(x):
        grad = np.zeros_like(x)
        dx_pert = 1e-4
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += dx_pert
            x_plus = x_plus * target_sum / np.sum(x_plus)
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            val_plus = np.max(corr_plus) * dx

            x_minus = x.copy()
            x_minus[i] -= dx_pert
            x_minus = x_minus * target_sum / np.sum(x_minus)
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')

            grad[i] = (val_plus - val_minus) / (2 * dx_pert)
