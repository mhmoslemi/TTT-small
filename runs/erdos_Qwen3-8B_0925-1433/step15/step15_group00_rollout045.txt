import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial(h):
        fft_h = np.fft.fft(h)
        phase = np.exp(2j * np.random.rand(len(fft_h)))
        h_transformed = np.fft.ifft(fft_h * phase).real
        h_transformed = np.clip(h_transformed, 0, 1)
        h_transformed = h_transformed * target_sum / np.sum(h_transformed)
        return h_transformed

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def eval_early_stop(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    kwargs = {'method': 'L-BFGS-B', 'bounds': [(0, 1)]*n_points, 'constraints': {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, 'tol': 1e-6}

    candidates = [initial_h]
    for _ in range(4):
        candidates.append(generate_initial(initial_h))
        noisy = initial_h + 0.05 * np.random.randn(n_points)
        noisy = np.clip(noisy, 0, 1)
        noisy = noisy * target_sum / np.sum(noisy)
        candidates.append(noisy)

    best_h = initial_h.copy()
    best_c5 = eval_early_stop(best_h)
    start_time = time.time()

    for h_start in candidates:
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(obj, h_start, **kwargs)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
