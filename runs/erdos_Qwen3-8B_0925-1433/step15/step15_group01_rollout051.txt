import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    def _clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    class PSO:
        def __init__(self, n_particles=10, max_iter=1000):
            self.n_particles = n_particles
            self.max_iter = max_iter
            self.pop = np.random.rand(n_particles, n_points) * 2 - 1
            self.pop = _clip_and_normalize(self.pop)
            self.vel = np.random.rand(n_particles, n_points) * 0.01
            self.pbest = self.pop.copy()
            self.pbest_c5 = np.full(n_particles, np.inf)
            self.gbest = self.pop[0].copy()
            self.gbest_c5 = np.inf
            self.inertia = 0.729
            self.c1, self.c2 = 1.494, 1.494

        def _evaluate(self, h):
            if np.sum(h) == 0:
                return np.inf
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        def run(self, budget_s):
            start_time = time.time()
            for i in range(self.max_iter):
                if time.time() - start_time > budget_s:
                    break
                for j in range(self.n_particles):
                    h = self.pop[j]
                    c5 = self._evaluate(h)
                    if c5 < self.pbest_c5[j]:
                        self.pbest[j] = h.copy()
                        self.pbest_c5[j] = c5
                    if c5 < self.gbest_c5:
                        self.gbest = h.copy()
                        self.gbest_c5 = c5
                for j in range(self.n_particles):
                    r1, r2 = np.random.rand(2)
                    self.vel[j] = (self.inertia * self.vel[j] +
                                  self.c1 * r1 * (self.pbest[j] - self.pop[j]) +

                    self.pop[j] = self.pop[j] + self.vel[j]

            return self.gbest, self.gbest_c5

    pso = PSO(n_particles=20, max_iter=1000)
    best_h, best_c5 = pso.run(budget_s)
    best_h = _clip_and_normalize(best_h)
