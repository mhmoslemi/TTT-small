import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

            fitness.append(c5)

            idxs = np.random.choice(pop_size, 5, replace=False)
            best_idx = idxs[np.argmin(fitness[idxs])]
            parents.append(population[best_idx])

        while len(offspring) < pop_size:
