def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial population
    population_size = 40
    particles = []
    velocities = []
    for _ in range(population_size):
        if np.random.rand() < 0.33:
            h = np.array(initial_h_values, dtype=float)
        elif np.random.rand() < 0.5:
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        else:
            h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    pbest = [np.array(p) for p in particles]
    gbest = np.array(particles[0])
    best_c5 = compute_c5(gbest)
    best_h = gbest.copy()

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            h = particles[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(h)

                j_plus = j + k_max
                term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0

                term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
