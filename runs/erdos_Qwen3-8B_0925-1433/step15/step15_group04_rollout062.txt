import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        x = x * target_sum / np.sum(x)
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def initializations(base):
        yield base * target_sum / np.sum(base)
        yield np.concatenate((base[::-1], base[1:])) * target_sum / np.sum(base)
        yield base[::-1] * target_sum / np.sum(base)
        yield np.roll(base, 1) * target_sum / np.sum(base)
        yield (base + np.concatenate((base[::-1], base[1:]))) / 2 * target_sum / np.sum(base)

    best_h = None
    best_c5 = float('inf')

    for init in initializations(initial_h_values):
        h = np.clip(init, 0, 1).copy()
        h = h * target_sum / np.sum(h)
        for _ in range(200):
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)

                hp = h.copy()
                hp[i] += eps

                hp = np.clip(hp, 0, 1)
                corr_plus = np.correlate(hp, 1 - hp, mode='full')[k_max]
                hm = h.copy()
                hm[i] -= eps
                hm = hm * target_sum / np.sum(hm)
                hm = np.clip(hm, 0, 1)
