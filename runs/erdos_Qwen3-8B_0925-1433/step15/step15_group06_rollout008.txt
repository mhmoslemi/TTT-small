import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        eps = 1e-4
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += eps
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= eps
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        return -grad

    def update_h(h, grad, lr=0.5, momentum=0.9, decay=0.999):
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay * v[i] + (1 - decay) * grad[i] ** 2
            biased_m = m[i] / (1 - momentum ** 100)
            biased_v = v[i] / (1 - decay ** 100)
            h[i] = h[i] + lr * biased_m / (np.sqrt(biased_v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        h = update_h(h, grad)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
