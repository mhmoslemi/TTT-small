            gbest_c5 = current_gbest_c5
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] * 0.729 + 1.494 * r1 * (pbest[i] - particles[i]) + 1.494 * r2 * (gbest - particles[i])
            v = np.clip(v, -0.1, 0.1)
            velocities[i] = v
            particles[i] += v
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        if gbest_c5 < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
