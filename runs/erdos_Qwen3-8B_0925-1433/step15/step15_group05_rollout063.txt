        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        if (step + 1) % perturb_interval == 0:
            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    for _ in range(3):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        res = minimize(objective, np.random.uniform(0, 1, n_points), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        res = minimize(objective, np.random.uniform(0, 1, n_points), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
