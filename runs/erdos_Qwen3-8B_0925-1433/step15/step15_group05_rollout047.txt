    best_h = h.copy()

    # Initialize AdaGrad parameters
    eps = 1e-8
    learning_rate = 0.05
    grad_sq = np.zeros(n_points)

    for iteration in range(150):

            break
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_worst = np.argmax(corr)
        m = k_worst - (n_points - 1)
        # Compute gradient for worst lag
        gradient = np.zeros(n_points)

            j = i + m
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]
        # Update AdaGrad
        grad_sq += gradient**2
        lr = learning_rate / (np.sqrt(grad_sq) + eps)

import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    # AdaGrad parameters
    eps = 1e-8
    learning_rate = 0.05
    grad_sq = np.zeros(n_points)

    for iteration in range(300):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_worst = np.argmax(corr)
        m = k_worst - (n_points - 1)
        # Compute gradient for worst lag
        gradient = np.zeros(n_points)
        j = np.arange(n_points) + m
        mask = (j >= 0) & (j < n_points)
        gradient[mask] = 1 - h[j[mask]]
        # AdaGrad update
        grad_sq += gradient ** 2
        lr = learning_rate / (np.sqrt(grad_sq) + eps)
        h -= lr * gradient

        h = np.clip(h, 0, 1)
