import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h, dtype=float)
    best_c5 = 0.381
    max_iterations = 100
    time_start = time.time()
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad
    
    def perturb(h):
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        h_pert = np.clip(h + noise, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        return h_pert
    
    for _ in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=compute_gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        best_h = perturb(best_h)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        c5_perturbed = np.max(corr) * dx
        if c5_perturbed < best_c5:
            best_h = best_h
            best_c5 = c5_perturbed
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
