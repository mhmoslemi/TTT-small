import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        x = np.linspace(0, 2, n_points)
        patterns = [np.array(initial_h_values, dtype=float)]
        sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        patterns.append(sinusoidal)
        triangular = np.minimum(1, np.abs(1 - 2 * (x / 2 - 0.5)))
        patterns.append(triangular)
        random = np.random.rand(n_points)
        patterns.append(random)
        regular = np.zeros(n_points)
        regular[::2] = 1.0
        patterns.append(regular)
        return [p * (target_sum / np.sum(p)) for p in patterns]

    patterns = generate_patterns()
    best_h = patterns[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1.0 if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()

    iter_count = 0
    for h in patterns:
        h = h.copy()
        for _ in range(20):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success:
                h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
                iter_count += 1
                break

    for _ in range(3):
        if time.time() - start_time > budget_s:

        noise = np.random.normal(0, 0.05, n_points)

        if res.success and res.fun < best_c5:
