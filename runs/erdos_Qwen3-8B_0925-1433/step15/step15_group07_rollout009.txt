import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

        total = np.sum(h_clipped)

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = project(x)

        corr = np.correlate(h, 1 - h, mode='full')
