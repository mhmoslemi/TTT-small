import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(x):
        return np.clip(x, 0, 1) * (target_sum / np.sum(x))

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_initial = rescale(np.array(initial_h_values, dtype=float))
    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)

    def gen_patterns(n):
        patterns = []
        for _ in range(n):
            patterns.append(rescale(np.random.rand(n_points)))
        for _ in range(n):
            p = np.zeros(n_points)
            shift = np.random.uniform(0.1, 0.5)
            p = np.roll(p, int(shift * n_points))
            p[:n_points//2] = 1
            patterns.append(rescale(p))
        for _ in range(n):
            p = np.random.randint(0, 2, n_points)
            p = rescale(p)
            patterns.append(p)
        return patterns

    patterns = gen_patterns(10)

    def obj(x):
        return evaluate(x)

    def grad(x):
        eps = 1e-4
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = rescale(x_plus)
            c_plus = evaluate(x_plus)
            x_minus = x.copy()
            x_minus[i] -= eps
            x_minus = rescale(x_minus)
            c_minus = evaluate(x_minus)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        return -grad

    start_time = time.time()
    for p in patterns:
        res = minimize(obj, p, method='L-BFGS-B',
                       bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=grad, tol=1e-7, maxiter=200)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=grad, tol=1e-9, maxiter=100)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
