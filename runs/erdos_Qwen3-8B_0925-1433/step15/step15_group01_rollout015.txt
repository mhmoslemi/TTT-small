import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_finite_diff(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        eps = 1e-4
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += eps
            h_plus = normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= eps
            h_minus = normalize(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        return -grad

    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    time_start = time.time()

    def optimize_candidate(start):
        res = minimize(
            objective,
            start,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(normalize(x)) - target_sum},
            jac=grad_finite_diff,
            tol=1e-5,
            options={'maxiter': 50}
        )
        if res.success:
            return normalize(res.x)
        return start

    candidates = [h_initial]
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        candidates.append(normalize(h_initial + perturb))

    for i, start in enumerate(candidates):
        if time.time() - time_start > budget_s:
            break
        h = optimize_candidate(start)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        # Add perturbations
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        h_pert = normalize(h + perturb)
        c5_pert = objective(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert

    return (best_h, best_c5, n_points)
