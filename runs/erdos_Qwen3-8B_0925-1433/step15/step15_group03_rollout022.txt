import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def get_initial_guesses():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = h0 * target_sum / np.sum(h0)
        h1 = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points) / 2)
        h1 = h1 * target_sum / np.sum(h1)
        h2 = np.random.uniform(0, 1, n_points)
        h2 = h2 * target_sum / np.sum(h2)
        return [h0, h1, h2]

    def project(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_guesses = get_initial_guesses()
    particles = [np.copy(g) for g in initial_guesses]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in initial_guesses]
    pbest = [g.copy() for g in particles]
    pbest_val = [obj(g) for g in particles]
    gbest = np.copy(particles[np.argmin(pbest_val)])
    gbest_val = np.min(pbest_val)

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            h = particles[i]
            vel = velocities[i]
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel = 0.7 * vel + 1.5 * r1 * (pbest[i] - h) + 1.5 * r2 * (gbest - h)
            # Add gradient-based adjustment
            if np.max(pbest_val) < 1e10:
                grad = compute_gradient(h)
                vel += 0.2 * grad
            # Update position
            h_new = h + vel
            h_new = project(h_new)
            particles[i] = h_new
            # Update pbest
            current_val = obj(h_new)
            if current_val < pbest_val[i]:
                pbest[i] = h_new.copy()
                pbest_val[i] = current_val
        # Update gbest
        idx = np.argmin(pbest_val)
        if pbest_val[idx] < gbest_val:
            gbest = pbest[idx].copy()
            gbest_val = pbest_val[idx]

    # Local refinement
    candidates = [gbest]
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = gbest + noise
        perturbed_h = project(perturbed_h)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient,
                       bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            candidates.append(res.x)
    best_h = np.min([obj(h) for h in candidates], axis=0)
    best_h = candidates[np.argmin([obj(h) for h in candidates])]
    best_h = project(best_h)
    best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
