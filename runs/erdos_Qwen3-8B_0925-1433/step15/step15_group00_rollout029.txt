from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h_phase_norm = h_phase_clipped * target_sum / np.sum(h_phase_clipped)
    
    h_noise = h_initial + np.random.normal(0, 0.05, h_initial.shape)
    h_noise_clipped = np.clip(h_noise, 0, 1)
    h_noise_norm = h_noise_clipped * target_sum / np.sum(h_noise_clipped)

    for pattern in initial_patterns:

            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode="full")
