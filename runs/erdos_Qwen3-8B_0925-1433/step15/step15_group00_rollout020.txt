import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_func(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += 1e-4
            x_plus = x_plus * target_sum / np.sum(x_plus)
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')[k_max]
            x_minus = x.copy()
            x_minus[i] -= 1e-4
            x_minus = x_minus * target_sum / np.sum(x_minus)
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    patterns = []
    h0 = np.array(initial_h_values, dtype=float)
    patterns.append(h0)
    fft0 = fft(h0)
    phase = np.exp(2j * np.pi * np.random.rand(n_points))
    h_phase = ifft(fft0 * phase).real
    h_phase = np.clip(h_phase, 0, 1)
    h_phase = h_phase * target_sum / np.sum(h_phase)
    patterns.append(h_phase)
    h_noise = h0 + 0.05 * np.random.randn(n_points)
    h_noise = np.clip(h_noise, 0, 1)
    h_noise = h_noise * target_sum / np.sum(h_noise)
    patterns.append(h_noise)
    h_reverse = h0[::-1].copy()
    h_reverse = h_reverse * target_sum / np.sum(h_reverse)
    patterns.append(h_reverse)

    for init_h in patterns:
        h = np.copy(init_h)
        h = h * target_sum / np.sum(h)
        start_time = time.time()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = grad_func(h)
            h += 0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
