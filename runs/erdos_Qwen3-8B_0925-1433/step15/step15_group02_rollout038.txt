            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                structured[idx_start:idx_end] = 1
        structured = structured * target_sum / np.sum(structured)
        initial_candidates.append(structured)
        binary = (np.random.rand(n_points) < 0.5).astype(float)
        binary = binary * target_sum / np.sum(binary)
        initial_candidates.append(binary)

    start_time = time.time()
    for candidate in initial_candidates:
        h_candidate = np.clip(candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=compute_gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        if time.time() - start_time > budget_s:
            break

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=compute_gradient, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
