    # Optimization parameters
    alpha = 0.05
    decay_rate = 0.99
    max_iter = 100
    time_start = time.time()
    
    for i in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_max = np.clip(k_max, 1, n_points - 1)  # Avoid edge cases
        
        # Compute gradient for worst lag
        grad = np.zeros(n_points)
        for idx in range(n_points):

        # Update h
