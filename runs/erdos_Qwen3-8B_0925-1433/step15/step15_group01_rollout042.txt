import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in locals():

    else:

    n_particles = 10

    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_c5 = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    global_best = particles[0].copy()

    for i in range(1, n_particles):
