import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_1 = np.array(initial_h_values, dtype=float)
    initial_2 = np.sin(np.pi * np.linspace(0, 2, n_points) / 2)
    initial_2 = initial_2.clip(0, 1) * target_sum / np.sum(initial_2)
    initial_3 = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))

    def obj(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.inf
        h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for init in [initial_1, initial_2, initial_3]:
        if time.time() - start_time >= budget_s:
            break
        result = differential_evolution(obj, bounds, popsize=20, strategy='best1bin', maxiter=100, tol=1e-6, seed=seed)
        if result.success:
            h = result.x
            current_c5 = obj(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
