from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initial patterns
    x = np.linspace(0, 2, n_points)

        p = p * target_sum / np.sum(p)
        candidates.append(p)

    best_h = candidates[0].copy()
    best_c5 = compute_c5(best_h, dx)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):

        corr = np.correlate(h, 1 - h, mode='full')

        for i in range(n_points):

            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0

        return grad

    start_time = time.time()

    # Initial batch optimization
    for h in candidates:
