import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate diverse initializations
    initializations = []

    # Original
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)

    h = initial_h_values + np.random.uniform(-0.2, 0.2, n_points)

    initializations.append(h)

    # Binary patch
    mask = np.random.choice(n_points, int(0.2 * n_points), replace=False)
    h = initial_h_values.copy()
    h[mask] = 0.0
    h = h * target_sum / np.sum(h)
    initializations.append(h)

    # Symmetric
    h = initial_h_values.copy()
    h = np.concatenate([h[:n_points//2], h[:n_points//2][::-1]])
    h = h * target_sum / np.sum(h)
    initializations.append(h)

    # Sine wave
    h = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 2, n_points) * 1.5)
    h = h * target_sum / np.sum(h)
    initializations.append(h)

    # Objective function
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    for init in initializations:

            if res.success and res.fun < best_c5:

                best_h = res.x.copy()
        except:
            pass

    # Final clipping
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
