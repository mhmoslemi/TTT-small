import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initializations
    init_points = []
    # Original
    h_initial = initial_h_values.copy()
    h_initial = h_initial * target_sum / np.sum(h_initial)
    init_points.append(h_initial)
    # Random noise
    h_noise = initial_h_values.copy()
    h_noise += np.random.normal(0, 0.1, n_points)
    h_noise = np.clip(h_noise, 0, 1)
    h_noise = h_noise * target_sum / np.sum(h_noise)
    init_points.append(h_noise)
    # Binary patches
    h_binary = initial_h_values.copy()
    binary_mask = np.zeros(n_points, dtype=bool)
    np.random.choice(range(n_points), size=int(n_points * 0.2), replace=False, out=binary_mask)
    h_binary[binary_mask] = 0
    binary_mask2 = np.zeros(n_points, dtype=bool)
    np.random.choice(range(n_points), size=int(n_points * 0.2), replace=False, out=binary_mask2)
    h_binary[binary_mask2] = 1
    h_binary = h_binary * target_sum / np.sum(h_binary)
    init_points.append(h_binary)
    # Symmetric
    h_sym = initial_h_values[::-1].copy()
    h_sym = h_sym * target_sum / np.sum(h_sym)
    init_points.append(h_sym)
    # Sine wave
    h_sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 1.5)
    h_sine = h_sine * target_sum / np.sum(h_sine)
    init_points.append(h_sine)

    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    start_time = time.time()
    for h0 in init_points:
        if time.time() - start_time > budget_s:
            break
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * target_sum / np.sum(h0)
        res = minimize(compute_c5, h0, method='SLSQP', bounds=[(0, 1)] * n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
