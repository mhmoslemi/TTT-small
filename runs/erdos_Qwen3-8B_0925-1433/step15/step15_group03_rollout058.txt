import numpy as np
from scipy.optimize import minimize
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
from scipy.stats import norm

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        sum_h = np.sum(h)
        factor = (target_sum) / sum_h
        return h * factor

    def generate_triangular():
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i]
            else:
                h[i] = 2 - x[i]
        sum_h = np.sum(h)
        factor = (target_sum) / sum_h
        return h * factor

    def generate_regular():
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 2 == 0 else 0
        sum_h = np.sum(h)
        factor = (target_sum) / sum_h
        return h * factor

    def generate_random():
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        factor = (target_sum) / sum_h
        return h * factor

    initial_candidates = [
        initial_h_values.copy(),
        generate_sinusoidal(),
        generate_random(),
        generate_triangular(),
        generate_regular(),
    ]

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    candidates = []
    for h in initial_candidates:
        candidates.append((h, compute_c5(h)))

    best_h, best_c5 = candidates[0]
    for h, c in candidates:
        if c < best_c5:
            best_h, best_c5 = h, c

    X = np.array([c[0] for c in candidates])
    y = np.array([c[1] for c in candidates])

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) * ConstantKernel(1.0, (1e-3, 1e3))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=0.1, n_restarts_optimizer=10)

    def acquisition(x):
        mu, std = gp.predict(x, return_std=True)
        mu = mu.flatten()
        std = std.flatten()
        z = (mu - best_c5) / std
        ei = (mu - best_c5) * norm.cdf(z) + std * norm.pdf(z)
        return -ei  # minimize

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        X_new = np.random.rand(1, n_points)
        y_new = compute_c5(X_new[0])
        X = np.vstack((X, X_new))
        y = np.append(y, y_new)
        gp.fit(X, y)
        X_next = np.atleast_2d(gp.kernel_.length_scale * np.random.randn(n_points))
        res = minimize(acquisition, X_next, bounds=[(0, 1)] * n_points, method='L-BFGS-B')
        X_next = res.x
        y_new = compute_c5(X_next)
        X = np.vstack((X, X_next))
        y = np.append(y, y_new)

    best_h = best_h
    for h, c in zip(X, y):
        if c < best_c5:
            best_h, best_c5 = h, c

    perturbed = best_h + 0.05 * np.random.randn(n_points)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    res = minimize(lambda x: compute_c5(x), perturbed, bounds=[(0, 1)] * n_points, method='L-BFGS-B')
    best_h = res.x
    best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
