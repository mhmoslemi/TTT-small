        h = h * scale
        h = np.clip(h, 0, 1)
        return h

        return h

        h[indices] = 1.0
        total = np.sum(h)
        scale = target_sum / total
        h = h * scale
        h = np.clip(h, 0, 1)
        return h
    initializations.append(generate_sparse_pattern(n_points))
