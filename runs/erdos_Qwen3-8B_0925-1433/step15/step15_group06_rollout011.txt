    h_bin[mask_0] = 0.0
    h_bin[mask_1] = 1.0
    h_bin = np.clip(h_bin, 0, 1)
    h_bin = h_bin * target_sum / np.sum(h_bin)
    initial_candidates.append(h_bin)

    # Symmetric
    h_sym = np.zeros(n_points)
    h_sym[:n_points//2] = initial_h_values[:n_points//2]
    h_sym[n_points//2+1:] = initial_h_values[:n_points//2][::-1]
    h_sym = np.clip(h_sym, 0, 1)
    h_sym = h_sym * target_sum / np.sum(h_sym)
    initial_candidates.append(h_sym)

    # Sine wave
    h_sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 1.5)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * target_sum / np.sum(h_sine)
    initial_candidates.append(h_sine)

    best_h = None
    best_c5 = float('inf')

    start_time = time.time()

    for init in initial_candidates:
        h = np.clip(init, 0, 1)
        h = h * target_sum / np.sum(h)
        try:
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    h_fallback = np.array(initial_h_values, dtype=float)
    h_fallback = np.clip(h_fallback, 0, 1)
    h_fallback = h_fallback * target_sum / np.sum(h_fallback)
    current_c5 = obj(h_fallback)
    if current_c5 < best_c5:
        best_h = h_fallback.copy()
        best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
