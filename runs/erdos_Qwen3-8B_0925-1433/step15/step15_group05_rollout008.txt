import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)

        h = np.array(x, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points

    # GA
    pop_size = 50
    generations = 10
    mutation_rate = 0.1
    crossover_rate = 0.7
    start_time = time.time()
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        new_pop = []
        for _ in range(pop_size // 2):
            a, b = np.random.choice(pop_size, 2, replace=False)
            if fitness[a] < fitness[b]:
                parent = population[a]
            else:
                parent = population[b]
            if np.random.rand() < crossover_rate:
                child = (parent + np.random.choice(population, 1)[0]) / 2
            else:
                child = parent.copy()
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
