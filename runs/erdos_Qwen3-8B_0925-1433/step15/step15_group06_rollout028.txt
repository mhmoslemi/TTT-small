            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.15 * mut[idx], 0.15 * mut[idx])
                mut[idx] += delta
            mut = mut * target_sum / np.sum(mut)
            mut = np.clip(mut, 0, 1)
            mutated.append(mut)
        sorted_idx = np.argsort(fitness)
        for i in range(len(mutated)):
            if i < len(sorted_idx):
                population[sorted_idx[-(i+1)]] = mutated[i]
        if fitness[best_idx] == best_c5:
            no_improvement += 1
        else:
            no_improvement = 0
        if no_improvement >= 5:
            break
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (np.clip(best_h, 0, 1), final_c5, n_points)
