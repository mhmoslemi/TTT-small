import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values and apply scaling
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = np.clip(x, 0, 1)
        cor = np.correlate(x, 1 - x, mode='full')
        return np.max(cor) * dx

    def compute_gradient(x):
        x = np.clip(x, 0, 1)
        cor = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(cor)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = np.clip(h_minus, 0, 1)
            cor_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (cor_plus - cor_minus) / (2e-4)
        return -grad

    # Generate initializations
    initializations = []
    initializations.append(h.copy())
    perturb = np.random.normal(0, 0.2, n_points)
    for _ in range(3):

        initializations.append(new_h.copy())

    for init_h in initializations:
        # Constraint: sum(x) = target_sum
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
