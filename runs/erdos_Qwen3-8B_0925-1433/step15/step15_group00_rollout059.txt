import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_points = [np.array(initial_h_values, dtype=float)]
    for _ in range(4):
        noise = np.random.normal(0, 0.03, n_points)
        perturbed = np.clip(initial_h_values + noise, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_points.append(perturbed)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for point in initial_points:
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            point,
            method='L-BFGS-B',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6
        )
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
