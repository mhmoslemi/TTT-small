            grad[i] = (overlap_perturbed - overlap_original) / (2 * epsilon_grad)
        
        grad_squared += grad ** 2
        learning_rate = initial_learning_rate / np.sqrt(grad_squared + epsilon)
        h = h - learning_rate * grad
        
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        if iteration % 50 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
