import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_to_sum(candidate):
        s = np.sum(candidate)
        if s == 0:
            return candidate
        scale = target_sum / s
        scaled = candidate * scale
        return np.clip(scaled, 0, 1)

    def objective(x):
        x = scale_to_sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return -np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    def compute_gradient(x):
        x = scale_to_sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - x[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = x[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_guesses = [initial_h_values]
    x = np.linspace(0, 2, n_points)
    initial_guesses.append(0.5 + 0.5 * np.sin(np.pi * x) * target_sum / np.sum(0.5 + 0.5 * np.sin(np.pi * x)))
    initial_guesses.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    best_h = initial_h_values
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx

    start_time = time.time()
    for guess in initial_guesses:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=20, mutation=(0.5, 1), recombination=0.9,
            tol=1e-6, maxiter=100, polish=True, callback=lambda x: None,
            options={'mutation_factor': 0.8, 'mutation': 'self', 'recombination': 'blend'}
        )
        current_c5 = -result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break

    result = minimize(

    )
    best_h = result.x
    best_c5 = -result.fun

        perturbed = scale_to_sum(perturbed)
        perturbed = np.clip(perturbed, 0, 1)
        result = minimize(
            objective, perturbed, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
