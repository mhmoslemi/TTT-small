import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    initializations = [initial_h.copy()]
    perturb = np.random.uniform(-0.1, 0.1, n_points)
    perturbed = np.clip(initial_h + perturb, 0, 1)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    initializations.append(perturbed)

    binary = np.copy(initial_h)
    indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
    binary[indices] = 0.0
    indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
    binary[indices] = 1.0
    binary = np.clip(binary, 0, 1)
    binary = binary * target_sum / np.sum(binary)
    initializations.append(binary)

    symmetric = np.copy(initial_h)
    symmetric = np.clip(symmetric, 0, 1)
    symmetric = symmetric * target_sum / np.sum(symmetric)
    symmetric = np.concatenate([symmetric[::-1], symmetric[1:-1]])

    sine_wave = sine_wave * target_sum / np.sum(sine_wave)
