import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / np.sum(h)

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = project(initial_h)
    pop_size = 50
    population = [initial_h.copy()] * pop_size
    for i in range(pop_size):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        population[i] = project(initial_h + noise)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    while time.time() - start_time < budget_s:
        c5_vals = np.array([compute_c5(h) for h in population])
        indices = np.argsort(c5_vals)
        top_indices = indices[:int(0.2 * pop_size)]
        top_candidates = [population[i] for i in top_indices]
        mutated = []
        for _ in range(int(0.5 * pop_size)):
            h = population[np.random.choice(pop_size)].copy()
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h += noise
            mutated.append(project(h))
        crossover = []
        for i in range(0, len(top_candidates), 2):
            if i + 1 < len(top_candidates):
                a, b = top_candidates[i], top_candidates[i + 1]
                child = (a + b) / 2
                crossover.append(project(child))
        new_pop = top_candidates + mutated + crossover
        new_pop = np.array(new_pop)
        new_pop = new_pop[:pop_size]
        for i in range(pop_size):

            for _ in range(5):
                g = grad(h)
                h -= 0.01 * g

            new_pop[i] = h
        population = new_pop
        best_c5 = min(best_c5, np.min(c5_vals))

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
