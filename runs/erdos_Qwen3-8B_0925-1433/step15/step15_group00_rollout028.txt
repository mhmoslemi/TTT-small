        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    for guess in initial_guesses:
        h = guess.copy()
        h = h * target_sum / np.sum(h)
        best_h_current = h.copy()
        best_c5_current = compute_c5(best_h_current)
        start_time = time.time()

        learning_rate = 0.3
        momentum = 0.8
        decay_rate = 0.99
        max_iterations = 300
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = momentum * m[i] + (1 - momentum) * grad[i]
                v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
                biased_corrected_m = m[i] / (1 - momentum ** (_ + 1))
                biased_corrected_v = v[i] / (1 - decay_rate ** (_ + 1))
                h[i] = h[i] + learning_rate * biased_corrected_m / (np.sqrt(biased_corrected_v) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5_current:
                best_h_current, best_c5_current = h.copy(), current_c5

        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, best_h_current, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
            if res.success and res.fun < best_c5_current:
                best_h_current, best_c5_current = res.x, res.fun

        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current

    return (np.clip(best_h, 0, 1), best_c5, n_points)
