        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_noise = initial_h + noise
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = (h_noise * target_sum / np.sum(h_noise)).clip(0, 1)
        patterns.append(h_noise)

    for h_initial in patterns:
        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1)] * n_points
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
