        rand_h = np.random.rand(n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        particles.append(rand_h.copy())

    velocities = np.zeros((len(particles), n_points))
    best_h = initial_h.copy()
    best_c5 = compute_C5(initial_h)

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            h = particles[i]
            current_c5 = compute_C5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        for i in range(len(particles)):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros(n_points)
            for j in range(n_points):
                j_plus = j + k_max
                term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
                j_minus = j - k_max
                term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
                grad[j] = dx * (term1 - term2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * np.random.rand() * (best_h - h) + 1.5 * np.random.rand() * (best_h - h) + 0.2 * grad
            new_h = h + velocities[i]
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            particles[i] = new_h.copy()

    top_candidates = sorted(particles, key=compute_C5)[:5]
    for h in top_candidates:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            new_h = res.x
            new_c5 = res.fun
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
