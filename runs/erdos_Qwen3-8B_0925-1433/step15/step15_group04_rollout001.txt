import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        if time.time() - start_time > budget_s:

        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = initial_h_values + perturb

        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds,
