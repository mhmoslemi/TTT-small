def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    best_h = initial_h.copy()

    population = [initial_h.copy()]
    for _ in range(19):
        if np.random.rand() < 0.5:
            rand_h = np.random.rand(n_points)
            rand_h = rand_h * target_sum / np.sum(rand_h)
            rand_h = np.clip(rand_h, 0, 1)
            population.append(rand_h)
        else:

            mirrored = mirrored * target_sum / np.sum(mirrored)
            mirrored = np.clip(mirrored, 0, 1)
            population.append(mirrored)

        corr = np.correlate(h, 1 - h, mode='full')

    best_c5 = evaluate(best_h)
