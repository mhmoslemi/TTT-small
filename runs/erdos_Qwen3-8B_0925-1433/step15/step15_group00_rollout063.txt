    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(best_h)
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = 0.9 * m[i] + 0.1 * grad[i]
            v[i] = 0.99 * v[i] + 0.01 * grad[i]**2
            best_h[i] = np.clip(best_h[i] + 0.3 * m[i] / (np.sqrt(v[i]) + 1e-8), 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
