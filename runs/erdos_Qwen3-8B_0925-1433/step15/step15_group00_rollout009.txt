from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        patterns = [np.array(initial_h_values, dtype=float)]
        h = np.zeros(n_points)
        num_ones = n_points // 2
        h[:num_ones] = 1.0
        h *= target_sum / np.sum(h)
        patterns.append(h)
        for i in range(10):
            h = np.array(initial_h_values, dtype=float)
            h += np.random.uniform(-0.2, 0.2, n_points)
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
            patterns.append(h)
        return patterns

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    patterns = generate_patterns()
    start_time = time.time()

    for h in patterns:
        if time.time() - start_time > budget_s:
            break
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h = np.clip(best_h, 0, 1)
