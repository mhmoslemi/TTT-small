import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(h):
        return h * target_sum / np.sum(h)

    def clip_to_bounds(h):
        return np.clip(h, 0, 1)

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Initialize population
    population = []
    parent = np.array(initial_h_values, dtype=float)
    population.append(parent)
    population.extend([np.random.uniform(0, 1, n_points) for _ in range(5)])
    for i in range(5):
        sym = np.concatenate([parent[:n_points//2], parent[n_points//2:][::-1]])
        population.append(sym)
    population = [normalize(clip_to_bounds(h)) for h in population]
    population = np.array(population)
    best_h = population[np.argmin([compute_c5(h) for h in population])]
    best_c5 = compute_c5(best_h)

    def tournament_selection(pop):
        idxs = np.random.choice(len(pop), size=5, replace=False)
        return pop[idxs[np.argmin([compute_c5(pop[i]) for i in idxs])]]

    for _ in range(100):
        if time.time() > time.time() + budget_s:
            break
        new_pop = []
        for _ in range(len(population) // 2):
            p1 = tournament_selection(population)
            p2 = tournament_selection(population)
            split = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([p1[:split], p2[split:]])
            child2 = np.concatenate([p2[:split], p1[split:]])
            child1 = normalize(clip_to_bounds(child1))
            child2 = normalize(clip_to_bounds(child2))
            new_pop.extend([child1, child2])
        population = np.array(new_pop)
        for _ in range(len(population) // 2):
            idx = np.random.randint(len(population))
            h = population[idx].copy()
            mask = np.random.rand(n_points) < 0.1
            h[mask] += np.random.uniform(-0.15, 0.15, np.sum(mask))
            h = normalize(clip_to_bounds(h))
            population[idx] = h
        curr_best = population[np.argmin([compute_c5(h) for h in population])]
        if compute_c5(curr_best) < best_c5:
            best_h, best_c5 = curr_best, compute_c5(curr_best)
        if np.allclose(best_c5, compute_c5(best_h), atol=1e-8):
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
