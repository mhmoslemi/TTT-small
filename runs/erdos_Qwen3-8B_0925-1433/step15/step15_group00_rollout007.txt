from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    def generate_population(parent, n_pop=200):
        pop = [np.array(parent, dtype=float)]
        for _ in range(n_pop-1):
            perturb = np.random.uniform(-0.15, 0.15, n_points)
            cand = parent + perturb
            cand = np.clip(cand, 0, 1)
            cand = cand * target_sum / np.sum(cand)
            pop.append(cand)
        return np.array(pop)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(h):
        return h * target_sum / np.sum(h)

    def local_refine(h):
        res = minimize(lambda x: compute_c5(x), h, method='L-BFGS-B',
                       bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        return res.x if res.success else h

    pop = generate_population(initial_h_values)
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    generations = 0
    while time.time() - start_time < budget_s:
        generations += 1
        # Select top 20%
        top_idx = np.argsort([compute_c5(p) for p in pop])[:int(0.2 * len(pop))]
        top_pop = pop[top_idx]
        
        # Crossover
        new_pop = []
        for i in range(0, len(top_pop), 2):
            if i+1 < len(top_pop):
                p1, p2 = top_pop[i], top_pop[i+1]
                child = (p1 + p2) / 2 + np.random.normal(0, 0.05, n_points)
                new_pop.append(project(np.clip(child, 0, 1)))
        
        # Mutation
        for p in top_pop:
            noise = np.random.normal(0, 0.05, n_points)
            mutant = p + noise
            mutant = project(np.clip(mutant, 0, 1))
            new_pop.append(mutant)
        
        # Local refinement
        for p in top_pop[:int(0.5 * len(top_pop))]:
            refined = local_refine(p)
            new_pop.append(refined)
        
        # Replace worst 50%
        all_scores = [compute_c5(p) for p in new_pop]
        replace_idx = np.argsort(all_scores)[-int(0.5 * len(new_pop)):]
        for i in replace_idx:

            new_pop[i] = project(new_pop[i])
