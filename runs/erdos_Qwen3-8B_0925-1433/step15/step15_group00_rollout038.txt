import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    h_initial = h_initial * target_sum / np.sum(h_initial)

        return np.max(corr) * dx
    
    pop = [h_initial.copy()]
    for _ in range(4):
        perturbed = h_initial + 0.1 * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        pop.append(perturbed)
    for _ in range(2):
        bin_h = np.random.rand(n_points) < 0.5
        bin_h = bin_h * target_sum / np.sum(bin_h)
        pop.append(bin_h)
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        evaluations = [(compute_c5(h), h) for h in pop]
        evaluations.sort()
        top_idx = int(len(pop) * 0.2)
        parents = evaluations[:top_idx]
        
        new_pop = []
        for i in range(0, len(parents), 2):
            h1, h2 = parents[i][1], parents[i+1][1]
            child = (h1 + h2) / 2.0
            child += 0.05 * np.random.randn(n_points)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        for _ in range(5):
            rand_h = np.random.rand(n_points) * 2.0 - 1.0
            rand_h = np.clip(rand_h, 0, 1)

            new_pop.append(rand_h)
        
        all_candidates = pop + new_pop
