    def objective(x):
        h_val = x * target_sum / np.sum(x)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()
