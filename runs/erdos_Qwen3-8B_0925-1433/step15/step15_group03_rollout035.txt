import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial candidates
    initial = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    sin_x = np.linspace(0, 2, n_points)
    sin_h = 0.5 + 0.5 * np.sin(np.pi * sin_x / 2)
    tri_h = np.clip(np.abs(np.linspace(-1, 1, n_points)), 0, 1)
    tri_h = tri_h * target_sum / np.sum(tri_h)
    rand_h = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    candidates = [initial, sin_h, tri_h, rand_h]

    best_h = initial
    best_c5 = obj(initial)

    start_time = time.time()

    # Global optimization
    try:
        result = differential_evolution(obj, bounds, strategy='best1bin', maxiter=50, popsize=8, tol=1e-6, mutation=(0.5, 0.9), recombination=0.8, polish=True, disp=False)
        best_h = result.x
        best_c5 = result.fun
    except:
        pass

    # Local refinement
    def grad(h):
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)

        for i in range(n_points):
