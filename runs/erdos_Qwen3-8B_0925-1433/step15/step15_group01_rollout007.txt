def run(seed=42, budget_s=600):

        h_candidate = h + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        h_copy = h_candidate.copy()

            corr = np.correlate(h_copy, 1 - h_copy, mode='full')
            c5 = np.max(corr) * dx

                best_h = h_copy.copy()
            k_max = np.argmax(corr)
