    for variation in variations:
        variation = variation * target_sum / np.sum(variation)
        try:
            res = minimize(obj, variation, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5
        except:
            pass

    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
