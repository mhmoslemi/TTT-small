import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - target_sum

    def update_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Initial candidates
    candidates = []
    base = np.array(initial_h_values, dtype=float)
    candidates.append(base)
    for _ in range(3):
        noise = np.random.uniform(-0.2, 0.2, n_points)
        h = base + noise
        h = update_h(h)
        candidates.append(h)

    best_h = base.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for h in candidates:
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1)]*n_points
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    # Perturbation phase
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.uniform(-0.15, 0.15, n_points)
        h_pert = best_h + perturb
        h_pert = update_h(h_pert)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        c5 = compute_c5(h_pert)
        if c5 < best_c5:
            best_h = h_pert
            best_c5 = c5

    best_h = update_h(best_h)
    return (best_h, best_c5, n_points)
