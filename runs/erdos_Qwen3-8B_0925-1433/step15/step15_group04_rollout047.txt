            h_plus = h_current.copy()
            h_plus[i] += eps
            norm_plus = np.sum(h_plus)
            h_plus = h_plus * target_sum / norm_plus
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k_max] * dx

            h_minus = h_current.copy()
            h_minus[i] -= eps
            norm_minus = np.sum(h_minus)
            h_minus = h_minus * target_sum / norm_minus
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k_max] * dx

            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad

    while time.time() - start_time < budget_s - 10:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        current_c5 = corr[k_max] * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        grad = grad_func(h, k_max)
        learning_rate = 0.5 / (iterations + 1)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        iterations += 1

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, final_c5)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
