        n_points = 45
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * 100 / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_scale(h):
        clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:
            return np.zeros_like(clipped)
        scaled = clipped * (target_sum / sum_clipped)
        return np.clip(scaled, 0.0, 1.0)

    population_size = 50
    population = []
    for _ in range(population_size):
        h = initial_h.copy()
        noise = np.random.normal(0.0, 0.05, size=n_points)
        h = h + noise
        h = clip_and_scale(h)
        population.append(h)

    generations = 30
    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()

        new_population = []
        for _ in range(population_size // 2):
            idx1, idx2 = np.random.choice(range(population_size), size=2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            child1 = (parent1 + parent2) / 2.0
            noise = np.random.normal(0.0, 0.05, size=n_points)
            child1 = child1 + noise
            child1 = clip_and_scale(child1)
            new_population.append(child1)
            child2 = (parent1 + parent2) / 2.0
            noise = np.random.normal(0.0, 0.05, size=n_points)
            child2 = child2 + noise
            child2 = clip_and_scale(child2)
            new_population.append(child2)

        temp = list(zip(fitness, population))
        temp.sort()
        population = [x[1] for x in temp[:population_size]]

        # Introduce diversity
        for _ in range(population_size // 10):
            h = np.random.rand(n_points)
            h = clip_and_scale(h)
            population.append(h)

    # Final optimization
    fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
