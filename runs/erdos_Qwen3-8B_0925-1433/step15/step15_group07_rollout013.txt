import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    best_h = h.copy()

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / total

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(x)

            i2 = (i - k_max) % len(x)
