import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    t0 = time.time()
    for _ in range(5):
        if time.time() - t0 > budget_s:
            break
        # Random restarts
        x0 = np.random.uniform(0, 1, n_points)
        x0 = x0 * target_sum / np.sum(x0)
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        # Perturb best_h
        perturb = np.random.normal(0, 0.08, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        # Local refinement
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
