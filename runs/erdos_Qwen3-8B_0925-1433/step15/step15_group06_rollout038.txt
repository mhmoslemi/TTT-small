            current_c5 = compute_c5(h)
            if current_c5 < particles[i]['pbest_value']:
                particles[i]['pbest'] = h.copy()
                particles[i]['pbest_value'] = current_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel = omega * velocity[i] + c1 * r1 * (particles[i]['pbest'] - h) + c2 * r2 * (best_h - h)
            vel = np.clip(vel, -0.1, 0.1)
            h_new = h + vel
            h_clamped = np.clip(h_new, 0, 1)
            h_normalized = h_clamped * target_sum / np.sum(h_clamped)
            particles[i]['position'] = h_normalized.copy()
            velocity[i] = vel.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
