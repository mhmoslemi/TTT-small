```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):  # Minimize max_k C(k)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Global optimization with DE
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-6,
        mutation=(0.8, 1),
        recombination=0.9,
        polish=True,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = initial_h_values
        h = np.array(h, dtype=float)
        h = h * target_sum / np.sum(h)
        return (h, np.max(np.correlate(h, 1 - h, mode='full')) * dx, n_points)

    best_h = result.x
    best_c5 = objective(best_h)

    # Local gradient descent
    def compute_gradient(x):
        h = x * (target_sum / np.sum(x))
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1.0 - (h[j_plus] if 0 <= j_plus < n_points else 0.0)
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    res = minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        jac=compute_gradient,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    # Perturb and refine
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        res = minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            jac=compute_gradient,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = objective(final_h)

    return (final_h, final_c5, n_points)
```
