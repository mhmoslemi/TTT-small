                b[i] = 0.5
        patterns.append(b)
        # Random binary

        r = np.where(r < 0.5, 0.0, 0.5)
        r = r * target_sum / np.sum(r)
        patterns.append(r)
        # Sparse
        s = np.zeros(n_points)
