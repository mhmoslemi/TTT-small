import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def structured_pattern(m, variation=0.1):
        intervals = np.zeros(n_points)
        spacing = 2.0 / (m + 1)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                length = idx_end - idx_start
                # Vary interval width
                length += np.random.uniform(-variation, variation)
                length = np.clip(length, 0.1, 2.0 - 0.1)
                intervals[idx_start:idx_start + int(length / dx + 0.5)] = 1
        intervals = intervals * target_sum / np.sum(intervals)
        return intervals

    patterns = [structured_pattern(m) for m in [8, 12, 16, 20, 24, 28]]
    for pattern in patterns:
        h_candidate = pattern.copy()
        res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-7, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        h_candidate = best_h + perturb
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-7, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
