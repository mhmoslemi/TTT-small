def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    # Generate structured initializations
    def block_pattern():
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 2/3:
                h[i] = 0.5
            elif x < 4/3:
                h[i] = 0.0
            else:
                h[i] = 0.5
        return h

    def symmetric_pattern():
        h = np.zeros(n_points)
        for i in range(n_points):
            mirror = 2 * (n_points // 2) - i
            h[i] = initial_h_values[mirror] if mirror < n_points else initial_h_values[n_points - 1 - (i - n_points // 2)]
        return h

    def random_pattern():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        return h

    def freq_pattern():

        h = np.clip(h, 0, 1)
        return h

        start_time = time.time()
