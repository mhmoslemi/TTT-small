import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def gen_pattern(pattern_type, x=None):
        if pattern_type == 'sin':
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        elif pattern_type == 'tri':
            x = np.linspace(0, 2, n_points)
            h = np.clip(np.sin(np.pi * x) / 2 + 0.5, 0, 1)
        elif pattern_type == 'reg':
            h = np.zeros(n_points)
            h[::2] = 1.0
        elif pattern_type == 'ran':
            h = np.random.rand(n_points)
        else:
            h = initial_h_values
        h = h * target_sum / np.sum(h)
        return h

    candidates = [gen_pattern('initial'), gen_pattern('sin'), gen_pattern('tri'), gen_pattern('reg'), gen_pattern('ran')]

    best_h = candidates[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = (1 - h[j_plus] if 0 <= j_plus < n_points else 0)
            j_minus = i - k_max
            term2 = (h[j_minus] if 0 <= j_minus < n_points else 0)
            grad[i] = dx * (term1 - term2)
        return grad

    start = time.time()

    for h in candidates:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(200):
        if time.time() - start > budget_s:
            break
        noise = np.random.normal(0, 0.03, n_points)
        pert = np.clip(best_h + noise, 0, 1)
        pert = pert * target_sum / np.sum(pert)
        res = minimize(obj, pert, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
