import time

def run(seed=42, budget_s=600):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    n_particles = 20
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [h.copy() for _ in range(n_particles)]
    pbest_c5 = [best_c5 for _ in range(n_particles)]
    gbest = h.copy()
    gbest_c5 = best_c5

    omega = 0.729
    c1 = c2 = 1.494
    max_vel = 0.1
    max_iterations = 200
    early_stop = 10
    iterations_without_improvement = 0

    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = current_c5
            if current_c5 < gbest_c5:
                gbest = h.copy()
                gbest_c5 = current_c5
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (pbest[i] - h)
            social = c2 * r2 * (gbest - h)
            v = omega * v + cognitive + social
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            h = h + v
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            particles[i] = h.copy()
        if np.abs(best_c5 - gbest_c5) < 1e-5:
