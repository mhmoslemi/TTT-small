import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_regular_pattern(spread):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % spread == 0:
                pattern[i] = 1.0
        pattern = pattern * target_sum / np.sum(pattern)
        return pattern

    def generate_triangular_pattern():
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                pattern[i] = 1.0
        pattern = pattern * target_sum / np.sum(pattern)
        return pattern

    def generate_alternating_pattern():
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0 or i % 3 == 2:
                pattern[i] = 1.0
        pattern = pattern * target_sum / np.sum(pattern)
        return pattern

    initial_patterns = [initial_h_values.copy()]

    for spread in [2, 3, 4, 5]:
        initial_patterns.append(generate_regular_pattern(spread))

    initial_patterns.append(generate_triangular_pattern())
    initial_patterns.append(generate_alternating_pattern())

    best_h = None
    best_c5 = 0.381

    start_time = time.time()

    for pattern in initial_patterns:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, pattern, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:

            best_h = res.x.copy()

    for _ in range(5):

        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)

        if res.success and res.fun < best_c5:
