import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [initial_h.copy()]
    for _ in range(19):
        if np.random.rand() < 0.5:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)

            h = h * target_sum / np.sum(h)
            h = np.concatenate([h[:n_points//2], h[n_points//2:][::-1]])
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        parents = []
        for _ in range(2):
            indices = np.random.choice(len(population), 5, replace=False)
            selected = [population[i] for i in indices]
            winner = min(selected, key=compute_c5)
            parents.append(winner)
        split = np.random.randint(1, n_points - 1)
