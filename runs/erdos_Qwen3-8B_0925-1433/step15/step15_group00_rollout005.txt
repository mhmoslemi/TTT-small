import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            idx_plus = k_max + i
            idx_minus = k_max - i
            val_plus = corr[idx_plus] if idx_plus < len(corr) else 0
            val_minus = corr[idx_minus] if idx_minus >= 0 else 0
            grad[i] = (val_plus - val_minus) * dx
        return grad

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    perturb = np.array(initial_h_values, dtype=float) + np.random.uniform(-0.1, 0.1, size=n_points)
    perturb = np.clip(perturb, 0, 1) * target_sum / np.sum(perturb)
    initial_guesses.append(perturb)

    fft_initial = np.fft.fft(initial_h_values)
    phase = np.exp(2j * np.random.uniform(0, 2 * np.pi, size=n_points))
    h_shifted = np.fft.ifft(fft_initial * phase).real
    h_shifted = np.clip(h_shifted, 0, 1) * target_sum / np.sum(h_shifted)
    initial_guesses.append(h_shifted)

    best_h = initial_guesses[0]
    best_c5 = objective(best_h)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
