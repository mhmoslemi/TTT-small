import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1.0 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    initial_candidates = [
        np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values),

        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    ]

    particles = [np.copy(h) for h in initial_candidates]
    velocities = [np.zeros(n_points) for _ in range(len(particles))]
