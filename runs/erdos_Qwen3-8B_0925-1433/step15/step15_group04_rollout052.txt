    h2[:n_points//2] = 0.5
    h2 = h2 * target_sum / np.sum(h2)
    patterns.append(h2)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    start_time = time.time()
    for pattern in patterns:
        h = np.array(pattern, dtype=float)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()
        if time.time() - start_time > budget_s - 10:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
