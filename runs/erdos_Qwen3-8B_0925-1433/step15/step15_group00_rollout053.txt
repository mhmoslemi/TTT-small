from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    def generate_transformed_patterns(base):
        patterns = []
        h_fft = np.fft.fft(base)

        h_new = np.fft.ifft(h_fft * np.exp(1j * phase)).real
        h_new = np.clip(h_new, 0, 1) * target_sum / np.sum(h_new)
        patterns.append(h_new)
        noise = np.random.normal(0, 0.05, n_points)
        h_noise = np.clip(base + noise, 0, 1) * target_sum / np.sum(np.clip(base + noise, 0, 1))
        patterns.append(h_noise)
        perm = np.random.permutation(base)
        h_perm = perm * target_sum / np.sum(perm)
        patterns.append(h_perm)
        return patterns
