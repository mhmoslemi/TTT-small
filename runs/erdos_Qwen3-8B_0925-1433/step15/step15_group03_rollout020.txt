    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        patterns = []
        h_init = np.array(initial_h_values, dtype=float)
        h_init = h_init * target_sum / np.sum(h_init)
        patterns.append(h_init)
        x = np.linspace(0, 2, n_points)
        sin_p = 0.5 + 0.5 * np.sin(np.pi * x / 2)
        sin_p = sin_p * target_sum / np.sum(sin_p)
        patterns.append(sin_p)
        tri_p = np.zeros(n_points)
        tri_p[:n_points//2] = np.linspace(0, 1, n_points//2)
        tri_p[n_points//2:] = np.linspace(1, 0, n_points//2)
        tri_p = tri_p * target_sum / np.sum(tri_p)
        patterns.append(tri_p)
        rand_p = np.random.rand(n_points)
        rand_p = rand_p * target_sum / np.sum(rand_p)
        patterns.append(rand_p)
        return patterns

    population = generate_patterns()
    def compute_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1-h, mode='full')
        k = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0
            j_minus = i - k
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0
            grad[i] = dx * (term1 - term2)
        return grad

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start = time.time()

    for _ in range(100):
        if time.time() - start > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        order = np.argsort(fitness)
        selected = [population[i] for i in order[:int(len(population)*0.5)]]
        new_pop = []
        for i in range(0, len(selected), 2):
            if i+1 < len(selected):
                p1, p2 = selected[i], selected[i+1]
                child = (p1 + p2) / 2 + np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_pop.append(child)
        for h in selected:
            grad = compute_gradient(h)
            h_mut = h - 0.05 * grad
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * target_sum / np.sum(h_mut)
            new_pop.append(h_mut)
        population = new_pop
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    if best_h is None:
        best_h = population[0]
        best_c5 = compute_c5(best_h)

    def obj(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)
    for _ in range(5):
        if time.time() - start > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
