import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h):
        h += np.random.normal(0, 0.05, h.shape)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    base = np.array(initial_h_values, dtype=float)
    pop = [base + np.random.normal(0, 0.05, n_points) for _ in range(20)]
    for i in range(len(pop)):
        pop[i] = np.clip(pop[i], 0, 1)
        pop[i] = pop[i] * target_sum / np.sum(pop[i])

    best_h = base.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        fits = [compute_c5(h) for h in pop]
        current_best = pop[np.argmin(fits)]
        if compute_c5(current_best) < best_c5:
            best_h = current_best.copy()
            best_c5 = compute_c5(best_h)

        new_pop = []
        for _ in range(10):
            idxs = np.random.randint(0, len(pop), 5)
            p1 = pop[np.argmin([compute_c5(pop[i]) for i in idxs])]
            idxs2 = np.random.randint(0, len(pop), 5)
            p2 = pop[np.argmin([compute_c5(pop[i]) for i in idxs2])]
            pt = np.random.randint(1, n_points - 1)
            c1 = np.concatenate([p1[:pt], p2[pt:]])
            c2 = np.concatenate([p2[:pt], p1[pt:]])
            new_pop.append(mutate(c1))
            new_pop.append(mutate(c2))
        pop = new_pop

    return np.clip(best_h, 0, 1), best_c5, n_points
