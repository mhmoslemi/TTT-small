import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        h_initial = np.array(initial_h_values, dtype=float).copy()
        h_initial = h_initial * target_sum / np.sum(h_initial)
        patterns = [h_initial]
        
        for _ in range(4):
            perturb = np.random.normal(0, 0.1, n_points)
            h_perturb = h_initial + perturb
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            patterns.append(h_perturb)
        
        h_uniform = np.zeros(n_points)
        m = int(target_sum)
        for i in range(m):
            h_uniform[i] = 1.0
        h_uniform = h_uniform * target_sum / np.sum(h_uniform)
        patterns.append(h_uniform)
        
        h_two = np.zeros(n_points)
        h_two[:n_points//2] = 1.0
        h_two = h_two * target_sum / np.sum(h_two)
        patterns.append(h_two)
        
        h_tri = np.linspace(0, 1, n_points)
        h_tri = h_tri * target_sum / np.sum(h_tri)
        patterns.append(h_tri)
        
        return patterns

    patterns = generate_patterns()
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h in patterns:
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, 
                       h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, 
                       tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
