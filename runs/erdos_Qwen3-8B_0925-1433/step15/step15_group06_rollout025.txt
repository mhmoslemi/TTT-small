import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = [np.array(initial_h_values, dtype=float)]
    random_init = np.random.rand(n_points)
    random_init = normalize(random_init)
    initial_guesses.append(random_init)

    for _ in range(2):
        idx = np.random.choice(n_points)
        perturb = np.random.randn(n_points)
        perturb[idx] = 0
        perturb = normalize(initial_h_values + 0.1 * perturb)
        initial_guesses.append(perturb)

    best_h = None
    best_c5 = 1.0

    for guess in initial_guesses:
        h = normalize(guess.copy())
        h = h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_p = h.copy()
                h_p[i] += 1e-4
                h_p = normalize(h_p)
                corr_p = np.correlate(h_p, 1 - h_p, mode='full')[k_max]
                h_m = h.copy()
                h_m[i] -= 1e-4
                h_m = normalize(h_m)
                corr_m = np.correlate(h_m, 1 - h_m, mode='full')[k_max]
                grad[i] = (corr_p - corr_m) / (2e-4)
            h = h - 0.5 * grad
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = obj(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
