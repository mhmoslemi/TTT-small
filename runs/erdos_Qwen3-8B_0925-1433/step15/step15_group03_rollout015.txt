import time

    pbest_fitness = np.full(len(particles), np.inf)

    def evaluate(h):

        return np.max(corr) * dx

    def compute_gradient(h):

        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()
