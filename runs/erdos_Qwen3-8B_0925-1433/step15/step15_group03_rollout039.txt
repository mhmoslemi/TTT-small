                grad[j] = dx * (term1 - term2)
            return grad
        
        res = minimize(obj, h_candidate, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < gbest_obj:
            gbest = res.x
            gbest_obj = res.fun
    
    h_final = gbest
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    best_c5 = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    return (h_final, best_c5, n_points)
