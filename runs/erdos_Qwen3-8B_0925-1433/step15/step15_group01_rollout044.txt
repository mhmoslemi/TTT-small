import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population = []
    if initial_h_values is not None:
        base = np.array(initial_h_values, dtype=float)
        for _ in range(5):
            pert = np.random.uniform(-0.05, 0.05, n_points)
            candidate = normalize(base + pert)
            population.append(candidate)
    else:
        for _ in range(20):
            h = np.random.uniform(0, 1, n_points)
            population.append(normalize(h))
    
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([compute_c5(ind) for ind in population])
        # Select parents via tournament
        tournament_size = 5
        parents = []
        for _ in range(20):
            idx = np.random.choice(n_points, tournament_size)
            winner = population[np.argmin(fitness[idx])]
            parents.append(winner)
        # Crossover and mutation
        new_population = []
        for i in range(0, 20, 2):
            if np.random.rand() < 0.8:
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
            else:
                child1, child2 = parents[i], parents[i+1]
            # Mutate
            for child in [child1, child2]:
                if np.random.rand() < 0.1:
                    pert = np.random.uniform(-0.1, 0.1, n_points)
                    child += pert
                child = np.clip(child, 0, 1)
                child = normalize(child)
            new_population.extend([child1, child2])
        population = new_population
        # Update best

            c5 = compute_c5(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
