import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_init = np.array(initial_h_values, dtype=float)
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def eval_func(h):
        return compute_c5(h)
    
    def grad_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad
    
    # Generate initial population
    pop_size = 50
    pop = [h_init]
    x = np.linspace(0, 2, n_points)
    pop.append(0.5 + 0.5 * np.sin(np.pi * x / 2))
    pop.append(np.random.rand(n_points))
    x_tri = np.clip(x, 0, 1)
    pop.append(x_tri)
    for _ in range(pop_size - len(pop)):
        perturb = np.random.normal(0, 0.1, n_points)
        h = h_init + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    
    best_h = h_init
    best_c5 = compute_c5(h_init)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in pop]
        idxs = np.argsort(fitness)
        top = idxs[:int(len(pop) * 0.2)]
        selected = [pop[i] for i in top]
        
        # Crossover
        new_pop = []
        for a, b in zip(selected[::2], selected[1::2]):
            child = (a + b) / 2
            new_pop.append(child)
        
        # Mutation
        for h in new_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            grad = grad_func(h)
            noise = np.random.normal(0, 0.05, n_points)
            h_perturbed = h + noise * grad / np.abs(grad).max()
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            new_pop.append(h_perturbed)
        
        # Replace population
        pop = new_pop + [pop[i] for i in top]
        for h in pop:
            c = compute_c5(h)
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c
    
    # Local refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(eval_func, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            if res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
