from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def create_initial_patterns():
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        yield h

        # Sinusoidal pattern
        sin_h = 0.5 + 0.5 * np.sin(np.linspace(0, np.pi, n_points))
        sin_h = sin_h * target_sum / np.sum(sin_h)
        yield sin_h

        # Random pattern
        rand_h = np.random.rand(n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        yield rand_h

        # Triangular pattern
        tri_h = np.linspace(0, 1, n_points)
        tri_h = tri_h * target_sum / np.sum(tri_h)
        yield tri_h

    population = list(create_initial_patterns())
    best_h = population[0]
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        new_population = []
        for h in population:
            # Crossover

                partner = np.random.choice(population)
                child = (h + partner) / 2.0
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                perturbation = np.random.normal(0, 0.05, n_points)
                mutated = h + perturbation
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * target_sum / np.sum(mutated)
                new_population.append(mutated)
        population = new_population
        for h in population:

                # Local refinement
