import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_c5_obj(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)
    
    def optimize_candidate(h0):
        res = minimize(compute_c5_obj, h0, method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            return res.x
        return h0
    
    # Initial population
    initial_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    # Generate diverse perturbations
    for _ in range(5):
        perturbed = initial_h + np.random.normal(0, 0.15, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        refined = optimize_candidate(perturbed)
        current_c5 = compute_c5(refined)
        if current_c5 < best_c5:
            best_h, best_c5 = refined, current_c5
    
    # Add binary patterns
    for _ in range(3):
        binary = np.random.choice([0, 1], n_points, p=[0.5, 0.5])
        binary = binary * target_sum / np.sum(binary)
        refined = optimize_candidate(binary)
        current_c5 = compute_c5(refined)
        if current_c5 < best_c5:
            best_h, best_c5 = refined, current_c5
    
    # Final refinement
    best_h = optimize_candidate(best_h)
    best_c5 = compute_c5(best_h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
