import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Default number of points if initial_h_values is not provided
    n_points = len(initial_h_values) if 'initial_h_values' in dir() else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Objective function
    def obj(x):
        h = x * (target_sum / np.sum(x))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial candidates
    candidates = []
    if 'initial_h_values' in dir():
        base_h = np.array(initial_h_values, dtype=float)
        for _ in range(10):
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            h_perturb = base_h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            h_scaled = h_perturb * (target_sum / np.sum(h_perturb))
            candidates.append(h_scaled)
    for _ in range(5):
        x = np.linspace(0, 2, n_points)
        h_func = 0.5 * (1 + np.sin(np.pi * x))
        h_func = np.clip(h_func, 0, 1)
        h_scaled = h_func * (target_sum / np.sum(h_func))
        candidates.append(h_scaled)
    for _ in range(5):
        binary = np.random.choice([0, 1], size=n_points)
        current_sum = np.sum(binary)
        if current_sum < target_sum:
            diff = target_sum - current_sum
            indices = np.where(binary == 0)[0]
            binary[indices[:diff]] = 1
        else:
            diff = current_sum - target_sum
            indices = np.where(binary == 1)[0]
            binary[indices[:diff]] = 0
        h_binary = binary * (target_sum / np.sum(binary))
        candidates.append(h_binary)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for candidate in candidates:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, candidate, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (np.clip(best_h, 0, 1), best_c5, n_points)
