        h = np.clip(h, 0, 1)
        curr = np.sum(h)
        if curr == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / curr

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = project(h_initial * target_sum / np.sum(h_initial))
    population = [h_initial.copy()]
    for _ in range(49):
        perturbed = h_initial + np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed = project(perturbed)
        population.append(perturbed)

    population = np.array(population)
    best_h = h_initial
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(h) for h in population])
        elite_idx = np.argsort(fitness)[:int(len(population)*0.2)]
        elites = population[elite_idx]

        mutated = []
        for h in population:
            if np.random.rand() < 0.5:
                noise = np.random.uniform(-0.05, 0.05, size=n_points)
                mutated_h = h + noise
                mutated_h = project(mutated_h)
                mutated.append(mutated_h)
        replace_idx = np.random.choice(len(population), len(population)//4, replace=False)
        population[replace_idx] = mutated[:len(replace_idx)]

        crossed = []
        for i in range(0, len(population)-1, 2):
            if np.random.rand() < 0.5:
                a, b = population[i], population[i+1]
                c1 = (a + b) / 2
                c2 = (a + b) / 2
                crossed.append(project(c1))
                crossed.append(project(c2))
        replace_idx = np.random.choice(len(population), len(population)//4, replace=False)
        population[replace_idx] = crossed[:len(replace_idx)]

        for i in range(len(population)):
            h = population[i].copy()
            for _ in range(5):
                g = grad(h)
                h -= 0.01 * g
                h = project(h)
            population[i] = h

        best_c5 = np.min(fitness)
        best_h = population[np.argmin(fitness)]

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
