import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    initial_patterns = []

    # Symmetric pattern
    h_sym = np.zeros(n_points)
    for i in range(n_points):
        h_sym[i] = initial_h_values[i]
        h_sym[n_points - 1 - i] = initial_h_values[i]
    initial_patterns.append(h_sym * target_sum / np.sum(h_sym))

    # Block pattern: 1/3, 0, 1/3
    h_block = np.zeros(n_points)
    h_block[:n_points//3] = 0.5
    h_block[n_points//3:2*n_points//3] = 0.0
    h_block[2*n_points//3:] = 0.5
    initial_patterns.append(h_block * target_sum / np.sum(h_block))

    # Random pattern
    h_rand = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    initial_patterns.append(h_rand)

    best_h = np.zeros(n_points)
    best_c5 = float('inf')

    for h in initial_patterns:
        h = np.clip(h, 0, 1)
        def obj(x):
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
