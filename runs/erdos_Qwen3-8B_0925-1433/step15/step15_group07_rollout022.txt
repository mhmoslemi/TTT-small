import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_guess():
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            h = np.zeros(n_points)
            m = n_points // 2
            i = np.random.choice(n_points, size=m, replace=False)
            h[i] = 1.0
            h = h * target_sum / np.sum(h)
        return h

    initial_guesses = [generate_initial_guess()]
    i = np.random.choice(n_points, size=n_points // 2, replace=False)
    h = np.zeros(n_points)
    h[i] = 1.0
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    best_h = np.zeros(n_points)
    best_c5 = float('inf')

    for _ in range(2):
        if time.time() > budget_s - 10:
            break
        initial = initial_guesses[_ % len(initial_guesses)]
        h = initial.copy()
        h = h * target_sum / np.sum(h)
        for _ in range(50):
            if time.time() > budget_s - 10:
                break
            def obj(x):
                h = x * target_sum / np.sum(x)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            def grad(x):
                h = x * target_sum / np.sum(x)
                corr = np.correlate(h, 1 - h, mode='full')
                k_max = np.argmax(corr)
                grad = np.zeros_like(x)
                dx_val = 2.0 / len(h)
                for i in range(len(h)):
                    j = (i + k_max) % len(h)
                    term1 = (1 - h[j]) * dx_val
                    i2 = (i - k_max) % len(h)
                    term2 = h[i2] * dx_val
                    grad[i] = term1 - term2
                return grad
            res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
