from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381

            h[start_idx:end_idx] += 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    initial_patterns = [best_h]
    for m in [10, 20, 30, 40]:
        for _ in range(2):
            h = generate_m_pattern(m)
            if h is not None:
                initial_patterns.append(h)

    def obj(x):

        corr = np.correlate(h, 1 - h, mode='full')

    def grad(x):

        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
