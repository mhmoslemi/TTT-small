import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def init_patterns():
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        yield h
        h_sym = h[::-1]
        yield h_sym
        h_bin = np.zeros(n_points)
        idx = np.random.choice(n_points, size=n_points//2, replace=False)
        h_bin[idx] = 0.5
        h_bin = h_bin * target_sum / np.sum(h_bin)
        yield h_bin
        h_blk = np.zeros(n_points)
        n_third = n_points // 3
        h_blk[:n_third] = 0.5
        h_blk[n_third+1:] = 0.5
        h_blk = h_blk * target_sum / np.sum(h_blk)
        yield h_blk

    best_c5 = float('inf')
    best_h = None

    for init in init_patterns():
        h = np.clip(init, 0, 1)
        h = h * target_sum / np.sum(h)
        h = h.clip(0, 1)
        h = h * target_sum / np.sum(h)

        def obj(x):
            x = x * target_sum / np.sum(x)
            x = x.clip(0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def cons(x):
            return np.sum(x) - target_sum

        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': cons}, tol=1e-5)
        if res.success:
            h = res.x
            h = h * target_sum / np.sum(h)
            h = h.clip(0, 1)
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    return (best_h, best_c5, n_points)
