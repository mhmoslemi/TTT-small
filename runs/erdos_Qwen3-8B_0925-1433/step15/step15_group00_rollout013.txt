        h = generate_log_stretched(initial_h_values)
        h = h * target_sum / np.sum(h)
        initial_candidates.append(h)
        noise = np.random.normal(0, 0.05, n_points)
        h = initial_h_values + noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        initial_candidates.append(h)

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    for candidate in initial_candidates:
        h = candidate * target_sum / np.sum(candidate)
        h_opt = h.copy()

        def objective(x):
            h_curr = x * target_sum / np.sum(x)
            corr = np.correlate(h_curr, 1 - h_curr, mode='full')
            return np.max(corr) * dx

        def gradient(x):
            h_curr = x * target_sum / np.sum(x)
            corr = np.correlate(h_curr, 1 - h_curr, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(x)
            i_plus = np.arange(n_points) + k_max
            term1_mask = i_plus < n_points
            term1 = np.zeros(n_points)
            term1[term1_mask] = 1.0 - h_curr[i_plus[term1_mask]]
            i_minus = np.arange(n_points) - k_max
            term2_mask = i_minus >= 0
            term2 = np.zeros(n_points)
            term2[term2_mask] = h_curr[i_minus[term2_mask]]
            grad = term1 - term2
            return grad

        start_time = time.time()
        h_opt = h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            g = gradient(h_opt)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g ** 2)
            m_hat = m / (1 - 0.9 ** (_ + 1))
            v_hat = v / (1 - 0.999 ** (_ + 1))
            h_opt = h_opt - 0.1 * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
            current_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            current_c5 = np.max(current_corr) * dx
            if current_c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = current_c5

        res = minimize(objective, h_opt, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
