import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(x):
        if np.all(x == 0):
            return float('inf')
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = obj(best_h)

    # Generate initial candidates
    initial_candidates = []
    # Random perturbations
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        initial_candidates.append(candidate)
    # Periodic patterns
    for m in [5, 10, 15]:
        pattern = np.zeros(n_points)
        spacing = 2.0 / (m + 1)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                pattern[idx_start:idx_end] = 1
        pattern = pattern * target_sum / np.sum(pattern)
        initial_candidates.append(pattern)
    # Jittered binary vectors
    for _ in range(5):
        binary = np.random.choice([0, 1], size=n_points)
        binary_sum = np.sum(binary)
        binary = binary * target_sum / binary_sum
        perturb = np.random.normal(0, 0.05, n_points)
        candidate = binary + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        initial_candidates.append(candidate)
