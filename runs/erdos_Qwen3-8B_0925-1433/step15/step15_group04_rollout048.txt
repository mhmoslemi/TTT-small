import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_c5(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    iterations = 0
    for _ in range(20):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        x0 = initial_h_values + perturb
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * target_sum / np.sum(x0)
        res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            current_h = res.x * target_sum / np.sum(res.x)
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        iterations += 1

    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        x0 = np.random.uniform(0, 1, n_points)
        x0 = x0 * target_sum / np.sum(x0)
        res = minimize(obj, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
