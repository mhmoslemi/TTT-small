        if time.time() - start_time > budget_s:
            break
        h_candidate = pattern.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h_candidate, method='SLSQP', bounds=[(0,1)]*n_points, constraints=constraints, tol=1e-5)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=constraints, tol=1e-5)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
