def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    max_iterations = 150
    learning_rate = 0.05
    ada_grad = np.zeros_like(h)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_worst = np.argmax(corr)
        grad = np.zeros_like(h)
        if k_worst >= 0:
            for i in range(n_points - k_worst):
                j = i + k_worst
                grad[i] += (1 - h[j])
                grad[j] -= h[i]

                j = i - m
                grad[i] += (1 - h[j])
