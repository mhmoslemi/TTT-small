import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h * target_sum / np.sum(initial_h)
    h = np.clip(h, 0, 1)

    def project(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.full_like(h, 0.5)
        h = h * target_sum / current_sum
        return h

    def obj(x):
        h = project(x, target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = project(x, target_sum)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        current_c5s = []
        for h_candidate in population:
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            current_c5s.append(np.max(corr) * dx)
        best_idx = np.argmin(current_c5s)
        best_c5 = current_c5s[best_idx]
        best_h = population[best_idx].copy()
        elite_size = int(0.2 * population_size)
        elite_indices = np.argsort(current_c5s)[:elite_size]
        elites = [population[i] for i in elite_indices]
        new_population = elites[:]
        np.random.shuffle(population)
        for _ in range(int(0.5 * population_size)):
            h = population[0].copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = project(h, target_sum)
            new_population.append(h)
        for _ in range(int(0.5 * population_size)):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            h = (population[idx1] + population[idx2]) / 2
            h = project(h, target_sum)
            new_population.append(h)
        while len(new_population) > population_size:
            new_population.pop(np.random.randint(len(new_population)))
        population = new_population
        for h_candidate in population:
            res = minimize(obj, h_candidate, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, tol=1e-5)
            if res.success:
                h_candidate[:] = res.x

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
