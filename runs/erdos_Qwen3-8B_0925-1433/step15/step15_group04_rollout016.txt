from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return compute_overlap(x)

    def gradient(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += 1e-4
            x_plus = x_plus * target_sum / np.sum(x_plus)

            x_minus[i] -= 1e-4

    def perturb_config(base, scale):
