import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)
    
    h = np.array(initial_h_values, dtype=float)
    h = project(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def grad(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad
    
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:

        h_pert = best_h + delta
        h_pert = project(h_pert)
