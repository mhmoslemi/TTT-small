import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k_max
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    def _init_population():
        def sin_pattern():
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
            return h * target_sum / np.sum(h)
        def rand_pattern():
            h = np.random.rand(n_points)
            return h * target_sum / np.sum(h)
        return [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values),
                sin_pattern(), rand_pattern()]

    candidates = _init_population()
    n_pop = len(candidates)

    def _update_velocity(p, pb, gb, v, k_max):
        grad = compute_gradient(p, k_max)
        v = 0.7 * v + 1.5 * np.random.rand(n_points) * (pb - p) + \
            1.5 * np.random.rand(n_points) * (gb - p) + 0.2 * grad
        return v

    pbest = [np.copy(c) for c in candidates]
    pbest_fit = [evaluate_overlap(c) for c in candidates]
    gbest = np.copy(candidates[np.argmin(pbest_fit)])
    gbest_fit = min(pbest_fit)
    velocities = [np.zeros(n_points) for _ in candidates]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(n_pop):
            if np.sum(candidates[i]) < target_sum - 1e-6 or np.sum(candidates[i]) > target_sum + 1e-6:
                candidates[i] = np.clip(candidates[i], 0, 1)
                candidates[i] = candidates[i] * target_sum / np.sum(candidates[i])
            current_fit = evaluate_overlap(candidates[i])
            k_max = np.argmax(np.correlate(candidates[i], 1 - candidates[i], mode='full'))
            velocities[i] = _update_velocity(candidates[i], pbest[i], gbest, velocities[i], k_max)
            candidates[i] += velocities[i]
            candidates[i] = np.clip(candidates[i], 0, 1)
            candidates[i] = candidates[i] * target_sum / np.sum(candidates[i])
            if current_fit < pbest_fit[i]:

        if min(pbest_fit) < gbest_fit:
            gbest = np.copy(pbest[np.argmin(pbest_fit)])
            gbest_fit = min(pbest_fit)

    def _refine(h):

            x = x * target_sum / np.sum(x)
            return evaluate_overlap(x)
        def jac(x):
            x = x * target_sum / np.sum(x)

            k_max = np.argmax(corr)
            return compute_gradient(x, k_max)
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_cand = best_cand * target_sum / np.sum(best_cand)
    return (best_cand, best_fit, n_points)
