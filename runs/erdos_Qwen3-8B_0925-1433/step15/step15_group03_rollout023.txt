            grad = np.zeros_like(h)
            for idx in range(n_points):
                j_plus = idx + k_max
                term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
                j_minus = idx - k_max
                term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
                grad[idx] = dx * (term1 - term2)
            r1, r2 = np.random.rand(), np.random.rand()
            vel_new = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - h) + 1.5 * r2 * (gbest - h) + 0.2 * grad
            velocities[i] = vel_new
            h_new = h + vel_new
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            particles[i] = h_new

    top_indices = np.argsort(fitness)[:5]
    for idx in top_indices:
        h = particles[idx]
        def obj(x):
            h_new = x * target_sum / np.sum(x)
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            return np.max(corr) * dx
        def grad(x):
            h_new = x * target_sum / np.sum(x)
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(x)
            for i in range(n_points):
                j_plus = i + k_max
                term1 = 1 - h_new[j_plus] if 0 <= j_plus < n_points else 0.0
                j_minus = i - k_max
                term2 = h_new[j_minus] if 0 <= j_minus < n_points else 0.0
                grad[i] = dx * (term1 - term2)
            return grad
        res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n_points, constraints={'type':'eq','fun':lambda x: np.sum(x)-target_sum})
        if res.success and -res.fun < -best_fitness:
            best_fitness = res.fun
            gbest = res.x

    best_h = gbest.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, -best_fitness, n_points)
