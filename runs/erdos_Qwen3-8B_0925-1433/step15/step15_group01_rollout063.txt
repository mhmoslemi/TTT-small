import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = np.max(corr_minus) * dx
            grad[i] = (max_plus - max_minus) / (2e-4)
        return grad

    base = np.array(initial_h_values, dtype=float)
    candidates = [base + np.random.normal(0, 0.05, n_points) for _ in range(15)]
    for h in candidates:
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    best_h = candidates[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for h in candidates:
            grad = compute_gradient(h)
