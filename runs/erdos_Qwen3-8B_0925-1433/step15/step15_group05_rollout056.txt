import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        return h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < rate:
                h[i] += np.random.normal(0, 0.05)
        return normalize(h)

    def crossover(a, b):
        p = np.random.randint(1, n_points - 1)
        return np.concatenate([a[:p], b[p:]])

    initial = np.array(initial_h_values, dtype=float)
    pop = [normalize(initial)] * 50

    for _ in range(10):
        if time.time() > time.time() + budget_s:
            break
        sel = [np.random.choice(pop) for _ in range(5)]
        parent1 = min(sel, key=evaluate)
        parent2 = min(sel, key=evaluate)
        child = crossover(parent1, parent2)
        mut = mutate(child)
        if evaluate(mut) < evaluate(pop[0]):
            pop[0] = mut

    best = normalize(pop[0])
    best_c5 = evaluate(best)

    def obj(x):
        return evaluate(x)

    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        res = minimize(obj, best, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

            best_c5 = res.fun
