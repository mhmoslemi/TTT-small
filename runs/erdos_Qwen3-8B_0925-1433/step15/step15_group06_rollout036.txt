import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and diversity
    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(9):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(5):
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        h = h[::-1]
        population.append(h)
    for h in population:
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.pop()
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    population_size = 20
    tournament_size = 5
    max_generations = 100
    start_time = time.time()
    no_improvement = 0

    for _ in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        parents = []
        for _ in range(population_size // 2):
            idxs = np.random.choice(len(population), tournament_size, replace=False)
            parent_idx = idxs[np.argmin(fitness[idxs])]
            parents.append(population[parent_idx].copy())
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parents[i][:split], parents[i+1][split:]])
                child2 = np.concatenate([parents[i+1][:split], parents[i][split:]])
                offspring.append(child1)
                offspring.append(child2)
        for h in offspring:
            mutate_idx = np.random.choice(n_points, int(0.1 * n_points), replace=False)
            h[mutate_idx] += np.random.uniform(-0.15, 0.15, size=len(mutate_idx)) * h[mutate_idx]
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
        combined = population + offspring
        fitness = [compute_c5(h) for h in combined]

        current_best = np.min(fitness)
        if current_best < best_c5:

            best_h = population[sorted_idx[np.argmin(fitness)]].copy()
            no_improvement = 0

            no_improvement += 1

            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
