import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    iters = 0
    max_iters = 5000

    while time.time() - start_time < budget_s and iters < max_iters:
        delta = np.random.normal(0, 0.15, n_points)
        h_pert = best_h + delta
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        if current_sum == 0:
            h_proj = np.full_like(h_pert, 0.5)
        else:
            h_proj = h_pert * target_sum / current_sum
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h_proj, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        iters += 1

    h_values = np.clip(best_h, 0, 1)
    current_sum = np.sum(h_values)
    if current_sum == 0:
        h_values = np.full_like(h_values, 0.5)
    else:
        h_values = h_values * target_sum / current_sum
    return (h_values, best_c5, n_points)
