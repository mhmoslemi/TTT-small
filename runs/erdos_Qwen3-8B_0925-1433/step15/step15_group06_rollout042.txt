import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def generate_initializations():
        h0 = initial_h_values * (target_sum / np.sum(initial_h_values))
        yield h0
        h1 = np.random.uniform(0.2, 0.8, n_points)
        h1 = h1 * (target_sum / np.sum(h1))
        yield h1
        h2 = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points * 0.2) * 2, replace=False)
        h2[indices[:int(n_points * 0.2)]] = 0
        h2[indices[int(n_points * 0.2):]] = 1
        remaining = np.setdiff1d(np.arange(n_points), indices)
        h2[remaining] = 0.5
        h2 = h2 * (target_sum / np.sum(h2))
        yield h2
        h3 = np.concatenate([initial_h_values[::-1], initial_h_values])
        h3 = h3[:n_points] * (target_sum / np.sum(h3[:n_points]))
        yield h3
        h4 = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 1.5)
        h4 = h4 * (target_sum / np.sum(h4))
        yield h4

    best_h = initial_h_values * (target_sum / np.sum(initial_h_values))
    best_c5 = obj(best_h)

    start_time = time.time()
    for h_init in generate_initializations():
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(
                obj,
                h_init,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                options={'ftol': 1e-6, 'maxiter': 100}
            )
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            continue

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = obj(best_h)
    return (best_h, final_c5, n_points)
