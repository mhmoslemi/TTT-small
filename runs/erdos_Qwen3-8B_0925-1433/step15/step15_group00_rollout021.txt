import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    def compute_corr(h):
        return np.correlate(h, 1 - h, mode='full') * dx

    def analytic_gradient(h):
        n = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)
        grad = np.zeros_like(h)
        for i in range(n):
            ti = i + shift
            if 0 <= ti < n:
                term1 = 1.0 - h[ti]
            else:
                term1 = 0.0
            ti = i - shift
            if 0 <= ti < n:
                term2 = h[ti]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def objective(h):
        return np.max(compute_corr(h))

    def gradient(h):
        return -analytic_gradient(h)

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    best_h = normalize(best_h)

    def perturb(h, noise=0.1):
        h = h.copy()
        h += np.random.normal(0, noise, h.shape)
        h = np.clip(h, 0, 1)
        h = normalize(h)
        return h

    transforms = [
        lambda h: h,
        lambda h: perturb(h, noise=0.05),
        lambda h: np.fft.ifft(np.fft.fft(h) * np.exp(2j * np.pi * np.random.rand(len(h)))).real,
        lambda h: np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points) * 1.5, h),
        lambda h: h[::-1]
    ]

    for t in transforms:
        h = t(initial_h)
        h = normalize(h)
        res = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(objective, best_h, method='Adam', jac=gradient, bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
