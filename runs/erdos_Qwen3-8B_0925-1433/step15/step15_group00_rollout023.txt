import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def init_variants():
        variants = [initial_h_values.copy()]
        for _ in range(4):
            h = np.array(initial_h_values, dtype=float)
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            variants.append(h)
        return variants

    variants = init_variants()

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, variant, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x)-target_sum}, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
