import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def structured_pattern(m):
        intervals = np.zeros(n_points)
        spacing = 2.0 / (m + 1)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            if idx_start < n_points:
                intervals[idx_start:idx_end] = 1
        intervals = intervals * target_sum / np.sum(intervals)
        return intervals

    initial_candidates = []

    # Random perturbations
    for _ in range(10):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        candidate = h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        initial_candidates.append(candidate)

    # Periodic patterns
    for m in [5, 10, 15, 20]:
        initial_candidates.append(structured_pattern(m))

    # Jittered binary
    for _ in range(5):
        binary = np.random.choice([0, 1], n_points)
        cnt = np.sum(binary)
        if abs(cnt - target_sum) < 1.5:
            binary = binary * target_sum / cnt
            initial_candidates.append(binary)

    for candidate in initial_candidates:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.choice(n_points, int(n_points * 0.2), replace=False)
        best_h[perturb] += np.random.uniform(-0.1, 0.1, len(perturb))
