import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return compute_c5(x)

    def compute_gradient(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_p = x.copy()
            x_p[i] += 1e-4
            x_p = np.clip(x_p, 0, 1)
            x_p = x_p * target_sum / np.sum(x_p)
            corr_p = np.correlate(x_p, 1 - x_p, mode='full')[k_max]
            x_m = x.copy()
            x_m[i] -= 1e-4
            x_m = np.clip(x_m, 0, 1)
            x_m = x_m * target_sum / np.sum(x_m)
            corr_m = np.correlate(x_m, 1 - x_m, mode='full')[k_max]
            grad[i] = (corr_p - corr_m) / (2e-4)
        return -grad

    def perturb(h):
        h = h.copy()
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        scaled_h = np.random.uniform(0, 1, n_points)
        scaled_h = scaled_h * target_sum / np.sum(scaled_h)
        res = minimize(obj, scaled_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6, jac=compute_gradient)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        h_perturbed = perturb(best_h)
        c5_perturbed = compute_c5(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed

    return (np.clip(best_h, 0, 1), best_c5, n_points)
