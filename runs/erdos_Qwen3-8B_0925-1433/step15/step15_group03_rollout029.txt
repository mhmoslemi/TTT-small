import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def eval_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def grad_func(h):
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0
            grad[i] = dx * (term1 - term2)
        return grad

    def scale_h(h):
        return h * target_sum / np.sum(h)

    def mutate(h):
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0
            grad[i] = dx * (term1 - term2)
        noise = np.random.normal(0, 0.01, n_points)
        h_mut = h + noise
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = scale_h(h_mut)
        return h_mut

    h = np.array(initial_h_values, dtype=float)
    h = scale_h(h)
    best_h = h.copy()
