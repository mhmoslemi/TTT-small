import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        max_k = np.argmax(np.correlate(h, 1 - h, mode='full'))
        for i in range(n_points):
            h_p = h.copy()
            h_p[i] += eps
            h_p = h_p * target_sum / np.sum(h_p)
            corr_p = np.correlate(h_p, 1 - h_p, mode='full')[max_k]
            h_m = h.copy()
            h_m[i] -= eps
            h_m = h_m * target_sum / np.sum(h_m)
            corr_m = np.correlate(h_m, 1 - h_m, mode='full')[max_k]
            grad[i] = (corr_p - corr_m) / (2 * eps)
        return -grad

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(best_h + perturbation, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        curr_c5 = np.max(np.correlate(perturbed, 1 - perturbed, mode='full')) * dx
        if curr_c5 < best_c5:
            best_h = perturbed
            best_c5 = curr_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
