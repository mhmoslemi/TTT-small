import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    best_h = h0.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    candidates = [(best_c5, best_h)]

    def calculate_c5(h):
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c * dx)

    def compute_gradient(h, k):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k
            term1 = 1 - h[j_plus] if 0 <= j_plus < n_points else 0.0
            j_minus = i - k
            term2 = h[j_minus] if 0 <= j_minus < n_points else 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()
    pop_size = 50
    population = []
    for _ in range(pop_size):
        if np.random.random() < 0.5:

        elif np.random.random() < 0.3:

            sin_h = 0.5 + 0.5 * np.sin(np.pi * x / 2)
            sin_h = sin_h * target_sum / np.sum(sin_h)
            h = sin_h.copy()
        elif np.random.random() < 0.15:

            h = h * target_sum / np.sum(h)
        else:

            tri_h = tri_h * target_sum / np.sum(tri_h)
            h = tri_h.copy()
        population.append(h)

    for _ in range(10):

            break
        scores = [calculate_c5(h) for h in population]
        top_idx = np.argsort(scores)[:int(len(population)*0.2)]
        top_h = [population[i] for i in top_idx]
