import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def scale_vector(x):
        return x * target_sum / np.sum(x)

    # Initial setup
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial_scaled = scale_vector(h_initial)
    best_h = h_initial_scaled.copy()
    best_c5 = compute_c5(best_h)

    # Generate diverse initial candidates
    candidates = [h_initial_scaled.copy()]
    for _ in range(10):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        cand = (h_initial + perturb) * target_sum / np.sum(h_initial + perturb)
        cand = np.clip(cand, 0, 1)
        candidates.append(cand)
    for m in [5, 10, 15, 20]:
        intervals = np.zeros(n_points)
        spacing = 2.0 / (m + 1)
        for i in range(m):
            start = i * spacing
            end = (i + 1) * spacing
            idx_start = int(start / dx + 0.5)
            idx_end = int(end / dx + 0.5)
            intervals[idx_start:idx_end] = 1
        cand = intervals * target_sum / np.sum(intervals)
        cand = np.clip(cand, 0, 1)
        candidates.append(cand)
    for _ in range(5):
        binary = np.random.choice([0, 1], n_points)
        bin_sum = np.sum(binary)
        bin_scale = target_sum / bin_sum if bin_sum != 0 else 0
        cand = binary * bin_scale
        cand = np.clip(cand, 0, 1)
        candidates.append(cand)

    # Optimization loop
    start_time = time.time()
    for cand in candidates:
        cand = np.clip(cand, 0, 1)
        cand = cand * target_sum / np.sum(cand)
        def obj(x):

            return compute_c5(h)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, cand, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    # Refinement
    if time.time() - start_time < budget_s:

            cand = best_h + perturb
