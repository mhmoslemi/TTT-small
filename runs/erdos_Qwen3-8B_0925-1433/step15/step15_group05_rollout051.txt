    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        res = minimize(objective, np.random.uniform(0, 1, n_points), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        res = minimize(objective, np.random.uniform(0, 1, n_points), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
