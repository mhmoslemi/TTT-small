from scipy.optimize import minimize

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_with_bounds(h_start):
