        lag = k_worst - (n_points - 1)
        gradient = (1 - h[(np.arange(n_points) + lag) % n_points]) * dx
        grad_sq += gradient**2
        current_lr = learning_rate / (np.sqrt(grad_sq) + 1e-8)
        current_lr = np.clip(current_lr, min_lr, 1.0)
        h = h - current_lr * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iter % 10 == 0 and iter > 0 and np.abs(current_c5 - best_c5) < 1e-4:
            break
        if iter % 50 == 0:
            perturb = np.random.normal(0, 0.02, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        if time.time() - start_time > budget_s:
            break
    return (np.clip(best_h, 0, 1), best_c5, n_points)
