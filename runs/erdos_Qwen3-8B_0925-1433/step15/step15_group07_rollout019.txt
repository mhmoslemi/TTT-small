import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        h = h * target_sum / s
        return h

    initial = np.array(initial_h_values, dtype=float)
    initial = project(initial)
    population = [initial.copy() for _ in range(50)]
    best_h = initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            grad[i] = (1 - h[j]) - h[(i - k_max) % len(h)]
        return grad

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        scores = [obj(p) for p in population]
        order = np.argsort(scores)
        elite = population[order[:5]]
        new_pop = elite.copy()

        for _ in range(45):
            idx = np.random.choice(len(population), 2, replace=False)
            a, b = population[idx[0]], population[idx[1]]
            child = (a + b) / 2
            child = project(child)
            new_pop.append(child)

        for _ in range(25):
            idx = np.random.choice(len(population), 1)
            h = population[idx].copy()
            noise = np.random.normal(0, 0.05, len(h))
            h += noise
            h = project(h)
            new_pop.append(h)

        for h in new_pop:
            res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n_points, tol=1e-6)
            if res.success:
                new_pop.append(res.x)
        population = new_pop[:50]
        for p in population:
            if obj(p) < best_c5:
                best_h, best_c5 = p, obj(p)

    best_h = project(best_h)
    h_values = np.clip(best_h, 0, 1)
    return (h_values, best_c5, n_points)
