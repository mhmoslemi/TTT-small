            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) != 0 else best_h
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
