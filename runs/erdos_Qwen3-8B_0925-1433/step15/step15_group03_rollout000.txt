    h_initial = h_initial * target_sum / np.sum(h_initial)
    initial_guesses.append(h_initial)

    # Sinusoidal pattern
    x = np.linspace(0, 2, n_points)
    h_sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    h_sinusoidal = h_sinusoidal * target_sum / np.sum(h_sinusoidal)
    initial_guesses.append(h_sinusoidal)

    # Random pattern
    h_random = np.random.rand(n_points)
    h_random = h_random * target_sum / np.sum(h_random)
    initial_guesses.append(h_random)

    best_h = initial_guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Function to compute objective
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Gradient function
    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1.0 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    # Optimization parameters
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()

    # Run optimization on each initial guess
    for guess in initial_guesses:
        res = minimize(obj, guess, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints=constraints)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    # Local refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Perturb best_h
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints=constraints)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Ensure best_h is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, best_c5, n_points)
