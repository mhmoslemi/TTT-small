import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        return h * target_sum / np.sum(h)

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = scale_h(h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    def compute_c5(h):
        h = scale_h(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, mutation_rate, target_sum, n_points):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, 0.05)
                h[i] = np.clip(h[i], 0, 1)
        h = scale_h(h)
        return h

    # Genetic algorithm parameters
    pop_size = 50
    gen_num = 10
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_count = 5
    start_time = time.time()

    # Initial population
    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        vec = np.random.rand(n_points)
        vec = scale_h(vec)
        population.append(vec)

    # GA loop
    for gen in range(gen_num):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [compute_c5(ind) for ind in population]
        elites = np.argsort(fitnesses)[:elite_count]
        parents = []
        for _ in range(pop_size // 2):
            indices = np.random.choice(pop_size, 5, replace=False)
            best_idx = indices[np.argmin(fitnesses[indices])]
            parents.append(population[best_idx])
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points)
                child1 = np.concatenate((p1[:point], p2[point:]))
                child2 = np.concatenate((p2[:point], p1[point:]))
            else:
                child1, child2 = p1.copy(), p2.copy()
            child1 = mutate(child1, mutation_rate, target_sum, n_points)
            child2 = mutate(child2, mutation_rate, target_sum, n_points)
            new_pop.append(child1)
            new_pop.append(child2)
        population = new_pop[:pop_size]

    best_idx = np.argmin([compute_c5(ind) for ind in population])
    best_h = population[best_idx].copy()
    best_h = scale_h(best_h)

    if time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        def objective(x):
            return compute_c5(x)
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
