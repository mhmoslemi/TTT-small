from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_vec(v):
        return np.clip(v, 0, 1) * (target_sum / np.sum(v))

    def generate_initial_candidates():
        candidates = []
        base = np.array(initial_h_values, dtype=float)
        perturbations = np.random.normal(0, 0.1, (10, n_points))
        for p in perturbations:
            candidates.append(scale_vec(base + p))
        for m in [4, 5, 6, 7, 8]:
            intervals = np.zeros(n_points)
            spacing = 2.0 / (m + 1)
            for i in range(m):
                start = i * spacing
                end = (i + 1) * spacing
                idx_start = int(start / dx + 0.5)
                idx_end = int(end / dx + 0.5)
                if idx_start < n_points:
                    intervals[idx_start:idx_end] = 1
            candidates.append(scale_vec(intervals))
        for _ in range(5):
            idx = np.random.choice(n_points, size=10, replace=False)
            val = np.random.choice([0, 1], size=n_points)
            val[idx] = 1 - val[idx]
            candidates.append(scale_vec(val))
        return candidates

    start_time = time.time()
    candidates = generate_initial_candidates()
    best_h = scale_vec(np.array(initial_h_values))
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = scale_vec(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = scale_vec(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-3
            h_plus = scale_vec(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-3
            h_minus = scale_vec(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-3)
