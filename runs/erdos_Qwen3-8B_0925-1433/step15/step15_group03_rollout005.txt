import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def init_population():
        base = np.array(initial_h_values, dtype=float)
        base = base * target_sum / np.sum(base)
        sin_h = 0.5 + 0.5 * np.sin(np.linspace(0, np.pi, n_points))
        sin_h = sin_h * target_sum / np.sum(sin_h)
        rand_h = np.random.rand(n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        tri_h = np.linspace(0, 1, n_points)
        tri_h = tri_h * target_sum / np.sum(tri_h)
        return [base, sin_h, rand_h, tri_h]

    pop = init_population()
    best_h = pop[0].copy()
    best_c5 = np.max(np.correlate(pop[0], 1 - pop[0], mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evaluate_fitness(h):
        return compute_c5(h)

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        if gen >= 100:
            break
        gen += 1
        
        # Evaluate and sort population
        fit = [evaluate_fitness(h) for h in pop]
        sorted_idx = np.argsort(fit)
        top_h = [pop[i] for i in sorted_idx[:20]]
        
        # Crossover
        new_pop = []
        for i in range(0, len(top_h), 2):
            if i + 1 < len(top_h):
                a = top_h[i]
                b = top_h[i+1]
                child = (a + np.random.rand(n_points) * b) / 2
                new_pop.append(child)
        
        # Mutation
        for h in top_h:
            k_max = np.argmax(np.correlate(h, 1 - h, mode='full'))
            grad = compute_gradient(h)
            noise = np.random.normal(0, 0.05, n_points)
            perturb = h + noise * grad
            perturb = np.clip(perturb, 0, 1)
            perturb = perturb * target_sum / np.sum(perturb)
            new_pop.append(perturb)
        
        # Replace worst
        new_pop += [pop[-1]]
        new_pop = np.array(new_pop)
        new_pop = new_pop[np.argsort(evaluate_fitness(new_pop))[:len(pop)]]
        pop = new_pop.tolist()
        
        # Local refinement
        for h in pop[:3]:
            res = minimize(lambda x: -compute_c5(x), h, method='L-BFGS-B',
                          jac=lambda x: -compute_gradient(x),
                          bounds=[(0, 1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success:
                candidate = res.x
                c5 = compute_c5(candidate)
                if c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
