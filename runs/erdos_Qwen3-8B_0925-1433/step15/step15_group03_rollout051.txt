        x = np.linspace(0, 2, n_points)
        tri = np.minimum(x, 2 - x)
        return tri * target_sum / np.sum(tri)

        reg[::2] = 1.0
        return reg * target_sum / np.sum(reg)

        return rand * target_sum / np.sum(rand)

    start_time = time.time()

    for pattern in initial_patterns:
