from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def generate_random_pattern(num_intervals):
        h = np.zeros(n_points)
        for _ in range(num_intervals):
            start = np.random.randint(0, n_points - 1)
            h[start:start + np.random.randint(1, n_points - start)] = 1
        scale = target_sum / np.sum(h)
        return h * scale

    initial_patterns = [np.array(initial_h_values) * target_sum / np.sum(initial_h_values)]
    for _ in range(5):
        initial_patterns.append(generate_random_pattern(np.random.randint(5, 15)))
        initial_patterns.append(generate_random_pattern(np.random.randint(10, 20)))

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[max_k]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[max_k]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    for pattern in initial_patterns:
        h_candidate = pattern.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(3):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
