from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / len(h)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    population = [h.copy() for _ in range(50)]
    for p in population:
        p += np.random.normal(0, 0.1, len(p))
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_population = []
        for cand in population:
            x = cand.copy()
            for _ in range(20):

                x = res.x

        population = new_population
