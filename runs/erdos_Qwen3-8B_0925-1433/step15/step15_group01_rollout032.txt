import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381  # Initial bound

    def generate_initial(m):
        l = 1.0 / m
        starts = np.random.uniform(0, 2 - l, m)
        h = np.zeros(n_points)
        for i in range(m):
            start = starts[i]
            end = start + l
            for j in range(n_points):
                x = j * dx
                if start <= x <= end:
                    h[j] = 1.0
        total = np.sum(h)
        if total == 0:
            return None
        h = h * target_sum / total
        return h

    # Generate diverse initial patterns
    for m in [20, 30, 40]:
        h = generate_initial(m)
        if h is not None:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
