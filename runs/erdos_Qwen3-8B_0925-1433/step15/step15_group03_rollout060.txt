def compute_gradient(x, target_sum, dx, n_points, k_max):

    grad = np.zeros_like(x)
    for i in range(n_points):
        j_plus = i + k_max
        if 0 <= j_plus < n_points:
            term1 = 1 - h[j_plus]
        else:
            term1 = 0.0
        j_minus = i - k_max
        if 0 <= j_minus < n_points:
            term2 = h[j_minus]
        else:
            term2 = 0.0

    return grad
