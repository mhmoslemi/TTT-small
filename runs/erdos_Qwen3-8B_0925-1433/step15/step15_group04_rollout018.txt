import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initializations = []

    # Generate various initializations

        return (init + noise) * target_sum / np.sum(init + noise)

        sparse = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.2:
                sparse[i] = 0.5
        return sparse * target_sum / np.sum(sparse)

    def generate_symmetric():
        symmetric = np.zeros(n_points)
        symmetric[::2] = 0.5
        return symmetric * target_sum / np.sum(symmetric)

    def generate_block():
        block = np.zeros(n_points)
        for i in range(n_points):
            block[i] = 0.5 if i % 3 == 0 else 0.0
        return block * target_sum / np.sum(block)

    def generate_binary():
        binary = np.random.binomial(1, 0.5, n_points)
        return binary * target_sum / np.sum(binary)

    initializations.append(np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values))
    initializations.append(generate_perturb(np.array(initial_h_values), 1.1))
    initializations.append(generate_sparse())
    initializations.append(generate_symmetric())
    initializations.append(generate_block())
    initializations.append(generate_binary())
    initializations.append(np.random.permutation(initial_h_values) * target_sum / np.sum(initial_h_values))

    best_c5 = 0.381
    best_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)

    start_time = time.time()
    for init in initializations:
        h = np.clip(init, 0, 1)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)

                h_plus[i] += 1e-4
                h_plus = h_plus * target_sum / np.sum(h_plus)
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]

                h_minus = np.clip(h_minus, 0, 1)

            h += 0.1 * grad

    # Final refinement with L-BFGS-B

        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
