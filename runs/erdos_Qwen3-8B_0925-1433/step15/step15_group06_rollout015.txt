import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)

    n_particles = 20
    particles = []
    particles.append(initial_h.copy())
    for _ in range(n_particles - 1):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        new_particle = initial_h + perturb
        new_particle = np.clip(new_particle, 0, 1)
        new_particle = new_particle * target_sum / np.sum(new_particle)
        particles.append(new_particle.copy())
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    pbest_scores = [compute_c5(p, dx) for p in particles]
    gbest = particles[np.argmin(pbest_scores)].copy()
    gbest_score = np.min(pbest_scores)
    best_h = gbest.copy()
    best_c5 = gbest_score

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    w = 0.729
    c1 = 1.494
    c2 = 1.494
    max_velocity = 0.1
    max_iterations = 200
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        scores = [compute_c5(p) for p in particles]
        for i in range(n_particles):
            if scores[i] < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = scores[i]
        best_score = np.min(scores)
        if best_score < gbest_score:

            gbest = particles[gbest_index].copy()

    best_h = np.clip(best_h, 0, 1)
