import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except NameError:
        n_points = 45
        initial_h_values = np.linspace(0, 2, n_points)
        initial_h_values = np.clip((initial_h_values / 2) * 2 - 1, 0, 1)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = 0.381
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    class Particle:
        def __init__(self, h):
            self.h = h.copy()
            self.pbest = h.copy()
            self.pbest_fitness = evaluate(h)
            self.velocity = np.random.uniform(-0.1, 0.1, size=n_points)
    
    particles = [Particle(np.clip(np.random.rand(n_points) * 2 - 1, 0, 1)) for _ in range(10)]
    for p in particles:
        p.h = p.h * target_sum / np.sum(p.h)
        p.pbest = p.h.copy()
        p.pbest_fitness = evaluate(p.h)
    gbest = min(particles, key=lambda p: p.pbest_fitness).pbest
    best_c5 = evaluate(gbest)
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for p in particles:
            p.velocity = 0.729 * p.velocity + 1.494 * np.random.rand(n_points) * (p.pbest - p.h) + 1.494 * np.random.rand(n_points) * (gbest - p.h)
            p.h = np.clip(p.h + p.velocity, 0, 1)
            p.h = p.h * target_sum / np.sum(p.h)
            fitness = evaluate(p.h)
            if fitness < p.pbest_fitness:
                p.pbest = p.h.copy()
                p.pbest_fitness = fitness
        gbest = min(particles, key=lambda p: p.pbest_fitness).pbest
        best_c5 = evaluate(gbest)
    
    best_h = gbest
    return (np.clip(best_h, 0, 1), best_c5, n_points)
