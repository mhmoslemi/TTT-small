import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = i + k_max
            if 0 <= j_plus < n_points:
                term1 = 1 - h[j_plus]
            else:
                term1 = 0.0
            j_minus = i - k_max
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            else:
                term2 = 0.0
            grad[i] = dx * (term1 - term2)
        return grad

    def lbfgs_refine(h):
        bounds = [(0, 1)] * n_points
        res = minimize(objective, h, method='L-BFGS-B', jac=compute_gradient, bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        return res.x

    initial_patterns = [np.array(initial_h_values, dtype=float)]
    x = np.linspace(0, 2, n_points)
    sinusoidal = 0.5 + 0.5 * np.sin(np.pi * x)
    sinusoidal = sinusoidal * target_sum / np.sum(sinusoidal)
    initial_patterns.append(sinusoidal)
    random = np.random.uniform(0, 1, n_points)
    random = random * target_sum / np.sum(random)
    initial_patterns.append(random)

    best_h = initial_patterns[0].copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    for idx, pattern in enumerate(initial_patterns):
        if time.time() - start_time > budget_s * 0.6:
            break
        result = differential_evolution(objective, [(0, 1)] * n_points, strategy='best1bin', maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.9, seed=seed)
        if result.success:
            candidate = result.x
            if objective(candidate) < best_c5:
                best_h = candidate
                best_c5 = objective(candidate)

    if time.time() - start_time > budget_s * 0.8:
        pass
    else:
        for _ in range(3):
            if time.time() - start_time > budget_s * 0.9:
                break
            best_h = lbfgs_refine(best_h)
            if objective(best_h) < best_c5:
                best_c5 = objective(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
