import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n / 2.0

        return np.max(corr) * dx

        h_t = h_t * target_sum / np.sum(h_t)
        return h_t

    def generate_perturbed(h):
        h_p = h * (1 + 0.1 * np.random.randn(n))
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * target_sum / np.sum(h_p)
