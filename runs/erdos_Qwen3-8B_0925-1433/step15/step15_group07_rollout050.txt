import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / current_sum

    h = np.array(initial_h_values, dtype=float)
    h = project(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def grad(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        dx_val = 2.0 / n_points
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = (i + k_max) % n_points
            term1 = (1 - h[j]) * dx_val
            i2 = (i - k_max) % n_points
            term2 = h[i2] * dx_val
            grad[i] = term1 - term2
        return grad

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x

    h_values = project(best_h)
    final_c5 = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * (2.0 / n_points)
    return (h_values, final_c5, n_points)
