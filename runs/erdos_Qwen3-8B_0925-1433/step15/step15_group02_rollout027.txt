        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[k_max]
            h_minus = x.copy()
            h_minus[i] -= 1e-4
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[k_max]
            grad[i] = (corr_plus - corr_minus) / (2e-4)
        return -grad

    start_time = time.time()
    patterns = []

    # Generate varied initial patterns
    for _ in range(5):
        m = np.random.randint(5, 15)
        lengths = np.random.rand(m)
        lengths /= lengths.sum()
        intervals = []
        for l in lengths:
            s = np.random.uniform(0, 2 - l)
            intervals.append((s, s + l))
        h_pattern = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            for a, b in intervals:
                if a <= x <= b:
                    h_pattern[i] = 1
                    break
        patterns.append(h_pattern)

    # Optimize each pattern
    for p in patterns:
        h_candidate = np.clip(p, 0, 1)
        res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            if time.time() - start_time > budget_s:
                break

    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq', 'fun':lambda x: np.sum(x) - target_sum}, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
