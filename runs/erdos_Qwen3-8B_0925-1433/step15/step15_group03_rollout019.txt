import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def make_pattern(base):
        return np.clip(base * target_sum / np.sum(base), 0, 1)

    x = np.linspace(0, 2, n_points)
    sin_p = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    tri_p = 1 - np.abs(x - 1)
    reg_p = np.zeros(n_points)
    reg_p[::2] = 1.0
    ran_p = np.random.rand(n_points)
    inc_p = np.linspace(0, 1, n_points)

    initial_candidates = [
        make_pattern(np.array(initial_h_values)),
        make_pattern(sin_p), make_pattern(tri_p), make_pattern(reg_p),
        make_pattern(ran_p), make_pattern(inc_p)
    ]

    best_h = initial_candidates[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_p = i + k_max
            if 0 <= j_p < n_points:
                t1 = 1 - h[j_p]
            else:
                t1 = 0
            j_m = i - k_max
            if 0 <= j_m < n_points:
                t2 = h[j_m]
            else:
                t2 = 0
            grad[i] = dx * (t1 - t2)
        return grad

    start = time.time()

    for h0 in initial_candidates:
        res = minimize(obj, h0, method='L-BFGS-B', jac=compute_grad,
                       bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            curr_c5 = res.fun
            if curr_c5 < best_c5:
                best_h, best_c5 = res.x.copy(), curr_c5

    for _ in range(3):
        if time.time() - start > budget_s:
            break
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', jac=compute_grad,
                       bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
