import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_regular_grid():
        grid = np.zeros(n_points)
        for i in range(n_points // 2):
            idx = (i * 2) % n_points
            grid[idx] = 1
        return grid * target_sum / np.sum(grid)
    
    initial_designs = []
    initial_designs.append(initial_h_values.copy())
    initial_designs.append(generate_regular_grid())
    noise = np.random.normal(0, 0.1, n_points)
    random_initial = initial_h_values.copy() + noise
    random_initial = np.clip(random_initial, 0, 1)
    random_initial = random_initial * target_sum / np.sum(random_initial)
    initial_designs.append(random_initial)
    
    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    for design in initial_designs:
        h_start = design.copy()
        if np.any(np.isnan(h_start)):
            continue
        result = minimize(objective, h_start, method='Nelder-Mead', bounds=[(0,1)]*n_points, tol=1e-5)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
