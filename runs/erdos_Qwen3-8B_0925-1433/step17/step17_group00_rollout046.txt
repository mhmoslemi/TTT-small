import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Generate multiple initial patterns
    initial_patterns = []
    initial_patterns.append(initial_h_values.copy())
    
    # Uniform spacing
    n_ones = int(round(target_sum))

    initial_patterns.append(uniform_h)
    
    # Random binary
    random_h = np.zeros(n_points)
    indices = np.random.choice(n_points, n_ones, replace=False)
    random_h[indices] = 1
    initial_patterns.append(random_h)
    
    # Geometric spacing
    geometric_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            geometric_h[i] = 1
    initial_patterns.append(geometric_h)
    
    # Evaluate initial patterns
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': constraint}
    
    best_h = initial_h_values.copy()

    start_time = time.time()
    for pattern in initial_patterns:
        res = minimize(objective, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:

            best_h = res.x

    for _ in range(5):

        if res.success and res.fun < best_c5:
