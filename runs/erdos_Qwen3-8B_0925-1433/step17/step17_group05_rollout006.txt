    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def compute_overlaps(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        overlaps = correlation[n_points-1:2*n_points-1] * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        return max_overlap, k_max

    start_time = time.time()
    max_iter = 300
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        current_c5, k_max = compute_overlaps(h)
        if current_c5 >= best_c5:
            break

        def obj(x):
            shifted = np.roll(x, k_max)
            return np.sum(x * (1 - shifted)) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1.0) for _ in range(n_points)]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_new = res.x
            h_new = project(h_new)
            new_c5, _ = compute_overlaps(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h = h_new
        if iter % 10 == 0:
            perturbation = np.random.normal(0, 0.15, n_points)
            h = project(h + perturbation)

    h_values = project(h)
    return (h_values, best_c5, n_points)
