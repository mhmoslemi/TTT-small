from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, rate=0.2):
        idxs = np.random.choice(n_points, int(n_points*rate), replace=False)
        h[idxs] *= np.random.uniform(0.5, 1.5)
        return project(h)

    def crossover(h1, h2):
        p = np.random.rand(n_points)
        mid = np.random.randint(1, n_points-1)
        return np.where(p < 0.5, h1, h2)

    best_h = project(initial_h_values * target_sum / np.sum(initial_h_values))
    best_c5 = fitness(best_h)

    start_time = time.time()
    pop_size = 50
    population = [project(initial_h_values * target_sum / np.sum(initial_h_values) + np.random.normal(0, 0.15, n_points)) for _ in range(pop_size)]
    population = [project(np.random.uniform(0, 1, n_points)) for _ in range(pop_size - len(population))]

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [fitness(ind) for ind in population]
        top_indices = np.argsort(fitnesses)[:int(pop_size * 0.1)]
        parents = [population[i] for i in top_indices]
        new_population = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        population = new_population + [population[i] for i in top_indices]
        for candidate in population:
            if fitness(candidate) < best_c5:
                best_h, best_c5 = candidate, fitness(candidate)

    for _ in range(10):

        h_pert = best_h + np.random.normal(0, 0.05, n_points)
