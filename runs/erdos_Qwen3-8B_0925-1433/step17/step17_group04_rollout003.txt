import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def create_candidate():
        return np.array([0.3 if i % 2 else 0.7 for i in range(n_points)])

        initial_h_values,

        create_candidate(),

    ]
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')
    start = time.time()

    for cand in candidates:
        h = cand * target_sum / np.sum(cand)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
