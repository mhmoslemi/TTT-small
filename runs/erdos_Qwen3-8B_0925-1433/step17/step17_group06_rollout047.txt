from scipy.stats import qmc

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

        return h

        h = np.zeros(n_points)

        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)

        return h
    
    def generate_random():
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        return h
    
    designs = [generate_halton(), generate_grid(), generate_perturb(), generate_random()]
    
    best_h = np.array(initial_h_values, dtype=float)
