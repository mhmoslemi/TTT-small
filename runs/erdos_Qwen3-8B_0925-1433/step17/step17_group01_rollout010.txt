    decay_rate = 0.999
    start_time = time.time()
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.argsort(corr)[::-1][:max_lags]
        grad = np.zeros(n_points)
        for k in lags:
            if k < 0:
                k += n_points
            grad += 1 - h[(np.arange(n_points) + k) % n_points]
        lr = lr_0 * (1 + np.cos(t / T)) / 2
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.linalg.norm(grad) < 1e-3:
            break
        lr_0 *= decay_rate
    return (np.clip(best_h, 0, 1), best_c5, n_points)
