            k = m - (n_points - 1)
            
            if k != 0:
                positions = []
                if k > 0:
                    for i in range(n_points - k):
                        positions.append((i, i + k))
                else:
                    for i in range(n_points + k):
                        positions.append((i, i + k))
                
                if positions:
                    i, j = random.choice(positions)
                    selected[i] += np.random.normal(0, mutation_strength)
                    selected[j] += np.random.normal(0, mutation_strength)
                    selected = np.clip(selected, 0, 1)
                    selected = selected * target_sum / np.sum(selected)
            
            new_population.append(selected)
        
        population = new_population
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
