def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)

        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)
    population = np.array(population)

    def evaluate(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h):
        h_mut = h + np.random.normal(0, 0.1, h.shape)
        h_mut = np.clip(h_mut, 0, 1)
        s = np.sum(h_mut)
        if s == 0:
            return np.full_like(h_mut, 0.5)
        return h_mut * (target_sum / s)

    def crossover(p1, p2):
        alpha = np.random.uniform(0.5, 1.5)
        c1 = p1 * alpha + p2 * (1 - alpha)
        c2 = p2 * alpha + p1 * (1 - alpha)
        c1 = np.clip(c1, 0, 1)
        c2 = np.clip(c2, 0, 1)
        s1, s2 = np.sum(c1), np.sum(c2)
        if s1 == 0:
            c1 = np.full_like(c1, 0.5)
        elif s2 == 0:
            c2 = np.full_like(c2, 0.5)
        else:
            c1 = c1 * (target_sum / s1)
            c2 = c2 * (target_sum / s2)
        return c1, c2

    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    gen = 0

    while time.time() - start_time < budget_s:
