from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initializations = []
    # First initialization: given
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    initializations.append(h_initial)
    # Second initialization: random
    h_random = np.random.rand(n_points)
    h_random = h_random * target_sum / np.sum(h_random)
    initializations.append(h_random)
    # Third initialization: regular pattern
    h_regular = np.zeros(n_points)
    for i in range(n_points):
        if (i * dx) % 0.5 <= 1e-6:
            h_regular[i] = 1.0
    h_regular = h_regular * target_sum / np.sum(h_regular)
    initializations.append(h_regular)

    best_c5 = float('inf')
    best_h = None

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    start_time = time.time()
    for h_candidate in initializations:
