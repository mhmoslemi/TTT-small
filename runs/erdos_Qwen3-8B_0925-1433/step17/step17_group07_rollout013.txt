from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / s)

    # Initial population

    pop = []

    h_start = normalize(h_start)

        h = normalize(h)
        pop.append(h)

    best_h = h_start
    best_c5 = compute_c5(h_start)

    def tournament_selection(pop, k=3):
        return min(np.random.choice(pop, k), key=compute_c5)

    start_time = time.time()

        if time.time() - start_time > budget_s:

            child += noise

        # Local refinement
        top = sorted(pop, key=compute_c5)[:5]

            res = minimize(lambda x: compute_c5(x), h, method='SLSQP',
