            i2 = (i - k_max) % len(x)
            term2 = x[i2] * dx
            grad[i] = term1 - term2
        return grad

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def optimize_with_bounds(h_start):
        bounds = Bounds([0.0] * n_points, [1.0] * n_points)
        res = minimize(objective, h_start, method='SLSQP', jac=gradient, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-7, options={'maxiter': 100})
        return res.x if res.success else h_start

    start_time = time.time()

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        h_candidate = optimize_with_bounds(best_h)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            print(f"New best C5: {best_c5}")

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
