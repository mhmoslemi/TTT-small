import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    start_time = time.time()

    for iteration in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        perturbation = np.random.normal(0, 0.15, n_points)
        h_pert = best_h + perturbation
        h_pert = project(h_pert)
        def obj(x):
            h = project(x)
            corr_val = np.correlate(h, 1 - h, mode='full')[k_max]
            return corr_val * dx
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success:
            new_h = res.x
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        if iteration % 10 == 0:
            noise = np.random.uniform(-0.15, 0.15, n_points)
            h_pert = best_h + noise
            h_pert = project(h_pert)
            new_c5 = np.max(np.correlate(h_pert, 1 - h_pert, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = h_pert
                best_c5 = new_c5

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
