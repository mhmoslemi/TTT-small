import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_vec(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constrained_objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    def perturb(h):
        noise = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + noise, 0, 1)

    h = initial_h * target_sum / np.sum(initial_h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        h_candidate = perturb(best_h)
        result = minimize(constrained_objective, h_candidate, method='SLSQP', jac=gradient, bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6,
                          options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                print(f"New best C5: {best_c5}")

    result_de = differential_evolution(objective_vec, bounds, strategy='best1bin', maxiter=50, popsize=30, tol=1e-6,
                                      mutation=(0.5, 1), recombination=0.9, disp=False)
    if result_de.success:
        h_candidate = result_de.x
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
