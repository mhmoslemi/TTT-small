            np.roll(np.random.rand(n_points) > 0.5, 1),
            np.random.rand(n_points) * 0.6 + 0.2,
            np.random.rand(n_points) * 0.4 + 0.3
        ]
        for pattern in init_patterns:
            pattern = pattern * target_sum / np.sum(pattern)
            res = minimize(obj, pattern, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
