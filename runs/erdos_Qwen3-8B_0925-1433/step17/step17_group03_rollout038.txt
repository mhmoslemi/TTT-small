import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    population = [initial_h]
    for _ in range(49):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        evals = []
        for h in population:
            c5 = compute_c5(h)
            evals.append((c5, h))
        evals.sort()
        top = evals[:5]
        new_pop = []
        for _ in range(50):
            a, b = np.random.choice(top, 2, replace=False)
            child = a[1] * np.random.rand() + b[1] * (1 - np.random.rand())
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
        for c, h in evals:
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c
        for c, h in evals[:3]:
            def obj(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            def grad(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                k = np.argmax(corr) - (n_points - 1)
                g = np.zeros(n_points)
                for i in range(n_points):
                    j = i + k
                    if 0 <= j < n_points:
                        g[i] = 1 - x[j]
                return g
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, jac=grad, tol=1e-6)
            if res.success:
                h_new = res.x
                h_new = h_new * target_sum / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c_new
    return (np.clip(best_h, 0, 1), best_c5, n_points)
