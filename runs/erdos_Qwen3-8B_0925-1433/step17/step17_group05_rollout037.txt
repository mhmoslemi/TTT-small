            if current_c5_pert < best_c5:
                best_h, best_c5 = perturbed_h, current_c5_pert

    def final_obj(x):
        h_pert = project(x)
        corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
        return np.max(corr_pert) * dx
    res_final = minimize(final_obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
    if res_final.success and res_final.fun < best_c5:
        best_h = res_final.x
        best_c5 = res_final.fun
    h_values = project(best_h)
    return (h_values, best_c5, n_points)
