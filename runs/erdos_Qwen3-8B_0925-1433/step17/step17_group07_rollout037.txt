from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_erdos(best_h, n_points, dx)
    best_c5 = min(best_c5, 0.381)

    def compute_erdos(h_vec, n, dx):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_local(h_start):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(lambda x: compute_erdos(x, n_points, dx), 
                      h_start, method='SLSQP', 
                      jac=lambda x: gradient(x, n_points, dx),
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-6, options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        if res.success:
            return res.x
        return h_start

    def gradient(x, n, dx):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n):
            j = (i + k_max) % n
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % n
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    pop_size = 50
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    pop = np.array(pop)

    max_gen = 200

    while gen < max_gen and time.time() - time_start < budget_s:

        fitness = np.array([compute_erdos(p, n_points, dx) for p in pop])
