from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial = np.array(initial_h_values, dtype=float)
    initial = initial * target_sum / np.sum(initial)
    best_h = initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    population = []
    for _ in range(50):
        if np.random.rand() < 0.7:
            pert = np.random.normal(0, 0.2, n_points)

        h = project(h)
        population.append(h)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:

        order = np.argsort(fitness)
        parents = [population[i] for i in order[:10]]
        new_population = []

        population = [project(ind) for ind in new_population]
