import time

    ELITE_COUNT = 5

        if np.random.rand() < 0.4:

            rand_h = rand_h * target_sum / np.sum(rand_h)
            population.append(rand_h)
        else:
            pattern = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
            pattern = pattern * target_sum / np.sum(pattern)
            population.append(pattern)

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    start_time = time.time()

        return np.max(corr) * dx
