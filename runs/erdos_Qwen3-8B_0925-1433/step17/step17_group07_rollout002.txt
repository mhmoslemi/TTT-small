import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    population_size = 50
    population = []

    # Start with initial_h_values
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population.append(h_initial)

    # Generate other individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = h_initial
    best_c5 = -1.0
    start_time = time.time()

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all individuals
        fitness_list = []
        for h in population:
            fitness_list.append(compute_c5(h))
        # Select parents via tournament
        parents = []
        for _ in range(population_size):
            indices = np.random.choice(len(population), 3, replace=False)
            candidates = [population[i] for i in indices]
            fitnesses = [fitness_list[i] for i in indices]
            best_idx = np.argmin(fitnesses)
            parents.append(candidates[best_idx])
        # Generate offspring via crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            parent1 = parents[i]
            parent2 = parents[i + 1]
            alpha = np.random.rand()
            child1 = parent1 * (1 - alpha) + parent2 * alpha
            child2 = parent2 * (1 - alpha) + parent1 * alpha
            offspring.append(child1)
            offspring.append(child2)
        # Mutate offspring
        mutated_offspring = []
        for child in offspring:
            noise = np.random.normal(0, 0.1, child.shape)
            child = np.clip(child + noise, 0, 1)
            current_sum = np.sum(child)
            if current_sum == 0:
                child = np.full(child.shape, 0.5)
            else:
                scale = target_sum / current_sum
                child *= scale
            mutated_offspring.append(child)
        # Replace worst individuals
        zipped = list(zip(fitness_list, population))
        zipped.sort()
        sorted_fitness, sorted_pop = zip(*zipped)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
