        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags_idx = np.argsort(corr)[::-1][:top_lag_count]
        top_lags = [idx - (n_points - 1) for idx in top_lags_idx]
        gradient = np.zeros(n_points)
        for lag in top_lags:
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    gradient[i] += 1.0 - h[j]
        grad_norm = np.max(gradient) + 1e-8
        gradient /= grad_norm
        learning_rate = learning_rate_initial / np.sqrt(accumulated + 1e-8)
        delta = learning_rate * gradient
        h = h - delta
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h > 0 else 1.0
        h = h * scale
        accumulated += gradient ** 2
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate_initial *= decay_rate

    return (np.clip(best_h, 0, 1), best_c5, n_points)
