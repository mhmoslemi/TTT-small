from sklearn.gaussian_process.kernels import RBF
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    initial_h = np.array(initial_h_values, dtype=float)

    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5)

    y = np.array([best_c5])
    gp.fit(X, y)
    X_train = X.copy()
    y_train = y.copy()

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break

        mean, std = gp.predict(X_train, return_std=True)
        noise = np.random.normal(0, std[0], size=n_points)

        next_h = next_h * target_sum / np.sum(next_h)
