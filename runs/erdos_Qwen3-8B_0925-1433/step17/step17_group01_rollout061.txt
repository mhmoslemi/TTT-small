    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        sorted_indices = np.argsort(corr)[::-1]
        top_indices = sorted_indices[:top_lags]
        lags = [idx - (n_points - 1) for idx in top_indices]
        g = 1.0 - h
        total_grad = np.zeros(n_points)
        for lag in lags:
            total_grad += np.roll(g, lag)
        lr = lr_0 * (1 + np.cos(t / T)) / 2.0
        h = h - lr * total_grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * 2.0 / n_points
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
