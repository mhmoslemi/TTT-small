import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    accumulated = np.zeros(n_points)
    learning_rate_initial = 0.05
    max_iterations = 200
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:5]
        top_lags = [idx - (n_points - 1) for idx in top_indices]
        gradient = np.zeros(n_points)
        for lag in top_lags:
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    gradient[i] += 1.0 - h[j]
        lr = [learning_rate_initial / np.sqrt(accumulated[i] + 1e-8) for i in range(n_points)]
        delta = np.array([lr[i] * gradient[i] for i in range(n_points)])
        h = h - delta
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h > 0 else 1.0
        h = h * scale
        for i in range(n_points):
            accumulated[i] += gradient[i] ** 2
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

        if iteration % 10 == 0:
            def objective(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                max_val = np.max([corr[idx] for idx in top_indices])
                return max_val * dx
            res = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
            if res.success:
