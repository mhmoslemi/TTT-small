    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        momentum_grad = momentum * momentum_grad + gradient
        h -= learning_rate * momentum_grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb_std = np.random.uniform(0.01, 0.05)
        perturb = np.random.normal(0, perturb_std, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr_perturbed) * dx
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
