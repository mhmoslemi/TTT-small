            parents.append(initial_population[idx1] if fit[idx1] < fit[idx2] else initial_population[idx2])
        
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            pt = np.random.randint(1, n_points-1)
            c1 = np.concatenate([p1[:pt], p2[pt:]])
            c2 = np.concatenate([p2[:pt], p1[pt:]])
            offspring.append(c1)
            offspring.append(c2)
        
        for i in range(len(offspring)):
            h = offspring[i]
            idx = np.random.randint(n_points)
            h[idx] += np.random.uniform(-0.1, 0.1) * (target_sum / n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            offspring[i] = h
        
        initial_population = parents + offspring
        for i in range(len(initial_population)):
            h = initial_population[i]
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            initial_population[i] = h

    return (np.clip(best_h, 0, 1), best_c5, n_points)
