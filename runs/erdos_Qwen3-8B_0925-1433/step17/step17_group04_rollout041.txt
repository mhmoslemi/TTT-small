import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_configs = [np.array(initial_h_values, dtype=float)]
    
    for _ in range(3):
        r = np.random.rand(n_points)
        r = r * target_sum / np.sum(r)
        initial_configs.append(r)
    
    structured = np.sin(2 * np.pi * np.arange(n_points) / (n_points + 1))
    structured = (structured + 1) / 2
    structured = structured * target_sum / np.sum(structured)
    initial_configs.append(structured)
    
    structured2 = np.zeros(n_points)
    structured2[::2] = 1.0
    structured2 = structured2 * target_sum / np.sum(structured2)
    initial_configs.append(structured2)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def optimize(cfg):
        res = minimize(obj, cfg, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        return res.x if res.success else cfg

    best_h = initial_configs[0]
    best_c5 = obj(best_h)

    start_time = time.time()
    
    for c in initial_configs:
        if time.time() - start_time > budget_s:
            break
        curr = optimize(c)
        curr_c5 = obj(curr)
        if curr_c5 < best_c5:
            best_h, best_c5 = curr, curr_c5
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.1, n_points)
        p = best_h.copy() + pert
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
        curr = optimize(p)
        curr_c5 = obj(curr)
        if curr_c5 < best_c5:
            best_h, best_c5 = curr, curr_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
