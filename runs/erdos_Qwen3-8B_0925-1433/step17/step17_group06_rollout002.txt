    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='Nelder-Mead', tol=1e-4)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, best_c5, n_points)
