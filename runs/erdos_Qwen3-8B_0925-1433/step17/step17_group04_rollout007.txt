import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            lag = k_max - (n_points - 1)
            if 0 <= i + lag < n_points:
                term1 = 1 - h[i + lag]
            else:
                term1 = 0
            if 0 <= i - lag < n_points:
                term2 = h[i - lag]
            else:
                term2 = 0
            grad[i] = (term1 - term2)
        return grad

    # Initial guess
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Structured patterns
    patterns = [h.copy()]
    m = n_points // 5
    patterns.append(np.zeros(n_points))
    patterns[-1][::m] = 1.0
    patterns[-1] = patterns[-1] * target_sum / np.sum(patterns[-1])
    
    for i in range(n_points):
        patterns.append(0.5 + 0.5 * np.sin(2*np.pi * i / n_points))
    for p in patterns:
        p = p * target_sum / np.sum(p)
        patterns[patterns.index(p)] = p

    # Optimizer
    for p in patterns:
        h = p.copy()
        h = h * target_sum / np.sum(h)
        for _ in range(300):
            if time.time() - time.time() > budget_s:
                break
            grad = compute_gradient(h)
            h = h - 0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    # Perturbation and refinement
    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        pert = np.random.normal(0, 0.05, n_points)
        h = best_h + pert
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                      constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
