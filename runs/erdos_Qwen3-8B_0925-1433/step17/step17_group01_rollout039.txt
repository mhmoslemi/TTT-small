        population[i] = population[i] * target_sum / np.sum(population[i])
    
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    max_generations = 100
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all candidates
        evals = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            evals.append((c5, h))
        evals.sort()
        current_best = evals[0][0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = evals[0][1]
        
        # Select parents via tournament
        tournament_size = 5
        parents = []
        for _ in range(pop_size // 2):
            idxs = np.random.choice(len(evals), tournament_size, replace=False)
            winner = min(idxs, key=lambda x: evals[x][0])
            parents.append(evals[winner][1])
        
        # Crossover and mutate
        new_population = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                child = (p1 + p2) / 2
                # Mutation based on top lag
                corr = np.correlate(child, 1 - child, mode='full')
                k_max = np.argmax(corr)
                k = k_max
                if k > len(child) - k - 1:
                    k = -k
                if k < len(child) - k:
                    for x in range(len(child) - k):
                        child[x] -= np.random.uniform(0.005, 0.015)
                        child[x + k] += np.random.uniform(0.005, 0.015)
                else:
                    for x in range(k, len(child)):
                        child[x - k] -= np.random.uniform(0.005, 0.015)
                        child[x] += np.random.uniform(0.005, 0.015)
                # Clip and normalize
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            else:
                new_population.append(parents[i])
        population = new_population
        population.append(best_h)
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
