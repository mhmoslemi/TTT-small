import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import norm
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, scale=0.1):
        pert = np.random.normal(0, scale, h.shape)
        return np.clip(np.abs(h + pert), 0, 1)

    # Initial exploration
    for _ in range(10):
        if time.time() > budget_s:
            break
        h_candidate = perturb(best_h, scale=0.2)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Local refinement with SLSQP
    def optimize_with_bounds(h_start):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, 
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                       tol=1e-7, options={'maxiter': 50, 'ftol': 1e-8})
        return res.x

    for _ in range(20):
        if time.time() > budget_s:
            break
        h_candidate = optimize_with_bounds(best_h)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Final refinement with perturbations
    for _ in range(10):
        if time.time() > budget_s:
            break
        h_candidate = perturb(best_h, scale=0.05)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
