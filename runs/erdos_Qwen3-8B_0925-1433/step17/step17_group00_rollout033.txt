    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n

    elapsed = 0
    for _ in range(30):
