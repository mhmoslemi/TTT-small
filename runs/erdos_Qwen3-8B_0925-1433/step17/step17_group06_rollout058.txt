import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def halton(n, base):
        seq = []
        for i in range(1, n+1):
            x = i
            r = 0
            while x > 0:
                x, rem = divmod(x, base)
                r += rem / (base ** (i))
            seq.append(r)
        return np.array(seq)

    def generate_initial_guess():
        h = np.zeros(n_points)
        halton_points = halton(n_points, 2)
        halton_points *= 2
        indices = np.round(halton_points * n_points / 2).astype(int)
        indices = np.clip(indices, 0, n_points-1)
        unique_indices = np.unique(indices)
        m = int(np.round(n_points / 2))
        selected = unique_indices[:m]
        h[selected] = 1
        return h

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    initial_guesses = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    initial_guesses.append(generate_initial_guess())

    start_time = time.time()
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        result = minimize(obj, guess, method='Nelder-Mead', options={'xatol': 1e-6, 'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
        if time.time() - start_time > budget_s - 10:
            break

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
