            m_hat = m / (1 - beta1**(i+1))
            v_hat = v / (1 - beta2**(i+1))
            h = h - 0.05 * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            cur_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5

    h = best_h.copy()
    m, v = np.zeros_like(h), np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    start = time.time()
    for i in range(200):
        if time.time() - start > budget_s:
            break
        h_curr = h * target_sum / np.sum(h)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)
        grad = np.zeros_like(h_curr)
        for j in range(n_points):
            ii = j + lag
            jj = j - lag
            if 0 <= ii < n_points:
                term1 = 1 - h_curr[ii]
            else:
                term1 = 0
            if 0 <= jj < n_points:
                term2 = h_curr[jj]
            else:
                term2 = 0
            grad[j] = term1 - term2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        h = h - 0.01 * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        cur_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if cur_c5 < best_c5:
            best_h, best_c5 = h.copy(), cur_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
