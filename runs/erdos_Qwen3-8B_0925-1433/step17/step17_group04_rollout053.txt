import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    POP_SIZE = 50
    MAX_GEN = 200

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Initialize population with initial solution, random, and sine patterns
    population = []
    for _ in range(POP_SIZE):
        if _ < 10:
            h = initial_h_values.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        elif _ < 20:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        else:
            h = 0.5 + 0.25 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
            h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(MAX_GEN):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h, best_c5 = population[best_idx].copy(), fitness[best_idx]

            parents.append(population[a] if fitness[a] < fitness[b] else population[b])

        offspring = []
        for i in range(0, len(parents), 2):
