import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.1
    decay_rate = 0.99
    momentum = 0.9
    velocity = np.zeros(n_points)
    max_iterations = 300
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        velocity = momentum * velocity + gradient
        h -= learning_rate * velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    # Additional refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr_perturbed) * dx
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
