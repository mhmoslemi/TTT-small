import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.995
    min_learning_rate = 0.001
    max_iterations = 200
    velocity = np.zeros(n_points)
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k_max = max(0, min(k_max, n_points - 1))
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1.0 - h[j]
        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        learning_rate *= decay_rate
        if learning_rate < min_learning_rate:
            learning_rate = min_learning_rate
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
