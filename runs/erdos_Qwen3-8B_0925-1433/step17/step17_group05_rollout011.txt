    target_sum = n_points / 2.0

    initial_h = project(initial_h)
    population = []

        perturbation = np.random.normal(0, 0.15, n_points)
        candidate = initial_h + perturbation
        candidate = project(candidate)
        population.append(candidate)
    for _ in range(20):
        x = np.linspace(0, 2, n_points)

        wave = np.clip(wave, 0, 1)

        population.append(candidate)

    population = np.array(population)
