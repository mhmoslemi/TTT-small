import numpy as np
from scipy.optimize import minimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def generate_rand():
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.full(n_points, 0.5)
        h = h * (target_sum / sum_h)
        h = np.clip(h, 0.0, 1.0)
        return h

    pop = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(49):
        pop.append(generate_rand())

    best_h = pop[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in pop]
        parents = []
        for _ in range(25):
            idxs = np.random.choice(len(pop), 3, replace=False)
            best_idx = idxs[np.argmin(fitness[idxs])]
            parents.append(pop[best_idx].copy())
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1 = parents[i]
                p2 = parents[i + 1]
                alpha = np.random.rand()
                c = p1 * alpha + p2 * (1 - alpha)
                noise = np.random.normal(0, 0.1, n_points)
                c += noise
                c = np.clip(c, 0, 1)
                s = np.sum(c)
                if s != 0:
                    c = c * (target_sum / s)
                else:
                    c = np.full(n_points, 0.5)
                offspring.append(c)
        pop = offspring + pop[:len(pop) - len(offspring)]
        top_idx = np.argsort(fitness)[:5]
        for i in top_idx:
            h = pop[i].copy()
            h = h * target_sum / np.sum(h)
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            def grad(x):
                corr = np.correlate(x, 1 - x, mode='full')
                k = np.argmax(corr)
                g = np.zeros_like(x)
                for i in range(len(x)):

                    g[i] = (1 - x[j]) * dx
                    i2 = (i - k) % len(x)
                    g[i] -= x[i2] * dx
