import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    initial_patterns = [np.array(initial_h_values, dtype=float)]
    uniform_h = np.zeros(n_points)
    num_ones = int(n_points / 2)
    uniform_h[:num_ones] = 1.0
    uniform_h[-1] += (target_sum - num_ones)
    initial_patterns.append(uniform_h)
    random_h = np.random.rand(n_points)
    random_h = random_h * target_sum / np.sum(random_h)
    initial_patterns.append(random_h)

    best_h = initial_patterns[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    for pattern in initial_patterns:
        h = pattern * target_sum / np.sum(pattern)
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
