import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x.copy()
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    # Generate diverse initial patterns
    h0s = []
    h0s.append(np.array(initial_h_values, dtype=float))
    h0s.append(np.full(n_points, 0.5))
    h0s.append(np.array([0.3 if i % 2 == 0 else 0.7 for i in range(n_points)]))
    h0s.append(np.array([0.5 + 0.2 * np.sin(2 * np.pi * i / n_points) for i in range(n_points)]))
    h0s.append(np.random.rand(n_points))
    for _ in range(4):
        h0s.append(np.random.permutation([0.0] * (n_points // 2) + [1.0] * (n_points // 2)))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    start_time = time.time()

    for h0 in h0s:
        if time.time() - start_time > budget_s:
            break
        h0 = h0 * target_sum / np.sum(h0)
        res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Final refinement
    if time.time() - start_time < budget_s:
        h = best_h.copy()
        h = h * target_sum / np.sum(h)
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 50})
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
