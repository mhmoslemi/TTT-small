import numpy as np
from scipy.optimize import minimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    population = []
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population.append(initial_h)
    for _ in range(49):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    def optimize_candidate(candidate):
        h = candidate.copy()
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        if res.success:
            return res.x
        return h

    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in population]

        offspring = []

            alpha = np.random.uniform(0, 1)

            h1 = population[idx]
