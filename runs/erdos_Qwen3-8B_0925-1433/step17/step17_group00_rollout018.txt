        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(best_h + noise, 0.0, 1.0)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * target_sum / np.sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h, final_c5, n_points)
