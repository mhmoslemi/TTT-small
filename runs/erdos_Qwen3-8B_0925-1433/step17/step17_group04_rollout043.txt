import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - target_sum

    def make_initial_patterns():
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
        yield h
        yield np.zeros(n_points) + 0.5
        yield np.array([0.3 if i % 2 == 0 else 0.7 for i in range(n_points)])
        yield 0.5 + 0.2 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        yield np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        yield np.random.permutation(np.concatenate([np.ones(10), np.zeros(10)])) * target_sum / np.sum(np.concatenate([np.ones(10), np.zeros(10)]))
        yield 0.5 * np.ones(n_points)
        yield np.random.rand(n_points) * np.random.rand(n_points)

    best_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_c5 = objective(best_h)

    start_time = time.time()
    for pattern in make_initial_patterns():
        if time.time() - start_time > budget_s:

        pattern = np.array(pattern, dtype=float)
        pattern = pattern * target_sum / np.sum(pattern)
        res = minimize(objective, pattern, method='SLSQP',
                      bounds=[(0, 1)] * n_points,

        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun
            if best_c5 < 0.3807:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
