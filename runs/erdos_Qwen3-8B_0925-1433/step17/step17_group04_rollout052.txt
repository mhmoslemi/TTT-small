from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):

        return np.max(corr) * dx

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Initial population
    POP_SIZE = 50
    population = []
    base_h = np.array(initial_h_values, dtype=float)
    base_h = base_h * target_sum / np.sum(base_h)
    population.append(base_h)
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h = clip_and_normalize(h)
        population.append(h)

    # Structured patterns
    for _ in range(POP_SIZE // 4):
        h = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
        h = clip_and_normalize(h)
        population.append(h)
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = clip_and_normalize(h)
        population.append(h)

    population = np.array(population)
    best_h = population[0].copy()

    start_time = time.time()

    for gen in range(200):

        # Selection

                parents.append(population[idx2])

        # Crossover
        offspring = []
