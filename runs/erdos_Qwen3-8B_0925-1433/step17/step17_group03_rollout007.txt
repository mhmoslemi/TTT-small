import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    def c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = c5(best_h)
    start_time = time.time()

    # Initialize population
    population = [h.copy()]
    for _ in range(49):
        mut = np.random.uniform(0, 1, n_points)
        mut = mut * target_sum / np.sum(mut)
        population.append(mut)

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [c5(ind) for ind in population]
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 5)
            best = idxs[np.argmin([fitness[i] for i in idxs])]
            parents.append(population[best].copy())
        
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < 0.7:
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy()
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        
        for i in range(len(new_pop)):
            if np.random.rand() < 0.1:
                mut = np.random.uniform(0, 1, n_points)
                mut = mut * target_sum / np.sum(mut)
                new_pop[i] = mut
        
        population = new_pop
        for ind in population:
            if c5(ind) < best_c5:
                best_h, best_c5 = ind.copy(), c5(ind)
        
        top_idx = np.argsort(fitness)[:5]
        for idx in top_idx:
            ind = population[idx].copy()
            corr = np.correlate(ind, 1 - ind, mode='full')
            max_idx = np.argmax(corr)
            lag = max_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] = 1.0 - ind[j]
            
            def objective(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            
            def jac(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                max_idx = np.argmax(corr)
                lag = max_idx - (n_points - 1)
                g = np.zeros(n_points)
                for i in range(n_points):
                    j = i + lag
                    if 0 <= j < n_points:
                        g[i] = 1.0 - x[j]
                return g
            
            res = minimize(objective, ind, method='L-BFGS-B', bounds=[(0,1)]*n_points, jac=jac)
            if res.success:
                best_h = res.x * target_sum / np.sum(res.x)
                best_c5 = c5(best_h)

    return (best_h, best_c5, n_points)
