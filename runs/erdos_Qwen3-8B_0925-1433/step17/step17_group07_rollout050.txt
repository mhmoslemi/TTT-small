            best_c5 = curr_best
            best_h = population[best_idx].copy()
        parents = []
        for _ in range(50):
            tour = np.random.choice(len(population), tournament_size, replace=False)
            tour_fit = [fitnesses[i] for i in tour]
            best_t = tour[np.argmin(tour_fit)]
            parents.append(population[best_t])
        offspring = []
        for i in range(0, 50, 2):
            p1, p2 = parents[i], parents[i+1]
            alpha = np.random.rand()
            c1 = p1 * alpha + p2 * (1 - alpha)
            c2 = p2 * alpha + p1 * (1 - alpha)
            c1 = np.clip(c1, 0, 1)
            c1 = c1 * (target_sum / np.sum(c1))
            c2 = np.clip(c2, 0, 1)
            c2 = c2 * (target_sum / np.sum(c2))
            offspring.extend([c1, c2])
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                mut = offspring[i] + normal(0, 0.1, n_points)
                mut = np.clip(mut, 0, 1)
                mut = mut * (target_sum / np.sum(mut))
                offspring[i] = mut
        offspring_fit = []
        for ind in offspring:
            corr = np.correlate(ind, 1 - ind, mode='full')
            fit = np.max(corr) * 2.0 / n_points
            offspring_fit.append(fit)
        all_cand = []
        for i in range(len(population)):
            all_cand.append((fitnesses[i], population[i]))
        for i in range(len(offspring)):
            all_cand.append((offspring_fit[i], offspring[i]))
        all_cand.sort()
        new_pop = [ind for fit, ind in all_cand[:50]]
        population = new_pop
        if gen > 0 and abs(curr_best - best_c5) < 1e-8:
            stagnation += 1
            if stagnation >= 20:
                mutation_rate *= 0.5
                stagnation = 0
        else:
            stagnation = 0
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * 2.0 / n_points
    return (best_h, final_c5, n_points)
