    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]

        new_population = []
        for _ in range(population_size):
            tournament = np.random.choice(len(population), 5, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            new_population.append(population[winner].copy())

        for i in range(population_size):
            if np.random.rand() < 0.7:
                j = np.random.randint(0, population_size)
                while j == i:
                    j = np.random.randint(0, population_size)
                parent1 = new_population[i]
                parent2 = new_population[j]
                child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 0:
                    child = child * target_sum / sum_child
                new_population[i] = child
            else:
                mutation = np.random.normal(0, 0.1, n_points)
                new_individual = new_population[i] + mutation
                new_individual = np.clip(new_individual, 0, 1)
                sum_child = np.sum(new_individual)
                if sum_child > 0:
                    new_individual = new_individual * target_sum / sum_child
                new_population[i] = new_individual
        population = new_population

        top_indices = np.argsort(fitness)[:5]
        for idx in top_indices:
            h_candidate = population[idx].copy()
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            max_idx = np.argmax(corr)
            lag = max_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] = 1.0 - h_candidate[j]
            def objective(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            bounds = [(0, 1)] * n_points
            result = minimize(objective, h_candidate, method='L-BFGS-B', bounds=bounds, jac=grad, tol=1e-6)
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                sum_h = np.sum(h_candidate)
                if sum_h > 0:
                    h_candidate = h_candidate * target_sum / sum_h
                population[idx] = h_candidate
                corr_new = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                new_c5 = np.max(corr_new) * dx
                if new_c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = new_c5

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    return (best_h, best_c5, n_points)
