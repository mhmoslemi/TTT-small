    # Final refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
    if res.success and res.fun < best_c5:
        best_c5 = res.fun
        best_h = res.x.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
