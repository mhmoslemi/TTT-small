from scipy.optimize import minimize
import time

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total < 1e-12:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def compute_overlap(h, k):
        rolled = np.roll(h, k)
        return np.dot(h, 1 - rolled) * dx

    def optimize_for_k(h, k):

            h_new = project(x)
            return compute_overlap(h_new, k)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(project(x)) - target_sum}
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-6)
        if res.success:
            return res.x
        return h
