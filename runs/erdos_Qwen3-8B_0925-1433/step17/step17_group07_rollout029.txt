from scipy.optimize import minimize

        return scale(h)

        return np.max(corr) * dx

        bounds = Bounds([0.0]*n_points, [1.0]*n_points)

                       bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
        return res.x if res.success else h_start
    
    def mutate(h, std_dev):

        return clip_and_scale(h + noise)
    
    def crossover(p1, p2):

        return (alpha * p1 + (1 - alpha) * p2)
