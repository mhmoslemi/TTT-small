import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            lag = k_max - (n_points - 1)
            if 0 <= i + lag < n_points:
                term1 = 1 - x[i + lag]
            else:
                term1 = 0
            if 0 <= i - lag < n_points:
                term2 = x[i - lag]
            else:
                term2 = 0
            grad[i] = (term1 - term2)
        return grad

    def optimize_with_slsqp(x, bounds):
        res = minimize(obj, x, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      jac=compute_gradient, tol=1e-8)
        return res.x if res.success else x

    def perturb(h, sigma=0.05):
        h = h.copy()
        noise = sigma * np.random.randn(n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        return h

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h = optimize_with_slsqp(h, Bounds([0.0, 0.0], [1.0, 1.0]))
        h = perturb(h)
        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
