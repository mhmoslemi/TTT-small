import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        return np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1))

    def objective(h):
        h_scaled = scale_h(h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def coordinate_descent(h):
        for _ in range(10):
            h = scale_h(h)
            max_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            for i in range(n_points):
                h_i = h.copy()
                for val in [0.0, 1.0]:
                    h_i[i] = val
                    h_i = scale_h(h_i)
                    corr = np.correlate(h_i, 1 - h_i, mode='full')
                    if np.max(corr) * dx < max_corr:
                        h = h_i
                        max_corr = np.max(corr) * dx
            if np.max(np.correlate(h, 1 - h, mode='full')) * dx >= max_corr:
                break
        return h

    initial_guesses = [initial_h_values]
    random_guess = np.random.rand(n_points)
    sinusoidal_guess = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
    for guess in [random_guess, sinusoidal_guess]:
        initial_guesses.append(guess)

    best_h = scale_h(np.array(initial_h_values))
    best_c5 = objective(best_h)

    for guess in initial_guesses:
        guess = scale_h(np.array(guess))
        res = minimize(objective, guess, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                      options={'maxiter': 200, 'disp': False})
        if res.success:
            h_candidate = scale_h(res.x)
            c5 = objective(h_candidate)
            if c5 < best_c5:
                best_h, best_c5 = h_candidate, c5

    best_h = coordinate_descent(best_h)
    best_c5 = objective(best_h)

    for _ in range(3):
        if time.time() - time.time() > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        h_candidate = scale_h(best_h + perturb)
        res = minimize(objective, h_candidate, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                      options={'maxiter': 50, 'disp': False})
        if res.success:
            c5 = objective(res.x)
            if c5 < best_c5:
                best_h, best_c5 = scale_h(res.x), c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
