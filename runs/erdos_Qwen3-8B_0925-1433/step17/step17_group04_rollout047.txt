        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    # Refinement
    if best_h is not None:
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x
    
    # Final clip and rescale
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, current_c5)
    
    return (best_h, best_c5, n_points)
