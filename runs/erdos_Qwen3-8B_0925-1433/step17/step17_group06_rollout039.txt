import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Generate shuffled initial designs
    initial_designs = []
    # Start with initial_h_values
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    initial_designs.append(h)
    # Generate Halton sequence-based design
    def generate_halton():
        n_halton = n_points // 2
        points = []
        for i in range(n_halton):
            d = i + 1
            xi = 0.0
            denom = 2
            while d > 0:
                xi += (d % 2) / denom
                d = d // 2
                denom *= 2
            xi *= 2.0  # scale to [0,2]
            points.append(xi)
        h = np.zeros(n_points)
        for x in points:
            idx = int(np.clip(x * n_points / 2, 0, n_points - 1))
            h[idx] = 1
        total = np.sum(h)
        h *= target_sum / total
        return h
    initial_designs.append(generate_halton())
    # Generate random perturbation
    h = np.random.rand(n_points)
    total = np.sum(h)
    h *= target_sum / total
    initial_designs.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Optimization function
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints and bounds
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1)] * n_points

    # Run optimizations
    start_time = time.time()
    for design in initial_designs:
        h = design.copy()
        h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

        # Early exit if time is up
        if time.time() - start_time > budget_s - 10:
            break

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
