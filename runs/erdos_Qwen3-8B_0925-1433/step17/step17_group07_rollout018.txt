import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            h[:] = 0.5
        else:
            scale = target_sum / current_sum
            h *= scale
            h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_candidate():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h[:] = 0.5
        else:
            scale = target_sum / current_sum
            h *= scale
            h = np.clip(h, 0, 1)
        return h

    for _ in range(49):
        population.append(generate_candidate())

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    generations = 0

    while time.time() - start_time < budget_s:
        generations += 1
        fitness = [compute_c5(ind) for ind in population]
        ranks = np.argsort(fitness)
        worst_indices = ranks[:25]

        parents = []
        for _ in range(50):
            indices = np.random.choice(len(population), 3, replace=False)
            best = indices[np.argmin([fitness[i] for i in indices])]
            parents.append(population[best])

        offspring = []
