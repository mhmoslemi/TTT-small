import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    initial_time = time.time()

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    for _ in range(5):
        if time.time() - initial_time > budget_s:
            break
        # Local refinement
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and compute_c5(res.x) < best_c5:
            best_h, best_c5 = res.x, compute_c5(res.x)
        # Random restart
        res = minimize(objective, np.random.uniform(0, 1, n_points), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and compute_c5(res.x) < best_c5:
            best_h, best_c5 = res.x, compute_c5(res.x)
        # Perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and compute_c5(res.x) < best_c5:
            best_h, best_c5 = res.x, compute_c5(res.x)

    # Final refinement
    for _ in range(3):
        if time.time() - initial_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
