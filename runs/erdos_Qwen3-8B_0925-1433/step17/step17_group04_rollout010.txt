    h_sine = h_sine * target_sum / np.sum(h_sine)

    best_h = h_initial.copy()
    best_c5 = compute_c5(h_initial, dx)
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            idx_plus = i + lag
            if 0 <= idx_plus < n_points:
                term1 = 1 - h[idx_plus]
            else:
                term1 = 0
            idx_minus = i - lag
            if 0 <= idx_minus < n_points:
                term2 = h[idx_minus]
            else:
                term2 = 0
            grad[i] = term1 - term2
        return grad

    # Optimization for initial_h_values
    res = minimize(obj, h_initial, method='SLSQP',
                   bounds=[(0, 1)] * n_points,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   jac=grad,
                   options={'maxiter': 300, 'ftol': 1e-9})
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    # Optimization for sine pattern
    res = minimize(obj, h_sine, method='SLSQP',
                   bounds=[(0, 1)] * n_points,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   jac=grad,
                   options={'maxiter': 300, 'ftol': 1e-9})
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
