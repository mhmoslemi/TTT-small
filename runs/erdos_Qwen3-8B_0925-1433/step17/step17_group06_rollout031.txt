import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_halton(n):
        def halton_index(i, base):
            f = 0
            while i > 0:
                i, remainder = divmod(i, base)
                f += remainder / (base ** (1 + np.log10(base)))
            return f

        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if np.floor(halton_index(i + 1, 2)) % 2 == 0 else 0.0
        return h

    def generate_regular(n):
        return np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return -np.max(corr) * dx

    def generate_initial_designs(n):
        return [
            (np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)),
            generate_halton(n),
            generate_regular(n),
            np.random.rand(n) * target_sum / (np.random.rand(n).sum())
        ]

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for init in generate_initial_designs(n_points):
        init = np.clip(init, 0, 1)
        init = init * target_sum / np.sum(init)
        res = minimize(
            objective,
            init,
            method='Nelder-Mead',
            bounds=[(0, 1)]*n_points,
            tol=1e-5,
            options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 200}
        )
        if res.success and -res.fun < best_c5:
            best_h = res.x
            best_c5 = -res.fun
            if time.time() - start_time > budget_s:
                break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
