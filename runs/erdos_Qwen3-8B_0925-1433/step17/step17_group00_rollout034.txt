import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    x = np.linspace(0, 2, n_points)

    def generate_pattern():
        if np.random.rand() < 0.6:
            h = 0.5 * (1 + np.sin(np.pi * x))
        elif np.random.rand() < 0.6:
            h = np.random.rand(n_points)
            h = (h - 0.5) * 2
            h = np.clip(h, 0, 1)
        elif np.random.rand() < 0.6:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 0.5 if (i % 3) == 0 else 0.25
        else:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 0.75 if (i % 5) == 0 else 0.2
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    initial_patterns = [generate_pattern() for _ in range(5)]
    best_h = initial_patterns[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for pattern in initial_patterns:
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(pattern + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(10):
        if time.time() > time.time() + budget_s:

        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:

    h_values = np.clip(best_h, 0, 1)
