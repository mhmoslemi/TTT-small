        h_start = design.copy()

        def objective(h):
            h_clipped = np.clip(h, 0, 1)
            sum_clipped = np.sum(h_clipped)
            if sum_clipped == 0:
                return float('inf')
            h_normalized = h_clipped * target_sum / sum_clipped
            corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
            return np.max(corr) * dx

        result = minimize(
            fun=objective,
            x0=h_start,
            method='Nelder-Mead',
            bounds=[(0, 1)] * n_points,
            tol=1e-6,
            options={'xatol': 1e-6, 'maxiter': 200}
        )

        if result.success:
            current_h = result.x
            current_h_clipped = np.clip(current_h, 0, 1)
            sum_clipped = np.sum(current_h_clipped)
            if sum_clipped == 0:
                current_c5 = float('inf')
            else:
                current_h_normalized = current_h_clipped * target_sum / sum_clipped
                corr = np.correlate(current_h_normalized, 1 - current_h_normalized, mode='full')
                current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = current_h_normalized
                best_c5 = current_c5
                if time.time() - start_time > budget_s - 20:
                    break

    best_h_clipped = np.clip(best_h, 0, 1)
    sum_clipped = np.sum(best_h_clipped)
    if sum_clipped == 0:
        best_c5 = float('inf')
    else:
        best_h_normalized = best_h_clipped * target_sum / sum_clipped
        corr = np.correlate(best_h_normalized, 1 - best_h_normalized, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h_normalized, best_c5, n_points)
