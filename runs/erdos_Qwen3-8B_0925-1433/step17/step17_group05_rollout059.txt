import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

        h = initial_h_values + pert
        population.append(project(h))

        h = 0.5 * (1 + np.sin(2 * np.pi * (np.linspace(0, 2, n_points) + phase)))
        population.append(project(h))
    population = np.array(population)

    best_h = initial_h_values
    best_c5 = float('inf')

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        idxs = np.argsort(fitness)
        top_idx = idxs[:20]
        new_pop = []
        for _ in range(50):
            p1, p2 = np.random.choice(top_idx, 2, replace=False)
            c1 = population[p1]
            c2 = population[p2]
            start = np.random.randint(n_points)

            if np.random.rand() < 0.2:

                end_m = np.random.randint(start_m, n_points)
                child[start_m:end_m] *= np.random.uniform(0.5, 1.5)
            new_pop.append(project(child))
