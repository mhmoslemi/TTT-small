import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    population = [h_initial]

    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = h_initial
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    max_gen = 200
    sigma = 0.1
    tournament_size = 3

    for gen in range(max_gen):
        f = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        if min(f) < best_c5:
            best_c5 = min(f)
            best_h = population[np.argmin(f)]
        parents = []
        for _ in range(50):
            cand = np.random.choice(population, size=tournament_size, replace=False)
            parents.append(min(cand, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx))
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1]
            alpha = np.random.rand()
            c1 = p1 * alpha + p2 * (1 - alpha)
            c2 = p2 * alpha + p1 * (1 - alpha)
            offspring.append(c1)
            offspring.append(c2)
        mutated = []
        for h in offspring:
            h_mut = h + np.random.normal(0, sigma, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            cs = np.sum(h_mut)
            if cs == 0:
                h_mut = np.full_like(h_mut, 0.5)
            else:
                h_mut *= target_sum / cs
            mutated.append(h_mut)
        rank = sorted(zip(f, population), key=lambda x: x[0])
        new_pop = rank[:25] + [(f[i], mutated[i]) for i in range(25)]
        population = [x[1] for x in new_pop]

    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    return (h_final, best_c5, n_points)
