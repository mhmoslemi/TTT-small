import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    def optimize(start):
        res = minimize(objective, start, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            return res.x
        return start

    for _ in range(3):
        if time.time() > time.time() + budget_s:
            break
        best_h = optimize(best_h)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        if np.random.rand() < 0.5:
            perturb = np.random.normal(0, 0.01, n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            best_h = optimize(perturbed)

    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        best_h = optimize(best_h)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    best_h_clip = np.clip(best_h, 0, 1)
    final_c5 = np.max(np.correlate(best_h_clip, 1 - best_h_clip, mode='full')) * dx
    return (best_h_clip, final_c5, n_points)
