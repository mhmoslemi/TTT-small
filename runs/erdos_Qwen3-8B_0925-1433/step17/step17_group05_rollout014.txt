import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(250):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        k = k_max
        grad = np.zeros(n_points)
        for m in range(n_points):
            idx1 = (m + k) % n_points
            idx2 = (m - k) % n_points
            grad[m] = (1 - best_h[idx1]) - best_h[idx2]

        step_size = 0.1
        h_new = best_h - step_size * grad
        h_new = project(h_new)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h, best_c5 = h_new, new_c5

        if (_ % 10) == 0:
            offset = np.random.randint(0, n_points)
            for i in range(n_points):
                h_pert = h_new * (1 + 0.2 * np.sin(np.pi * (i + offset) / n_points))
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = project(h_pert)
                new_c5_pert = compute_c5(h_pert)
                if new_c5_pert < best_c5:
                    best_h, best_c5 = h_pert, new_c5_pert

    def final_objective(x):
        h = project(x)
        return compute_c5(h)

    result = minimize(final_objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
