import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx

    start_time = time.time()
    max_iter = 1000
    T = max_iter
    lr_0 = 0.05
    top_lags = 5
    update_freq = 10

    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        sorted_indices = np.argsort(corr)[::-1]
        top_indices = sorted_indices[:top_lags]
        k_values = [idx - (n_points - 1) for idx in top_indices]
        grad = np.zeros(n_points)
        for k in k_values:
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] += 1.0 - h[j]
        learning_rate = lr_0 * (1.0 + np.cos(t / T)) / 2.0
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if t % update_freq == 0 and t != 0:
            lr_0 *= 0.999

    return (np.clip(best_h, 0, 1), best_c5, n_points)
