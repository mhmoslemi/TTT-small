import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_configs = [np.array(initial_h_values, dtype=float)]

    initial_configs.append(sin_pattern)
    alt_pattern = np.zeros(n_points)
    for i in range(n_points):

        return np.sum(x) - target_sum

                term2 = h[i - lag]
