            overlaps.append(overlap)
        return overlaps

    def objective(x, worst_lag):
        h_opt = x
        shifted = np.roll(h_opt, -worst_lag)
        return np.sum(h_opt * (1 - shifted))

    corr_values = compute_overlaps(h)
    best_c5 = max(corr_values)

    start_time = time.time()
    for iter in range(300):
        if time.time() - start_time > budget_s:
            break
        corr_values = compute_overlaps(best_h)
        worst_lag = corr_values.index(max(corr_values))
        res = minimize(lambda x: objective(x, worst_lag), best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success:
            h_new = res.x
            h_new_projected = project(h_new)
            current_overlap = compute_overlaps(h_new_projected)[worst_lag]
            if current_overlap < best_c5:
                best_h = h_new_projected
                best_c5 = current_overlap
        if iter % 10 == 0:
            perturbation = np.random.normal(0, 0.15, n_points)
            h_pert = best_h + perturbation
            h_pert = project(h_pert)
            res_pert = minimize(lambda x: objective(x, worst_lag), h_pert, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
            if res_pert.success:
                h_new_pert = res_pert.x
                h_new_pert_proj = project(h_new_pert)
                current_overlap_pert = compute_overlaps(h_new_pert_proj)[worst_lag]
                if current_overlap_pert < best_c5:
                    best_h = h_new_pert_proj
                    best_c5 = current_overlap_pert

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
