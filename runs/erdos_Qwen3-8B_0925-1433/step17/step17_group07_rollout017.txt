import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(objective, bounds, popsize=50, strategy='best1bin', maxiter=100, tol=1e-6)
    best_h = result.x

    def compute_c5(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k = np.argmax(corr)
        g = np.zeros_like(x)
        for i in range(len(x)):
            j = (i + k) % len(x)
            g[i] = (1 - x[j]) * dx
            g[i] -= x[(i - k) % len(x)] * dx
        return g

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    res = minimize(compute_c5, best_h, method='SLSQP', jac=grad, bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   tol=1e-7, options={'maxiter': 100})
    best_h = res.x
    best_c5 = compute_c5(best_h)

    for _ in range(5):
        h_perturb = best_h + np.random.normal(0, 0.1, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        c5 = compute_c5(h_perturb)
        if c5 < best_c5:
            best_h = h_perturb
            best_c5 = c5

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
