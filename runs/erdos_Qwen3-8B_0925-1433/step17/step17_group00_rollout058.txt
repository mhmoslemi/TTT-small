import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        h_val = x * target_sum / np.sum(x) if np.sum(x) > 0 else x
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0., 1.) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def generate_patterns():
        patterns = [initial_h_values.copy()]
        m = n_points // 5
        block = np.zeros(n_points)
        for i in range(n_points):
            if (i // m) % 2 == 0:
                block[i] = 1.
        block = block * target_sum / np.sum(block)
        patterns.append(block)

        x = np.linspace(0, 2, n_points)
        y = np.exp(-((x - 1.)**2)/(0.5**2))
        y = y / np.max(y) if np.max(y) > 0 else y
        y = y * target_sum / np.sum(y)
        patterns.append(y)

        random_binary = np.random.rand(n_points) < 0.5
        random_binary = random_binary * target_sum / np.sum(random_binary)
        patterns.append(random_binary)

        sparse = np.zeros(n_points)
        for i in range(n_points):
            if i % (n_points // 10) == 0:
                sparse[i] = 1.
        sparse = sparse * target_sum / np.sum(sparse)
        patterns.append(sparse)
        return patterns

    start_time = time.time()
    patterns = generate_patterns()
    for pattern in patterns:
        initial_guess = np.array(pattern, dtype=float)
        initial_guess = initial_guess * target_sum / np.sum(initial_guess)
        result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        if time.time() - start_time > budget_s - 10:
            break

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        perturb = np.random.normal(0., 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0., 1.)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return (np.clip(best_h, 0., 1.), best_c5, n_points)
