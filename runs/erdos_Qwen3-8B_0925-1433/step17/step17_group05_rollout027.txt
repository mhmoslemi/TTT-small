import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        return 1.0 / obj(h)

    def mutate(h, scale_strength=0.5):
        idx = np.random.choice(n_points, 5, replace=False)
        h[idx] *= np.random.uniform(0.5, 1.5)
        return project(h)

    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return project(child)

    def generate_population():
        pop = []
        base = project(initial_h_values)
        for _ in range(30):
            pop.append(mutate(base, 0.3))
        for _ in range(20):
            theta = np.random.uniform(0, 2*np.pi)
            freq = np.random.uniform(0.1, 0.5)
            h = 0.5 + 0.5 * np.cos(2*np.pi*freq*np.arange(n_points) + theta)
            pop.append(project(h))
        return [project(np.random.uniform(0, 1, n_points)) for _ in range(10)]

    population = generate_population()
    best_h = project(initial_h_values)
    best_c5 = obj(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness_scores = [fitness(h) for h in population]
        top_indices = np.argsort(fitness_scores)[-10:]
        parents = [population[i] for i in top_indices]
        new_population = []
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            new_population.append(mutate(child))
        population = new_population + [h for h in population if h is not None]
        current_best = min(population, key=obj)
        if obj(current_best) < best_c5:
            best_h, best_c5 = current_best, obj(current_best)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = project(np.random.uniform(0, 1, n_points))
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
