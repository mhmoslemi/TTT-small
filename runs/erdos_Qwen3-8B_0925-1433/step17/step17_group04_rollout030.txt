        if time.time() - start_time > budget_s:
            break
        optimized = optimize_pattern(p)
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_h, best_c5 = optimized, current_c5

    # Perturb and refine
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        optimized = optimize_pattern(perturbed)
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_h, best_c5 = optimized, current_c5

    # Final refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        optimized = optimize_pattern(perturbed)
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_h, best_c5 = optimized, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
