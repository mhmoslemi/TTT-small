import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_designs = [np.array(initial_h_values, dtype=float)]
    random_h = np.random.rand(n_points)
    random_h = random_h * target_sum / np.sum(random_h)
    initial_designs.append(random_h)

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for init in initial_designs:
        h = init.copy()
        h = h * target_sum / np.sum(h)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0., 1.) for _ in range(n_points)]
        try:
            res = minimize(lambda x: compute_c5(x), h, method='COBYLA', bounds=bounds, constraints=constraints, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0., 1.)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
