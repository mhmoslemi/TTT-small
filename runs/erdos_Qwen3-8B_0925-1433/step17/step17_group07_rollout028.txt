from scipy.optimize import minimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_h(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = scale_h(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h_vec):
        h = scale_h(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_c5(x)

    def gradient(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    def optimize_with_bounds(h_start):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(objective, h_start, method='SLSQP', jac=gradient, bounds=bounds, 
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                       tol=1e-6, options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        if res.success:
            return res.x
        return h_start

    def blend_crossover(parent1, parent2, alpha=0.5):
        child = (1 - alpha) * parent1 + alpha * parent2
        return child

    def mutate(h, sigma=0.1):
        noise = np.random.normal(0, sigma, h.shape)
        h = h + noise
        return scale_h(h)

    population = [initial_h]

        h = scale_h(h)
        population.append(h)
