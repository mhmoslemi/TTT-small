import numpy as np
from scipy.optimize import minimize
import time
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def scale(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return h * target_sum / s

    def mutate(h, sigma=0.05):
        noise = np.random.normal(0, sigma, len(h))
        return scale(np.clip(h + noise, 0, 1))

    def blend_crossover(p1, p2):
        alpha = np.random.uniform(0.5, 1.5)
        return scale(np.clip(p1 * np.random.rand() + p2 * (1 - np.random.rand()), 0, 1))

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = scale(initial_h)
    population = [initial_h.copy()]
    for _ in range(49):
        rand = np.random.rand(n_points)
        rand = scale(rand)
        population.append(rand)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        current_best = population[np.argmin(fitness)]
        current_c5 = min(fitness)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5

        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 3, replace=False)
            candidates = [population[i] for i in idxs]
            fit = [fitness[i] for i in idxs]
            winner = candidates[np.argmin(fit)]
            parents.append(winner)

        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                child1 = blend_crossover(parents[i], parents[i+1])
                child2 = blend_crossover(parents[i+1], parents[i])
                offspring.extend([child1, child2])

        for h in offspring:
            if np.random.random() < 0.1:
                h = mutate(h, sigma=0.1)

        combined = list(zip(population, fitness))
        combined.extend([(h, compute_c5(h)) for h in offspring])
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:50]]

        top = population[:10]
        for h in top:
            bounds = Bounds([0]*n_points, [1]*n_points)
            res = minimize(lambda x: compute_c5(x), h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res.success:
                population[population.index(h)] = res.x

    h_values = scale(best_h)
    return (h_values, best_c5, n_points)
