from scipy.optimize import minimize

            # Evenly spaced 1s
            h[::2] = 1.0  # Start with basic even spacing
        elif pattern_type == 'random':
            h = np.random.rand(n_points)
        elif pattern_type == 'geometric':
            h = np.zeros(n_points)
            idx = np.ceil(np.linspace(0, n_points-1, 5)).astype(int)
            h[idx] = 1.0
        elif pattern_type == 'sin':
            h = 0.5 * (1 + np.sin(np.linspace(0, 2*np.pi, n_points)))
        elif pattern_type == 'binary':
            h = np.random.rand(n_points) > 0.5
            h = h.astype(float)
        h = h * target_sum / np.sum(h)
        return h
    
    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': constraint}

    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def optimize_pattern(start_h):
        res = minimize(objective, start_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, maxiter=200)

        return start_h
