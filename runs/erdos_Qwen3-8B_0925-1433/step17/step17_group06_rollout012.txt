```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    initial_designs = [
        np.array(initial_h_values, dtype=float),
        np.zeros(n_points),
        np.random.rand(n_points) < 0.5,
        np.array(initial_h_values, dtype=float) + np.random.normal(0, 0.1, n_points)
    ]
    initial_designs = [d.astype(float) for d in initial_designs]
    for i in range(len(initial_designs)):
        initial_designs[i] = initial_designs[i] * target_sum / np.sum(initial_designs[i])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for design in initial_designs:
        def objective(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        res = minimize(objective, design, method='Nelder-Mead', bounds=[(0,1)]*n_points, options={'maxiter': 200, 'xatol': 1e-6})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    h_final = np.clip(best_h, 0, 1) * target_sum / np.sum(np.clip(best_h, 0, 1))
    return (h_final, best_c5, n_points)
```
