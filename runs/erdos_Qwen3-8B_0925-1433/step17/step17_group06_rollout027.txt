def halton(n, base):
    # Generate the nth term of the Halton sequence in base 'base'
    # Assumes n >= 0
    n = int(n)
    if n == 0:
        return 0.0
    digits = []
    while n > 0:
        digits.append(n % base)
        n = n // base
    # Reverse the digits
    digits = digits[::-1]
    # Convert to fraction
    val = 0.0
    for d in digits:
        val = val / base + d / base
    return val
