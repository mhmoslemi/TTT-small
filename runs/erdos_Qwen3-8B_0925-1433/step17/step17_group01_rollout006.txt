                        parent[i] += np.random.uniform(-0.02, 0.02)
                    else:
                        parent[j] += np.random.uniform(-0.02, 0.02)
                parent = np.clip(parent, 0, 1)
                parent = parent * target_sum / np.sum(parent)
            new_pop.append(parent)
        population = new_pop
        for c5, cand in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand.copy()
        mut_rate *= decay

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
