import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_designs = []
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    initial_designs.append(h_initial)

    # Add a random perturbation
    h_perturbed = h_initial + np.random.normal(0, 0.1, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
    initial_designs.append(h_perturbed)

    # Add a random binary vector
    h_binary = np.random.rand(n_points)
    h_binary = np.round(h_binary)
    h_binary = h_binary * target_sum / np.sum(h_binary)
    initial_designs.append(h_binary)

    # Add a regular grid if possible
    if n_points % 2 == 0:
        h_grid = np.zeros(n_points)
        h_grid[::2] = 1.0
        h_grid = h_grid * target_sum / np.sum(h_grid)
        initial_designs.append(h_grid)

    best_c5 = float('inf')
    best_h = initial_designs[0]

    start_time = time.time()
    for h0 in initial_designs:
        res = minimize(objective, h0, method='Nelder-Mead', options={'maxiter': 200, 'xatol': 1e-6})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
