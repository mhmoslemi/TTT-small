import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    h = project(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = project(x)
        return evaluate(h)

    start_time = time.time()
    population = []
    for _ in range(50):
        if 'initial_h_values' in globals():
            p = initial_h_values.copy()
        else:
            p = np.random.rand(n_points)
        p = p * (n_points / 2) / np.sum(p)
        p = project(p)
        population.append(p)

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(p) for p in population])
        elite = population[np.argsort(fitness)[:10]]
        new_pop = []
        for _ in range(50):
            parent1, parent2 = np.random.choice(elite, 2, replace=False)
            start = np.random.randint(0, n_points)
            length = np.random.randint(1, n_points - start)
            child = np.concatenate([parent1[:start], parent2[start:start+length], parent1[start+length:]])
            if np.random.rand() < 0.2:
                indices = np.random.choice(n_points, 5, replace=False)
                child[indices] *= np.random.uniform(0.5, 1.5)
                child = project(child)
            new_pop.append(project(child))
        population = new_pop
        for p in population:
            if evaluate(p) < best_c5:
                best_h = p.copy()
                best_c5 = evaluate(p)

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        h_pert = best_h + np.random.normal(0, 0.05, n_points)
        h_pert = project(h_pert)
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
