import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    else:
        n_points = 45
        initial_h_values = np.zeros(n_points)
        initial_h_values[:n_points//2] = 1
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum if h_sum != 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx

    def generate_initials():
        designs = []
        h0 = np.array(initial_h_values, dtype=float)
        h0 = h0 * target_sum / np.sum(h0)
        designs.append(h0.copy())
        h_reg = np.zeros(n_points)
        h_reg[::2] = 1
        h_reg = h_reg * target_sum / np.sum(h_reg)
        designs.append(h_reg.copy())

        idxs = np.random.choice(n_points, size=n_points//2, replace=False)
        h_rand[idxs] = 1
        h_rand = h_rand * target_sum / np.sum(h_rand)
        designs.append(h_rand.copy())
        h_even = np.zeros(n_points)
        h_even[::3] = 1
        h_even = h_even * target_sum / np.sum(h_even)
        designs.append(h_even.copy())
        return designs

    best_h = np.zeros(n_points)
    best_c5 = float('inf')

    designs = generate_initials()
    for h_initial in designs:
        res = minimize(obj, h_initial, method='Nelder-Mead', options={'maxiter': 150, 'xatol': 1e-6})
