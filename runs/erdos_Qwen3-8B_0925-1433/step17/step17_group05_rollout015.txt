import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * (target_sum / np.sum(h))
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        total = np.sum(h_vec)
        if total == 0:
            return np.full_like(h_vec, 0.5)
        return h_vec * (target_sum / total)

    start_time = time.time()
    max_iterations = 300
    perturbation_frequency = 10

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        def obj_for_k(h_vec):
            return np.dot(h_vec, 1 - np.roll(h_vec, lag)) / n_points
        def grad_for_k(h_vec):
            return 1 - np.roll(h_vec, lag) - np.roll(h_vec, -lag)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        try:
            res = minimize(obj_for_k, h, method='SLSQP', jac=grad_for_k, bounds=bounds, constraints=cons, tol=1e-4)
            new_h = res.x

            if current_c5 < best_c5:

            h = project(h + perturbation)

    h_values = np.clip(h_values, 0, 1)
