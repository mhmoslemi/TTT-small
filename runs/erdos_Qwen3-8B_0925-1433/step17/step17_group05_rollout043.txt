import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = project(h_initial)
    population = [h_initial]

    for _ in range(49):
        if np.random.rand() < 0.7:
            pert = np.random.normal(0, 0.1, n_points)
            h = h_initial + pert
        else:
            freq = np.random.uniform(0.5, 2.0)
            phase = np.random.uniform(0, 2 * np.pi)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
        population.append(project(h))

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_initial
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        idxs = np.argsort([evaluate(p) for p in population])
        parents = population[idxs[:5]]
        new_pop = []
        for _ in range(len(population) // 2):
            a, b = np.random.choice(parents, 2, replace=False)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([a[:split], b[split:]])
            child = project(child)
            if np.random.rand() < 0.2:
                idxs = np.random.choice(n_points, 5, replace=False)
                child[idxs] *= np.random.uniform(0.5, 1.5)
                child = project(child)
            new_pop.append(child)
        population = population[idxs[len(population)//2:]] + new_pop
        curr = evaluate(population[0])
        if curr < best_c5:
            best_h, best_c5 = population[0], curr

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = project(best_h)
    return (best_h, best_c5, n_points)
