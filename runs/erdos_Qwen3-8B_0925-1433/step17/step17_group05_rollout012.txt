        if res.success:
            new_h = res.x
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        if i % 10 == 0:
            perturbation = np.random.uniform(-0.2, 0.2, n_points)
            best_h = project(best_h + perturbation)

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
