import numpy as np
from scipy.stats.qmc import Halton
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h_values.copy()
    best_c5 = 0.381
    start_time = time.time()
    
    def obj(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def generate_halton(n):
        sample = Halton(d=1)
        points = sample.random(n)
        points *= 2
        h = np.zeros(n_points)
        for p in points:
            i = np.round(p / dx).astype(int)
            if 0 <= i < n_points:
                h[i] = 1
        h = h * target_sum / np.sum(h)
        return h
    
    def generate_grid():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 1
        h = h * target_sum / np.sum(h)
        return h
    
    def generate_random():
        h = initial_h_values + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    initial_designs = [
        generate_halton(n_points // 2),
        generate_grid(),
        generate_random(),
    ]
    
    for h_initial in initial_designs:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h_initial, method='Nelder-Mead', bounds=[(0, 1)]*n_points, tol=1e-6, options={'maxiter': 200})
        if res.success:
            current_c5 = -res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
