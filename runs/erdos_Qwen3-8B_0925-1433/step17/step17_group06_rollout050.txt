    h = np.zeros(n_points)
    h[:target] = 1
    h = h * target_sum / np.sum(h)

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        res = minimize(obj, best_h, method='Nelder-Mead', options={'xatol': 1e-6, 'fatol': 1e-6})
        if res.success:
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
            if np.isclose(res.fun, best_c5, atol=1e-6):
                break

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
