            noise = np.random.normal(0, 0.1, n_points)
            child_mut = np.clip(child + noise, 0, 1)
            sum_mut = np.sum(child_mut)
            child_mut = (child_mut * target_sum / sum_mut) if sum_mut != 0 else np.full_like(child_mut, 0.5)
            mutated.append(child_mut)

        combined = population + mutated
        combined_fitness = [compute_c5(h) for h in combined]
        sorted_indices = np.argsort(combined_fitness)
        population = [combined[i] for i in sorted_indices[:population_size]]

    best_h_values = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h_values)
    best_h_values = (best_h_values * target_sum / sum_best) if sum_best != 0 else np.full_like(best_h_values, 0.5)
    return (best_h_values, best_c5, n_points)
