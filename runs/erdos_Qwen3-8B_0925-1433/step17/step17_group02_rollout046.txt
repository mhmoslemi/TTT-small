import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    h = np.array(initial_h_values, dtype=float)

    learning_rate = 0.1
    max_iterations = 200
    decay_rate = 0.998
    perturb_freq = 20
    perturb_std = 0.015
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            i_minus_k = (i - k) % n_points
            gradient[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        h -= learning_rate * gradient

        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
