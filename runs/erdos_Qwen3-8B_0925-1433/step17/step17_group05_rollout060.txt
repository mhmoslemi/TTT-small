from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = project(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_index = np.argmax(corr)
        def objective(x):
            h_opt = project(x)
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            return corr_opt[k_index] * dx
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success:
            h_candidate = res.x

            current_c5 = compute_c5(h_candidate)

                best_c5 = current_c5
                best_h = h_candidate
