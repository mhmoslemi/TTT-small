        0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points),
    ]
    for _ in range(2):
        initial_configurations.append(np.random.rand(n_points))

    start_time = time.time()

    for config in initial_configurations:
        h_scaled = config * target_sum / np.sum(config)
        h_scaled = np.clip(h_scaled, 0, 1)
        try:
            res = minimize(
                objective, h_scaled,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                jac=gradient,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h.copy()
        idx = np.random.randint(n_points)
        perturbed[idx] = np.random.rand()
        perturbed = perturbed * target_sum / np.sum(perturbed)
        perturbed = np.clip(perturbed, 0, 1)
        try:
            res = minimize(
                objective, perturbed,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                jac=gradient,
                options={'ftol': 1e-8, 'maxiter': 50}
            )
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
