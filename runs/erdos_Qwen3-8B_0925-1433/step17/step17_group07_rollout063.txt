import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_c5(x)
    
    def optimize_with_bounds(h_start):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6, options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        return res.x if res.success else h_start
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        h_candidate = optimize_with_bounds(best_h)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
