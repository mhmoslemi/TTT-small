        res = minimize(obj, best_h, method='Nelder-Mead', bounds=[(0,1)]*n_points, tol=1e-6, options={'xatol': 1e-4, 'maxiter': 20})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
