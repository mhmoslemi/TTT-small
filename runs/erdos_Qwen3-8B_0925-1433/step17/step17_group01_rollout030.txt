import numpy as np
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population_size = 50
    population = []
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    for _ in range(population_size):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        h = h_initial + perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_h = h_initial.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    max_generations = 80
    elite_fraction = 0.2
    mutation_rate = 0.2
    tournament_size = 5
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates
        c5_values = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5_values.append(np.max(corr) * dx)
        
        # Track best
        best_index = np.argmin(c5_values)
        if c5_values[best_index] < best_c5:
            best_h = population[best_index].copy()
            best_c5 = c5_values[best_index]
        
        # Elites
        elite_count = int(elite_fraction * population_size)
        elite_indices = np.argsort(c5_values)[:elite_count]
        elites = [population[i] for i in elite_indices]
        
        # Generate new population
        new_population = elites[:]
        for _ in range(population_size - elite_count):
            # Tournament selection
            tournament = np.random.choice(range(population_size), tournament_size, replace=False)
            vals = [c5_values[i] for i in tournament]
            winner = tournament[np.argmin(vals)]
            parent1 = population[winner].copy()
            
            # Second parent
            tournament2 = np.random.choice(range(population_size), tournament_size, replace=False)
            vals2 = [c5_values[i] for i in tournament2]
            winner2 = tournament2[np.argmin(vals2)]
            parent2 = population[winner2].copy()
            
            # Blend crossover
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            
            # Directional mutation
            corr = np.correlate(child, 1 - child, mode='full')
            k_max = np.argmax(corr)
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    j = (i + k_max) % n_points
                    if np.random.rand() < 0.5:
                        child[i] += np.random.uniform(-0.05, 0.05)
                    else:
                        child[j] += np.random.uniform(-0.05, 0.05)
                    child = np.clip(child, 0, 1)
                    child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        population = new_population
    
    # Final adjustments
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h, final_c5, n_points)
