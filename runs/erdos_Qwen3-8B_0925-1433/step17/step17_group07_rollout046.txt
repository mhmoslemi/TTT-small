import numpy as np
from scipy.optimize import differential_evolution, Bounds
from scipy.optimize import minimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h_initial.copy()
    best_c5_val = compute_c5(best_h)

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    try:
        result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=100, popsize=20,
            tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed
        )
        best_h = result.x
        best_c5_val = objective(best_h)
    except:
        pass

    def objective_slsqp(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_slsqp(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = (i + k_max) % n_points
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % n_points
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    bounds_slsqp = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    if time.time() - start_time < budget_s:
        res = minimize(
            objective_slsqp, best_h, method='SLSQP', jac=grad_slsqp,
            bounds=bounds_slsqp, constraints=cons, tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        best_h = res.x
        best_c5_val = objective_slsqp(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, best_c5_val, n_points)
