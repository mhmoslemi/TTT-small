import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)

    best_c5 = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
    best_h = h_scaled.copy()

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        x_scaled = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=50, maxiter=100,
        tol=1e-6, mutation=(0.5, 1), recombination=0.9, callback=lambda x: None,
        disp=False
    )
    best_h = result_de.x
    best_c5 = objective(best_h)

    def objective_slsqp(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient_slsqp(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):

            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)

            grad[i] = term1 - term2

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_slsqp = Bounds([0.0]*n_points, [1.0]*n_points)
    result_slsqp = minimize(
        objective_slsqp, best_h, method='SLSQP', jac=gradient_slsqp,
        bounds=bounds_slsqp, constraints=cons, tol=1e-7,

    best_h = result_slsqp.x

    start_time = time.time()
