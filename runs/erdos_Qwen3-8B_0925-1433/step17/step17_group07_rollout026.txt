                noise = np.random.normal(0, 0.1, size=h.shape)
                h_mut = np.clip(h + noise, 0, 1)
                sum_h = np.sum(h_mut)
                if sum_h == 0:
                    h_mut = np.full_like(h_mut, 0.5)
                else:
                    h_mut = h_mut * target_sum / sum_h
                mutated_offspring.append(h_mut)
            else:
                mutated_offspring.append(h)

        combined = population + mutated_offspring
        combined_fitness = [compute_c5(h) for h in combined]
        sorted_indices = np.argsort(combined_fitness)
        new_population = [combined[i] for i in sorted_indices[:50]]
        population = new_population

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
