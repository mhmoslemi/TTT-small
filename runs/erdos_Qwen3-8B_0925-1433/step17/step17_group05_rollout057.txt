import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    restarts = 5
    for _ in range(restarts):
        if time.time() - start_time > budget_s:
            break
        # Structured perturbation: invert a random segment
        start = np.random.randint(0, n_points - 5)
        end = start + np.random.randint(3, 6)
        perturbed_h = best_h.copy()
        perturbed_h[start:end] = 1 - perturbed_h[start:end]
        perturbed_h = project(perturbed_h)
        # Run optimization
        res = minimize(obj, perturbed_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-5, maxiter=100)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    # Ensure valid final projection
    final_h = project(best_h)
    return (final_h, best_c5, n_points)
