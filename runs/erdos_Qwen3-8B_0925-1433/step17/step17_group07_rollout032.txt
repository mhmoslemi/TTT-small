import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            term1 = (1 - h[j]) * dx
            i2 = (i - k_max) % len(h)
            term2 = h[i2] * dx
            grad[i] = term1 - term2
        return grad

    def optimize_with_bounds(h_start):
        bounds_slsqp = Bounds([0.0]*n_points, [1.0]*n_points)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(compute_c5, h_start, method='SLSQP', jac=gradient, bounds=bounds_slsqp, 
                       constraints=constraints, tol=1e-7, options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        return res.x

    start_time = time.time()

    result_de = differential_evolution(objective, bounds, strategy='best1bin', 
                                      popsize=20, maxiter=50, tol=1e-6, 
                                      disp=False, workers=-1, x0=h_initial)
    best_de = result_de.x

    best_h = best_de
    best_c5 = compute_c5(best_h)

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.uniform(-0.1, 0.1, size=n_points)
        h_candidate = np.clip(perturbed, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            print(f"New best C5: {best_c5}")

    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    return (final_h, best_c5, n_points)
