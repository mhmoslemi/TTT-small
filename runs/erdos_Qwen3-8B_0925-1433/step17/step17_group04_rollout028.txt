    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
        n_points = len(initial_h_values)
    else:
        initial_h = np.random.rand(45)
        n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population with initial solution
    POP_SIZE = 50
    population = [initial_h.copy()]
    for _ in range(POP_SIZE - 1):
        rand = np.random.rand(n_points)
        rand = rand * target_sum / np.sum(rand)
        population.append(rand)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = evaluate(population[0])

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(h) for h in population]
        current_best_idx = np.argmin(fitness)
        if evaluate(population[current_best_idx]) < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = evaluate(population[current_best_idx])

        # Tournament selection
        parents = []
        for _ in range(POP_SIZE // 2):
            idx1, idx2 = np.random.choice(POP_SIZE, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parents.append(population[idx1])
            else:
                parents.append(population[idx2])

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            pt = np.random.randint(1, n_points-1)
            child1 = np.concatenate([p1[:pt], p2[pt:]])
            child2 = np.concatenate([p2[:pt], p1[pt:]])
            offspring.append(child1)
            offspring.append(child2)

        # Mutation
        for j in range(len(offspring)):
            child = offspring[j].copy()
            idx = np.random.randint(n_points)
            perturbation = np.random.uniform(-0.1, 0.1)
            child[idx] += perturbation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring[j] = child

        # Replace population
        population = parents + offspring
        population = population[:POP_SIZE]

    final_candidate = best_h
    final_c5 = evaluate(final_candidate)
    return (np.clip(final_candidate, 0, 1), final_c5, n_points)
