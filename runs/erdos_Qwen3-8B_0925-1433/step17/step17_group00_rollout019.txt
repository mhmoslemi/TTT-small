import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_block_pattern(n):
        block_size = n // 5
        p = np.zeros(n)
        for i in range(5):
            if i % 2: p[i*block_size:(i+1)*block_size] = 1.0
        return p

    def generate_sine_pattern(n):
        x = np.linspace(0, 2, n)
        sin_wave = 0.5 * (1 + np.sin(2 * np.pi * (x - 1.0)/2.0) + np.sin(2 * np.pi * (x - 1.0)/2.0))
        return np.clip(sin_wave, 0, 1)

    def generate_random(n):
        return np.random.uniform(0, 1, n)

    initial_patterns = [
        np.array(initial_h_values, dtype=float),
        generate_block_pattern(n_points),
        generate_sine_pattern(n_points),
        generate_random(n_points)
    ]

    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    bounds = [(0, 1)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for pattern in initial_patterns:
        pattern = pattern * target_sum / np.sum(pattern)
        res = minimize(objective, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
