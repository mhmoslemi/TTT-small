from scipy.optimize import minimize, Bounds
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)

    # Initialize GP
    kernel = C(1.0, (1e-2, 1e2)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)
    gp.fit(h_samples, c5_values)

    best_h = h.copy()
    best_c5 = c5_values[0]

    for _ in range(300):

        def ucb_func(x):

        # Maximize UCB function
        res = minimize(lambda x: -ucb_func(x), x0=best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)

    # Local refinement with SLSQP
    def objective(x):
