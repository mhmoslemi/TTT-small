import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total <= 1e-10:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        start_idx = np.random.randint(0, n_points - 5)
        end_idx = start_idx + 5
        perturbed_h = h.copy()
        perturbed_h[start_idx:end_idx] = 1 - perturbed_h[start_idx:end_idx]
        perturbed_h = project(perturbed_h)
        res = minimize(obj, perturbed_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6, maxiter=100)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
