    decay_rate = 0.9995

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m_idx = np.argsort(corr)[::-1][:top_lag_count]
        m = m_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for lag in m:
            for i in range(n_points):
                j = i + int(lag)
                if 0 <= j < n_points:
                    gradient[i] += (1 - h[j])
        if np.max(gradient) > 1e-8:
            gradient /= np.max(gradient)
        lr = learning_rate_initial / (1 + iteration)
        delta = np.array([lr * gradient[i] for i in range(n_points)])
        h = h - delta
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            scale = target_sum / sum_h
        else:
            scale = 1.0
        h = h * scale
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if iteration % 10 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            top_lags = np.argsort(corr)[::-1][:5]
            top_lags = top_lags - (n_points - 1)
            def objective(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                max_val = np.max(corr[top_lags])
                return max_val * dx
            res = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
            if res.success:
                h = res.x
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                if sum_h > 0:
                    scale = target_sum / sum_h
                else:
                    scale = 1.0
                h = h * scale
                current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
    return (np.clip(best_h, 0, 1), best_c5, n_points)
