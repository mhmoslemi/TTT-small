        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        if time.time() - start_time > budget_s:
            break

    return (np.clip(best_h, 0, 1), best_c5, n_points)
