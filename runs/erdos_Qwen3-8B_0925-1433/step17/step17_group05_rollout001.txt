import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    restarts = 5
    for _ in range(restarts):
        if time.time() - start_time > budget_s:
            break
        # Generate structured perturbation: flip a segment
        seg_len = np.random.randint(3, n_points // 3)
        start_idx = np.random.randint(0, n_points - seg_len)
        pert = np.zeros(n_points)
        pert[start_idx:start_idx + seg_len] = 1
        perturbed = best_h * (1 - pert) + (1 - best_h) * pert
        perturbed = project(perturbed)
        
        # Optimization
        res = minimize(obj, perturbed, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
