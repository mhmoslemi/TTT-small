from scipy.optimize import minimize, Bounds
import time
import random

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_c5(h_vec):
        h = h_vec * target_sum / np.sum(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb_h(h_vec):
        h_perturb = h_vec + np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        return h_perturb

    def optimize_with_bounds(h_start):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(objective, h_start, method='SLSQP', 
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                       tol=1e-6, options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        if res.success:
            return res.x
        return h_start

    start_time = time.time()

        # Perturb current best and optimize

        # Also optimize from current best

            print(f"New best C5: {best_c5}")
        # Local refinement
        def refined_objective(x):

            h = x * target_sum / np.sum(x)

            k_max = np.argmax(corr)

                i2 = (i - k_max) % len(h)
