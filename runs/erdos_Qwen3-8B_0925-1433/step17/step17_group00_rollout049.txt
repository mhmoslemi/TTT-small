import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    for _ in range(3):
        if time.time() > time.time() + budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        res = minimize(objective, np.random.uniform(0, 1, n_points), method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
