import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        x_normalized = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_normalized, 1 - x_normalized, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    # Run Nelder-Mead with early stopping
    res = minimize(obj, best_h, method='Nelder-Mead', 
                  options={'xatol': 1e-5, 'maxiter': 100, 'maxfev': 500})
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    # Final clipping and normalization
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
