import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_designs = [initial_h_values.copy()]
    random_initial = np.random.rand(n_points)
    random_initial = random_initial * target_sum / np.sum(random_initial)
    initial_designs.append(random_initial)
    regularized = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            regularized[i] = 1
    regularized = regularized * target_sum / np.sum(regularized)
    initial_designs.append(regularized)

    best_h = None
    best_c5 = float('inf')

    def optimize_design(design):
        design = design.copy()
        design = design * target_sum / np.sum(design)
        start = time.time()
        while time.time() - start < budget_s:
            res = minimize(obj, design, method='COBYLA', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x.copy()
                return True
            else:
                design = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        return False

    for design in initial_designs:
        optimize_design(design)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
