import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        total = np.sum(h)
        h = h * target_sum / total if total > 0 else np.zeros_like(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initial designs
    initial_designs = []
    initial_designs.append(np.array(initial_h_values, dtype=float))
    np.random.seed(seed + 1)
    noise = np.random.normal(0, 0.03, n_points)
    initial_designs.append(np.clip(np.array(initial_h_values) + noise, 0, 1) * target_sum / np.sum(np.clip(np.array(initial_h_values) + noise, 0, 1)))
    regular = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            regular[i] = 1
    initial_designs.append(regular * target_sum / np.sum(regular))
    spaced = np.zeros(n_points)
    for i in range(n_points // 2):
        spaced[2*i] = 1
    initial_designs.append(spaced * target_sum / np.sum(spaced))

    best_c5 = float('inf')
    best_h = initial_designs[0].copy()

    start_time = time.time()
    for design in initial_designs:
        try:
            result = minimize(objective, design, method='Nelder-Mead',
                              options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 200})
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x.copy()
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    best_h = best_h * target_sum / total if total > 0 else np.zeros_like(best_h)
    return (best_h, best_c5, n_points)
