import numpy as np
from scipy.optimize import minimize

from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    population = [h_initial]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    population = np.array(population)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()
    
    def rescale(h):
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        h = h * (target_sum / total)
        h = np.clip(h, 0, 1)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def optimize_candidate(h):
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        res = minimize(lambda x: compute_c5(x * target_sum / np.sum(x)), h, method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6, options={'maxiter': 100})
        if res.success:
            return res.x
        return h
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(p) for p in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 3, replace=False)
            best = idxs[np.argmin(fitness[idxs])]
            parents.append(population[best])
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1]

            c1 = rescale(c1)

            c2 = rescale(c2)
            offspring.append(c1)

            if np.random.rand() < 0.1:

                offspring[j] += noise
                offspring[j] = rescale(offspring[j])
        all_candidates = np.vstack([population, offspring])

        all_fitness = np.concatenate([fitness, fitness_off])
