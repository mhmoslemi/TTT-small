import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def halton_index(i, base=2):
        v = i
        r = 0.0
        m = 0
        while v > 0:
            v, remainder = divmod(v, base)
            r += remainder / (base ** (m + 1))
            m += 1
        return r

    def generate_h():
        x = dx * np.arange(n_points)
        h = np.zeros(n_points)
        halton_points = np.array([halton_index(i) * 2.0 for i in range(n_points)])
        sorted_points = np.sort(halton_points)
        num_ones = int(target_sum)
        indices = []
        for i in range(num_ones):
            idx = np.argmin(np.abs(x - sorted_points[i]))
            indices.append(idx)
        for idx in indices:
            h[idx] = 1.0
        h = h * (target_sum / np.sum(h))
        return h

    h = generate_h()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x * (target_sum / np.sum(x))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project(x):
        x = np.clip(x, 0, 1)
        x = x * (target_sum / np.sum(x))
        return x

    start_time = time.time()
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='Nelder-Mead', bounds=[(0, 1)]*n_points, tol=1e-6)
        if result.success:
            h_candidate = project(result.x)
            c5_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if c5_candidate < best_c5:
                best_h, best_c5 = h_candidate, c5_candidate

    h_values = project(best_h)
    return (h_values, best_c5, n_points)
