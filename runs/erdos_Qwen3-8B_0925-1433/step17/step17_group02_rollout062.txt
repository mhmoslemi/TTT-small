import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n_points
    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 500
    start_time = time.time()
    eps = 0.05
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr_initial = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr_initial)
        k = max_index - (n_points - 1)
        shifted_forward = (np.arange(n_points) + k) % n_points
        shifted_backward = (np.arange(n_points) - k) % n_points
        gradient = (1 - h[shifted_forward]) - h[shifted_backward]
        h_update = h + learning_rate * (gradient + eps * (1 - 2 * h))
        h = np.clip(h_update, 0, 1)
        if np.max(h) == np.min(h):
            h = np.full_like(h, target_sum / n_points)
        else:
            h = (h - h.min()) / (h.max() - h.min()) * target_sum
        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * 2.0 / n_points
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        if np.max(perturbed) == np.min(perturbed):
            perturbed = np.full_like(perturbed, target_sum / n_points)
        else:
            perturbed = (perturbed - perturbed.min()) / (perturbed.max() - perturbed.min()) * target_sum

        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
