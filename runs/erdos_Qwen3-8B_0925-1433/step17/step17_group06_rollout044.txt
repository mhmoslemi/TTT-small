import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_points = [np.array(initial_h_values, dtype=float)]
    for _ in range(3):
        perturbed = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.05, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_points.append(perturbed)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    for h_initial in initial_points:
        try:
            res = minimize(obj, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-5)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        except:
            pass
        if time.time() - start_time > budget_s - 5:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
