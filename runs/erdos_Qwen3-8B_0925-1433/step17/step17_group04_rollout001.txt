import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def scale(h):
        return h * target_sum / np.sum(h)

    def obj(x):
        h = scale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = scale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            lag = k_max - (n_points - 1)
            if 0 <= i + lag < n_points:
                term1 = 1 - h[i + lag]
            else:
                term1 = 0
            if 0 <= i - lag < n_points:
                term2 = h[i - lag]
            else:
                term2 = 0
            grad[i] = (term1 - term2)
        return grad

    def coord_descent(h):
        for _ in range(10):
            for i in range(n_points):
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.05), 0, 1)
                h_new = scale(h_new)
                if obj(h_new) < obj(h):
                    h = h_new
        return h

    initial_guesses = [initial_h_values]
    for _ in range(2):
        initial_guesses.append(np.random.rand(n_points))
    for _ in range(2):
        t = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * t)
        initial_guesses.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = obj(best_h)

    for idx, guess in enumerate(initial_guesses):
        h = guess.copy()
        h = scale(h)
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)]*n_points,
                      constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:

        if idx == 0:
