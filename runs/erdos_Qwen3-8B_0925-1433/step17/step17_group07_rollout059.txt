    target_sum = n_points / 2.0

    # Scale initial_h_values to meet the sum constraint
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population = [initial_h.copy()]

    # Generate additional individuals
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Evaluate initial population
    population_fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]

    # GA parameters
    num_generations = 200
    tournament_size = 3
    mutation_rate = 0.1
    crossover_alpha = 0.5

    start_time = time.time()

    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break

        # Evaluate fitness
        current_fitness = [compute_c5(h) for h in population]

        # Tournament selection
        parents = []
        for _ in range(50):
            tournament = np.random.choice(len(population), tournament_size, replace=False)
            winner_idx = tournament[np.argmin([current_fitness[i] for i in tournament])]
            parents.append(population[winner_idx])

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i + 1]
                alpha = np.random.uniform(0, crossover_alpha)
                child1 = parent1 * alpha + parent2 * (1 - alpha)
                child2 = parent2 * alpha + parent1 * (1 - alpha)
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i])

        # Mutation
        mutated_offspring = []
        for h in offspring:
            noise = np.random.normal(0, 0.1, n_points)
            h_mutated = h + noise
            h_mutated = np.clip(h_mutated, 0.0, 1.0)
            current_sum = np.sum(h_mutated)
            if current_sum == 0:
                h_mutated = np.full(n_points, 0.5)
            else:
                h_mutated = h_mutated * target_sum / current_sum
            mutated_offspring.append(h_mutated)

        # Combine and sort
        combined = population + mutated_offspring
        combined_fitness = [compute_c5(h) for h in combined]
        sorted_indices = np.argsort(combined_fitness)
        new_population = [combined[i] for i in sorted_indices[:50]]
        population = new_population

        # Track best solution
        if np.min(population_fitness) < best_c5:
            best_c5 = np.min(population_fitness)
            best_h = population[np.argmin([compute_c5(h) for h in population])]

    # Final adjustments
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
