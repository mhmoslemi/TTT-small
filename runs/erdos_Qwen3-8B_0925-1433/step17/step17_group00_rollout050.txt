        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
