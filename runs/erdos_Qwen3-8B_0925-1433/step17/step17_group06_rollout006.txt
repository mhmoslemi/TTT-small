import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def halton_sequence(n, base=2):
        seq = []
        for i in range(1, n+1):
            d, x = i, 0.0
            k = 0
            while d > 0:
                d, r = divmod(d, base)
                x += r / (base ** (k + 1))
                k += 1
            seq.append(x)
        return np.array(seq)

    def generate_halton_design():
        points = halton_sequence(n_points // 2, 2)
        h = np.zeros(n_points)
        for p in points:
            idx = np.round(p / dx).astype(int)
            if 0 <= idx < n_points:
                h[idx] = 1
        h = h * target_sum / np.sum(h)
        return h

    def generate_regular_design():
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * target_sum / np.sum(h)
        return h

    def generate_perturbed_design():
        h = np.array(initial_h_values, dtype=float)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def generate_random_design():
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, n_points // 2, replace=False)
        h[idx] = 1
        h = h * target_sum / np.sum(h)
        return h

    designs = [generate_halton_design(), generate_regular_design(), generate_perturbed_design(), generate_random_design()]

    for design in designs:
        h_start = design.copy()
        def obj(x):
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x)
            corr = np.correlate(x, 1 - x, mode='full')

        res = minimize(obj, h_start, method='Nelder-Mead', bounds=[(0, 1)] * n_points, tol=1e-5, options={'maxiter': 150})

    best_h = best_h * target_sum / np.sum(best_h)
