import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    def generate_uniform():
        return np.where(np.arange(n_points) % 2 == 0, 1, 0)
    
    def generate_random():
        ones_indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h = np.zeros(n_points)
        h[ones_indices] = 1
        return h
    
    patterns = [np.array(initial_h_values, dtype=float), generate_uniform(), generate_random()]
    
    def objective_avg(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.mean(corr) * dx
    
    def evaluate_c5(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1)] * n_points
    
    for pattern in patterns:
        pattern = pattern * target_sum / np.sum(pattern)
        res = minimize(objective_avg, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = evaluate_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective_avg, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = evaluate_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
