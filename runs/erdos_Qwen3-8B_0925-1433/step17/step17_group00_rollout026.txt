import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_initial = initial_h_values
    else:
        n_points = 100
        h_initial = np.zeros(n_points)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_patterns = []
    if 'initial_h_values' in globals():
        initial_patterns.append(initial_h_values)
    else:
        initial_patterns.append(np.zeros(n_points))
    x = np.linspace(0, 2, n_points)
    initial_patterns.append(0.5 + 0.5 * np.sin(np.pi * (x - 0.5)))
    ip3 = np.zeros(n_points)
    for i in range(n_points):
        ip3[i] = 1.0 if i % 2 == 0 else 0.0
    initial_patterns.append(ip3)
    ip4 = np.random.choice([0, 1], size=n_points)
    initial_patterns.append(ip4)
    ip5 = np.zeros(n_points)
    for i in range(n_points):
        ip5[i] = 0.5 * (np.exp(-((x[i] - 0.3)**2) / 0.08) + np.exp(-((x[i] - 1.7)**2) / 0.08))
    initial_patterns.append(ip5)

    best_h = np.copy(initial_patterns[0])
    best_c5 = objective(best_h)

    for pattern in initial_patterns:
        if 'initial_h_values' in globals():
            pattern = pattern.copy()
        else:
            pattern = np.copy(pattern)
        res = minimize(objective, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    for _ in range(5):
        if time.time() > time.time() + budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(best_h + perturb, 0, 1)
        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
