import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_normalize(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Initial guess with small perturbation
    perturbed = np.array(initial_h_values, dtype=float) + np.random.normal(0, 0.01, n_points)
    perturbed = clip_and_normalize(perturbed)
    best_h = perturbed.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
