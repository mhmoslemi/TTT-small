import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = []

    h_init = np.array(initial_h_values, dtype=float)

    population.append(h_init)

    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    # Add structured patterns
    h_sine = 0.5 + 0.25 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    h_sine = h_sine * target_sum / np.sum(h_sine)

        h_block[i + block_size:i + 2 * block_size] = 0.0
    h_block = h_block * target_sum / np.sum(h_block)
    population.append(h_block)

    population = population[:POP_SIZE]

    best_h = population[0].copy()
