        learning_rate *= decay_rate

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(h + perturb, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        corr_perturbed = np.correlate(perturbed, 1 - perturbed, mode='full')
        current_c5 = np.max(corr_perturbed) * dx
        if current_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
