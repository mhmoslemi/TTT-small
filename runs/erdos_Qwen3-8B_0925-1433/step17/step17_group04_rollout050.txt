        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = -(n_points - 1) + max_idx
        grad = np.zeros_like(x)
        for j in range(n_points):
            j_plus = j + lag
            term1 = 0.0
            if 0 <= j_plus < n_points:
                term1 = 1.0 - h[j_plus]
            j_minus = j - lag
            term2 = 0.0
            if 0 <= j_minus < n_points:
                term2 = h[j_minus]
            grad[j] = (term1 - term2) * dx
        return grad

    learning_rate = 0.3
    momentum = 0.9
    decay_rate = 0.999
    max_iterations = 200
    start_time = time.time()

    h = h * target_sum / np.sum(h)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        m, v = np.zeros_like(grad), np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
            biased_m = m[i] / (1 - momentum ** (_ + 1))
            biased_v = v[i] / (1 - decay_rate ** (_ + 1))
            h[i] = h[i] - learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        random_h = np.random.rand(n_points)
        random_h = random_h * target_sum / np.sum(random_h)
        res = minimize(obj, random_h, method='SLSQP', bounds=[(0, 1)]*n_points, 
                      constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
