import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial configurations
    initial_configs = [initial_h_values.copy()]

    # Add regular grid pattern
    regular_grid = np.zeros(n_points, dtype=float)
    regular_grid[:n_points//2] = 1.0
    initial_configs.append(regular_grid)

    # Add random permutation
    random_perm = np.random.permutation(n_points)
    random_config = np.zeros(n_points, dtype=float)
    random_config[random_perm[:n_points//2]] = 1.0
    initial_configs.append(random_config)

    # Add perturbed versions
    for _ in range(3):
        perturb = np.random.normal(0, 0.1, n_points)
        perturb_h = initial_h_values.copy() + perturb
        perturb_h = np.clip(perturb_h, 0, 1)
        perturb_h = perturb_h * target_sum / np.sum(perturb_h)
        initial_configs.append(perturb_h)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()

    # Define objective function
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Run optimization for each initial configuration
    for config in initial_configs:
        current_h = config.copy()
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        result = minimize(objective, current_h, method='Nelder-Mead',
                          options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        # Early exit if time is up
        if time.time() - start_time > budget_s:
            break

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    # Recalculate C₅ to ensure no errors
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx

    return (best_h, final_c5, n_points)
