import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    h = np.array(initial_h_values, dtype=float)
    h = project(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    for it in range(400):
        if time.time() - start_time > budget_s:
            break
        overlaps = np.correlate(h, 1 - h, mode='full')
        worst_k = np.argmax(overlaps)
        k = worst_k
        def obj(x):
            h_p = project(x)
            corr = np.correlate(h_p, 1 - h_p, mode='full')
            return corr[k] * dx
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success:
            h = res.x
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if it % 10 == 0:
            pert = np.random.normal(0, 0.1, n_points)
            h = project(h + pert)
    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
