import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (target_sum / total)

    def obj(x):
        h = project(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    restarts = 0
    max_restarts = 50  # Adjust based on budget
    
    while restarts < max_restarts and time.time() - start_time < budget_s:
        restarts += 1
        # Create a new perturbed starting point
        h_pert = best_h.copy()
        start_idx = np.random.randint(0, n_points - 3)
        end_idx = start_idx + np.random.randint(2, 5)
        h_pert[start_idx:end_idx] = 1 - h_pert[start_idx:end_idx]
        h_pert = project(h_pert)
        
        # Run optimization
        res = minimize(obj, h_pert, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5
        # Check for early termination if time is up
        if time.time() - start_time >= budget_s:
            break

    h_values = project(best_h)
    h_values = np.clip(h_values, 0, 1)
    return (h_values, best_c5, n_points)
