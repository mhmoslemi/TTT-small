from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    accumulated = np.zeros(n_points)
    learning_rate = 0.05
    max_iterations = 200
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m_idx = np.argmax(corr)
        m = m_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:
                gradient[i] = 1.0 - h[j]
        lr = learning_rate / (np.sqrt(accumulated + 1e-8))
        delta = gradient * lr
        h = h - delta
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            scale = target_sum / sum_h
        else:
            scale = 1.0
        h = h * scale
        accumulated += gradient ** 2
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if iteration % 10 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            m_idx = np.argmax(corr)

            def objective(x):
                x = x * target_sum / np.sum(x)
                corr = np.correlate(x, 1 - x, mode='full')
                max_val = corr[m_idx]

            res = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
