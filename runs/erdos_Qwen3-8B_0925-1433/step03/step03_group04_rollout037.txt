import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        h = np.full(n_points, 0.5)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    λ = 100.0
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x

            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx

            h = res.x
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if np.random.rand() < 0.1:

                h = np.full_like(h, 0.5)

                h = np.clip(h * scale, 0, 1)
        if np.random.rand() < 0.05:

        elif np.random.rand() < 0.05:
            λ *= 0.9
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
