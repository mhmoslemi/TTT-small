    for h in population:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append((-c5, h))
        fitness.sort()
        selected = [h for (_, h) in fitness[:int(0.2 * len(population))]]
        new_pop = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                p1, p2 = selected[i], selected[i + 1]
                split = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
                new_pop.extend([c1, c2])
        while len(new_pop) < 50:
            new_pop.append(np.random.uniform(0, 1, n_points))
        while len(new_pop) > 50:
            new_pop.pop()
        for i in range(len(new_pop)):
            if np.random.rand() < 0.1:
                h_mut = new_pop[i].copy()
                perturb = np.random.uniform(-0.05, 0.05, size=n_points)
                h_mut += perturb
                h_mut = np.clip(h_mut, 0, 1)
                sum_h_mut = np.sum(h_mut)
                if sum_h_mut == 0:
                    h_mut[:] = 0.5
                else:
                    scale = (n_points / 2) / sum_h_mut
                    h_mut[:] = h_mut * scale
                new_pop[i] = h_mut
        population = new_pop
        for (_, h) in fitness:
            c5 = -_.item()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)
