        scale = (n_points / 2) / sum_initial
    h = np.clip(h * scale, 0, 1)
    sum_perturbed = np.sum(h)
    if sum_perturbed == 0:
        scale_perturbed = 0
    else:
        scale_perturbed = (n_points / 2) / sum_perturbed
    h = np.clip(h * scale_perturbed, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    T = 1.0
    cooling_rate = 0.995
    max_steps = 5000
    start_time = time.time()
    for step in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed == 0:
            scale_perturbed = 0
        else:
            scale_perturbed = (n_points / 2) / sum_perturbed
        h_perturbed = np.clip(h_perturbed * scale_perturbed, 0, 1)
        current_c5 = compute_c5(h, dx, n_points)
        candidate_c5 = compute_c5(h_perturbed, dx, n_points)
        delta = candidate_c5 - current_c5
        if candidate_c5 < current_c5:
            h = h_perturbed.copy()
            current_c5 = candidate_c5
        else:
            if np.random.random() < np.exp(-delta / T):
                h = h_perturbed.copy()
                current_c5 = candidate_c5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        T *= cooling_rate
    return (best_h, best_c5, n_points)
