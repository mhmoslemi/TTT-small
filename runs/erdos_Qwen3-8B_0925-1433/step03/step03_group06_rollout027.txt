        x_scaled = np.clip(x * scale, 0, 1)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')

            scale = (n_points / 2) / sum_candidate

            initial_population.append(candidate)

            scale = (n_points / 2) / sum_candidate
            candidate = np.clip(candidate * scale, 0, 1)
            initial_population.append(candidate)

        maxiter=100,
