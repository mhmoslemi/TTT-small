            mutant = np.clip(mutant, 0, 1)
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            if sum_trial == 0:
                trial = np.full_like(trial, 0.5)
            else:
                scale = (n_points / 2) / sum_trial
                trial = np.clip(trial * scale, 0, 1)
            trial_c5 = compute_c5(trial)
            target_c5 = compute_c5(target)
            if trial_c5 < target_c5:
                new_pop.append(trial)
            else:
                new_pop.append(target)
        for idx in range(len(new_pop)):
            vec = new_pop[idx]
            vec = np.clip(vec, 0, 1)
            sum_vec = np.sum(vec)
            if sum_vec == 0:
                vec = np.full_like(vec, 0.5)
            else:
                scale = (n_points / 2) / sum_vec
                vec = np.clip(vec * scale, 0, 1)
            new_pop[idx] = vec
        pop = new_pop
        for vec in pop:
            current_c5 = compute_c5(vec)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = vec.copy()
    return (best_h, best_c5, n_points)
