import numpy as np
import time
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        n_points = 200
        h = np.zeros(n_points)
        m = int(np.sqrt(n_points))
        for j in range(n_points):
            if j % m == m // 2:
                h[j] = 1
            else:
                h[j] = 0
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        for j in range(n_points):
            def objective(x):
                h_new = h.copy()
                h_new[j] = x
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                return np.max(corr) * dx
            res = minimize_scalar(objective, bounds=(0, 1), method='bounded')
            h[j] = res.x
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if np.random.random() < 0.1:

                h = np.clip(h * scale, 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
