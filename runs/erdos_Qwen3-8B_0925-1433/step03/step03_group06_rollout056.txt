import numpy as np
import time
from scipy.optimize import minimize

def _obj(x, n_points):
    x = np.clip(x, 0, 1)
    sum_x = np.sum(x)
    if sum_x == 0:
        return 1e6
    scale = (n_points / 2) / sum_x
    x = np.clip(x * scale, 0, 1)
    corr = np.correlate(x, 1 - x, mode='full')
    return np.max(corr) * (2.0 / n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
