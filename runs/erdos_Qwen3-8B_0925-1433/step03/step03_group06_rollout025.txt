import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (len(h)/2) / sum_h
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    population = [h.copy() for _ in range(10)]
    for i in range(len(population)):
        perturb = np.random.normal(0, 0.1, len(population[i]))
        population[i] += perturb
        population[i] = np.clip(population[i], 0, 1)
        sum_pop = np.sum(population[i])
        scale = (len(population[i])/2) / sum_pop
        population[i] = np.clip(population[i] * scale, 0, 1)
    while time.time() - start_time < budget_s:
        for candidate in population:
            res = minimize(obj, candidate, method='SLSQP', bounds=[(0,1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2}])
            if res.success:
                current_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x.copy()
        perturb = np.random.normal(0, 0.05, len(best_h))
        best_h += perturb
        best_h = np.clip(best_h, 0, 1)
        sum_b = np.sum(best_h)
        scale = (len(best_h)/2) / sum_b
        best_h = np.clip(best_h * scale, 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def obj(x):
    x = np.clip(x, 0, 1)
    sum_x = np.sum(x)
    if sum_x == 0:
        return 1e6
    scale = (len(x)/2) / sum_x
    x = np.clip(x * scale, 0, 1)
    corr = np.correlate(x, 1 - x, mode='full')
    return np.max(corr) * 2.0 / len(x)
