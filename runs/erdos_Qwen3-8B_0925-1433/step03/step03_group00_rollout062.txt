                new_c5 = compute_c5(new_h, dx, n_points)
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h.copy(), new_c5
                h_candidate[:] = new_h.copy()

        # Mutate and replace candidates
        for _ in range(2):
            h_mut = np.random.normal(0, 0.05, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            sum_mut = np.sum(h_mut)
            scale_mut = (n_points / 2) / sum_mut if sum_mut != 0 else 1.0
            h_mut = np.clip(h_mut * scale_mut, 0, 1)
            new_c5 = compute_c5(h_mut, dx, n_points)
            if new_c5 < best_c5:
                best_h, best_c5 = h_mut.copy(), new_c5
            candidates.append(h_mut.copy())
            if len(candidates) > 10:
                candidates.pop(0)
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
