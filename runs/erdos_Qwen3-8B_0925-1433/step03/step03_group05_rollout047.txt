import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start = time.time()
    max_iter = 200
    for i in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            s = np.sum(x)
            if s == 0:
                return 1e6
            x = np.clip(x * (n_points / 2) / s, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_idx] * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
        bounds = [(0, 1)]*n_points
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = np.clip(h * (n_points/2)/sum_h, 0, 1)
            else:
                h = np.ones(n_points)*0.5
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        
        if i % 20 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = np.clip(h * (n_points/2)/sum_h, 0, 1)
            else:
                h = np.ones(n_points)*0.5
        
        if time.time() - start > budget_s:
            break
    
    return (best_h, best_c5, n_points)
