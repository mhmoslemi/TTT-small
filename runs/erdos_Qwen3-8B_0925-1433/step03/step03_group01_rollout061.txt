import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    
    def rescale(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = (n_points / 2) / s
        return np.clip(x * scale, 0, 1)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_pattern(func):
        x = np.linspace(0, 2, n_points)
        h = func(x)
        return rescale(h)
    
    patterns = [
        lambda x: 0.5 + 0.5 * np.sin(np.pi * x),
        lambda x: 0.5 + 0.5 * np.abs(np.sin(np.pi * x)),
        lambda x: 1 - np.exp(-2 * (x - 1)**2),
        lambda x: 0.5 + 0.5 * np.sign(np.sin(np.pi * x))
    ]
    
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = rescale(h)
    else:
        h = generate_pattern(np.random.choice(patterns))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    def obj(x):
        x = rescale(x)
        return compute_c5(x)
    
    def local_optimize(x0):
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1)] * n_points
        result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x
    
    start_time = time.time()
    temp = 1.0
    cooling = 0.995
    iter_count = 0
    max_iter = 200
    
    while temp > 1e-4 and iter_count < max_iter and time.time() - start_time < budget_s:
        h_perturb = best_h + np.random.normal(0, temp, n_points)
        h_perturb = rescale(h_perturb)
        current_c5 = compute_c5(h_perturb)
        if current_c5 < best_c5:
            best_h = h_perturb
            best_c5 = current_c5
        temp *= cooling
        iter_count += 1
    
    best_h = local_optimize(best_h)
    best_c5 = compute_c5(best_h)
    
    while time.time() - start_time < budget_s:
        for _ in range(5):

            if current_c5 < best_c5:
                best_h = h_perturb
                best_c5 = current_c5
        best_h = local_optimize(best_h)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
