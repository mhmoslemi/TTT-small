import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 450
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n_points

    n_particles = 20
    inertia = 0.9
    cognitive = 1.5
    social = 1.5
    v_max = 0.1

    velocities = np.zeros_like(h)
    pbest = h.copy()
    pbest_c5 = compute_c5(pbest)
    gbest = h.copy()
    gbest_c5 = compute_c5(gbest)

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1

        for _ in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities += (inertia * velocities +
                          cognitive * r1 * (pbest - h) +
                          social * r2 * (gbest - h))
            np.clip(velocities, -v_max, v_max, out=velocities)
            h += velocities
            np.clip(h, 0, 1, out=h)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)

            current_c5 = compute_c5(h)
            if current_c5 < pbest_c5:
                pbest = h.copy()
                pbest_c5 = current_c5
                if current_c5 < gbest_c5:
                    gbest = h.copy()
                    gbest_c5 = current_c5

        if iteration % 500 == 0:
            def obj(x):
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)
                if sum_x == 0:
                    return 1e6
                scale = (n_points / 2) / sum_x
                x = np.clip(x * scale, 0, 1)
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx

            res = minimize(obj, gbest, method='SLSQP',
                          bounds=[(0,1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])

            best_c5 = gbest_c5
            best_h = gbest.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
