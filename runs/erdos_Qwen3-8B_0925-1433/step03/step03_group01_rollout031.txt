        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = get_max_corr(best_h) * dx

    def obj(theta):
        a = theta[:10]
        b = theta[10:]
        x = np.linspace(0, 2, n_points)
        h = 0.5 + np.sum([a[n] * np.cos((n+1)*np.pi*x) + b[n] * np.sin((n+1)*np.pi*x) for n in range(10)], axis=0)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1e6
        h = h * (n_points / 2) / sum_h
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    bounds = [(-0.5, 0.5)] * 20
    result = BayesSearchCV(obj, bounds, n_iter=200, random_state=seed).fit({})
    best_theta = result.best_params_.values
    def refine(x):
        a = x[:10]
        b = x[10:]
        x_vals = np.linspace(0, 2, n_points)
        h = 0.5 + np.sum([a[n] * np.cos((n+1)*np.pi*x_vals) + b[n] * np.sin((n+1)*np.pi*x_vals) for n in range(10)], axis=0)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1e6
        h = h * (n_points / 2) / sum_h
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    cons = ({'type': 'eq', 'fun': lambda x: np.sum([x[i] for i in range(20)])})
    res = minimize(refine, best_theta, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    final_h = 0.5 + np.sum([res.x[i] * np.cos((i+1)*np.pi*np.linspace(0,2,n_points)) + res.x[i+10] * np.sin((i+1)*np.pi*np.linspace(0,2,n_points)) for i in range(10)], axis=0)
    final_h = np.clip(final_h, 0, 1)
    sum_h = np.sum(final_h)
    final_h = final_h * (n_points / 2) / sum_h
    c5 = get_max_corr(final_h) * dx
    return (final_h, c5, n_points)

def get_max_corr(h):
    return np.max(np.correlate(h, 1-h, mode='full'))
