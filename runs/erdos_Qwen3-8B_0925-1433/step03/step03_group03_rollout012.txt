    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    scale_initial = target_sum / sum_initial
    scaled_initial = np.clip(initial_h * scale_initial, 0, 1)
    population = [scaled_initial]
    for _ in range(20):
        perturbed = scaled_initial + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale_perturbed = target_sum / sum_perturbed
        perturbed = perturbed * scale_perturbed
        population.append(perturbed)
    velocities = [np.zeros(n_points) for _ in population]
    personal_bests = [h.copy() for h in population]
    best_h = min(personal_bests, key=lambda x: compute_c5(x, dx, n_points))
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    n_iterations = 800
    max_stagnation = 50
    stagnation_counter = 0
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    for iteration in range(n_iterations):
        if time.time() - start_time > budget_s:
            break
        current_c5 = [(compute_c5(h, dx, n_points), i) for i, h in enumerate(population)]
        for i in range(len(population)):
            c5_i, idx = current_c5[i]
            if c5_i < compute_c5(personal_bests[i], dx, n_points):
                personal_bests[i] = population[i].copy()
        current_global = min(population, key=lambda x: compute_c5(x, dx, n_points))
        curr_global_c5 = compute_c5(current_global, dx, n_points)
        if curr_global_c5 < best_c5:
            best_c5 = curr_global_c5
            best_h = current_global.copy()
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocity = velocities[i]
            velocity = omega * velocity + c1 * r1 * (personal_bests[i] - population[i]) + c2 * r2 * (best_h - population[i])
            velocities[i] = velocity
            population[i] += velocity
            h = population[i]
            h_clipped = np.clip(h, 0, 1)
            sum_h_clipped = np.sum(h_clipped)
            if sum_h_clipped == 0:
                continue
            scale = target_sum / sum_h_clipped
            h_clipped = h_clipped * scale
            population[i] = h_clipped.copy()
        current_global = min(population, key=lambda x: compute_c5(x, dx, n_points))
        curr_global_c5 = compute_c5(current_global, dx, n_points)
        if np.isclose(curr_global_c5, best_c5, atol=1e-6):
            stagnation_counter += 1
        else:
            stagnation_counter = 0
        if stagnation_counter >= max_stagnation:
            omega = max(0.4, omega - 0.01)
            c1 = max(0.5, c1 - 0.1)
            c2 = max(0.5, c2 - 0.1)
            stagnation_counter = 0
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
