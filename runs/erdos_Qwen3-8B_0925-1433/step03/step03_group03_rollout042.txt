import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    current_h = h.copy()
    current_c5 = compute_c5(current_h, dx, n_points)
    best_h = current_h.copy()
    best_c5 = current_c5
    temp = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        candidate = current_h + noise
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        scale_candidate = (n_points / 2) / sum_candidate
        candidate = np.clip(candidate * scale_candidate, 0, 1)
        c5_candidate = compute_c5(candidate, dx, n_points)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = candidate.copy()
        delta = c5_candidate - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = candidate.copy()
            current_c5 = c5_candidate
        temp *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
