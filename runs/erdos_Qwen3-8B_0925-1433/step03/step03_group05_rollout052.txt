        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        # Early stopping check
        iter_count += 1
        if iter_count % 100 == 0:
            learning_rate *= 0.99

    return (best_h, best_c5, n_points)
