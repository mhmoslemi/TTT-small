import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 120  # low-res first, then refine if time allows
    dx = 2.0 / n_points

    # Helper to repair sum and clamp
    def repair(h):
        h_clamped = np.clip(h, 0, 1)
        s = np.sum(h_clamped)
        if s == 0:
            return np.full_like(h, 0.5)
        scale = (n_points / 2) / s
        return np.clip(h_clamped * scale, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = repair(h)
    else:
        h = np.random.rand(n_points)
        h = repair(h)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    population = [h.copy() for _ in range(50)]
    start_time = time.time()

    while time.time() - start_time < budget_s:
        improvements = 0
        for i in range(len(population)):
            target = population[i]
            # Select three distinct indices
            idxs = np.random.choice(len(population), 3, replace=False)
            vec1, vec2, vec3 = [population[j] for j in idxs]
            # Mutation: best/2 strategy
            F = 0.8 + (np.random.rand() - 0.5) * 0.2
            mutant = target + F * (vec3 - vec2)
            # Crossover
            CR = 0.9
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = repair(trial)
            if compute_c5(trial) < compute_c5(target):
                population[i] = trial.copy()
                if compute_c5(trial) < best_c5:
                    best_c5 = compute_c5(trial)
                    best_h = trial.copy()
                improvements += 1
        if improvements == 0:
            break  # no improvement, early exit
    return (best_h, best_c5, n_points)
