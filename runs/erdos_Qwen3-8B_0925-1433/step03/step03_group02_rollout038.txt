        for i in range(n_points):

            h_neg = h.copy()
            h_neg[i] -= eps
            sum_neg = np.sum(h_neg)
            if sum_neg == 0:

            else:
                scale_neg = (n_points / 2) / sum_neg

        # Adaptive step size

            step_size = 0.5 / grad_norm

        noise = np.random.uniform(-0.05, 0.05, n_points)
        # Update
        h_trial = h + step_size * (-grad) + noise

        h_trial = np.clip(h_trial, 0, 1)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    scale = (n_points / 2) / np.sum(initial_h_values)
    h = np.clip(initial_h_values * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        grad = np.zeros(n_points)
        eps = 1e-8
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            sum_perturb = np.sum(h_perturb)
            if sum_perturb == 0:
                perturb_val = 1e6
            else:
                scale_perturb = (n_points / 2) / sum_perturb
                h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
                overlap = 0.0
                for j in range(n_points):
                    jj = j + k
                    if 0 <= jj < n_points:
                        overlap += h_perturb[j] * (1 - h_perturb[jj])
                perturb_val = overlap
            h_neg = h.copy()
            h_neg[i] -= eps
            sum_neg = np.sum(h_neg)
            if sum_neg == 0:
                perturb_neg = 1e6
            else:
                scale_neg = (n_points / 2) / sum_neg
                h_neg = np.clip(h_neg * scale_neg, 0, 1)
                overlap_neg = 0.0
                for j in range(n_points):
                    jj = j + k
                    if 0 <= jj < n_points:
                        overlap_neg += h_neg[j] * (1 - h_neg[jj])
                perturb_neg = overlap_neg
            grad[i] = (perturb_val - perturb_neg) / (2 * eps)
        grad_norm = np.linalg.norm(grad)
        step_size = 0.5 / (grad_norm + 1e-10)
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_trial = h + step_size * (-grad) + noise
        h_trial = np.clip(h_trial, 0, 1)
        sum_trial = np.sum(h_trial)
        if sum_trial == 0:
            h_trial = np.ones(n_points) * 0.5
        else:
            scale_trial = (n_points / 2) / sum_trial
            h_trial = np.clip(h_trial * scale_trial, 0, 1)
        current_c5 = compute_c5(h_trial, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_trial.copy()
        h = h_trial.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    max_overlap = 0.0
    for k in range(-n_points + 1, n_points):
        overlap = 0.0
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx
