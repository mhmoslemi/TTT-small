            h_neg = h.copy()
            h_neg[i] -= eps
            sum_neg = np.sum(h_neg)
            if sum_neg == 0:
                perturb_neg_corr = 1e6
            else:
                scale_neg = (n_points / 2) / sum_neg
                h_neg = np.clip(h_neg * scale_neg, 0, 1)
                corr_neg = np.correlate(h_neg, 1 - h_neg, mode='full')
                perturb_neg_corr = corr_neg[k]
            grad[i] = (perturb_corr - perturb_neg_corr) / (2 * eps)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-6:
            step_size = 0.1
        else:
            step_size = 0.5 / grad_norm
        noise = np.random.uniform(-0.01, 0.01, n_points)
        trial_h = h - grad * step_size + noise
        trial_h = np.clip(trial_h, 0, 1)
        trial_sum = np.sum(trial_h)
        if trial_sum == 0:
            trial_h = np.ones(n_points) * 0.5
        else:
            scale_trial = (n_points / 2) / trial_sum
            trial_h = np.clip(trial_h * scale_trial, 0, 1)
        trial_c5 = compute_c5(trial_h, dx, n_points)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h.copy()
        h = trial_h
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
