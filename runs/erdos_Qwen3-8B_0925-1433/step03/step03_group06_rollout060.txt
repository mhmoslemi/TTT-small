        for i in range(len(particles)):
            pos, vel = particles[i]
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel = inertia_weight * vel + cognitive_coeff * r1 * (personal_bests[i][0] - pos) + social_coeff * r2 * (swarm_best_pos - pos)
            new_pos = pos + vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_pos = np.sum(new_pos)
            penalty = penalty_coeff * np.abs(sum_pos - n_points / 2)
            curr_val = evaluate_erdos_solution(new_pos, n_points) + penalty
            particles[i] = (new_pos, vel)
            if curr_val < personal_bests[i][1]:
                personal_bests[i] = (new_pos, curr_val)
                if curr_val < swarm_best_val:
                    swarm_best_val = curr_val
                    swarm_best_pos = new_pos.copy()
                    best_h = swarm_best_pos.copy()
                    best_c5 = swarm_best_val
    return (best_h, best_c5, n_points)
