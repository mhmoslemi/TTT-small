        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2}])

        if res.success:
            h = res.x.copy()

        if iteration % 100 == 0:
            perturb = np.random.normal(0, 0.05, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (len(h)/2) / sum_h
                h = np.clip(h * scale, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
