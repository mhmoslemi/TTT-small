    start_time = time.time()
    iterations = 0
    temp = 0.3
    cooling_rate = 0.995
    max_iterations = 100000

    while time.time() - start_time < budget_s:
        iterations += 1
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        h_candidate = h + perturb
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate == 0:
            h_candidate = np.full_like(h_candidate, 0.5)
        else:
            scale = (n_points / 2) / sum_candidate
            h_candidate = np.clip(h_candidate * scale, 0, 1)
        c5_candidate = compute_c5(h_candidate)
        current_c5 = compute_c5(h)
        delta = c5_candidate - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_candidate.copy()
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        if iterations % 100 == 0:
            temp *= cooling_rate
        if iterations % 1000 == 0:
            if np.abs(best_c5 - compute_c5(h)) < 1e-7:
                break
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
