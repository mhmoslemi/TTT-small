                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = scale_and_clip(h_opt)
                population[idx] = h_opt
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt
                    if best_c5 < 0.3807:
                        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
