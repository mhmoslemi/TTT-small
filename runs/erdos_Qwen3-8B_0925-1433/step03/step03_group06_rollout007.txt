import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//3] = 0.75
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    perturbation_std = 0.05
    no_improvement = 0
    max_no_improvement = 500
    
    while time.time() - start_time < budget_s:
        new_h = h + np.random.normal(0, perturbation_std, n_points)
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            continue
        scale_factor = (n_points / 2) / sum_new
        new_h = np.clip(new_h * scale_factor, 0, 1)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            no_improvement = 0
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = new_h
                best_c5 = new_c5
                no_improvement = 0
            else:
                no_improvement += 1
        if no_improvement > max_no_improvement:

    scale_factor = (n_points / 2) / sum_best
    best_h = np.clip(best_h * scale_factor, 0, 1)
    return (best_h, best_c5, n_points)

    corr = np.correlate(h, 1 - h, mode='full')
