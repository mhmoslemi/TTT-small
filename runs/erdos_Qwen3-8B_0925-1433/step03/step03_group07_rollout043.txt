import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600):

            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)

    c2 = 1.5

        iteration += 1

            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx

                best_c5 = c5
                gbest = h.copy()

            v_new = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)

            particles[i] = h_new

        if iteration % 500 == 0:
            def refine_obj(x):
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)

                    return 1e6
                scale = (n_points / 2) / sum_x
                x = np.clip(x * scale, 0, 1)

                return np.max(corr) * dx
            res = minimize(refine_obj, gbest, method='SLSQP', bounds=[(0,1)]*n_points,

            if res.success:
                gbest = res.x.copy()
                best_c5 = compute_c5(gbest)

def compute_c5(h):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
