import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = get_max_corr(best_h) * dx
    start_time = time.time()
    
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points/2) / sum_x
        x = np.clip(x * scale, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1)] * n_points
    
    def gen_pattern(pattern_type):
        if pattern_type == 'sine':
            return 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
        elif pattern_type == 'tri':
            return 0.5 * (1 + np.abs(2 * np.arange(n_points) / n_points - 1))
        elif pattern_type == 'alt':
            return np.concatenate([np.full(n_points//3, 0.5), np.full(n_points//3, 0), np.full(n_points//3, 0.5)])
        elif pattern_type == 'rand':
            return np.random.rand(n_points)
    
    patterns = ['sine', 'tri', 'alt', 'rand']
    for pt in patterns:
        guess = gen_pattern(pt)
        sum_guess = np.sum(guess)
        scale = (n_points / 2) / sum_guess
        guess = np.clip(guess * scale, 0, 1)
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = get_max_corr(current_h) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    while time.time() - start_time < budget_s:

        sum_perturb = np.sum(h_perturb)

        if res_perturb.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return (best_h, best_c5, n_points)

def get_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)
