import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 450
    dx = 2.0 / n_points

    def rescale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

    def get_max_corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr)

    if initial_h_values is not None:
        h = rescale(initial_h_values.copy())
    else:
        h = np.random.rand(n_points)
        h = rescale(h)

    candidates = [h]
    x = np.linspace(0, 2, n_points)
    for _ in range(3):
        if np.random.rand() > 0.5:
            # Triangular wave
            htri = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
        else:
            # Sinusoidal wave
            htri = 0.5 + 0.5 * np.sin(np.pi * x)
        htri = rescale(htri)
        candidates.append(htri)

    best_h = h.copy()
    best_c5 = get_max_corr(best_h) * dx

    start_time = time.time()
    T = 1.0
    tau = 0.99
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        for h_candidate in candidates:
            perturb = np.random.normal(0, T, n_points)
            h_perturb = h_candidate + perturb
            h_perturb = rescale(h_perturb)
            current_c5 = get_max_corr(h_perturb) * dx
            delta = current_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = h_perturb.copy()
                best_c5 = current_c5
        T *= tau

    def obj(x):
        x = rescale(x)
        return get_max_corr(x) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = get_max_corr(best_h) * dx

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
