                best_h = res.x.copy()
        if time.time() - start_time >= budget_s:
            break
    
    while time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        scale_perturb = (n_points / 2) / sum_perturb
        h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
        res_perturb = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res_perturb.success:
            current_c5 = get_max_corr(res_perturb.x) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res_perturb.x.copy()
        if time.time() - start_time >= budget_s:
            break
    
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / sum_h
    return (best_h, best_c5, n_points)
