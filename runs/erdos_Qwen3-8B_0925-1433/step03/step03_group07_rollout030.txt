        angles = np.pi * (N-1)/N * np.arange(N) * (np.arange(N)+1)
        z = np.cos(angles)
        h_binary = (z >= 0.5).astype(float)
        sum_h = np.sum(h_binary)
        delta = sum_h - N/2
        if delta > 0:
            indices = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(indices, int(delta), replace=False)
            h_binary[flip_indices] = 0
        else:
            indices = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(indices, int(-delta), replace=False)
            h_binary[flip_indices] = 1
        h = h_binary

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_points = len(h)
    dx = 2.0 / n_points

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        def obj(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2}]
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=constraints)

        if res.success:
            h = res.x.copy()

        if iteration % 100 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (len(h)/2) / sum_h
                h = np.clip(h * scale, 0, 1)

    return (best_h, best_c5, n_points)
