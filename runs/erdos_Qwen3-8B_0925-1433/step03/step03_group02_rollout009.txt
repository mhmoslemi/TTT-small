import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    population = [h.copy() for _ in range(50)]

    best_c5 = compute_c5(best_h, dx, n_points)

        for i in range(len(h)):

            h_perturb[i] += eps
            sum_perturb = np.sum(h_perturb)

                perturb_corr = 1e6

                scale_perturb = (n_points / 2) / sum_perturb
                h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
                corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
                perturb_corr = np.max(corr_perturb) * dx
            h_neg = h.copy()
            h_neg[i] -= eps
            sum_neg = np.sum(h_neg)
            if sum_neg == 0:
                perturb_neg_corr = 1e6
            else:
                scale_neg = (n_points / 2) / sum_neg
                h_neg = np.clip(h_neg * scale_neg, 0, 1)

                perturb_neg_corr = np.max(corr_neg) * dx
            grad[i] = (perturb_corr - perturb_neg_corr) / (2 * eps)
        return grad
    
    def local_refine(h):
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale_x = (n_points / 2) / sum_x
