import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        n_points = 45
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.6 if i % 2 == 0 else 0.4
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    iteration = 0
    lambda_val = 100.0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        curr_c5 = compute_c5(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        
        def obj(x):
            x_clipped = np.clip(x, 0, 1)
            sum_x = np.sum(x_clipped)
            if sum_x == 0:
                return 1e6
            scale = (len(x_clipped)/2) / sum_x
            x_scaled = np.clip(x_clipped * scale, 0, 1)
            corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return np.sum(np.exp(-lambda_val * corr)) * dx
        
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2}])
