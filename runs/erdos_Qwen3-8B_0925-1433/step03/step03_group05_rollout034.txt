import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / s
        return np.clip(h * scale, 0, 1)

    def objective(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return 1e6
        x = np.clip(x * (n_points / 2) / s, 0, 1)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        return compute_c5(h)

    def tournament_selection(pop, k=5):
        return min(np.random.choice(pop, size=k), key=fitness)

    h_initial = scale_h(initial_h_values)
    pop_size = 50
    population = [scale_h(np.random.uniform(0, 1, n_points)) for _ in range(pop_size)]
    for i in range(pop_size):
        population[i] = scale_h(np.random.uniform(0, 1, n_points))

    best_h = h_initial
    best_c5 = compute_c5(h_initial)
    generation = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
            child1 = scale_h(child1)
            child2 = scale_h(child2)
            new_pop.append(child1)
            new_pop.append(child2)
        
        for i in range(len(new_pop)):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.1, n_points)
                new_pop[i] = scale_h(new_pop[i] + noise)

        population = new_pop
        for h in population:
            if fitness(h) < best_c5:
                best_h = h.copy()
                best_c5 = fitness(h)
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        
        if generation % 10 == 0:
            for h in population[:5]:
                res = minimize(objective, h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
                if res.success:
                    h_opt = scale_h(res.x)
                    if compute_c5(h_opt) < best_c5:
                        best_h = h_opt
                        best_c5 = compute_c5(h_opt)
                        if best_c5 < 0.3807:
                            return (best_h, best_c5, n_points)

        generation += 1

    return (best_h, best_c5, n_points)
