import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.6 if i % 2 == 0 else 0.4
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = 0.5
    cooling_rate = 0.995
    max_iter = 50000
    no_improve = 0
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.02, size=n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed == 0:
            h_perturbed = np.full_like(h_perturbed, 0.5)
