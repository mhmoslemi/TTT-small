import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Normalize initial solution
    sum_initial = np.sum(initial_h_values)
    scale = (n_points / 2) / sum_initial
    initial_h = np.clip(initial_h_values * scale, 0, 1)
    # Initialize population
    population = [initial_h]
    for _ in range(20):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_h = (n_points / 2) / sum_h
        h = np.clip(h * scale_h, 0, 1)
        population.append(h)
    # PSO parameters
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    velocities = [np.zeros(n_points) for _ in population]
    personal_bests = [h.copy() for h in population]
    global_best = min(population, key=lambda h: compute_c5(h, dx, n_points))
    best_c5 = compute_c5(global_best, dx, n_points)
    start_time = time.time()
    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        for i in range(len(population)):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (personal_bests[i] - population[i]) + c2 * r2 * (global_best - population[i])
            population[i] += velocities[i]
            # Clip and rescale
            population[i] = np.clip(population[i], 0, 1)
            sum_pop = np.sum(population[i])
            if sum_pop > 0:
                population[i] *= (n_points / 2) / sum_pop
        # Evaluate and update
        new_c5 = [compute_c5(h, dx, n_points) for h in population]
        for i in range(len(population)):
            if new_c5[i] < compute_c5(personal_bests[i], dx, n_points):
                personal_bests[i] = population[i].copy()
        current_global = min(population, key=lambda h: compute_c5(h, dx, n_points))
        if compute_c5(current_global, dx, n_points) < best_c5:
            best_c5 = compute_c5(current_global, dx, n_points)
            global_best = current_global.copy()
        # Adaptive parameters
        if _ % 100 == 0 and _ > 50:
            inertia = max(0.4, inertia - 0.005)
            c1 = max(0.5, c1 - 0.05)
            c2 = max(0.5, c2 - 0.05)
    return (global_best, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    return np.max(corr) * dx
