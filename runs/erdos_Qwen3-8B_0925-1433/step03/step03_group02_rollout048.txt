import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values)

    h = np.clip(initial_h_values * scale, 0, 1)
    population = [h.copy() for _ in range(20)]

    best_h = h.copy()

        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind, dx, n_points) for ind in population]
        top_indices = np.argsort(fitness)[:10]
        top_candidates = [population[i] for i in top_indices]
        
        new_population = []
        for candidate in top_candidates:
            noise = np.random.normal(0, 0.05, n_points)
            mutated = candidate + noise
            mutated = np.clip(mutated, 0, 1)
            sum_mut = np.sum(mutated)
            if sum_mut == 0:
                mutated = np.ones(n_points) * 0.5
            else:
                mutated = np.clip(mutated * (n_points / 2) / sum_mut, 0, 1)
            new_population.append(mutated)
        
        for i in range(len(new_population)):
            h_refine = new_population[i].copy()
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(lambda x: compute_c5(x, dx, n_points), h_refine, method='SLSQP', bounds=bounds, constraints=cons)
            new_population[i] = result.x

        for ind in population:
            current_c5 = compute_c5(ind, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
