            return np.max(corr) * dx
        n_points = len(h)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1)] * n_points
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            h = res.x.copy()
        if iteration % 50 == 0:
            indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
            h[indices] += np.random.uniform(-0.05, 0.05, size=len(indices))
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
    return (best_h, best_c5, len(h))
