    no_improvement = 0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p, dx, n_points) for p in particles]
        gbest_idx = np.argmin(c5_values)
        gbest = particles[gbest_idx].copy()
        for i in range(len(particles)):
            p = particles[i]
            p_current_c5 = c5_values[i]
            pbest = pbests[i]
            pbest_c5 = compute_c5(pbest, dx, n_points)
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = omega * velocities[i] + c1 * r1 * (pbest - p) + c2 * r2 * (gbest - p)
            velocities[i] = new_velocity
            new_p = p + new_velocity
            new_p = repair(new_p, T)
            particles[i] = new_p
            if compute_c5(new_p, dx, n_points) < pbest_c5:
                pbests[i] = new_p.copy()
        current_best_c5 = compute_c5(gbest, dx, n_points)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = gbest.copy()
        if abs(current_best_c5 - best_c5) < 1e-6:
            no_improvement += 1
            if no_improvement >= stagnation_threshold:
                omega += 0.1
                c1 = max(c1 - 0.1, 0.1)
                c2 = max(c2 - 0.1, 0.1)
                no_improvement = 0
        else:
            no_improvement = 0
    return (best_h, best_c5, n_points)
