                term1 = 1.0 - h[iplus]
            term2 = 0.0
            if 0 <= iminus < n_points:
                term2 = h[iminus]
            grad[i] = term1 - term2
        direction = -grad
        noise = np.random.uniform(-0.05, 0.05, n_points)
        grad_norm = np.linalg.norm(grad)
        step_size = 0.5 / grad_norm if grad_norm > 1e-6 else 0.1
        h_update = h + step_size * direction + noise
        h_update = np.clip(h_update, 0, 1)
        sum_h = np.sum(h_update)
        if sum_h == 0:
            h_update = np.ones(n_points) * 0.5
        else:
            scale = (n_points / 2) / sum_h
            h_update = np.clip(h_update * scale, 0, 1)
        corr_new = np.correlate(h_update, 1 - h_update, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_update.copy()
        h = h_update.copy()
        if _ > 100 and np.allclose(c5_new, best_c5, atol=1e-5):
            break
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n_points)
