        sum_h_square = np.sum(h_square)
        h_square_rescaled = h_square * (n_points / 2) / sum_h_square
        initial_guesses.append(h_square_rescaled)
        # Triangle wave
        h_triangle = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
        h_triangle = np.clip(h_triangle, 0, 1)
        sum_h_triangle = np.sum(h_triangle)
        h_triangle_rescaled = h_triangle * (n_points / 2) / sum_h_triangle
        initial_guesses.append(h_triangle_rescaled)
        # Gaussian-like
        beta = 100.0
        h_gauss = np.exp(-beta * (x - 1)**2)
        h_gauss = np.clip(h_gauss, 0, 1)
        sum_h_gauss = np.sum(h_gauss)
        h_gauss_rescaled = h_gauss * (n_points / 2) / sum_h_gauss
        initial_guesses.append(h_gauss_rescaled)

    for guess in initial_guesses:
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1)] * n_points
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        if time.time() - start_time >= budget_s:
            break

    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        if best_h is None:
            break
        perturb = np.random.normal(0, 0.02, n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        scale_perturb = (n_points / 2) / sum_perturb
        h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
        def obj_perturb(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        cons_perturb = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds_perturb = [(0, 1)] * n_points
        res_perturb = minimize(obj_perturb, h_perturb, method='SLSQP', bounds=bounds_perturb, constraints=cons_perturb, tol=1e-6)
        if res_perturb.success:
            current_c5 = res_perturb.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res_perturb.x

    return (best_h, best_c5, n_points)
