import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = np.clip(h * scale, 0, 1)
    
    # PSO parameters
    pop_size = 20
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    rho = 1000
    perturb_std = 0.05
    
    # Initialize population
    population = [h.copy() for _ in range(pop_size)]
    velocities = [np.random.randn(n_points) * 0.01 for _ in range(pop_size)]
    personal_bests = [p.copy() for p in population]
    swarm_best = np.copy(population[0])
    best_c5 = compute_c5(population[0])
    
    start_time = time.time()
    best_h = population[0].copy()
    
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_bests[i] - population[i]) +
                             social * r2 * (swarm_best - population[i]))
            
            # Update position
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            
            # Constraint penalty
            sum_p = np.sum(population[i])
            penalty = rho * abs(sum_p - n_points / 2)

            corr = np.correlate(population[i], 1 - population[i], mode='full')
            obj_val = np.max(corr) * (2.0 / n_points) + penalty

                personal_bests[i] = population[i].copy()

            if obj_val < compute_c5(swarm_best):
                swarm_best = population[i].copy()
        
        # Random perturbation
        for i in range(pop_size):

                population[i] += np.random.normal(0, perturb_std, n_points)
                population[i] = np.clip(population[i], 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
