import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 150
    pop_size = 50
    best_h = None
    best_c5 = float('inf')
    initial_h = initial_h_values if initial_h_values is not None else np.random.rand(n_points)
    h_pop = []
    for _ in range(pop_size):
        h = np.clip(initial_h.copy(), 0, 1) if initial_h_values is not None else np.random.rand(n_points)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full(n_points, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        h_pop.append(h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(h) for h in h_pop]
        idx = np.argsort(fitness)
        top_idx = idx[:int(0.2 * pop_size)]
        new_pop = []
        for i in range(0, len(top_idx), 2):
            p1, p2 = h_pop[top_idx[i]], h_pop[top_idx[i+1]]
            split = np.random.randint(1, n_points - 1)
            new_pop.append(np.concatenate([p1[:split], p2[split:]]))
            new_pop.append(np.concatenate([p2[:split], p1[split:]]))
        for i in range(len(new_pop)):
            if np.random.rand() < 0.1:
                h = new_pop[i]
                delta = np.random.uniform(-0.05, 0.05, size=n_points)
                h = np.clip(h + delta, 0, 1)
                sum_h = np.sum(h)
                if sum_h == 0:
                    h = np.full(n_points, 0.5)
                else:
                    scale = (n_points / 2) / sum_h
                    h = np.clip(h * scale, 0, 1)
                new_pop[i] = h
        combined = h_pop + new_pop
        combined.sort(key=lambda x: compute_c5(x))
        h_pop = combined[:pop_size]
        if min(fitness) < best_c5:
            best_c5 = min(fitness)
            best_h = h_pop[np.argmin(fitness)].copy()
    return (best_h, best_c5, n_points)
