            best_h = res.x
            best_c5 = get_max_corr(best_h) * dx
        if time.time() - start_time >= budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)

def rescale(arr):
    arr = np.clip(arr, 0, 1)
    sum_arr = np.sum(arr)
    if sum_arr == 0:
        return np.zeros_like(arr)
    scale = (n_points / 2) / sum_arr
    return arr * scale
