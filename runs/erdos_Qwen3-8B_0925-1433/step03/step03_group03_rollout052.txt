import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    scale = (n_points / 2) / sum_initial
    initial_h = initial_h * scale
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    scale_correct = (n_points / 2) / sum_initial
    initial_h = initial_h * scale_correct
    initial_h = np.clip(initial_h, 0, 1)
    population = [initial_h.copy()]
    for _ in range(20):
        noise = np.random.normal(0, 0.1, n_points)
        h = initial_h_values.copy() + noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
        population.append(h)
    velocities = [np.zeros(n_points) for _ in population]
    pbest = [h.copy() for h in population]
    pbest_fitness = [compute_c5(h, dx, n_points) for h in population]
    best_h = pbest[np.argmin(pbest_fitness)].copy()
    best_c5 = np.min(pbest_fitness)
    omega = 0.8
    c1, c2 = 1.5, 1.5
    max_iter = 800
    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h, dx, n_points) for h in population]
        gbest_idx = np.argmin(fitness)
        gbest = population[gbest_idx].copy()
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            population[i] = population[i] + velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            sum_h = np.sum(population[i])
            scale = (n_points / 2) / sum_h
            population[i] = population[i] * scale
            population[i] = np.clip(population[i], 0, 1)
            new_fitness = compute_c5(population[i], dx, n_points)
            if new_fitness < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = new_fitness
            if new_fitness < best_c5:
                best_h = population[i].copy()
                best_c5 = new_fitness
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
