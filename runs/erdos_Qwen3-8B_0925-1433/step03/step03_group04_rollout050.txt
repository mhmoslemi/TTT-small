        sum_h = np.sum(h_initial)
        scale = (n_points / 2) / sum_h
        h_initial = np.clip(h_initial * scale, 0, 1)
        population = [h_initial.copy()]
        for _ in range(49):
            h_rand = np.random.rand(n_points)
            sum_rand = np.sum(h_rand)
            scale_rand = (n_points / 2) / sum_rand
            h_rand = np.clip(h_rand * scale_rand, 0, 1)
            population.append(h_rand.copy())
    else:
        n_points = 150
        population = []
        for _ in range(50):
            h_rand = np.random.rand(n_points)
            sum_rand = np.sum(h_rand)
            scale_rand = (n_points / 2) / sum_rand
            h_rand = np.clip(h_rand * scale_rand, 0, 1)
            population.append(h_rand.copy())
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            fitness.append(c5)
        
        # Select top 20%
        sorted_indices = np.argsort(fitness)
        selected_h = [population[i] for i in sorted_indices[:int(0.2 * len(population))]]
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected_h), 2):
            if i+1 >= len(selected_h):
                break
            parent1 = selected_h[i]
            parent2 = selected_h[i+1]
            split_idx = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:split_idx], parent2[split_idx:]])
            child2 = np.concatenate([parent2[:split_idx], parent1[split_idx:]])
            offspring.append(child1)
            offspring.append(child2)
        
        # Mutate 10% of the population
        mutate_indices = np.random.choice(len(population), size=int(0.1 * len(population)), replace=False)
        for idx in mutate_indices:
            h = population[idx].copy()
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full(n_points, 0.5)
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            population[idx] = h
        
        # Combine selected and offspring
        candidates = selected_h + offspring
        candidate_fitness = []
        for h in candidates:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            candidate_fitness.append(c5)
        
        # Sort candidates and take top 50
        sorted_candidates = sorted(zip(candidate_fitness, candidates), key=lambda x: x[0])
        population = [c[1] for c in sorted_candidates[:50]]
        
        # Update best solution
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
