        # Compute gradient for max lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Forward perturbation
            h_perturbed = h.copy()
            h_perturbed[i] += 1e-5
            h_perturbed = np.clip(h_perturbed, 0, 1)
            sum_h_perturbed = np.sum(h_perturbed)
            scale_perturbed = (n_points / 2) / sum_h_perturbed
            h_perturbed = np.clip(h_perturbed * scale_perturbed, 0, 1)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_perturbed = corr_perturbed[lag_index] * dx
            
            # Backward perturbation
            h_perturbed_neg = h.copy()
            h_perturbed_neg[i] -= 1e-5
            h_perturbed_neg = np.clip(h_perturbed_neg, 0, 1)
            sum_h_neg = np.sum(h_perturbed_neg)
            scale_neg = (n_points / 2) / sum_h_neg
            h_perturbed_neg = np.clip(h_perturbed_neg * scale_neg, 0, 1)
            corr_neg = np.correlate(h_perturbed_neg, 1 - h_perturbed_neg, mode='full')
            c5_neg = corr_neg[lag_index] * dx
            
            grad[i] = (c5_perturbed - c5_neg) / (2 * 1e-5)
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        
        # Track best solution
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
