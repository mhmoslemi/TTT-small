import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 0.25
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    dx = 2.0 / n_points
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 0
        h_scaled = np.clip(h * scale, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=int((budget_s - (time.time() - start_time)) / 10),
        disp=False
    )
    
    best_h = np.clip(result.x, 0, 1)
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 0
    best_h = np.clip(best_h * scale, 0, 1)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
