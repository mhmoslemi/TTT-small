import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    temp = 1.0
    cooling = 0.995
    start_time = time.time()
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        candidate = h + noise
        candidate = np.clip(candidate, 0, 1)
        sum_c = np.sum(candidate)
        scale_c = (n_points / 2) / sum_c
        candidate = np.clip(candidate * scale_c, 0, 1)
        c5 = compute_c5(candidate, dx, n_points)
        if c5 < best_c5:

            best_h = candidate.copy()
        elif np.random.rand() < np.exp((best_c5 - c5) / temp):
            best_h = candidate.copy()
        temp *= cooling
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
