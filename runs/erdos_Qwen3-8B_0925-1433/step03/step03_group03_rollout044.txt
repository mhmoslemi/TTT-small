import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    scale = (n_points / 2) / sum_initial
    initial_h = np.clip(initial_h * scale, 0, 1)
    population = [initial_h]
    for _ in range(20):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    
    pbest = [np.copy(h) for h in population]
    gbest = population[0].copy()
    best_c5 = float('inf')
    best_h = gbest.copy()
    velocities = [np.zeros(n_points) for _ in population]
    omega = 0.8
    c1, c2 = 1.5, 1.5
    max_iter = 800
    stagnation = 50
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(len(population)):
            current_c5 = compute_c5(population[i], dx, n_points)
            if current_c5 < compute_c5(pbest[i], dx, n_points):
                pbest[i] = population[i].copy()
        
        fitness = [compute_c5(h, dx, n_points) for h in population]
        gbest_idx = np.argmin(fitness)
        if fitness[gbest_idx] < best_c5:
            best_c5 = fitness[gbest_idx]
            best_h = population[gbest_idx].copy()
        
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - population[i]) + c2 * r2 * (gbest - population[i])
            new_h = population[i] + velocities[i]
            new_h = np.clip(new_h, 0, 1)

    corr = np.correlate(h, shifted, mode='full')
