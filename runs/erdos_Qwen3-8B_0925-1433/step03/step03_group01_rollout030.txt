    a_params = best_params[:N]
    b_params = best_params[N:]
    x = np.linspace(0, 2, n_points)
    h = 0.5 + np.sum([a_params[i] * np.cos((i+1)*np.pi*x) + b_params[i] * np.sin((i+1)*np.pi*x) for i in range(N)], axis=0)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = h * scale
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return (h, c5, n_points)
