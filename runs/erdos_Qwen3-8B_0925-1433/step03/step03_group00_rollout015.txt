    sum_s = np.sum(h_sine)
    scale_s = (n_points / 2) / sum_s if sum_s != 0 else 1.0
    h_sine = np.clip(h_sine * scale_s, 0, 1)
    candidates.append(h_sine)
    h_rand = np.random.choice([0, 1], size=n_points)
    sum_r = np.sum(h_rand)
    scale_r = (n_points / 2) / sum_r if sum_r != 0 else 1.0
    h_rand = np.clip(h_rand * scale_r, 0, 1)
    candidates.append(h_rand)
    h_uni = np.zeros(n_points)
    h_uni[:n_points//2] = 1.0
    sum_u = np.sum(h_uni)
    scale_u = (n_points / 2) / sum_u if sum_u != 0 else 1.0
    h_uni = np.clip(h_uni * scale_u, 0, 1)
    candidates.append(h_uni)

    for h_cand in candidates:
        h_cand = np.clip(h_cand, 0, 1)
        sum_hc = np.sum(h_cand)
        scalehc = (n_points / 2) / sum_hc if sum_hc != 0 else 1.0
        h_scaled = np.clip(h_cand * scalehc, 0, 1)
        c = compute_c5(h_scaled, dx, n_points)
        if c < best_c5:
            best_c5 = c
            best_h = h_scaled.copy()

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        h_candidate = np.random.choice(candidates)
        h_c = h_candidate.copy()
        h_c = np.clip(h_c, 0, 1)
        sum_hc = np.sum(h_c)
        scalehc = (n_points / 2) / sum_hc if sum_hc != 0 else 1.0
        h_c_scaled = np.clip(h_c * scalehc, 0, 1)

        def obj(x):
            x = np.clip(x, 0, 1)
            s = np.sum(x)
            if s == 0:
                return 1e10
            x_s = x * (n_points / 2) / s
            corr = np.correlate(x_s, 1 - x_s, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h_c_scaled, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            sum_n = np.sum(new_h)
            scale_n = (n_points / 2) / sum_n if sum_n != 0 else 1.0
            new_h_scaled = np.clip(new_h * scale_n, 0, 1)
            new_c = compute_c5(new_h_scaled, dx, n_points)
            if new_c < best_c5:
                best_c5 = new_c
                best_h = new_h_scaled.copy()
        else:
            new_h = h_c_scaled.copy()
            new_h = np.clip(new_h, 0, 1)
            sum_n = np.sum(new_h)
            scale_n = (n_points / 2) / sum_n if sum_n != 0 else 1.0
            new_h_scaled = np.clip(new_h * scale_n, 0, 1)
            new_c = compute_c5(new_h_scaled, dx, n_points)
            if new_c < best_c5:
                best_c5 = new_c
                best_h = new_h_scaled.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
