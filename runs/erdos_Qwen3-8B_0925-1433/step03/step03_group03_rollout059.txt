import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = h * scale
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    temp = 1.0
    cool = 0.995
    start_time = time.time()
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed > 0:
            scale_perturbed = (n_points / 2) / sum_perturbed
            h_perturbed *= scale_perturbed
        else:
            h_perturbed = h.copy()
        new_c5 = compute_c5(h_perturbed, dx, n_points)
        if new_c5 < best_c5:
            h = h_perturbed.copy()
            best_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - best_c5) / temp)
            if np.random.rand() < prob:
                h = h_perturbed.copy()
                best_c5 = new_c5
        temp *= cool
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
