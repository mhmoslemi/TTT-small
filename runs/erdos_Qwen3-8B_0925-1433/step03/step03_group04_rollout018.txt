            h = np.clip(h * scale, 0., 1.)
            population.append(h.copy())
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    while time.time() - start_time < budget_s:
        scores = []
        for h in population:
            c5 = compute_c5(h)
            scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        sorted_indices = np.argsort(scores)
        top_size = int(len(population) * 0.2)
        parents = [population[i] for i in sorted_indices[:top_size]]
        indices_to_replace = sorted_indices[top_size:]
        new_individuals = []
        for i in indices_to_replace:
            p1 = np.random.choice(parents)
            p2 = np.random.choice(parents)
            idx = np.random.randint(1, n_points - 1)
            child = np.concatenate((p1[:idx], p2[idx:]))
            delta = np.random.uniform(-0.05, 0.05, len(child))
            mutated = child + delta
            mutated = np.clip(mutated, 0., 1.)
            sum_m = np.sum(mutated)
            if sum_m == 0:
                scale = 1.
            else:
                scale = (n_points / 2) / sum_m
            mutated = np.clip(mutated * scale, 0., 1.)
            new_individuals.append(mutated)
        for i, idx in enumerate(indices_to_replace):
            population[idx] = new_individuals[i]
    return (best_h, best_c5, n_points)
