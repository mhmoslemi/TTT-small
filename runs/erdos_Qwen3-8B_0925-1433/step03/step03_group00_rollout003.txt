import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def rescale(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

        return np.max(corr) * dx

    population = []
    initial = initial_h_values.copy()

    for _ in range(20):
        h = initial.copy() + np.random.normal(0, 0.05, n_points)
        population.append(rescale(h))
    for _ in range(10):
        even = np.zeros(n_points)
        even[::2] = 1
        population.append(rescale(even))
    for _ in range(10):
        sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        population.append(rescale(sine))
    for _ in range(10):
        bin_h = np.random.choice([0, 1], size=n_points)
        population.append(rescale(bin_h))
    np.random.shuffle(population)
    population = population[:50]

    best_h = population[0]
    best_c5 = compute_c5(best_h)
