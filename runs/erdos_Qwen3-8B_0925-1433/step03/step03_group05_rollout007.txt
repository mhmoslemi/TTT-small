import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        s = (n_points / 2) / np.sum(h)
        return np.clip(h * s, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = scale_h(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    max_iter = 200
    iter_count = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and iter_count < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_idx = max_idx - (n_points - 1)

        def objective(x):
            x = np.clip(x, 0, 1)
            x = scale_h(x)
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_idx] * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)

        if res.success:

            h = scale_h(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:

            h_perturbed = np.clip(h + perturb, 0, 1)
            h_perturbed = scale_h(h_perturbed)
            h = h_perturbed

        iter_count += 1
