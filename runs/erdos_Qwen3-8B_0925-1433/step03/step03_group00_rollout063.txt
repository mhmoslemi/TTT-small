                    if i_minus_k >= 0:
                        term2 = h[i_minus_k] * dx
                    dcorr = term1 - term2
                    grad[i] += (np.exp(gamma * corr[k]) / Z) * dcorr
            return grad * (1 / gamma)

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='SLSQP', jac=grad, tol=1e-6)
        if res.success:
            h_candidate = res.x.copy()
        else:
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_p = np.sum(h_candidate)
            scale = (n_points / 2) / sum_p if sum_p != 0 else 1.0
            h_candidate = np.clip(h_candidate * scale, 0, 1)
        c5 = compute_c5(h_candidate, dx, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
