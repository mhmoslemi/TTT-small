import numpy as np
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 150
    if initial_h_values is not None:
        n_points = len(initial_h_values)
    population = []
    if initial_h_values is not None:
        h_initial = np.clip(initial_h_values.copy() * (n_points / 2) / np.sum(initial_h_values), 0, 1)
        population.append(h_initial)
        for _ in range(49):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
            population.append(h)
    else:
        for _ in range(50):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                h = np.clip(h * (n_points / 2) / sum_h, 0, 1)
            population.append(h)
    best_h = population[0].copy()
    best_c5 = float('inf')
    start_time = time.time()
    for _ in range(120):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        top_idx = np.argsort(fitness)[:10]
        top = [population[i] for i in top_idx]
        offspring = []
        for _ in range(50 - len(top)):
            p1, p2 = random.sample(top, 2)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            offspring.append(child)
        for _ in range(5):
            h = random.choice(population)
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h_mut = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h_mut)
            if sum_h == 0:
                h_mut = np.full_like(h_mut, 0.5)
            else:
                h_mut = np.clip(h_mut * (n_points / 2) / sum_h, 0, 1)
            offspring.append(h_mut)
        new_pop = top + offspring
        new_pop.sort(key=lambda x: np.correlate(x, 1 - x, mode='full').max() * (2.0 / n_points))
        population = new_pop[:50]
        current_c5 = np.correlate(population[0], 1 - population[0], mode='full').max() * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0].copy()
    return (best_h, best_c5, n_points)
