def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)

        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return 1e6
        scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx
    
    population = [initial_h_values.copy() for _ in range(50)]
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
