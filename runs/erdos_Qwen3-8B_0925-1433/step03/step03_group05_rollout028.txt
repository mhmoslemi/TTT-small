            h = np.ones(n_points) * 0.5

        noise = np.random.normal(0, 0.01, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5

    shifted = 1 - h
    cross_corr = np.correlate(h, shifted, mode='full')
    best_c5 = np.max(cross_corr) * dx
    return (h, best_c5, n_points)
