        )
    except:
        result = differential_evolution(
            objective, bounds, strategy='best2bin',
            popsize=50, tol=1e-5, maxiter=200, seed=seed
        )

    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
