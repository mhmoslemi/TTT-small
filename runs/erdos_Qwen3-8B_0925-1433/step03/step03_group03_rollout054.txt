        corr = np.correlate(h, shifted, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + max_k < n_points:
                grad[i] += 1 - h[i + max_k]
            if i - max_k >= 0:
                grad[i] -= h[i - max_k]
        
        eta = 0.1
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
