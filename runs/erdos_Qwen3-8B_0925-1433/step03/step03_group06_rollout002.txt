import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    global initial_h_values

    else:
        n = 45

        h[2*n//3:] = 0.1
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (len(h)/2) / sum_h
    h = np.clip(h * scale, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    candidates = [h.copy() for _ in range(5)]
    
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        scale = (len(x)/2) / sum_x
        x = np.clip(x * scale, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - len(x)/2
    
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1)] * n_points
    
    while time.time() - start_time < budget_s:
        for h in candidates:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                current_c5 = compute_c5(res.x)
                if current_c5 < best_c5:

                    best_h = res.x.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
