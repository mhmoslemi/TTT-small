import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 200
        h = np.random.rand(n)
        h = np.clip(h * (n/2 / np.sum(h)), 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points

    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1)] * n_points

    initial_guesses = [h.copy()]
    x = np.linspace(0, 2, n_points)

    # Sine wave
    h_sine = 0.5 + 0.5 * np.sin(2 * np.pi * x / (n_points / 2))
    h_sine = np.clip(h_sine, 0, 1)
    sum_h_sine = np.sum(h_sine)
    scale_sine = (n_points / 2) / sum_h_sine
    initial_guesses.append(h_sine * scale_sine)

    # Triangular wave
    h_tri = np.clip(np.abs(x - 1) - 0.5, 0, 1)
    sum_h_tri = np.sum(h_tri)
    scale_tri = (n_points / 2) / sum_h_tri
    initial_guesses.append(h_tri * scale_tri)

    # Gaussian bump
    h_gauss = np.exp(-((x - 1) / 0.5)**2)
    h_gauss = np.clip(h_gauss, 0, 1)
    sum_h_gauss = np.sum(h_gauss)
    scale_gauss = (n_points / 2) / sum_h_gauss
    initial_guesses.append(h_gauss * scale_gauss)

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        if time.time() - start_time >= budget_s:
            break

    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        perturbation = 0.1 * np.sin(2 * np.pi * np.linspace(0, 2, n_points) / (n_points / 2))
        h_perturb = best_h + perturbation
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        scale_perturb = (n_points / 2) / sum_perturb
        h_perturb = h_perturb * scale_perturb
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h
    best_h = np.clip(best_h * scale, 0, 1)
    best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
