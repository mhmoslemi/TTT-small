import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 50
    if initial_h_values is not None:
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        j = np.arange(n_points)
        z = np.cos(np.pi * (n_points - 1)/n_points * j * (j + 1))
        h_initial = (z >= 0.5).astype(float)
        sum_h = np.sum(h_initial)
        desired_sum = n_points / 2
        diff = sum_h - desired_sum
        if diff > 0:
            indices_ones = np.where(h_initial == 1)[0]
            num_flips = int(np.round(diff))
            if num_flips > 0:
                h_initial[indices_ones[:num_flips]] = 0
        elif diff < 0:
            indices_zeros = np.where(h_initial == 0)[0]
            num_flips = int(np.round(-diff))
            if num_flips > 0:
                h_initial[indices_zeros[:num_flips]] = 1
        h = h_initial.copy()
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n_points
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
        if res.success:
            best_h = res.x.copy()
        if iteration % 100 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h = np.clip(best_h + perturb, 0, 1)
            sum_h = np.sum(best_h)
            if sum_h == 0:
                best_h = np.full_like(best_h, 0.5)
            else:
                scale = (n_points / 2) / sum_h
                best_h = np.clip(best_h * scale, 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
