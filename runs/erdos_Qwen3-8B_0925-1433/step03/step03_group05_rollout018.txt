import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def scale_h(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

    pop = [scale_h(initial_h_values.copy() + np.random.normal(0, 0.1, n_points)) for _ in range(50)]
    best_h = scale_h(initial_h_values.copy())
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = np.random.choice(pop, 20, replace=False)
        new_pop = []
        for i in range(0, 20, 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
            else:
                child1, child2 = p1.copy(), p2.copy()
            for child in [child1, child2]:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise

            new_pop.extend([child1, child2])
        pop = new_pop

        if current_c5 < best_c5:
            best_h = current_best
