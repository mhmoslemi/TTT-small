    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def objective(x):
    x = np.clip(x, 0, 1)
    sum_x = np.sum(x)
    if sum_x < 1e-6:
        return 1e6
    corr = np.correlate(x, 1 - x, mode='full')
    return np.max(corr) * (2.0 / len(x))
