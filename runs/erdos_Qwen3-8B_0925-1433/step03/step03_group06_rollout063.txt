import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values) if initial_h_values is not None else 200
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    stagnation = 0
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.05, n_points)
        h_new = h + noise
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new == 0:
            h = best_h.copy()
            continue
        scale = (n_points / 2) / sum_new
        h_scaled = h_new * scale
        new_c5 = compute_c5(h_scaled)
        current_c5 = compute_c5(h)
        if new_c5 < current_c5:
            h = h_scaled.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.random() < prob:
                h = h_scaled.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        temp *= cooling_rate
        stagnation += 1 if new_c5 >= current_c5 else 0
        if stagnation > 100:
            temp = 1.0
            h = best_h.copy()
            stagnation = 0
        iterations += 1
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
