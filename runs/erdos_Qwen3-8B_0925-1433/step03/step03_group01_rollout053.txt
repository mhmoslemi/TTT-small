import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 450
    dx = 2.0 / n_points
    x_values = np.linspace(0, 2, n_points)
    
    def rescale(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        return h
    
    def compute_corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    initial_guesses = []
    
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = rescale(h)
        initial_guesses.append(h)
    else:
        h = np.random.rand(n_points)
        h = rescale(h)
        initial_guesses.append(h)
    
    h_sine = 0.5 * np.sin(np.pi * x_values) + 0.5
    h_sine = rescale(h_sine)
    initial_guesses.append(h_sine)
    
    h_tri = 0.5 * (1 + np.abs(np.sin(np.pi * x_values)))
    h_tri = rescale(h_tri)
    initial_guesses.append(h_tri)
    
    h_square = 0.5 + 0.5 * np.sign(np.sin(np.pi * x_values))
    h_square = rescale(h_square)
    initial_guesses.append(h_square)

    h_exp = rescale(h_exp)
    initial_guesses.append(h_exp)
    
    best_h = None
    best_c5 = float('inf')
    for guess in initial_guesses:
        c5 = compute_corr(guess)
        if c5 < best_c5:
            best_h = guess.copy()
            best_c5 = c5
    
    current_h = best_h.copy()
    temp = 1.0
    cooling_rate = 0.99

        if time.time() - start_time >= budget_s:

        perturb = np.random.normal(0, 0.05, n_points)
