import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    scale = (n_points / 2) / sum_initial
    initial_h = np.clip(initial_h * scale, 0, 1)
    population = [initial_h]
    for _ in range(20):
        noise = np.random.normal(0, 0.05, n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    velocities = [np.zeros(n_points) for _ in population]
    personal_best = [h.copy() for h in population]
    global_best = min(population, key=lambda h: compute_c5(h, dx, n_points))
    best_c5 = compute_c5(global_best, dx, n_points)
    start_time = time.time()
    for _ in range(800):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h, dx, n_points) for h in population]
        for i in range(len(population)):
            if fitness[i] < compute_c5(personal_best[i], dx, n_points):
                personal_best[i] = population[i].copy()
        current_global = min(population, key=lambda h: compute_c5(h, dx, n_points))
        if compute_c5(current_global, dx, n_points) < best_c5:
            best_c5 = compute_c5(current_global, dx, n_points)
            global_best = current_global.copy()
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - population[i]) + c2 * r2 * (global_best - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            sum_pop = np.sum(population[i])
            if sum_pop != 0:
                population[i] *= (n_points / 2) / sum_pop
    return (global_best, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
