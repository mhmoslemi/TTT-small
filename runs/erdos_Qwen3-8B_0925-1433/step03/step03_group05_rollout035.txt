from scipy.optimize import minimize
import time

        return max_corr * dx

        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        return scale_h(h)
    
    initial_h = scale_h(initial_h_values)

    best_c5 = compute_c5(best_h)
