            current_c5 = test_c5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
