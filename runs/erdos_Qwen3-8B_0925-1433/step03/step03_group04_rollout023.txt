import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values
    else:
        n_points = 120
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(x):
        x = np.clip(x, 1e-8, 1 - 1e-8)
        sum_x = np.sum(x)
        penalty = 1e6 * abs(sum_x - n_points / 2)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx + penalty

    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=50, tol=1e-5,
        mutation=(0.5, 1), recombination=0.9, seed=seed, maxiter=200,
        callback=lambda x: None
    )
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
