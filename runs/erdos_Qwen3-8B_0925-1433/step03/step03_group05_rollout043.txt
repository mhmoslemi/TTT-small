        return max_corr * dx

    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h_initial = np.clip(h_initial * scale, 0, 1)
    else:
        h_initial = np.ones(n_points) * 0.5
    
    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    
    def fitness(h):
        return compute_c5(h)
    
    population = [repair(h_initial + np.random.normal(0, 0.1, n_points)) for _ in range(50)]

    best_c5 = fitness(best_h)
    
    start_time = time.time()

        fitnesses = [fitness(h) for h in population]
