    scaled = np.clip(scaled, 0, 1)
    return scaled

def compute_c5(h, dx):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    h = rescale_and_clip(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    if best_c5 < 0.3807:
        return (best_h, best_c5, n_points)
    learning_rate = 0.1
    max_iter = 1000
    time_start = time.time()
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr_idx = np.argmax(corr)
        epsilon = 1e-4
        grad = np.zeros(n_points)
        original_overlap = corr[max_corr_idx] * dx
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += epsilon
            h_perturb = rescale_and_clip(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            overlap_perturb = corr_perturb[max_corr_idx] * dx
            grad[i] = (overlap_perturb - original_overlap) / (2 * epsilon)
        h = h - learning_rate * grad
        h = rescale_and_clip(h)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
