def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5

    h = np.clip(h * scale, 0, 1)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale_x = (n_points / 2) / sum_x
        x = np.clip(x * scale_x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        penalty = 1000 * abs(np.sum(x) - n_points / 2)
        return c5 + penalty
    
    bounds = [(0, 1)] * n_points
    start_time = time.time()
