import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    decay = 0.9995
    learning_rate = 0.05
    while time.time() - start_time < budget_s:
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        shift_index = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + shift_index
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        c5 = compute_c5(h, dx, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        learning_rate *= decay
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
