import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / s, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    def generate_initial_pop():
        pop = []
        h0 = initial_h_values.copy()
        pop.append(h0)
        h_rand = np.random.rand(n_points)
        pop.append(scale_h(h_rand))
        h_sine = 0.5 + 0.5 * np.sin(2*np.pi*np.arange(n_points)/n_points)
        pop.append(scale_h(h_sine))
        h_bin = np.zeros(n_points)
        h_bin[::3] = 1
        h_bin = np.clip(h_bin, 0, 1)
        pop.append(scale_h(h_bin))
        return pop

    population = generate_initial_pop()

    best_h = np.array(population[0])
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    iterations = 0
    max_gen = 50

    while time.time() - start_time < budget_s and iterations < max_gen:
        scores = [compute_c5(h) for h in population]
        idxs = np.argsort(scores)
        top_idx = idxs[:20]
        top_candidates = [population[i] for i in top_idx]

        # Mutation
        new_pop = []
        for _ in range(20):
            if np.random.rand() < 0.6:
                a, b = np.random.choice(top_idx, 2, replace=False)
                child = (population[a] + population[b]) * 0.5
                child = scale_h(child)
            else:
                child = np.random.rand(n_points)
                child = scale_h(child)
            new_pop.append(child)
        population += new_pop
        population = [h for h in population if np.sum(h) > 0]

        # Local optimization
        for h in top_candidates:
            def obj(x):
                h_clipped = np.clip(x, 0, 1)
                s = np.sum(h_clipped)
                if s == 0:
                    return 1e9
                h_scaled = h_clipped * (n_points / 2) / s
                corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
                return np.max(corr) * dx
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                h_opt = res.x
                if compute_c5(h_opt) < compute_c5(h):
                    h = h_opt
        population = [h for h in population if np.sum(h) > 0]
        population = population[:len(population)//2] + top_candidates
        scores = [compute_c5(h) for h in population]
        idxs = np.argsort(scores)
        population = [population[i] for i in idxs]
        best_h = population[0]
        best_c5 = compute_c5(best_h)
        iterations += 1

    return (best_h, best_c5, n_points)
