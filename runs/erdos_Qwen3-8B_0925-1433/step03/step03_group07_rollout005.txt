    best_h = h.copy()

    n_points = len(h)
    dx = 2.0 / n_points

        def obj(x):

            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
