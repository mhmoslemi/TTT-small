            sum_perturb = np.sum(h_perturb)
            if sum_perturb == 0:
                perturb_corr_k = 1e6
            else:
                scale_perturb = (n_points / 2.0) / sum_perturb
                h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
                corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
                perturb_corr_k = corr_perturb[k] * dx
            h_neg = h.copy()
            h_neg[i] -= eps
            sum_neg = np.sum(h_neg)
            if sum_neg == 0:
                perturb_neg_corr_k = 1e6
            else:
                scale_neg = (n_points / 2.0) / sum_neg
                h_neg = np.clip(h_neg * scale_neg, 0, 1)
                corr_neg = np.correlate(h_neg, 1 - h_neg, mode='full')
                perturb_neg_corr_k = corr_neg[k] * dx
            gradient[i] = (perturb_corr_k - perturb_neg_corr_k) / (2 * eps)
        alpha = 1.0
        improved = False
        for _ in range(20):
            trial_h = h + alpha * (-gradient)
            sum_trial = np.sum(trial_h)
            if sum_trial == 0:
                trial_h = np.ones(n_points) * 0.5
            else:
                scale_trial = (n_points / 2.0) / sum_trial
                trial_h = np.clip(trial_h * scale_trial, 0, 1)
            corr_trial = np.correlate(trial_h, 1 - trial_h, mode='full')
            trial_max_corr = np.max(corr_trial) * dx
            if trial_max_corr < max_corr:
                h = trial_h.copy()
                max_corr = trial_max_corr
                improved = True
                break
            else:
                alpha *= 0.5
        if not improved:
            continue
        sum_h_new = np.sum(h)
        if sum_h_new == 0:
            h = np.ones(n_points) * 0.5
        else:
            scale_h = (n_points / 2.0) / sum_h_new
            h = np.clip(h * scale_h, 0, 1)
        corr_current = np.correlate(h, 1 - h, mode='full')
        current_max_corr = np.max(corr_current) * dx
        if current_max_corr < best_c5:
            best_c5 = current_max_corr
            best_h = h.copy()
        if iteration > 100 and abs(current_max_corr - best_c5) < 1e-5:
            break
    return (best_h, best_c5, n_points)
