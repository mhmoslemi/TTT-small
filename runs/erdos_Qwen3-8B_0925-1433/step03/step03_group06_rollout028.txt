import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = 45 if initial_h_values is not None else 250
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//3] = 0.75
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points/2) / sum_h
    h = np.clip(h * scale, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    particles = [h.copy() for _ in range(25)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(25)]
    personal_best = [p.copy() for p in particles]
    personal_best_vals = [compute_c5(p) for p in particles]
    swarm_best = h.copy()
    swarm_best_val = best_c5
    rho = 1000
    iter_no_improve = 0
    max_iter_no_improve = 100
    while time.time() - start_time < budget_s:
        for i in range(len(particles)):
            pos = particles[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            vel += 0.8 * (np.random.rand() * (personal_best[i] - pos) * 1.5) + \
                   0.8 * (np.random.rand() * (swarm_best - pos) * 1.5)
            new_pos = pos + vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            penalty = rho * abs(sum_new - n_points/2)
            corr = np.correlate(new_pos, 1 - new_pos, mode='full')
            current_c5 = np.max(corr) * dx + penalty
            if current_c5 < personal_best_vals[i]:
                personal_best[i] = new_pos.copy()
                personal_best_vals[i] = current_c5
            if current_c5 < swarm_best_val:
                swarm_best = new_pos.copy()
                swarm_best_val = current_c5
            particles[i] = new_pos
            velocities[i] = vel
        if swarm_best_val < best_c5:
            best_c5 = swarm_best_val
            best_h = swarm_best.copy()
        iter_no_improve += 1
        if iter_no_improve >= max_iter_no_improve:
            iter_no_improve = 0
            for i in range(len(particles)):
                particles[i] = personal_best[i].copy()
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    return (np.clip(best_h, 0, 1), best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
