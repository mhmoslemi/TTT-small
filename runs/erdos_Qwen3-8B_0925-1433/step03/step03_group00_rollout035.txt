        iterations += 1
        if iterations > max_iterations:
            break

        c5_list = [compute_c5(p) for p in population]
        sort_indices = np.argsort(c5_list)
        top_candidates = [population[i] for i in sort_indices[:20]]

        new_population = []
        for i in range(0, len(top_candidates), 2):
            if i + 1 < len(top_candidates):
                p1 = top_candidates[i]
                p2 = top_candidates[i+1]
                child = 0.8 * p1 + 0.2 * p2
                child = np.clip(child, 0, 1)
                sum_c = np.sum(child)
                scale = (n_points / 2) / sum_c if sum_c != 0 else 1.0
                child = np.clip(child * scale, 0, 1)
                new_population.append(child)
            else:
                new_population.append(top_candidates[i])

        for child in new_population:
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    child[j] += np.random.normal(0, 0.05)
                    child[j] = np.clip(child[j], 0, 1)
            sum_c = np.sum(child)
            scale = (n_points / 2) / sum_c if sum_c != 0 else 1.0
            child = np.clip(child * scale, 0, 1)
        new_population = [np.clip(p, 0, 1) for p in new_population]

        population = [population[i] for i in sort_indices[:len(population)-len(new_population)]] + new_population
        while len(population) > n_points:
            population.pop()
        while len(population) < n_points:
            population.append(np.random.choice([0, 1], size=n_points))
            sum_rand = np.sum(population[-1])
            scale_rand = (n_points / 2) / sum_rand if sum_rand != 0 else 1.0
            population[-1] = np.clip(population[-1] * scale_rand, 0, 1)

        for p in population:
            c5 = compute_c5(p)
            if c5 < best_c5:
                best_h = p.copy()
                best_c5 = c5
        if best_c5 < 0.3807:
            break

    if best_c5 < 0.3807:
        return (best_h, best_c5, n_points)

    top_indices = np.argsort([compute_c5(p) for p in population])[:5]
    for idx in top_indices:
        p = population[idx].copy()
        def obj(x):
            h = x.copy()
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)
            corr = np.correlate(h, 1 - h, mode='full')
            values = corr * dx
            gamma = 100
            return (1/gamma) * np.log(np.sum(np.exp(gamma * values)))
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, p, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
        if res.success:
            p = res.x
            p = np.clip(p, 0, 1)
            sum_p = np.sum(p)
            scale_p = (n_points / 2) / sum_p if sum_p != 0 else 1.0
            p = np.clip(p * scale_p, 0, 1)
            population[idx] = p
            c5 = compute_c5(p)
            if c5 < best_c5:
                best_h = p.copy()
                best_c5 = c5
        else:
            population[idx] = np.clip(p, 0, 1)
            sum_p = np.sum(population[idx])
            scale_p = (n_points / 2) / sum_p if sum_p != 0 else 1.0
            population[idx] = np.clip(population[idx] * scale_p, 0, 1)
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
