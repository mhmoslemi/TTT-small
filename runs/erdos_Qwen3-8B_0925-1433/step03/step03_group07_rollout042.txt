    gbest = h.copy()

                scale = (n_points / 2) / sum_p
                p_new = np.clip(p_new * scale, 0, 1)

            c5 = compute_c5(particles[i])

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
