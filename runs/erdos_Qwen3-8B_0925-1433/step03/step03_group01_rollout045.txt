        if time.time() - start_time >= budget_s:
            break
        current_h = np.random.choice(structured_guesses)
        perturbation = np.random.normal(0, temp, n_points)
        perturbed = current_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale = (n_points / 2) / sum_perturbed
        perturbed = perturbed * scale
        new_c5 = compute_c5(perturbed)
        if new_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                best_h = perturbed.copy()
                best_c5 = new_c5
        temp *= cooling_rate
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x = np.clip(x * scale, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h
    best_h = best_h * scale
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
