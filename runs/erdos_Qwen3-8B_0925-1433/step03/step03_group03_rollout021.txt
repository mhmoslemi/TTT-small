def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1.0 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Initialize population
    initial_h = np.array(initial_h_values, dtype=np.float64)
    sum_initial = np.sum(initial_h)
    scale = target_sum / sum_initial
    initial_h = np.clip(initial_h * scale, 0, 1)
    population = [initial_h]
    for _ in range(20):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h_rand = initial_h + noise
        h_rand = np.clip(h_rand, 0, 1)
        sum_rand = np.sum(h_rand)
        scale_rand = target_sum / sum_rand
        h_rand = np.clip(h_rand * scale_rand, 0, 1)
        population.append(h_rand)

    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 800
    early_stop = 100
    best_C5 = float('inf')
    best_h = initial_h.copy()

    start_time = time.time()
    velocities = [np.zeros(n_points) for _ in population]
    for iter in range(max_iter):

        current_best = min(c5_values)
        best_idx = c5_values.index(current_best)
        if current_best < best_C5:
            best_C5 = current_best
            best_h = population[best_idx].copy()
        if iter > 0 and abs(best_C5 - current_best) < 1e-6:
