    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    assert abs(np.sum(h) - n_points / 2) < 1e-7, "Invalid initial sum"
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    max_iterations = 400
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        eps = 1e-8
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            sum_perturb = np.sum(h_perturb)
            if sum_perturb == 0:
                perturb_corr = 1e6
            else:
                scale_perturb = (n_points / 2) / sum_perturb
                h_perturb_clipped = np.clip(h_perturb * scale_perturb, 0, 1)
                corr_perturb = np.correlate(h_perturb_clipped, 1 - h_perturb_clipped, mode='full')
                perturb_corr = np.max(corr_perturb) * dx
            h_neg = h.copy()
            h_neg[i] -= eps
            sum_neg = np.sum(h_neg)
            if sum_neg == 0:
                perturb_neg_corr = 1e6
            else:
                scale_neg = (n_points / 2) / sum_neg
                h_neg_clipped = np.clip(h_neg * scale_neg, 0, 1)
                corr_neg = np.correlate(h_neg_clipped, 1 - h_neg_clipped, mode='full')
                perturb_neg_corr = np.max(corr_neg) * dx
            grad[i] = (perturb_corr - perturb_neg_corr) / (2 * eps)
        direction = -grad
        alpha = 1.0
        while alpha > 1e-8:
            trial_h = h + alpha * direction
            trial_h = np.clip(trial_h, 0, 1)
            sum_trial = np.sum(trial_h)
            if sum_trial == 0:
                trial_h = np.ones(n_points) * 0.5
            else:
                scale_trial = (n_points / 2) / sum_trial
                trial_h = np.clip(trial_h * scale_trial, 0, 1)
            corr_trial = np.correlate(trial_h, 1 - trial_h, mode='full')
            trial_max = np.max(corr_trial) * dx
            if trial_max < best_c5:
                h = trial_h.copy()
                best_c5 = trial_max
                break
            else:
                alpha *= 0.5
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.ones(n_points) * 0.5
        else:
            scale_h = (n_points / 2) / sum_h
            h = np.clip(h * scale_h, 0, 1)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
