            current_c5 = correlate(current_h, 1 - current_h, mode='full').max() * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    h_final = np.clip(best_h, 0, 1)
    sum_final = np.sum(h_final)
    scale_final = (n_points/2) / sum_final
    h_final = h_final * scale_final
    final_c5 = correlate(h_final, 1 - h_final, mode='full').max() * dx
    return (h_final, final_c5, n_points)
