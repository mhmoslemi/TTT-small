from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values and scale to meet integral constraint
    h_current = initial_h_values.copy()

        h_current = np.ones(n_points) * 0.5

    best_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx

    iteration = 0
    while time.time() - start_time < budget_s:

        target_lag = max_index
