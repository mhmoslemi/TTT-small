def run(seed=42, budget_s=600):

    velocities = []
    personal_bests = []
    personal_best_values = []
    # Start with initial_h_values
    particles.append(h.copy())
    velocities.append(np.random.rand(n_points) * 0.1)
    personal_bests.append(h.copy())

    # Add other particles
    for _ in range(num_particles - 1):
        # Create a random variation
        new_h = np.random.rand(n_points)
        sum_new = np.sum(new_h)
        scale_new = target_sum / sum_new
        new_h = np.clip(new_h * scale_new, 0, 1)
        particles.append(new_h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        personal_bests.append(new_h.copy())
        personal_best_values.append(float('inf'))
    # Initialize global best
    global_best = particles[0].copy()
    global_best_value = float('inf')
    # PSO parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    rho = 1000.0
    # Main loop
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = (inertia_weight * velocities[i] +
                 cognitive_coeff * r1 * (personal_bests[i] - particles[i]) +
                 social_coeff * r2 * (global_best - particles[i]))

            new_pos = np.clip(new_pos, 0, 1)
            # Compute sum and penalty

            penalty = rho * np.abs(sum_new - target_sum)

            corr = np.correlate(new_pos, 1 - new_pos, mode='full')
            current_max = np.max(corr) * dx
            objective = current_max + penalty
            # Update personal best
            if objective < personal_best_values[i]:
                personal_best_values[i] = objective
                personal_bests[i] = new_pos.copy()
            # Update global best
            if objective < global_best_value:

                global_best = new_pos.copy()
            # Update particles and velocities
            particles[i] = new_pos
            velocities[i] = v
        # Check for early termination if global best is very good
        if global_best_value < 0.3807:

        iteration += 1
    # Clip and rescale the global best to ensure sum is correct
    sum_gb = np.sum(global_best)
    scale_gb = target_sum / sum_gb
    final_best = np.clip(global_best * scale_gb, 0, 1)
    # Re-evaluate to get accurate C5
    corr_final = np.correlate(final_best, 1 - final_best, mode='full')
