import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 150
    population_size = 50
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.full_like(h, 0.5)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    # Initialize population
    if initial_h_values is not None:
        h_initial = initial_h_values.copy()
        h_initial = scale_h(h_initial)
        population = [h_initial]
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = scale_h(h)
            population.append(h)
    else:
        population = [np.random.rand(n_points) for _ in range(population_size)]
