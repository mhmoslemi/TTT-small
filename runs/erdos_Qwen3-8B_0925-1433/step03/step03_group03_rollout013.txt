import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / initial_sum
    initial_h = np.clip(initial_h_values * scale_factor, 0, 1)
    population = [initial_h.copy()]
    for _ in range(20):
        noise = np.random.normal(0, 0.1, n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 800
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    velocities = [np.zeros(n_points) for _ in population]
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx, n_points)
            fitness.append((c5, h))
        fitness.sort()
        gbest = fitness[0][1]
        for i in range(len(population)):
            h = population[i]
            v = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            v_new = omega * v + c1 * r1 * (gbest - h) + c2 * r2 * (gbest - h)
            velocities[i] = v_new
            h_new = h + v_new
            h_new = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_new)
            scale = (n_points / 2) / h_sum
            h_new = np.clip(h_new * scale, 0, 1)
            population[i] = h_new
        for c5, h in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
