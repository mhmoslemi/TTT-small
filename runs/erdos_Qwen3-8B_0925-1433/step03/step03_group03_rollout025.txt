import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    learning_rate = 0.01
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k_opt = np.argmax(corr)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k_opt < n_points:
                term1 = 1.0 - h[j + k_opt]
            else:
                term1 = 0.0
            x = j - k_opt
            if x >= 0 and x + k_opt < n_points:
                term2 = -h[x]
            else:
                term2 = 0.0
            grad[j] = term1 + term2
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        scale_new = (n_points / 2) / sum_h_new
        h_new = np.clip(h_new * scale_new, 0, 1)
        new_c5 = compute_c5(h_new, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
