import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = 250
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//3] = 0.75
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)

    # PSO parameters
    pop_size = 25
    inertia = 0.8
    cogn_coeff = 1.5
    soc_coeff = 1.5
    rho = 1000
    max_iter = 1000
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Initialize population
    particles = [h.copy() for _ in range(pop_size)]
    velocities = [np.zeros(n_points) for _ in range(pop_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_vals = [compute_c5(p) for p in particles]
    global_best = h.copy()
    global_best_val = best_c5

    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + cogn_coeff * r1 * (personal_best[i] - particles[i]) + soc_coeff * r2 * (global_best - particles[i])
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            penalty = rho * abs(sum_p - n_points / 2)
            # Compute C5
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            obj = c5 + penalty
            # Update personal best
            if obj < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = obj
        # Update global best
        for i in range(pop_size):
            if personal_best_vals[i] < global_best_val:
                global_best_val = personal_best_vals[i]

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
