import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    best_h = h.copy()

    temp = 1.0
    cooling_rate = 0.995
    max_iter = 5000
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale_perturbed = (n_points / 2) / sum_perturbed

        current_c5 = compute_c5(h, dx, n_points)
