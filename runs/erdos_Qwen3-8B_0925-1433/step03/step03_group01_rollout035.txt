        h_perturb = np.clip(h_perturb, 0, 1)
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constr}, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    return (best_h, best_c5, n_points)
