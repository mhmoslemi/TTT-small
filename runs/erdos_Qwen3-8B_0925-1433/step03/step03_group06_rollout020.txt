import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = 500
    if initial_h_values is not None:
        h = initial_h_values
    else:
        h = np.zeros(n_points)
        h[:n_points//3] = 0.7
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.3
        h = np.clip(h, 0, 1)
    h = np.clip(h * (n_points/2) / np.sum(h), 0, 1)
    candidates = [h.copy() for _ in range(25)]
    np.random.shuffle(candidates)
    pop_size = 25
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    rho = 1000
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        positions = np.array([c.copy() for c in candidates])
        velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
        personal_best = positions.copy()
        swarm_best = positions[0].copy()
        for i in range(pop_size):
            pos = positions[i]
            vel = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel = inertia * vel + cognitive * r1 * (personal_best[i] - pos) + social * r2 * (swarm_best - pos)
            new_pos = pos + vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            penalty = rho * abs(sum_new - n_points/2)
            corr = np.correlate(new_pos, 1 - new_pos, mode='full')
            obj = np.max(corr) * (2.0 / n_points) + penalty
            if obj < compute_c5(personal_best[i]):
                personal_best[i] = new_pos
            if obj < compute_c5(swarm_best):
                swarm_best = new_pos
            positions[i] = new_pos
            velocities[i] = vel
        if compute_c5(swarm_best) < best_c5:

            best_h = swarm_best.copy()

    best_h = np.clip(best_h * (n_points/2) / np.sum(best_h), 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
