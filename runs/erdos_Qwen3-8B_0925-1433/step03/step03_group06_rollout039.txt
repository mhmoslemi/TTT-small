import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (len(h)/2) / sum_h
    h = np.clip(h * scale, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    patience = 100
    no_improvement = 0
    
    while time.time() - start_time < budget_s:
        for _ in range(10):
            noise = np.random.normal(0, 0.05, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if sum_h_new == 0:
                continue
            scale_new = (len(h_new)/2) / sum_h_new
            h_scaled = np.clip(h_new * scale_new, 0, 1)
            current_c5 = compute_c5(h)
            new_c5 = compute_c5(h_scaled)
            if new_c5 < current_c5:
                h = h_scaled
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                no_improvement = 0
            else:

                    h = h_scaled
                    no_improvement = 0
                else:
                    no_improvement += 1
        if no_improvement > patience:

            no_improvement = 0
        temp *= cooling_rate
        iterations += 1
    
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
