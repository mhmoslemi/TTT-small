import numpy as np

from pre_imported import evaluate_erdos_solution, initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)

    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(h)
    start_time = time.time()
    current_temp = 0.5
    cooling_rate = 0.995
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            new_h = np.full_like(new_h, 0.5)
        else:
            scale = (n_points / 2) / sum_new
            new_h = new_h * scale
        new_c5 = evaluate_erdos_solution(new_h)
        current_c5 = evaluate_erdos_solution(h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
            h = new_h.copy()
        if new_c5 < best_c5:
            best_c5 = new_c5

        current_temp *= cooling_rate
