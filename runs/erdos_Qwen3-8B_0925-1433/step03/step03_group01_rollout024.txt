        if time.time() - start_time >= budget_s:
            break

    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        scale_perturb = (n_points / 2) / sum_perturb
        h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
        res = minimize(obj, h_perturb, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    return (best_h, best_c5, n_points)
