import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):

    global initial_h_values
    if initial_h_values is not None:

        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)

        n_points = 45
        h = np.zeros(n_points)

            if i % 3 == 0:
                h[i] = 0.8
            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0.2
        sum_h = np.sum(h)
        scale = (len(h)/2) / sum_h
        h = np.clip(h * scale, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    particles = [h.copy() for _ in range(20)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(20)]
    pbest = [p.copy() for p in particles]
    gbest = particles[0].copy()
    best_c5 = compute_c5(gbest)
    inertias = np.linspace(0.9, 0.4, 20)
    
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        for i in range(20):
            h = particles[i].copy()
            v = velocities[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < compute_c5(pbest[i]):
                pbest[i] = h.copy()
                compute_c5(pbest[i])

                best_h = h.copy()
                best_c5 = c5

            phi1 = np.random.rand()

            v = inertias[i] * v + 1.5 * phi1 * (pbest[i] - h) + 1.5 * phi2 * (gbest - h)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
