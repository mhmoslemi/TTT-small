        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1)] * n_points

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        try:
            res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_c5 = objective(res.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x.copy()
        except:
            pass
        if time.time() - start_time >= budget_s:
            break

    while time.time() - start_time < budget_s:
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        perturb_h = best_h + perturb
        perturb_h = np.clip(perturb_h, 0, 1)
        sum_perturb = np.sum(perturb_h)
        scale_perturb = (n_points / 2) / sum_perturb
        perturb_h_scaled = np.clip(perturb_h * scale_perturb, 0, 1)
        try:
            res_perturb = minimize(objective, perturb_h_scaled, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res_perturb.success:
                current_c5 = objective(res_perturb.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res_perturb.x.copy()
        except:
            pass
        if time.time() - start_time >= budget_s:
            break

    sum_best = np.sum(best_h)
    scale_best = (n_points / 2) / sum_best
    best_h = np.clip(best_h * scale_best, 0, 1)

    return (best_h, best_c5, n_points)
