import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 0.25
        h = np.clip(h, 0, 1)
    n_points = len(h)
    # Rescale to meet the integral constraint
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    # Initialize population
    population = [h.copy()]
    for _ in range(5):
        pert = np.random.normal(0, 0.05, n_points)
        candidate = h + pert
        candidate = np.clip(candidate, 0, 1)
        sum_c = np.sum(candidate)
        scale_c = (n_points / 2) / sum_c
        candidate = candidate * scale_c
        population.append(candidate)
    # Evaluate initial candidates
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    def constraint(x):
        return np.sum(x) - n_points / 2
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': constraint}]
    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Select top candidates from population
        pop_candidates = [c.copy() for c in population]
        for i in range(5):
            candidate = pop_candidates[i]
            res = minimize(obj, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x.copy()
        # Evolve population
        population.sort(key=lambda x: obj(x))
        population = population[:5]
        # Generate new candidates
        for _ in range(5):
            pert = np.random.normal(0, 0.1, n_points)
            new_candidate = best_h + pert
            new_candidate = np.clip(new_candidate, 0, 1)
            sum_c = np.sum(new_candidate)
            scale_c = (n_points / 2) / sum_c
            new_candidate = new_candidate * scale_c
            population.append(new_candidate)
    return (best_h, best_c5, n_points)
