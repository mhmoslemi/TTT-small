    personal_best = [p.copy() for p in particles]
    swarm_best = min(particles, key=compute_c5)
    best_h = particles[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    rho = 1000
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (swarm_best - particles[i])
            new_position = particles[i] + new_velocity
            new_position = np.clip(new_position, 0, 1)
            sum_new = np.sum(new_position)
            penalty = rho * abs(sum_new - n_points / 2)
            corr = np.correlate(new_position, 1 - new_position, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            objective = current_c5 + penalty
            if objective < compute_c5(particles[i]) + rho * abs(np.sum(particles[i]) - n_points / 2):
                particles[i] = new_position.copy()
                velocities[i] = new_velocity.copy()
                personal_best[i] = new_position.copy()
        new_swarm_best = min(particles, key=compute_c5)
        if compute_c5(new_swarm_best) < compute_c5(swarm_best):
            swarm_best = new_swarm_best
            best_c5 = compute_c5(swarm_best)
            best_h = swarm_best.copy()
        for i in range(pop_size):
            if np.random.random() < 0.05:
                perturb = np.random.normal(0, 0.05, n_points)
                new_p = particles[i] + perturb
                new_p = np.clip(new_p, 0, 1)
                sum_new = np.sum(new_p)
                penalty = rho * abs(sum_new - n_points / 2)
                corr = np.correlate(new_p, 1 - new_p, mode='full')
                current_c5 = np.max(corr) * (2.0 / n_points)
                objective = current_c5 + penalty
                if objective < compute_c5(particles[i]) + rho * abs(np.sum(particles[i]) - n_points / 2):
                    particles[i] = new_p.copy()
                    velocities[i] = np.random.rand(n_points) * 0.1
    final_sum = np.sum(best_h)
    scale = (n_points / 2) / final_sum
    best_h = np.clip(best_h * scale, 0, 1)
    return (best_h, best_c5, n_points)
