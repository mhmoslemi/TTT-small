            mutant = h_target + F * (h1 - h2)
            mutant = np.clip(mutant, 0, 1)
            trial = h_target.copy()
            for i in range(len(trial)):
                if np.random.rand() < CR:
                    trial[i] = mutant[i]
            sum_trial = np.sum(trial)
            if sum_trial == 0:
                trial = np.full_like(trial, 0.5)
            else:
                scale = (n_points / 2) / sum_trial
                trial = np.clip(trial * scale, 0, 1)
            trial_c5 = compute_c5(trial)
            target_c5 = compute_c5(h_target)
            if trial_c5 < target_c5:
                new_pop.append(trial)
            else:
                new_pop.append(h_target)
        pop = new_pop
        for h in pop:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if iteration > 1000:
            break
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
