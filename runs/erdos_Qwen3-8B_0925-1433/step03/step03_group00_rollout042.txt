import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def create_even_spacing():
        h = np.zeros(n_points)
        h[::2] = 1.0
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        return np.clip(h, 0, 1)

    def create_sine_wave():
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        return np.clip(h, 0, 1)

    def create_binary():
        h = np.random.choice([0, 1], size=n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        return np.clip(h, 0, 1)

    initial_pop = [initial_h_values.copy()]
    for _ in range(20):
        h = initial_h_values + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h else 1.0
        h = np.clip(h * scale, 0, 1)
        initial_pop.append(h)

    for _ in range(10):
        initial_pop.append(create_even_spacing())
    for _ in range(10):
        initial_pop.append(create_sine_wave())
    for _ in range(10):
        initial_pop.append(create_binary())

    pop = np.array([np.array(p) for p in initial_pop])
    pop = np.clip(pop, 0, 1)
    best_idx = np.argmin([compute_c5(p) for p in pop])
    best_h = pop[best_idx].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        scores = [compute_c5(p) for p in pop]
        top_idx = np.argsort(scores)[:20]
        new_pop = pop[top_idx].copy()
        for _ in range(20):
            p1, p2 = np.random.choice(top_idx, 2, replace=False)
            child = 0.8 * pop[p1] + 0.2 * pop[p2]
            child = np.clip(child, 0, 1)
            sum_h = np.sum(child)
            scale = (n_points / 2) / sum_h if sum_h else 1.0
            child = np.clip(child * scale, 0, 1)
            new_pop = np.vstack([new_pop, child])
        pop = new_pop
        for _ in range(10):
            idx = np.random.randint(len(pop))
            h = pop[idx].copy()
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h else 1.0
            h = np.clip(h * scale, 0, 1)
            pop[idx] = h

        scores = [compute_c5(p) for p in pop]
        best_idx = np.argmin(scores)
        best_h = pop[best_idx].copy()
        best_c5 = scores[best_idx]

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        top_idx = np.argsort([compute_c5(p) for p in pop])[:5]
        for idx in top_idx:
            h = pop[idx].copy()
            def obj(x):
                h = x.copy()
                sum_h = np.sum(h)
                scale = (n_points / 2) / sum_h if sum_h else 1.0
                h = np.clip(h * scale, 0, 1)
                corr = np.correlate(h, 1 - h, mode='full')
                gamma = 100

            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
            if res.success:
                pop[idx] = res.x
        scores = [compute_c5(p) for p in pop]
        best_idx = np.argmin(scores)
        best_h = pop[best_idx].copy()
        best_c5 = scores[best_idx]

    return (best_h, best_c5, n_points)
