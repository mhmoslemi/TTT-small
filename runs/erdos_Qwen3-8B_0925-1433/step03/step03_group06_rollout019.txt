import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (len(h)/2) / sum_h
    h = np.clip(h * scale, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (len(x)/2) / sum_x
        x = np.clip(x * scale, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0., 1.) for _ in range(n_points)]
    
    try:
        result = differential_evolution(objective, bounds, strategy='best1bin',
                                       maxiter=100, popsize=15, tol=1e-6,
                                       mutation=(0.5, 1), recombination=0.8,
                                       seed=seed, polish=True, disp=False,
                                       vectorized=True)
        best_h = result.x
        best_c5 = objective(best_h)
    except:
        pass
    
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
