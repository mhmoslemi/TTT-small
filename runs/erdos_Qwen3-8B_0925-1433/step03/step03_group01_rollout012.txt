import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values
        h = np.clip(h, 0, 1)
        scale = (len(h)/2) / np.sum(h)

        h = np.clip(h * scale, 0, 1)
    dx = 2.0 / n_points

    def rescale(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = (n_points / 2) / s
        return np.clip(x * scale, 0, 1)

    def get_max_corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr)

    candidates = [h]

    candidates.append(rescale(0.5 + 0.5 * np.abs(np.sin(np.pi * x))))

    best_h = candidates[0].copy()
    best_c5 = get_max_corr(best_h) * dx

    start_time = time.time()

    cooling_rate = 0.9995
    iter_count = 0
