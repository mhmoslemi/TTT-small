            term1 = 1.0 - h[j] if 0 <= j < n_points else 0.0
            m = i - k_max
            term2 = h[m] if 0 <= m < n_points else 0.0
            grad[i] = (term1 - term2) * dx
        h -= eta * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
