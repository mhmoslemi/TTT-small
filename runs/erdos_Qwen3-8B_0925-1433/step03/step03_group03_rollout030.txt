        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        k_samples = max_corr_index - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + k_samples
            if i_plus < n_points:
                grad[i] += 1 - h[i_plus]
            i_minus = i - k_samples
            if i_minus >= 0:
                grad[i] -= h[i_minus]
        h = h - learning_rate * grad
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iteration % 100 == 0 and learning_rate > 0.001:
            learning_rate *= 0.95
        iteration += 1
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
