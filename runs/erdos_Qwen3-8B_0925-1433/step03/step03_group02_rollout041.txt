import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values)
    scale = (n_points / 2) / sum_initial
    h = np.clip(initial_h_values * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    population = [h.copy() for _ in range(20)]
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(ind, dx, n_points) for ind in population]
        sorted_idx = np.argsort(fitness)
        top_idx = sorted_idx[:10]
        
        new_pop = []
        for i in top_idx:
            h_cand = population[i].copy()
            noise = np.random.normal(0, 0.02, n_points)
            h_perturb = h_cand + noise
            sum_perturb = np.sum(h_perturb)
            if sum_perturb == 0:
                h_perturb = np.ones(n_points) * 0.5
            else:
                scale_perturb = (n_points / 2) / sum_perturb
                h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
            new_pop.append(h_perturb)
        
        population = population[:10] + new_pop
        
        for i in range(10):
            h_refine = population[i].copy()
            def obj(x):
                x_sum = np.sum(x)
                if x_sum == 0:
                    return 1e6
                scale_x = (n_points / 2) / x_sum
                x_clipped = np.clip(x * scale_x, 0, 1)
                return compute_c5(x_clipped, dx, n_points)
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            res = minimize(obj, h_refine, bounds=bounds, constraints=cons, method='SLSQP')
            if res.success:
                cand = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = cand.copy()
    
    return (best_h, best_c5, n_points)
