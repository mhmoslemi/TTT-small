import numpy as np
from skopt import BayesSearch
from skopt.utils import use_named_args
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        n_points = len(initial_h_values)
    else:
        n_points = 450
    dx = 2.0 / n_points
    N = 10
    if initial_h_values is not None:
        x = np.linspace(0, 2, n_points)
        a = np.zeros(N)
        b = np.zeros(N)
        for n in range(1, N+1):
            a[n-1] = dx * np.sum(initial_h_values * np.cos(n * np.pi * x))
            b[n-1] = dx * np.sum(initial_h_values * np.sin(n * np.pi * x))
        theta_init = np.concatenate((a, b))
    else:
        theta_init = np.random.uniform(-0.5, 0.5, 2*N)

    def obj(theta):
        a_theta = theta[:N]
        b_theta = theta[N:]
        x = np.linspace(0, 2, n_points)
        h = 0.5 + np.sum([a_theta[i] * np.cos((i+1)*np.pi*x) + b_theta[i] * np.sin((i+1)*np.pi*x) for i in range(N)], axis=0)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    @use_named_args
    def objective(**kwargs):
        theta = np.array([kwargs[f'a_{i}'] for i in range(N)] + [kwargs[f'b_{i}'] for i in range(N)])
        return obj(theta)

    bounds = [(-0.5, 0.5)] * (2*N)
    optimizer = BayesSearch(objective, bounds, n_iter=200)
    optimizer.run()
    best_theta = optimizer.x
    a_best = best_theta[:N]
    b_best = best_theta[N:]
    x = np.linspace(0, 2, n_points)
    h = 0.5 + np.sum([a_best[i] * np.cos((i+1)*np.pi*x) + b_best[i] * np.sin((i+1)*np.pi*x) for i in range(N)], axis=0)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h_scaled = h * scale
    best_c5 = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
    return (h_scaled, best_c5, n_points)
