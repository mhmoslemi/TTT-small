import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 45
        h = np.zeros(n)
        h[:n//3] = 0.75
        h[n//3:2*n//3] = 0.5
        h[2*n//3:] = 0.25
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = np.clip(h * (n / 2) / sum_h, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    iterations = 0
    candidates = [h.copy()]
    
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constrain(x):
        return np.sum(x) - n_points / 2
    
    while time.time() - start_time < budget_s:
        if len(candidates) < 5:
            candidates.append(np.random.uniform(0, 1, n_points))
            candidates[-1] = np.clip(candidates[-1] * (n_points / 2) / np.sum(candidates[-1]), 0, 1)
        for h in candidates:
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': constrain}])
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        new_h = best_h + np.random.normal(0, 0.05, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = np.clip(new_h * (n_points / 2) / np.sum(new_h), 0, 1)
        candidates[0] = new_h
        iterations += 1
    
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
