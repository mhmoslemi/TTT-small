import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def objective(x, n_points, dx):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x = np.clip(x * scale, 0, 1)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    def generate_initial_population(size):
        pop = []
        for _ in range(size):
            if np.random.rand() < 0.6:
                h = initial_h_values.copy()
            elif np.random.rand() < 0.3:
                h = np.random.uniform(0, 1, n_points)
            else:
                h = np.sin(np.linspace(0, np.pi, n_points)) + 1
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5
            pop.append(h)
        return pop

    pop_size = 50
    population = generate_initial_population(pop_size)

    start_time = time.time()

        new_population = []
        for h in population:
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5

            def obj(x):
                return objective(x, n_points, dx)

            cons = [{'type': 'eq', 'fun': constraint}]
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x

                sum_h_opt = np.sum(h_opt)
                if sum_h_opt != 0:
                    scale = (n_points / 2) / sum_h_opt
