import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        population = [h.copy() for _ in range(50)]
    else:
        population = []
        for _ in range(50):
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
            population.append(h.copy())
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        new_pop = []
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            best = population[np.argmin([compute_c5(sol) for sol in population])]
            mutant = best + F * (a - b)
            mutant = np.clip(mutant, 0, 1)
            trial = np.copy(best)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            sum_t = np.sum(trial)
            if sum_t == 0:
                trial = np.full_like(trial, 0.5)
            else:
                scale = (n_points / 2) / sum_t
                trial = np.clip(trial * scale, 0, 1)
            c5 = compute_c5(trial)
            if c5 < compute_c5(population[i]):
                new_pop.append(trial)
            else:
                new_pop.append(population[i])
        population = new_pop
        for sol in population:
            c5 = compute_c5(sol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = sol.copy()
        if gen % 50 == 0:
            F *= 0.98
            CR += 0.01
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
