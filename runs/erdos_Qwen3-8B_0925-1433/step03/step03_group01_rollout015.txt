import numpy as np
from skopt import BayesianOptimization
from skopt.utils import use_named_args
from scipy.optimize import minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = (h * (n_points / 2) / np.sum(h)).clip(0, 1)
        fft = np.fft.fft(h)
        N = 10
        freqs = np.fft.fftshift(np.fft.fftfreq(n_points, d=dx))
        a = np.zeros(N)
        b = np.zeros(N)
        for i in range(N):
            idx = np.argmin(np.abs(freqs - (i+1)*np.pi))
            a[i] = fft[idx].real
            b[i] = fft[idx].imag
        theta = np.concatenate([a, b])
    else:
        N = 10
        theta = np.random.uniform(-0.5, 0.5, 2*N)
    
    def objective(theta):
        a, b = theta[:N], theta[N:]
        x = np.linspace(0, 2, n_points)
        h = 0.5 + np.sum([a[i] * np.cos((i+1)*np.pi*x) + b[i] * np.sin((i+1)*np.pi*x) for i in range(N)], axis=0)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    @use_named_args
    def opt_func(**params):
        theta = np.array([params[f'theta_{i}'] for i in range(2*N)])
        return objective(theta)
    
    bounds = [(-0.5, 0.5)] * (2 * N)
    opt = BayesianOptimization(opt_func, {f'theta_{i}': bounds[i] for i in range(2*N)})
    opt.minimize(n_iter=150)
    best_theta = opt.x
    a_best, b_best = best_theta[:N], best_theta[N:]
    x = np.linspace(0, 2, n_points)
    h = 0.5 + np.sum([a_best[i] * np.cos((i+1)*np.pi*x) + b_best[i] * np.sin((i+1)*np.pi*x) for i in range(N)], axis=0)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return (h, c5, n_points)
