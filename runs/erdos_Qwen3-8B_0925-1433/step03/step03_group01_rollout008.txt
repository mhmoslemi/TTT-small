    else:
        n = 500
        h = np.random.rand(n)
        sum_h = np.sum(h)
        scale = (n/2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = get_max_corr(best_h) * (2.0 / len(best_h))
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    
    def generate_sine(n):
        x = np.linspace(0, 2, n)
        wave = 0.5 * (1 + np.sin(np.pi * x))
        sum_wave = np.sum(wave)
        scale = (n / 2) / sum_wave
        return np.clip(wave * scale, 0, 1)
    
    def generate_triangular(n):
        x = np.linspace(0, 2, n)
        wave = np.clip(2 * (x - 1), -1, 1)
        sum_wave = np.sum(wave)
        scale = (n / 2) / sum_wave
        return np.clip(wave * scale, 0, 1)
    
    def generate_gaussian(n):
        x = np.linspace(0, 2, n)
        wave = np.exp(-((x - 1)/0.5)**2)
        sum_wave = np.sum(wave)
        scale = (n / 2) / sum_wave
        return np.clip(wave * scale, 0, 1)
    
    def generate_alternating(n):
        wave = np.zeros(n)
        wave[::2] = 0.5
        wave[1::2] = 1.0
        sum_wave = np.sum(wave)
        scale = (n / 2) / sum_wave
        return np.clip(wave * scale, 0, 1)
    
    structured_guesses = [generate_sine(n_points), generate_triangular(n_points),
                          generate_gaussian(n_points), generate_alternating(n_points)]
    if initial_h_values is not None:

    for guess in structured_guesses:
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:

def get_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)
