import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 120
    pop_size = 50
    F = 0.8
    CR = 0.9
    best_h = None
    best_c5 = float('inf')
    
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        pop = [h.copy() for _ in range(pop_size)]
    else:
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            pop.append(h.copy())
    
    for h in pop:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            indices = np.random.choice(pop_size, 3, replace=False)
            v1, v2, v3 = pop[indices[0]], pop[indices[1]], pop[indices[2]]
            mutant = v3 + F * (v1 - v2)
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            sum_t = np.sum(trial)
            if sum_t == 0:
                trial = np.full_like(trial, 0.5)
            else:
                scale = (n_points / 2) / sum_t
                trial = np.clip(trial * scale, 0, 1)
            t_c5 = compute_c5(trial)
            if t_c5 < compute_c5(pop[i]):
                pop[i] = trial.copy()
                if t_c5 < best_c5:
                    best_c5 = t_c5
                    best_h = trial.copy()
        if any(np.abs(compute_c5(h) - best_c5) < 1e-4 for h in pop):
            break
    return (best_h, best_c5, n_points)
