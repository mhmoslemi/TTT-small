                return np.sum(x) - target_sum
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(objective, h, method='COBYLA', constraints={'type': 'eq', 'fun': constraint}, bounds=bounds)
            if result.success:
                population[i] = result.x
            else:
                population[i] = h
        for ind in population:
            current_c5 = compute_c5(ind, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
