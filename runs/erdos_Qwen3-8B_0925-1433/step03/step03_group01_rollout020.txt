import time
from scipy.optimize import minimize

    best_h = h.copy()
    best_c5 = get_max_corr(best_h) * dx
    start_time = time.time()

    def rescale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return np.zeros_like(x)
        scale = (n_points / 2) / sum_x
        return np.clip(x * scale, 0, 1)

    def compute_corr(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    def energy(h_val):
        return compute_corr(h_val)

    def perturb(h, temp):
        noise = np.random.normal(0, temp, n_points)
        return rescale(h + noise)

def get_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
