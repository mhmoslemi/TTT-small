import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n = 200
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_scaled = x * scale
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        maxiter=300, popsize=15, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, 
        seed=seed, disp=False
    )
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h
    best_h = np.clip(best_h * scale, 0, 1)
    final_c5 = compute_c5(best_h)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(5):
            perturb = np.random.normal(0, 0.08, n_points)
            h_trial = best_h + perturb
            h_trial = np.clip(h_trial, 0, 1)
            sum_trial = np.sum(h_trial)
            scale_trial = (n_points / 2) / sum_trial
            h_trial = np.clip(h_trial * scale_trial, 0, 1)
            c5_trial = compute_c5(h_trial)
            if c5_trial < final_c5:
                final_c5 = c5_trial
                best_h = h_trial
    
    return (best_h, final_c5, n_points)
