import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    result = differential_evolution(lambda x: compute_c5(x), bounds, constraints=constraints, strategy='best1bin', 
                                    popsize=50, mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=int(budget_s / 3))
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
