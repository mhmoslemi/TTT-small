import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0
    initial_h = np.array(initial_h_values)
    if np.abs(np.sum(initial_h) - sum_required) > 1e-6:
        initial_h *= sum_required / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    best_h = initial_h.copy()
    best_c5 = obj(initial_h)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]

    def scale_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return h
        return h * (sum_required / np.sum(h))

    start_time = time.time()
    for variation in [0.1, 0.15, 0.2]:
        n_fft = n_points // 2 + 1
        freqs = np.linspace(0, 0.5, n_fft)
        H = np.ones(n_fft)
        low_mask = (variation <= freqs) & (freqs <= variation + 0.05)
        H[low_mask] = 0.0
        h_fft = np.fft.irfft(H)
        h_fft = scale_h(h_fft)
        result = minimize(obj, h_fft, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 100, 'disp': False})
        h_best = result.x
        h_best = scale_h(h_best)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5
        if time.time() - start_time > budget_s:
            break
