from scipy.optimize import minimize

        initial_h = initial_h * (n_points / 2) / total

    dx = 2.0 / n_points

    def obj(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_initial_guesses():

        # Trigonometric polynomial
        x = np.linspace(0, 2, n_points)
        c = 0.5 * (1 + np.cos(np.pi * x))
        total_c = np.sum(c)
        c *= (n_points / 2) / total_c
        guesses.append(c)
        # Random binary pattern
        r = np.random.rand(n_points)
        r = (r > 0.5).astype(float)
        total_r = np.sum(r)
        r *= (n_points / 2) / total_r
        guesses.append(r)
        # Periodic alternating pattern
        p = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                p[i] = 1.0
            else:
                p[i] = 0.0
        total_p = np.sum(p)
        if total_p != 0:
            p *= (n_points / 2) / total_p
        guesses.append(p)
        return guesses

    def evaluate_guess(h):
        h = np.clip(h, 0.0, 1.0)
        total_h = np.sum(h)
        if abs(total_h - n_points / 2) > 1e-6:

    start_time = time.time()
    best_h = initial_h.copy()

    guesses = generate_initial_guesses()
