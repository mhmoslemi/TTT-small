import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)

        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iters = 0
    
    def objective(x):
        h_candidate = x.astype(float)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate == 0:
            return 1e6
        scale = (n_points / 2) / sum_candidate
        h_candidate = np.clip(h_candidate * scale, 0, 1)
        shifted = 1 - h_candidate
        corr = np.correlate(h_candidate, shifted, mode='full')
        lag_index = np.argmax(corr)
        val = corr[lag_index] * dx
        return val
    
    while time.time() - start_time < budget_s and iters < 100:
        iters += 1
        
        # Perturb and scale
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_pert = h + perturb
        h_pert = np.clip(h_pert, 0, 1)
        sum_pert = np.sum(h_pert)
        if sum_pert == 0:
            h_pert = np.ones(n_points) * 0.5
        else:
            scale = (n_points / 2) / sum_pert
            h_pert = np.clip(h_pert * scale, 0, 1)
        
        current_c5 = compute_c5(h_pert, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_pert.copy()
        
        # Local optimization
        res = minimize(objective, best_h, bounds=[(0,1) for _ in range(n_points)], constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}, method='SLSQP', tol=1e-6)
        if res.success:
            h = res.x
        else:
            h = best_h.copy()
        
        # Random explore
        if np.random.rand() < 0.1:
            perturb = np.random.uniform(-0.2, 0.2, size=n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.ones(n_points) * 0.5
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
