import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Initialize population with perturbed clones
    population_size = 20
    population = [initial_h_values.copy() for _ in range(population_size)]
    for i in range(population_size):
        perturb = np.random.normal(0, 0.05, n_points)
        pop = initial_h_values + perturb
        pop = np.clip(pop, 0, 1)
        sum_pop = np.sum(pop)
        if sum_pop != 0:
            scale = (n_points / 2) / sum_pop
            population[i] = pop * scale
        else:
            population[i] = np.ones(n_points) * 0.5
    best_c5 = float('inf')
    best_h = population[0].copy()
    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        # Evaluate and sort population
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        sorted_indices = np.argsort(fitness)
        # Select top 10 parents
        parents = [population[i] for i in sorted_indices[:10]]
        # Generate offspring
        offspring = []
        for i in range(0, len(parents), 2):
            if i+1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                # Uniform crossover
                child = np.random.choice([p1, p2], size=n_points)
                # Mutate
                mutation = np.random.normal(0, 0.02, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child != 0:
                    scale = (n_points / 2) / sum_child
                    child = child * scale
                else:
                    child = np.ones(n_points) * 0.5
                offspring.append(child)
        # Replace worst half with offspring
        population = parents[:10] + offspring
        np.random.shuffle(population)
        # Track best
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')

            if c5 < best_c5:

                best_h = h.copy()
        # Early stopping
        if generation > 100 and np.isclose(best_c5, best_c5, atol=1e-10):

        generation += 1
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
