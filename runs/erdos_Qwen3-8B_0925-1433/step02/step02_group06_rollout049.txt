            check_h[i] = 0
    total = np.sum(check_h)
    check_h = check_h * (n_points / 2) / total
    initial_guesses.append(check_h)

    # Random perm
    if n_points % 2 == 0:
        part = [1] * (n_points // 2) + [0] * (n_points // 2)
        np.random.shuffle(part)
        initial_guesses.append(part)
    else:
        part = [1] * (n_points // 2 + 1) + [0] * (n_points // 2)
        np.random.shuffle(part)
        part = part[:n_points]
        total = np.sum(part)
        part = part * (n_points / 2) / total
        initial_guesses.append(part)

    # Random noise
    noise = np.random.rand(n_points)
    noise = (noise > 0.5).astype(float)
    total = np.sum(noise)
    if total != 0:
        noise = noise * (n_points / 2) / total
    else:
        noise = np.full(n_points, 0.5)
    initial_guesses.append(noise)

    best_h = initial_h.copy()
    best_c5 = obj(initial_h)

    for h_start in initial_guesses:
        h_start = np.clip(h_start, 0.0, 1.0)
        total = np.sum(h_start)
        if total != 0:
            h_start = h_start * (n_points / 2) / total
        else:
            h_start = np.full(n_points, 0.5)
        res = minimize(obj, h_start, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'ftol': 1e-8, 'maxiter': 500})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        else:
            current_h = np.clip(h_start, 0.0, 1.0)
            total = np.sum(current_h)
            if total != 0:
                current_h = current_h * (n_points / 2) / total
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = np.clip(best_h, 0.0, 1.0)
    total_h = np.sum(best_h)
    if abs(total_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
