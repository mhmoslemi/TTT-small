    np.random.seed(seed)
    n_points = 480
    required_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_initial, initial_h)
        h = h * required_sum / np.sum(h)
        initial_guesses.append(h)
    else:
        x = np.linspace(0, 2, n_points)
        sym = 0.5 * (1 + np.cos(np.pi * x))
        sym = sym * required_sum / np.sum(sym)
        initial_guesses.append(sym)
        def gen_periodic(T):
            h = 0.5 * (1 + np.cos(2 * np.pi * x / T))
            h = h * required_sum / np.sum(h)
            return h
        T_values = [2/3, 2/4, 2/5, 2/6]
        for T in T_values:
            initial_guesses.append(gen_periodic(T))
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * required_sum / np.sum(h)
        initial_guesses.append(h)
    best_c5 = float('inf')
    best_h = None
    elapsed = 0
    for h_initial in initial_guesses:
        start_time = time.time()
        h = h_initial.copy()
        h = np.clip(h, 0, 1)
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            return np.max(cc * dx)
        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum
        bounds = [(0, 1)] * n_points
        result = minimize(objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_eq}], options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8})
        elapsed += time.time() - start_time
        if elapsed > budget_s - 10:
            break
        if result.success:
            candidate = np.clip(result.x.reshape(n_points), 0, 1)
            candidate = candidate * required_sum / np.sum(candidate)
            cc = correlate(candidate, 1 - candidate, mode='full')
            c5 = np.max(cc * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
        else:
            candidate = np.clip(h_initial, 0, 1)
            candidate = candidate * required_sum / np.sum(candidate)
            cc = correlate(candidate, 1 - candidate, mode='full')
            c5 = np.max(cc * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
    return (best_h, best_c5, n_points)
