import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():

        return np.max(corr) * dx

    pop_size = 30

    c1 = 1.5
    c2 = 1.5
    max_iter = 200
    velocity_limit = 0.1

    particles = [h.copy()]

    pbest = [h.copy()]
    pbest_fitness = [obj(h)]

    gbest_fitness = obj(h)

    for _ in range(pop_size - 1):
        h_new = np.random.rand(n_points)
        h_new = (h_new > 0.5).astype(float)
        total = np.sum(h_new)
        h_new = h_new * (n_points / 2) / total
        particles.append(h_new)
        velocities.append(np.random.rand(n_points) * velocity_limit * 2 - velocity_limit)
        pbest.append(h_new.copy())
        pbest_fitness.append(obj(h_new))

            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
