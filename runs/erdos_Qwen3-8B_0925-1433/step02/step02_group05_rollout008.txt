import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Generate diverse initial guesses
    guesses = [initial_h_values.copy()]
    for period in [3, 4]:
        temp = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                temp[i] = 1.0
        temp = temp * (sum_required / np.sum(temp))
        temp = np.clip(temp, 0, 1)
        guesses.append(temp)

    x = np.linspace(0, 2, n_points)
    sine_pattern = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_pattern = np.clip(sine_pattern, 0, 1)
    sine_pattern = sine_pattern * (sum_required / np.sum(sine_pattern))
    guesses.append(sine_pattern)

    cosine_pattern = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    cosine_pattern = np.clip(cosine_pattern, 0, 1)
    cosine_pattern = cosine_pattern * (sum_required / np.sum(cosine_pattern))
    guesses.append(cosine_pattern)

    random_pattern = np.random.rand(n_points)
    random_pattern = random_pattern * (sum_required / np.sum(random_pattern))
    random_pattern = np.clip(random_pattern, 0, 1)
    guesses.append(random_pattern)

    # Optimize each guess
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - sum_required) > 1e-6:
            h_best = h_best * (sum_required / sum_h)
            h_best = np.clip(h_best, 0, 1)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5

    return (best_h, best_c5, n_points)
