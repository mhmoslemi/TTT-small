import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / s
        return np.clip(h * scale, 0, 1)
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    h = normalize(initial_h_values.copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    particles = [normalize(np.random.rand(n_points)) for _ in range(50)]
    velocities = np.random.uniform(-0.1, 0.1, size=(50, n_points))
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        for p in range(50):

            h_p = normalize(h_p)
            particles[p] = h_p
        
        costs = np.array([compute_c5(p) for p in particles])
        best_idx = np.argmin(costs)

        if best_c5 < best_c5:

        if best_c5 < 0.3807:
