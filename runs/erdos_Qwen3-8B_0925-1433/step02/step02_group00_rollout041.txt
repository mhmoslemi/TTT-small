import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in locals() else None

            scale = (len(h)/2) / sum_h

        n = 150
        h = np.zeros(n)

            if i < n//3:
