import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-3:

    h = np.clip(h, 0, 1)
    population = [h.copy() for _ in range(20)]
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        fitness = [compute_c5(c, dx, n_points) for c in population]
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:len(population)//2]]
        new_population = []
        for i in range(len(selected) // 2):
            parent1 = selected[i]
            parent2 = selected[-i-1]
            child = (parent1 + parent2) * 0.5
            noise = np.random.uniform(-0.1, 0.1, n_points)
            child += noise
            sum_child = np.sum(child)
            if sum_child == 0:

            new_population.append(child)
        population = selected + new_population
        top_candidates = population[:len(population)//4]
        for c in top_candidates:
            def obj(x):
                h_opt = x.copy()
                sum_h = np.sum(h_opt)
                if sum_h == 0:
                    return 1e6
                scale = (n_points / 2) / sum_h

                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
