import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial_h_values
    h = initial_h_values.copy()
    total = np.sum(h)
    scale = (n_points / 2) / total if total > 0 else 1.0
    h = h * scale
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total != 0:
        scale = (n_points / 2) / total
        h = h * scale

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    alpha = 0.1
    decay_rate = 0.9995
    max_iterations = 10000
    mutation_prob = 0.1

    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)

        g = (1 - h[(np.arange(n_points) + lag) % n_points]) * dx
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0, 1)
        total_new = np.sum(h_new)
        if total_new != 0:
            scale = (n_points / 2) / total_new
        else:
            scale = 1.0
        h_new = h_new * scale

        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h.copy()

        alpha *= decay_rate

        if np.random.rand() < mutation_prob:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if total != 0:
                scale = (n_points / 2) / total
            else:
                scale = 1.0
            h = h * scale

    return (best_h, best_c5, n_points)
