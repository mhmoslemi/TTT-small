    n_points = 480
    dx = 2.0 / n_points
    required_sum = n_points / 2
    best_c5 = float('inf')
    best_h = None

    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_initial, initial_h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (required_sum / sum_h)
        initial_guesses.append(h)
    else:
        # Generate symmetric, periodic, and binary initial guesses
        x = np.linspace(0, 2, n_points)
        # Symmetric
        h_sym = 0.5 * (1 + np.cos(np.pi * x))
        h_sym = np.clip(h_sym, 0, 1)
        sum_h_sym = np.sum(h_sym)
        h_sym = h_sym * (required_sum / sum_h_sym)
        initial_guesses.append(h_sym)
        # Periodic with T=0.5
        h_period = 0.5 * (1 + np.cos(2 * np.pi * x / 0.5))
        h_period = np.clip(h_period, 0, 1)

        initial_guesses.append(h_period)

        initial_guesses.append(h_bin)

    # Objective and constraints

        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        c_values = cc * dx
        return np.max(c_values)

    def constraint_eq(h_flat):
        return np.sum(h_flat.reshape(n_points)) - required_sum

    bounds = [(0, 1)] * n_points
