    h_cos = 0.5 * (1 + np.cos(np.pi * x))
    total = np.sum(h_cos) * (2.0 / n_points)
    h_cos = h_cos * (n_points / 2) / total
    initial_guesses.append(h_cos)

    x = np.linspace(0, 2, n_points)
    h_tri = np.zeros_like(x)
    h_tri[:n_points//2] = x[:n_points//2]
    h_tri[n_points//2:] = 2 - x[n_points//2:]
    total = np.sum(h_tri) * (2.0 / n_points)
    h_tri = h_tri * (n_points / 2) / total
    initial_guesses.append(h_tri)

    r = np.random.rand(n_points)
    r = (r > 0.5).astype(float)
    total_r = np.sum(r) * (2.0 / n_points)
    r = r * (n_points / 2) / total_r
    initial_guesses.append(r)

    p = np.zeros(n_points)
    for i in range(n_points):
        pos = dx * i
        part = int(pos / (n_points / 3))
        if part % 2 == 0:
            p[i] = 1
        else:
            p[i] = 0
    total_p = np.sum(p) * (2.0 / n_points)
    if total_p != 0:
        p = p * (n_points / 2) / total_p
    initial_guesses.append(p)

    best_h = initial_h
    best_c5 = obj(initial_h)

    for h_start in initial_guesses:
        h_start = np.clip(h_start, 0.0, 1.0)
        sum_h = np.sum(h_start)
        if sum_h != n_points / 2:
            scale = (n_points / 2) / sum_h
            h_start *= scale
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
        else:
            current_c5 = obj(np.clip(h_start, 0.0, 1.0))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_start

    best_h = np.clip(best_h, 0.0, 1.0)
    total_h = np.sum(best_h)
    if abs(total_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
