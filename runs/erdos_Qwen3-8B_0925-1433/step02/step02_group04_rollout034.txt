from scipy.optimize import minimize
from scipy.signal import correlate

        h *= (n_points / 2.0)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):

    bounds = [(0.0, 1.0) for _ in range(n_points)]
