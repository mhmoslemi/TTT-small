from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_dim = len(h)

            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, h_dim), h)
        h /= (np.sum(h) * dx) * (n_points / 2.0)
        h *= (n_points / 2.0)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        # Ensure sum(h) matches
        sum_h = np.sum(h) * dx
        h *= (n_points / 2.0) / sum_h

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 500,
        'ftol': 1e-10,
        'eps': 1e-8,
        'disp': False,
        'iprint': -1
    }
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
    cc = correlate(best_h, 1 - best_h, mode='full')

    best_h = np.clip(best_h, 0.0, 1.0)
