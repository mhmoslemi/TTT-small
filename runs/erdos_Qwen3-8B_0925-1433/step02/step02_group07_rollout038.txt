            particles[i] = particles[i] * required_sum / np.sum(particles[i])
        pbest[i] = particles[i].copy()
        cc = correlate(particles[i], 1 - particles[i], mode='full')
        pbest_obj[i] = np.max(cc * dx)
    best_idx = np.argmin(pbest_obj)
    gbest = pbest[best_idx].copy()
    gbest_obj = pbest_obj[best_idx]
    omega = 0.729
    phi1 = 1.494
    phi2 = 1.494
    max_iter = 500
    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + phi1 * r1 * (pbest[i] - particles[i]) + phi2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * required_sum / np.sum(particles[i])
            cc = correlate(particles[i], 1 - particles[i], mode='full')
            current_obj = np.max(cc * dx)
            if current_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = current_obj
        best_idx = np.argmin(pbest_obj)
        if pbest_obj[best_idx] < gbest_obj:
            gbest = pbest[best_idx].copy()
            gbest_obj = pbest_obj[best_idx]
    best_h = gbest
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * required_sum / np.sum(best_h)
    return (best_h, c5_bound, n_points)
