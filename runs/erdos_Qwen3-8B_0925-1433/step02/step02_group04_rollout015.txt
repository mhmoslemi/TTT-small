from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    n_points = 256

    if 'initial_h_values' in globals() and initial_h_values is not None:

        initial_h_dim = len(initial_h)

        initial_h /= (np.sum(initial_h) * dx) * (n_points / 2.0)

        sum_h = np.sum(initial_h) * dx
        initial_h *= (n_points / 2.0) / sum_h

    def objective(h):

        return np.max(cc) * dx

    def constraint_sum(h):

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        'disp': False,
        'iprint': -1
    }
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)
