import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 480
    required_sum = n_points / 2
    dx = 2.0 / n_points
    initial_guesses = []

    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h_values)
        h_initial = h_initial * required_sum / np.sum(h_initial)
        initial_guesses.append(h_initial)

    def make_binary():
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * required_sum / np.sum(h)
        return h

    def make_smooth():
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1.0 - np.cos(np.pi * x))
        h = h * required_sum / np.sum(h)
        return h

    initial_guesses.append(make_binary())
    initial_guesses.append(make_smooth())

    best_h = None
    best_c5 = float('inf')

    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        bounds = [(0, 1)] * n_points

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            return np.max(cc * dx)

        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum

        try:
            result = minimize(
                objective,
                guess.flatten(),
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_eq}],
                options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
            )
        except:
            continue

        if result.success:
            current_h = result.x.reshape(n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * required_sum / np.sum(current_h)
            cc = correlate(current_h, 1 - current_h, mode='full')
            c5 = np.max(cc * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h

    if best_h is None:
        best_h = make_binary()
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    return (best_h, c5_bound, n_points)
