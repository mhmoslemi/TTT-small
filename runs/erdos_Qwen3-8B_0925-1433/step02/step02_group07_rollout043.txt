from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 480
    dx = 2.0 / n_points
    required_sum = n_points / 2
    guesses = []

    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        current_sum = np.sum(h_initial)
        h_initial = h_initial * required_sum / current_sum
        guesses.append(h_initial)
    
    # Binary pattern
    binary = np.zeros(n_points)
    binary[::2] = 1.0
    guesses.append(binary)
    
    # Smooth pattern
    x = np.linspace(0, 2, n_points)
    smooth = 0.5 * (1 - np.cos(np.pi * (x - 1.0))) + 0.5
    smooth = smooth * 0.5
    guesses.append(smooth)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    for initial_guess in guesses:
        initial_guess = np.clip(initial_guess, 0, 1)
        current_sum = np.sum(initial_guess)
        initial_guess = initial_guess * required_sum / current_sum
        
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            c_values = cc * dx
            return np.max(c_values)
        
        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum
        
        bounds = [(0.0, 1.0)] * n_points
        initial_flat = initial_guess.flatten()
        
        result = minimize(
            fun=objective,
            x0=initial_flat,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False}
        )
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        
        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            current_sum = np.sum(h_opt)

            cc = correlate(h_opt, 1 - h_opt, mode='full')
