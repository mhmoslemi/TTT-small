import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        n = 150
        base = np.zeros(n)
        for i in range(n):
            if i % 3 == 0:
                base[i] = 0.6
            elif i % 3 == 1:
                base[i] = 0.3
            else:
                base[i] = 0.1
        sum_base = np.sum(base)
        scale = (n/2) / sum_base
        h = np.clip(base * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_points = len(h)
    dx = 2.0 / n_points

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points/2) / sum_x
            x_scaled = np.clip(x * scale, 0, 1)
            corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return np.max(corr) * dx

        res = minimize(obj, h, method='COBYLA', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
        if res.success:
            h = res.x.copy()

        if iteration % 50 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (n_points/2) / sum_h
                h = np.clip(h * scale, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
