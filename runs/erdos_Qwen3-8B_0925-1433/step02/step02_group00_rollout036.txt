import numpy as np
import time
from scipy.optimize import basinhopping, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)

    minimizer_kwargs = {
        'method': 'SLSQP',
        'bounds': [(0, 1)] * n_points,

    start_time = time.time()

    best_h = res.x

    sum_best = np.sum(best_h)
    if sum_best == 0:

    elapsed = time.time() - start_time
