import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total > 0 else 1.0
        return h * scale

    h = initial_h_values.copy()
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    max_iterations = 500
    learning_rate = 0.1
    decay_rate = 0.999
    no_improvement_count = 0
    max_no_improvement = 300
    improvement_threshold = 1e-5

    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_index = np.argmax(corr)
        k = lag_index - (n_points - 1)

        grad = np.zeros(n_points)
        if k >= 0:
            for i in range(n_points - k):
                j = i + k
                grad[i] = (1 - h[j]) * dx
        else:
            m = -k
            for i in range(m, n_points):
                j = i + k
                grad[i] = (1 - h[j]) * dx

        h_new = h - learning_rate * grad
        h_new = clip_and_normalize(h_new)

        current_c5 = evaluate_erdos_solution(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                break

        learning_rate *= decay_rate

        if _ % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = clip_and_normalize(h + noise)
        else:
            h = h_new

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = clip_and_normalize(best_h)
    return (best_h, evaluate_erdos_solution(best_h), n_points)
