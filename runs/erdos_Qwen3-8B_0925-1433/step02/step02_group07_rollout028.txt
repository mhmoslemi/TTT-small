import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 480
    required_sum = n_points / 2
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    structured = []

    # Symmetric pattern
    h_sym = 0.5 * (1 + np.cos(np.pi * x))
    structured.append(h_sym)

    # Periodic patterns
    for T in [1.0, 0.5, 0.25]:
        h_per = 0.5 * (1 + np.cos(2 * np.pi * x / T))
        structured.append(h_per)

    # Binary patterns
    h_bin = np.zeros(n_points)
    h_bin[:n_points//2] = 1.0
    structured.append(h_bin)
    h_bin2 = np.zeros(n_points)
    h_bin2[::2] = 1.0
    structured.append(h_bin2)

    if 'initial_h_values' in globals():

        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, xi, initial_h_values)
        structured.append(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc * dx)

    def constraint_eq(h_flat):
        return np.sum(h_flat.reshape(n_points)) - required_sum

    bounds = [(0, 1)] * n_points
    best_c5 = float('inf')
    best_h = None

    for guess in structured:
        guess = np.clip(guess, 0, 1)
        guess_sum = np.sum(guess)
        if guess_sum > 0:
            guess = guess * (required_sum / guess_sum)
        initial = guess.flatten()
        start = time.time()
        res = minimize(objective, initial, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_eq}], options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False})
        elapsed = time.time() - start
        if elapsed > budget_s - 10:
            continue
        if res.success:
            h = res.x.reshape(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (required_sum / h_sum)
            cc = correlate(h, 1 - h, mode='full')
            c5 = np.max(cc * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    if best_h is None:
        best_h = h_sym
    cc = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(cc * dx)
    return (best_h, best_c5, n_points)
