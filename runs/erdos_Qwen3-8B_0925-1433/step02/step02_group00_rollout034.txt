        best_c5 = compute_c5(best_h)
    
    # Perturb and optimize
    for _ in range(3):
        if time.time() - start >= budget_s:
            break
        perturb = np.random.uniform(-0.05, 0.05, half)
        variables = np.clip(res.x + perturb, 0, 1)
        res = minimize(objective, variables, method='SLSQP', bounds=bounds, constraints=constraints)
        if res.success:
            h_opt = np.zeros(m)
            h_opt[:half] = res.x
            for i in range(half):
                h_opt[m - 1 - i] = res.x[i]
            h_opt = np.clip(h_opt, 0, 1)
            sum_h_opt = np.sum(h_opt)
            if sum_h_opt == 0:
                h_opt = np.full_like(h_opt, 0.5)
            else:
                scale = (m / 2) / sum_h_opt
                h_opt = np.clip(h_opt * scale, 0, 1)
            best_c5 = compute_c5(h_opt)
            best_h = h_opt
    
    return (best_h, best_c5, n_points)
