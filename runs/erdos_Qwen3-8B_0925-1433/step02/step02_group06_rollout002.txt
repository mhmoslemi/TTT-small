import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

        n_points = 95

        initial_h[::2] = 1.0
        initial_h[1::2] = 0.0
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guesses = [initial_h.copy()]
    
    # Fallback periodic pattern
    period = int(n_points / 3)
    p = np.zeros(n_points)

        if part % 2 == 0:
            p[i] = 1
        else:
            p[i] = 0
    total_p = np.sum(p)
    if total_p != 0:
        p *= (n_points / 2) / total_p
    initial_guesses.append(p)
    
    # Fallback cosine-like curve
    x = np.linspace(0, 2, n_points)

    initial_guesses.append(c)
    
    # Fallback random binary pattern

    interval_length = 1.0 / m

    best_h = initial_h
