import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h_dim = len(h)

        h *= (n_points / 2.0)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 500,
        'ftol': 1e-8,
        'eps': 1e-8,

        'iprint': -1
