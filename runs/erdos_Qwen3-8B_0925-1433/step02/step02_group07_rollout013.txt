            cc = correlate(best_h, 1 - best_h, mode='full')
            c5_bound = np.max(cc * dx)
            if c5_bound < best_c5:
                best_c5 = c5_bound

    # Fallback to symmetric initial guess if no improvement
    if best_h is None:
        best_h = initial_guesses[0]
        cc = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(cc * dx)

    # Final accuracy check
    cc = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(cc * dx)

    return (best_h, best_c5, n_points)
