    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    temp = 0.5
    cool = 0.995
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        # Sensitivity computation
        eps = 1e-5
        sens = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            sum_p = np.sum(h_plus)
            if sum_p != 0:
                h_plus = np.clip(h_plus * ((n_points / 2) / sum_p), 0, 1)
            else:
                h_plus = np.ones(n_points) * 0.5
            c5p = compute_c5(h_plus, dx, n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            sum_m = np.sum(h_minus)
            if sum_m != 0:
                h_minus = np.clip(h_minus * ((n_points / 2) / sum_m), 0, 1)
            else:
                h_minus = np.ones(n_points) * 0.5
            c5m = compute_c5(h_minus, dx, n_points)
            
            sens[i] = (c5p - c5m) / (2 * eps)
        
        # Select top 20%
        top = np.argsort(sens)[-int(0.2 * n_points):]
        perturb = np.zeros(n_points)
        perturb[top] = np.random.normal(0, 0.05, len(top))
        
        # Apply perturbation & normalize
        h_candidate = h + perturb
        sum_c = np.sum(h_candidate)
        if sum_c != 0:
            h_candidate = np.clip(h_candidate * (n_points / 2) / sum_c, 0, 1)
        else:
            h_candidate = np.ones(n_points) * 0.5
        
        new_c5 = compute_c5(h_candidate, dx, n_points)
        
        # Acceptance
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5)/temp):
            best_c5 = new_c5
            best_h = h_candidate.copy()
        
        # Cool down
        temp *= cool
        
        # Random all-perturb every 500 steps
        if iter_count % 500 == 0:
            perturb_all = np.random.normal(0, 0.05, n_points)
            h_candidate = h + perturb_all
            sum_c = np.sum(h_candidate)
            if sum_c != 0:
                h_candidate = np.clip(h_candidate * (n_points / 2) / sum_c, 0, 1)
            else:
                h_candidate = np.ones(n_points) * 0.5
            new_c5 = compute_c5(h_candidate, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_candidate.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
