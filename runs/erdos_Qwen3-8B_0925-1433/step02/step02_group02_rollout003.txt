import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)

    dx = 2.0 / n_points

    else:

    population.append(base_h)
    
    for _ in range(population_size - 1):
        h = base_h + np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.ones(n_points) * 0.5
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        population.append(h)
    
    best_h = base_h.copy()

        max_iter -= 1

        idxs = np.argsort(fitness)
        top = idxs[:int(len(population)*0.2)]
        top_h = [population[i] for i in top]
        
        # Generate new population
        new_pop = []
        for _ in range(population_size):

            child = np.clip(child, 0, 1)

def compute_c5(h, dx, n_points):
