import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values) * dx
    if abs(sum_initial - 1.0) > 1e-6:
        initial_h_values *= 1.0 / sum_initial
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def create_guess(low, high, trans='linear'):
        freqs = np.fft.rfftfreq(n_points, dx)
        H = np.zeros(len(freqs))
        if trans == 'linear':
            for i, f in enumerate(freqs):
                if f < low: H[i] = 0.0
                elif f > high: H[i] = 1.0
                else: H[i] = (f - low) / (high - low)
        elif trans == 'cos':
            for i, f in enumerate(freqs):
                if f < low: H[i] = 0.0
                elif f > high: H[i] = 1.0
                else: H[i] = 0.5 * (1 + np.cos(np.pi * (f - low) / (high - low)))
        h = np.fft.irfft(H)
        h = np.clip(h, 0, 1)
        s = np.sum(h) * dx
        if s > 1e-12:
            h *= 1.0 / s
        return h

    guesses = [create_guess(0.2, 0.8, 'linear'), create_guess(0.2, 0.8, 'cos'),
               create_guess(0.15, 0.85, 'linear'), create_guess(0.25, 0.75, 'linear'),
               initial_h_values]

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        guess = np.clip(guess, 0, 1)
        sum_guess = np.sum(guess) * dx
        if abs(sum_guess - 1.0) > 1e-6:
            guess *= 1.0 / sum_guess
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-8, 'maxiter': 200})
        h = np.clip(res.x, 0, 1)
        s_h = np.sum(h) * dx
        if abs(s_h - 1.0) > 1e-6:
            h *= 1.0 / s_h
        h = np.clip(h, 0, 1)
        c5 = obj(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return (best_h, best_c5, n_points)
