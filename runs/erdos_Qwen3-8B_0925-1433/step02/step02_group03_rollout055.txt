    dx = 2.0 / n_points

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total > 0 else 1.0
        return h * scale

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()

    learning_rate = 0.1
    max_iterations = 400
    for iteration in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_lag = i + lag
            term1 = (1 - h[i_plus_lag]) * dx if 0 <= i_plus_lag < n_points else 0.0
            i_minus_lag = i - lag
            term2 = -h[i_minus_lag] * dx if 0 <= i_minus_lag < n_points else 0.0
            gradient[i] = term1 + term2
        h = h - learning_rate * gradient
        h = clip_and_normalize(h)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = clip_and_normalize(h)
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n_points)
