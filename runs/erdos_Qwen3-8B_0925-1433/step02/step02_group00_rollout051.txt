    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)
            if res.success:
                x0 = res.x
            else:
                break
        except:
            break

    best_h = np.zeros(m)
    best_h[:half] = x0
    best_h[half:] = x0[::-1]
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        scale = (m / 2) / sum_h
        best_h = best_h * scale
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, m)
