    best_c5 = 0.0
    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.995
    total_iterations = 1000
    for iteration in range(total_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            i_plus = (j + lag) % n_points
            i_minus = (j - lag) % n_points
            gradient[j] = dx * ((1 - h[i_plus]) - h[i_minus])
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        total_current = np.sum(h)
        scale = (n_points / 2) / total_current if total_current > 0 else 1.0
        h *= scale
        current_c5 = max_val * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                h = np.clip(h + noise, 0, 1)
                total_current = np.sum(h)
                scale = (n_points / 2) / total_current if total_current > 0 else 1.0
                h *= scale
        learning_rate *= decay_rate
    h = np.clip(h, 0, 1)
    total_current = np.sum(h)
    scale = (n_points / 2) / total_current if total_current > 0 else 1.0
    h *= scale
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, best_c5, n_points)
