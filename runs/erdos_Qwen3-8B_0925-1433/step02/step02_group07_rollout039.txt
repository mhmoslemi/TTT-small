import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    required_sum = n_points / 2
    dx = 2.0 / n_points
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_initial, initial_h)
        h_interp = h_interp * required_sum / np.sum(h_interp)
        particles = np.random.rand(20, n_points)
        particles = particles * required_sum / np.sum(particles, axis=1, keepdims=True)
        particles = np.vstack([h_interp, particles])
    else:
        particles = np.random.rand(20, n_points)
        particles = particles * required_sum / np.sum(particles, axis=1, keepdims=True)
    velocities = np.zeros((20, n_points))
    pbest = particles.copy()
    pbest_obj = np.full(20, np.inf)
    gbest = particles[0].copy()
    gbest_obj = np.inf
    def evaluate(h):
        return np.max(correlate(h, 1 - h, mode='full') * dx)
    for _ in range(300):
        t0 = time.time()
        obj = np.array([evaluate(p) for p in particles])
        for i in range(20):
            if obj[i] < pbest_obj[i]:
                pbest_obj[i] = obj[i]
                pbest[i] = particles[i].copy()
        if np.min(obj) < gbest_obj:
            gbest_obj = np.min(obj)
            gbest = particles[np.argmin(obj)].copy()
        r1, r2 = np.random.rand(20), np.random.rand(20)
        velocities = 0.729*velocities + 1.494*r1*(pbest - particles) + 1.494*r2*(gbest - particles)

        particles = particles * required_sum / np.sum(particles, axis=1, keepdims=True)
        if time.time() - t0 > budget_s - 10:
