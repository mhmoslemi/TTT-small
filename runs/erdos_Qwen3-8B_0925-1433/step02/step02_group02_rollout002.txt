            if sum_offspring == 0:
                offspring = np.full(n_points, 0.5)
            else:
                scale = (n_points / 2) / sum_offspring
                offspring = np.clip(offspring * scale, 0, 1)
            new_population.append(offspring)
        
        population = new_population
        
        for h in top_candidates:
            def objective(x):
                x = x.astype(float)
                sum_x = np.sum(x)
                if sum_x == 0:
                    return 1e6
                scale = (n_points / 2) / sum_x
                x = np.clip(x * scale, 0, 1)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            res = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                h[:] = res.x
        
        if best_c5 < 0.3807:
            break
    
    return (best_h, best_c5, n_points)
