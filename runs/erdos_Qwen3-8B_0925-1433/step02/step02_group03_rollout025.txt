import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial_h_values
    initial_h = np.clip(initial_h_values, 0, 1)
    total = np.sum(initial_h)

    h = initial_h * scale
    h = np.clip(h, 0, 1)

        return np.max(corr) * dx

        total = np.sum(h)
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)

    def objective(h):
        return evaluate_c5(h)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break
        # Perturb the current best solution
        perturb = np.random.normal(0, 0.015, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = normalize(perturbed)

        res = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            current_h = res.x
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

        # Add random starting points
        random_start = np.random.rand(n_points)
        random_start = np.clip(random_start, 0, 1)

        if res.success:

            current_c5 = evaluate_c5(current_h)

                best_h = current_h

    best_h = normalize(best_h)
