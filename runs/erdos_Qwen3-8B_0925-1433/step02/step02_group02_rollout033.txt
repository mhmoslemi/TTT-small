import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values and scale
    def scale_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / np.sum(h)
        return np.clip(h * scale, 0, 1)
    
    h = scale_h(initial_h_values.copy())
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb and scale
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_candidate = scale_h(h + perturb)
        current_c5 = compute_c5(h_candidate, dx, n_points)
        
        # Local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        def objective(x):
            h_candidate = x.astype(float)
            sum_h = np.sum(h_candidate)
            if sum_h == 0:
                return 1e6
            scale = (n_points / 2) / sum_h
            h_candidate = np.clip(h_candidate * scale, 0, 1)
            shifted = 1 - h_candidate
            corr = np.correlate(h_candidate, shifted, mode='full')
            lag_index = np.argmax(corr)
            return corr[lag_index] * dx
        
        res = minimize(objective, h_candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-5)
        if res.success:
            h = res.x
        else:
            h = h_candidate
        
        # Update best solution
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Random exploration
        if np.random.rand() < 0.02:
            h = scale_h(np.random.uniform(0, 1, n_points))
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
