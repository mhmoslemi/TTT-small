        h_rand = np.random.rand(n_points)
        h_rand = (h_rand > 0.5).astype(float)
        sum_rand = np.sum(h_rand)
        h_rand = h_rand * (n_points / 2) / sum_rand
        initial_guesses.append(h_rand)

    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    elapsed = 0

    for guess in initial_guesses:
        start_time = time.time()
        result = cma.fmin(obj=objective, x0=guess, sigma=0.5, options={'maxiter': 100, 'ftol': 1e-6})
        elapsed += time.time() - start_time
        if elapsed >= budget_s:
            break
        if result is not None and result[1] < best_c5:
            best_h = result[0]
            best_c5 = result[1]

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    if sum_best != 0:
        scale = (n_points / 2) / sum_best
        best_h = best_h * scale
    else:
        best_h = h_initial
        best_c5 = objective(best_h)

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_corr = np.max(corr)
    best_c5 = max_corr * dx
    return (best_h, best_c5, n_points)
