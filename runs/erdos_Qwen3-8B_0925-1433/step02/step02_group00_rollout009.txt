import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    if initial_h_values is not None:
        h_initial = initial_h_values.copy()
        h_initial = np.clip(h_initial, 0, 1)

        h_initial = np.clip(h_initial * scale, 0, 1)
        initial_guesses = [h_initial]
        for _ in range(4):
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturb = np.clip(h_initial + perturb, 0, 1)
            sum_h = np.sum(h_perturb)
            scale = (n_points / 2) / sum_h
            h_perturb = np.clip(h_perturb * scale, 0, 1)
            initial_guesses.append(h_perturb)
    else:
        h_default = np.full(n_points, 1.0 / n_points)
        sum_h = np.sum(h_default)
        scale = (n_points / 2) / sum_h
        h_default = np.clip(h_default * scale, 0, 1)
        initial_guesses = [h_default]
    
    start_time = time.time()
    for idx, h0 in enumerate(initial_guesses):
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1)] * n_points

        if res.success:
