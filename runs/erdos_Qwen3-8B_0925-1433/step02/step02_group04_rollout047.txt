from scipy.optimize import minimize
from scipy.signal import correlate

        h *= (n_points / 2.0)
    else:
        h = np.full(n_points, 0.5)

        h[1::spacing] = 1.0
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        h *= (n_points / 2.0) / sum_h

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        'disp': False,

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
