import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    guesses = [initial_h_values]
    for _ in range(5):
        rng = np.random.RandomState(seed)
        rand = rng.rand(n_points)
        rand = (rand / np.sum(rand)) * sum_required
        guesses.append(rand)
    for i in range(n_points):
        if i % 3 == 0:
            guesses.append(np.zeros(n_points))
            guesses[-1][i] = 1.0
    x = np.linspace(0, 2, n_points)
    guesses.append(0.5 + 0.5 * np.sin(np.pi * x))
    guesses.append(0.5 + 0.5 * np.cos(2 * np.pi * x))
    for i in range(n_points):
        if i % 2 == 0:
            guesses.append(np.zeros(n_points))
            guesses[-1][i] = 1.0

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        try:
            result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                              options={'ftol': 1e-8, 'maxiter': 100, 'disp': False})
            h_best = result.x
            h_best = np.clip(h_best, 0, 1)
            if np.abs(np.sum(h_best) - sum_required) > 1e-6:

            c5 = obj(h_best)
            if c5 < best_c5:
                best_h = h_best
                best_c5 = c5
        except:
            pass
