from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        sum_h = np.sum(h) * dx
        h *= (n_points / 2.0) / sum_h

    def objective(h):

        return np.max(cc) * dx

        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
