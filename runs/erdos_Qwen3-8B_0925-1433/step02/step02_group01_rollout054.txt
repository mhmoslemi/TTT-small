import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Optimization
        def objective(x):
            h_opt = x.copy()
            sum_h = np.sum(h_opt)
            if sum_h == 0:
                return 1e6
            scale = (n_points / 2) / sum_h
            h_opt = np.clip(h_opt * scale, 0, 1)
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            return np.max(corr) * dx
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-4)
        if res.success:
            h = res.x

                h = np.clip(h * scale, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
