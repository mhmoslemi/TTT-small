from scipy.optimize import minimize
import time

        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guess = initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess *= (n_points / 2) / np.sum(initial_guess)
    initial_guess = np.clip(initial_guess, 0, 1)

    guesses = [initial_guess.copy()]

    # Generate structured patterns
    def make_triangle_pattern(n):
        x = np.linspace(0, 2, n)
        return 0.5 + 0.5 * np.abs(np.sin(np.pi * x / 1.0))
    
    def make_sawtooth_pattern(n):
        x = np.linspace(0, 2, n)
        return 0.5 + 0.5 * (x % 1.0) / 1.0

    def make_modulated_pattern(n):
        freq = 3
        x = np.linspace(0, 2, n)
        return 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    
    def make_perturbed_pattern(base, std=0.05):
        return np.clip(base + np.random.normal(0, std, n_points), 0, 1)

    for pattern_func in [make_triangle_pattern, make_sawtooth_pattern, make_modulated_pattern]:
        guess = pattern_func(n_points)
        guess = np.clip(guess, 0, 1)
        guess *= (n_points / 2) / np.sum(guess)
        guess = np.clip(guess, 0, 1)
        guesses.append(guess)

    for _ in range(2):
        guess = make_perturbed_pattern(initial_guess)
        guesses.append(guess)

    best_h = initial_guess
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:

        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,

        h_best = result.x
