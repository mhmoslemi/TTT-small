    def validate_initial_h():
        sum_h = np.sum(initial_h_values)
        if np.isclose(sum_h, n_points / 2, atol=1e-6):
            return np.clip(initial_h_values, 0, 1)
        scale = (n_points / 2.0) / sum_h
        return np.clip(initial_h_values * scale, 0, 1)

    initial_guess = validate_initial_h()

    len_rfft = n_points // 2 + 1
    freqs = np.linspace(0, 0.5, len_rfft)
    H = np.zeros(len_rfft)
    a, b = 0.15, 0.25
    for k in range(len_rfft):
        f = freqs[k]
        if f < a:
            H[k] = 0
        elif f > b:
            H[k] = 1
        else:
            H[k] = (f - a) / (b - a)
    h_fft = np.fft.irfft(H)
    h_fft = np.clip(h_fft, 0, 1)
    sum_h_fft = np.sum(h_fft) * dx
    scale_fft = (1.0) / sum_h_fft
    freq_initial_guess = h_fft * scale_fft

    def obj(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = initial_guess.copy()
    best_c5 = obj(best_h)

    result = minimize(obj, freq_initial_guess, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
    h_best = np.clip(result.x, 0, 1)
    sum_h = np.sum(h_best)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_best = h_best * (n_points / 2) / sum_h
        h_best = np.clip(h_best, 0, 1)
    c5 = obj(h_best)
    if c5 < best_c5:
        best_h = h_best
        best_c5 = c5

    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = initial_guess.copy()
        best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
