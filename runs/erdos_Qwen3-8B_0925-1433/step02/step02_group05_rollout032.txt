import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    guesses = [initial_h_values.copy()]
    
    # Random pattern
    rand = np.random.rand(n_points)
    rand *= sum_required / np.sum(rand)
    rand = np.clip(rand, 0, 1)
    guesses.append(rand)
    
    # Sine wave
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = np.clip(sine, 0, 1)
    sine *= sum_required / np.sum(sine)
    sine = np.clip(sine, 0, 1)
    guesses.append(sine)
    
    # Cosine wave
    cosi = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    cosi = np.clip(cosi, 0, 1)
    cosi *= sum_required / np.sum(cosi)
    cosi = np.clip(cosi, 0, 1)
    guesses.append(cosi)
    
    # Square wave
    sqw = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0:
            sqw[i] = 1.0
    sqw *= sum_required / np.sum(sqw)
    sqw = np.clip(sqw, 0, 1)
    guesses.append(sqw)
    
    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 100, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - sum_required) > 1e-6:

        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
