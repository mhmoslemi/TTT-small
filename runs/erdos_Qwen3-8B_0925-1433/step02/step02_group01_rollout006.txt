            sensitivity[i] = (dp - c5) / eps
        
        # Select top 25% indices by sensitivity
        top_indices = np.argsort(sensitivity)[-int(n_points * 0.25):]
        
        # Generate new candidate
        h_candidate = h.copy()
        noise = np.random.normal(0, 0.05, size=n_points)
        h_candidate[top_indices] += noise[top_indices]
        sum_hc = np.sum(h_candidate)
        if sum_hc == 0:
            h_candidate = np.ones(n_points) * 0.5
        else:
            scale_c = (n_points / 2) / sum_hc
            h_candidate = np.clip(h_candidate * scale_c, 0, 1)
        
        # Acceptance criteria
        corr_c = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        max_corr_c = np.max(corr_c)
        c5_c = max_corr_c * dx
        if c5_c < c5:
            h = h_candidate
        else:
            T = 0.5 * (1 - iter_count / (budget_s))
            if np.random.rand() < np.exp(-(c5_c - c5) / T):
                h = h_candidate
        
        if iter_count % 200 == 0:
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
