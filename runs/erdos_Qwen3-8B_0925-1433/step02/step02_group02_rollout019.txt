import numpy as np
import time
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()

        else:
            candidate = np.random.uniform(0, 1, n_points)
        sum_candidate = np.sum(candidate)
        if sum_candidate != 0:
            scale = (n_points / 2) / sum_candidate
            candidate = np.clip(candidate * scale, 0, 1)
        else:
            candidate = np.ones(n_points) * 0.5

                scale = (n_points / 2) / sum_child

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
