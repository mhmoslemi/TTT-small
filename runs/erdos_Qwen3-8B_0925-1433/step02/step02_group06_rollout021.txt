import numpy as np
import cma
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 95
        initial_h = np.full(n_points, 0.5)
        initial_h[::2] = 1.0
        initial_h[1::2] = 0.0
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def fitness(x):
        h = np.clip(x, 0.0, 1.0)
        sum_h = np.sum(h)
        penalty = 100.0 * abs(sum_h - n_points / 2)
        c5_val = compute_c5(h)
        return c5_val + penalty

    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)

    initial_guesses = [initial_h]

    # Add structured initial guesses
    x = np.linspace(0, 2, n_points)
    trig_func = 0.5 + 0.5 * np.cos(np.pi * x / 1.2)
    trig_func = np.clip(trig_func, 0, 1)
    total_trig = np.sum(trig_func)
    trig_func = trig_func * (n_points / 2) / total_trig
    initial_guesses.append(trig_func)

    # Random binary pattern
    np.random.seed(seed)
    rand_binary = np.random.rand(n_points)
    rand_binary = (rand_binary > 0.5).astype(float)
    total_rand = np.sum(rand_binary)
    rand_binary = rand_binary * (n_points / 2) / total_rand
    initial_guesses.append(rand_binary)

    # Sparse binary pattern
    alt_guess = np.zeros(n_points)
    alt_guess[::3] = 1.0
    alt_guess[1::3] = 0.5
    alt_guess[2::3] = 0.0
    total_alt = np.sum(alt_guess)
    alt_guess = alt_guess * (n_points / 2) / total_alt
    initial_guesses.append(alt_guess)

    for guess in initial_guesses:
        guess = np.clip(guess, 0.0, 1.0)
        try:
            sigma = 0.1 * np.std(guess)
            es = cma.CMAEvolutionStrategy(guess, sigma, {'bounds': [(0.0, 1.0) for _ in range(n_points)]})
            for _ in range(100):
                es.optimize(fitness, iter_cb=lambda _: None)
                best_x = es.bestsolution
                best_h_candidate = np.clip(best_x, 0.0, 1.0)
                sum_candidate = np.sum(best_h_candidate)
                if abs(sum_candidate - n_points / 2) < 1e-6:
                    current_c5 = compute_c5(best_h_candidate)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = best_h_candidate
        except:
            pass

    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
