import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h_dim = len(h)
        if h_dim != n_points:
            x_new = np.linspace(0, 2, n_points)
            x_old = np.linspace(0, 2, h_dim)
            h = np.interp(x_new, x_old, h)
        sum_h = np.sum(h) * dx
        scale = (n_points / 2.0) / sum_h
        h *= scale
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0

    def objective(h):
        cc = correlate(h, 1.0 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 500,
        'ftol': 1e-10,
        'disp': False,
        'iprint': -1,
        'maxfun': 10000
    }
    result = minimize(objective, x0, method='COBYLA', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h) * dx
    scale = (n_points / 2.0) / sum_h
    best_h *= scale
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = (best_h * (1.0 / final_sum)) * (n_points / 2.0)

    cc = correlate(best_h, 1.0 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    return (best_h, c5_bound, n_points)
