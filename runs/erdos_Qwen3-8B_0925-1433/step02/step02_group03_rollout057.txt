import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial guess
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h else 1.0
    h = h * scale
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    max_iter = 100
    for _ in range(max_iter):
        # Perturb current best solution
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        scale_perturb = (n_points / 2) / sum_perturb if sum_perturb else 1.0
        h_perturb = h_perturb * scale_perturb

        # Run optimizer
        res = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'eps': 1e-6})
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    scale_best = (n_points / 2) / sum_best if sum_best else 1.0
    best_h = best_h * scale_best
    return (best_h, compute_c5(best_h), n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
