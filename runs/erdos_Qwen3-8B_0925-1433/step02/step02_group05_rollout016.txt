import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    def generate_initial_patterns():
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        yield h

        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        yield h

        h = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        yield h

        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        h = np.clip(h, 0, 1)
        yield h

        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        h = np.clip(h, 0, 1)
        yield h

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in generate_initial_patterns():
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - target_sum) > 1e-6:
