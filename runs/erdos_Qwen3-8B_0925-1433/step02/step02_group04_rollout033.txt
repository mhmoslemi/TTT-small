from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    else:
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        h = h * (n_points / 2.0) / sum_h

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 500,

        'eps': 1e-8,
        'disp': False,
        'iprint': -1
    }
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
