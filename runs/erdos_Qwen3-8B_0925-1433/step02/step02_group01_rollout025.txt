import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iter_count = 0
    pop_size = 20
    population = [np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1) for _ in range(pop_size)]
    population = [np.clip(p * (n_points / 2) / np.sum(p), 0, 1) for p in population]
    
    while time.time() - start_time < budget_s:
        iter_count += 1

        best_h = population[best_idx].copy()

        offspring = []
        for _ in range(pop_size - len(elite)):
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = np.clip(child * (n_points / 2) / np.sum(child), 0, 1)
            offspring.append(child)
        population = elite + offspring

            population = [np.clip(p + perturb, 0, 1) for p in population]

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
