from scipy.optimize import minimize, differential_evolution

    def obj_func(x):

        sum_clipped = np.sum(h_clipped)

        scale = (n_points / 2) / sum_clipped
        scaled_h = np.clip(h_clipped * scale, 0, 1)

        return np.max(corr) * dx
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
