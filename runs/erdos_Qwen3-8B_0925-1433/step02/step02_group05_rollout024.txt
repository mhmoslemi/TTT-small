import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_guesses():
        guesses = []
        # Random guess
        r = np.random.rand(n_points)
        r = r * (target_sum / np.sum(r))
        guesses.append(r)
        # Sine wave
        x = np.linspace(0, 2, n_points)
        s = 0.5 + 0.5 * np.sin(np.pi * x)
        s = np.clip(s, 0, 1)
        s = s * (target_sum / np.sum(s))
        guesses.append(s)
        # Cosine wave
        c = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        c = np.clip(c, 0, 1)
        c = c * (target_sum / np.sum(c))
        guesses.append(c)
        # Square wave
        sq = np.random.rand(n_points)
        sq = np.where(sq < 0.5, 0, 1)
        sq = sq * (target_sum / np.sum(sq))
        sq = np.clip(sq, 0, 1)
        guesses.append(sq)
        # Initial guess
        guesses.append(initial_h_values.copy())
        return guesses

    guesses = generate_guesses()

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-8, 'maxiter': 200})
        h_best = res.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - target_sum) > 1e-6:
            h_best = h_best * (target_sum / sum_h)
            h_best = np.clip(h_best, 0, 1)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5

    sum_h = np.sum(best_h)
    if abs(sum_h - target_sum) > 1e-6:
        best_h = initial_h_values.copy()
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
