import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = np.interp(np.linspace(0, 2, 200), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = np.zeros(200)
        for i in range(200):
            x = i * (2.0 / 200)
            dist = abs(x - 1.0)
            h[i] = 0.5 * (1 - dist)
    
    def obj(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if np.isclose(sum_x, 0):
            return 1e6
        scale = (200 / 2) / sum_x
        x = np.clip(x * scale, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / 200)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 100}]
    bounds = [(0, 1)] * 200
    start_time = time.time()
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
    elapsed = time.time() - start_time
    while elapsed < budget_s and res.success:
        res = minimize(obj, res.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        elapsed = time.time() - start_time
