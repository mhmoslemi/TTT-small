import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h_dim = len(h)
        if h_dim != n_points:
            x_new = np.linspace(0, 2, n_points)
            x_old = np.linspace(0, 2, h_dim)
            h = np.interp(x_new, x_old, h)
        sum_h = np.sum(h) * dx
        h *= (n_points / 2.0) / sum_h
    else:
        indices = np.random.choice(n_points, n_points // 2, replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0

        cc = correlate(h, 1 - h, mode='full')

        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = result.x

    sum_h = np.sum(best_h) * dx
    scale = (n_points / 2.0) / sum_h
    best_h *= scale
