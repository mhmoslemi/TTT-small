```python
import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Prepare bounds and initial population
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            scale = 0.5
        else:
            scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initial population
    pop_size = 20
    initial_pop = [initial_h_values.copy() for _ in range(pop_size)]
    for i in range(pop_size - 1):
        perturb = np.random.uniform(-0.2, 0.2, n_points)
        initial_pop[i+1] = initial_h_values + perturb
        initial_pop[i+1] = np.clip(initial_pop[i+1], 0, 1)
        sum_p = np.sum(initial_pop[i+1])
        if sum_p != 0:
            scale = (n_points / 2) / sum_p
            initial_pop[i+1] = np.clip(initial_pop[i+1] * scale, 0, 1)
    
    best_vals = [objective(p) for p in initial_pop]
    best_idx = np.argmin(best_vals)
    best_h = np.array(initial_pop[best_idx])
    best_c5 = best_vals[best_idx]
    
    start_time = time.time()
    
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   population_size=pop_size, tol=1e-6, 
                                   maxiter=100, recombination=0.9, mutation=0.2, 
                                   polish=True, workers=-1)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        final_h = result.x
        final_c5 = objective(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Ensure constraints are met
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h_clipped)
    if sum_best == 0:
        scale = 0.5
    else:
        scale = (n_points / 2) / sum_best
    best_h_scaled = np.clip(best_h_clipped * scale, 0, 1)
    best_c5 = objective(best_h_scaled)
    
    return (best_h_scaled, best_c5, n_points)
```
