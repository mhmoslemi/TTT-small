            if abs(sum_h_best - sum_required) > 1e-6:
                h_best = h_best * (sum_required / sum_h_best)
                h_best = np.clip(h_best, 0, 1)
            c5 = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx
            if c5 < best_c5:
                best_h = h_best.copy()
                best_c5 = c5
    return (best_h, best_c5, n_points)
