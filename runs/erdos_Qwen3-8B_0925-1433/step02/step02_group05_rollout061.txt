    sum_best = np.sum(h_best) * dx
    if abs(sum_best - 1.0) > 1e-6:
        h_best = h_best * (1.0 / sum_best)
    c5 = obj(h_best)
    if c5 < best_c5:
        best_h = h_best
        best_c5 = c5

    return (best_h, best_c5, n_points)
