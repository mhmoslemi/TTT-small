import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Initialize with provided h_values, clip and scale to meet sum constraint
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    
    # Define bounds and objective with penalty for sum constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        h_candidate = x.copy()
        sum_candidate = np.sum(h_candidate)
        penalty = 1e6 * np.abs(sum_candidate - n_points / 2)
        shifted = 1 - h_candidate
        corr = np.correlate(h_candidate, shifted, mode='full')
        max_corr_val = np.max(corr)
        return max_corr_val * dx + penalty
    
    # Run differential evolution for max iterations within budget
    try:
        result = differential_evolution(
            objective, bounds, 
            strategy='best1bin', 
            maxiter=int(budget_s / 4), 
            popsize=15, 
            tol=1e-6,
            mutation=(0.5, 1), 
            recombination=0.8
        )
        best_h = result.x
        best_c5 = compute_c5(best_h, dx, n_points)
    except:
        pass
    
    # Final check against initial solution
    if compute_c5(best_h, dx, n_points) > compute_c5(initial_h_values, dx, n_points):
        best_h = initial_h_values
        best_c5 = compute_c5(best_h, dx, n_points)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
