        particles.append(h)
        velocities.append(np.zeros_like(h))
    
    pbest = [h.copy() for h in particles]
    pbest_scores = np.full(n_particles, np.inf)

    # Evaluate initial pbest_scores
    for i in range(n_particles):
        h = particles[i]
        cc = correlate(h, 1 - h, mode='full')
        c_values = cc * dx
        score = np.max(c_values)
        pbest_scores[i] = score
        if score < pbest_scores[i]:
            pbest[i] = h.copy()
    
    # Initialize gbest
    best_index = np.argmin(pbest_scores)
    gbest = pbest[best_index].copy()
    gbest_score = pbest_scores[best_index]

    # Handle initial_h_values if available
    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h_values)
        h_initial = h_initial * required_sum / np.sum(h_initial)
        particles[best_index] = h_initial.copy()
        cc = correlate(h_initial, 1 - h_initial, mode='full')
        c_values = cc * dx
        score = np.max(c_values)
        pbest_scores[best_index] = score
        pbest[best_index] = h_initial.copy()
        if score < gbest_score:
            gbest_score = score
            gbest = h_initial.copy()

    start_time = time.time()
    iterations = 0
    max_iterations = 500
    while iterations < max_iterations and time.time() - start_time < budget_s - 10:
        # Update velocities and positions
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            v = velocities[i] * 0.729 + 1.494 * r1 * (pbest[i] - particles[i]) + 1.494 * r2 * (gbest - particles[i])
            velocities[i] = v
            particles[i] += v
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * required_sum / np.sum(particles[i])
        
        # Evaluate scores and update pbest
        for i in range(n_particles):
            h = particles[i]
            cc = correlate(h, 1 - h, mode='full')
            c_values = cc * dx
            score = np.max(c_values)
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = h.copy()
        
        # Update gbest
        best_index = np.argmin(pbest_scores)
        if pbest_scores[best_index] < gbest_score:
            gbest_score = pbest_scores[best_index]
            gbest = pbest[best_index].copy()
        iterations += 1

    # Recompute final C5
    cc = correlate(gbest, 1 - gbest, mode='full')
    final_c5 = np.max(cc * dx)
    return (gbest, final_c5, n_points)
