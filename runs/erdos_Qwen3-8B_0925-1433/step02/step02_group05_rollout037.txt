import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def scale_and_clip(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-6:
            scale = 1 / sum_h
            h = h * scale
        return h

    # Initial h
    h_initial = initial_h_values.copy()
    h_initial = scale_and_clip(h_initial)
    c5_initial = obj(h_initial)
    best_h = h_initial
    best_c5 = c5_initial
    start_time = time.time()

    # Frequency-domain guess
    frequencies = np.fft.rfftfreq(n_points, dx)
    H = np.ones_like(frequencies)
    mask = (frequencies >= 0.1) & (frequencies <= 0.2)
    H[mask] = 0
    h_fft = np.fft.irfft(H)
    h_fft = scale_and_clip(h_fft)
    c5_fft = obj(h_fft)
    if c5_fft < best_c5:
        best_h = h_fft
        best_c5 = c5_fft

    # Optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
    if time.time() - start_time < budget_s:
        h_opt = scale_and_clip(result.x)
        c5_opt = obj(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    return (best_h, best_c5, n_points)
