        # Evaluate population

def normalize(h, n_points):
    sum_h = np.sum(h)
    if sum_h == 0:
        return np.full(n_points, 0.5)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    return h

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
