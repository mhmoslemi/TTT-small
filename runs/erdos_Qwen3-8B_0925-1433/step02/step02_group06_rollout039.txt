        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def adjust_sum(h, target_sum):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-6:
            return h
        delta = target_sum - s
        if delta > 0:
            mask = h < 1.0
            if np.any(mask):
                count = np.sum(mask)
                delta_per = delta / count
                h[mask] += delta_per
                h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if np.any(mask):
                count = np.sum(mask)
                delta_per = abs(delta) / count
                h[mask] -= delta_per
                h = np.clip(h, 0.0, 1.0)
        return h

    np.random.seed(seed)
    n_particles = 30
    w, c1, c2 = 0.8, 1.5, 1.5
    vel_clip = 0.1
    max_iter = 500
    best_h = initial_h
    best_c5 = obj(initial_h)
    start_time = np.random.rand()

    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    for _ in range(n_particles):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum(h, n_points / 2)
        particles.append(h)
        velocities.append(np.random.uniform(-vel_clip, vel_clip, size=n_points))
        pbest.append(h)
        pbest_vals.append(obj(h))

    gbest = particles[0]
    gbest_val = pbest_vals[0]
    for i in range(1, n_particles):
        if pbest_vals[i] < gbest_val:
            gbest = particles[i]
            gbest_val = pbest_vals[i]

    time_used = 0
    for _ in range(max_iter):
        if time_used >= budget_s:
            break
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -vel_clip, vel_clip)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = adjust_sum(particles[i], n_points / 2)
            current_val = obj(particles[i])
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
        for i in range(n_particles):
            if pbest_vals[i] < gbest_val:
                gbest = pbest[i].copy()
                gbest_val = pbest_vals[i]
        time_used += 1
        if time_used >= budget_s:
            break

    if gbest_val < best_c5:
        best_h = gbest
        best_c5 = gbest_val

    return (best_h, best_c5, n_points)
