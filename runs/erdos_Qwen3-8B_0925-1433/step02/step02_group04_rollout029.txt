from scipy.optimize import minimize
from scipy.signal import correlate

    dx = 2.0 / n_points

        h /= (np.sum(h) * dx) * (n_points / 2.0)
        h *= (n_points / 2.0)
    else:
        indices = np.random.permutation(n_points)
        h = np.zeros(n_points)
        h[indices[:n_points // 2]] = 1.0
        h = np.clip(h, 0, 1)

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 1000,
        'ftol': 1e-10,
        'eps': 1e-10,
        'disp': False,
        'iprint': -1

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
