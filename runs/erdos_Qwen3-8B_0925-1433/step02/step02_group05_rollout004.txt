        if abs(f) < low_freq:
            H[i] = 0.0
        elif abs(f) > high_freq:
            H[i] = 1.0
        else:
            x = (abs(f) - low_freq) / (high_freq - low_freq)
            H[i] = 1.0 / (1.0 + np.exp(-10.0 * x))
    h_fft = np.fft.irfft(H, n_points)
    h_fft = np.clip(h_fft, 0.0, 1.0)
    sum_h_fft = np.sum(h_fft) * dx
    if abs(sum_h_fft - 1.0) > 1e-6:
        h_fft *= 1.0 / sum_h_fft
    h_fft = np.clip(h_fft, 0.0, 1.0)
    
    # Optimization
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}]
    result = minimize(obj, h_fft, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
    h_best = result.x
    h_best = np.clip(h_best, 0.0, 1.0)
    sum_h_best = np.sum(h_best) * dx
    if abs(sum_h_best - 1.0) > 1e-6:
        h_best *= 1.0 / sum_h_best
    h_best = np.clip(h_best, 0.0, 1.0)
    c5 = np.max(np.correlate(h_best, 1.0 - h_best, mode='full')) * dx
    if c5 < best_c5:
        best_h = h_best
        best_c5 = c5

    return (best_h, best_c5, n_points)
