            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1)] * n_points

        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints)
        if res.success:
            best_h = res.x.copy()

        if iteration % 50 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h = np.clip(best_h + perturb, 0, 1)
            sum_h = np.sum(best_h)
            if sum_h == 0:
                best_h = np.full_like(best_h, 0.5)
            else:
                scale = (n_points / 2) / sum_h
                best_h = np.clip(best_h * scale, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
