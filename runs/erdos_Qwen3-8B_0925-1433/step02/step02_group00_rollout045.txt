import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 150
    if initial_h_values is not None:
        h_initial = initial_h_values.copy()
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if sum_h == 0:
            h_initial = np.full_like(h_initial, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h_initial = np.clip(h_initial * scale, 0, 1)
    else:
        h_initial = np.random.rand(n_points)
        sum_h = np.sum(h_initial)
        scale = (n_points / 2) / sum_h
        h_initial = np.clip(h_initial * scale, 0, 1)
    
    dx = 2.0 / n_points

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return 1e6
        scale = (n_points / 2) / sum_h
        h_scaled = h_clipped * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}

    start_time = time.time()
    result = basinhopping(

        minimizer_kwargs={

            'bounds': bounds,
            'constraints': [constraint]

        disp=False
    )

        scale = (n_points / 2) / sum_best
