import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Process initial_h_values
    h = initial_h_values.copy()

    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.full(n_points, 0.5)

        scale = (n_points / 2) / sum_h

    # Genetic algorithm with differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    def objective(x):
        h = x.reshape(n_points)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full(n_points, 0.5)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def evaluator(x):
        return objective(x)

    # Run differential evolution
    result = differential_evolution(evaluator, bounds, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, callback=lambda x: None, disp=False, polish=True)
    best_h = result.x.reshape(n_points)
    best_c5 = result.fun

    # Final check of constraint and scaling
    sum_h = np.sum(best_h)
    if sum_h == 0:
        best_h = np.full(n_points, 0.5)
    else:
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
    best_c5 = compute_c5(best_h, dx, n_points)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):

    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
