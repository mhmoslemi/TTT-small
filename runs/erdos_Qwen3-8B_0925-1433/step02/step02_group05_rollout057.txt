import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    guesses = [initial_h_values.copy()]

    # Random guess
    rand = np.random.rand(n_points)
    rand *= (n_points / 2) / np.sum(rand)
    rand = np.clip(rand, 0, 1)
    guesses.append(rand)

    # Sine wave
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = np.clip(sine, 0, 1)

    guesses.append(cosine)

    block = np.zeros(n_points)

    block *= (n_points / 2) / np.sum(block)
    block = np.clip(block, 0, 1)
    guesses.append(block)

    guesses.append(square)
