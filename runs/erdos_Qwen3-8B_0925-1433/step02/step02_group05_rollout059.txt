import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]

    def generate_patterns():
        h_parent = np.clip(initial_h_values, 0.0, 1.0)
        h_perturbed = h_parent + np.random.normal(0.0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0) * (sum_required / np.sum(h_perturbed))
        yield h_perturbed

        t = np.linspace(0, 2 * np.pi, n_points)
        h_cos = 0.5 + 0.5 * np.cos(t)
        h_cos = np.clip(h_cos, 0.0, 1.0) * (sum_required / np.sum(h_cos))
        yield h_cos

        t = np.linspace(0, 1.0, n_points)
        h_sym = 0.5 + 0.5 * (1.0 - np.abs(t - 0.5))
        h_sym = np.clip(h_sym, 0.0, 1.0) * (sum_required / np.sum(h_sym))
        yield h_sym

        h_alt = np.zeros(n_points)
        h_alt[::2] = 1.0
        h_alt = np.clip(h_alt, 0.0, 1.0) * (sum_required / np.sum(h_alt))
        yield h_alt

        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0.0, 1.0) * (sum_required / np.sum(h_rand))
        yield h_rand

        h_fft = np.random.randn(n_points)
        h_fft = np.fft.irfft(np.fft.rfft(h_fft, n=n_points) * np.exp(-0.5 * (np.arange(n_points)/n_points)**2))
        h_fft = np.clip(h_fft, 0.0, 1.0) * (sum_required / np.sum(h_fft))
        yield h_fft

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in generate_patterns():
        if time.time() - start_time > budget_s:
            break
        h_guess = guess.copy()
        result = minimize(obj, h_guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0.0, 1.0)
        if np.abs(np.sum(h_best) - sum_required) > 1e-6:
            h_best *= sum_required / np.sum(h_best)

        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5

    return (best_h, best_c5, n_points)
