import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values)
    if abs(sum_initial) < 1e-6:
        h = np.full(n_points, 0.5)
    else:
        scale = (n_points / 2) / sum_initial
        h = np.clip(initial_h_values * scale, 0, 1)
    population = [h.copy() for _ in range(20)]
    for i in range(20):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        p = np.clip(h + perturb, 0, 1)
        sum_p = np.sum(p)
        if abs(sum_p) < 1e-6:
            p = np.full(n_points, 0.5)
        else:
            p = np.clip(p * (n_points / 2) / sum_p, 0, 1)
        population[i] = p
    best_h = population[0].copy()
    best_c5 = compute_c5(population[0], dx, n_points)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        c5_values = []
        for h in population:
            c5_values.append(compute_c5(h, dx, n_points))
        elite_size = int(len(population) * 0.2)
