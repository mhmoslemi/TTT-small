def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 150
    if initial_h_values is not None:
        h = initial_h_values.copy()

        sum_h = np.sum(h)

        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1e6
        scale = (n_points / 2) / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2

    def local_opt(x0):
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        return res.x if res.success else x0

    def callback(x, accept):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            x = np.full_like(x, 0.5)
        else:
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
        return x

    result = basinhopping(
        objective,
        h,
        T=1.0,
        stepsize=0.1,
        niter=100,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': [(0, 1)] * n_points,

    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
