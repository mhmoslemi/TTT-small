import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        initial_h = np.full(n_points, 0.5)
        initial_h[::3] = 1.0
        initial_h[1::3] = 0.0
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def obj(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h
    best_c5 = obj(initial_h)

    # Generate diverse initial guesses
    def generate_patterns():
        patterns = []
        # Periodic three-bit pattern
        p = np.zeros(n_points)
        period = int(n_points / 3)
        for i in range(n_points):
            pos = dx * i
            part = int(pos / period)
            if part % 2 == 0:
                p[i] = 1.0
            else:
                p[i] = 0.0
        total_p = np.sum(p)
        if total_p != 0:
            p *= (n_points / 2) / total_p
        patterns.append(p)
        # Cosine-shaped
        x = np.linspace(0, 2, n_points)
        c = 0.5 * (1 + np.cos(np.pi * x))
        total_c = np.sum(c)
        c *= (n_points / 2) / total_c
        patterns.append(c)
        # Random binary
        r = np.random.rand(n_points)
        r = (r > 0.5).astype(float)
        total_r = np.sum(r)
        r *= (n_points / 2) / total_r
        patterns.append(r)
        # Alternating trapezoidal
        trap = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0:
                trap[i] = 1.0
            elif i % 4 == 1:
                trap[i] = 1.0
            elif i % 4 == 2:
                trap[i] = 0.5
            else:
                trap[i] = 0.0
        total_trap = np.sum(trap)
        trap = trap * (n_points / 2) / total_trap
        patterns.append(trap)
        return patterns

    initial_guesses = generate_patterns()

    for h_start in initial_guesses:
        h_start = np.clip(h_start, 0.0, 1.0)
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons,
