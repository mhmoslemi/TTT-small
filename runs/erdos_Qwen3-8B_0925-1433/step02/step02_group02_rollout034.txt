import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import Bounds

    global initial_h_values
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Preprocess initial_h_values
    def scale_h(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.full(n_points, 0.5)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)
    h_initial = scale_h(initial_h_values.copy())
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    # Generate initial population
    population = []
    for _ in range(20):
        h = h_initial + np.random.uniform(-0.1, 0.1, n_points)
        h = scale_h(h)
        population.append(h)
    for _ in range(15):
        h = np.random.rand(n_points)
        h = scale_h(h)
        population.append(h)
    population = np.array(population)
    
    start_time = time.time()
    
    # Main optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate C5
        c5_values = np.array([compute_c5(population[i], dx, n_points) for i in range(len(population))])
        
        # Select elites
        elite_indices = np.argsort(c5_values)[:int(len(population)*0.1)]
        elites = population[elite_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (p1 + p2) / 2.0
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
            child = scale_h(child)
            offspring.append(child)
        population = np.vstack((elites, offspring))
        
        # Local refinement on top 5
        top_idx = np.argsort(c5_values)[:5]
        for i in top_idx:
            h = population[i].copy()
            def obj(x):
                x = scale_h(x)
                shifted = 1 - x
                corr = np.correlate(x, shifted, mode='full')
                return np.max(corr) * dx
            res = minimize(obj, h, bounds=[(0,1)]*n_points, method='L-BFGS-B', tol=1e-6)
            if res.success:
                population[i] = res.x
        
        # Update best
        current_idx = np.argmin(c5_values)
        current_c5 = c5_values[current_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[current_idx].copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
