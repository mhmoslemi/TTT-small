    sum_fft = np.sum(h_fft) * dx
    h_fft_scaled = h_fft * (1.0 / sum_fft) if sum_fft != 0 else h_fft
    h_fft_scaled = np.clip(h_fft_scaled, 0, 1)

    # Define objective function
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization
    best_h = initial_h_values_scaled.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    result = minimize(obj, h_fft_scaled, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
    h_best = result.x
    h_best = np.clip(h_best, 0, 1)
    sum_h = np.sum(h_best) * dx
    if abs(sum_h - 1.0) > 1e-6:
        h_best = h_best * (1.0 / sum_h)
        h_best = np.clip(h_best, 0, 1)
    c5 = obj(h_best)
    if c5 < best_c5:
        best_h = h_best
        best_c5 = c5

    # Fallback to initial_scaled if needed
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = initial_h_values_scaled.copy()
        best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
