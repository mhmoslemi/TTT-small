import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        return h * scale

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = clip_and_scale(h)

    def objective(h):
        return evaluate_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h = result.x
    best_c5 = evaluate_c5(best_h)

    num_perturbations = 5
    for _ in range(num_perturbations):
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = clip_and_scale(h_perturbed)
        res = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            current_c5 = evaluate_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    best_h = clip_and_scale(best_h)
    return (best_h, best_c5, n_points)
