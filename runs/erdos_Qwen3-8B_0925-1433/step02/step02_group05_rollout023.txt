import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_freq_guess(low_cut, high_cut):
        freqs = np.fft.rfftfreq(n_points, dx)
        H = np.zeros_like(freqs)
        mask_low = freqs <= low_cut
        mask_high = freqs >= high_cut
        mask_trans = ~mask_low & ~mask_high
        H[mask_high] = 1.0
        H[mask_low] = 0.0
        H[mask_trans] = (freqs[mask_trans] - low_cut) / (high_cut - low_cut)
        return H

    def generate_init_guesses(variations):
        guesses = []
        for (low, high) in variations:
            H = create_freq_guess(low, high)
            h = np.fft.irfft(H)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h) * dx
            if sum_h == 0:
                scale = 1.0
            else:
                scale = target_sum / sum_h
            h = h * scale
            guesses.append(h)
        return guesses

    variations = [
        (0.2, 0.4),
        (0.25, 0.4),
        (0.15, 0.5),
        (0.1, 0.45),
        (0.2, 0.45),
        (0.2, 0.35)
    ]

    initial_guesses = generate_init_guesses(variations)
    initial_guesses.append(initial_h.copy())

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = obj(best_h)
    start = time.time()

    for guess in initial_guesses:
        if time.time() - start > budget_s:
            break
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        h = res.x
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - target_sum) > 1e-6:
            scale = target_sum / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)
        c5 = obj(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    if best_c5 > obj(initial_h):
        best_h = initial_h.copy()
        best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
