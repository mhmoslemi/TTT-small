import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values) * dx
    scale_factor = (n_points / 2.0) / initial_sum if initial_sum != 0 else 1.0
    initial_h = np.clip(initial_h_values * scale_factor, 0, 1)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_freq_guess(low, high, mid, k):
        freqs = np.fft.rfftfreq(n_points, d=dx)
        H = np.zeros_like(freqs)
        H[freqs >= high] = 1.0
        H[freqs <= low] = 0.0
        mask = (freqs > low) & (freqs < high)
        H[mask] = 1 / (1 + np.exp(-k * (freqs[mask] - mid)))
        h = np.fft.irfft(H)
        h = np.clip(h, 0, 1)
        if np.sum(h) * dx != 0:
            h = h * (n_points / 2.0) / (np.sum(h) * dx)
        return h

    variations = [
        (0.1, 0.9, 0.5, 10),
        (0.2, 0.8, 0.5, 15),
        (0.25, 0.75, 0.5, 20),
        (0.3, 0.7, 0.5, 5),
    ]

    start_time = time.time()
    for low, high, mid, k in variations:
        h_guess = generate_freq_guess(low, high, mid, k)
        result = minimize(obj, h_guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        if result.success:
            h_best = np.clip(result.x, 0, 1)
            sum_h = np.sum(h_best) * dx
            if abs(sum_h - 1) > 1e-6:
                h_best = h_best * (1 / sum_h)
                h_best = np.clip(h_best, 0, 1)
            current_c5 = obj(h_best)
            if current_c5 < best_c5:
                best_h = h_best.copy()
                best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
