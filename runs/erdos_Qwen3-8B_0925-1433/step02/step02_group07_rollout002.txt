        if elapsed > budget_s - 10:
            break
        
        if result.success:
            current_h = result.x.reshape(n_points)
            current_h = np.clip(current_h, 0, 1)
            current_sum = np.sum(current_h)
            current_h *= required_sum / current_sum
            cc = correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(cc * dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return (best_h, best_c5, n_points)
