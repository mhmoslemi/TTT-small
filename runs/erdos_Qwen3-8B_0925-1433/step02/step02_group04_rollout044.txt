import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 320  # Sufficient resolution for better bounds
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h_dim = len(h)
        if h_dim != n_points:

            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x, h)
        h = h.astype(np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0  # Structured binary pattern for even distribution

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 750,  # More iterations for better convergence
        'ftol': 1e-9,    # Tighter tolerance
        'eps': 1e-8,
        'disp': False,
        'iprint': -1

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1.0) > 1e-6:
