import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, initial_h_values=None):
    if initial_h_values is None:
        n_points = 45
        initial_h_values = np.random.rand(n_points)
    else:
        n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute current correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        current_c5 = corr[max_idx] * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Define objective function for optimization
        def objective(x):
            h_opt = x.copy()
            sum_h = np.sum(h_opt)
            if sum_h == 0:
                return 1e6
            scale = (n_points / 2) / sum_h
            h_opt = np.clip(h_opt * scale, 0, 1)
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            max_idx = np.argmax(corr)
            return corr[max_idx] * dx
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints)
        if res.success:
            h = res.x
        
        # Perturb every 100 iterations
        if iter_count % 100 == 0:

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
