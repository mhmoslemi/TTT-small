        scale_factor = required_sum / current_sum
        h_flat *= scale_factor
        
        def objective(x):
            h = x.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            return np.max(cc * dx)
        
        def constraint(x):
            return np.sum(x) - required_sum
        
        bounds = [(0, 1)] * n_points
        result = minimize(
            fun=objective,
            x0=h_flat,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False}
        )
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        
        h = result.x
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h) * dx
        h *= (required_sum / current_sum)
        cc = correlate(h, 1 - h, mode='full')
        c5 = np.max(cc * dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    if best_h is None:
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            x_interpolate = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_interpolate, h)
            h = np.clip(h, 0.0, 1.0)
            current_sum = np.sum(h) * dx
            h *= (required_sum / current_sum)
        else:
            h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)])
            h = np.clip(h, 0.0, 1.0)
            current_sum = np.sum(h) * dx
            h *= (required_sum / current_sum)
        cc = correlate(h, 1 - h, mode='full')
        best_c5 = np.max(cc * dx)
        best_h = h
    
    return (best_h, best_c5, n_points)
