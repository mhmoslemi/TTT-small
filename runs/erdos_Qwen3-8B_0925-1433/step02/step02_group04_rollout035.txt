            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, interp_x, interp_y)
        else:

        sum_h = np.sum(h) * dx
        h = h * (n_points / 2.0) / sum_h
        initial_h = h

        h[::4] = 1.0
        h[1::4] = 0.5

        h = h * (n_points / 2.0) / sum_h
        initial_h = h

    h = cp.Variable(n_points)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2.0,
        h >= 0.0,
        h <= 1.0
    ]

    for k in range(1, n_points):
        # Compute overlap integral for lag k
        # Value at position i: h[i] * (1 - h[(i + k) % n_points])
        # Use linear expression for each lag
        overlap = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += h[i] * (1.0 - h[j])
        constraints.append(overlap <= t)

    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        prob.solve(solver=cp.SCS, verbose=False)

    best_h = np.clip(h.value, 0.0, 1.0)

    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = (best_h * (1.0 / final_sum) * 1.0) * (n_points / 2.0)

    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    return (best_h, c5_bound, n_points)
