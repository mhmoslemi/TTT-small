    h_bin = h_bin * required_sum / np.sum(h_bin)
    structured_guesses.append(h_bin)

    best_h = h.copy()
    best_c5 = float('inf')
    elapsed = 0

    for guess in structured_guesses:
        guess_clipped = np.clip(guess, 0, 1)
        guess_rescaled = guess_clipped * required_sum / np.sum(guess_clipped)
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            c = correlate(h, 1 - h, mode='full') * dx
            return np.max(c)
        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum
        bounds = [(0, 1)] * n_points
        initial_guess = guess_rescaled.flatten()
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        elapsed = time.time() - time.time()
        if not result.success:
            continue
        curr_h = result.x
        curr_h = np.clip(curr_h.reshape(n_points), 0, 1)
        curr_h = curr_h * required_sum / np.sum(curr_h)
        c_vals = correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(c_vals)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h
        if elapsed > budget_s - 10:
            break

    return (best_h, best_c5, n_points)
