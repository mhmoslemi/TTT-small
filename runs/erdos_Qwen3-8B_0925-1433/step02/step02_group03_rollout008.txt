import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    # Normalize initial guess
    initial_h = initial_h_values.copy()
    total = np.sum(initial_h)
    scale = (n_points / 2) / total if total > 0 else 1.0
    initial_h = np.clip(initial_h * scale, 0, 1)

    def perturb(h):
        noise = np.random.normal(0, 0.05, len(h))
        h_perturb = h + noise
        h_perturb = np.clip(h_perturb, 0, 1)
        total = np.sum(h_perturb)
        scale = (n_points / 2) / total if total > 0 else 1.0
        return h_perturb * scale

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Run multiple local optimizations
    for _ in range(10):
        current_h = perturb(best_h)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

            result = minimize(

                current_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}

            if result.success:
                current_c5 = result.fun
            else:
                current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x if result.success else current_h
        except:
            pass
        if time.time() - start_time >= budget_s:

    # Final normalization
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
