from scipy.optimize import minimize
from scipy.signal import correlate

        sum_h = np.sum(h) * dx
        h *= (n_points / 2.0) / sum_h

        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        'eps': 1e-9,
        'disp': False,

    }
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)
