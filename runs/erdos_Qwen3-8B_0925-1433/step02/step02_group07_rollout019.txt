        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_initial, initial_h)
        h = h * required_sum / np.sum(h)
        initial_guesses.append(h)

    binary_h = np.zeros(n_points)
    binary_h[::2] = 1
    initial_guesses.append(binary_h)

    x = np.linspace(0, 2, n_points)
    smooth_h = 0.5 * (1 + np.sin(np.pi * x))
    initial_guesses.append(smooth_h)

    best_c5 = np.inf
    best_h = None
    start_time = time.time()

    for guess in initial_guesses:
        guess = guess.reshape(n_points)
        guess = np.clip(guess, 0, 1)
        current_sum = np.sum(guess)
        if abs(current_sum - required_sum) > 1e-6:
            guess = guess * required_sum / current_sum

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            return np.max(cc * dx)

        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum

        bounds = [(0, 1)] * n_points
        result = minimize(
            fun=objective,
            x0=guess.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False}
        )

        if result.success:
            current_h = np.clip(result.x.reshape(n_points), 0, 1)
            current_h = current_h * required_sum / np.sum(current_h)
            cc = correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(cc * dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

    if best_h is None:
        best_h = np.zeros(n_points)
        best_h[::2] = 1
        cc = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(cc * dx)

    return (best_h, best_c5, n_points)
