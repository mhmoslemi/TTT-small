import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    target_sum = n_points / 2.0
    initial_h = initial_h_values.copy()
    if abs(np.sum(initial_h) - target_sum) > 1e-6:
        initial_h *= target_sum / np.sum(initial_h)
    best_h = initial_h
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode="full")) * dx

    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    guesses = []
    np.random.seed(seed)
    # Perturbed parent
    guesses.append(initial_h + np.random.normal(0, 0.02, n_points))
    # Periodic pattern
    t = np.linspace(0, 2*np.pi, n_points)
    guesses.append(0.5 + 0.5 * np.cos(t))
    # Symmetric pattern
    guesses.append(0.5 + 0.5 * (1 - np.abs(np.arange(n_points) - n_points/2) / (n_points/2)))
    # Alternating pattern
    guesses.append(0.5 + 0.5 * (1 - (np.arange(n_points) % 2)))
    # Random oscillation
    guesses.append(np.random.rand(n_points))
    # Filtered oscillation
    rfft = np.fft.rfft(np.random.rand(n_points))
    rfft *= np.exp(-0.5 * (np.arange(n_points) / n_points)**2)
    guesses.append(np.fft.irfft(rfft))

    start_time = time.time()

        if time.time() - start_time > budget_s:
            break
        guess = np.clip(guess, 0, 1)
        if abs(np.sum(guess) - target_sum) > 1e-6:
            guess *= target_sum / np.sum(guess)
        guess = np.clip(guess, 0, 1)
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 150, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        if abs(np.sum(h_best) - target_sum) > 1e-6:
