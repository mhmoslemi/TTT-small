    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_initial, initial_h)
        h = h * required_sum / np.sum(h)
        initial_guesses.append(h)
    x = np.linspace(0, 2, n_points)
    h_binary = np.zeros(n_points)
    h_binary[::2] = 1
    h_binary = h_binary * required_sum / np.sum(h_binary)
    initial_guesses.append(h_binary)
    h_smooth = 0.5 * (1 - np.cos(2 * np.pi * x))
    h_smooth = h_smooth * required_sum / np.sum(h_smooth)
    initial_guesses.append(h_smooth)
    def obj(h_flat):
        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc * dx)
    def cons(h_flat):
        return np.sum(h_flat.reshape(n_points)) - required_sum
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    for h_initial in initial_guesses:
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * required_sum / np.sum(h_initial)
        bounds = [(0, 1)] * n_points
        result = minimize(obj, h_initial.flatten(), method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': cons}], options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False})
        if time.time() - start_time > budget_s - 10:
            break
        if result.success:
            h_opt = np.clip(result.x.reshape(n_points), 0, 1)
            h_opt = h_opt * required_sum / np.sum(h_opt)
            cc = correlate(h_opt, 1 - h_opt, mode='full')
            current_c5 = np.max(cc * dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt
    if best_h is None:
        best_h = initial_guesses[0]
        cc = correlate(best_h, 1 - best_h, mode='full') * dx
        best_c5 = np.max(cc)
    return (best_h, best_c5, n_points)
