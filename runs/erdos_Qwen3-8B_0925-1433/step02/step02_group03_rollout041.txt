```python
import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / total
        return h * scale

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()

    for _ in range(100):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = h + noise
        perturbed = clip_and_normalize(perturbed)
        
        def obj(x):
            return evaluate_c5(x)
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
        bounds = [(0, 1)] * n_points
        
        try:
            res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
            if res.success:
                new_h = res.x
                new_c5 = evaluate_c5(new_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        except:
            pass
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (clip_and_normalize(best_h), best_c5, n_points)
```
