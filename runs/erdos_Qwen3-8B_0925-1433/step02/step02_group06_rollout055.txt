            h = h * T / s if s > 0 else np.zeros_like(h)
            pop.append(h)
        vel = [np.zeros(n_points) for _ in range(pop_size)]
        for i in range(pop_size):
            fit = compute_c5(pop[i])
            pfit.append(fit)
            pbest.append(pop[i].copy())
            if fit < gfit:
                gfit = fit
                gbest = pop[i].copy()

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel[i] = w * vel[i] + c1 * r1 * (pbest[i] - pop[i]) + c2 * r2 * (gbest - pop[i])
            vel[i] = np.clip(vel[i], -vel_limit, vel_limit)
            pop[i] += vel[i]
            pop[i] = np.clip(pop[i], 0.0, 1.0)
            s = np.sum(pop[i])
            pop[i] = pop[i] * T / s if s > 0 else np.zeros_like(pop[i])
            fit = compute_c5(pop[i])
            if fit < pfit[i]:
                pfit[i] = fit
                pbest[i] = pop[i].copy()
            if fit < gfit:
                gfit = fit
                gbest = pop[i].copy()
    best_h = gbest
    return (best_h, gfit, n_points)
