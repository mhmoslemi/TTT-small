import numpy as np
import time
from scipy.optimize import basinhopping

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    if initial_h_values is not None:
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    start_time = time.time()
    result = basinhopping(objective, h, niter=50, T=1.0, stepsize=0.1,
                          minimizer_kwargs={"method": "SLSQP", "bounds": bounds, "constraints": constraints})
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2):
        scale = (n_points / 2) / sum_h
        best_h = np.clip(best_h * scale, 0, 1)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
