import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale

    h = clip_scale(initial_h_values.copy())
    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()

    for _ in range(20):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = clip_scale(best_h + perturb)
        def obj(x):
            return evaluate_erdos_solution(x)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            current_h = res.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = clip_scale(best_h)
    return (best_h, evaluate_erdos_solution(best_h), n_points)
