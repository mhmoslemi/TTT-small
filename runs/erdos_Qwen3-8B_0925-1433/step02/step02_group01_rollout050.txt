def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    population_size = 40
    pop = []
    for _ in range(population_size):

        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale = (n_points / 2) / sum_perturbed
        h = np.clip(perturbed * scale, 0, 1)
        pop.append(h)
    best_h = pop[0].copy()
    best_c5 = compute_c5(pop[0], dx, n_points)
    
    start_time = time.time()
    generations = 0
    while time.time() - start_time < budget_s and generations < 500:
        generations += 1
        new_pop = []
        for i in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)
            p1, p2, p3 = pop[indices[0]], pop[indices[1]], pop[indices[2]]

            mask = np.random.rand(n_points) < 0.9
            trial[mask] = mutant[mask]

            scale = (n_points / 2) / sum_trial

            c5_trial = compute_c5(trial, dx, n_points)

            if c5_trial < c5_original:

                perturb = np.random.normal(0, 0.05, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
