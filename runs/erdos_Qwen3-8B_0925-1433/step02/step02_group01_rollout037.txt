import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    pop_size = 20
    population = [np.clip(h * np.random.rand() + h * (1 - np.random.rand()), 0, 1) for _ in range(pop_size)]
    for p in population:
        s = np.sum(p)
        if s != 0:
            p = np.clip(p * (n_points / 2) / s, 0, 1)
        else:
            p = np.full(n_points, 0.5)
    best_pop = best_h.copy()
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = np.array([compute_c5(p, dx, n_points) for p in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Selection
        top_idx = np.argsort(fitness)[:int(pop_size/2)]
        top_pop = [population[i] for i in top_idx]
        
        # Crossover and mutation
        new_pop = []
        for _ in range(pop_size - len(top_pop)):
            a, b = np.random.choice(top_idx, 2, replace=False)
            alpha = np.random.uniform(0.5, 1.5)
            child = top_pop[a] * alpha + top_pop[b] * (1 - alpha)
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            s = np.sum(child)
            if s != 0:
                child = np.clip(child * (n_points / 2) / s, 0, 1)
            else:
                child = np.full(n_points, 0.5)
            new_pop.append(child)
        
        population = top_pop + new_pop
        population = np.array(population)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
