import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total > 0 else 1.0
        return h * scale

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_initial = initial_h_values.copy()
    h_initial = clip_and_normalize(h_initial)
    best_h = h_initial
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    def objective(h):
        return evaluate_c5(h)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1)] * n_points

    num_iterations = int(budget_s / 10)
    for _ in range(num_iterations):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = clip_and_normalize(h_initial + perturbation)
        res = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = clip_and_normalize(best_h)
    return (best_h, best_c5, n_points)
