        symmetric_h = h.copy()
        half_m = m // 2
        if m % 2 == 0:
            symmetric_h[half_m:] = h[:half_m][::-1]
        else:
            symmetric_h[half_m + 1:] = h[:half_m][::-1]
        variables = symmetric_h[:half_m] if m % 2 == 0 else np.concatenate([symmetric_h[:half_m], [symmetric_h[half_m]]])
        dx = 2.0 / m
    
    best_c5 = float('inf')
    best_h = symmetric_h
    
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        def objective(x):
            if m % 2 == 0:
                full_h = np.zeros(m)
                full_h[:half_m] = x
                full_h[half_m:] = x[::-1]
            else:
                full_h = np.zeros(m)
                full_h[:half_m] = x[:-1]
                full_h[half_m + 1:] = x[:-1][::-1]
                full_h[half_m] = x[-1]
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0, 1) for _ in range(len(variables))]
        if m % 2 == 0:
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - m / 4}]
        else:
            constraints = [{'type': 'eq', 'fun': lambda x: 2 * np.sum(x[:-1]) + x[-1] - m / 2}]
        
        res = minimize(objective, variables, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            if m % 2 == 0:
                new_h = np.zeros(m)
                new_h[:half_m] = res.x
                new_h[half_m:] = res.x[::-1]
            else:
                new_h = np.zeros(m)
                new_h[:half_m] = res.x[:-1]
                new_h[half_m + 1:] = res.x[:-1][::-1]
                new_h[half_m] = res.x[-1]
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
        
        if iteration % 50 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=len(variables))
            variables = np.clip(variables + perturb, 0, 1)
            if m % 2 == 0:
                sum_var = np.sum(variables)
                scale = (m / 4) / sum_var if sum_var != 0 else 1
                variables = np.clip(variables * scale, 0, 1)
            else:
                sum_var = 2 * np.sum(variables[:-1]) + variables[-1]
                scale = (m / 2) / sum_var if sum_var != 0 else 1
                variables = np.clip(variables * scale, 0, 1)
    
    return (best_h, best_c5, m)
