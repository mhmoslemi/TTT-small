import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial guess
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = h * scale

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(100):
        # Perturb and scale
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_p = np.sum(h_perturbed)
        scale_p = (n_points / 2) / sum_p if sum_p != 0 else 1.0
        h_perturbed = h_perturbed * scale_p

        # Run optimizer
        res = minimize(obj, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            curr_h = res.x
            curr_c5 = obj(curr_h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
