import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 480
    required_sum = n_points / 2
    dx = 2.0 / n_points

    # Initialize particles
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h)
        h_initial = h_initial * required_sum / np.sum(h_initial)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * required_sum / np.sum(h_initial)

    num_particles = 40
    particles = np.tile(h_initial[np.newaxis, :], (num_particles, 1))
    velocities = np.zeros_like(particles)

    # PSO parameters
    omega = 0.729
    c1 = 1.494
    c2 = 1.494
    max_iter = 500
    personal_best = np.copy(particles)
    obj_personal = np.inf * np.ones(num_particles)
    start_time = time.time()

    # Initial objective evaluation
    obj_vals = np.zeros(num_particles)
    for i in range(num_particles):
        h = particles[i]
        fft_h = np.fft.fft(h)
        fft_g = np.fft.fft(1 - h)
        cc_fft = fft_h * np.conj(fft_g)
        cc = np.fft.ifft(cc_fft).real
        obj_vals[i] = np.max(cc * dx)

    for i in range(num_particles):
        if obj_vals[i] < obj_personal[i]:
            personal_best[i] = particles[i].copy()
            obj_personal[i] = obj_vals[i]

    global_best_index = np.argmin(obj_personal)
    global_best = personal_best[global_best_index]

    # PSO loop
    for iter in range(max_iter):
        # Compute objectives
        obj_vals = np.zeros(num_particles)
        for i in range(num_particles):
            h = particles[i]
            fft_h = np.fft.fft(h)
            fft_g = np.fft.fft(1 - h)
            cc_fft = fft_h * np.conj(fft_g)
            cc = np.fft.ifft(cc_fft).real
            obj_vals[i] = np.max(cc * dx)

        # Update personal bests
        for i in range(num_particles):
            if obj_vals[i] < obj_personal[i]:
                personal_best[i] = particles[i].copy()
                obj_personal[i] = obj_vals[i]

        # Update global best
        global_best_index = np.argmin(obj_personal)
        global_best = personal_best[global_best_index]

        # Update velocities and positions
        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]

            # Clamp and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * required_sum / np.sum(particles[i])

        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break

    # Final evaluation
    best_h = np.clip(personal_best[global_best_index], 0, 1)
    best_h = best_h * required_sum / np.sum(best_h)
    fft_h = np.fft.fft(best_h)
    fft_g = np.fft.fft(1 - best_h)
    cc_fft = fft_h * np.conj(fft_g)
    cc = np.fft.ifft(cc_fft).real
    c5_bound = np.max(cc * dx)

    return (best_h, c5_bound, n_points)
