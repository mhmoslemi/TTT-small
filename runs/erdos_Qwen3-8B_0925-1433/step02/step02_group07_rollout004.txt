        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def generate_binary(n_points, m):
        dx = 2.0 / n_points
        segment_length = 1.0 / m
        n_samples_per_segment = int(round(segment_length / dx))
        h = np.zeros(n_points)
        for i in range(2 * m):
            start = i * n_samples_per_segment
            end = (i + 1) * n_samples_per_segment
            start = np.clip(start, 0, n_points)
            end = np.clip(end, 0, n_points)
            if i % 2 == 0:
                h[start:end] = 1.0
            else:
                h[start:end] = 0.0
        current_sum = np.sum(h) * dx
        scale = 1.0 / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    initial_guesses = []
    if 'initial_h_values' in globals():
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        current_sum = np.sum(h) * dx
        scale = 1.0 / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guesses.append(h)
    else:
        initial_guesses.append(generate_symmetric(n_points))
        initial_guesses.append(generate_periodic(n_points, 1.5))
        initial_guesses.append(generate_periodic(n_points, 2.0))
        initial_guesses.append(generate_binary(n_points, 3))
        initial_guesses.append(generate_binary(n_points, 4))

    best_h = None
    best_c5 = float('inf')

    for guess in initial_guesses:
        h = np.clip(guess, 0, 1)
        current_sum = np.sum(h) * dx
        scale = 1.0 / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        h_flat = h.flatten()

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            return np.max(cc * dx)

        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum

        bounds = [(0, 1)] * n_points
        result = minimize(
            objective,
            h_flat,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq}],
            options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )

        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt.reshape(n_points), 0, 1)
            current_sum_opt = np.sum(h_opt)
            h_opt *= required_sum / current_sum_opt
            cc = correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(cc * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    if best_h is None:
        best_h = generate_symmetric(n_points)
        best_c5 = np.max(correlate(best_h, 1 - best_h, mode='full') * dx)

    return (best_h, best_c5, n_points)
