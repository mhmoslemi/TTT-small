import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        n_points = 128
        initial_h = np.full(n_points, 0.5)
        initial_h[::2] = 1.0
        initial_h[1::2] = 0.0
        total = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / total
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guesses = [initial_h.copy()]
    x = np.linspace(0, 2, n_points)
    c = 0.5 * (1 + np.cos(np.pi * x))
    total_c = np.sum(c)
    c *= (n_points / 2) / total_c
    initial_guesses.append(c)
    np.random.seed(seed)
    r = np.random.rand(n_points)
    r = (r > 0.5).astype(float)
    total_r = np.sum(r)
    r *= (n_points / 2) / total_r
    initial_guesses.append(r)

    best_h = initial_h
    best_c5 = obj(initial_h)

    for h_start in initial_guesses:
        h_start = np.clip(h_start, 0.0, 1.0)
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-8, 'maxiter': 300, 'disp': False})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        current_c5 = obj(np.clip(h_start, 0.0, 1.0))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_start

    best_h = np.clip(best_h, 0.0, 1.0)
    total_h = np.sum(best_h)
    if abs(total_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
