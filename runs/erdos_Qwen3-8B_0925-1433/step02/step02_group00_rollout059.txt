def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:
        m = 150
        first_half = m // 2
        h = np.zeros(m)
        values = [0.25 + 0.5 * (i / (first_half - 1)) for i in range(first_half)]
        for i in range(first_half):
            h[i] = values[i]
            h[m - 1 - i] = values[i]
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (m/2) / sum_h
            h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        bounds = [(0, 1)] * (n_points // 2)
        def objective(x):
            x = np.clip(x, 0, 1)
            h_curr = np.zeros(n_points)
            for i in range(n_points // 2):
                h_curr[i] = x[i]
                h_curr[n_points - 1 - i] = x[i]
            sum_curr = np.sum(h_curr)
            if sum_curr == 0:
                return 1e6
            scale = (n_points / 2) / sum_curr
            h_scaled = np.clip(h_curr * scale, 0, 1)
            corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - (n_points / 4)}]
        res = minimize(objective, h[:n_points//2], method='SLSQP', bounds=bounds, constraints=constraints)
        if res.success:
            h[:n_points//2] = res.x
            for i in range(n_points//2):
                h[i] = res.x[i]
                h[n_points - 1 - i] = res.x[i]
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        if iteration % 50 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points//2)
            h[:n_points//2] = np.clip(h[:n_points//2] + perturb, 0, 1)
            sum_curr = np.sum(h[:n_points//2])
            if sum_curr == 0:
                h[:n_points//2] = np.full_like(h[:n_points//2], 0.5)
            else:
                scale = (n_points / 4) / sum_curr
                h[:n_points//2] = np.clip(h[:n_points//2] * scale, 0, 1)
            for i in range(n_points//2):
                h[i] = h[:n_points//2][i]
                h[n_points - 1 - i] = h[:n_points//2][i]
    return (best_h, best_c5, n_points)
