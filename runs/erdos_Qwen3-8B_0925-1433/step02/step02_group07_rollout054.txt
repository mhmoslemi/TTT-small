    final_cc = correlate(gbest, 1 - gbest, mode='full') * dx
    final_c5 = np.max(final_cc)
    gbest = np.clip(gbest, 0, 1)
    current_sum = np.sum(gbest)
    if current_sum < 1e-12:
        gbest = np.zeros_like(gbest)
    else:
        gbest *= required_sum / current_sum
    return (gbest, final_c5, n_points)
