    initial_guesses.append(rand_h)

    best_h = initial_h
    best_c5 = obj(initial_h)

    for h_start in initial_guesses:
        h_start = np.clip(h_start, 0.0, 1.0)
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        else:
            current_c5 = obj(np.clip(h_start, 0.0, 1.0))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_start

    best_h = np.clip(best_h, 0.0, 1.0)
    total_h = np.sum(best_h)
    if abs(total_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
