        h_binary = (h_vals > 0.5).astype(float)
        intervals = []
        start = None
        for i in range(n_points):
            if h_binary[i] == 1 and (i == 0 or h_binary[i-1] == 0):
                start = i * dx
            elif h_binary[i] == 0 and (i == 0 or h_binary[i-1] == 1):
                if start is not None:
                    end = i * dx
                    intervals.append((start, end))
        if not intervals or len(intervals) < 3:
            intervals = [(0.0, 0.25), (0.5, 0.75), (1.0, 1.25), (1.5, 1.75)]
    else:
        n_points = 95
        dx = 2.0 / n_points
        intervals = [(0.0, 0.25), (0.5, 0.75), (1.0, 1.25), (1.5, 1.75)]

    m = len(intervals)
    start_vars = np.array([i[0] for i in intervals])
    length_vars = np.array([i[1] - i[0] for i in intervals])
    variables = np.hstack((start_vars, length_vars))

    def constraint_sum_lengths(x):
        return np.sum(x[1::2]) - 1.0

    def constraint_pos(x):
        return [x[2*i+1] + x[2*i] - 2.0 for i in range(m)]

    def constraint_order(x):
        return [x[2*i+1] + x[2*i] - x[2*(i+1)] for i in range(m-1)]

    cons = [{'type': 'eq', 'fun': constraint_sum_lengths}]
    for cons_func in [constraint_pos, constraint_order]:
        cons.append({'type': 'ineq', 'fun': lambda x, f=cons_func: f(x)})

    def objective(x):
        h = np.zeros(n_points)
        for i in range(m):
            s = x[2*i]
            l = x[2*i + 1]
            end = s + l
            for j in range(n_points):
                xj = j * dx
                if s <= xj < end:
                    h[j] = 1.0
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    res = minimize(objective, variables, method='SLSQP', constraints=cons, options={'ftol': 1e-8, 'maxiter': 500})

    h_vals = np.zeros(n_points)
    for i in range(m):
        s = res.x[2*i]
        l = res.x[2*i + 1]
        end = s + l
        for j in range(n_points):
            xj = j * dx
            if s <= xj < end:
                h_vals[j] = 1.0

    while np.abs(np.sum(h_vals) * dx - 1.0) > 1e-6:
        h_vals = initial_h_values.copy()
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return (h_vals, c5, n_points)

    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5 = np.max(corr) * dx
    return (h_vals, c5, n_points)
