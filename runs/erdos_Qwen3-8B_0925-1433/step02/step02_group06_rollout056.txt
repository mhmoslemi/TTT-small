    initial_guesses = [initial_h.copy()]
    x = np.linspace(0, 2, n_points)

    # Add structured guesses
    square_wave = np.where(np.sin(np.pi * x) > 0, 1, 0)
    square_wave *= (n_points / 2) / np.sum(square_wave)
    initial_guesses.append(square_wave)

    triangle_wave = np.abs(np.sin(np.pi * x))
    triangle_wave *= (n_points / 2) / np.sum(triangle_wave)
    initial_guesses.append(triangle_wave)

    c_high_freq = 0.5 * (1 + np.cos(2 * np.pi * x))
    c_high_freq *= (n_points / 2) / np.sum(c_high_freq)
    initial_guesses.append(c_high_freq)

    # Add randomized guess
    np.random.seed(seed)
    r = np.random.rand(n_points)
    r = (r > 0.5).astype(float)
    r *= (n_points / 2) / np.sum(r)
    initial_guesses.append(r)

    best_h = initial_h
    best_c5 = obj(initial_h)

    for h_start in initial_guesses:
        h_start = np.clip(h_start, 0.0, 1.0)
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-8, 'maxiter': 1000, 'disp': False})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        else:
            current_c5 = obj(np.clip(h_start, 0.0, 1.0))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_start

    best_h = np.clip(best_h, 0.0, 1.0)
    total_h = np.sum(best_h)
    if abs(total_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
