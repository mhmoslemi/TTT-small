import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 480
    required_sum = n_points / 2
    dx = 2.0 / n_points
    bounds = [(0, 1)] * n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc * dx)

    def constraint_eq(h_flat):
        return np.sum(h_flat.reshape(n_points)) - required_sum

    constraints = [{'type': 'eq', 'fun': constraint_eq}]

    initial_guesses = []

    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_parent = np.interp(x_new, x_initial, initial_h_values)
        h_perturb = h_parent + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * required_sum / np.sum(h_perturb)
        initial_guesses.append(h_perturb)
        x = np.linspace(0, 2, n_points)
        h_sine = 0.5 + 0.5 * np.sin(np.pi * x)
        h_sine = h_sine * required_sum / np.sum(h_sine)
        initial_guesses.append(h_sine)
        h_square = np.zeros(n_points)
        for i in range(n_points):
            h_square[i] = 1.0 if i % 2 == 0 else 0.0
        h_square = h_square * required_sum / np.sum(h_square)
        initial_guesses.append(h_square)
    else:
        x = np.linspace(0, 2, n_points)

        h_sine = h_sine * required_sum / np.sum(h_sine)
        initial_guesses.append(h_sine)
        h_square = np.zeros(n_points)

        h_square = h_square * required_sum / np.sum(h_square)
        initial_guesses.append(h_square)
        h_tri = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
        h_tri = h_tri * required_sum / np.sum(h_tri)
