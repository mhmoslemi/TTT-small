    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    pop_size = 20
    population = []
    for _ in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        candidate = initial_h_values + perturb
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        scale = (n_points / 2) / sum_candidate
        candidate = np.clip(candidate * scale, 0, 1)
        population.append(candidate)
    population = np.array(population)
    best_h = population[0]
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    generations = 0
    while time.time() - start_time < budget_s:
        generations += 1
        new_population = []
        for i in range(population.shape[0]):
            a, b, c = np.random.choice(population.shape[0], 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            sum_trial = np.sum(trial)
            if sum_trial == 0:
                trial = np.full(n_points, 0.5)
            else:
                scale = (n_points / 2) / sum_trial
                trial = np.clip(trial * scale, 0, 1)
            new_population.append(trial)
        population = np.array(new_population)
        for candidate in population:
            current_c5 = compute_c5(candidate, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate.copy()
        if generations % 100 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            for i in range(population.shape[0]):
                population[i] = np.clip(population[i] + perturb, 0, 1)
                sum_perturb = np.sum(population[i])
                if sum_perturb == 0:
                    population[i] = np.full(n_points, 0.5)
                else:
                    scale = (n_points / 2) / sum_perturb
                    population[i] = np.clip(population[i] * scale, 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
