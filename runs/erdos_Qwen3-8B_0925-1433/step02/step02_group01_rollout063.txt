import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()

    start_time = time.time()
    T_initial = 0.5
    cooling_rate = 0.995
    iterations = 0
    max_iterations = int(budget_s / 10)  # Allocate ~10 iterations per second
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        iterations += 1
        # Compute correlation once
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)

        j = (np.arange(n_points) + k_max) % n_points

        h_candidate = h + perturb

        else:

            h_candidate = np.clip(h_candidate * scale_candidate, 0, 1)
        # Evaluate new_c5
        new_c5 = compute_c5(h_candidate, dx, n_points)
        current_c5 = compute_c5(h, dx, n_points)
        # Acceptance criterion

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
