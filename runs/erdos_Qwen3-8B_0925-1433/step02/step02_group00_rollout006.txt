    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_points = len(h)
    dx = 2.0 / n_points
    
    start_time = time.time()
    iteration = 0
    max_iter = 300
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        
        # Check if current solution is better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Local optimization to reduce max_corr
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points/2) / sum_x
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
        if res.success:
            h = res.x.copy()
        
        # Perturb every 10 iterations
        if iteration % 10 == 0:
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.full_like(h, 0.5)
            else:
                scale = (n_points/2) / sum_h
                h = np.clip(h * scale, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
