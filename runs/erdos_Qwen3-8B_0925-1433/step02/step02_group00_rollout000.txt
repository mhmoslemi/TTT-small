            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
    else:

        h = np.clip(h * scale, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    def obj(x):

        if sum_x == 0:

        scale = (n_points / 2) / sum_x
        x = np.clip(x * scale, 0, 1)

        return np.max(corr) * dx

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
