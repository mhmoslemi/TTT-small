        intervals = []
        start = 0
        for i in range(len(h_binary)):
            if h_binary[i] == 1:
                if start != i:
                    intervals.append((start, i-1))
                    start = i
        if start < len(h_binary):
            intervals.append((start, len(h_binary)-1))
        def is_valid(intervals):
            for i in range(len(intervals)-1):
                if intervals[i][1] >= intervals[i+1][0]:
                    return False
            return True
        if not is_valid(intervals):
            n_points = len(h_initial)
            intervals = []
            n_intervals = 4
            spacing = 2.0 / n_intervals
            for i in range(n_intervals):
                s = i * spacing
                e = s + spacing
                intervals.append((s, e))
        h = np.zeros(len(h_initial))
        dx = 2.0 / len(h_initial)
        for s, e in intervals:
            start_idx = int(np.round(s / dx))
            end_idx = int(np.round(e / dx))
            if start_idx < 0:
                start_idx = 0
            if end_idx >= len(h):
                end_idx = len(h) - 1
            h[start_idx:end_idx+1] = 1
    else:
        n_points = 100
        h = np.zeros(n_points)
        intervals = []
        spacing = 2.0 / 4
        for i in range(4):
            s = i * spacing
            e = s + spacing
            intervals.append((s, e))
        dx = 2.0 / n_points
        for s, e in intervals:
            start_idx = int(np.round(s / dx))
            end_idx = int(np.round(e / dx))
            if start_idx < 0:
                start_idx = 0
            if end_idx >= n_points:
                end_idx = n_points - 1
            h[start_idx:end_idx+1] = 1

    m = 4
    variables = []
    for s, e in intervals:
        variables.append(s)
        variables.append(e)
    dx = 2.0 / len(h)

    def objective(vars):
        intervals_opt = []
        for i in range(m):
            s = vars[2*i]
            e = vars[2*i + 1]
            if e < s:
                return float('inf')
            intervals_opt.append((s, e))
        for j in range(m-1):
            if intervals_opt[j][1] >= intervals_opt[j+1][0]:
                return float('inf')
        h = np.zeros(len(h))
        for s, e in intervals_opt:
            start_idx = int(np.round(s / dx))
            end_idx = int(np.round(e / dx))
            if start_idx < 0:
                start_idx = 0
            if end_idx >= len(h):
                end_idx = len(h) - 1
            h[start_idx:end_idx+1] = 1
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_total_length(vars):
        total = 0
        for i in range(m):
            s = vars[2*i]
            e = vars[2*i + 1]
            total += e - s
        return total - 1.0

    constraints = [{'type': 'eq', 'fun': constraint_total_length}]
    for j in range(m-1):
        def constraint_non_overlap_j(vars, j=j):
            s_prev = vars[2*j]
            e_prev = vars[2*j + 1]
            s_next = vars[2*(j+1)]
            return s_next - e_prev
        constraints.append({'type': 'ineq', 'fun': lambda x, j=j: constraint_non_overlap_j(x, j)})

    bounds = [(0.0, 2.0) for _ in range(m*2)]

    res = minimize(objective, variables, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-8, 'maxiter': 500})
    if res.success:
        best_vars = res.x
    else:
        best_vars = variables

    intervals_opt = []
    for i in range(m):
        s = best_vars[2*i]
        e = best_vars[2*i + 1]
        intervals_opt.append((s, e))
    h_final = np.zeros(len(h))
    for s, e in intervals_opt:
        start_idx = int(np.round(s / dx))
        end_idx = int(np.round(e / dx))
        if start_idx < 0:
            start_idx = 0
        if end_idx >= len(h):
            end_idx = len(h) - 1
        h_final[start_idx:end_idx+1] = 1
    total = np.sum(h_final) * dx
    if abs(total - 1.0) > 1e-5:
        h_final = h
    return (h_final, res.fun, len(h_final))
