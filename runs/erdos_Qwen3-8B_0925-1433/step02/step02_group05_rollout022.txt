def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Scale initial_h_values
    sum_initial = np.sum(initial_h_values)
    scale_initial = (n_points / 2.0) / sum_initial if sum_initial != 0 else 1.0
    initial_scaled = np.clip(initial_h_values * scale_initial, 0, 1)
    # Precompute frequencies
    freqs = np.linspace(0, 1/(2*dx), n_points)
    # Generate variations
    variations = []
    # Variation 1
    H1 = np.zeros_like(freqs)
    mask_low = freqs < 0.2
    mask_high = freqs > 0.8
    H1[mask_low] = 0
    H1[mask_high] = 1
    mid_mask = ~mask_low & ~mask_high
    H1[mid_mask] = 1 / (1 + np.exp(-50 * (freqs[mid_mask] - 0.2)))
    h1 = np.fft.irfft(H1)
    h1 = np.clip(h1, 0, 1)
    sum_h1 = np.sum(h1)
    scale_h1 = (n_points / 2.0) / sum_h1 if sum_h1 != 0 else 1.0
    h1 = h1 * scale_h1
    variations.append(h1)
    # Variation 2
    H2 = np.zeros_like(freqs)
    mask_low = freqs < 0.3
    mask_high = freqs > 0.9
    H2[mask_low] = 0
    H2[mask_high] = 1
    mid_mask = ~mask_low & ~mask_high
    H2[mid_mask] = 1 / (1 + np.exp(-100 * (freqs[mid_mask] - 0.3)))
    h2 = np.fft.irfft(H2)
    h2 = np.clip(h2, 0, 1)
    sum_h2 = np.sum(h2)
    scale_h2 = (n_points / 2.0) / sum_h2 if sum_h2 != 0 else 1.0
    h2 = h2 * scale_h2
    variations.append(h2)
    # Variation 3
    H3 = np.zeros_like(freqs)
    mask_low = freqs < 0.1
    mask_high = freqs > 0.7
    H3[mask_low] = 0
    H3[mask_high] = 1
    mid_mask = ~mask_low & ~mask_high
    H3[mid_mask] = 1 / (1 + np.exp(-30 * (freqs[mid_mask] - 0.1)))
    h3 = np.fft.irfft(H3)
    h3 = np.clip(h3, 0, 1)
    sum_h3 = np.sum(h3)
    scale_h3 = (n_points / 2.0) / sum_h3 if sum_h3 != 0 else 1.0
    h3 = h3 * scale_h3
    variations.append(h3)
    # Variation 4
    H4 = np.zeros_like(freqs)
    mask_low = freqs < 0.25
    mask_high = freqs > 0.95
    H4[mask_low] = 0
    H4[mask_high] = 1
    mid_mask = ~mask_low & ~mask_high
    H4[mid_mask] = 1 / (1 + np.exp(-40 * (freqs[mid_mask] - 0.25)))
    h4 = np.fft.irfft(H4)
    h4 = np.clip(h4, 0, 1)
    sum_h4 = np.sum(h4)
    scale_h4 = (n_points / 2.0) / sum_h4 if sum_h4 != 0 else 1.0
    h4 = h4 * scale_h4
    variations.append(h4)
    variations.append(initial_scaled)
    # Define objective and constraints
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_scaled.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    for guess in variations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - n_points / 2.0) > 1e-6:
            h_best = h_best * (n_points / 2.0) / sum_h
            h_best = np.clip(h_best, 0, 1)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best.copy()
            best_c5 = c5
    # Final check
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2.0) > 1e-6:
        best_h = initial_scaled.copy()
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
