            return corr[selected_lag] * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]

        # Perform local optimization
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 150})
        if result.success:
            h = clip_and_normalize(result.x)
        else:
            h = clip_and_normalize(h)

        # Check if improvement
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            # Random perturbation if no improvement
            noise = np.random.normal(0, 0.05, n_points)
            h = clip_and_normalize(h + noise)

    return (best_h, best_c5, n_points)
