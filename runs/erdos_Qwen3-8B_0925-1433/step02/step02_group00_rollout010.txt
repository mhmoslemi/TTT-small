        x_initial += delta
    else:
        third = half_m // 3
        x_initial = np.zeros(half_m)
        x_initial[:third] = 0.75
        x_initial[third:2*third] = 0.5
        x_initial[2*third:] = 0.25
        current_sum = np.sum(x_initial)
        delta = (m/4 - current_sum) / half_m
        x_initial += delta
    
    def build_h(x):
        h = np.zeros(m)
        h[:len(x)] = x
        h[len(x):] = x[::-1]
        return h
    
    def objective(x):
        h = build_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return -max_corr * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - m/4}]
    bounds = [(0, 1) for _ in range(half_m)]
    result = minimize(objective, x_initial, method='SLSQP', bounds=bounds, constraints=cons)
    
    best_x = result.x if result.success else x_initial
    best_h = build_h(best_x)
    best_c5 = -result.fun if result.success else -objective(x_initial)
    
    return (best_h, best_c5, m)
