def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

        population[i] = np.clip(population[i] + perturb, 0, 1)
        sum_perturb = np.sum(population[i])
        scale = (n_points / 2) / sum_perturb if sum_perturb != 0 else 1
        population[i] = np.clip(population[i] * scale, 0, 1)

    gen = 0
    while time.time() - start < budget_s:
        scores = [compute_c5(h, dx, n_points) for h in population]
        top = np.argsort(scores)[:len(population)//2]
        selected = [population[i] for i in top]
        new_pop = []
        for _ in range(len(population)//2):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = (p1 + p2) * 0.5
            mut = np.random.uniform(-0.2, 0.2, n_points)
            child += mut
            s = np.sum(child)
            scale = (n_points / 2) / s if s != 0 else 1
            child = np.clip(child * scale, 0, 1)
            new_pop.append(child)
        population = selected + new_pop
        top_idx = np.argsort(scores)[:10]
        for i in top_idx:
            h = population[i].copy()
            grad = np.zeros(n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
