        res = minimize(objective, init_vars, method='SLSQP', bounds=bounds, tol=1e-6)
        if res.success:
            new_h = np.zeros(n_points)
            new_h[:len(res.x)] = res.x
            new_h[len(res.x):] = res.x[::-1]
            sum_h = np.sum(new_h)
            if sum_h > 1e-6:
                scale = (n_points / 2) / sum_h
                new_h = np.clip(new_h * scale, 0, 1)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
            return True
        return False
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        if optimize():
            perturb = np.random.uniform(-0.05, 0.05, size=n_points // 2)
            perturb = np.clip(perturb, -0.1, 0.1)
            h[:n_points // 2] += perturb
            h[:n_points // 2] = np.clip(h[:n_points // 2], 0, 1)
            sum_h = np.sum(h[:n_points // 2])
            if sum_h > 1e-6:
                scale = (n_points / 2) / sum_h
                h[:n_points // 2] = np.clip(h[:n_points // 2] * scale, 0, 1)
            h[n_points // 2:] = h[:n_points // 2][::-1]
        else:
            # Fallback to initial guess
            best_h = h.copy()
            best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
