import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h_dim = len(h)
        if h_dim != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, h_dim), h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1.0) > 1e-6:
            h = h * (1.0 / sum_h) * 1.0 * (n_points / 2.0)
    else:
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[indices] = 1.0
        h = np.clip(h, 0, 1)

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {
        'maxiter': 500,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False,
        'iprint': -1
    }
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        best_h = best_h * (1.0 / sum_h) * 1.0 * (n_points / 2.0)

    return (best_h, c5_bound, n_points)
