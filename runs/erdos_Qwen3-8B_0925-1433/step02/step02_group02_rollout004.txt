from scipy.optimize import minimize

        p = np.clip(p, 0, 1)

            p = np.ones(n_points) * 0.5
        else:
            scale = (n_points / 2) / sum_p
            p = np.clip(p * scale, 0, 1)
        particles.append(p)
        velocities.append(np.zeros(n_points))
