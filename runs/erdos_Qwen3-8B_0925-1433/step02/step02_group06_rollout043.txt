            best_h = res.x
        else:
            current_c5 = obj(np.clip(h_start, 0.0, 1.0))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_start

    best_h = np.clip(best_h, 0.0, 1.0)
    total_h = np.sum(best_h)
    if abs(total_h - n_points / 2) > 1e-6:
        best_h = initial_h
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
