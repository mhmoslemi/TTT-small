    import time

    # Assume initial_h_values is provided

    scale = (n_points / 2) / sum_h

    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    T = 1.0
    cooling_rate = 0.99
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        # Compute correlation to find max k
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        # Compute approximate sensitivity
        sensitivity = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k_max < n_points:
                sensitivity[i] += (1 - h[i + k_max])
            if 0 <= i - k_max < n_points:
                sensitivity[i] -= h[i - k_max]
        # Select top 20% indices
        top_indices = np.argsort(np.abs(sensitivity))[-int(0.2 * n_points):]
        # Generate perturbation
        perturb = np.zeros(n_points)
        perturb[top_indices] = np.random.normal(0, 0.05, len(top_indices))
        h_candidate = h + perturb
        h_candidate = np.clip(h_candidate * scale, 0, 1)
        c5_candidate = compute_c5(h_candidate, dx, n_points)
        # Acceptance criterion
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        else:

            if np.random.random() < np.exp(-delta / T):
                best_h = h_candidate.copy()

    corr = np.correlate(h, shifted, mode='full')
