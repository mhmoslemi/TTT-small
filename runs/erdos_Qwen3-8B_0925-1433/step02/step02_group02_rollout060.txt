    personal_best = particles.copy()
    personal_c5 = [compute_c5(p, dx, n_points) for p in particles]
    global_best = np.argmin(personal_c5)
    best_h = personal_best[global_best].copy()
    best_c5 = personal_c5[global_best]

    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        for i in range(len(particles)):
            c5 = compute_c5(particles[i], dx, n_points)
            if c5 < personal_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_c5[i] = c5
        global_best_idx = np.argmin(personal_c5)
        global_best_c5 = personal_c5[global_best_idx]
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5
            best_h = personal_best[global_best_idx].copy()
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_i = vel[i]
            pbest = personal_best[i]
            gbest = personal_best[global_best_idx]
            vel_i = 0.9 * vel_i + 1.5 * r1 * (pbest - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            vel[i] = vel_i
            particles[i] += vel_i
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            scale = (n_points / 2) / sum_p if sum_p != 0 else 1.0
            particles[i] = np.clip(particles[i] * scale, 0, 1)

        top_idx = np.argsort(personal_c5)[:5]
        for idx in top_idx:
            h_candidate = particles[idx].copy()
            def obj(x):
                h = x.copy()
                sum_h = np.sum(h)
                scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
                h = np.clip(h * scale, 0, 1)
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                particles[idx] = res.x
            else:
                particles[idx] = np.clip(h_candidate, 0, 1)
                sum_p = np.sum(particles[idx])
                scale = (n_points / 2) / sum_p if sum_p != 0 else 1.0
                particles[idx] = np.clip(particles[idx] * scale, 0, 1)
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
