import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_pop = [initial_h_values.copy() for _ in range(20)]
    for i in range(20):
        h_pop[i] = np.clip(h_pop[i] + np.random.uniform(-0.1, 0.1, n_points), 0, 1)
        h_pop[i] = np.clip(h_pop[i] * (n_points / 2.0) / np.sum(h_pop[i]), 0, 1)
    best_h = min(h_pop, key=lambda x: compute_c5(x, dx, n_points))
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        scores = [compute_c5(h, dx, n_points) for h in h_pop]
        top_indices = np.argsort(scores)[:10]
        parents = [h_pop[i] for i in top_indices]
        offspring = []
        for _ in range(10):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = np.zeros(n_points)
            for i in range(n_points):
                child[i] = p1[i] * np.random.rand() + p2[i] * (1 - np.random.rand())
            child = np.clip(child, 0, 1)
            child = np.clip(child * (n_points / 2.0) / np.sum(child), 0, 1)
            offspring.append(child)
        h_pop = [h for h in h_pop if h is not None]
        h_pop.extend(offspring)
        h_pop.sort(key=lambda x: compute_c5(x, dx, n_points))
        h_pop = h_pop[:20]
        if gen % 50 == 0 and np.random.rand() < 0.3:
            for i in range(len(h_pop)):
                h_pop[i] = np.clip(h_pop[i] + np.random.normal(0, 0.05, n_points), 0, 1)
                h_pop[i] = np.clip(h_pop[i] * (n_points / 2.0) / np.sum(h_pop[i]), 0, 1)
        current_best = min(h_pop, key=lambda x: compute_c5(x, dx, n_points))
        if compute_c5(current_best, dx, n_points) < best_c5:
            best_c5 = compute_c5(current_best, dx, n_points)
            best_h = current_best
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
