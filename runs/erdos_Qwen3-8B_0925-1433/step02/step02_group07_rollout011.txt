import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 480
    required_sum = n_points / 2
    dx = 2.0 / n_points

    initial_guesses = []

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h)
        initial_guesses.append(h_initial)

    # Add smooth pattern
    x = np.linspace(0, 2, n_points)
    h_smooth = 0.5 * (1 - np.cos(np.pi * x))
    initial_guesses.append(h_smooth)

    # Add binary pattern
    binary_initial = np.where(np.arange(n_points) % 2 == 0, 1, 0)
    initial_guesses.append(binary_initial)

    best_h = None
    best_c5 = np.inf
    start_time = time.time()

    for h_initial in initial_guesses:
        h_flat = h_initial.flatten()
        h_flat = np.clip(h_flat, 0, 1)

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            cc = correlate(h, 1 - h, mode='full')
            c_values = cc * dx
            return np.max(c_values)

        def constraint_eq(h_flat):
            return np.sum(h_flat.reshape(n_points)) - required_sum

        bounds = [(0, 1)] * n_points
        result = minimize(
            fun=objective,
            x0=h_flat,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq}],
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
        )

        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            current_sum = np.sum(h_opt)
            h_opt *= required_sum / current_sum
            cc = correlate(h_opt, 1 - h_opt, mode='full')
            current_c5 = np.max(cc * dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt

        cc = correlate(h_opt, 1 - h_opt, mode='full')
        best_c5 = np.max(cc * dx)
        best_h = h_opt

    return (best_h, best_c5, n_points)
