import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_required}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_guesses():
        guesses = []
        # Perturbed parent
        perturb = np.random.normal(0, 0.03, n_points)
        guess = np.clip(initial_h + perturb, 0, 1)
        guess = guess * (sum_required / np.sum(guess))
        guesses.append(guess)
        # Periodic pattern
        i = np.arange(n_points)
        guess = 0.5 + 0.5 * np.cos(2 * np.pi * i / n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess * (sum_required / np.sum(guess))
        guesses.append(guess)
        # Symmetric pattern
        guess = 0.5 + 0.5 * (1 - np.abs(i - n_points / 2) / (n_points / 2))
        guess = np.clip(guess, 0, 1)
        guess = guess * (sum_required / np.sum(guess))
        guesses.append(guess)
        # Alternating pattern
        guess = 0.5 + 0.5 * (1 - (i % 2))
        guess = np.clip(guess, 0, 1)
        guess = guess * (sum_required / np.sum(guess))
        guesses.append(guess)
        # Random oscillation
        guess = np.random.rand(n_points) * 2 - 1
        guess = np.clip(guess, -1, 1)
        guess = 0.5 + 0.5 * guess
        guess = np.clip(guess, 0, 1)
        guess = guess * (sum_required / np.sum(guess))
        guesses.append(guess)
        # Filtered oscillation
        guess = np.random.rand(n_points)
        guess = np.fft.irfft(np.fft.rfft(guess, n=n_points) * np.exp(-0.5 * (np.arange(n_points) / n_points)**2))
        guess = np.clip(guess, 0, 1)
        guess = guess * (sum_required / np.sum(guess))
        guesses.append(guess)
        return guesses

    guesses = generate_guesses()
    best_h = np.copy(initial_h)
    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - sum_required) > 1e-6:
            h_best = h_best * (sum_required / sum_h)
            h_best = np.clip(h_best, 0, 1)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5

    sum_h = np.sum(best_h)
    if abs(sum_h - sum_required) > 1e-6:
        best_h = initial_h.copy()
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
