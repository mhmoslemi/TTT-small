import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_scale(h):
        total = np.sum(h)
        scale = (n_points / 2) / total if total > 0 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    h = initial_h_values.copy()
    h = clip_and_scale(h)
    best_h = h
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        worst_k_index = np.argmax(corr)
        worst_lag = worst_k_index - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            lag = worst_lag
            j_plus = j + lag
            if j_plus >= n_points:
                h_plus = 0.0
            elif j_plus < 0:
                h_plus = 0.0
            else:
                h_plus = h[int(j_plus)]
            j_minus = j - lag
            if j_minus >= n_points:
                h_minus = 0.0
            elif j_minus < 0:
                h_minus = 0.0
            else:
                h_minus = h[int(j_minus)]
            grad[j] = dx * (1.0 - h_plus - h_minus)
        alpha = 0.1 * (0.99 ** _)
        h_update = h - alpha * grad
        h_update = np.clip(h_update, 0, 1)
        h_update = clip_and_scale(h_update)
        corr_new = np.correlate(h_update, 1 - h_update, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_h = h_update
            best_c5 = new_c5
        h = h_update
        if np.random.rand() < 0.01:
            h = np.random.rand(n_points)
            h = clip_and_scale(h)

    return (best_h, best_c5, n_points)
