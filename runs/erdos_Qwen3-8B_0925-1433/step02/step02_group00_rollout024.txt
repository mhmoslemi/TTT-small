import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initialize h based on initial_h_values if available, else default
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.full_like(h, 0.5)
        else:
            scale = (len(h)/2) / sum_h
            h = np.clip(h * scale, 0, 1)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.zeros(n_points)
        h[:n_points//3] = 0.75
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.25
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        h = np.clip(h * scale, 0, 1)

    # Create multiple initial guesses
    n_initial = 5
    initial_guesses = [h.copy() for _ in range(n_initial)]
    for i in range(n_initial):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        initial_guesses[i] += noise
        initial_guesses[i] = np.clip(initial_guesses[i], 0, 1)
        sum_h = np.sum(initial_guesses[i])
        scale = target_sum / sum_h
        initial_guesses[i] = np.clip(initial_guesses[i] * scale, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_time = time.time()

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1

        # Randomly select a guess to optimize
        guess = np.random.choice(initial_guesses)
        guess = guess.copy()

        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = target_sum / sum_x
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        res = minimize(obj, guess, method='SLSQP', bounds=[(0,1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success:
            current_h = res.x.copy()
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

        # Perturb the current best solution
        if iteration % 10 == 0:
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            perturbed = np.clip(best_h + perturb, 0, 1)
            sum_perturbed = np.sum(perturbed)
            scale = target_sum / sum_perturbed
            best_h = np.clip(perturbed * scale, 0, 1)

    return (best_h, best_c5, n_points)
