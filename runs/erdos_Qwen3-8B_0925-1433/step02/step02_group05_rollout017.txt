import numpy as np

import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    guesses = [initial_h_values.copy()]

    x = np.linspace(0, 2, n_points)

        sine_pattern *= (n_points / 2) / sum_sin
    guesses.append(sine_pattern)

    cosine_pattern = np.clip(cosine_pattern, 0, 1)
    sum_cos = np.sum(cosine_pattern)
    if sum_cos > 1e-6:

    guesses.append(square_pattern)

    random_pattern = np.random.rand(n_points)
