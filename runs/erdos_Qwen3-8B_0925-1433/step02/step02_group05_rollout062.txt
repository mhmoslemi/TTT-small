    # Validate initial_h_values
    sum_initial = np.sum(initial_h_values)
    if sum_initial != 0:
        scale_initial = (n_points / 2) / sum_initial
    else:
        scale_initial = 1.0
    h_initial = np.clip(initial_h_values * scale_initial, 0, 1)
    sum_h_initial = np.sum(h_initial) * dx

    # Objective function
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def create_freq_guess(low_frac, high_frac):
        if low_frac >= high_frac:
            return h_initial
        n_fft = n_points
        freqs = np.fft.rfftfreq(n_fft, dx)
        H = np.zeros_like(freqs)
        len_freqs = len(freqs)
        for m in range(len_freqs):
            frac = m / len_freqs
            if frac < low_frac:
                H[m] = 0
            elif frac > high_frac:
                H[m] = 1
            else:
                H[m] = (frac - low_frac) / (high_frac - low_frac)
        h_guess = np.fft.irfft(H)
        h_guess = np.clip(h_guess, 0, 1)
        sum_h_guess = np.sum(h_guess) * dx
        scale = (1.0 / sum_h_guess) if sum_h_guess != 0 else 1.0
        return h_guess * scale

    initial_guesses = [h_initial]
    low_fracs = [0.15, 0.25]
    high_fracs = [0.65, 0.85]
    for low in low_fracs:
        for high in high_fracs:
            if low < high:
                initial_guesses.append(create_freq_guess(low, high))

    best_h = h_initial
    best_c5 = obj(h_initial)
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best) * dx
        if abs(sum_h - 1) > 1e-6:
            h_best = h_best * (1.0 / sum_h)
            h_best = np.clip(h_best, 0, 1)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5

    return (best_h, best_c5, n_points)
