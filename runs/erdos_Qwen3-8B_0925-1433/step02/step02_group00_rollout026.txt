import numpy as np

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize

    if initial_h_values is not None:
        h = initial_h_values.copy()
        m = len(h)
    else:
        m = 150
        h = np.zeros(m)
        for i in range(m):
            if i < m//3:
                h[i] = 0.75
            elif i < 2*m//3:
                h[i] = 0.5
            else:
                h[i] = 0.25
        sum_h = np.sum(h)
        scale = (m / 2) / sum_h
        h = np.clip(h * scale, 0, 1)

    dx = 2.0 / m

    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - m/2}]
    bounds = [(0, 1)] * m

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-5)

    best_h = result.x
    best_c5 = objective(best_h)

    perturb = np.random.uniform(-0.05, 0.05, size=m)
    h_perturb = np.clip(best_h + perturb, 0, 1)
    sum_perturb = np.sum(h_perturb)
    if sum_perturb != 0:
        scale_perturb = (m / 2) / sum_perturb
        h_perturb = np.clip(h_perturb * scale_perturb, 0, 1)
    c5_perturb = objective(h_perturb)
    if c5_perturb < best_c5:
        best_h = h_perturb
        best_c5 = c5_perturb

    return (best_h, best_c5, m)
