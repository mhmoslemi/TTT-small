import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    pop_size = 20
    population = []
    for _ in range(pop_size):
        h = initial_h_values.copy()
        h += np.random.uniform(-0.2, 0.2, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    generations = 0
    while time.time() - start_time < budget_s:
        generations += 1
        new_population = []
        for i in range(len(population)):
            r1, r2, r3 = np.random.choice(len(population), 3, replace=False)
            mutant = population[r1] + 0.8 * (population[r2] - population[r3])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            sum_trial = np.sum(trial)
            scale = (n_points / 2) / sum_trial
            trial = np.clip(trial * scale, 0, 1)
            c5_trial = compute_c5(trial, dx, n_points)
            c5_current = compute_c5(population[i], dx, n_points)
            if c5_trial < c5_current:
                new_population.append(trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
        population = new_population
        if generations % 100 == 0:
            for i in range(len(population)):
                population[i] += np.random.uniform(-0.1, 0.1, n_points)
                population[i] = np.clip(population[i], 0, 1)
                sum_pop = np.sum(population[i])
                scale = (n_points / 2) / sum_pop
                population[i] = np.clip(population[i] * scale, 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
