import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h * (target_sum / np.sum(initial_h))
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    learning_rate = 0.5
    max_iterations = 2000
    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        k_max = np.argmax(corr)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            idx1 = (j + k_max) % n_points
            idx2 = (j - k_max) % n_points
            gradient[j] = dx * (1 - h[idx1]) - dx * h[idx2]
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum != 0:
            h = h_new * target_sum / h_new_sum
        else:
            h = h_new
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if new_c5 < current_c5:
            learning_rate *= 1.01
        else:
            learning_rate *= 0.99
        if time.time() - start_time > budget_s:
            break
    return (best_h, best_c5, n_points)
