        k = max_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]
            else:
                gradient[i] = 0
        step = learning_rate * gradient
        new_h = h - step
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            scale = target_sum / sum_new
            new_h = new_h * scale
        current_c5 = compute_c5(new_h, 2.0 / n_points, n_points)
        if current_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = current_c5
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = h + noise
            perturbed = np.clip(perturbed, 0, 1)
            sum_perturbed = np.sum(perturbed)
            if sum_perturbed > 0:
                perturbed = perturbed * (target_sum / sum_perturbed)
            current_c5_p = compute_c5(perturbed, 2.0 / n_points, n_points)
            if current_c5_p < best_c5:
                best_h = perturbed.copy()
                best_c5 = current_c5_p
        learning_rate *= decay_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
