    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max
        valid = np.arange(len(h) - k)
        pairs = [(i, i + k) for i in valid]
        indices = list(set([i for pair in pairs for i in pair]))
        m = len(indices)
        x = cp.Variable(m)
        sum_original = sum(h[i] for i in indices)
        terms = [x[indices.index(i)] * x[indices.index(j)] for i, j in pairs]
        obj = cp.Maximize(cp.sum(terms))
        constraints = [cp.sum(x) == sum_original, 0 <= x, x <= 1]
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS)
        except:
            continue
        for idx, val in zip(indices, x.value):
            h[idx] = max(0, min(1, val))
        h = h * (target_sum / np.sum(h))
        curr_c5 = compute_c5(h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
