import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population

    start_time = time.time()

            break

        fitness = [(-compute_c5(h, dx, n_points), h) for h in population]

        # Generate offspring
        offspring = []
        for _ in range(len(population) - len(elites)):
            # Crossover
            parent1, parent2 = np.random.choice(elites, 2, replace=False)
            child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
            # Mutation

                child[idxs] += np.random.normal(0, 0.05, len(idxs))

            child = np.clip(child, 0, 1)

        # Replace population

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
