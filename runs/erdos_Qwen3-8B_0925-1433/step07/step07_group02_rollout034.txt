import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_sum = np.sum(initial_h)
    initial_scaled = initial_h * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iterations = 0
    max_iterations = 1000
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        shifted = (np.arange(n_points) + k_max) % n_points
        def objective(x):
            return np.dot(x, 1 - x[shifted])
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
