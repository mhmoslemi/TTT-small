    target_sum = n_points / 2
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    max_iterations = 10000
    delta = 0.05
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        m = np.argmax(corr)
        lag = m - (n_points - 1)
        if lag >= 0:
            valid_i_start = 0
            valid_i_end = n_points - 1 - lag
        else:
            valid_i_start = -lag
            valid_i_end = n_points - 1
        valid_indices = np.arange(valid_i_start, valid_i_end + 1)
        if not valid_indices.size:
            break
        max_idx = None
        max_val = -np.inf
        for i in valid_indices:
            val = h[i] * (1 - h[i + lag])
            if val > max_val:
                max_val = val
                max_idx = i
        if max_idx is None:
            break
        h_new = h.copy()
        h_new[max_idx] = max(0, h_new[max_idx] - delta)
        h_new[max_idx + lag] = min(1, h_new[max_idx + lag] + delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        new_c5 = compute_c5(h_new, dx, n_points)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        if _ % 500 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * target_sum / np.sum(h)
        delta *= 0.9995
    return (best_h, best_c5, n_points)
