        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        if k > 0:
            # Sinusoidal perturbation at lag k frequency
            freq = np.pi * k / (n_points)
            perturb = 0.01 * np.sin(freq * np.arange(n_points))
        else:
            # Random noise for other lags
            perturb = 0.015 * np.random.normal(size=n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        # Enforce integral constraint
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8, options={'x_scale': 'jac', 'ftol': 1e-9})
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
