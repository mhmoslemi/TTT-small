import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()

    def clip_and_rescale(h, target_sum, n_points):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = target_sum / current_sum if current_sum > 0 else 1.0
        return h * scale

    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        return np.max(corr) * (2.0 / n_points)

    def refine(x):
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, x, method='SLSQP', bounds=bounds, constraints=cons)
        return res.x if res.success else x

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h + np.random.normal(0, 0.1, n_points)
        perturbed = clip_and_rescale(perturbed, target_sum, n_points)
        perturbed = refine(perturbed)
        curr_c5 = compute_c5(perturbed, 2.0 / n_points, n_points)
        if curr_c5 < best_c5:
            best_h = perturbed.copy()
            best_c5 = curr_c5

    for _ in range(50):

        ref = refine(best_h)

        if curr_c5 < best_c5:
            best_h = ref.copy()
            best_c5 = curr_c5

    return best_h, best_c5, n_points

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
