import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    if initial_h_values is not None:
        h_pop = [initial_h_values.copy() * target_sum / np.sum(initial_h_values) if np.sum(initial_h_values) > 0 else np.zeros(n_points)]

            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            h_pop.append(np.clip(h, 0, 1))
    else:
        h_pop = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(30)]
        h_pop = [np.clip(h, 0, 1) for h in h_pop]

    h_pop = np.array(h_pop)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_pop[0]
    best_c5 = compute_c5(best_h)
    population_size = 30
    mutation_rate = 0.15
    mutation_std = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    max_iterations = 350
    early_stopping = 50
    start_time = time.time()
    iteration = 0

    while iteration < max_iterations and time.time() - start_time < budget_s:
        fitness = np.array([compute_c5(ind) for ind in h_pop])

                child = parents[i]
