```python
import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else np.zeros(n_points)

    def de_obj(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped)) if np.sum(h_clipped) > 0 else np.zeros_like(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    result_de = differential_evolution(de_obj, bounds, strategy='best1bin', 
                                      popsize=15, tol=1e-8, maxiter=250,
                                      mutation=(0.5, 1.0), recombination=0.9)
    best_h = result_de.x

    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp,
                            constraints=cons, tol=1e-10)
    best_h_refined = result_slsqp.x
    c5 = result_slsqp.fun

    return (best_h_refined, c5, n_points)
```
