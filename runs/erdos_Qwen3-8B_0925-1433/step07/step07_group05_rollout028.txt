import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = initial_h_values.copy()
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iter = 1000
    step_size = 0.1
    decay_rate = 0.999
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        l = np.argmax(corr)
        k = l - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = j + k
            term1 = 1.0 - (current_h[j_plus_k] if 0 <= j_plus_k < n_points else 0.0)
            j_minus_k = j - k
            term2 = (current_h[j_minus_k] if 0 <= j_minus_k < n_points else 0.0)
            grad[j] = term1 - term2
        current_h -= step_size * grad

        current_h = current_h * target_sum / np.sum(current_h)
        new_c5 = compute_c5(current_h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        step_size *= decay_rate
    return (best_h, best_c5, n_points)
