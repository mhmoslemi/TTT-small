import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    import numpy as np
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    np.random.seed(seed)
    population_size = 50
    population = []
    for _ in range(population_size):
        h = initial_h_values.copy()
        h = h * target_sum / np.sum(h)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        c5_values = [compute_c5(h, dx, n_points) for h in population]
        elite_idx = np.argsort(c5_values)[:int(0.3 * population_size)]
        elites = [population[i] for i in elite_idx]
        new_population = elites
        for _ in range(population_size - len(elites)):
            p1, p2 = np.random.choice(elites, size=2, replace=False)
            child = (p1 + p2) / 2
            corr = np.correlate(child, 1 - child, mode='full')
            top_lags = np.argsort(corr)[-int(0.2 * len(corr)):]
            for lag in top_lags:
                if lag >= len(child):
                    continue
                noise = np.random.normal(0, 0.1, len(child))
                child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        for h in population:
            c5 = compute_c5(h, dx, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if best_c5 < 0.3805:
                break
    return (best_h, best_c5, n_points)
