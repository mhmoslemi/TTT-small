import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()

    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)

    best_c5 = result.fun

    # Apply structured perturbations with sinusoidal and block swapping
    for _ in range(200):
        if time.time() - start_time > budget_s:
