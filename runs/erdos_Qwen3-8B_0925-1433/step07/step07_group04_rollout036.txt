import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population_size = 20

    # Initialize population
    population = []
    initial_h = initial_h_values.copy()
    initial_h = initial_h * target_sum / np.sum(initial_h)
    population.append(initial_h)
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = initial_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        population.append(new_h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(h, dx, n_points) for h in population]
        elites = np.argsort(fitness)[:int(len(population)*0.1)]
        # Generate offspring
        offspring = []
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            alpha = np.random.rand()
            child = p1 * alpha + p2 * (1 - alpha)
            mutate_mask = np.random.rand(n_points) < 0.1
            child[mutate_mask] += np.random.normal(0, 0.01, np.sum(mutate_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Combine and sort
        combined = population + offspring
        combined_fitness = [compute_c5(h, dx, n_points) for h in combined]
        combined.sort(key=lambda x: compute_c5(x, dx, n_points))
        population = combined[:population_size]
        # Update best
        current_best = population[0]
        current_c5 = compute_c5(current_best, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
