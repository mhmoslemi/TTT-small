import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    x = np.linspace(0, 2, n_points)
    freq1 = 2 * np.pi * (n_points / 4)
    freq2 = 2 * np.pi * (n_points / 8)
    h = initial_h_values * (1 + 0.2 * np.sin(freq1 * x) + 0.1 * np.sin(freq2 * x + np.pi / 4))
    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr * dx)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Structured perturbations
        freqs = [1, 2, 3, 4, 5]
        for k in freqs:
            perturb = 0.05 * np.sin(2 * np.pi * k * x / n_points)
            new_h = best_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
