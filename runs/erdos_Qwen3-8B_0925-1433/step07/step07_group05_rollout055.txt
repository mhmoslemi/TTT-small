    best_h = current_h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iterations = 1000
    eps = 0.05
    improvement_count = 0
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr)
        max_contrib = -1
        max_index = 0
        for i in range(n_points - k):
            contrib = current_h[i] * (1 - current_h[i + k])
            if contrib > max_contrib:
                max_contrib = contrib
                max_index = i
        i = max_index
        j = i + k
        temp_h = current_h.copy()
        if current_h[i] > 0.5 and current_h[j] > 0.5:
            temp_h[i] = max(0.0, current_h[i] - eps)
            temp_h[j] = max(0.0, current_h[j] - eps)
        elif current_h[i] > current_h[j]:
            temp_h[i] = max(0.0, current_h[i] - eps)
        else:
            temp_h[j] = max(0.0, current_h[j] - eps)
        temp_h = np.clip(temp_h, 0, 1)
        sum_temp = np.sum(temp_h)
        if sum_temp == 0:
            continue
        temp_h *= target_sum / sum_temp
        new_c5 = compute_c5(temp_h, dx, n_points)
        if new_c5 < best_c5:
            current_h = temp_h
            best_c5 = new_c5
            improvement_count = 0
        else:
            improvement_count += 1
            if improvement_count > 100:
                perturb = np.random.normal(0, 0.05, n_points)
                temp_h = current_h + perturb
                temp_h = np.clip(temp_h, 0, 1)
                sum_temp = np.sum(temp_h)
                if sum_temp > 0:
                    temp_h *= target_sum / sum_temp
                new_c5 = compute_c5(temp_h, dx, n_points)
                if new_c5 < best_c5:
                    current_h = temp_h
                    best_c5 = new_c5
                improvement_count = 0
        if best_c5 < compute_c5(best_h, dx, n_points):
            best_h = current_h.copy()
    return (best_h, best_c5, n_points)
