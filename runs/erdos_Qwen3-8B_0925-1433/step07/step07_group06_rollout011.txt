import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    learning_rate = 0.1
    decay_rate = 0.995
    max_iterations = 500
    perturbation_prob = 0.2
    perturb_scale = 0.05
    early_stop = 0
    max_early_stop = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        k = max_corr_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] = 1 - h[i + k]
            else:
                grad[i] = 0
        
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        if sum_h_new == 0:
            h_new = np.zeros_like(h_new)
        else:
            h_new = h_new * target_sum / sum_h_new
        
        new_c5 = compute_c5(h_new, 2.0 / n_points, n_points)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            early_stop = 0
        else:
            if np.random.random() < perturbation_prob:
                perturb = np.random.normal(0, perturb_scale, n_points)
                h_perturb = h + perturb
                h_perturb = np.clip(h_perturb, 0, 1)
                sum_perturb = np.sum(h_perturb)
                h_perturb = h_perturb * target_sum / sum_perturb
                perturb_c5 = compute_c5(h_perturb, 2.0 / n_points, n_points)
                if perturb_c5 < best_c5:
                    h = h_perturb.copy()
                    best_c5 = perturb_c5
                    early_stop = 0
            else:
                early_stop += 1
                if early_stop > max_early_stop:
                    break
        
        learning_rate *= decay_rate
    
    return (h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
