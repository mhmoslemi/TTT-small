def run(seed=42, budget_s=600):

    start_time = time.time()
    
    T = 1.0
    cooling = 0.997
    perturb_step = 0.02
    no_improvement = 0
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-5:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        grad = np.zeros(n_points)

            for i in range(n_points - k):
                grad[i] += (1 - h[i + k]) * dx
                grad[i + k] += -h[i] * dx

            for i in range(m, n_points):
                grad[i] += (1 - h[i - m]) * dx
                grad[i - m] += -h[i] * dx
