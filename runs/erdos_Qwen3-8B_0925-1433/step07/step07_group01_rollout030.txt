def run(seed=42, budget_s=600):

    # Initialize h
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points) + 0.5  # fallback for all zeros

    # Objective function for DE
    def obj(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    # Run differential evolution
    try:
        result = differential_evolution(
            obj,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=30,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True,
            seed=seed,
            disp=False
        )
    except:
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, tol=1e-8)

    best_h = result.x
    best_c5 = result.fun

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        constraints=cons,
        tol=1e-8,

    )
    best_h, best_c5 = result_slsqp.x, result_slsqp.fun

    # Final check for time
    if time.time() - start_time > budget_s:
        # Early exit
        pass  # best_h and best_c5 already set

    return (best_h, best_c5, n_points)
