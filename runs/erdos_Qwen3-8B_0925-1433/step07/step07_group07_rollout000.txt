def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    T = 1.0
    cooling = 0.995
    n_perturb = 15
    perturb_std = 0.03
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        
        valid_indices = []
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                valid_indices.append(i)
        
        if not valid_indices:
            continue
        
        contributions = [h[i] * (1 - h[i + lag]) for i in valid_indices]
        sorted_contrib_indices = np.argsort(contributions)[::-1]
        top_indices = sorted_contrib_indices[:n_perturb]
        selected_indices = [valid_indices[i] for i in top_indices]
        
        h_perturb = h.copy()
        for idx in selected_indices:
            h_perturb[idx] += np.random.normal(0, perturb_std)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        
        curr_c5 = compute_c5(h_perturb, dx, n_points)
        delta = curr_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_perturb.copy()
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        T *= cooling
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
