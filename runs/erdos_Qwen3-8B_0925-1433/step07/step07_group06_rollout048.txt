import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    for _ in range(100):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        # SLSQP optimization
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, 2.0 / n_points, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        # Perturbation
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res_p = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res_p.success:
            curr_h_p = res_p.x
            curr_c5_p = compute_c5(curr_h_p, 2.0 / n_points, n_points)
            if curr_c5_p < best_c5:
                best_c5 = curr_c5_p
                best_h = curr_h_p.copy()
        # Prune
        prune_mask = (best_h > 0.4) & (best_h < 0.6)
        pruned_h = best_h.copy()
        pruned_h[prune_mask] = np.random.choice([0, 1], size=sum(prune_mask))
        pruned_sum = np.sum(pruned_h)
        if pruned_sum > 0:
            pruned_h *= (target_sum / pruned_sum)
        curr_c5_p = compute_c5(pruned_h, 2.0 / n_points, n_points)
        if curr_c5_p < best_c5:
            best_c5 = curr_c5_p
            best_h = pruned_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
