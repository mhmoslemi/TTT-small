import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    x = np.linspace(0, 2, n_points)
    amps = [0.2, 0.1, 0.05]
    freqs = [2 * np.pi * (n_points / 4), 2 * np.pi * (n_points / 6), 2 * np.pi * (n_points / 8)]
    h = initial_h_values * (1 + sum(amp * np.sin(freq * x) for amp, freq in zip(amps, freqs)))
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def perturb_h(h):
        length = int(n_points * 0.2)
        start1 = np.random.randint(0, n_points - length + 1)
        start2 = np.random.randint(0, n_points - length + 1)
        while abs(start1 - start2) < length:
            start2 = np.random.randint(0, n_points - length + 1)
        h_swapped = h.copy()
        h_swapped[start1:start1+length], h_swapped[start2:start2+length] = h_swapped[start2:start2+length], h_swapped[start1:start1+length]
        return h_swapped
    
    start_time = time.time()
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        new_h = perturb_h(best_h)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
