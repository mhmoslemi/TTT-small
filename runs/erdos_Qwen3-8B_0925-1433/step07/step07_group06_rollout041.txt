            curr_c5 = compute_c5(curr_h, 2.0 / n_points, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        # Phase 2: Perturb and optimize
        perturbed = best_h + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res_p = minimize(obj, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if res_p.success:
            curr_h_p = res_p.x
            curr_c5_p = compute_c5(curr_h_p, 2.0 / n_points, n_points)
            if curr_c5_p < best_c5:
                best_c5 = curr_c5_p
                best_h = curr_h_p.copy()
        # Phase 3: Prune values close to 0.5
        pruned_h = best_h.copy()
        for i in range(n_points):
            if pruned_h[i] < 0.4:
                pruned_h[i] = 0
            elif pruned_h[i] > 0.6:
                pruned_h[i] = 1
        pruned_h = pruned_h * target_sum / np.sum(pruned_h)
        res_prune = minimize(obj, pruned_h, method='SLSQP', bounds=bounds, constraints=cons)
        if res_prune.success:
            curr_h_prune = res_prune.x
            curr_c5_prune = compute_c5(curr_h_prune, 2.0 / n_points, n_points)
            if curr_c5_prune < best_c5:
                best_c5 = curr_c5_prune
                best_h = curr_h_prune.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
