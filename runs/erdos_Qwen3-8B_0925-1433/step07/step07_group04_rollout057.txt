                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-7, 'maxiter': 100}
            )
            if res_perturb.success:
                curr_h_perturb = res_perturb.x
                curr_c5_perturb = compute_c5(curr_h_perturb, dx, n_points)
                if curr_c5_perturb < best_c5:
                    best_c5 = curr_c5_perturb
                    best_h = curr_h_perturb.copy()

    # Final check
    final_c5 = compute_c5(best_h, dx, n_points)
    return (best_h, final_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
