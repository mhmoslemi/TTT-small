                best_c5 = new_c5
                best_h = new_h.copy()
        # Perturb and explore
        perturbed = best_h + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = np.clip(perturbed * target_sum / np.sum(perturbed), 0, 1)
        res_perturb = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                               perturbed, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'ftol': 1e-6, 'maxiter': 50})
        if res_perturb.success:
            curr_c5 = compute_c5(res_perturb.x, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = res_perturb.x.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
