from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize h
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.inf

    # SA parameters
    temp = 1.0
    cooling = 0.995
    iter_sa = 600
    sa_time = 0
    start_time = time.time()

    for _ in range(iter_sa):
        if time.time() - start_time > budget_s:
            break
        # Perturb
        perturb = np.zeros(n_points)
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.05, 5)
        perturb[indices] = noise
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_h = h.copy()
        else:
            new_h = new_h * target_sum / new_sum
        # Compute C5
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        current_c5 = np.max(corr) * dx
        # Accept with probability
        if current_c5 < best_c5:
            best_h, best_c5 = new_h.copy(), current_c5
        else:
