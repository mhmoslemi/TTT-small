import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 50:

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - n_points + 1

        products = best_h[valid_indices] * (1 - best_h[valid_indices + k])
