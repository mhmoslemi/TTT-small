        res = minimize(objective, best_h, method='L-BFGS-B', jac=grad_func, bounds=bounds, constraints=cons)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, 2.0 / n_points, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        
        # Perturbation with cooling schedule
        temp = max(0.1, 0.8 ** (_ / 10))
        perturbed = best_h + np.random.normal(0, 0.05 * temp, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res_p = minimize(objective, perturbed, method='L-BFGS-B', jac=grad_func, bounds=bounds, constraints=cons)
        if res_p.success:
            curr_h_p = res_p.x
            curr_c5_p = compute_c5(curr_h_p, 2.0 / n_points, n_points)
            if curr_c5_p < best_c5:
                best_c5 = curr_c5_p
                best_h = curr_h_p.copy()
    
    return (best_h, best_c5, n_points)

def rescale(h):
    total = np.sum(h)
    if total == 0:
        return np.zeros_like(h)
    return h * (len(h) / 2) / total

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
