import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        A = 0.05 * np.random.rand()
        freq = np.random.randint(1, 7)
        perturb = A * np.sin(freq * np.arange(n_points))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        try:
            res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        except:
            pass

    return (best_h, best_c5, n_points)
