        if res_p.success:
            curr_h_p = res_p.x
            curr_c5_p = compute_c5(curr_h_p, 2.0 / n_points, n_points)
            if curr_c5_p < best_c5:
                best_c5 = curr_c5_p
                best_h = curr_h_p.copy()
    
    # Pruning phase
    prune_threshold = 0.4
    for i in range(n_points):
        if best_h[i] > prune_threshold:
            best_h[i] = 1.0
        elif best_h[i] < 1 - prune_threshold:
            best_h[i] = 0.0
    best_h = best_h * target_sum / np.sum(best_h)
    curr_c5_pruned = compute_c5(best_h, 2.0 / n_points, n_points)
    if curr_c5_pruned < best_c5:
        best_c5 = curr_c5_pruned
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
