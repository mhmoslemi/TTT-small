import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    # Define cost function
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    
    # Genetic algorithm
    population = [copy.deepcopy(h) + np.random.normal(0, 0.05, n_points) for _ in range(50)]
    for p in population:
        p = np.clip(p, 0, 1)
        p = p * target_sum / np.sum(p)
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(p, 2.0 / n_points, n_points) for p in population]
        sorted_idx = np.argsort(fitness)
        parents = [population[i] for i in sorted_idx[:20]]
        offspring = []
        for _ in range(len(population) - len(parents)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            cross = np.random.randint(1, n_points)
            child = np.concatenate((p1[:cross], p2[cross:]))
            mutate = np.random.normal(0, 0.03, n_points)
            child += mutate
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        new_pop = parents + offspring
        new_pop.sort(key=lambda x: compute_c5(x, 2.0 / n_points, n_points))
        population = new_pop[:50]
        current_best = population[0]
        current_c5 = compute_c5(current_best, 2.0 / n_points, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    # Local refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        top_indices = np.argsort([compute_c5(p, 2.0 / n_points, n_points) for p in population])[:5]
        for idx in top_indices:
            h = population[idx].copy()
            res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                h = res.x
                corr = np.correlate(h, 1.0 - h, mode='full')
                current_c5 = np.max(corr) * (2.0 / n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
