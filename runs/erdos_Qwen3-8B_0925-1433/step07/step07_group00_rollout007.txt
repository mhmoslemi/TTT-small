import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with given or sinusoidal pattern
    h = np.clip(initial_h_values, 0, 1).copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(len(x) - lag):
            grad[i] += 1 - x[i + lag]
            grad[i + lag] -= x[i]
        return max_corr * dx, grad

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, jac=True, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        lag = np.argmax(corr)
        delta = 0.015 * np.sin(2 * np.pi * lag * np.arange(n_points) / n_points)
        new_h = np.clip(best_h + delta, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, jac=True, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
