import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    h = initial_h_values * target_sum / initial_sum
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    iterations = 0
    max_iterations = 500
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        m = max_corr_idx - (n_points - 1)
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + m) % n_points
            A[i, j] = 1
        
        h_var = cp.Variable(n_points)
        objective = cp.Maximize(cp.quad_form(h_var, A))
        constraints = [cp.sum(h_var) == target_sum, 0 <= h_var, h_var <= 1]
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'OPTIMAL':
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                new_sum = np.sum(h_new)
                if new_sum == 0:
                    h_new = h.copy()
                else:
                    h_new = h_new * target_sum / new_sum
                new_c5 = compute_c5(h_new, dx, n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                h = h_new
        except:
            pass
        
        iterations += 1
    
    return (best_h, best_c5, n_points)
