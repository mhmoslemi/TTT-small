def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    MAX_ITERATIONS = 1000
    step_size = 0.01
    for _ in range(MAX_ITERATIONS):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        if lag >= 0:
            m = lag
            for i in range(n_points):
                if i + m < n_points:
                    term1 = (1 - h[i + m]) * dx
                else:
                    term1 = 0.0
                if i - m >= 0:
                    term2 = h[i - m] * dx
                else:
                    term2 = 0.0
                grad[i] = term1 - term2
        else:
            m = -lag
            for i in range(n_points):
                if i - m >= 0:
                    term1 = (1 - h[i - m]) * dx
                else:
                    term1 = 0.0
                if i + m < n_points:
                    term2 = h[i + m] * dx
                else:
                    term2 = 0.0
                grad[i] = term1 - term2
        
        abs_grad = np.abs(grad)
        indices = np.argsort(-abs_grad)
        selected_indices = indices[:10]
        
        h_perturb = h.copy()
        for i in selected_indices:
            h_perturb[i] = np.clip(h[i] - step_size * grad[i], 0, 1)
        
        new_sum = np.sum(h_perturb)
        if new_sum == 0:
            h_perturb = h.copy()
        else:
            h_perturb = h_perturb * (target_sum) / new_sum
        
        new_c5 = compute_c5(h_perturb, dx, n_points)
        if new_c5 < best_c5:

            best_h = h_perturb.copy()
            step_size *= 1.05
        else:
            step_size *= 0.95
    
    return (best_h, best_c5, n_points)
