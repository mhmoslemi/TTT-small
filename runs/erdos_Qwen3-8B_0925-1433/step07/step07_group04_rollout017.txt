from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize the initial guess
    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    current_sum = np.sum(h)
    scaling = target_sum / current_sum
    h = np.clip(h * scaling, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for _ in range(300):  # 300 iterations for more exploration
        if time.time() - start_time > budget_s:
            break
        # Compute correlation to find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        # Generate targeted perturbation
        delta = 0.02
        perturbed = h.copy()
        if max_index + 1 < n_points:
            perturbed[max_index] -= delta
            perturbed[max_index + 1] += delta
        else:
            perturbed[max_index] -= delta
