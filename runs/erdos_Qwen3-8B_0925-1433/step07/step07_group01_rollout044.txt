import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros_like(h)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def evaluate(h):
        return obj(repair(h))
    
    def ga_step(pop):
        new_pop = []
        for _ in range(len(pop)):
            a, b = np.random.choice(pop, 2, replace=False)
            idx = np.random.choice(n_points)
            child = a.copy()
            child[idx] = b[idx]
            child = repair(child)
            new_pop.append(child)
        return new_pop
    
    if 'initial_h_values' in globals():
        h = repair(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = repair(h)
    
    pop_size = 30
    pop = [repair(np.random.rand(n_points)) for _ in range(pop_size)]
    best_h, best_c5 = h, evaluate(h)
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = ga_step(pop)
        scores = np.array([evaluate(p) for p in new_pop])
        top_idx = np.argmin(scores)
        new_h = new_pop[top_idx]
        if evaluate(new_h) < best_c5:
            best_h, best_c5 = new_h, evaluate(new_h)
        pop = new_pop
    
    res = minimize(obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   tol=1e-8)
    best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
