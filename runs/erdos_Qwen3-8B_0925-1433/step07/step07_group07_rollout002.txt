    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    T = 1.0
    cooling = 0.997
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        valid_indices = [i for i in range(n_points) if 0 <= i + max_lag < n_points]
        perturb_indices = set()
        for i in valid_indices:
            perturb_indices.add(i)
            perturb_indices.add(i + max_lag)
        gradients = np.zeros(n_points)
        for i in valid_indices:
            gradients[i] = (1 - h[i + max_lag]) * dx
            gradients[i + max_lag] -= h[i] * dx
        sorted_indices = np.argsort(np.abs(gradients))[::-1]
        top_indices = sorted_indices[:10]
        perturb = np.zeros(n_points)
        perturb[top_indices] = np.random.normal(0, 0.02, len(top_indices))
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        new_c5 = compute_c5(h_perturb, dx, n_points)
        if new_c5 < best_c5:
            h = h_perturb.copy()
            best_c5 = new_c5
            best_h = h.copy()
        elif np.random.rand() < np.exp(-(new_c5 - best_c5)/T):
            h = h_perturb.copy()
        T *= cooling
    return (best_h, best_c5, n_points)
