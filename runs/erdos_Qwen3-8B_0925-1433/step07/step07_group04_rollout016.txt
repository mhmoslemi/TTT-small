import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize and scale
    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    h *= target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for _ in range(300):  # 300 iterations for thorough exploration
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        curr_c5 = corr[max_k] * dx

        # Directed perturbation targeting worst lag
        perturb = np.zeros(n_points)
        for i in range(n_points):
            if i + max_k < n_points:
                perturb[i] -= 0.02 * best_h[i]
                perturb[i + max_k] += 0.02 * best_h[i]
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 50}
        )

        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()

        # Local optimization on best_h
        res_best = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 50}
        )
        if res_best.success:
            curr_h_best = res_best.x
            curr_c5_best = compute_c5(curr_h_best, dx, n_points)
            if curr_c5_best < best_c5:
                best_c5 = curr_c5_best
                best_h = curr_h_best.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
