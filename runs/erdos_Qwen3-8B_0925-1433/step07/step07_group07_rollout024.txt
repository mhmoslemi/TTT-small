import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    T = 1.0
    cooling = 0.995
    N_perturb = 15
    max_iter = 1000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        m = np.argmax(corr)
        k = m - (n_points - 1)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                gradient[i] = (1 - h[j]) * dx
        
        top_indices = np.argsort(gradient)[-N_perturb:]
        h_perturb = h.copy()
        for idx in top_indices:
            delta = np.random.normal(0, 0.03)
            h_perturb[idx] += delta
            h_perturb[idx] = np.clip(h_perturb[idx], 0, 1)
        
        sum_perturb = np.sum(h_perturb)
        if sum_perturb == 0:
            h_perturb = h.copy()
        else:
            h_perturb = h_perturb * target_sum / sum_perturb
        
        curr_c5 = compute_c5(h_perturb, dx, n_points)
        if curr_c5 < best_c5 or np.random.rand() < np.exp(-(curr_c5 - best_c5)/T):
            h = h_perturb.copy()
            best_c5 = curr_c5
            best_h = h.copy()
        T *= cooling
    
    return (best_h, best_c5, n_points)
