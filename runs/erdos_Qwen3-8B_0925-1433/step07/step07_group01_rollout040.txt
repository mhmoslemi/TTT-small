        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = current_h.copy()
        obj_new = compute_obj(new_h)

        # Acceptance criteria
        if obj_new < current_obj:
            current_h = new_h.copy()
            current_obj = obj_new
        elif np.random.rand() < np.exp((current_obj - obj_new) / T):
            current_h = new_h.copy()
            current_obj = obj_new

        # Update best solution
        if obj_new < best_c5:
            best_h = new_h.copy()
            best_c5 = obj_new
        T *= cooling_rate

    # Refinement with SLSQP
    result = minimize(compute_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    best_h = result.x
    best_c5 = result.fun

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    else:
        best_h = np.zeros(n_points)
    best_c5 = compute_obj(best_h)

    return (best_h, best_c5, n_points)
