        res = minimize(obj, best_h, method='L-BFGS-B', bounds=Bounds(0, 1),
                       constraints=LinearConstraint(np.ones(n_points), [target_sum], [target_sum]),
                       options={'ftol': 1e-6, 'maxiter': 50})
        if res.success:
            curr_h = res.x
            curr_h = np.clip(curr_h, 0, 1)
            curr_sum = np.sum(curr_h)
            curr_h *= target_sum / curr_sum
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        # Perturb to escape local minima
        perturb = np.random.normal(0, 0.03, n_points)
        curr_h = best_h + perturb
        curr_h = np.clip(curr_h, 0, 1)
        curr_sum = np.sum(curr_h)
        curr_h *= target_sum / curr_sum
        curr_c5 = compute_c5(curr_h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
