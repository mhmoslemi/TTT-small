import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h with structured pattern
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    # Initial optimization with trust-constr
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    # Refinement loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Apply sinusoidal perturbation
        freq = int(np.random.choice([1, 2, 3, 4, 5]))
        A = np.random.uniform(0.01, 0.05)
        perturb = A * np.sin(2 * np.pi * freq * np.arange(n_points) * dx)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        # Optimize with trust-constr
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        # Add random swapping for diversity
        if np.random.rand() < 0.1:
            idx = np.random.choice(n_points, size=2, replace=False)
            best_h[idx[0]], best_h[idx[1]] = best_h[idx[1]], best_h[idx[0]]
            # Clip and rescale
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h

    return (best_h, best_c5, n_points)
