from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    def de_obj(x):
        h_candidate = np.clip(x, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    try:
        result = differential_evolution(
            de_obj,
            bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-8,
            maxiter=300,
            mutation=(0.5, 1.0),
            recombination=0.9,
            disp=False
        )
    except:
        result = None

    best_h = h
    best_c5 = np.inf
    if result is not None:
        best_h = result.x
        best_c5 = result.fun

    def obj(x):

        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_sqp = [(0.0, 1.0)] * n_points

                method='SLSQP',
                bounds=bounds_sqp,

                tol=1e-8
