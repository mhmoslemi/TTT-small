import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize with sinusoidal modulation
    h = initial_h_values.copy()
    wave = np.sin(2 * np.pi * (n_points // 4) * np.linspace(0, 2, n_points))
    h = h * (1 + 0.2 * wave)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def obj(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def repair(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        return x * target_sum / s
    
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Add structured frequency-based perturbations
        k = np.random.randint(1, 6)
        perturb = 0.05 * np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
        new_h = repair(best_h + perturb)
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
