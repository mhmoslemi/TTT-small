import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize initial guess
    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    current_sum = np.sum(h)
    scaling = target_sum / current_sum
    h = np.clip(h * scaling, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break

        # Find worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)

        # Define objective function to minimize the worst lag
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[max_index] * dx

        # Perturb current best
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        chebyshev = np.sum(perturbed) * dx
        scaling = target_sum / chebyshev if chebyshev != 0 else 1.0
        perturbed = np.clip(perturbed * scaling, 0, 1)

        # Constrained optimization
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        res = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )

        if res.success:
            curr_h = res.x
            curr_corr = np.correlate(curr_h, 1 - curr_h, mode='full')
            new_c5 = np.max(curr_corr) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr_h.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
