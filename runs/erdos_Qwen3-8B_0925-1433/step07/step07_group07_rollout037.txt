import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    alpha = 0.02
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c5_values = corr * dx
        k_max = np.argmax(c5_values)
        grad = np.zeros_like(h)
        for i in range(n_points - k_max):
            grad[i] += dx * (1 - h[i + k_max])
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        curr_c5 = compute_c5(h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
