import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    import numpy as np
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def mutate(ind):
        noise = np.random.normal(0, 0.05, len(ind))
        new_ind = np.clip(ind + noise, 0, 1)
        new_sum = np.sum(new_ind)
        return new_ind * (target_sum / new_sum) if new_sum else new_ind

    def crossover(p1, p2):
        child = np.random.rand(len(p1))
        child = np.where(child < 0.5, p1, p2)
        child_sum = np.sum(child)
        return child * (target_sum / child_sum) if child_sum else child

    population = [h]
    for _ in range(29):
        new_h = np.random.rand(n_points)
        new_h = new_h * (target_sum / np.sum(new_h)) if np.sum(new_h) > 0 else new_h
        new_h = np.clip(new_h, 0, 1)
        population.append(new_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        tournament = np.random.choice(population, size=5, replace=False)
        best1 = tournament[np.argmin([obj(ind) for ind in tournament])]
        tournament = np.random.choice(population, size=5, replace=False)
        best2 = tournament[np.argmin([obj(ind) for ind in tournament])]
        child1 = mutate(best1)
        child2 = mutate(best2)
        child = crossover(child1, child2)
        fitness = [obj(ind) for ind in population]
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child

    best_h = min(population, key=obj)
    best_c5 = obj(best_h)

    def refine(x):
        return obj(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(refine, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-10)
    best_h = res.x
    best_c5 = res.fun

    return (best_h, best_c5, n_points)
