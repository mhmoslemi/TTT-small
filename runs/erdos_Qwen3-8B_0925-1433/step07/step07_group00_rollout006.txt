import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Sine-based initialization with frequency modulation
    f = 2 * np.pi * n_points / 3.5
    mod = 1 + 0.15 * np.sin(f * np.linspace(0, 2, n_points))
    h = initial_h_values * mod
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    # Perturbation refinement
    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        norm = np.sum(new_h) or 1
        new_h = new_h * target_sum / norm
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
