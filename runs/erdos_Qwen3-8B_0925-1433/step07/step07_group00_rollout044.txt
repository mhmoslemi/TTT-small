import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guess with structured binary pattern
    m = int(np.sqrt(n_points))
    interval_length = 2.0 / m
    x = np.linspace(0, 2, n_points)
    binary = np.zeros_like(x)
    for i in range(m):
        if i % 2 == 1:
            start = i * interval_length
            end = (i + 1) * interval_length
            idx = np.where((x >= start) & (x < end))[0]
            binary[idx] = 1
    # Rescale to target_sum
    binary = binary * target_sum / np.sum(binary) if np.sum(binary) > 0 else np.zeros(n_points)
    h = np.clip(binary, 0, 1)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial optimization
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    # Refinement loop with enhanced perturbations
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Random swap of non-overlapping segments
        seg_len = int(n_points * 0.2)
        swaps = np.random.permutation(n_points // seg_len)
        for i in range(len(swaps) // 2):
            a, b = swaps[i], swaps[-1 - i]
            best_h[a*seg_len:(a+1)*seg_len], best_h[b*seg_len:(b+1)*seg_len] = \
                best_h[b*seg_len:(b+1)*seg_len], best_h[a*seg_len:(a+1)*seg_len]
        # Apply sinusoidal perturbation
        freqs = np.random.choice([0.1, 0.3, 0.5, 0.7, 0.9], size=n_points)
        noise = 0.05 * np.sin(2 * np.pi * freqs * np.arange(n_points))
        new_h = np.clip(best_h + noise, 0.0, 1.0)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    return (best_h, best_c5, n_points)
