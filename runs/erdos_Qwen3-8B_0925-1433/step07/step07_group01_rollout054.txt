import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize from provided data
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    # Objective with constraint penalty
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_overlap = np.max(corr)
        penalty = 1e3 * abs(np.sum(x) - target_sum)
        return max_overlap * dx + penalty

    bounds = [(0, 1) for _ in range(n_points)]

    # Differential evolution with adaptive settings
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=30,
        maxiter=200, mutation=(0.5, 1), recombination=0.8,
        tol=1e-6, seed=seed, disp=False
    )

    best_h = result.x
    best_c5 = result.fun

    # SLSQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_refine = minimize(
        objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 100}
    )
    best_h, best_c5 = result_refine.x, result_refine.fun

    # Final verification
    if np.isnan(best_c5) or np.isinf(best_c5):
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
