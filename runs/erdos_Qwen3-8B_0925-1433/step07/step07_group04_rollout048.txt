import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize the initial guess
    h = initial_h_values.copy()

        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    current_sum = np.sum(h)
    scaling = target_sum / current_sum
    h = np.clip(h * scaling, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

        if time.time() - start_time > budget_s:
            break
        # Compute cross-correlation and find the max lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        lag_index = np.argmax(corr)
        current_max = corr[lag_index] * dx

        # Define objective function to minimize the specific lag
        def objective(x):
            return np.correlate(x, 1 - x, mode='full')[lag_index] * dx

        # Perturb best_h
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = np.clip(perturbed * target_sum / np.sum(perturbed), 0, 1)

        # Run optimization
        res = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=Bounds(0, 1),

        # Occasionally optimize the entire function to prevent local minima

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
