import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    for _ in range(150):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        # SLSQP optimization with numerical gradient
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_points)
        def jac(x):
            eps = 1e-5
            grad = np.zeros_like(x)
            for i in range(len(x)):
                x_pert = x.copy()
                x_pert[i] += eps
                corr_pert = np.correlate(x_pert, 1 - x_pert, mode='full')
                grad[i] = (np.max(corr_pert) - np.max(np.correlate(x, 1 - x, mode='full'))) / eps
            return grad
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=jac)
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, 2.0 / n_points, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        # Perturbation
        perturbed = best_h + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res_p = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons, jac=jac)
        if res_p.success:

            curr_c5_p = compute_c5(curr_h_p, 2.0 / n_points, n_points)

                best_h = curr_h_p.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
