import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, initial_h_values=None):
    global initial_h_values
    if initial_h_values is None:
        initial_h_values = np.zeros(500)
        for i in range(250):
            initial_h_values[2*i] = 1.0
        initial_h_values = np.clip(initial_h_values, 0, 1)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    best_h, best_c5 = h.copy(), obj(h)

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        h = res.x
        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        # Structured perturbations

        h = np.clip(h + perturb, 0, 1)
