            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_max] * dx
        
        constraints = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        
        if res.success:
            new_h = res.x
            corr_new = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
                if best_c5 < 0.38:
                    mutation_amp *= 0.975
                elif best_c5 > 0.385:
                    mutation_amp *= 1.025
        else:
            new_h = perturbed.copy()
            corr_new = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
        
        iterations += 1
    
    return (best_h, best_c5, n_points)
