import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

        perturbed = np.clip(perturbed, 0, 1)

        if s == 0:
            perturbed = np.zeros_like(perturbed)
        else:
            perturbed = perturbed * (target_sum / s)
        population.append(perturbed)
    
    best_h = population[0]
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all in population
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx, n_points)
            fitnesses.append(-c5)
        best_idx = np.argmax(fitnesses)
        best_c5 = -fitnesses[best_idx]
        best_h = population[best_idx]
        
        # Select elites (top 10%)
        elite_size = int(len(population) * 0.1)
        elites = [population[i] for i in np.argsort(fitnesses)[-elite_size:]]
        
        # Generate offspring
        offspring = []
        for _ in range(len(population) - elite_size):
            # Crossover
            p1, p2 = np.random.choice(population, 2, replace=False)
            if np.random.rand() < 0.8:
                alpha = np.random.rand(n_points)
                child = p1 * alpha + p2 * (1 - alpha)
            else:
                child = p1.copy()

            if np.random.rand() < 0.1:

            if s == 0:

            else:

            offspring.append(child)
        
        # Replace population
        population = elites + offspring
    
    # Final check and return best solution
    best_c5 = compute_c5(best_h, dx, n_points)
