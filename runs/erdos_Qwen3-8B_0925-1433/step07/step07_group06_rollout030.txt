import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    # Simulated Annealing
    T = 1.0
    cooling_rate = 0.995
    max_temp = 100
    max_iter_per_temp = 1000
    current_h = h.copy()
    
    for _ in range(max_temp):
        if time.time() - start_time > budget_s:
            break
        for _ in range(max_iter_per_temp):
            noise = np.random.normal(0, T * 0.05, n_points)
            perturbed = current_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            new_c5 = compute_c5(perturbed, 2.0 / n_points, n_points)
            if new_c5 < best_c5:
                current_h = perturbed.copy()
                best_c5 = new_c5
                best_h = current_h.copy()
            else:
                if np.random.rand() < np.exp((best_c5 - new_c5) / T):
                    current_h = perturbed.copy()
        T *= cooling_rate
    
    # Local Refinement
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons, options={'maxiter': 100})
    if res.success:
        best_h = res.x
        best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    return (best_h, final_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
