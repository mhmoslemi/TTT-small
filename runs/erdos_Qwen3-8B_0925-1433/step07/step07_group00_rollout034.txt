import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Sinusoidal modulation of initial guess if available
    if initial_h_values is not None:
        h = initial_h_values.copy()
        freq = 2 * np.pi * (n_points / 4)
        h = h * (1 + 0.2 * np.sin(freq * np.linspace(0, 2, n_points)))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    else:
        # Structured initial guess with 2*sqrt(n_points) intervals
        m = int(2 * np.sqrt(n_points))
        h = np.zeros(n_points)
        for i in range(m):
            idx = np.linspace(i / m, (i + 1) / m, n_points, endpoint=False)
            h[idx.astype(int)] = 1
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    # Perturbations with sinusoidal variation
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Add structured sinusoidal perturbations
        perturb = 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
