import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial guess (clipped and normalized)
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Use trust-constr with Jacobian for better convergence
    def jac(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            for j in range(n_points):
                if idx == j:
                    grad[i] += -((1 - x[i]) if i == j else -(x[j] * (1 - x[i]))) * dx
        return grad

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, jac=jac, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    # Apply structured perturbations and refine
    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        # Add sinusoidal perturbation with varying frequency
        amplitude = 0.05
        k_values = [1, 2, 3, 4, 5]
        best_h_perturbed = best_h.copy()
        for k in k_values:
            perturbation = amplitude * np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
            best_h_perturbed += perturbation
        best_h_perturbed = np.clip(best_h_perturbed, 0, 1)
