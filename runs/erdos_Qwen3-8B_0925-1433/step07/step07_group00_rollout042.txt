from scipy.optimize import minimize
import time

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Optimization with trust-constr
    res = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = res.x, res.fun

    start_time = time.time()
