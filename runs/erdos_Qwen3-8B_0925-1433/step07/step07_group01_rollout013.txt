import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    def create_individual():
        if np.sum(initial_h_values) > 0:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    pop_size = 30
    pop = [create_individual() for _ in range(pop_size)]
    best_h = pop[0]
    best_c5 = float('inf')

    def evaluate(ind):
        corr = np.correlate(ind, 1 - ind, mode='full')
        return np.max(corr) * dx

    def mutate(ind):
        mut = ind.copy()
        idx = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        mut[idx] += np.random.normal(0, 0.05, len(idx))
        mut = np.clip(mut, 0, 1)
        mut = mut * target_sum / np.sum(mut)
        return mut

    def crossover(a, b):
        child = a.copy()
        mask = np.random.rand(n_points) < 0.5
        child[mask] = b[mask]
        return child

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        tournament = np.random.choice(pop, size=5)
        parents = [tournament[np.argmin([evaluate(p) for p in tournament])]]
        # Mutate
        new_pop = []
        for p in parents:
            child = mutate(p)
            new_pop.append(child)
        # Crossover
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(pop, size=2, replace=False)
            child = crossover(p1, p2)
            new_pop.append(child)
        pop = new_pop
        # Update best
        for ind in pop:
            curr_c5 = evaluate(ind)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = ind

    # Local refinement
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
