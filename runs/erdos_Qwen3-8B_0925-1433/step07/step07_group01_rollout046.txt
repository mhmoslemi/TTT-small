import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        if np.sum(h_clipped) == 0:
            return float('inf')
        h_rescaled = h_clipped * (target_sum / np.sum(h_clipped))
        corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    # Differential Evolution for global search
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20,
        tol=1e-6, mutation=(0.5, 1), recombination=0.8,
        maxiter=100, disp=False, polish=False
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    # SLSQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_slsqp = [(0, 1)] * n_points

    result_slsqp = minimize(
        objective, best_h, method='SLSQP', bounds=bounds_slsqp,
        constraints=cons, tol=1e-8, options={'maxiter': 100, 'ftol': 1e-9}
    )
    if result_slsqp.fun < best_c5:
        best_h, best_c5 = result_slsqp.x, result_slsqp.fun

    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(final_corr) * dx
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, c5_bound, n_points)
    return (best_h, c5_bound, n_points)
