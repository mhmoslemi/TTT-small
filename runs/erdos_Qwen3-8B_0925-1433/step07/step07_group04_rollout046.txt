import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize and scale the initial guess
    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]

        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)

        # Define objective to minimize specific lag overlap
        def obj_func(x):
            corr_new = np.correlate(x, 1 - x, mode='full')
            return corr_new[k] * dx

        # Optimization constraints and bounds
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)

        # Perform SLSQP optimization
        res = minimize(
            obj_func,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 100}
        )

        if res.success:

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
