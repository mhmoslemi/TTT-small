import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h with a structured pattern to balance the integral
    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / np.sum(h)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial optimization with trust-constr
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    # Refinement with targeted perturbations
    for _ in range(200):
        if time.time() - time.time() > budget_s:
            break
        # Compute correlation and identify max lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = (max_idx - (n_points - 1)) // 2  # Adjust for full mode
    
        # Apply localized perturbation to reduce overlap
        w = np.hanning(25)  # Windowed perturbation
        perturb = 0.02 * w
        if k + 25 < n_points:
            best_h[k:k+25] += perturb
        if k - 25 >= 0:
            best_h[k-25:k] += perturb
        best_h = np.clip(best_h, 0, 1)
        # Rescale to maintain integral constraint
        if np.sum(best_h) > 0:
            best_h = best_h * target_sum / np.sum(best_h)
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
