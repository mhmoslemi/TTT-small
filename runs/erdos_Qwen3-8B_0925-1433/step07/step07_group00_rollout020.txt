import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    # Enhanced perturbation strategies
    for _ in range(120):
        if time.time() - start_time > budget_s:
            break
        # Structured swap perturbations
        if np.random.rand() < 0.4:
            idx1 = np.random.choice(n_points, 2, replace=False)
            idx2 = np.random.choice(n_points, 2, replace=False)
            swap_len = min(np.random.randint(1, n_points // 4), n_points - idx1[0])
            new_h = best_h.copy()
            new_h[idx1[0]:idx1[0]+swap_len], new_h[idx2[0]:idx2[0]+swap_len] = \
                new_h[idx2[0]:idx2[0]+swap_len], new_h[idx1[0]:idx1[0]+swap_len]
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        # Waveform perturbations
        elif np.random.rand() < 0.4:
            freq = np.random.uniform(1, 5)
            noise = 0.15 * np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
            new_h = np.clip(best_h + noise, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
        # Random walk with local search
