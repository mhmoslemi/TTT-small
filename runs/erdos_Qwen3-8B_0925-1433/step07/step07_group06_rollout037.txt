import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    # Genetic Algorithm parameters
    pop_size = 50
    generations = 100
    elite_frac = 0.2
    mutation_rate = 0.05
    
    # Initialize population
    population = [h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        perturb = np.random.normal(0, 0.05, n_points)
        population[i] = h + perturb
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind, 2.0 / n_points, n_points) for ind in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        elite_count = int(elite_frac * pop_size)
        elites = sorted_pop[:elite_count]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elite_count):
            a, b = np.random.choice(pop_size, 2, replace=False)
            parent1, parent2 = sorted_pop[a], sorted_pop[b]
            child = (parent1 + parent2) / 2
            mutate_mask = np.random.rand(n_points) < mutation_rate
            child[mutate_mask] += np.random.normal(0, 0.01, np.sum(mutate_mask))

            child = child * target_sum / np.sum(child)
            offspring.append(child)
        population = elites + offspring

                best_c5 = curr_c5
                best_h = ind.copy()
    
    # Local refinement with COBYLA
    top_candidates = sorted_pop[:5]
    for candidate in top_candidates:
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
