import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_val = np.clip(x, 0, 1)
        if np.sum(h_val) == 0:
            return float('inf')
        h_val = h_val * target_sum / np.sum(h_val)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Global search with DE
    result_de = differential_evolution(obj, bounds, popsize=30, strategy='best1bin', tol=1e-10, maxiter=100, seed=seed)
    best_h = result_de.x
    best_c5 = result_de.fun

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    best_h, best_c5 = result_slsqp.x, result_slsqp.fun

    # Early stop if time is up
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
