import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def compute_c5_and_worst_lag(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    k_max = np.argmax(corr)
    return max_corr, k_max

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5, _ = compute_c5_and_worst_lag(best_h, dx, n_points)
    
    start_time = time.time()
    mutation_amp = 0.025
    iterations = 0
    max_iterations = 1000
    while iterations < max_iterations and time.time() - start_time < budget_s:
        noise = np.random.normal(0, mutation_amp, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum == 0:
            perturbed = best_h
        else:
            perturbed = perturbed * target_sum / current_sum
        
        current_c5, k_max = compute_c5_and_worst_lag(perturbed, dx, n_points)
        
        def obj(x):
            curr_corr = np.correlate(x, 1 - x, mode='full')
            return curr_corr[k_max] * dx
        
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, perturbed, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-8)
