import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scale = target_sum / np.sum(h)
        return h * scale

    population = [np.array(initial_h_values)]
    for _ in range(19):
        perturbed = initial_h_values + np.random.normal(0, 0.05, n_points)
        perturbed = normalize_h(perturbed)
        population.append(perturbed)

    best_h = population[0]
    best_score = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([compute_c5(h, dx, n_points) for h in population])
        current_best = population[np.argmin(scores)]
        best_score = min(best_score, compute_c5(current_best, dx, n_points))
        best_h = current_best.copy()

        elite_size = max(1, int(len(population) * 0.1))
        elites = np.array([population[i] for i in np.argsort(scores)[:elite_size]])

        offspring = []
        for _ in range(len(population) - elite_size):
            p1, p2 = np.random.choice(population, 2, replace=False)
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            mask = np.random.rand(n_points) < 0.1
            child[mask] += np.random.normal(0, 0.05, np.sum(mask))
            child = normalize_h(child)
            offspring.append(child)

        population = np.vstack([elites, offspring])
        population = population[:len(population)]

    return (best_h, best_score, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
