import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):

            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            c5_perturbed = compute_c5(h_perturbed, dx, n_points)
            grad[i] = (c5_perturbed - best_c5) / eps

    # PSO phase
