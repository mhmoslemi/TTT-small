import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    # Simulated Annealing
    T = 1.0
    temp_decay = 0.995
    iter_count = 0
    max_iter = int(budget_s * 0.6)
    while iter_count < max_iter and time.time() - start_time < budget_s:
        for _ in range(10):
            perturb = np.random.uniform(-0.2, 0.2, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            new_c5 = obj(new_h)
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = new_h
                best_c5 = new_c5
        T *= temp_decay
        iter_count += 1

    # Local Refinement
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        h = best_h

    return (best_h, best_c5, n_points)
