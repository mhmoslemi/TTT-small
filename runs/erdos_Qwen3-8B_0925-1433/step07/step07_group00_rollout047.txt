import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 45
        f = 2 * np.pi * (n_points / 3.5)
        h = 0.5 + 0.5 * np.sin(f * np.linspace(0, 2, n_points))
        h = np.clip(h, 0, 1)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
