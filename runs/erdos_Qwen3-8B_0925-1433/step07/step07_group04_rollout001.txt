import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    current_sum = np.sum(h)
    scaling = target_sum / current_sum
    h = np.clip(h * scaling, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_idx = np.argmax(corr)
        lag = k_idx - (n_points - 1)
        if lag > 0:
            perturb_start = 0
            perturb_end = int((2 - lag) * n_points / 2)
            indices = np.arange(perturb_start, min(perturb_end, n_points))
        elif lag < 0:
            perturb_start = int((lag + 2) * n_points / 2)
            perturb_end = n_points
            indices = np.arange(perturb_start, perturb_end)
        else:
            indices = np.arange(n_points)

        perturbed = best_h.copy()
        perturbed[indices] += np.random.normal(0, 0.03, len(indices))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = np.clip(perturbed * target_sum / np.sum(perturbed), 0, 1)

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        if res.success:
            curr_h = res.x
            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        
        res_best = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        if res_best.success:
            curr_h_best = res_best.x
            curr_c5_best = compute_c5(curr_h_best, dx, n_points)
            if curr_c5_best < best_c5:
                best_c5 = curr_c5_best
                best_h = curr_h_best.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
