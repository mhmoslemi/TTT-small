import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def objective(x):
        x_sum = np.sum(x)
        if x_sum == 0:
            return float('inf')
        scaling = target_sum / x_sum
        adjusted = x * scaling
        adjusted = np.clip(adjusted, 0, 1)
        corr = np.correlate(adjusted, 1 - adjusted, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=30,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        disp=False
    )
