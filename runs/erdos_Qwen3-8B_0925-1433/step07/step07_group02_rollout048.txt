import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    mutation_amp = 0.025
    iterations = 0
    max_iterations = 1000
    while iterations < max_iterations and time.time() - start_time < budget_s:
        # Perturb
        perturbed = best_h + np.random.normal(0, mutation_amp, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) == 0:
            perturbed = best_h
        else:
            perturbed = perturbed * target_sum / np.sum(perturbed)
        
        # Find worst lag
        current_corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        k_max = np.argmax(current_corr)
        
        # Objective function
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_max] * dx
        
        # Optimization
        res = minimize(objective, perturbed, method='Nelder-Mead', bounds=[(0, 1) for _ in range(n_points)])
        if res.success:
            new_h = res.x
        else:
            new_h = perturbed.copy()
        
        # Clip and scale
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) == 0:
            new_h = best_h
        else:
            new_h = new_h * target_sum / np.sum(new_h)
        
        # Evaluate
        new_c5 = compute_c5(new_h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
            if best_c5 < 0.38:
                mutation_amp *= 0.975
            elif best_c5 > 0.385:
                mutation_amp *= 1.025
        iterations += 1
    return (best_h, best_c5, n_points)
