import time

def compute_c5(h, dx, n_points):
    max_overlap = 0.0
    for k in range(n_points):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

    initial_sum = np.sum(initial_h_values)

    best_h = h.copy()

    start_time = time.time()
    learning_rate = 0.1
    iterations = 0

        # Compute current C5 to find worst lag
        current_c5 = compute_c5(h, dx, n_points)

            best_h = h.copy()

            learning_rate *= 1.01
        elif current_c5 > best_c5 + 1e-5:

        iterations += 1
    return (best_h, best_c5, n_points)

    max_overlap = 0.0
    for k in range(n_points):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap
