import time
from scipy.optimize import minimize, Bounds, LinearConstraint

        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2
    def compute_c5(h_vec, dx):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    pop_size = 50
    generations = 120
    elite_frac = 0.2
    mutation_rate = 0.15
    mutation_scale = 0.05
    
    # Generate initial population
    def gen_pop():
        pop = []
        for _ in range(pop_size):
            vec = h.copy()
            vec += np.random.normal(0, mutation_scale, n_points)
            vec = np.clip(vec, 0, 1)
            vec = vec * target_sum / np.sum(vec)
            pop.append(vec)
        return np.array(pop)
    pop = gen_pop()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fit = np.array([compute_c5(p, 2.0 / n_points) for p in pop])
        # Select elites
        elite_idx = np.argsort(fit)[:int(elite_frac * pop_size)]
        elites = pop[elite_idx]

        parent_idx = np.random.choice(pop_size, size=int(pop_size * 0.6), replace=False)
        parents = pop[parent_idx]
        # Crossover
        offspring = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            child = np.copy(p1)
            mask = np.random.rand(n_points) < 0.5
            child[mask] = p2[mask]
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Mutation
        for i in range(len(offspring)):
            vec = offspring[i]
            if np.random.rand() < mutation_rate:
                vec += np.random.normal(0, mutation_scale, n_points)
                vec = np.clip(vec, 0, 1)
                vec = vec * target_sum / np.sum(vec)
            offspring[i] = vec
        # Combine elites and offspring
        pop = np.vstack((elites, offspring))
        # Track best
        curr_best = pop[np.argmin(fit)]
        if compute_c5(curr_best, 2.0 / n_points) < best_c5:
