from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Start with initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    best_h, best_c5 = result.x, result.fun

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Generate a random perturbation

    # Final refinement with L-BFGS-B
