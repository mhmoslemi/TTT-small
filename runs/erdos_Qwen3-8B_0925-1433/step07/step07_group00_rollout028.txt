import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h with a structured pattern to balance the integral
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial optimization with COBYLA
    start_time = time.time()
    result = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    # Refinement with structured perturbations
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Add structured perturbations: swap non-overlapping segments
        if n_points < 10:
            continue
        seg_len = int(n_points * 0.3)
        seg1 = np.random.choice(n_points - seg_len, 1)[0]

        new_h = best_h.copy()
