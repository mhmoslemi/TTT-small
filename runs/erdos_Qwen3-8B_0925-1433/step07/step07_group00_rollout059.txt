import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate initial guess with sinusoidal modulation
    freq = 2 * np.pi * (n_points / 4)
    mod = 0.3 * np.sin(freq * np.linspace(0, 2, n_points))
    h = initial_h_values * (1 + mod)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Apply structured perturbation
        k = np.random.randint(1, 5)
        perturb = 0.04 * np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
        new_h = np.clip(best_h + perturb, 0, 1)

        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
