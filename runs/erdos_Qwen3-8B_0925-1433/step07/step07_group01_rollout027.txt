import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points) + 0.5  # Fallback

    # Define DE objective function
    def de_obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_rescaled = h_clipped * target_sum / sum_h
        corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Run differential evolution
    start_time = time.time()
    result_de = differential_evolution(
        de_obj, bounds, strategy='best1bin',
        popsize=20, tol=1e-8, maxiter=300,
        mutation=(0.5, 1.0), recombination=0.9
    )
    best_h_de = result_de.x
    best_c5_de = result_de.fun

    # Early exit if time budget exhausted
    if time.time() - start_time > budget_s:
        return (best_h_de, best_c5_de, n_points)

    # Define SQP objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints for SQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_sqp = minimize(
        obj, best_h_de, method='SLSQP',
        bounds=bounds, constraints=cons, tol=1e-8
    )

    best_h = result_sqp.x
    best_c5 = result_sqp.fun

    # Final feasibility checks
    assert np.all(0 <= best_h <= 1), "h values out of bounds"
    assert np.isclose(np.sum(best_h), target_sum), "Sum constraint violated"

    return (best_h, best_c5, n_points)
