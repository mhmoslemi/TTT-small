from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 30

            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0]
    best_c5 = obj(best_h)

        current_best = population[best_idx]

        if current_best_c5 < best_c5:
