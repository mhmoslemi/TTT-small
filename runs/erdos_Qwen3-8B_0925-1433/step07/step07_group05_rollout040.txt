        h_shifted = current_h[x_shifted]
        if h_x == 1 and h_shifted == 0:
            new_h_x = max(0, h_x - 0.1)
            current_h[x] = new_h_x
        elif h_x == 0 and h_shifted == 1:
            new_h_x = min(1, h_x + 0.1)
            current_h[x] = new_h_x
        total = np.sum(current_h)
        scale = target_sum / total
        current_h = current_h * scale
    
    return (best_h, best_c5, n_points)
