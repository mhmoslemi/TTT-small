                grad[i] += 1 - h[i + k]
            if 0 <= i - k < n_points:
                grad[i] -= h[i - k]
        
        # Gradient descent with adaptive learning
        learning_rate = 0.1 / (np.sqrt(np.sum(grad**2) + 1e-8))
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Stochastic perturbation
        perturbed = h + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        current_c5_p = compute_c5(perturbed, dx, n_points)
        if current_c5_p < best_c5:
            best_h = perturbed.copy()
            best_c5 = current_c5_p
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
