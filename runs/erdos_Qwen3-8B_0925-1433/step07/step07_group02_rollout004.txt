import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import cvxpy as cp

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    h = initial_h_values.copy() * (target_sum / initial_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iterations = 0
    max_iterations = 500
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        delta = m - (n_points - 1)
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + delta) % n_points
            A[i, j] = 1
        x = cp.Variable(n_points)
        prob = cp.Problem(cp.Maximize(x.T @ A @ x),
                          [cp.Min(x, 1), cp.Max(x, 0), cp.sum(x) == target_sum])
        try:

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h
        except:
            pass
        iterations += 1
    return (best_h, best_c5, n_points)
