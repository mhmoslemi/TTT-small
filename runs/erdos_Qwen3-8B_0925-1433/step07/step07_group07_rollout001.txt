        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        
        perturb_indices = np.argsort(-np.abs(grad))[:15]
        h_perturb = h.copy()
        h_perturb[perturb_indices] += np.random.normal(0, perturb_size, len(perturb_indices))
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        
        curr_c5 = compute_c5(h_perturb, dx, n_points)
        delta = curr_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_perturb.copy()
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        T *= cooling
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
