import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    def objective(x):
        h_candidate = np.clip(x, 0, 1)
        if np.sum(h_candidate) > 1e-12:
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        else:
            h_candidate = x
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    # Differential evolution optimization
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=30, tol=1e-8, mutation=(0.5, 1.0), 
        recombination=0.8, polish=True, 
        maxiter=100, seed=seed,
        disp=False
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    # SLSQP refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(
        objective, best_h, method='SLSQP',
        bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 100}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    # Check time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 10:
        return (best_h, best_c5, n_points)
    
    # Early perturbation
    perturb = np.random.normal(0, 0.05, n_points)
    new_h = np.clip(best_h + perturb, 0, 1)
    new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 1e-12 else best_h
    res = minimize(
        objective, new_h, method='SLSQP',
        bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    if res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
