def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize the initial guess
    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    current_sum = np.sum(h)
    scaling = target_sum / current_sum
    h = np.clip(h * scaling, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    # Prepare constraints and bounds
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_c5 = corr[max_idx] * dx

        # Define objective for the worst lag
        def objective(x):
            corr_current = np.correlate(x, 1 - x, mode='full')
            return corr_current[max_idx] * dx

        # Perform optimization
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 100}
        )

        if res.success:
            curr_h = res.x
            # Clip and rescale to ensure bounds and sum
            curr_h = np.clip(curr_h, 0, 1)
            curr_sum = np.sum(curr_h)
            curr_h = curr_h * target_sum / curr_sum

            curr_c5 = compute_c5(curr_h, dx, n_points)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
        else:
            break  # Optimization failed, exit

    return (best_h, best_c5, n_points)
