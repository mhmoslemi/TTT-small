    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        h = h.copy()

        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:

        return h

    start_time = time.time()

    best_h, best_c5 = result.x, result.fun

    for _ in range(100):
