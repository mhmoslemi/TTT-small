import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()

    start_time = time.time()

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx

            corr = np.correlate(x, 1 - x, mode='full')
            return corr[max_idx] * dx

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 100}
        )

        if res.success:
            curr_h = res.x
            curr_h = np.clip(curr_h, 0, 1)
            curr_sum = np.sum(curr_h)

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
