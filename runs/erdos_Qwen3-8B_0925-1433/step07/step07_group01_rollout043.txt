import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    np.random.seed(seed)
    bounds = [(0, 1)] * n_points
    
    def de_obj(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) == 0:
            return np.inf
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    res = differential_evolution(
        de_obj, bounds, strategy='best1bin', maxiter=100, popsize=20,
        tol=1e-8, mutation=(0.5, 1.0), recombination=0.9
    )
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        best_h = res.x
        best_c5 = res.fun
    else:
        def sqp_obj(x):
            h = np.clip(x, 0, 1)
            if np.sum(h) == 0:
                return np.inf
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(sqp_obj, res.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        best_h = result.x
        best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h
