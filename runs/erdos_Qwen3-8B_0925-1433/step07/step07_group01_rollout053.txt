from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    h = validate(h)

    best_h = h.copy()
    best_c5 = obj(h)

    def evaluate(h):
        return obj(validate(h))

    for gen in range(200):
        if time.time() > time.time() + budget_s:
            break

        for i in range(0, population_size, 2):
