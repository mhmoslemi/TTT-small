    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    T = 1.0
    cooling = 0.995
    steps = 0
    max_steps = 1000
    while steps < max_steps and time.time() - start_time < budget_s:
        steps += 1
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        if k_max == 0:
            continue
        indices_range = np.arange(0, n_points - k_max)
        contributions = h[indices_range] * (1 - h[indices_range + k_max])
        top_indices = np.argsort(contributions)[-10:]
        perturbation_indices = []
        for i in top_indices:
            perturbation_indices.append(i)
            perturbation_indices.append(i + k_max)
        perturbation_indices = list(set(perturbation_indices))
        if len(perturbation_indices) < 1:
            continue
        h_perturb = h.copy()
        h_perturb[perturbation_indices] += np.random.normal(0, 0.02, len(perturbation_indices))
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        curr_c5 = compute_c5(h_perturb, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_perturb.copy()
        h = h_perturb
        T *= cooling
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
