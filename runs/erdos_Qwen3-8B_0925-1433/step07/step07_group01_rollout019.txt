from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def de_obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    result = differential_evolution(
        de_obj, bounds, strategy='best1bin', popsize=20,
        tol=1e-8, maxiter=300, mutation=(0.5, 1.0), recombination=0.9,
        seed=seed, polish=True, disp=False
    )
    best_h = result.x
    best_c5 = result.fun

    def sqp_obj(x):

        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(sqp_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
