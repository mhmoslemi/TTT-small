import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h using initial_h_values if available
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    # Objective function for DE
    def de_obj(x):
        h_norm = x * target_sum / np.sum(x)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Bounds and DE parameters
    bounds = [(0, 1)] * n_points
    result_de = differential_evolution(
        de_obj, bounds, strategy='best1bin', popsize=20,
        tol=1e-8, maxiter=300, mutation=(0.5, 1.0), recombination=0.9,
        workers=-1, polish=False, vectorized=True
    )

    # Refine with SLSQP
    def obj(x):
        h_norm = x * target_sum / np.sum(x)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_sqp = minimize(obj, result_de.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)

    # Validate and return
    h_values = result_sqp.x
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    final_c5 = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    return (h_values, final_c5, n_points)
