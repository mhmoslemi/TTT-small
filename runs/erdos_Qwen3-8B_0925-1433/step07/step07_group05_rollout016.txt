import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    current_h = np.array(initial_h_values, dtype=float)
    current_h = current_h * target_sum / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iterations = 1000
    delta = 0.05
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        start_x = max(0, -lag)
        end_x = n_points - 1 - lag
        if end_x < start_x:
            continue
        contributions = []
        for x in range(start_x, end_x + 1):
            if x + lag < 0 or x + lag >= n_points:
                continue
            contrib = current_h[x] * (1 - current_h[x + lag])
            contributions.append((x, contrib))
        if not contributions:
            continue
        max_contrib_x = max(contributions, key=lambda t: t[1])[0]
        x = max_contrib_x
        if current_h[x] > delta:
            current_h[x] -= delta
        for y in range(n_points):
            if y == x or current_h[y] >= 1:
                continue
            current_h[y] += delta
            break
        new_c5 = compute_c5(current_h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
    return (best_h, best_c5, n_points)
