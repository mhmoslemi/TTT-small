    while time.time() - start_time < budget_s:
        iter_count += 1
        if iter_count > 500:
            break
        
        # Compute cross-correlation and find worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        relevant_corr = corr[-n_points:]
        k_idx = np.argmax(relevant_corr)
        max_overlap = relevant_corr[k_idx] * dx

        # Objective function to minimize overlap at k_idx
        def objective(x):
            corr_new = np.correlate(x, 1 - x, mode='full')
            relevant_corr_new = corr_new[-n_points:]
            return relevant_corr_new[k_idx] * dx
        
        # Constraints: sum(x) = target_sum and 0 <= x <= 1
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)

        # Perform optimization
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-7, 'maxiter': 100}
        )

        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * (target_sum / new_sum)
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            # Perturbation and re-optimization
            perturbed = best_h + np.random.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = np.clip(perturbed * target_sum / np.sum(perturbed), 0, 1)
            res_perturbed = minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-7, 'maxiter': 100}
            )
            if res_perturbed.success:
                new_h_perturbed = res_perturbed.x
                new_h_perturbed = np.clip(new_h_perturbed, 0, 1)
                new_sum_perturbed = np.sum(new_h_perturbed)
                if new_sum_perturbed > 0:
                    new_h_perturbed = new_h_perturbed * (target_sum / new_sum_perturbed)
                new_c5_perturbed = compute_c5(new_h_perturbed, dx, n_points)
                if new_c5_perturbed < best_c5:
                    best_c5 = new_c5_perturbed
                    best_h = new_h_perturbed.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    relevant_corr = corr[-n_points:]
    return np.max(relevant_corr) * dx
