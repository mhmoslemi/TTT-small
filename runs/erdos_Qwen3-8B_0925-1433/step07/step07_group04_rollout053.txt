import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = []
    for _ in range(20):
        h = initial_h_values.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(h) for h in population])
        top_idx = np.argsort(fitness)[:int(len(population)*0.1)]
        elites = [population[i] for i in top_idx]
        new_pop = elites[:]
        while len(new_pop) < len(population):
            a, b = np.random.choice(elites, 2, replace=False)
            child = np.random.rand(n_points) * a + (1 - np.random.rand(n_points)) * b
            if np.random.rand() < 0.1:
                idx = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                child[idx] += np.random.normal(0, 0.05, len(idx))
                child = np.clip(child, 0, 1)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
        curr_c5 = compute_c5(population[0])
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = population[0].copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
