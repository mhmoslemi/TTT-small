from scipy.optimize import minimize
import time

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = obj(h)

    temp = 1.0
    cooling = 0.995

    sa_solutions = []

    while time.time() - start_time < budget_s * 0.8:

    # Refinement phase with SLSQP
