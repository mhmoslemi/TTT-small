import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def de_obj(x):
        h_clipped = np.clip(x, 0, 1)
        h_rescaled = h_clipped * target_sum / np.sum(h_clipped) if np.sum(h_clipped) > 0 else h_clipped
        corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    start_time = time.time()

    # Run differential evolution with time budget
    try:
        result_de = differential_evolution(
            de_obj, bounds, 
            strategy='best1bin', popsize=20, 
            tol=1e-8, maxiter=300, 
            mutation=(0.5, 1.0), recombination=0.9,
            options={'mutation': (0.5, 1.0), 'recombination': 0.9})
    except:
        result_de = differential_evolution(de_obj, bounds, strategy='best1bin', popsize=20, tol=1e-8, maxiter=100)

    elapsed = time.time() - start_time
    best_h_de = result_de.x

    # Refine with SQP if time remains
    if elapsed < budget_s:
        def sqp_obj(x):
            h_clipped = np.clip(x, 0, 1)
            h_rescaled = h_clipped * target_sum / np.sum(h_clipped) if np.sum(h_clipped) > 0 else h_clipped
            corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        try:
            result_sqp = minimize(sqp_obj, best_h_de, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        except:
            result_sqp = minimize(sqp_obj, best_h_de, method='L-BFGS-B', bounds=bounds)
        best_h = result_sqp.x
        best_c5 = result_sqp.fun
    else:
        best_h = best_h_de
        best_c5 = result_de.fun

    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values) if np.sum(h_values) > 0 else h_values
    return (h_values, best_c5, n_points)
