        h = np.random.rand(45)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        i = np.arange(n_points)
        term1 = np.where(i + m < n_points, 1 - h[i + m], 0)
        term2 = np.where(i - m >= 0, h[i - m], 0)
        grad = term1 - term2
        h += 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (best_h, best_c5, n_points)
