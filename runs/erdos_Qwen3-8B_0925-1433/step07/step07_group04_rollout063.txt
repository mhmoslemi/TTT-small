import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize the initial guess
    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    current_sum = np.sum(h)
    scaling = target_sum / current_sum
    h = np.clip(h * scaling, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_idx = np.argmax(corr)
        current_max = corr[k_idx] * dx

        # Define objective function to minimize the overlap at k_idx
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_idx] * dx

        # Optimization setup
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])

        # Perform optimization
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': 100}
        )

        if res.success:
            new_h = res.x
            # Clip and rescale if necessary
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum != target_sum:
                new_h = new_h * (target_sum / new_sum)
            new_h = np.clip(new_h, 0, 1)
            # Compute new c5
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

        # Optionally, perform a perturbation step
        # for every 10 iterations, or some other condition
        # to avoid getting stuck in local minima
        if np.random.rand() < 0.1:
            # Perturb best_h
            perturbed = best_h + np.random.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = np.clip(perturbed * target_sum / np.sum(perturbed), 0, 1)
            # Optimize perturbed
            res_perturb = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-6, 'maxiter': 100}

                    best_h = curr_h_perturb.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
