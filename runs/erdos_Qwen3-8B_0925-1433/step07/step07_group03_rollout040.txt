import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    step_size = 0.05
    mutation_amp = 0.02
    eps = 1e-5
    while time.time() - start_time < budget_s:
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            c5_plus = compute_c5(h_plus, dx, n_points)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            c5_minus = compute_c5(h_minus, dx, n_points)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        new_c5 = compute_c5(h_new, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            step_size *= 1.01
        else:
            h = h_new.copy()
            step_size *= 0.99
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, mutation_amp, n_points)
            h_perturbed = h_new + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            new_c5_perturbed = compute_c5(h_perturbed, dx, n_points)
            if new_c5_perturbed < best_c5:
                best_c5 = new_c5_perturbed
                best_h = h_perturbed.copy()
        if best_c5 < 0.3805:
            break
    return (best_h, best_c5, n_points)
