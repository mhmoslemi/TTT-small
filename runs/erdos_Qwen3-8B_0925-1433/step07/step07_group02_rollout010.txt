            shifted = np.roll(x, k_max)
            return np.dot(x, 1 - shifted) * dx
        
        bounds = Bounds(0.0, 1.0)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons)
        
        if result.success:
            new_h = result.x
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
    
    return (best_h, best_c5, n_points)
