import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    target_sum = n_points / 2
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_points, n_points)
    start_time = time.time()
    
    temperature = 1.0
    cooling_rate = 0.995
    time_remaining = budget_s
    
    while temperature > 1e-3 and time_remaining > 0:
        start_temp_time = time.time()
        for _ in range(1000):
            if time.time() - start_temp_time > time_remaining:
                break
            noise = np.random.normal(0, temperature * 0.05, n_points)
            perturbed = h + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            new_c5 = compute_c5(perturbed, 2.0 / n_points, n_points)
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5)/temperature):
                h = perturbed.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = perturbed.copy()
        temperature *= cooling_rate
        time_remaining -= time.time() - start_temp_time
    
    # Local refinement
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    bounds = Bounds(0, 1)
    res = minimize(lambda x: compute_c5(x, 2.0/n_points, n_points), best_h, method='COBYLA', bounds=bounds, constraints=cons)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
