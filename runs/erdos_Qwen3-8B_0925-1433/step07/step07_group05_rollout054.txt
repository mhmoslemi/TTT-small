        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx
        contrib = []
        for i in range(n_points - k):
            contrib.append(h[i] * (1 - h[i + k]))
        if contrib:
            contrib = np.array(contrib)
            top_indices = np.argsort(contrib)[-3:]
            adjusted = False
            for i_idx in top_indices:
                i = i_idx
                j = i + k
                if h[i] > 0.5:
                    h[i] -= eps
                    adjusted = True
                elif h[j] < 0.5:
                    h[j] += eps
                    adjusted = True
            if adjusted:
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
    return (best_h, best_c5, n_points)
