from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate a structured initial guess if initial_h_values is not optimal
    def generate_structured_initial(n_points):
        m = n_points // 2
        h = np.zeros(n_points)
        for i in range(m):
            idx = 2 * i
            if idx < n_points:
                h[idx] = 1.0
        return h

    # Use provided initial guess or structured one if initial_h_values is None
    if initial_h_values is None:
        h = generate_structured_initial(n_points)
    else:
        h = np.clip(initial_h_values, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Optimize with trust-constr for better convergence
    def optimize_with_trust_constr(h):
        try:
            result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
            return result
        except:
            return minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

    start_time = time.time()
    result = optimize_with_trust_constr(h)
    best_h, best_c5 = result.x, result.fun

    # Apply structured perturbations
    for _ in range(150):
        if time.time() - start_time > budget_s:

        # Sinusoidal perturbation
