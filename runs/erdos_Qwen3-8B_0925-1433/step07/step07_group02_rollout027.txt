import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    initial_h_scaled = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iterations = 1000
    iterations = 0
    while time.time() - start_time < budget_s and iterations < max_iterations:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            break
        def obj(x):
            total = 0.0
            for i in range(n_points):
                j = (i + k_max) % n_points
                total += x[i] * (1 - x[j])
            return total * dx
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            if np.sum(new_h) == 0:
                new_h = best_h.copy()
            else:
                new_h = new_h * target_sum / np.sum(new_h)
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
        iterations += 1
    return (best_h, best_c5, n_points)
