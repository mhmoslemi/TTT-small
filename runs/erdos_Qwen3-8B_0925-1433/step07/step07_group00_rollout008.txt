import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def optimize(x0):
        try:
            res = minimize(obj, x0, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
            return res.x, res.fun if res.success else x0, np.inf
        except:
            return x0, np.inf

    start_time = time.time()
    best_h, best_c5 = optimize(h)

    for _ in range(30):
        if time.time() - start_time > budget_s - 10:
            break
        # Sinusoidal perturbation
        k = np.random.randint(1, 5)
        A = 0.05 + 0.01 * np.random.rand()
        perturb = A * np.sin(2 * np.pi * k * np.arange(n_points))
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        new_h, new_c5 = optimize(new_h)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5

    return (best_h, best_c5, n_points)
