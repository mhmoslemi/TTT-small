import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale_and_clip(h):
        clipped = np.clip(h, 0, 1)
        current_sum = np.sum(clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        return clipped * (target_sum / current_sum)

    population = [initial_h_values.copy()]
    for _ in range(19):
        h = initial_h_values.copy() + np.random.normal(0, 0.05, n_points)
        h = rescale_and_clip(h)
        population.append(h)
    population = [rescale_and_clip(h) for h in population]

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h) for h in population]
        elite_idx = np.argsort(fitness)[:int(len(population) * 0.1)]
        elites = [population[i] for i in elite_idx]
        
        offspring = []
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = p1 * np.random.rand() + p2 * (1 - np.random.rand())
            mutate_idx = np.random.choice(n_points, int(n_points * 0.1))
            child[mutate_idx] += np.random.normal(0, 0.05, len(mutate_idx))
            child = rescale_and_clip(child)
            offspring.append(child)
        
        new_pop = elites + offspring
        new_pop = np.random.choice(new_pop, len(population), replace=False)
        population = new_pop
        
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
