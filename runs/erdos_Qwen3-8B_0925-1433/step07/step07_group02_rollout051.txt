        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            sum_h = np.sum(new_h)
            if not np.isclose(sum_h, target_sum):
                new_h = new_h * target_sum / sum_h
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
        else:
            new_h = best_h.copy()
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
        iterations += 1
    return (best_h, best_c5, n_points)
