                h_new = h_var.value
                new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
        except:
            pass

        h = np.clip(best_h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h = h * (target_sum / total)
        else:
            h = best_h

    return (best_h, best_c5, n_points)
