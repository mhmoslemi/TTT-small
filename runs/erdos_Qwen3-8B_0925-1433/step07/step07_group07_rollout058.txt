                contrib_j = -h[i] * dx
                grad_contrib[i] += abs(contrib_i)
                grad_contrib[j] += abs(contrib_j)
            
            indices = np.argsort(grad_contrib)[::-1][:10]
            h_perturb = h.copy()
            for idx in indices:
                h_perturb[idx] += np.random.normal(0, perturbation_step, 1)
                h_perturb[idx] = np.clip(h_perturb[idx], 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            
            new_c5 = compute_c5(h_perturb, dx, n_points)
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_perturb.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        T *= cooling_rate
        if no_improvement_count > 100:
            perturbation_step += 0.005
            no_improvement_count = 0
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
