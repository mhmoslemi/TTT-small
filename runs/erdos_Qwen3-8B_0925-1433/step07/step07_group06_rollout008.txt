from scipy.optimize import minimize, Bounds, LinearConstraint

    bounds = Bounds(0, 1)
    
    for _ in range(100):  # Dynamic optimization phases
        if time.time() - start_time > budget_s:
            break
        # SLSQP with adaptive perturbation

        res = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:

            curr_c5 = compute_c5(curr_h, 2.0 / n_points, n_points)
