import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()

    start_time = time.time()
    
    T = 1.0
    cooling = 0.995
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for j in range(n_points):
            term1 = (1 - h[j + k_max]) * dx if j + k_max < n_points else 0
            term2 = -h[j - k_max] * dx if j - k_max >= 0 else 0
            grad[j] = term1 + term2
        selected = np.argsort(np.abs(grad))[::-1][:15]
        h_perturb = h.copy()
        for idx in selected:
