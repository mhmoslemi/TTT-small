import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def de_obj(x):
        h_candidate = np.clip(x, 0, 1)
        sum_h = np.sum(h_candidate)
        if sum_h == 0:
            h_candidate = np.zeros_like(h_candidate)
        else:
            h_candidate = h_candidate * target_sum / sum_h
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result_de = differential_evolution(de_obj, bounds, strategy='best1bin', popsize=20, tol=1e-8,
                                      maxiter=300, mutation=(0.5, 1.0), recombination=0.9, seed=seed)

    best_h_de = result_de.x
    best_c5_de = result_de.fun
    remaining_time = budget_s - (time.time() - start_time)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_sqp = [(0, 1)] * n_points

    best_h = best_h_de
    best_c5 = best_c5_de

    if remaining_time > 0:
        result_sqp = minimize(obj, best_h_de, method='SLSQP', bounds=bounds_sqp, constraints=cons,
                              tol=1e-10, options={'maxiter': 100})
        best_h_sqp = result_sqp.x
        best_c5_sqp = result_sqp.fun
        if best_c5_sqp < best_c5:
            best_h = best_h_sqp
            best_c5 = best_c5_sqp

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h, final_c5, n_points)
