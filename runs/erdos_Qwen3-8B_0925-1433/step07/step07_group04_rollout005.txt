import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with variations of initial_h_values
    population_size = 20
    population = []
    for _ in range(population_size):
        h = initial_h_values.copy()
        perturb = np.random.normal(0, 0.01, n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        # Select elites (top 20%)
        elite_indices = np.argsort(fitness)[:population_size // 5]
        elites = [population[i] for i in elite_indices]
        # Generate offspring
        offspring = []
