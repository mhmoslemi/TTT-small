        h = h * target_sum / total
        # Update best solution
        curr_c5 = compute_c5(h, 2.0 / n_points, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
