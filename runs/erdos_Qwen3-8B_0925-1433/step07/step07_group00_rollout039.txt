import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    x = np.linspace(0, 2, n_points)
    if initial_h_values is not None:
        h = initial_h_values
    else:
        freq = 2 * np.pi * n_points / (3.5 * 2)
        h = 0.5 + 0.5 * np.sin(freq * x)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x_values):
        corr = np.correlate(x_values, 1 - x_values, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
    if time.time() - start_time > budget_s:
        best_h = h
        best_c5 = obj(h)
    else:
        best_h = result.x
        best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h

    return (best_h, best_c5, n_points)
