    for phase in range(3):

            break
        # Adjust perturbation step based on phase
        perturb_scale = 0.05 if phase == 0 else 0.02
        for _ in range(50):  # Max iterations per phase
            if time.time() - start_time > budget_s:
                break
            # Phase 1: Gradient descent with SLSQP
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n_points)
            cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
            bounds = Bounds(0, 1)
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons)
            if res.success:
                curr_h = res.x
                curr_c5 = compute_c5(curr_h, 2.0 / n_points, n_points)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5

            # Phase 2: Perturbation and re-optimization
            noise = np.random.normal(0, perturb_scale, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0, 1)

            res_p = minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons)
            if res_p.success:

def compute_c5(h, dx, n_points):
