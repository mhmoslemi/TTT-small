import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    freq = 2 * np.pi * (n_points / 4)
    modulation = 0.2 * np.sin(freq * np.linspace(0, 2, n_points))
    h = np.clip(h * (1 + modulation), 0, 1)
    h = h * target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        freq_perturb = 2 * np.pi * np.random.uniform(0.1, 0.5)
        perturb = 0.05 * np.sin(freq_perturb * np.linspace(0, 2, n_points))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
