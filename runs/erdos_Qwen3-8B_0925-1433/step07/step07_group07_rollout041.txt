    T = 1.0
    cooling = 0.995
    num_perturb = 15
    perturbation_size = 0.03
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        
        # Compute correlation and find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        
        # Compute gradient contributions for the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] = (1 - h[i + lag]) * dx
            else:
                grad[i] = 0.0
        
        # Select points to perturb
        selected_indices = np.argsort(grad)[-num_perturb:]
        
        # Create perturbation
        h_perturb = h.copy()
        for idx in selected_indices:
            h_perturb[idx] -= perturbation_size
        
        # Clip and rescale
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        
        # Compute current C5
        curr_c5 = compute_c5(h_perturb, dx, n_points)
        
        # Acceptance probability
        delta = curr_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_perturb.copy()
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        
        T *= cooling
    
    return (best_h, best_c5, n_points)
