import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.inf
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        # Random restart
        random_h = np.random.rand(n_points)
        random_h = random_h * target_sum / np.sum(random_h)
        random_h = np.clip(random_h, 0, 1)
        res_rand = minimize(obj, random_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res_rand.success and res_rand.fun < best_c5:
            best_h, best_c5 = res_rand.x, res_rand.fun
        # Larger perturbation for exploration
        perturb_large = np.random.normal(0, 0.05, n_points)
        new_h_large = np.clip(best_h + perturb_large, 0, 1)
        new_h_large = new_h_large * target_sum / np.sum(new_h_large) if np.sum(new_h_large) > 0 else best_h
        res_large = minimize(obj, new_h_large, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

    return (best_h, best_c5, n_points)
