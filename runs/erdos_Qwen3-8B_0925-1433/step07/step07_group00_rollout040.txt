import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    # Initial optimization with trust-constr
    res = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun

    # Perturbations with sinusoidal mixing and segment swaps
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Sinusoidal perturbation
        freqs = [4, 5, 8]
        A = 0.04
        freqs = np.array([4, 5, 8])
        perturb = np.zeros(n_points)
        for f in freqs:
            perturb += A * np.sin(2 * np.pi * f * np.linspace(0, 2, n_points))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        # Swap segments to diversify patterns
        def swap_segments(h):
            idx = np.argsort(np.random.rand(n_points))
            mid = n_points // 4
            return np.concatenate([h[idx[:mid]], h[idx[mid:]]])
        new_h = swap_segments(new_h)
        res = minimize(obj, new_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (best_h, best_c5, n_points)
