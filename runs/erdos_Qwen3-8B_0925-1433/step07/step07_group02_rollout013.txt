import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * target_sum / initial_sum
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    iterations = 0
    max_iterations = 500
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        m = max_index - (n_points - 1)
        m_shift = m % n_points
        def obj(x):
            shifted = np.roll(x, shift=m_shift)
            return np.dot(x, shifted)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(lambda x: -obj(x), best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

            if new_c5 < best_c5:

                best_h = new_h
