from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with provided starting point
    h = np.array(initial_h_values, dtype=float)

    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    # DE objective function with constraint enforcement
    def de_obj(x):
        h_clip = np.clip(x, 0, 1)
        h_rescaled = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Run differential evolution
    start_time = time.time()
    result_de = differential_evolution(
        de_obj,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-8,
        maxiter=300,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun
    time_used = time.time() - start_time

    # Local refinement with SQP
    if time_used < budget_s - 100:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds_sqp = [(0, 1)] * n_points

        def sqp_obj(x):

            return np.max(corr) * dx

        result_sqp = minimize(
            sqp_obj,
            best_h,
            method='SLSQP',

            constraints=cons,
            tol=1e-10,
            options={'maxiter': 100}
        )
        if result_sqp.fun < best_c5:
