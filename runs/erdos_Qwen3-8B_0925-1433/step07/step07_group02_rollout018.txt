import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    max_iterations = 1000
    iterations = 0
    
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Define objective function for the worst lag
        def obj(x):
            k = k_max
            idx = (np.arange(n_points) + k) % n_points
            return np.dot(x, 1 - x[idx]) * dx
        
        # Setup constraints and bounds
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        
        # Add small perturbation to initial guess
        perturbation = np.random.normal(0, 0.01, n_points)
        initial_guess = best_h + perturbation
        initial_guess = np.clip(initial_guess, 0, 1)
        initial_guess = initial_guess * target_sum / np.sum(initial_guess)
        
        res = minimize(obj, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum == 0:
                new_h = best_h
            else:
                new_h = new_h * target_sum / new_sum
            new_c5 = compute_c5(new_h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
        else:
            new_h = best_h
            new_c5 = current_c5
        
        iterations += 1
    
    return (best_h, best_c5, n_points)
