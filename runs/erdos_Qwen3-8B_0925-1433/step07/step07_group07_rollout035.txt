import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    step_size = 0.02
    cooling_factor = 0.997
    T = 1.0
    epochs_without_improvement = 0
    max_epochs = 1000

    for _ in range(max_epochs):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            epochs_without_improvement = 0
        else:
            epochs_without_improvement += 1
        
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus = i + k_max
            i_minus = i - k_max
            term1 = 0.0
            if 0 <= i_plus < n_points:
                term1 = 1.0 - h[i_plus]
            term2 = 0.0
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            grad[i] = (term1 - term2) * dx
        
        indices = np.argsort(np.abs(grad))[::-1]
        selected_indices = indices[:10]
