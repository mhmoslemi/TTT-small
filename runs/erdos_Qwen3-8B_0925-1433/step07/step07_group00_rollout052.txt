import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    # Structured initial guess
    freq = 2 * np.pi * (n_points / 4)
    h = h * (1.0 + 0.2 * np.sin(freq * np.linspace(0, 2, n_points)))
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        # Apply sinusoidal perturbation
        k = np.random.randint(1, 6)
        perturb = 0.05 * np.sin(2 * np.pi * k * np.linspace(0, 2, n_points))
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        # Apply segment swap
        len_seg = np.random.randint(20, 40)
        start1 = np.random.randint(0, n_points - len_seg)
        start2 = np.random.randint(0, n_points - len_seg)
        new_h = best_h.copy()
        new_h[start1:start1+len_seg], new_h[start2:start2+len_seg] = new_h[start2:start2+len_seg], new_h[start1:start1+len_seg]

        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
