import numpy as np
import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    max_iterations = 500
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        for x in range(n_points):
            shifted_x = x + k
            if 0 <= shifted_x < n_points:
                if h[x] > 0.9 and h[shifted_x] < 0.1:
                    h[x] = max(h[x] - 0.1, 0.0)
                elif h[x] < 0.1 and h[shifted_x] > 0.9:
                    h[x] = min(h[x] + 0.1, 1.0)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h = h * target_sum / sum_h
        if np.random.rand() < 0.2:
            perturbed = h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed * target_sum / np.sum(perturbed)
            new_c5 = compute_c5(perturbed, dx, n_points)
            if new_c5 < best_c5:
                h = perturbed
                best_c5 = new_c5
                best_h = h.copy()
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)
