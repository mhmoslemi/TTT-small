import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_scaled = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    max_iterations = 1000
    iterations = 0
    while iterations < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        objective = lambda x: np.dot(x, 1 - x[(np.arange(n_points) + k_max) % n_points]) * dx
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
        else:
            new_h = best_h.copy()
        new_sum = np.sum(new_h)
        if abs(new_sum - target_sum) > 1e-6:
            new_h = new_h * target_sum / new_sum
        new_h = np.clip(new_h, 0, 1)
        new_c5 = compute_c5(new_h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
        iterations += 1
    return (best_h, best_c5, n_points)
