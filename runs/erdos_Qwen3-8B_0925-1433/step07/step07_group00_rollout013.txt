import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate improved initial guess with multi-frequency sinusoidal adjustment
    x = np.linspace(0, 2, n_points)
    freq1 = 2 * np.pi * (n_points / 4)
    freq2 = 2 * np.pi * (n_points / 6)
    h = initial_h_values * (1 + 0.1 * np.sin(freq1 * x) + 0.1 * np.cos(freq2 * x))
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Swap non-overlapping segments
        seg_len = max(1, n_points // 5)
        start1 = np.random.randint(0, n_points - seg_len)
        start2 = np.random.randint(0, n_points - seg_len)
        while abs(start1 - start2) < seg_len:
            start2 = np.random.randint(0, n_points - seg_len)
        h_swap = best_h.copy()
        h_swap[start1:start1+seg_len], h_swap[start2:start2+seg_len] = h_swap[start2:start2+seg_len].copy(), h_swap[start1:start1+seg_len].copy()
        # Apply sinusoidal perturbation with random frequency
        perturb = 0.05 * np.sin(2 * np.pi * np.random.randint(1, 6) * x)
        new_h = np.clip(h_swap + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
