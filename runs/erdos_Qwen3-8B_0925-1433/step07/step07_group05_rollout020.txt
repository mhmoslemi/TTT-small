        current_sum = np.sum(h_new)
        scale = target_sum / current_sum
        h_new = h_new * scale
        h = h_new.copy()
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        step_size *= 0.999
    
    return (best_h, best_c5, n_points)
