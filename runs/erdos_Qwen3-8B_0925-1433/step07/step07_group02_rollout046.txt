import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * (target_sum / np.sum(initial_h_values))
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    max_iterations = 1000
    learning_rate = 0.1

    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            if best_c5 < 0.38:
                break
        k_max = np.argmax(corr)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            idx1 = (j + k_max) % n_points
            idx2 = (j - k_max) % n_points
            gradient[j] = dx * (1 - h[idx1]) - dx * h[idx2]
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new != 0:
            h = h_new * target_sum / sum_new
        else:
            h = h_new
        if current_c5 < best_c5:
            learning_rate *= 1.01
        else:
            learning_rate *= 0.99
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    return (best_h, best_c5, n_points)
