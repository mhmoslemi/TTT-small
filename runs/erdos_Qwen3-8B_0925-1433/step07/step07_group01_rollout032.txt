import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if 'initial_h_values' not in globals():
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    else:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full(n_points, target_sum / n_points)
    dx = 2.0 / n_points

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = obj(h)

    temp = 1.0
    cooling_rate = 0.995
    iterations = 1000
    start_time = time.time()

    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * target_sum / new_sum
        else:
            new_h = np.full(n_points, target_sum / n_points)
        new_c5 = obj(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = new_h.copy()
            best_c5 = new_c5
            if best_c5 < obj(best_h):
                best_h = h.copy()
        temp *= cooling_rate

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if best_sum > 0:
        best_h = best_h * target_sum / best_sum
    else:
        best_h = np.full(n_points, target_sum / n_points)

    return (best_h, result.fun, n_points)
