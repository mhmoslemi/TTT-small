import numpy as np
import cma
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize from initial_h_values if available
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    # Define objective function for CMA-ES
    def obj_cma(x):
        x_clipped = np.clip(x, 0, 1)
        x_rescaled = x_clipped * target_sum / np.sum(x_clipped)
        corr = np.correlate(x_rescaled, 1 - x_rescaled, mode='full')
        return np.max(corr) * dx

    # CMA-ES setup
    mean = h.copy()
    sigma = 0.5
    es = cma.CMAEvolutionStrategy(mean, sigma, {
        'bounds': [(0, 1)] * n_points,
        'popsize': 30,
        'maxiter': 200
    })

    best_h = h.copy()
    best_c5 = obj_cma(h)
    start_time = time.time()

    # Run CMA-ES
    for i in range(es.popsize):
        if time.time() - start_time > budget_s:
            break
        pop = es.ask()
        pop_eval = []
        for x in pop:
            x_clipped = np.clip(x, 0, 1)
            x_rescaled = x_clipped * target_sum / np.sum(x_clipped)
            corr = np.correlate(x_rescaled, 1 - x_rescaled, mode='full')
            pop_eval.append(np.max(corr) * dx)
        best_idx = np.argmin(pop_eval)
        if pop_eval[best_idx] < best_c5:
            best_c5 = pop_eval[best_idx]
            best_h = pop[best_idx].copy()
        es.tell(pop, pop_eval)
        es.rebalance()

    # SLSQP refinement
    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8)

    best_c5 = res.fun
