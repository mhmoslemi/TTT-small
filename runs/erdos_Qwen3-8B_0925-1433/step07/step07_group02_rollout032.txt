import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)

    best_h = initial_h * target_sum / initial_sum
    best_c5 = compute_c5(best_h, dx, n_points)

        if time.time() - start_time > budget_s:

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_lag = np.argmax(corr)
        obj = lambda x: corr[worst_lag] * dx if np.all(0 <= x) and np.all(x <= 1) else np.inf
