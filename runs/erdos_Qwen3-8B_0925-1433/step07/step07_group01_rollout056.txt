import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    h = np.array(initial_h_values, copy=True)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 1e-10 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 1e-10 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # SA parameters
    temp = 1.0
    cooling = 0.995
    sa_iter = 1000
    sa_sol = [h.copy()]

    start_time = time.time()
    for _ in range(sa_iter):
        if time.time() - start_time > budget_s:
            break
        # Generate neighbor
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 1e-10 else h
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 1e-10 else h

        # Evaluate
        curr_c5 = obj(h)
        new_c5 = obj(new_h)
        delta = new_c5 - curr_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = new_h
        if np.random.rand() < 0.1:
            sa_sol.append(h.copy())
        temp *= cooling

    # Local refinement
    best_h = h.copy()
    best_c5 = obj(h)
    sa_sol = np.array(sa_sol)
    sa_sol = sa_sol[np.argsort(obj(sa_sol))][:10]
    for sol in sa_sol:
        res = minimize(obj, sol, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 1e-10 else np.zeros(n_points)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 1e-10 else np.zeros(n_points)
    final_c5 = obj(best_h)

    return (best_h, final_c5, n_points)
