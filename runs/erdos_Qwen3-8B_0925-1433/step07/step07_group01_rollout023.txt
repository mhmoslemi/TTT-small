import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)

    def objective(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) < 1e-10:
            return 1e10
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    # Global optimization with DE
    try:
        result_de = differential_evolution(
            objective, bounds,
            strategy='best1bin', mutation=(0.5, 1), recombination=0.9,
            popsize=30, tol=1e-8, maxiter=300, disp=False
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        best_h = h_initial
        best_c5 = objective(h_initial)

    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(
        objective, best_h, method='SLSQP',
        bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-10
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    # Final check for time and refinement
    if time.time() - start_time < budget_s:
        perturb = np.random.normal(0, 0.02, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
