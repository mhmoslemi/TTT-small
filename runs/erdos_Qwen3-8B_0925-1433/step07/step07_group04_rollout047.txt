import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    for i in range(n_points // 2):
        h[i], h[n_points - 1 - i] = h[n_points - 1 - i], h[i]
    h = np.clip(h * target_sum / np.sum(h), 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        lag_idx = np.argmax(corr)
        curr_max_val = corr[lag_idx] * dx

        # Objective is to minimize overlap at worst lag
        def objective(x):
            return np.correlate(x, 1 - x, mode='full')[lag_idx] * dx

        cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
        bounds = Bounds(0, 1)
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-6, 'maxiter': 100})
        if res.success:
            curr_h = res.x
            curr_corr = np.correlate(curr_h, 1 - curr_h, mode='full')
            curr_c5 = np.max(curr_corr) * dx
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()

        # Random perturbation to escape local minima
        perturb = np.random.normal(0, 0.03, n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = np.clip(perturbed * target_sum / np.sum(perturbed), 0, 1)

        res_perturb = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-6, 'maxiter': 100})
        if res_perturb.success:
            curr_h_perturb = res_perturb.x
            curr_corr_perturb = np.correlate(curr_h_perturb, 1 - curr_h_perturb, mode='full')
            curr_c5_perturb = np.max(curr_corr_perturb) * dx
            if curr_c5_perturb < best_c5:
                best_c5 = curr_c5_perturb
                best_h = curr_h_perturb.copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
