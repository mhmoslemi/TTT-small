import time

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    # Initialize h
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    # Hyperparameters
    step_size = 0.05
    max_iterations = 1000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        index_max = np.argmax(corr)
        lag_k = index_max - (n_points - 1)
        # Compute gradient for lag_k
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + lag_k
            if 0 <= i_plus_k < n_points:
                h_plus = h[i_plus_k]
            else:
                h_plus = 0.0
            i_minus_k = i - lag_k
            if 0 <= i_minus_k < n_points:
                h_minus = h[i_minus_k]
            else:
                h_minus = 0.0
            grad[i] = (1 - h_plus) - h_minus
        # Update h
        h_new = h - step_size * grad
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize sum
        sum_new = np.sum(h_new)
        if sum_new == 0:
            # Avoid division by zero
            h = h_new
        else:
            scale_factor = target_sum / sum_new
            h = h_new * scale_factor
        # Compute current c5
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Check if we've improved enough
        # ... maybe add early stopping if max c5 stops improving for several steps
    return (best_h, best_c5, n_points)
