import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()
    
    max_iterations = 1000
    cooling_factor = 0.997
    T = 1.0
    improved = 0
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        if T < 1e-6:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        start_i = max(0, -lag)
        end_i = min(n_points - 1, n_points - 1 - lag)
        valid_indices = list(range(start_i, end_i + 1))
        
        grad_contributions = []
        for i in valid_indices:
            j = i + lag
            if 0 <= j < n_points:
                contrib = (1 - h[j]) * dx + h[i] * dx
                grad_contributions.append((contrib, i))
        
        grad_contributions.sort(reverse=True)
        top_indices = [i for (c, i) in grad_contributions[:10]]
        
        perturbations = np.random.normal(0, 0.01, len(top_indices))
        for idx, pos in enumerate(top_indices):
            h[pos] += perturbations[idx]
            h[pos] = np.clip(h[pos], 0, 1)
        
        h = h * target_sum / np.sum(h)
        
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            improved += 1
        else:
            if np.random.rand() < np.exp((best_c5 - new_c5) / T):
                best_h = h.copy()
                best_c5 = new_c5
        
        T *= cooling_factor
        if improved > 100:
            improved = 0
            T *= 0.95
    
    return (best_h, best_c5, n_points)
