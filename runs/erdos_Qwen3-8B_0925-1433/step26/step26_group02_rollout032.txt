import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2.0 / n_points

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def compute_gradient(h):
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1.0
            h_plus = h + eps * e
            h_plus = clip_rescale(h_plus)
            h_minus = h - eps * e
            h_minus = clip_rescale(h_minus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        return grad

    def generate_guesses():
        guesses = []
        phi = np.random.rand() * 2 * np.pi
        guesses.append(0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1) + phi)))
        guesses.append(np.zeros(n_points))
        for i in range(n_points // 4):
            guesses[-1][i] = 1.0
        guesses[-1] = np.roll(guesses[-1], np.random.randint(n_points // 4))
        guesses[-1] = clip_rescale(guesses[-1])
        return [clip_rescale(g) for g in guesses]

    guesses = generate_guesses()
    for guess in guesses:
        h = guess.copy()
        for _ in range(300):
            grad = compute_gradient(h)
            h = np.clip(h - 0.01 * grad, 0, 1)
            h = clip_rescale(h)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if time.time() - time.time() > budget_s:
                break

    result = minimize(compute_c5, best_h, method='L-BFGS-B',
                      bounds=[(0, 1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-6, options={'maxiter': 50})
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x.copy()
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
