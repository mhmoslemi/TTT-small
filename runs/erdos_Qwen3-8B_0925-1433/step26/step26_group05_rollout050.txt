    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            h = particles[i, :]
            corr = np.correlate(h, 1 - h, mode='full')
            current_fitness = np.max(corr) * dx
        for i in range(n_particles):
            if current_fitness[i] < pbest_fitness[i]:
                pbest_fitness[i] = current_fitness[i]
                pbest_positions[i, :] = particles[i, :].copy()
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < gbest_fitness:
            gbest_fitness = current_fitness[best_idx]
            gbest_position = particles[best_idx, :].copy()
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i, :] = 0.8 * velocities[i, :] + 1.5 * r1 * (pbest_positions[i, :] - particles[i, :]) + 1.5 * r2 * (gbest_position - particles[i, :])
            velocities[i, :] = np.clip(velocities[i, :], -0.2, 0.2)
            particles[i, :] += velocities[i, :]
            particles[i, :] = np.clip(particles[i, :], 0, 1)
            sum_h = np.sum(particles[i, :])
            if sum_h != 0:
                particles[i, :] = particles[i, :] * target_sum / sum_h
            else:
                particles[i, :] = 0.5 * np.ones(n_points)

    best_h = gbest_position
    best_c5 = gbest_fitness
    return (best_h, best_c5, n_points)
