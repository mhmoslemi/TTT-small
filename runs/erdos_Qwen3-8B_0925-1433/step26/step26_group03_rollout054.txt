    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    corr = np.correlate(h, 1 - h, mode='full')
    relevant_corr = corr[n-1:]
    best_c5 = np.max(relevant_corr) * dx
    best_c5 = min(best_c5, 0.381)
    best_h = h.copy()
    start_time = time.time()
    alpha = 0.1
    decay_rate = 0.95
    restart_interval = 500
    iterations = 0
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        relevant_corr = corr[n-1:]
        max_overlap = relevant_corr.max()
        m_max = np.argmax(relevant_corr)
        shifted_indices = (np.arange(n_points) + m_max) % n_points
        grad = (1 - h[shifted_indices]) * dx
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = max_overlap * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if iterations % restart_interval == 0:
            perturb = np.random.normal(0, 0.2, size=n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        iterations += 1
        if iterations % 100 == 0:
            alpha *= decay_rate
        if best_c5 < 0.3808:
            break
    return (best_h, best_c5, n_points)
