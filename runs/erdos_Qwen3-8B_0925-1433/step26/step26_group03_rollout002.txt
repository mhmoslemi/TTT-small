import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    start_time = time.time()
    step_size = 0.5
    decay = 0.999
    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        overlaps = []
        for m in range(n_points):
            valid_len = n_points - m
            if valid_len <= 0:
                overlaps.append(0.0)
                continue
            overlap = np.dot(h[:valid_len], 1 - h[m:])
            overlaps.append(overlap * dx)
        max_overlap = max(overlaps)
        m_max = overlaps.index(max_overlap)
        grad = np.zeros(n_points)
        idx1 = np.arange(n_points) + m_max
        valid1 = idx1 < n_points
        grad[valid1] += 1 - h[idx1[valid1]]
        idx2 = np.arange(n_points) - m_max
        valid2 = idx2 >= 0
        grad[valid2] -= h[idx2[valid2]]
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        step_size *= decay
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
