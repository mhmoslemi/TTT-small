    time_start = time.time()

    while time.time() - time_start < budget_s and iter_count < 10000:

        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)

        shifted_minus = (np.arange(n_points) - shift) % n_points

        m = beta1 * m + (1 - beta1) * gradient

    return (best_h, best_c5, n_points)
