    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    learning_rate = 0.5
    max_iterations = int(30000)
    tolerance = 1e-4

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        max_overlap = 0.0
        k_max = 0
        for k in range(n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                k_max = k
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = (i + k_max) % n_points
            j_minus = (i - k_max) % n_points
            grad[i] = (1 - h[j_plus]) - h[j_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = 0.5 * np.ones(n_points)
        else:
            h = h * target_sum / sum_h
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if _ % 100 == 0:
            h += np.random.normal(0, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        learning_rate *= 0.9998

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    n_points = len(h)
    max_overlap = 0.0
    for k in range(n_points):
        shifted = np.roll(h, -k)
        overlap = np.sum(h * (1 - shifted)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap
