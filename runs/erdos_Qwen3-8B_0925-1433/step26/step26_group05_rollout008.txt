    decay_rate = 0.9995
    max_iterations = 1000
    K = 5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlaps[k] = np.dot(h, 1 - shifted) * dx
        top_k_indices = np.argsort(overlaps)[-K:]
        grad = np.zeros(n_points)
        for k in top_k_indices:
            shifted_plus = np.roll(h, k)
            shifted_minus = np.roll(h, -k)
            grad += (1 - shifted_plus) - shifted_minus
        t = iteration + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = 0.5 * np.ones(n_points)
        else:
            h = h * target_sum / sum_h
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if iteration % 100 == 0:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
