import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.round(initial_h_values).astype(int)
    current_sum = np.sum(h)
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        if current_sum < target_sum:
            idx = np.random.choice(np.where(h == 0)[0])
            h[idx] = 1
        else:
            idx = np.random.choice(np.where(h == 1)[0])
            h[idx] = 0
    assert np.isclose(np.sum(h), target_sum, atol=1e-6)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    for _ in range(100000):
        if time.time() - start_time > budget_s:
            break
        candidate = best_h.copy()
        i = np.random.randint(n_points)
        candidate[i] ^= 1
        cs = np.sum(candidate)
        if cs != target_sum:
            if cs > target_sum:
                j = np.random.choice(np.where(candidate == 1)[0])
                candidate[j] = 0
            else:
                j = np.random.choice(np.where(candidate == 0)[0])
                candidate[j] = 1
        new_c5 = compute_c5(candidate, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate.copy()
        else:
            if np.random.rand() < np.exp(-(new_c5 - best_c5)/T):
                best_h = candidate.copy()
                best_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)
