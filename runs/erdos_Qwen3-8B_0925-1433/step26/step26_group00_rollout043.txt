import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 80
        h = np.zeros(n_points)
        n_blocks = 8
        size = n_points // (2 * n_blocks)
        for _ in range(n_blocks):
            start = np.random.randint(0, n_points - size)
            h[start:start+size] = 1.0
        h = np.clip(h, 0, 1)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros_like(h)
        for j in range(n_points):
            j_plus_k = j + k
            if j_plus_k < n_points:
                grad[j] += (1 - h[j_plus_k]) * dx
            j_minus_k = j - k
            if j_minus_k >= 0:
                grad[j] -= h[j_minus_k] * dx
        return grad
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = compute_gradient(h, lag, n_points, dx)
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if np.abs(np.sum(h) - target_sum) > 1e-3:
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
