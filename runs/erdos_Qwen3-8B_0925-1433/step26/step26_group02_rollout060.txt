import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h_ref = np.zeros(n_points)
        h_ref[:n_points//2] = 1.0
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    n_points = len(h_ref)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k):
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            h_perturb = clip_rescale(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            val = corr_perturb[k] * dx
            h_perturb = h.copy()
            h_perturb[i] -= eps
            h_perturb = clip_rescale(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            val -= corr_perturb[k] * dx
            grad[i] = val / (2 * eps)
        return grad

    start_time = time.time()
    h = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        grad = compute_gradient(h, k)
        h = h - 0.01 * grad
        h = clip_rescale(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        temp = 1.0
        for _ in range(100):
            if temp < 1e-5:
                break
            h_perturb = h + np.random.normal(0, 0.1, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            c5_perturb = compute_c5(h_perturb)
            delta = c5_perturb - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):

            temp *= 0.99
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

                      bounds=[(0, 1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-6, options={'maxiter': 100})
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x.copy()
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
