import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    corr = np.correlate(h_ref, 1 - h_ref, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h_ref.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = min(best_c5, 0.381)

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1.0
            h_plus = h + eps * e
            h_plus = clip_rescale(h_plus)
            h_minus = h - eps * e
            h_minus = clip_rescale(h_minus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            obj_plus = corr_plus[max_idx] * dx
            obj_minus = corr_minus[max_idx] * dx
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    start = time.time()

    def generate_guesses():
        phi = np.random.rand()
        guess1 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1) + phi))
        guess1 = np.clip(guess1, 0, 1)
        guess1 = guess1 * target_sum / np.sum(guess1)
        return [guess1]

    guesses = generate_guesses()
    for guess in guesses:
        if time.time() - start > budget_s:
            break
        h = guess.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        for _ in range(50):
            if time.time() - start > budget_s:
                break
            grad = compute_gradient(h)
            h = h - 0.01 * grad
            h = clip_rescale(h)
            c5 = compute_c5(h)
            if c5 < best_c5:

                best_h = h.copy()
    for _ in range(5):
        if time.time() - start > budget_s:

        if result.success:

            best_c5 = result.fun

    return (best_h, best_c5, n_points)
