import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):

    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * (n_points / 2.0) / np.sum(h)

            h[i] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def get_corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = get_corr(best_h)
    best_c5 = min(best_c5, 0.381)

    def generate_random_blocks(n):
        h = np.zeros(n)
        n_blocks = np.random.randint(4, 8)
        block_sizes = np.random.randint(1, 5, size=n_blocks)
        total = np.sum(block_sizes)
        if total > n:
            block_sizes = block_sizes * (n // total)
        start = 0
        for size in block_sizes:
            end = start + size
            if end > n:
                end = n
            h[start:end] = 1.0
            start = end
        return h * (n_points / 2.0) / np.sum(h)

    initial_guesses = [generate_random_blocks(n_points) for _ in range(3)]

                                    popsize=20, maxiter=50, tol=1e-6, mutation=(0.8, 1.05))
        h_candidate = res.x.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2.0) / np.sum(h_candidate)
        current_c5 = get_corr(h_candidate)
        if current_c5 < best_c5:

            best_c5 = current_c5

    bounds = [(0, 1)] * n_points
    res_slsqp = minimize(objective, best_h, method='SLSQP',

    h_candidate = h_candidate * (n_points / 2.0) / np.sum(h_candidate)
    best_c5 = min(best_c5, get_corr(h_candidate))

    return (h_candidate, best_c5, n_points)
