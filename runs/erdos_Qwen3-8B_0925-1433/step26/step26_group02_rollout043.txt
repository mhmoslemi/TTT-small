import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        dx = 2.0 / n_points
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k] * dx
        return grad

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        h = best_h.copy()
        h = clip_rescale(h)
        grad = compute_gradient(h)
        h = h - 0.1 * grad
        h = clip_rescale(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(compute_c5, best_h, method='L-BFGS-B',
                          bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
