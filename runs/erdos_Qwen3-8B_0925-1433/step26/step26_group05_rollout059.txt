import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    population = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    for _ in range(19):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)
    population = np.array(population)
    velocities = np.random.uniform(-0.2, 0.2, size=(len(population), n_points))
    pbest = population.copy()
    gbest = population[0].copy()
    fitness = np.array([compute_c5(p, dx) for p in population])
    best_c5 = np.min(fitness)
    start_time = time.time()
    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * 0.8 + 1.5 * r1 * (pbest[i] - population[i]) + 1.5 * r2 * (gbest - population[i])
            vel = np.clip(vel, -0.2, 0.2)
            velocities[i] = vel

            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * target_sum / np.sum(new_pos)
            population[i] = new_pos
        fitness = np.array([compute_c5(p, dx) for p in population])
        for i in range(len(population)):
            if fitness[i] < compute_c5(pbest[i], dx):
                pbest[i] = population[i].copy()
        min_index = np.argmin(fitness)
        if compute_c5(gbest, dx) > fitness[min_index]:
            gbest = population[min_index].copy()
        best_c5 = np.min(fitness)
    best_h = gbest
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.min(corr) * dx
