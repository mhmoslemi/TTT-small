import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    max_iter = 100000
    iter_count = 0
    while time.time() - start_time < budget_s:
        if iter_count >= max_iter:
            break
        iter_count += 1
        h_neighbor = h.copy()
        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.01, 0.01)
        h_neighbor[i] += delta
        h_neighbor[i] = np.clip(h_neighbor[i], 0, 1)
        current_sum = np.sum(h_neighbor)
        delta_sum = target_sum - current_sum
        if abs(delta_sum) > 1e-6:
            j = np.random.randint(n_points)
            while j == i:
                j = np.random.randint(n_points)
            h_neighbor[j] += delta_sum

        new_c5 = compute_c5(h_neighbor, dx)
        if new_c5 < best_c5:

        else:
            prob = np.exp((best_c5 - new_c5) / T)
            if np.random.rand() < prob:
                h = h_neighbor.copy()
                best_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
