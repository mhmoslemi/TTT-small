            starts[i] = starts[i-1] + best_widths[i-1]
        ends = starts + best_widths
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for j in range(m):
                if starts[j] <= x <= ends[j]:
                    h[i] = 1.0
                    break
        best_c5 = objective(best_widths)
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res_l_bfgs = minimize(objective, best_widths, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
        if res_l_bfgs.success:
            best_widths = res_l_bfgs.x
            m = len(best_widths)
            starts = np.zeros(m)
            starts[0] = 0.0
            for i in range(1, m):
                starts[i] = starts[i-1] + best_widths[i-1]
            ends = starts + best_widths
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * dx
                for j in range(m):
                    if starts[j] <= x <= ends[j]:
                        h[i] = 1.0
                        break
            best_c5 = objective(best_widths)
    
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
