import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    bounds = [(0, 1)] * n_points

    def objective(x):
        x = clip_rescale(x)
        return compute_c5(x)

    def constraints(x):
        return np.sum(x) - target_sum

    start_time = time.time()

    def generate_initial_population(size=20):
        pop = []
        for _ in range(size):
            if np.random.rand() < 0.5:
                phi = np.random.rand()
                guess = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points-1) + phi))
                guess = np.clip(guess, 0, 1)
                guess = guess * target_sum / np.sum(guess)
            else:
                guess = np.zeros(n_points)
                indices = np.random.choice(n_points, size=n_points // 5, replace=False)
                guess[indices] = 1.0
                guess = np.clip(guess, 0, 1)
                guess = guess * target_sum / np.sum(guess)
            pop.append(guess)
        return pop

    population = generate_initial_population(20)
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            mutant = clip_rescale(mutant)
            if np.random.rand() < 0.7:
                mutant = np.array([np.random.choice([population[i][j], mutant[j]]) for j in range(n_points)])
            new_population.append(mutant)
        population += new_population
        population = np.array([clip_rescale(p) for p in population])
        population = np.array([p for p in population if np.sum(p) > target_sum - 1e-6 and np.sum(p) < target_sum + 1e-6])
        population = np.array([p for p in population if np.all(0 <= p <= 1)])
        population = np.array([p for p in population if np.sum(p) < target_sum + 1e-6 and np.sum(p) > target_sum - 1e-6])
        population = population[:20]
        for p in population:
            c5 = compute_c5(p)
            if c5 < best_c5:

                best_h = p.copy()

    result = minimize(compute_c5, best_h, method='L-BFGS-B',
                      bounds=[(0, 1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],

    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
