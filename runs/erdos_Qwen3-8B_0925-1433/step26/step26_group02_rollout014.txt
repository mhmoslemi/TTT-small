import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    result_de = differential_evolution(compute_c5, bounds, 
                                      strategy='best1bin', 
                                      popsize=20, mutation=(0.5, 1),
                                      recombination=0.7, 
                                      tol=1e-6, maxiter=50,
                                      seed=seed)
    
    best_h_de = rescale(result_de.x)
    best_c5_de = compute_c5(best_h_de)
    best_c5 = min(best_c5, best_c5_de)
    best_h = best_h_de
    
    result_lbgfs = minimize(compute_c5, best_h, method='L-BFGS-B',
                            bounds=[(0, 1)]*n_points,
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            tol=1e-6, options={'maxiter': 200})
    
    best_h = rescale(result_lbgfs.x)
    best_c5 = result_lbgfs.fun
    best_c5 = min(best_c5, compute_c5(h_initial))
    
    return (best_h, best_c5, n_points)
