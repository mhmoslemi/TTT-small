import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    
    # Define the convex optimization problem
    def optimize_convex():
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        for k in range(-n_points + 1, n_points):
            corr = cp.sum(h_var * (1 - h_var[(np.arange(n_points) + k) % n_points]))
            constraints.append(corr <= t)
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                return h_var.value
        except:
            pass
        return h
    
    # Fallback gradient descent with corrected gradient
    def gradient_descent(h):
        best_h = h.copy()
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        for _ in range(1000):
            corr = np.correlate(h, 1 - h, mode='full')
            k_idx = np.argmax(corr)
            shift = k_idx - (n_points - 1)
            index1 = (np.arange(n_points) + shift) % n_points
            index2 = (np.arange(n_points) - shift) % n_points
            gradient = (1 - h[index1]) - h[index2]
            gradient *= dx
            h -= 0.1 * gradient
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        return best_h, best_c5
    
    # Run convex optimization
    h_opt = optimize_convex()
    if h_opt is not None:
        h = h_opt
    else:
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
    
    # Run gradient descent with multiple restarts
    h, best_c5 = gradient_descent(h)
