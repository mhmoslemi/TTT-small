import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_candidates = [np.array(initial_h_values, dtype=float)]
    
    # Generate 4 more initial candidates with random perturbations
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        initial_candidates.append(h)
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    max_iterations = 1200
    learning_rate = 0.05
    
    time_start = time.time()
    for candidate in initial_candidates:
        h = candidate.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        for t in range(max_iterations):
            if time.time() - time_start >= budget_s:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k_idx = np.argmax(corr)
            gradient = 1.0 - h[(np.arange(n_points) + k_idx) % n_points]
            # Adam update with bias correction
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * (gradient ** 2)
            m_hat = m / (1 - beta1 ** (t + 1))
            v_hat = v / (1 - beta2 ** (t + 1))
            h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
