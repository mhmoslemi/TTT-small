import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(39):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    best_h = population[0].copy()
    best_c5 = compute_c5(population[0], dx)
    start_time = time.time()
    POP_SIZE = 40
    MAX_GEN = 1500
    MUTATION_STD = 0.05
    TOURNAMENT_SIZE = 5

    for gen in range(MAX_GEN):
        if time.time() - start_time > budget_s:
            break
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(np.max(corr) * dx)
        new_pop = []
        for _ in range(POP_SIZE // 2):
            p1_idx = min(np.random.choice(len(population), TOURNAMENT_SIZE, replace=False),
                         key=lambda i: fitness[i])
            p2_idx = min(np.random.choice(len(population), TOURNAMENT_SIZE, replace=False),
                         key=lambda i: fitness[i])
            p1, p2 = population[p1_idx], population[p2_idx]
            split = np.random.randint(1, n_points - 1)
            c1, c2 = np.concatenate((p1[:split], p2[split:])), np.concatenate((p2[:split], p1[split:]))
            c1 += np.random.normal(0, MUTATION_STD, n_points)
            c1 = np.clip(c1, 0, 1)
            delta = (target_sum - np.sum(c1)) / n_points
            c1 += delta
            c1 = np.clip(c1, 0, 1)
            c2 += np.random.normal(0, MUTATION_STD, n_points)
            c2 = np.clip(c2, 0, 1)
            delta = (target_sum - np.sum(c2)) / n_points
            c2 += delta
            c2 = np.clip(c2, 0, 1)
            new_pop.extend([c1, c2])
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx[:POP_SIZE//2]] + new_pop
        for h in population:
            cur_c5 = compute_c5(h, dx)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()

    return (best_h, best_c5, n_points)
