import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    
    learning_rate = 0.1
    decay_rate = 0.998
    iter_count = 0
    time_start = time.time()
    restarts = 0
    
    while time.time() - time_start < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n_points - 1)
        gradient = 1.0 - h[(np.arange(n_points) + shift) % n_points]
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        iter_count += 1
        learning_rate *= decay_rate
        
        if iter_count % 1000 == 0:
            h_perturb = h + np.random.normal(0, 0.1, size=n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            h = h_perturb.copy()
            restarts += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
