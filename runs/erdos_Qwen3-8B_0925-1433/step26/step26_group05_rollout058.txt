import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 20
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []

    # Initialize population
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    particles.append(initial_h)
    velocities.append(np.random.uniform(-0.2, 0.2, n_points))
    pbest.append(initial_h.copy())
    pbest_fitness.append(compute_c5(initial_h, dx))

    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.2, 0.2, n_points))
        pbest.append(h.copy())
        pbest_fitness.append(compute_c5(h, dx))

    gbest = pbest[0].copy()
    gbest_fitness = pbest_fitness[0]

    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 2000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Update pbest and gbest
        for i in range(pop_size):
            fit = compute_c5(particles[i], dx)
            if fit < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fit
            if fit < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = fit

        # Update velocities and positions
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i]
            vel += omega * vel
            vel += c1 * r1 * (pbest[i] - particles[i])
            vel += c2 * r2 * (gbest - particles[i])
            vel = np.clip(vel, -0.2, 0.2)
            velocities[i] = vel

            particles[i] += vel
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h == 0:
                particles[i] = np.zeros_like(particles[i]) + 0.5
            else:
                particles[i] = particles[i] * target_sum / sum_h

    return (gbest, gbest_fitness, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
