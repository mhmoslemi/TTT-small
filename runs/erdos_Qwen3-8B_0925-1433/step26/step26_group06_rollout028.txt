        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    def generate_block(n):
        h = np.zeros(n)
        spacing = n // 3
        for i in range(0, n, spacing):
            h[i:i+2] = 1.0
        return h
    
    def generate_spline(n, peak=3):
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % peak == 0) else 0.0
        return h
    
    initial_guesses = [generate_block(n_points), generate_spline(n_points)]
    
    start_time = time.time()
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=Bounds(0, 1), tol=1e-6, options={'maxiter': 150})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=Bounds(0, 1), tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
