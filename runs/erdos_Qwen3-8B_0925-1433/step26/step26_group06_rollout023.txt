            if 0 <= idx < m:
                h_vals[i] = 1.0
        return h_vals
    
    def objective(w):
        h_vals = generate_h(w)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 2}]
    bounds = [(1e-6, 2) for _ in range(m)]
    
    result = minimize(objective, initial_widths, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    
    best_h = generate_h(result.x) if result.success else h
    best_c5 = objective(result.x) if result.success else objective(initial_widths)
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * dx
    if abs(total - 1) > 1e-6:
        best_h = h
    
    return (best_h, best_c5, n_points)
