            else:
                new_population.append(population[i])
        population = new_population
        for candidate in population:
            c5_candidate = compute_c5(candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(lambda x: compute_c5(x), best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 200})
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
