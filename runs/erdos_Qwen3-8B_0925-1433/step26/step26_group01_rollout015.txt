                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child > 0:
                child = child * target_sum / sum_child
            offspring.append(child)
        
        new_population = parents + offspring
        new_fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in new_population]
        combined = list(zip(new_fitness, new_population))
        combined.sort()
        population = [h for (f, h) in combined[:len(population)]]
        
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[sorted_indices[0]]
        
        if abs(best_c5 - previous_best) < 1e-7:
            early_stop_counter += 1
            if early_stop_counter >= early_stop:
                break
        else:
            early_stop_counter = 0
        previous_best = best_c5

    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * target_sum / final_sum
    c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
    h_values = final_h.copy()
    return (h_values, c5, n_points)
