import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    start_time = time.time()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        c5 = compute_c5(h)
        penalty = 1e6 * np.abs(np.sum(h) - target_sum)
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Differential Evolution
    try:
        result = optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            polish=True,
            x0=h_initial,
            seed=seed,
            workers=1
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    except:
        best_h = h_initial
        best_c5 = compute_c5(best_h)
    
    # L-BFGS-B Refinement
    try:
        result_lbgfgs = optimize.minimize(
            compute_c5,
            best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        best_h = result_lbgfgs.x
        best_c5 = compute_c5(best_h)
    except:
        pass
    
    # Final rescale
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
