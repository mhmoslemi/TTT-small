import numpy as np

def run(seed=42, budget_s=600):
    import time
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    best_h = h.copy()
    
    def compute_gradient(m):
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + m
            if 0 <= j < n_points:
                grad[i] = (1.0 - h[j])
        return grad
    
    start_time = time.time()
    step_size = 0.1
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        m = max_idx - (n_points - 1)
        grad = compute_gradient(m)
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = np.max(corr) * dx
        if new_c5 >= best_c5:
            step_size *= 0.9
        else:
            best_c5 = new_c5
            best_h = h.copy()
        if iterations > 1000:
            break
    return (best_h, best_c5, n_points)
