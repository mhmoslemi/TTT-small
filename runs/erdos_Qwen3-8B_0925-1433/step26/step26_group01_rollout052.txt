def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.03
    decay = 0.999
    momentum = 0.9
    perturb_prob = 0.1
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = full_corr.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        top_indices = np.argsort(full_corr)[::-1][:8]
        grads = np.zeros(n_points)
        for idx in top_indices:
            k = idx - (n_points - 1)
            i_plus_k = np.arange(n_points) + k
            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            term1 = np.zeros(n_points)
            term1[valid_plus] = (1 - h[i_plus_k[valid_plus]]) * dx
