        aggregate_grad = np.zeros(n_points)
        for idx in top_indices:
            k = idx - (n_points - 1)
            i_plus_k = np.arange(n_points) + k
            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            term1 = (1 - h[i_plus_k[valid_plus]])
