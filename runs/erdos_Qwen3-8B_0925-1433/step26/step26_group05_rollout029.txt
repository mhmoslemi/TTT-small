import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    POP_SIZE = 40
    MAX_GENERATIONS = 1500
    MUTATION_STD = 0.05
    TOURNAMENT_SIZE = 5
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = compute_c5(best_h, dx)
    best_c5 = min(best_c5, 0.381)
    population = [best_h]
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    start_time = time.time()
    for gen in range(MAX_GENERATIONS):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h, dx) for h in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        parents = []
        for _ in range(2):
            tournament = np.random.choice(range(POP_SIZE), TOURNAMENT_SIZE, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            parents.append(population[winner])
        split = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parents[0][:split], parents[1][split:]])
        child2 = np.concatenate([parents[1][:split], parents[0][split:]])
        for child in [child1, child2]:
            noise = np.random.normal(0, MUTATION_STD, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            delta = (target_sum - current_sum) / n_points
            child += delta
            child = np.clip(child, 0, 1)
        idxs = np.argsort(fitness)[-2:]
        population[idxs[0]] = child1
        population[idxs[1]] = child2
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
