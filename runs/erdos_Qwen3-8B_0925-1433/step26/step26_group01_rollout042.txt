import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.05
    decay = 0.999
    momentum = 0.9
    perturb_prob = 0.1
    
    velocity = np.zeros(n)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        lags = np.argsort(full_corr)[-5:]
        grad = np.zeros(n)
        
        for k in lags:
            i_plus = np.arange(n) + k
            valid_plus = (i_plus >= 0) & (i_plus < n)
            term1 = np.zeros(n)
            term1[valid_plus] = (1 - h[i_plus[valid_plus]]) * dx
            i_minus = np.arange(n) - k
            valid_minus = (i_minus >= 0) & (i_minus < n)
            term2 = np.zeros(n)
            term2[valid_minus] = -h[i_minus[valid_minus]] * dx
            grad += term1 + term2
        
        velocity = momentum * velocity + grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.05, n)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = np.max(full_corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n)
