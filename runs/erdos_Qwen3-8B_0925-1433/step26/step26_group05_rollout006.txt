            grad[i] = (1 - h[j_plus]) - h[j_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = 0.5 * np.ones(n_points)
        else:
            h = h * target_sum / sum_h
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if _ % 100 == 0:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        learning_rate *= 0.9995

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    max_overlap = 0.0
    for k in range(len(h)):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap
