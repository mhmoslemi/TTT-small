    while time.time() - start_time < budget_s:
        generations += 1
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:

            c5 = np.max(corr) * dx
            fitness.append(c5)

            # Mutate
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
