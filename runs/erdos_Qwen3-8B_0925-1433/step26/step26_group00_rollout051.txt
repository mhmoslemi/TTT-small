import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 80
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        block_size = n_points // 8
        for _ in range(8):
            start = np.random.randint(0, n_points - block_size)
            h[start:start+block_size] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        m = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + m < n_points:
                term1 = 1 - h[i + m]
            else:
                term1 = 0.0
            if i - m >= 0:
                term2 = h[i - m]
            else:
                term2 = 0.0
            grad[i] = (term1 - term2) * dx
        h = h - 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                   x0=best_h,
                   method='L-BFGS-B',
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': 100})
    if res.success:
        best_h = res.x.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
