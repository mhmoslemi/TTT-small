import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    POP_SIZE = 40
    MAX_GEN = 1500
    MUT_STD = 0.05
    TOURN_SIZE = 5
    
    def init_pop():
        pop = [np.array(initial_h_values, dtype=float)]
        for _ in range(POP_SIZE - 1):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
            pop.append(h)
        return np.array([h * target_sum / np.sum(h) for h in pop])
    
    pop = init_pop()
    best_h = pop[0]
    best_c5 = compute_c5(best_h, dx)
    
    def select_parents():
        indices = np.random.choice(len(pop), size=TOURN_SIZE, replace=False)
        return pop[indices[np.argmin([compute_c5(pop[i], dx) for i in indices])]]
    
    for gen in range(MAX_GEN):
        if time.time() - time.time() > budget_s:
            break

            child2 = np.concatenate([p2[:split], p1[split:]])

                child += noise
                child = np.clip(child, 0, 1)

            new_pop.extend([child1, child2])

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
