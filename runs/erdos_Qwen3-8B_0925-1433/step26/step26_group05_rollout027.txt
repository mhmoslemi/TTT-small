import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    cogn_coeff = 1.5
    soc_coeff = 1.5
    vel_limits = (-0.2, 0.2)
    max_iter = 2000

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)
    vel = np.random.uniform(*vel_limits, size=(pop_size, n_points))

    pbest = population.copy()
    pbest_vals = np.array([compute_c5(p) for p in population])
    gbest = population[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    start_time = time.time()

        if time.time() - start_time > budget_s:
