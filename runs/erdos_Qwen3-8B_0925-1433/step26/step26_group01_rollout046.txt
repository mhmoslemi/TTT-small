import numpy as np

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    max_gen = int(budget_s / 30)
    for gen in range(max_gen):
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            fitness.append(np.max(corr))
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx].copy()
        
        parents = np.argsort(fitness)[:int(population_size * 0.2)]
        offspring = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            alpha = np.random.uniform(0.2, 0.8)
