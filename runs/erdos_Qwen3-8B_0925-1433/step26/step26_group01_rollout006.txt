def run(seed=42, budget_s=600):

    else:

    population_size = 100
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        population.append(perturbed)
    
    max_generations = int(budget_s / 30)
    best_h = initial_h.copy()
    best_c5 = float('inf')
    
    for gen in range(max_generations):
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append((c5, h))
        fitness_sorted = sorted(fitness, key=lambda x: x[0])
        
        parents = [h for (c5, h) in fitness_sorted[:int(0.2 * len(population))]]
        num_offspring = len(population) // 2
        offspring = []
