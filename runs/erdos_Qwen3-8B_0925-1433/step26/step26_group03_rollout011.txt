import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    learning_rate = 0.2
    decay_rate = 0.999
    max_iter = 300
    time_start = time.time()

    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx

        indices = (np.arange(n_points) + k) % n_points
        gradient = 1.0 - best_h[indices]
        best_h -= learning_rate * gradient
        best_h = clip_rescale(best_h)

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
        elif abs(current_c5 - best_c5) < 1e-6:
            break
        learning_rate *= decay_rate

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
