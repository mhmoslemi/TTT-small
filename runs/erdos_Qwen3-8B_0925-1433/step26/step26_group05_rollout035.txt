        h = h * target_sum / np.sum(h)
        particles.append(h)
    velocities = [np.random.uniform(-0.2, 0.2, n_points) for _ in range(population_size)]
    
    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    personal_best = [p.copy() for p in particles]
    global_best = personal_best[0].copy()
    best_c5 = compute_fitness(global_best)
    
    start_time = time.time()
    max_iterations = 2000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_fitness(p) for p in particles]
        for i in range(population_size):
            if fitness[i] < compute_fitness(personal_best[i]):
                personal_best[i] = particles[i].copy()
        current_global = personal_best[0].copy()
        for p in personal_best:
            if compute_fitness(p) < compute_fitness(current_global):
                current_global = p.copy()
        if compute_fitness(current_global) < best_c5:
            best_c5 = compute_fitness(current_global)
            global_best = current_global.copy()
        for i in range(population_size):
            h = particles[i]
            v = velocities[i]
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v_new = 0.8 * v + 1.5 * r1 * (personal_best[i] - h) + 1.5 * r2 * (global_best - h)
            v_new = np.clip(v_new, -0.2, 0.2)
            velocities[i] = v_new
            h_new = h + v_new
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            particles[i] = h_new
    return (global_best, best_c5, n_points)
