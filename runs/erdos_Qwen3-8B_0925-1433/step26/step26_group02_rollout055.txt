import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    initial_guesses = []
    initial_guesses.append(np.array(initial_h_values, dtype=float))

    guess_sine = np.clip(guess_sine, 0, 1)
    guess_sine = guess_sine * target_sum / np.sum(guess_sine)
    initial_guesses.append(guess_sine)
    block_size = n_points // 4
    guess_block = np.zeros(n_points)
    for i in range(n_points):

    guess_block = np.roll(guess_block, np.random.randint(0, block_size))
    guess_block = np.clip(guess_block, 0, 1)
    guess_block = guess_block * target_sum / np.sum(guess_block)
    initial_guesses.append(guess_block)
    guess_sparse = np.zeros(n_points)

    def objective(x):

    start_time = time.time()
