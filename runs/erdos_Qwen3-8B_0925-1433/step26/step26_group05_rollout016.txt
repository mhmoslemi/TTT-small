import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)
    lr = 0.1
    max_iter = 10000
    K = 5
    decay = 0.9995
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        overlaps = np.array([compute_c5(h, dx) for _ in range(n_points)])
        top_k = np.argsort(overlaps)[-K:].astype(int)
        grad = np.zeros(n_points)
        for k in top_k:
            for i in range(n_points):
                j_p = (i + k) % n_points
                j_m = (i - k) % n_points
                grad[i] += (1 - h[j_p]) - h[j_m]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h < 1e-6:
            h = 0.5 * np.ones(n_points)
        else:
            h = h * target_sum / sum_h
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if t % 100 == 0:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        lr *= decay
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
