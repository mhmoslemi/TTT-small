import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    candidates = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
    
    # Generate diversified initializations
    perturb = np.random.normal(0, 0.05, n_points)
    h_pert = np.clip(np.array(initial_h_values) + perturb, 0, 1)
    h_pert = h_pert * target_sum / np.sum(h_pert)
    candidates.append(h_pert)
    
    for _ in range(2):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        candidates.append(h_rand)
    
    best_h = np.copy(candidates[0])
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    time_start = time.time()
    
    for h in candidates:
        m, v = np.zeros(n_points), np.zeros(n_points)
        lr = 0.1
        t = 1
        iter_cnt = 0
        while time.time() - time_start < budget_s and iter_cnt < 10000:
            corr = np.correlate(h, 1 - h, mode='full')
            k_idx = np.argmax(corr)
            shift = k_idx - (n_points - 1)
            grad = 1.0 - h[(np.arange(n_points) + shift) % n_points]
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            h -= lr * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            t += 1
            iter_cnt += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
