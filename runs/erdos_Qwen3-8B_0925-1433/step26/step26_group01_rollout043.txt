def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)

    decay = 0.995
    momentum = 0.9
    perturb_prob = 0.1
    L = 5  # number of top lags to consider
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        # Get top L lags
        top_indices = np.argsort(-full_corr)[:L]
        grads = np.zeros_like(h)
        for idx in top_indices:
            k = idx - (n_points - 1)
            i_plus_k = (np.arange(n_points) + k) % n_points
            i_minus_k = (np.arange(n_points) - k) % n_points
            grads_k = (1 - h[i_plus_k]) - h[i_minus_k]
            grads += grads_k * dx  # Apply dx scaling correction
        
        # Gradient descent update

        alpha = learning_rate * (1 - decay * t / max_iter)
