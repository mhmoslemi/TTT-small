    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    L = 5  # number of top lags to consider
    learning_rate = 0.05
    decay = 0.995
    momentum = 0.9
    perturb_prob = 0.1
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        indices = np.argsort(full_corr)[-L:]
        selected_lags = [m - (n_points - 1) for m in indices]
        aggregate_grad = np.zeros(n_points)
        for k in selected_lags:
            j = np.arange(n_points) + k
            valid = (j >= 0) & (j < n_points)
            grad = np.zeros(n_points)
            grad[valid] = 1 - h[j[valid]]
            aggregate_grad += grad
        velocity = momentum * velocity - learning_rate * aggregate_grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        current_c5 = full_corr.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
