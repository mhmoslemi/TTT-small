import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    POP_SIZE = 40
    MAX_GENERATIONS = 1500
    TOURNAMENT_SIZE = 5
    MUTATION_STD = 0.05
    
    # Initialize population
    population = []
    # Use initial_h_values as the first candidate
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    population.append(h_initial)
    # Generate remaining candidates
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for gen in range(MAX_GENERATIONS):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        # Track best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        # Selection and crossover
        new_population = []
        for _ in range(POP_SIZE // 2):
            # Tournament selection for two parents
            tournament = np.random.randint(0, POP_SIZE, size=TOURNAMENT_SIZE)
            parent1 = population[np.argmin([fitness[i] for i in tournament])]
            parent2 = population[np.argmin([fitness[i] for i in tournament])]
            while parent1 is parent2:
                parent2 = population[np.argmin([fitness[i] for i in tournament])]
            # Crossover
            split = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
            # Mutation

                h_mut = h + noise

                delta = (target_sum - np.sum(h_mut)) / n_points

            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
