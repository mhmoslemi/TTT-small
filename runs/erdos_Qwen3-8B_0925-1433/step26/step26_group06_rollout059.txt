        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i + size] = 1.0
        return h
    
    def generate_random_blocks(n, block_size_range=5):
        h = np.zeros(n)
        n_blocks = np.random.randint(4, 8)
        block_sizes = np.random.randint(1, block_size_range, size=n_blocks)
        total = np.sum(block_sizes)
        if total > n:
            block_sizes = block_sizes * (n // total)
        start = 0
        for size in block_sizes:
            end = start + size
            if end > n:
                end = n
            h[start:end] = 1.0
            start = end
        return h
    
    initial_guesses = []
    for phase in [0, 1, 2]:
        guess = generate_evenly_spaced(n_points, phase)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points))
    initial_guesses.append(generate_random_blocks(n_points))
    
    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res_de = differential_evolution(objective_de, [(0, 1)] * n_points, 
                                       popsize=30, maxiter=100, tol=1e-5)
        h_candidate = res_de.x.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(objective, best_h, method='SLSQP',
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 200})
        if res_slsqp.success:
            h_candidate = res_slsqp.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B',
                             bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res_l_bfgs.success:
            h_candidate = res_l_bfgs.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
