import numpy as np
import time

def run(seed=42, budget_s=600):

    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    POP_SIZE = 40
    population = [initial_h.copy()]
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h *= target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    for gen in range(1500):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.zeros(POP_SIZE)
        for i in range(POP_SIZE):
            fitness[i] = compute_c5(population[i], dx)
        
        # Tournament selection
        parents = []
        for _ in range(2):
            tournament = np.random.choice(POP_SIZE, size=5, replace=False)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner].copy())
        
        # Crossover
        split = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parents[0][:split], parents[1][split:]])
        child2 = np.concatenate([parents[1][:split], parents[0][split:]])
        
        # Mutation
        def mutate(child):
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            delta = (target_sum - np.sum(child)) / n_points
            child += delta
            child = np.clip(child, 0, 1)
            return child
        
        child1 = mutate(child1)
        child2 = mutate(child2)
        
        # Replace worst
        worst_idx = np.argmax(fitness)

        worst_idx = np.argmax(fitness)

        # Update best

            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = population[i].copy()

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
