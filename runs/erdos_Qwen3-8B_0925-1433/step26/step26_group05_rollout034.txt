import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    pop_size = 20
    particles = [initial_h.copy() for _ in range(pop_size)]
    velocities = np.random.uniform(-0.2, 0.2, (pop_size, n_points))
    pbest = particles.copy()
    pbest_fitness = np.full(pop_size, np.inf)
    gbest = initial_h.copy()
    gbest_fitness = np.inf
    
    def calc_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for _ in range(2000):
        if time.time() - time.time() > budget_s:
            break
        for i in range(pop_size):
            h = particles[i]
            fitness = calc_fitness(h)
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = h.copy()
                gbest_fitness = fitness
        
        for i in range(pop_size):
            h = particles[i]
            pbest_h = pbest[i]
            gbest_h = gbest
            r1, r2 = np.random.rand(2)
            vel = velocities[i] * 0.8 + 1.5 * r1 * (pbest_h - h) + 1.5 * r2 * (gbest_h - h)
            np.clip(vel, -0.2, 0.2, out=vel)
            velocities[i] = vel
            new_h = h + vel
            np.clip(new_h, 0, 1, out=new_h)
            new_h = new_h * target_sum / np.sum(new_h)
            particles[i] = new_h
    
    best_h = gbest
    best_c5 = gbest_fitness
    return (best_h, best_c5, n_points)
