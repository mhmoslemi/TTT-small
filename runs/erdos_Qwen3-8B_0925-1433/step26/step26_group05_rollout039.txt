import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    learning_rate = 0.1
    max_iterations = 10000
    decay_rate = 0.9995

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        max_overlap = 0.0
        k_max = 0
        for k in range(n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                k_max = k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k_max) % n_points
            grad[i] = 1.0 - h[j_plus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
