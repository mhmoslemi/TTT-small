        v_hat = v / (1 - beta2 ** t)
        
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if np.abs(new_c5 - best_c5) < 1e-5:
            break
        learning_rate *= 0.999
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
