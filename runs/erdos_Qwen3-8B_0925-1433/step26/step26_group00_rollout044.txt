import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.resize(h, n_points)
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
        block_size = 5
        n_blocks = 8
        for i in range(n_blocks):
            start = np.random.randint(0, n_points - block_size)
            h[start:start+block_size] = 1.0
        h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    start_time = time.time()
    lambda_penalty = 0.1

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        overlaps = np.correlate(h, 1 - h, mode='full')
        current_max = overlaps.max() * dx
        violation = abs(np.sum(h) - target_sum)

        if violation > 1e-3:
            lambda_penalty *= 1.1
        else:
            lambda_penalty *= 0.99

        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            overlaps_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (overlaps_plus.max() * dx - current_max) / 1e-4

        h = h - 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if np.correlate(h, 1 - h, mode='full').max() * dx < best_c5:
            best_h = h.copy()
            best_c5 = np.correlate(h, 1 - h, mode='full').max() * dx

        if _ % 50 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        if _ % 100 == 0:
            def obj(x):
                x = np.clip(x, 0, 1)
                x = x * target_sum / np.sum(x)
                return np.correlate(x, 1 - x, mode='full').max() * dx
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter': 100})
            if res.success:
                h = res.x.copy()
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
                current_c5 = obj(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5

    return (best_h, best_c5, n_points)
