from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_gradient(h):
        eps = 1e-5
        grad = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = clip_rescale(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = corr_plus[k_star] * dx

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = clip_rescale(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            val_minus = corr_minus[k_star] * dx

            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad

    start_time = time.time()

    def generate_guesses():
        n = n_points
        phi = np.random.rand()
        guess1 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n-1) + phi))
        guess1 = np.clip(guess1, 0, 1)
        guess1 = guess1 * target_sum / np.sum(guess1)
        block_size = n // 4
