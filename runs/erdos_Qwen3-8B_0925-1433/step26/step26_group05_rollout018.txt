import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 20
    particles = np.zeros((population_size, n_points))
    velocities = np.zeros((population_size, n_points))
    for i in range(population_size):
        if i == 0:
            h = np.array(initial_h_values, dtype=float)
            h = h * target_sum / np.sum(h)
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        particles[i] = h
        velocities[i] = np.random.uniform(-0.2, 0.2, n_points)

    pbest = particles.copy()
    pbest_fitness = np.zeros(population_size)
    for i in range(population_size):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        pbest_fitness[i] = np.max(corr) * dx

    gbest = pbest[0].copy()
    gbest_fitness = pbest_fitness[0]
    start_time = time.time()
    max_iter = 2000

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        current_fitness = np.zeros(population_size)
        for i in range(population_size):
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_fitness[i] = np.max(corr) * dx

        for i in range(population_size):
            if current_fitness[i] < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness[i]

        min_idx = np.argmin(pbest_fitness)
        gbest = pbest[min_idx].copy()
        gbest_fitness = pbest_fitness[min_idx]

        inertia = 0.8
        c1 = 1.5
        c2 = 1.5
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h == 0:
                particles[i] = 0.5 * np.ones(n_points)
            else:
                particles[i] = particles[i] * target_sum / sum_h

    best_idx = np.argmin(pbest_fitness)
    best_h = pbest[best_idx].copy()
    best_c5 = pbest_fitness[best_idx]
    return (best_h, best_c5, n_points)
