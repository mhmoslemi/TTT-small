import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    # Hyperparameters
    L = 5              # number of top lags to consider
    learning_rate = 0.05
    decay = 0.995
    momentum = 0.9
    perturb_prob = 0.1
    max_iter = int(budget_s / 2)  # adjust based on time estimate per iteration
    early_stop = 100
    
    velocity = np.zeros(n_points)
    start_time = time.time()
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        top_lags = np.argsort(full_corr)[-L:]
        # For each lag in top_lags, compute gradient and sum
        total_grad = np.zeros(n_points)
        for k in top_lags:
            # Compute gradient for lag k
            i_plus_k = np.arange(n_points) + k
            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            term1 = np.zeros(n_points)
            term1[valid_plus] = (1 - h[i_plus_k[valid_plus]]) * dx
            
            i_minus_k = np.arange(n_points) - k
            valid_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
            term2 = np.zeros(n_points)
            term2[valid_minus] = -h[i_minus_k[valid_minus]] * dx
            
            grad_k = term1 + term2
            total_grad += grad_k
        
        velocity = momentum * velocity + total_grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = full_corr.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
