import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points) * 2
        h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def eval_and_clip(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def generate_block(n, block_size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * block_size):
            h[i:i + block_size] = 1.0
        return h

    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h

    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h

    def generate_block_var(n, block_size=5):
        h = np.zeros(n)
        for i in range(0, n, block_size + 1):
            h[i:i + 1] = 1.0
        return h

    initial_guesses = []
    for phase in [0, 1, 2]:
        guess = generate_evenly_spaced(n_points, phase)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points, 6))
    initial_guesses.append(generate_block_var(n_points, 6))

    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for guess in initial_guesses[:3]:
        if time.time() - start_time > budget_s:
            break
        res_de = differential_evolution(
            objective, [(0, 1)] * n_points, 
            popsize=100, maxiter=200, tol=1e-5,
            mutation=(0.5, 1), recombination=0.8
        )
        h_candidate = eval_and_clip(res_de.x)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(
            objective, best_h, method='SLSQP',
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            bounds=[(0, 1)] * n_points, tol=1e-8, options={'maxiter': 1000}
        )
        if res_slsqp.success:
            h_candidate = eval_and_clip(res_slsqp.x)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res_l_bfgs = minimize(
            objective, best_h, method='L-BFGS-B',
            bounds=[(0, 1)] * n_points, tol=1e-8, options={'maxiter': 500}
        )
        if res_l_bfgs.success:
            h_candidate = eval_and_clip(res_l_bfgs.x)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    best_h = eval_and_clip(best_h)
    return (best_h, best_c5, n_points)
