import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:

        for _ in range(8):

            h[start:start+5] = 1.0
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.01
    max_iterations = 200
    lambda_penalty = 0.1

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_lag = np.argmax(corr)
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += epsilon
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            corr_p = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            h_perturb[i] -= 2 * epsilon
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            corr_p2 = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            grad[i] = (np.max(corr_p) - np.max(corr_p2)) / (2 * epsilon)

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if abs(np.sum(h) - target_sum) > 1e-4:

        else:
            lambda_penalty *= 0.9

    def objective(x):
        h_scaled = x * target_sum / np.sum(x)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')

    bounds = [(0, 1)] * n_points

    if res.success:

    best_c5 = np.max(corr) * dx
