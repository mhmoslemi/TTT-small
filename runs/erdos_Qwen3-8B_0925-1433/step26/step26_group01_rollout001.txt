import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.05
    decay = 0.99
    momentum = 0.95
    perturb_prob = 0.2
    perturb_scale = 0.1
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        j_max = np.argmax(corr)
        k = j_max - (n_points - 1)
        
        i_plus_k = np.arange(n_points) + k
        valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
        term1 = (1 - h[i_plus_k[valid_plus]]) * dx
        
        i_minus_k = np.arange(n_points) - k
        valid_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
        term2 = -h[i_minus_k[valid_minus]] * dx
        
        grad = term1 + term2
        
        velocity = momentum * velocity + grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
