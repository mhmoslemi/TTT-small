def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.05
    decay = 0.9995
    momentum = 0.95

        k = j_max - (n_points - 1)

        valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
        term1 = np.zeros(n_points)
