    best_c5 = min(best_c5, 0.381)
    
    # Adam optimizer parameters
    learning_rate = 0.1
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    iter_count = 0
    time_start = time.time()
    
    while time.time() - time_start < budget_s and iter_count < 10000:
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        
        # Compute gradient for the maximum lag
        gradient = np.zeros(n_points)
        for m_idx in range(n_points):
            i_plus_k = (m_idx + k_max) % n_points
            i_minus_k = (m_idx - k_max) % n_points
            gradient[m_idx] = (1 - h[i_plus_k]) - h[i_minus_k]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * gradient**2
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Periodic perturbation to escape local optima
        if iter_count % 100 == 0:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        iter_count += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
