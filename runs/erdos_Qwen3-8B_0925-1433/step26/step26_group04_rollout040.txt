import numpy as np

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    
    def optimize_candidate(cand):
        cand = np.clip(cand, 0, 1)
        cand = cand * target_sum / np.sum(cand)
        lr = 0.15
        momentum = 0.9
        decay = 0.999
        velocities = np.zeros_like(cand)
        time_start = time.time()
        for _ in range(4000):
            if time.time() - time_start > budget_s:
                break
            corr = np.correlate(cand, 1 - cand, mode='full')
            k_idx = np.argmax(corr)
            shift = k_idx - (n_points - 1)
            grad = 1.0 - cand[(np.arange(n_points) + shift) % n_points]

            lr *= decay
        return best_c5, best_h
