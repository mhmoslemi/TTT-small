import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    guesses = [initial_h.copy()]

    phi = np.random.rand()
    guess_sine = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points-1) + phi))
    guess_sine = np.clip(guess_sine, 0, 1)
    guess_sine = guess_sine * target_sum / np.sum(guess_sine)
    guesses.append(guess_sine)

    block_size = n_points // 4
    guess_block = np.zeros(n_points)
    for i in range(n_points):
        if (i // block_size) % 2 == 0:
            guess_block[i] = 1.0
    guess_block = np.roll(guess_block, np.random.randint(0, block_size))
    guess_block = np.clip(guess_block, 0, 1)
    guess_block = guess_block * target_sum / np.sum(guess_block)
    guesses.append(guess_block)

    indices = np.random.choice(n_points, size=n_points // 5, replace=False)
    guess_sparse = np.zeros(n_points)
    guess_sparse[indices] = 1.0
    guess_sparse = np.clip(guess_sparse, 0, 1)
    guess_sparse = guess_sparse * target_sum / np.sum(guess_sparse)
    guesses.append(guess_sparse)

    population = [guess.copy() for guess in guesses]
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    pop_size = 20
    generations = 100
    mutation_factor = 0.5
    crossover_prob = 0.7

    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:

            c5_trial = compute_c5(trial)
            if c5_trial < compute_c5(population[i]):
