                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 200})
        if res_slsqp.success:
            h_candidate = res_slsqp.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B',
                             bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res_l_bfgs.success:
            h_candidate = res_l_bfgs.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
