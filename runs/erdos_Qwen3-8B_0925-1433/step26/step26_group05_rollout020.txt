import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    POP_SIZE = 40
    MAX_GEN = 1500
    MUT_STDEV = 0.05
    TOURNAMENT_SIZE = 5

    population = [initial_h.copy()]
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        h = np.clip(h, 0, 1)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(population[0], dx)
    start_time = time.time()

    for gen in range(MAX_GEN):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(h, dx) for h in population]
        new_pop = []
        for _ in range(POP_SIZE // 2):
            t = np.random.choice(population, TOURNAMENT_SIZE)
            p1 = min(t, key=lambda x: compute_c5(x, dx))
            t = np.random.choice(population, TOURNAMENT_SIZE)
            p2 = min(t, key=lambda x: compute_c5(x, dx))
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            child += np.random.normal(0, MUT_STDEV, n_points)
            child = np.clip(child, 0, 1)
            ch_sum = np.sum(child)
            child += (target_sum - ch_sum) / n_points
            child = np.clip(child, 0, 1)
            new_pop.append(child)
        combined = population + new_pop
        fitness_combined = [compute_c5(h, dx) for h in combined]
        indices = np.argsort(fitness_combined)
        population = [combined[i] for i in indices[:POP_SIZE]]
        current_best = min(population, key=lambda x: compute_c5(x, dx))
        if compute_c5(current_best, dx) < best_c5:
            best_h = current_best
            best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n_points)
