def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.05
    decay = 0.995
    momentum = 0.9
    perturb_prob = 0.2
    eps = 1e-8
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        j_max = np.argmax(full_corr)
        k = j_max - (n_points - 1)
        k = int(k)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            val1 = (j + k) % n_points
            val2 = (j - k) % n_points
            grad[j] = (1 - h[val1]) - h[val2]
        
        velocity = momentum * velocity + grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, eps, 1 - eps)
        h_new = h_new * target_sum / np.sum(h_new)
