        h_clipped = np.clip(best_h, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        max_overlap = np.max(corr)
        max_lag_idx = np.argmax(corr)
        
        # Finite difference gradient for max lag
        epsilon = 1e-4
        grad = np.zeros_like(best_h)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += epsilon
            h_plus_clipped = np.clip(h_plus, 0, 1)
            h_plus_scaled = h_plus_clipped * target_sum / np.sum(h_plus_clipped)
            corr_plus = np.correlate(h_plus_scaled, 1 - h_plus_scaled, mode='full')
            overlap_plus = corr_plus[max_lag_idx] * dx
            
            h_minus = best_h.copy()
            h_minus[i] -= epsilon
            h_minus_clipped = np.clip(h_minus, 0, 1)
            h_minus_scaled = h_minus_clipped * target_sum / np.sum(h_minus_clipped)
            corr_minus = np.correlate(h_minus_scaled, 1 - h_minus_scaled, mode='full')
            overlap_minus = corr_minus[max_lag_idx] * dx
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        
        best_h -= learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        
        current_sum = np.sum(best_h)
        if abs(current_sum - target_sum) > 1e-3:
            lambda_penalty *= 1.1
        else:
            lambda_penalty *= 0.9
    
    # Local refinement with L-BFGS-B
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 100})
    if res.success:
        best_h = res.x.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
