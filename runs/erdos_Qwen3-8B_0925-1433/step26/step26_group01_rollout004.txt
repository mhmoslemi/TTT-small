    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.05
    decay = 0.9995
    momentum = 0.95
    perturb_prob = 0.1
    T = 10  # Select top 10 lags for optimization
    early_stop = 50
    no_improvement_count = 0
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        corr_values = full_corr[:]
        k_indices = np.argpartition(corr_values, -T)[-T:]
        selected_lags = k_indices
        
        weights = full_corr[selected_lags]
        weights /= np.sum(weights)
        
        grad_total = np.zeros(n_points)
        for k in selected_lags:
