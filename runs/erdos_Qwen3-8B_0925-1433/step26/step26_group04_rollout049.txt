import numpy as np
import time

def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    
    learning_rate = 0.2
    momentum = 0.9
    decay_rate = 0.999
    velocity = np.zeros(n_points)
    time_start = time.time()
    iter_count = 0
    worse_count = 0
    
    while time.time() - time_start < budget_s and iter_count < 5000:
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n_points - 1)
        grad = 1.0 - h[(np.arange(n_points) + shift) % n_points]
        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:

            best_h = h.copy()
            worse_count = 0
        else:

    best_h = best_h * target_sum / np.sum(best_h)
