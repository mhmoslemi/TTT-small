    def constraint1(x):
        return positions[1:] - positions[:-1]
    
    bounds = Bounds([0.0] * (m-1), [2.0] * (m-1))
    
    res_slsqp = minimize(objective_pos, np.zeros(m-1), method='SLSQP', bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint1}])
    
    if res_slsqp.success:
        positions_opt = np.array([0.0] + list(res_slsqp.x) + [2.0])
        widths_opt = positions_opt[1:] - positions_opt[:-1]
        h_block = np.zeros(n_points)
        for i, (s, e) in enumerate([(positions_opt[i], positions_opt[i+1]) for i in range(len(positions_opt)-1)]):
            start_idx = int(round(s / dx))
            end_idx = int(round(e / dx))
            h_block[start_idx:end_idx] = 1.0
        current_c5 = compute_c5(h_block)
        if current_c5 < best_c5:
            best_h = h_block.copy()
            best_c5 = current_c5
    
    # Local refinement
    for _ in range(2):
        if time.time() - time.time() > budget_s:
            break
        res_l_bfgs = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res_l_bfgs.success:
            h_candidate = res_l_bfgs.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
