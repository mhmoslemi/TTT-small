import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # PSO parameters
    population_size = 20
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    vel_lim = 0.2
    max_iter = 2000
    
    # Initialize population
    h_pop = np.zeros((population_size, n_points))
    vel_pop = np.zeros((population_size, n_points))
    for i in range(population_size):
        if i == 0:
            h_pop[i] = initial_h
        else:
            noise = np.random.normal(0, 0.1, n_points)
            h_pop[i] = np.clip(initial_h + noise, 0, 1)
            h_pop[i] = h_pop[i] * target_sum / np.sum(h_pop[i])
    pbest = h_pop.copy()
    gbest = h_pop[0].copy()
    pbest_fitness = np.zeros(population_size)
    gbest_fitness = float('inf')
    
    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for iter in range(max_iter):
        if time.time() > budget_s:
            break
        # Evaluate fitness
        fitness = np.zeros(population_size)
        for i in range(population_size):
            fitness[i] = compute_fitness(h_pop[i])
        # Update pbest and gbest
        for i in range(population_size):
            if fitness[i] < pbest_fitness[i]:
                pbest[i] = h_pop[i].copy()
                pbest_fitness[i] = fitness[i]
        gbest = pbest[np.argmin(pbest_fitness)].copy()
        gbest_fitness = np.min(pbest_fitness)
        # Update velocities and positions
        for i in range(population_size):
            r1, r2 = np.random.rand(2)
            vel_pop[i] = (inertia * vel_pop[i] +
                          cognitive * r1 * (pbest[i] - h_pop[i]) +
                          social * r2 * (gbest - h_pop[i]))
            vel_pop[i] = np.clip(vel_pop[i], -vel_lim, vel_lim)

        # Early termination
        if gbest_fitness < 0.3808:
