    bounds = [(0, 2) for _ in range(m)]
    result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-5, mutation=(0.5, 1), recombination=0.9)
    best_weights = result.x
    best_c5 = result.fun

    def objective_slsqp(x):
        x = np.array(x)
        total = np.sum(x)
        if total == 0:
            return np.inf
        x = x / total * 2
        block_starts = np.zeros(m + 1)
        block_starts[0] = 0
        for i in range(1, m):
            block_starts[i] = block_starts[i-1] + x[i-1]
        block_starts[m] = 2
        h_vals = np.zeros(n_points)
        for i in range(n_points):
            x_val = i * dx
            idx = np.searchsorted(block_starts, x_val)
            if idx < m:
                h_vals[i] = 1
            else:
                h_vals[i] = 0
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(corr)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 2}]
    res_slsqp = minimize(objective_slsqp, best_weights, method='SLSQP', bounds=[(0, 2) for _ in range(m)], constraints=constraints, tol=1e-6, options={'maxiter': 50})
    if res_slsqp.success:
        best_weights = res_slsqp.x
        best_c5 = res_slsqp.fun

    block_starts = np.zeros(m + 1)
    block_starts[0] = 0
    for i in range(1, m):
        block_starts[i] = block_starts[i-1] + best_weights[i-1]
    block_starts[m] = 2
    h_vals = np.zeros(n_points)
    for i in range(n_points):
        x_val = i * dx
        idx = np.searchsorted(block_starts, x_val)
        if idx < m:
            h_vals[i] = 1
        else:
            h_vals[i] = 0
    sum_h = np.sum(h_vals)
    if abs(sum_h - target_sum) > 1e-6:
        h_vals = h_vals * target_sum / sum_h
    return (h_vals, best_c5, n_points)
