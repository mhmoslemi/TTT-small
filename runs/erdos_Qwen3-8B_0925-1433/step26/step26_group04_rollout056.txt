        # Update momentum
        momentum_buffer = momentum * momentum_buffer + (1 - momentum) * gradient
        h -= learning_rate * momentum_buffer
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        learning_rate *= decay_rate
        iter_count += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
