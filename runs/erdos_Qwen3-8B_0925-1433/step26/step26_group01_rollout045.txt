    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.1)
    T = 8  # number of top lags to consider
    learning_rate = 0.05
    decay = 0.99
    momentum = 0.8
    perturb_prob = 0.1
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        top_indices = np.argsort(full_corr)[-T:]
        
        lags = [idx - (n_points - 1) for idx in top_indices]
        gradient = np.zeros(n_points)
        for k in lags:
            shifted_h = np.roll(h, shift=-k)
            gradient += (1 - shifted_h) * dx
        
        velocity = momentum * velocity + gradient
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = np.max(full_corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
