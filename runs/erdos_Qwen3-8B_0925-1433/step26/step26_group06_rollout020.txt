        h = np.zeros(n_points)
        for i in range(len(block_indices)-1):
            if block_indices[i+1] > block_indices[i]:
                h[block_indices[i]:block_indices[i+1]] = 1.0
        return h

    def objective(x):
        block_widths = x
        h = generate_blocks(block_widths)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def constraint(x):
        return np.sum(x) - 2.0

    # Initial guess from initial_h_values
    if 'initial_h_values' in globals():
        block_starts = [0.0]
        block_widths = []
        for i in range(n_points):
            if h[i] == 1 and (i == 0 or h[i-1] == 0):
                start = i
            elif h[i] == 0 and (i > 0 and h[i-1] == 1):
                end = i
                block_widths.append((end - start) * dx)
        if not block_widths:
            block_widths = [2.0 / len(h)] * len(h)
    else:
        block_widths = [2.0 / len(h)] * len(h)

    initial_block_widths = np.array(block_widths)

    # Optimization
    result = minimize(objective, initial_block_widths, method='SLSQP',
                      constraints=[{'type': 'eq', 'fun': constraint}],
                      bounds=[(0.001, 2.0) for _ in initial_block_widths],
                      options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})
    best_block_widths = result.x
    best_h = generate_blocks(best_block_widths)
    best_c5 = objective(best_block_widths)

    # Local refinement
    for _ in range(3):
        if time.time() - time.time() > budget_s:
            break
        result = minimize(objective, best_block_widths, method='L-BFGS-B',
                          bounds=[(0.001, 2.0) for _ in best_block_widths],
                          options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})
        best_block_widths = result.x
        best_h = generate_blocks(best_block_widths)
        best_c5 = objective(best_block_widths)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
