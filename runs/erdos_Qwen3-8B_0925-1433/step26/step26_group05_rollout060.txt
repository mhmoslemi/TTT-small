def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    POP_SIZE = 40
    MUTATION_STD = 0.05
    TOURNAMENT_SIZE = 5
    MAX_GENERATIONS = 1500
    
    population = [initial_h]
    for _ in range(POP_SIZE - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    np.random.shuffle(population)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    generations = 0
    
    def compute_c5(h, dx):
        n = len(h)
        max_overlap = 0.0

            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:

        return max_overlap

        return h

    while generations < MAX_GENERATIONS and time.time() - start_time < budget_s:
