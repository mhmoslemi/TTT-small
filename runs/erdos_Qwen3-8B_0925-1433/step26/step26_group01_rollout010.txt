def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.2)
    L = 5
    learning_rate = 0.1
    decay = 0.999
    momentum = 0.9
    perturb_prob = 0.1
    velocity = np.zeros(n_points)
    
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        indices = np.argsort(full_corr)[::-1]
        top_lags_indices = indices[:L]
        lags = [i - (n_points - 1) for i in top_lags_indices]
        grad = np.zeros(n_points)
        for k in lags:
            for j in range(n_points):
                j_plus_k = j + k
                if 0 <= j_plus_k < n_points:
                    grad[j] += 1 - h[j_plus_k]
                j_minus_k = j - k
                if 0 <= j_minus_k < n_points:
                    grad[j] -= h[j_minus_k]
        velocity = momentum * velocity - learning_rate * grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
