        for j in range(n_points):
            shifted_plus = (j + k) % n_points
            shifted_minus = (j - k) % n_points
            gradient[j] = (1.0 - h[shifted_plus]) - h[shifted_minus]
        velocities = momentum * velocities + gradient
        h -= learning_rate * velocities
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        learning_rate *= decay_rate
        if iter % 100 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            perturb_count += 1

    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, final_c5, n_points)
