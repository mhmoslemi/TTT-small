import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)
    t = 1
    learning_rate = 0.1
    
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        overlaps = [np.sum(h * (1 - np.roll(h, k))) * dx for k in range(n_points)]
        top_k = np.argsort(overlaps)[-5:]
        grad = np.zeros(n_points)
        for k in top_k:
            j_plus = (np.arange(n_points) + k) % n_points
            j_minus = (np.arange(n_points) - k) % n_points
            grad += (1 - h[j_plus]) - h[j_minus]
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= learning_rate * m_hat / np.sqrt(v_hat + epsilon)
