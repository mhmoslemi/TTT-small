            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = 0.5 * np.ones(n_points)
            else:
                h = h * target_sum / sum_h
        learning_rate *= 0.9995
    return (best_h, best_c5 * dx, n_points)
