import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points) * 2
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def scale_to_sum(x):
        x = np.clip(x, 0, 1)
        return x * target_sum / np.sum(x)

    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)

    def generate_block(n):
        h = np.zeros(n)
        for i in range(0, n, 5):
            h[i:i+5] = 1.0
        return h

    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h

    def generate_evenly(n):
        h = np.zeros(n)
        spacing = n // 2
        for i in range(n):
            if (i - 0) % spacing == 0:
                h[i] = 1.0
        return h

    def generate_sparse(n):
        h = np.zeros(n)
        step = n // 4
        for i in range(0, n, step):
            h[i] = 1.0
        return h

    initial_guesses = [generate_block(n_points), generate_sine(n_points), generate_evenly(n_points), generate_sparse(n_points)]
    for guess in initial_guesses:
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, [(0, 1)] * n_points, popsize=40, maxiter=200, tol=1e-5, mutation=(0.5, 1.0), recombination='exponential')
        h_candidate = res.x.copy()
        h_candidate = scale_to_sum(h_candidate)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP',
                        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                        bounds=[(0, 1)] * n_points, tol=1e-8, options={'maxiter': 500})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = scale_to_sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B',
                        bounds=[(0, 1)] * n_points, tol=1e-8, options={'maxiter': 200})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = scale_to_sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    best_h = scale_to_sum(best_h)
    return (best_h, best_c5, n_points)
