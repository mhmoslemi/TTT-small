import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def compute_gradient(h, k):
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1.0
            h_plus = h + eps * e
            h_plus = clip_rescale(h_plus)
            h_minus = h - eps * e
            h_minus = clip_rescale(h_minus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            obj_plus = corr_plus[k] * dx
            obj_minus = corr_minus[k] * dx
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    start_time = time.time()
    best_c5 = np.inf
    best_h = h_ref.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        grad = compute_gradient(best_h, k_max)
        best_h -= 0.05 * grad
        best_h = clip_rescale(best_h)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(compute_c5, best_h, method='L-BFGS-B',
                          bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6, options={'maxiter': 200, 'disp': False})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
    
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
