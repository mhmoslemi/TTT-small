import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def de_bounds():
        return [(0, 1)] * n_points

    def de_fitness(x):
        x = clip_rescale(x)
        return objective(x)

    def generate_structured_guesses():
        n = n_points
        guess1 = np.zeros(n)
        for i in range(n):
            if (i % (n // 3)) == 0:
                guess1[i] = 1.0
        guess1 = np.roll(guess1, np.random.randint(0, n // 3))
        guess1 = clip_rescale(guess1)

        guess2 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        guess2 = np.clip(guess2, 0, 1)
        guess2 = guess2 * target_sum / np.sum(guess2)

        block_size = n // 4
        guess3 = np.zeros(n)
        for i in range(n):
            if (i // block_size) % 2 == 0:
                guess3[i] = 1.0
        guess3 = np.roll(guess3, np.random.randint(0, block_size))
        guess3 = clip_rescale(guess3)

        guess4 = 0.5 * np.abs(np.sin(np.pi * np.arange(n) / (n // 2)))
        guess4 = np.clip(guess4, 0, 1)
        guess4 = guess4 * target_sum / np.sum(guess4)

        guess5 = np.zeros(n)
        k = np.random.randint(0, n)
        for i in range(n):
            if (i - k) % (n // 5) == 0:
                guess5[i] = 1.0
        guess5 = np.roll(guess5, np.random.randint(0, n // 5))
        guess5 = clip_rescale(guess5)

        return [guess1, guess2, guess3, guess4, guess5]

    initial_guesses = generate_structured_guesses()
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(de_fitness, de_bounds(), 
                                       strategy='best1bin', 
                                       maxiter=200, 
                                       popsize=50, 
                                       tol=1e-6, 
                                       mutation=(0.8, 1.2), 
                                       recombination=0.9, 
                                       disp=False)
        if result.success:
            res_h = result.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_h.copy()

    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP',
                          bounds=[(0, 1)] * n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6, options={'maxiter': 300})
        if result.success and result.fun < best_c5:

    for _ in range(3):

        result = minimize(objective, best_h, method='L-BFGS-B',
