import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(45) * 2
    n_points = len(h)

    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def evaluate(solution):
        solution = np.clip(solution, 0, 1)
        solution = solution * target_sum / np.sum(solution)
        corr = np.correlate(solution, 1 - solution, mode='full')
        return np.max(corr) * dx
    
    def generate_block(n, size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i+size] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        return np.clip(h, 0, 1)
    
    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    population = [h.copy()]
    for _ in range(49):
        guess = generate_block(n_points, 5)
        guess = guess * target_sum / np.sum(guess)
        population.append(guess)
        guess = generate_sine(n_points)
        guess = guess * target_sum / np.sum(guess)
        population.append(guess)
        guess = generate_evenly_spaced(n_points, 0)
        guess = guess * target_sum / np.sum(guess)
        population.append(guess)
        guess = generate_evenly_spaced(n_points, 1)
        guess = guess * target_sum / np.sum(guess)
        population.append(guess)
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(sol) for sol in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[best_idx].copy()
        
        parents = np.random.choice(len(population), size=50, replace=False)
        new_population = []
        for i in range(0, 50, 2):
            p1, p2 = parents[i], parents[i+1]
            child1, child2 = population[p1].copy(), population[p2].copy()
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child1[j] = population[p2][j]

            for child in [child1, child2]:
