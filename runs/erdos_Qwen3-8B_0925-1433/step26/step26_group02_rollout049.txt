import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)

    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        return np.clip(x * target_sum / np.sum(x), 0, 1)

    def compute_gradient(h):
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = clip_rescale(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = clip_rescale(h_minus)
            corr_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx

            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        return grad

    start_time = time.time()
