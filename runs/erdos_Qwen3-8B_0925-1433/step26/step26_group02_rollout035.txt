import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def generate_guesses():
        n = n_points
        phi = np.random.rand()
        guess1 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n-1) + phi))
        guess1 = clip_rescale(guess1)
        block_size = n // 4
        guess2 = np.zeros(n)
        guess2[::block_size] = 1.0
        guess2 = np.roll(guess2, np.random.randint(0, block_size))
        guess2 = clip_rescale(guess2)
        guess3 = np.zeros(n)
        guess3[np.random.choice(n, size=n//5, replace=False)] = 1.0
        guess3 = clip_rescale(guess3)
        return [guess1, guess2, guess3]

    start_time = time.time()
    guesses = generate_guesses()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        h = guess.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            grad = np.zeros(n_points)
            for i in range(n_points):
                e = np.zeros(n_points)
                e[i] = 1.0
                h_plus = h + 1e-4 * e
                h_plus = clip_rescale(h_plus)

                h_minus = clip_rescale(h_minus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
