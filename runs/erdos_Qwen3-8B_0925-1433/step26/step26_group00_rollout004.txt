            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            new_max = max(np.correlate(h_plus, 1 - h_plus, mode='full'))
            grad[i] = (new_max - current_max) / epsilon
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_max = max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_max < best_c5:
            best_h = h.copy()
            best_c5 = new_max

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    res = minimize(objective, best_h, method='SLSQP',
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                   bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
    if res.success:
        best_h = res.x.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
