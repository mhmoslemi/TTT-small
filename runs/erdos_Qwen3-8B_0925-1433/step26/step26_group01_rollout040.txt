import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.1
    decay = 0.999
    momentum = 0.9
    perturb_prob = 0.2
    
    velocity = np.zeros(n_points)
    no_improve = 0
    early_stop = 50
    
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        top_k_indices = np.argsort(full_corr)[-3:]
        grads = []
        
        for idx in top_k_indices:
            k = idx - (n_points - 1)
            i_plus_k = np.arange(n_points) + k
            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            term1 = np.zeros(n_points)
            term1[valid_plus] = (1 - h[i_plus_k[valid_plus]]) * dx
            
            i_minus_k = np.arange(n_points) - k
            valid_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
            term2 = np.zeros(n_points)
            term2[valid_minus] = -h[i_minus_k[valid_minus]] * dx
            
            grad = term1 + term2
            grads.append(grad)
        
        if grads:
            grad = np.sum(grads, axis=0) / len(grads)
        else:
            grad = np.zeros(n_points)
        
        velocity = momentum * velocity + grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.1, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = full_corr.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= early_stop:
            break
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
