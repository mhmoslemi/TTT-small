import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    start_time = time.time()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m, v = np.zeros(n_points), np.zeros(n_points)

    K = 5
    learning_rate = 0.1
    max_iterations = 10000

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        overlaps = np.array([np.sum(h * (1 - np.roll(h, k))) * dx for k in range(n_points)])
        k_indices = np.argsort(overlaps)[-K:]
        grad = np.zeros(n_points)

                j_plus = (i + k) % n_points
                j_minus = (i - k) % n_points
                grad[i] += (1 - h[j_plus]) - h[j_minus]
        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + eps)

        sum_h = np.sum(h)

            h = 0.5 * np.ones(n_points)

        learning_rate *= 0.9995

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
