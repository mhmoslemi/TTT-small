from scipy.optimize import differential_evolution, minimize
import time

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)

            h[i:i + size] = 1.0
        return h
    
    def generate_phase_shifted(n, phase=1, size=5):

        for i in range(n):

                h[i] = 1.0
        return h
    
    def generate_sparse(n, density=0.2):

        h[indices] = 1.0
        return h
    
    initial_guesses = [h]
    initial_guesses.append(generate_block(n_points, size=5))
    initial_guesses.append(generate_phase_shifted(n_points, phase=1, size=7))
