def run(seed=42, budget_s=600):

    L = 5

    decay = 0.995
    momentum = 0.9
    perturb_prob = 0.1
    velocity = np.zeros(n_points)
    
    for t in range(max_iter):

        indices = np.argsort(full_corr)[-L:]
        grad = np.zeros(n_points)
        for idx in indices:

            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            term1 = np.zeros(n_points)
            term1[valid_plus] = (1 - h[i_plus_k[valid_plus]]) * dx
            i_minus_k = np.arange(n_points) - k
            valid_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
            term2 = np.zeros(n_points)
            term2[valid_minus] = -h[i_minus_k[valid_minus]] * dx
            grad += term1 + term2
        velocity = momentum * velocity + grad
