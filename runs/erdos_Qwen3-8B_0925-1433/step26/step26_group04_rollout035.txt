        # Manual gradient descent with momentum
        # Compute new correlation for current k
        current_corr = np.correlate(h, 1 - h, mode='full')[k_idx]
        if current_corr < best_c5:
            best_c5 = current_corr
            best_h = h.copy()
            best_k = k
        
        learning_rate *= decay
        iter_count += 1
    
    # Fallback to gradient descent if initial method not successful
    if best_c5 > 0.381:
        h = best_h.copy()
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n_points - 1)
        k = shift
        targets = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[0] = 1
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.cos(np.pi * (targets[i] - shift) / 2))
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
        
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
