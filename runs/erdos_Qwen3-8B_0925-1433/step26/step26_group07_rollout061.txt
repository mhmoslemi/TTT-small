import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(45) * 2
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def generate_block(n):
        h = np.zeros(n)
        l = 1.0
        m = 5
        w = l / m
        for i in range(m):
            pos = 0.5 + i * (2.0 / m)
            start = pos - w / 2
            end = pos + w / 2
            idx_start = int(start / dx)
            idx_end = int(end / dx)
            h[idx_start:idx_end] = 1.0
        return h * target_sum / np.sum(h)
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)
    
    def generate_evenly_spaced(n):
        spacing = n // 4
        h = np.zeros(n)
        for i in range(n):
            if (i - spacing % n) % spacing == 0:
                h[i] = 1.0
        return h * target_sum / np.sum(h)
    
    initial_guesses = [np.copy(h), generate_block(n_points), generate_sine(n_points), generate_evenly_spaced(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res_de = differential_evolution(objective, [(0, 1)] * n_points, popsize=20, maxiter=100, tol=1e-5)
        h_candidate = res_de.x.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(objective, best_h, method='SLSQP',
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 200})
        if res_slsqp.success:
            h_candidate = res_slsqp.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)

                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
