    def objective(h_vec):

        return np.max(corr) * dx

    def clip_rescale(x):

        x = x * target_sum / np.sum(x)
        return x

        return [(0, 1)] * n_points

        x = clip_rescale(x)
        return objective(x)

        return [(0, 1)] * n_points

        return np.sum(x) - target_sum
