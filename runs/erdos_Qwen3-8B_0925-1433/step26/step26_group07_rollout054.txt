import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(45)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(np.clip(x, 0, 1)) - target_sum
    
    def generate_block(n):
        h = np.zeros(n)
        for i in range(0, n, 5):
            h[i:i+5] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        return np.clip(h, 0, 1)
    
    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    initial_guesses = [
        generate_block(n_points), generate_sine(n_points), 
        generate_evenly_spaced(n_points, 0), generate_evenly_spaced(n_points, 1),
        generate_evenly_spaced(n_points, 2)
    ]
    initial_guesses = [g * target_sum / np.sum(g) for g in initial_guesses]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-6,
                       options={'maxiter': 200})
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-6,
