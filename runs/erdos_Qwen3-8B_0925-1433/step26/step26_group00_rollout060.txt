            grad[i] = (new_overlaps.max() - current_max) / 1e-4

        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if iteration % perturb_interval == 0 and iteration > 0:
            perturb_mask = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[perturb_mask] += np.random.uniform(-perturb_strength, perturb_strength, size=len(perturb_mask))
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break

        res = minimize(
            lambda x: compute_c5(x),
            x0=best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )

        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
