from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)

    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_gradient(h):
        eps = 1e-4
        grad = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1.0
            h_plus = h + eps * e
            h_plus = clip_rescale(h_plus)
            h_minus = h - eps * e
            h_minus = clip_rescale(h_minus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            obj_plus = np.max(corr_plus) * dx
            obj_minus = np.max(corr_minus) * dx
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(best_h)
        best_h = best_h - 0.01 * grad
        best_h = clip_rescale(best_h)

        if current_c5 < best_c5:

            best_h = best_h.copy()

    result = minimize(objective, best_h, method='L-BFGS-B',
