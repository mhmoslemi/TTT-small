    for _ in range(population_size - 1):
        parent = np.random.choice(population)
        mutated = parent + np.random.normal(0, 0.05, n_points)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * target_sum / np.sum(mutated)
        population.append(mutated)
    best_c5 = float('inf')
    best_h = initial_h.copy()
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = corr.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    max_generations = int(budget_s / 30)
    for gen in range(max_generations):
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            fitness.append(corr.max())
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        num_parents = int(len(sorted_population) * 0.2)
        parents = sorted_population[:num_parents]
        new_population = []
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        new_pop = sorted_population[:50] + new_population
        population = new_pop
        curr_c5 = float('inf')
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = corr.max()
            if c5 < curr_c5:
                curr_c5 = c5
                if curr_c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    h_values = best_h
    return (h_values, best_c5, n_points)
