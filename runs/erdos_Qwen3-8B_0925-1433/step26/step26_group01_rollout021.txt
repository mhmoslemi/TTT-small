import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.05
    decay = 0.9995
    momentum = 0.9
    perturb_prob = 0.1
    T = 12  # Select top 12 lags for gradient aggregation
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        indices = np.argsort(full_corr)[::-1]
        selected_lags = indices[:T]
        total_grad = np.zeros(n_points)
        for k in selected_lags:
            i_plus_k = np.arange(n_points) + k
            valid = (i_plus_k >= 0) & (i_plus_k < n_points)
            term = np.zeros(n_points)
            term[valid] = (1 - h[i_plus_k[valid]]) * full_corr[k]
            total_grad += term
        
        velocity = momentum * velocity + total_grad
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.03, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = full_corr.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
    h_values = h_values * target_sum / np.sum(h_values)
    return (h_values, best_c5, n_points)
