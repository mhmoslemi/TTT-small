import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(np.array(initial_h_values, dtype=float), 0, 1)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    best_c5 = min(best_c5, 0.381)
    
    learning_rate = 0.2
    momentum = 0.9
    decay_rate = 0.999
    iter_count = 0
    time_start = time.time()
    velocity = np.zeros(n_points)
    perturbation = 0.05
    restart_threshold = 100
    no_improvement = 0
    
    while time.time() - time_start < budget_s and iter_count < 100000:
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n_points - 1)
        gradient = 1.0 - h[(np.arange(n_points) + shift) % n_points]
        velocity = momentum * velocity - learning_rate * gradient
        h += velocity
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > restart_threshold:
            h += np.random.uniform(-perturbation, perturbation, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            no_improvement = 0
        learning_rate *= decay_rate
        iter_count += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
