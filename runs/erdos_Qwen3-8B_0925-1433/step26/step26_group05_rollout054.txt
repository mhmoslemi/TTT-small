import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.round(initial_h_values * 2).astype(int)
    current_sum = np.sum(h)
    target = int(target_sum)
    if current_sum != target:
        diff = current_sum - target
        if diff > 0:
            for _ in range(diff):
                idx = np.random.choice(np.where(h == 1)[0])
                h[idx] = 0
        else:
            for _ in range(-diff):
                idx = np.random.choice(np.where(h == 0)[0])
                h[idx] = 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    best_c5 = min(best_c5, 0.381)
    T = 1.0
    cooling_rate = 0.995
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h_neighbor = h.copy()
        i = np.random.randint(n_points)
        if h_neighbor[i] == 0:
            h_neighbor[i] = 1
            j = np.random.choice(np.where(h_neighbor == 1)[0])
            h_neighbor[j] = 0
        else:
            h_neighbor[i] = 0
            j = np.random.choice(np.where(h_neighbor == 0)[0])
            h_neighbor[j] = 1
        new_c5 = compute_c5(h_neighbor, dx)
        if new_c5 < best_c5:
            h = h_neighbor.copy()
            best_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - best_c5) / T)
            if np.random.rand() < prob:
                h = h_neighbor.copy()
                best_c5 = new_c5
        T *= cooling_rate
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
