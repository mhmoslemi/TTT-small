import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate initial guesses
    h0 = np.array(initial_h_values, dtype=float)
    h0 = h0 * target_sum / np.sum(h0)
    guesses = [h0]

    phi = np.random.rand()

    guess_sine = clip_rescale(guess_sine)
    guesses.append(guess_sine)

    block_size = n_points // 4
    guess_block = np.zeros(n_points)
    for i in range(n_points):
        if (i // block_size) % 2 == 0:
            guess_block[i] = 1.0
    guess_block = np.roll(guess_block, np.random.randint(0, block_size))

    guesses.append(guess_block)

    guess_sparse = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points // 5, replace=False)
    guess_sparse[indices] = 1.0
    guess_sparse = clip_rescale(guess_sparse)
    guesses.append(guess_sparse)

    population = [np.copy(g) for g in guesses]
    best_h = np.copy(population[0])
