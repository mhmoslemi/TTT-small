def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    max_iter = int(budget_s / 0.15)
    learning_rate = 0.05
    decay = 0.997
    momentum = 0.9
    perturb_prob = 0.2
    early_stop_threshold = 100
    no_improve_counter = 0
    
    velocity = np.zeros(n_points)
    for t in range(max_iter):
        full_corr = np.correlate(h, 1 - h, mode='full') * dx
        top_lags_indices = np.argsort(full_corr)[-4:]  # top 4 lags
        top_lags = [j - (n_points - 1) for j in top_lags_indices]
        gradient = np.zeros(n_points)
        for k in top_lags:
            i_plus_k = np.arange(n_points) + k
            valid_plus = (i_plus_k >= 0) & (i_plus_k < n_points)
            term1 = (1 - h[i_plus_k[valid_plus]]) * dx
            i_minus_k = np.arange(n_points) - k
            valid_minus = (i_minus_k >= 0) & (i_minus_k < n_points)
            term2 = -h[i_minus_k[valid_minus]] * dx

            grad_for_k[valid_plus] = term1
            grad_for_k[valid_minus] = term2
            gradient += grad_for_k
        
        velocity = momentum * velocity + gradient
        alpha = learning_rate * (1 - decay * t / max_iter)
        h_new = h - alpha * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        if np.random.rand() < perturb_prob:
            perturbation = np.random.normal(0, 0.08, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
        
        current_c5 = full_corr.max()

            no_improve_counter += 1
        
        if no_improve_counter >= early_stop_threshold:

        h = h_new
    
    h_values = np.clip(best_h, 0, 1)
