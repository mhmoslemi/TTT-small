from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_gradient(h):
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1.0
            h_plus = h + eps * e
            h_plus = clip_rescale(h_plus)
            h_minus = h - eps * e
            h_minus = clip_rescale(h_minus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            obj_plus = np.max(corr_plus) * dx
            obj_minus = np.max(corr_minus) * dx
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    start_time = time.time()

    def generate_guesses():
        n = n_points
        phi = np.random.rand()
        guess1 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n-1) + phi))
        guess1 = np.clip(guess1, 0, 1)
        guess1 = guess1 * target_sum / np.sum(guess1)
        block_size = n // 4
        guess2 = np.zeros(n)
        for i in range(n):
            if (i // block_size) % 2 == 0:
                guess2[i] = 1.0
        guess2 = np.roll(guess2, np.random.randint(0, block_size))
        guess2 = np.clip(guess2, 0, 1)
        guess2 = guess2 * target_sum / np.sum(guess2)
        guess3 = np.zeros(n)
        indices = np.random.choice(n, size=n//5, replace=False)
        guess3[indices] = 1.0
        guess3 = np.clip(guess3, 0, 1)

    guesses = generate_guesses()
    for guess in guesses:
        if time.time() - start_time > budget_s:

        for _ in range(50):
            if time.time() - start_time > budget_s:

            h = h - 0.01 * grad
            h = clip_rescale(h)
            c5 = compute_c5(h)
