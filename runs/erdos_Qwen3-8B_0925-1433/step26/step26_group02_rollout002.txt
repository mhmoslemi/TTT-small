import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_gradient(h):
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            e = np.zeros(n_points)
            e[i] = 1.0
            h_plus = h + eps * e
            h_plus = clip_rescale(h_plus)
            h_minus = h - eps * e
            h_minus = clip_rescale(h_minus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            obj_plus = np.max(corr_plus) * dx
            obj_minus = np.max(corr_minus) * dx
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    start_time = time.time()

    def generate_focused_guess():
        phi = np.random.rand()
        guess = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1) + phi))
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        return guess

    initial_guess = generate_focused_guess()
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(best_h)
        best_h -= 0.05 * grad
        best_h = clip_rescale(best_h)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1)] * n_points

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
