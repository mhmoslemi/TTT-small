    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    
    # Parameters
    alpha = 0.1
    decay_rate = 0.99
    max_iterations = 10000
    restarts = 0
    restart_interval = 500
    time_start = time.time()
    
    for iter in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        positive_shifts = corr[-n_points:]
        max_index = np.argmax(positive_shifts)
        k = max_index
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = (i + k) % n_points
            grad[i] = (1 - h[shifted_i]) * dx
        # Update h
        h = h - alpha * grad
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Check if we need to restart
        if iter % restart_interval == 0 and iter != 0:
            # Random perturbation
            h += np.random.uniform(-0.2, 0.2, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        # Compute current C5
        curr_c5 = np.max(positive_shifts) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        # Adjust alpha
        alpha *= decay_rate
    # Final check
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, curr_c5)
    return (best_h, best_c5, n_points)
