import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    learning_rate = 0.5
    decay_rate = 0.997
    iter_count = 0
    time_start = time.time()
    momentum = 0.9
    velocities = np.zeros(n_points)
    
    while time.time() - time_start < budget_s and iter_count < 5000:
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n_points - 1)
        indices_plus = (np.arange(n_points) + shift) % n_points
        indices_minus = (np.arange(n_points) - shift) % n_points
        gradient = (1 - h[indices_plus]) - h[indices_minus]
        velocities = momentum * velocities + (1 - momentum) * gradient
        h -= learning_rate * velocities
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        learning_rate *= decay_rate
        iter_count += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
