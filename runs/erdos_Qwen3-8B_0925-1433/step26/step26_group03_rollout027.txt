    iteration = 0

    while iteration < max_iterations and time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()

        m_shift = m_pos - (n_points - 1)
