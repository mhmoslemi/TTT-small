import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(45) * 2
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def compute_c5(h_vals):
        h_scaled = h_vals * target_sum / np.sum(h_vals)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        bounds = [(0., 1.)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
