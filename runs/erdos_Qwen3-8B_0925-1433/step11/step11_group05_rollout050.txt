import numpy as np
from scipy.optimize import minimize
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals():

            h = h * (n_points / 2) / sum_h

        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def perturb(h, std=0.05):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Bayesian Optimization with GP
    gp_kernel = Matern(length_scale=0.3, nu=1.5)
    gp = GaussianProcessRegressor(kernel=gp_kernel, alpha=1e-5)
    X = np.array([np.arange(n_points) / n_points]).T
    y = np.array([compute_c5(h)])
    
    iter_count = 0
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        iter_count += 1
        if iter_count % 10 == 0:
            std = max(0.01, 0.05 * (1 - iter_count / 100))
        else:
            std = 0.05
        
        # Evaluate current best
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Generate new candidate
        if len(X) < 10:
            new_h = perturb(best_h, std)
        else:
            # GP-guided perturbation

                    pert = np.random.normal(0, std, n_points)

                    candidate = best_h + pert
                    candidate = np.clip(candidate, 0, 1)
                    candidate = candidate * target_sum / np.sum(candidate)
                    candidates.append(candidate)
                new_h = candidates[np.random.choice(5)]

        # Update GP

        y_new = np.array([new_c5])
