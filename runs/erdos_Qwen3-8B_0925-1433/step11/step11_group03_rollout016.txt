from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    if initial_h_values is not None:
        h = initial_h_values * target_sum / np.sum(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    population = [h.copy()]
    for _ in range(49):  # Increased initial population
        perturb = np.random.normal(0, 0.2, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        population.append(candidate)

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(300):  # Increased generations
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        elite_indices = np.argsort(fitness)[:int(len(population)*0.2)]
        elites = [population[i] for i in elite_indices]
        new_population = elites.copy()
        for _ in range(len(population) - len(elites)):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            idx = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:idx], parent2[idx:]])
            mut = np.random.normal(0, 0.05, n_points)
            child = np.clip(child + mut, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        if gen % 10 == 0 and current_c5 >= best_c5:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Multi-stage refinement
    refined = []
    for _ in range(5):
