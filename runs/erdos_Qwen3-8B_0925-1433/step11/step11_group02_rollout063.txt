        for i in range(n_particles):
            h = particles[i]
            c5 = compute_c5(rescale(h))
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = rescale(h)
        best_idx = np.argmin(pbest_vals)
        if pbest_vals[best_idx] < gbest_val:
            gbest_val = pbest_vals[best_idx]
            gbest = pbest[best_idx].copy()
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = rescale(particles[i])

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        r1, r2 = np.random.rand(), np.random.rand()
        velocities = 0.4 * velocities + 2.0 * r1 * (gbest - particles) + 2.0 * r2 * (gbest - particles)
        particles += velocities
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = rescale(particles[i])

    return (gbest, gbest_val, n_points)
