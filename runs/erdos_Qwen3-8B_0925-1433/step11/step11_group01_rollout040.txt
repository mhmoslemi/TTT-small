import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0
    decay_rate = 0.995
    learning_rate = 0.05

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(best_h)
        epsilon = 1e-4
        for i in range(n_points):
            h_perturbed = best_h.copy()
            h_perturbed[i] += epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            sum_h_p = np.sum(h_perturbed)
            if sum_h_p > 0:
                h_perturbed = h_perturbed * target_sum / sum_h_p
            corr_p = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad[i] = (corr_p[k_max] - corr[k_max]) / epsilon
        best_h = best_h - learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * target_sum / sum_h
        new_c5 = obj(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break
        learning_rate *= decay_rate

    for _ in range(150):
        if time.time() - start_time > budget_s:

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
