from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0

    # Initial refinement with SLSQP
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h, best_c5 = res.x, res.fun
    else:
        early_stop = 100

    # Simulated annealing phase
    for temp in np.logspace(0, 1, 100):
        if time.time() - start_time > budget_s:
            break
        new_h = np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        new_c5 = obj(new_h)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            best_h, best_c5 = new_h, new_c5
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 50:
            break

    # Local optimization
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)
