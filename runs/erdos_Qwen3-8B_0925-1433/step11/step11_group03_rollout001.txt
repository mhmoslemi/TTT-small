import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.copy(initial_h_values)
    h = h * target_sum / np.sum(h)

        return np.max(corr) * dx

        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
