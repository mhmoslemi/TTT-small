import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 45
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return h * target_sum / sum_h

    def perturb(h, std):  
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        new_h = rescale(new_h)
        return new_h

    def compute_population_fitness(pop):
        return np.array([compute_c5(p) for p in pop])

    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, len(parent1)-1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child = rescale(child)
        return child

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    population = [perturb(best_h, 0.02) for _ in range(50)]
    fitness = compute_population_fitness(population)
    for i in range(300):
        selected = np.random.choice(len(population), size=50, replace=False)
        new_population = []
        for j in range(len(population)):
            parent1, parent2 = population[selected[j % 50]], population[selected[(j+1) % 50]]
            child = crossover(parent1, parent2)
            new_h = perturb(child, 0.01)
            new_population.append(new_h)
        population = new_population
        fitness = compute_population_fitness(population)
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_h = population[idx].copy()

        if i % 10 == 0:

    return (np.clip(best_h, 0, 1), best_c5, n_points)
