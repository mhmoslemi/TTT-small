    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(v):
        return v * target_sum / np.sum(v) if np.sum(v) > 0 else v

    initial = np.array(initial_h_values, dtype=float)
    initial = normalize(initial)
    pop = [initial]
    for _ in range(18):
        perturb = np.random.normal(0, 0.1, n_points)
        h = np.clip(initial + perturb, 0, 1)
        h = normalize(h)
        pop.append(h)
        if np.random.rand() < 0.2:
            h = np.random.rand(n_points)
            h = normalize(h)
            pop.append(h)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial.copy()
    best_c5 = eval_c5(best_h)
    start_time = time.time()
    gen = 0

    while time.time() - start_time < budget_s and gen < 500:
        gen += 1
        new_pop = []
        elites = sorted(pop, key=eval_c5)[:5]
        new_pop.extend(elites)
        for _ in range(len(pop) - 5):
            idxs = np.random.choice(len(pop), 4, replace=False)
            candidates = [pop[i] for i in idxs]
            parents = sorted(candidates, key=eval_c5)[:2]
            p1, p2 = parents
            idx = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:idx], p2[idx:]])
            mut = np.random.normal(0, 0.02, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_pop.append(child)
        pop = new_pop
        current_c5 = eval_c5(pop[0])
        if current_c5 < best_c5:
            best_h, best_c5 = pop[0], current_c5

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    for h in pop[:5]:
        if time.time() - start_time >= budget_s:
            break
        def obj(x):
            h = np.clip(x, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            return eval_c5(h)
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
