import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    def generate_patterns():
        patterns = [initial_h_values]
        # Binary half-half
        patterns.append(np.zeros(n_points) + (n_points // 2 < n_points))
        # Prouhet-Thue-Morse
        patterns.append(np.array([bin(i).count('1') % 2 == 0 for i in range(n_points)]))
        # Interleaved
        i1 = np.zeros(n_points)
        i1[::2] = 1.0
        i2 = np.zeros(n_points)
        i2[::4] = 1.0
        patterns.append(np.vstack([i1, i2]).flatten())
        return patterns

    for patt in generate_patterns():
        h = np.clip(patt, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        def obj(x):
            h_vec = np.clip(x, 0, 1)
            sum_h_vec = np.sum(h_vec)
            if sum_h_vec > 0:
                h_vec = h_vec * target_sum / sum_h_vec
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if time.time() - start_time > budget_s:
            break

    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    return (h, best_c5, n_points)
