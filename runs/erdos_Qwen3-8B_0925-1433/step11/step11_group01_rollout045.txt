        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_idx = np.argmax(corr)
        k_shift = (max_idx - n_points + 1) * dx
        valid_x = np.arange(n_points)
        valid_x = valid_x[(valid_x + k_shift) <= 2.0]
        valid_x_idx = np.clip(valid_x.astype(int), 0, n_points - 1)
        contributions = h_[valid_x_idx] * (1 - h_[np.clip(valid_x_idx + int(k_shift / dx), 0, n_points - 1)])
        if len(contributions) > 0:
            sorted_indices = np.argsort(contributions)[::-1]
            selected_x = valid_x_idx[sorted_indices[:len(valid_x_idx)//5]]
        else:
            selected_x = []
        perturb = np.zeros_like(best_h)
        if selected_x.size > 0:
            perturb[selected_x] += np.random.normal(0, 0.01, selected_x.size)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
