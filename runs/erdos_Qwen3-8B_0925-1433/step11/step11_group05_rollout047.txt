import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 45
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    temp = 1.0
    cooling_rate = 0.996
    iterations = 0
    pop_size = 5
    population = [best_h.copy() for _ in range(pop_size)]

    start_time = time.time()
    while temp > 1e-3 and time.time() - start_time < budget_s:
        for _ in range(pop_size):
            candidate = population[np.random.randint(pop_size)]
            noise = np.random.normal(0, 0.015, len(candidate))
            new_h = np.clip(candidate + noise, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            e_old = compute_c5(candidate)
            e_new = compute_c5(new_h)
            if e_new < e_old or np.random.rand() < np.exp(-(e_new - e_old)/temp):
                population[_] = new_h.copy()
                if e_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = e_new
        temp *= cooling_rate
        iterations += 1
        if iterations % 50 == 0:
            for i in range(pop_size):
                noise = np.random.normal(0, 0.005, len(population[i]))
                population[i] = np.clip(population[i] + noise, 0, 1)
                sum_p = np.sum(population[i])
                population[i] = population[i] * target_sum / sum_p

    bounds = Bounds(0, 1)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
