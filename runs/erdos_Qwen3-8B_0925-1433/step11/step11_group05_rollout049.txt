import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 45
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.02):
        noise = np.random.normal(0, std, len(h))
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    population = [h.copy() for _ in range(50)]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tournament_size = 3
    crossover_rate = 0.8
    mutation_rate = 0.5
    std = 0.02
    max_iter = 500

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(50):
            t = [np.random.choice(population) for _ in range(tournament_size)]
            p1 = min(t, key=compute_c5)
            t = [np.random.choice(population) for _ in range(tournament_size)]
            p2 = min(t, key=compute_c5)
            if np.random.rand() < crossover_rate:
                cp = np.random.randint(1, n_points)
                child = np.concatenate([p1[:cp], p2[cp:]])
            else:
                child = p1.copy()
            if np.random.rand() < mutation_rate:
                child = perturb(child, std)
            new_pop.append(child)
        population = new_pop
        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5

    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
