from scipy.optimize import differential_evolution, minimize

    bounds = [(0, 1)] * n_points
    
    start_time = time.time()
    
    # Run DE with adaptive parameters

            de_obj,

            maxiter=50,
            polish=True,
            seed=seed,
            vectorized=True
        )
    
    best_h_de = result_de.x
    best_c5_de = de_obj(best_h_de)

    if final_sum > 0:
        final_h = final_h * target_sum / final_sum
    
    final_corr = np.correlate(final_h, 1 - final_h, mode='full')

    return (final_h, best_c5, n_points)
