    best_c5 = compute_c5(h)
    best_h = h.copy()

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        cross_corr = corr_full[n_points - 1:2 * n_points - 1]
        k_max = np.argmax(cross_corr)
        value = cross_corr[k_max] * dx

        def objective(x):
            h_ = np.clip(x, 0, 1)
            if np.sum(h_) > 0:
                h_ = h_ * target_sum / np.sum(h_)
            total = 0.0
            for i in range(n_points):
                j = (i + k_max) % n_points
                total += h_[i] * (1 - h_[j])
            return total * dx

        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            if np.sum(new_h) > 0:
                new_h = new_h * target_sum / np.sum(new_h)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5

        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = best_h + perturb
            new_h = np.clip(new_h, 0, 1)
            if np.sum(new_h) > 0:
                new_h = new_h * target_sum / np.sum(new_h)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5

    return (best_h, best_c5, n_points)
