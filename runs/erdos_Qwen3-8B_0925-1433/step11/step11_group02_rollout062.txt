import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

    else:
        base = np.zeros(45)

    n_particles = 50
    particles = [base + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in particles]
    velocities = [np.random.normal(0, 0.01, n_points) for _ in range(n_particles)]
    pbest = particles
    pbest_vals = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    gbest = np.array(pbest[np.argmin(pbest_vals)])
    gbest_val = gbest_vals = np.min(pbest_vals)

    start = time.time()
    for _ in range(500):
        if time.time() - start > budget_s:

        current_gbest_val = np.min(pbest_vals)
