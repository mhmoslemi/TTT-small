    from scipy.optimize import minimize

    np.random.seed(seed)

        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    else:
        h = np.random.rand(n_points)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    start_time = time.time()
    for _ in range(200):  # number of iterations
        if time.time() - start_time > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')

        lag_index = max_index
        lag = lag_index - (n_points - 1)
        # Create A matrix for the current lag
        n = n_points
        A = np.zeros((n, n))

            if 0 <= j < n:

        # Convert to minimization by negating
        obj = cp.Minimize(-cp.quad_form(h, A))
        # Constraints
        constraints = [
            cp.sum(h) == target_sum,
            cp.constraints.NonNegativity(h)
        ]
        # Solve QP
        try:
