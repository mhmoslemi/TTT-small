import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    
    def objective(h_vec):
        h = h_vec
        corr = np.correlate(h, 1 - h, mode='full') * dx
        beta = 100
        return np.log(np.sum(np.exp(beta * corr))) / beta
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun
    
    # Compute true C5
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    true_c5 = np.max(corr)
    
    if time.time() - start_time > budget_s:
        return (best_h, true_c5, n_points)
    
    # Additional refinement
    def refined_objective(h_vec):
        h = h_vec
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    result = minimize(refined_objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    
    return (best_h, final_c5, n_points)
