        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag_shift
            if 0 <= j < n_points:
                gradient[i] = dx * (1 - h[j])
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay
        iter_count += 1
        if iter_count % perturbation_freq == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    best_c5 = compute_c5(h)
    return (h, best_c5, n_points)
