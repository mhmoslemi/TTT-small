import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600):

        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:

        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.01):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    def local_refine(h):
        res = minimize(compute_c5, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success:
            return res.x
        return h

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    population = [best_h.copy()]
    temp = 1.0
    cooling_rate = 0.996
    iterations = 0
