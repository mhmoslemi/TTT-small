import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    learning_rate = 0.1
    learning_rate_decay = 0.995
    max_iterations = 1000
    early_stop = 50

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        s = max_index - (n_points - 1)

        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]

        h -= learning_rate * gradient
        learning_rate *= learning_rate_decay

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(h)
