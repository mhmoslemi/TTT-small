import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 200
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 if i % 2 == 0 else 0.5
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    h = clip_and_normalize(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    learning_rate = 0.1
    max_iterations = 1000
    tolerance = 1e-6
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, h, mode='full')
        min_index = np.argmin(corr)
        k_min = min_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx1 = (i + k_min) % n_points
            idx2 = (i - k_min) % n_points
            grad[i] = h[idx1] + h[idx2]
        step = learning_rate
        h_new = h + step * grad
        h_new = clip_and_normalize(h_new)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        learning_rate *= 0.999
        if abs(new_c5 - best_c5) < tolerance:
            break
        h = h_new

    def obj(x):
        h = clip_and_normalize(x)
        return -compute_c5(h)

    bounds = [(0, 1)] * n_points

    for _ in range(3):
