    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    else:
        best_h = best_h
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Final check
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, best_c5, n_points)
