import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    # Simulated Annealing
    T0 = 1.0
    alpha = 0.995
    steps = 200
    for step in range(steps):
        if time.time() - start_time > budget_s:
            break
        T = T0 * alpha**step
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.zeros(n_points)
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        c5_new = np.max(corr) * dx
        if c5_new < best_c5:
            best_h, best_c5 = new_h.copy(), c5_new
        else:
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h, best_c5 = new_h.copy(), c5_new

    # Local optimization
    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1.0
        h_scaled = h * target_sum / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    return best_h, best_c5, n_points
