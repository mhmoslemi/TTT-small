import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial = np.array(initial_h_values, dtype=float)
    initial = (initial * target_sum) / np.sum(initial) if np.sum(initial) > 0 else initial
    pop = [initial]
    for _ in range(19):
        perturb = np.random.normal(0, 0.1, n_points)
        h = np.clip(initial + perturb, 0, 1)
        h = (h * target_sum) / np.sum(h) if np.sum(h) > 0 else h
        pop.append(h)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial.copy()
    best_c5 = eval_c5(best_h)
    gen = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and gen < 500:
        gen += 1
        new_pop = []
        elites = sorted(pop, key=eval_c5)[:5]
        new_pop.extend(elites)
        for _ in range(len(pop) - 5):
            idxs = np.random.choice(len(pop), 4, replace=False)
            candidates = [pop[i] for i in idxs]
            parents = sorted(candidates, key=eval_c5)[:2]
            p1, p2 = parents
            idx = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:idx], p2[idx:]])
            mut = np.random.normal(0, 0.02, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child = (child * target_sum) / np.sum(child) if np.sum(child) > 0 else child
            new_pop.append(child)
        pop = new_pop
        current_c5 = eval_c5(pop[0])
        if current_c5 < best_c5:
            best_h, best_c5 = pop[0], current_c5

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return eval_c5(h)

    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    for h in pop[:3]:
        if time.time() - start_time >= budget_s:
            break
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
