            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_std, n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 0:
                    child = child * target_sum / sum_child
            new_population.append(child)
        population = new_population
        # Update best
        for h_candidate in population:
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
        # Adaptive mutation
        mutation_std *= cooling_factor

    # Local refinement
    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
