        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)

        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5

        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points)
            new_c5_perturb = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5_perturb < best_c5:
                best_h = h.copy()
                best_c5 = new_c5_perturb

        learning_rate *= decay_rate

    h = best_h.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    return (h, final_c5, n_points)
