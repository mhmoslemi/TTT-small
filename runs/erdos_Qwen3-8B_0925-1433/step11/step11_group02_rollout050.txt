import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    max_iter = 500
    learning_rate = 0.02

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        h_eval = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_eval)
        if sum_h > 0:
            h_eval = h_eval * target_sum / sum_h
        corr = np.correlate(h_eval, 1 - h_eval, mode='full')
        max_k = np.argmax(corr)
        k = int(np.clip(max_k, 0, n_points - 1))
        grad = np.zeros_like(best_h)
        for i in range(n_points):
            term1 = 0.0
            if i + k < n_points:
                term1 = (1 - best_h[i + k]) * dx
            term2 = 0.0
            if i - k >= 0:
                term2 = best_h[i - k] * dx
            grad[i] = term1 - term2
        perturb = -learning_rate * grad
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            corr = np.correlate(h_, 1 - h_, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        if res.success and res.fun < best_c5:
