        for i in range(n_points):
            j = i - shift
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
        learning_rate = 0.05 * (0.995 ** iteration)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
