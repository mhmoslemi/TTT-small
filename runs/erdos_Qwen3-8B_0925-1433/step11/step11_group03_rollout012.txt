import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        lag = max_corr_index - (n_points - 1)
        P = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                P[i, j] = 1.0
        x = cp.Variable(n_points)
        obj = cp.Maximize(cp.quad_form(x, P))
        constraints = [cp.sum(x) == target_sum, 0 <= x, x <= 1]
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            continue
        if prob.status != 'OPTIMAL':
            continue
        new_h = x.value
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        h = new_h
    corr = correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    return (h, c5_bound, n_points)
