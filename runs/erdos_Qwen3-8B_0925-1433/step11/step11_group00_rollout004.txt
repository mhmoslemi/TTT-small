        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()

    for iter in range(500):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n_points - 1)

        def objective(x):
            h_ = np.clip(x, 0, 1)
            sum_h_ = np.sum(h_)
            if sum_h_ > 0:
                h_ = h_ * target_sum / sum_h_
            total = 0.0
            for i in range(n_points):
                j = (i + lag) % n_points
                total += h_[i] * (1 - h_[j])
            return total * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        if iter % 10 == 0:
            perturb_std = 0.01
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                new_c5 = np.max(np.correlate(res.x, 1 - res.x, mode='full')) * dx
                if new_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = new_c5
    return (best_h, best_c5, n_points)
