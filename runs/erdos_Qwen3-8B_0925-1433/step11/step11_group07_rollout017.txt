                child = child * target_sum / sum_c
            offspring.append(child)
        
        # Combine and sort
        combined = population + offspring
        combined_fitnesses = []
        for h in combined:
            h_clipped = np.clip(h, 0, 1)
            sum_h = np.sum(h_clipped)
            if sum_h == 0:
                combined_fitnesses.append(float('inf'))
            else:
                h_norm = h_clipped * target_sum / sum_h
                corr = np.correlate(h_norm, 1 - h_norm, mode='full')
                combined_fitnesses.append(np.max(corr) * dx)
        
        sorted_idx = np.argsort(combined_fitnesses)
        population = [combined[i] for i in sorted_idx[:population_size]]
    
    # Local optimization
    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        curr_c5 = res.fun
    else:
        curr_c5 = best_c5
    
    # Final normalization
    best_h_clipped = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h_clipped)
    if sum_best == 0:
        best_h = np.zeros(n_points)
    else:
        best_h = best_h_clipped * target_sum / sum_best
    
    return (best_h, curr_c5, n_points)
