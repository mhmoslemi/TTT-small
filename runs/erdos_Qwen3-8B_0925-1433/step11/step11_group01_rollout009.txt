import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_idx = np.argmax(corr)
        max_k = int(max_idx - (n_points - 1))
        overlap_end = n_points - 1 - max_k
        overlap_start = 0
        if overlap_end < 0:
            overlap_start = 0
            overlap_end = 0
        perturb = np.zeros_like(best_h)
        if overlap_end >= 0:
            perturb[overlap_start:overlap_end + 1] += np.random.normal(0, 0.005, overlap_end - overlap_start + 1)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    for _ in range(100):
        if time.time() - start_time > budget_s:

        perturb = np.random.normal(0, 0.003, n_points)

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
