    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h) if np.sum(h) > 0 else 1.0
    early_stop = 0

    # Genetic algorithm phase
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        h_current = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_current)
        if sum_h > 0:
            h_current = h_current * target_sum / sum_h
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        m = np.argmax(corr)
        lag_k = m - (n_points - 1)
        shifted = np.roll(h_current, lag_k)
        product = h_current * (1 - shifted)
        top_indices = np.argsort(product)[::-1][:int(0.2 * n_points)]
        perturb = np.zeros_like(best_h)
        for idx in top_indices:
            perturb[idx] += np.random.normal(0, 0.03, 1)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        new_c5 = obj(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    # Local refinement with SLSQP
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        if np.abs(obj(best_h) - best_c5) < 1e-10:
            break

    return (best_h, best_c5, n_points)
