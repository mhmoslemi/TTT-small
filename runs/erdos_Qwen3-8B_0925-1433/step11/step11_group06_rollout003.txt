        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        s = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_s = i + s
            if 0 <= i_plus_s < n_points:
                term1 = (1 - h[i_plus_s]) * dx
            else:
                term1 = 0.0
            i_minus_s = i - s
            if 0 <= i_minus_s < n_points:
                term2 = -h[i_minus_s] * dx
            else:
                term2 = 0.0
            grad[i] = term1 + term2
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iter % perturbation_freq == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points)
        learning_rate *= decay_factor

    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return (best_h, best_c5, n_points)
