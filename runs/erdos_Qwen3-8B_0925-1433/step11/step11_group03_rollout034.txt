    best_c5 = compute_c5(best_h)
    time_start = time.time()
    learning_rate = 0.1

    for step in range(2000):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(corr)
        k = m - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            shift = i + k
            if 0 <= shift < n_points:
                gradient[i] = (1 - h[shift]) * dx
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        h = h_new
        learning_rate *= 0.999

    def obj(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    try:
        result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        best_h = result.x
        best_c5 = compute_c5(best_h)
    except:
        pass

    return (best_h, best_c5, n_points)
