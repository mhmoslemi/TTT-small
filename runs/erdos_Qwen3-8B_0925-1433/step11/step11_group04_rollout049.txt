        perturb = np.random.normal(0, 0.15, n_points)
        h = np.clip(initial + perturb, 0, 1)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial.copy()
    best_c5 = eval_c5(best_h)
    gen = 0
    start_time = time.time()
    pop_size = 40
    
    while time.time() - start_time < budget_s and gen < 150:
        gen += 1
        new_pop = []
        elites = sorted(pop, key=eval_c5)[:5]
        new_pop.extend(elites)
        for _ in range(pop_size - 5):
            idxs = np.random.choice(len(pop), 4, replace=False)
            parents = [pop[i] for i in idxs]
            p1, p2 = sorted(parents, key=eval_c5)[:2]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            mut = np.random.normal(0, 0.04, n_points)
            child += mut

    def obj(x):
