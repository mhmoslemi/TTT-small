import numpy as np
from scipy.optimize import minimize
import time
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(v):
        return v * target_sum / np.sum(v) if np.sum(v) > 0 else v

    initial = np.array(initial_h_values, dtype=float)
    initial = normalize(initial)
    population = [initial]
    for _ in range(15):
        perturb = np.random.normal(0, 0.05, n_points)
        h = np.clip(initial + perturb, 0, 1)
        h = normalize(h)
        population.append(h)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial.copy()
    best_c5 = eval_c5(best_h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        elites = sorted(population, key=eval_c5)[:5]
        new_population.extend(elites)
        for _ in range(len(population) - 5):
            idxs = np.random.choice(len(population), 4, replace=False)
            candidates = [population[i] for i in idxs]
            parents = sorted(candidates, key=eval_c5)[:2]
            p1, p2 = parents
            idx = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:idx], p2[idx:]])
            mut = np.random.normal(0, 0.02, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        population = new_population
        current_c5 = eval_c5(population[0])
        if current_c5 < best_c5:
            best_h, best_c5 = population[0], current_c5

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    for h in population[:5]:
        if time.time() - start_time >= budget_s:
            break
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
