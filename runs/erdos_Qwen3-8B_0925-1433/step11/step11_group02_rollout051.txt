import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial solution
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial_sum = np.sum(h_initial)
    if h_initial_sum > 0:
        h_initial = h_initial * target_sum / h_initial_sum

    def objective(solution):
        h = np.clip(solution, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=100, popsize=20, tol=1e-7,
        mutation=(0.5, 1), recombination=0.8,
        polish=True, disp=False
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 20:
        best_h = h_initial
        best_c5 = objective(best_h)
    else:
        best_h = result_de.x
        best_c5 = result_de.fun

    def local_objective(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_local = [(0., 1.) for _ in range(n_points)]

    result_slsqp = minimize(
        local_objective, best_h, method='SLSQP',
        bounds=bounds_local, constraints=cons, tol=1e-7
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    return (best_h, best_c5, n_points)
