import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if hasattr(initial_h_values, '__len__'):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    learning_rate = 0.1
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        shift = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            if 0 <= m + shift < n_points:
                grad[m] += (1 - h[m + shift]) * dx
            if 0 <= m - shift < n_points:
                grad[m] -= h[m - shift] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    return (best_h, best_c5, n_points)
