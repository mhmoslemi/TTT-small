import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 50
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, std):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    def crossover(parent1, parent2):
        if np.random.rand() < 0.8:
            idx = np.random.randint(1, len(parent1)-1)
            child1 = np.concatenate([parent1[:idx], parent2[idx:]])
            child2 = np.concatenate([parent2[:idx], parent1[idx:]])
            return child1, child2
        return parent1, parent2

    def tournament_selection(population, size=3):
        contestants = np.random.choice(population, size, replace=False)
        return min(contestants, key=compute_c5)

    population = [h.copy() for _ in range(50)]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    iterations = 0
    start_time = time.time()
    sigma = 0.02
    elite_ratio = 0.05

    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(len(population)):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, sigma)
            child2 = mutate(child2, sigma)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population
        elites = sorted(population, key=compute_c5)[:int(len(population)*elite_ratio)]
        population = elites + [tournament_selection(population) for _ in range(len(population)-len(elites))]
        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        if iterations % 100 == 0 and sigma > 0.005:
            sigma *= 0.95
        iterations += 1

    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
