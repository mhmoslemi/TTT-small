                                   popsize=15, mutation=(0.5, 1), recombination=0.9,
                                   tol=1e-6, maxiter=100, seed=seed, vectorized=True)
    best_h_de = result.x

    def objective_local(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    bounds_local = [(0.0, 1.0) for _ in range(n_points)]

    result_local = minimize(objective_local, best_h_de, method='SLSQP',
                           bounds=bounds_local, constraints={'type': 'eq', 'fun': constraint},
                           tol=1e-8, options={'maxiter': 100})

    best_h = result_local.x
    best_c5 = objective_local(best_h)

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else h_initial
    corr_initial = np.correlate(h_initial, 1 - h_initial, mode='full')
    c5_initial = np.max(corr_initial) * dx
    if c5_initial < best_c5:
        best_h = h_initial
        best_c5 = c5_initial

    return (np.clip(best_h, 0, 1), best_c5, n_points)
