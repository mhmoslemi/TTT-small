import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial guess
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    
    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()
    
    # Differential evolution for global search
    try:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=20, tol=1e-6, mutation=(0.8, 1.0),
            recombination=0.9, maxiter=100, polish=True, 
            disp=False, vectorized=True
        )
        best_h = result_de.x
        best_c5 = objective(best_h)
    except:
        best_h = h
        best_c5 = objective(h)
    
    # SLSQP refinement
    def objective_slsqp(h_vec):
        h_ = np.clip(h_vec, 0, 1)
        sum_h_ = np.sum(h_)
        if sum_h_ > 0:
            h_ = h_ * target_sum / sum_h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_slsqp = [(0.0, 1.0)] * n_points
    
    try:
        result_slsqp = minimize(
            objective_slsqp, best_h, method='SLSQP',
            bounds=bounds_slsqp, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    except:
        pass
    
    # Early stopping if time exceeds budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
