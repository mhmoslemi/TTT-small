import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def compute_c5(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    def compute_overlap(h, k, dx, n_points):
        total = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            if j < n_points:
                total += h[i] * (1 - h[j]) * dx
        return total

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    step = 0
    max_steps = 1000

    for _ in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        step += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        epsilon = 1e-5
        gradient = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            sum_hp = np.sum(h_plus)
            if sum_hp > 0:
                h_plus = h_plus * target_sum / sum_hp
            o_plus = compute_overlap(h_plus, lag, dx, n_points)
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            sum_hm = np.sum(h_minus)
            if sum_hm > 0:
                h_minus = h_minus * target_sum / sum_hm
            o_minus = compute_overlap(h_minus, lag, dx, n_points)
            gradient[j] = (o_plus - o_minus) / (2 * epsilon)
        alpha = 0.05 * (1 - step / max_steps)
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return compute_c5(h)

    bounds = [(0, 1)] * n_points

    if sum_best > 0:
        best_h = best_h * target_sum / sum_best

    return (best_h, best_c5, n_points)
