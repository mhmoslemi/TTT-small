            overlap_end = n_points - 1
        overlap_start = max(0, overlap_start)
        overlap_end = min(n_points - 1, overlap_end)
        # Targeted mutation on overlapping region
        perturb = np.zeros(n_points)
        for i in range(overlap_start, overlap_end + 1):
            perturb[i] += np.random.normal(0, 0.15, 1)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.zeros(n_points)
        new_c5 = compute_c5(new_h, dx)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5

    # Local SLSQP optimization
    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n_points)
