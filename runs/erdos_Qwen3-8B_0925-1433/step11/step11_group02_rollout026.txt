from scipy.optimize import minimize
import time

            h = h * target_sum / s
        return np.clip(h, 0, 1)

    def obj(x):
        x = scale_h(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points

    best_h = h.copy()

        if time.time() - start_time > budget_s:

        h_eval = scale_h(best_h)

        max_k = np.argmax(corr)

        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = scale_h(new_h)
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
