def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    n_particles = 50
    inertia_weight = 0.7
    c1, c2 = 1.5, 1.5
    max_iter = 500
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))

    for i in range(n_particles):
        particles[i] = base_h + np.random.normal(0, 0.05, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        sum_p = np.sum(particles[i])
        particles[i] = particles[i] * target_sum / sum_p if sum_p > 0 else base_h

    pbest_positions = particles.copy()
    pbest_scores = np.zeros(n_particles)
    for i in range(n_particles):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        pbest_scores[i] = np.max(corr) * dx
        pbest_positions[i] = particles[i].copy()
    gbest_score = np.min(pbest_scores)
    gbest_position = pbest_positions[np.argmin(pbest_scores)]

    best_c5 = gbest_score
    best_h = gbest_position
    start_time = time.time()

    for _ in range(max_iter):
