import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()

    # Compute initial maximum overlap
    best_c5 = 0.0
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            overlap += h[i] * (1 - h[(i + k) % n_points]) * dx
        best_c5 = max(best_c5, overlap)

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Compute all lags' overlaps
        overlaps = []
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                overlap += h[i] * (1 - h[(i + k) % n_points]) * dx
            overlaps.append(overlap)
        k_max = np.argmax(overlaps)
        current_max = overlaps[k_max]
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        # Define objective to minimize worst lag
        def obj(x):
            h_clipped = np.clip(x, 0, 1)
            sum_h = np.sum(h_clipped)
            if sum_h > 0:
                h_clipped = h_clipped * target_sum / sum_h
            overlap = 0.0
            for i in range(n_points):
                overlap += h_clipped[i] * (1 - h_clipped[(i + k_max) % n_points]) * dx
            return overlap

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            # Update best_c5 with new overlaps
            overlaps_new = []
            for k in range(n_points):
                overlap = 0.0
                for i in range(n_points):
                    overlap += h[i] * (1 - h[(i + k) % n_points]) * dx
                overlaps_new.append(overlap)
            current_max_new = max(overlaps_new)
            if current_max_new < best_c5:
                best_c5 = current_max_new
                best_h = h.copy()

        # Add random perturbations every 10 iterations
        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h

    return (best_h, best_c5, n_points)
