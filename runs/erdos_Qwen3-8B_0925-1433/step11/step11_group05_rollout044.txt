import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        h = np.random.rand(45)
        h = h * (45/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    population = [perturb(best_h, 0.05) for _ in range(50)]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        correlations = [compute_c5(p) for p in population]
        elite_idx = np.argsort(correlations)[:5]
        new_population = []
        for _ in range(50):
            parent = population[np.random.choice(elite_idx)]
            child = perturb(parent, 0.02)
            new_population.append(child)
        population = new_population
        if np.min(correlations) < best_c5:
            best_c5 = np.min(correlations)
            best_h = population[np.argmin(correlations)].copy()

    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
