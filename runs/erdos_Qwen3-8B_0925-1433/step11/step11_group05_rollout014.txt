        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 45
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.01):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    max_iter = 500
    pop_size = 5
    population = [perturb(best_h) for _ in range(pop_size)]

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for _ in range(pop_size):
            candidate = population[np.random.randint(pop_size)]
            new_h = perturb(candidate)
            e_old = compute_c5(candidate)
            e_new = compute_c5(new_h)
            if e_new < e_old or np.random.rand() < np.exp(-(e_new - e_old)/temp):
                population[_] = new_h
                if e_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = e_new
        temp *= cooling_rate
        iterations += 1
        if iterations % 100 == 0:
            std = max(0.005, 0.02 * (temp / 1.0))
            population = [perturb(population[i], std) for i in range(pop_size)]

    # Final refinement with L-BFGS
    def objective(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
