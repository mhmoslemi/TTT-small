import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    population_size = 10
    population = []
    for _ in range(population_size):
        perturb = np.random.normal(0, 0.15, n_points)
        candidate = h + perturb
        candidate = np.clip(candidate, 0, 1)
        sum_cand = np.sum(candidate)
        if sum_cand > 0:
            candidate = candidate * target_sum / sum_cand
        population.append(candidate)

    def compute_c5(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    T = 1.0
    cooling_rate = 0.998
    max_temp_steps = 1200
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for temp_step in range(max_temp_steps):
        if time.time() - start_time > budget_s:
            break
        idx = np.random.randint(population_size)
        candidate = population[idx].copy()
        delta = np.random.normal(0, 0.1, n_points)
        new_h = candidate + delta
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = candidate

        E_old = compute_c5(candidate)
        E_new = compute_c5(new_h)

        if E_new < E_old:
            population.append(new_h)
            population.pop(idx)
            if E_new < best_c5:
                best_h = new_h.copy()
                best_c5 = E_new
        else:
            prob = np.exp((E_old - E_new) / T)
            if np.random.rand() < prob:
                population.append(new_h)
                population.pop(idx)
                if E_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = E_new

        T *= cooling_rate

    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:

    return (best_h, best_c5, n_points)
