import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    population_size = 20
    population = []
    
    # Initialize with parent and random variations
    h_initial = np.clip(initial_h_values, 0, 1).copy()
    h_sum = np.sum(h_initial)
    if h_sum != 0:
        h_initial = h_initial * target_sum / h_sum
    population.append(h_initial)
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = (h * target_sum) / np.sum(h)
        population.append(np.clip(h, 0, 1))
    
    best_solution = population[0].copy()
    best_fitness = float('inf')
    tournament_size = 5
    mutation_rate = 0.1
    max_generations = 500
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = []
        for h in population:
            h_clipped = np.clip(h, 0, 1)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        
        # Track best
        best_idx = np.argmin(fitness)
        current_best = population[best_idx].copy()
        current_best_fit = fitness[best_idx]
        if current_best_fit < best_fitness:
            best_fitness = current_best_fit
            best_solution = current_best.copy()
        
        # Generate new population
        new_population = []
        for _ in range(population_size // 2):
            # Tournament selection
            idxs = np.random.choice(len(population), tournament_size, replace=False)
            best_idx_t = idxs[np.argmin(fitness[idxs])]
            p1 = population[best_idx_t].copy()
            
            idxs = np.random.choice(len(population), tournament_size, replace=False)
            best_idx_t = idxs[np.argmin(fitness[idxs])]
            p2 = population[best_idx_t].copy()
            
            # Crossover
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.01, n_points)
                child += noise
            
            # Clip and normalize
            child_clipped = np.clip(child, 0, 1)
            child_sum = np.sum(child_clipped)
            if child_sum == 0:
                new_population.append(np.zeros(n_points))
            else:
                new_population.append(child_clipped * target_sum / child_sum)
        
        # Combine and select top individuals

        combined_fitness = []
        for h in combined:
            h_clipped = np.clip(h, 0, 1)

            combined_fitness.append(np.max(corr) * (2.0 / n_points))

        best_solution = best_solution * target_sum / h_sum

        best_solution = np.zeros(n_points)
