import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        h = h * target_sum / s if s > 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 20
    population = [initial_h_values.copy()]
    for _ in range(population_size - 1):
        rand_h = np.random.rand(n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    gen = 0
    
    while time.time() - start_time < budget_s:
        gen += 1
        
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        best_idx = int(np.argmin(fitness))
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Generate offspring
        offspring = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(range(population_size), 2, replace=False)
            parent1 = population[p1]
            parent2 = population[p2]
            alpha = np.random.rand()
            child = parent1 * alpha + parent2 * (1 - alpha)
            offspring.append(child)
        
        # Mutate offspring
        for child in offspring:
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                s = np.sum(child)
                child = child * target_sum / s if s > 0 else child
        
        # Combine and select new population
        combined = population + offspring
        fitness_combined = [compute_c5(ind) for ind in combined]
        sorted_indices = np.argsort(fitness_combined)
        population = [combined[i] for i in sorted_indices[:population_size]]
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    best_h = best_h * target_sum / s if s > 0 else best_h
    
    return (best_h, best_c5, n_points)
