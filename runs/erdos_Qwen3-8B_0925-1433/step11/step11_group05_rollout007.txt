        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def perturb(h, std=0.03):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    population_size = 5
    population = [perturb(best_h) for _ in range(population_size)]
    elite = best_h.copy()

    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    max_iter = 500
    while temp > 1e-4 and time.time() - start_time < budget_s - 60:
        for _ in range(population_size):
            candidate = population[np.random.randint(population_size)]
            new_h = perturb(candidate)
            e_old = compute_c5(candidate)

            if e_new < e_old or np.random.rand() < np.exp(-(e_new - e_old)/temp):
                population[_] = new_h.copy()
                if e_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = e_new
                    elite = new_h.copy()
        temp *= cooling_rate
        iterations += 1
        if iterations % 100 == 0:

    def objective(x):
