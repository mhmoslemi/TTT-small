from scipy.optimize import minimize
from scipy.optimize import COBYLA
import time

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.02):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    def crossover(parent1, parent2):
        mid = len(parent1) // 2
        child1 = np.concatenate([parent1[:mid], parent2[mid:]])
        child2 = np.concatenate([parent2[:mid], parent1[mid:]])
        sum_c1 = np.sum(child1)
        sum_c2 = np.sum(child2)
        if sum_c1 > 0:
            child1 = child1 * target_sum / sum_c1
        if sum_c2 > 0:
            child2 = child2 * target_sum / sum_c2
        return child1, child2

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    population = [perturb(best_h) for _ in range(50)]
    start_time = time.time()

    while time.time() - start_time < budget_s:

            break
