    learning_rate = 0.1
    decay_factor = 0.995
    convergence_threshold = 1e-4
    time_start = time.time()

    for step in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        lag = max_index - (n_points - 1)

        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
            else:
                grad[i] = 0.0

        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (target_sum / current_sum)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if np.abs(current_c5 - best_c5) < convergence_threshold:
            break
        learning_rate *= decay_factor

    h = np.clip(best_h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (target_sum / h_sum)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, best_c5, n_points)
