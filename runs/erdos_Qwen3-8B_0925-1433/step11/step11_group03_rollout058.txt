        max_index = np.argmax(corr)
        s = max_index - (n_points - 1)

        grad = np.zeros_like(h)
        for j in range(n_points):
            shifted_j = j + s
            if 0 <= shifted_j < n_points:
                grad[j] = (1 - h[shifted_j]) * dx

        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    h = np.clip(best_h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return (h, np.max(np.correlate(h, 1 - h, mode='full')) * dx, n_points)
