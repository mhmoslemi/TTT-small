import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1).copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_c5(h_vals):
        max_ov = 0.0
        for k in range(n_points):
            ov = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                ov += h_vals[i] * (1 - h_vals[j]) * dx
            max_ov = max(max_ov, ov)
        return max_ov

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for iter in range(500):
        if time.time() - start_time > budget_s:
            break
        # Compute overlaps and find worst lag
        max_ov = 0.0
        k_max = 0
        for k in range(n_points):
            ov = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                ov += h[i] * (1 - h[j]) * dx
            if ov > max_ov:
                max_ov = ov
                k_max = k
        # Define objective for this lag
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            ov = 0.0
            for i in range(n_points):
                j = (i + k_max) % n_points
                ov += h_[i] * (1 - h_[j]) * dx
            return ov
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h = new_h * target_sum / sum_new
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        # Random perturbation every 10 iterations
        if iter % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
    return (best_h, best_c5, n_points)
