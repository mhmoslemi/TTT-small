import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0

    # Global phase with targeted mutations
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)  # Compute lag from index
        if lag > 0:
            contrib_indices = np.arange(n_points - lag)
            contrib_values = h_[contrib_indices] * (1 - h_[contrib_indices + lag])
        elif lag < 0:
            lag_abs = -lag
            contrib_indices = np.arange(n_points - lag_abs)
            contrib_values = h_[contrib_indices + lag_abs] * (1 - h_[contrib_indices])
        else:
            contrib_indices = np.arange(n_points)
            contrib_values = h_[contrib_indices] * (1 - h_[contrib_indices])
        sorted_indices = np.argsort(contrib_values)[::-1]
        m = max(1, int(0.2 * len(sorted_indices)))
        top_indices = sorted_indices[:m]
        perturb = np.zeros_like(best_h)
        perturb[top_indices] += np.random.normal(0, 0.05, len(top_indices))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    # Local refinement phase with SLSQP
    for _ in range(150):
        if time.time() - start_time > budget_s:

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
