        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def grad_func(h_vec, beta):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full') * dx
        exp_corr = np.exp(beta * corr)
        sum_exp = np.sum(exp_corr)
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            sum_hp = np.sum(h_plus)
            if sum_hp > 0:
                h_plus = h_plus * target_sum / sum_hp
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            exp_corr_plus = np.exp(beta * corr_plus)
            smooth_plus = np.log(np.sum(exp_corr_plus)) / beta
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            sum_hm = np.sum(h_minus)
            if sum_hm > 0:
                h_minus = h_minus * target_sum / sum_hm
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx
            exp_corr_minus = np.exp(beta * corr_minus)
            smooth_minus = np.log(np.sum(exp_corr_minus)) / beta
            
            grad[i] = (smooth_plus - smooth_minus) / (2 * epsilon)
        return grad
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    current_h = h.copy()
    beta = 50
    learning_rate = 0.05
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        grad = grad_func(current_h, beta)
        current_h = current_h - learning_rate * grad
        current_h = np.clip(current_h, 0, 1)
        sum_current = np.sum(current_h)
        if sum_current > 0:
            current_h = current_h * target_sum / sum_current
        else:
            current_h = np.zeros(n_points)
        current_c5 = obj(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    final_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(final_h)
    if sum_h > 0:
        final_h = final_h * target_sum / sum_h
    else:
        final_h = np.zeros(n_points)
    final_c5 = obj(final_h)
    
    return (final_h, final_c5, n_points)
