            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            return corr[max_index] * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

        if res.success:
            new_h = res.x
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5
                h = new_h
        else:
            h = best_h.copy()

        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h

    return (best_h, best_c5, n_points)
