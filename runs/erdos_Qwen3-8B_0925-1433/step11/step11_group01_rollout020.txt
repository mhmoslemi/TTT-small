from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    best_c5_val = best_c5
    early_stop = 0
    pop_size = 5
    elites = [best_h]

    for generation in range(300):
        if time.time() - start_time > budget_s:
            break
        if len(elites) < pop_size:
            new_h = np.random.rand(n_points)
            new_h = new_h * target_sum / np.sum(new_h)
            elites.append(new_h)
        else:
            # Select top 2 elites
            sorted_elites = sorted(elites, key=obj)
            top_h = sorted_elites[:2]
            elite = top_h[0]
            # Compute current max lag
            h_ = np.clip(elite, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')

            # Identify contributors to max lag
            k = max_k
            overlap_start = max(0, int(2.0 * n_points / (n_points + 1) - k * n_points / (n_points + 1)) - 1)

            contrib = h_[overlap_start:overlap_end+1] * (1 - h_[overlap_start + k:overlap_end + k + 1])

            # Apply sparse mutation

            new_h = np.clip(elite + perturb, 0, 1)

            elites.append(new_h)

        # Evaluate elites

    # Local optimization
