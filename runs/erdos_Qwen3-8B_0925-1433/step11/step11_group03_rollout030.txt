        lag_index = np.argmax(corr)
        lag = lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus = (j + lag) % n_points
            j_minus = (j - lag) % n_points
            grad[j] = (1 - h[j_plus]) - h[j_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        new_corr = np.correlate(h, 1 - h, mode="full")
        new_max = np.max(new_corr)
        if new_max < max_corr:
            best_h = h.copy()
            best_c5 = new_max * dx
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 100:
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
                stagnation = 0
        learning_rate *= decay_rate
        if iter_count % 100 == 0:
            h = best_h.copy()
            stagnation = 0

    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    return h, c5, n_points
