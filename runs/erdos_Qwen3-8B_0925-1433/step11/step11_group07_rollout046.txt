    initial_patterns.append(periodic)

    interleaved = np.zeros(n_points)
    for i in range(n_points):
        interleaved[i] = 1 if i % 2 == 0 else 0
    initial_patterns.append(interleaved)

    permuted = np.random.permutation(initial_h_values)
    initial_patterns.append(permuted)

    def obj(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    start_time = time.time()

    for h0 in initial_patterns:
        if time.time() - start_time > budget_s:
            break
        h = np.clip(h0, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    return (best_h, best_c5, n_points)
