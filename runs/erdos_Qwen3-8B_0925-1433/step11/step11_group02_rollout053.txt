import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def objective(x):
        h = np.clip(x, 0, 1)
        s = np.sum(h)
        if s == 0:
            return float('inf')
        h_scaled = h * target_sum / s
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        polish=True,
        seed=seed
    )
    elapsed = time.time() - start_time
    best_h = result.x
    best_c5 = result.fun
    
    if elapsed < budget_s:
        def s_obj(x):
            x = np.clip(x, 0, 1)
            s = np.sum(x)
            if s == 0:
                return float('inf')
            x_scaled = x * target_sum / s
            corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(s_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        best_h, best_c5 = result.x, result.fun
    
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if s > 0:
        best_h = best_h * target_sum / s
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
