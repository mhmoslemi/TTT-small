import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-6:
        h = h * target_sum / sum_h

    learning_rate = 0.1
    decay_rate = 0.995
    max_iterations = 1000
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        gradient = np.zeros_like(h)
        if k > 0:
            gradient[:n_points - k] = 1 - h[k:]
        elif k < 0:
            gradient[-k:] = 1 - h[:-k]
        else:
