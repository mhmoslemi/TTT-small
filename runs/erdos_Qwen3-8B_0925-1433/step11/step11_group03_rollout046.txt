    start_time = time.time()
    learning_rate = 0.1

    max_iterations = 1000

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize h
    if initial_h_values is not None:
        h = initial_h_values * target_sum / np.sum(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 1000

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (n_points - 1)
        j = (np.arange(n_points) + s) % n_points
        grad = 1 - h[j]
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum == 0:
            h_new = np.zeros(n_points)
        else:
            h_new = h_new * target_sum / h_new_sum
        h = h_new
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate
        if _ % 100 == 0:
            print(f"Iteration {_}, best C5: {best_c5}")

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    result = minimize(lambda x: compute_c5(x), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
