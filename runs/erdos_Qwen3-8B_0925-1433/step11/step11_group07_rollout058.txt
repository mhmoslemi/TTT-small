        temperature *= cooling_factor
    
    # Gradient-Based Optimization
    def smooth_obj(h_vec, beta=100):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        exp_corr = np.exp(beta * corr)
        return np.log(np.sum(exp_corr)) / beta
    
    def grad_smooth(h_vec, beta=100):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        exp_corr = np.exp(beta * corr)
        sum_exp = np.sum(exp_corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            eps = 1e-6
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            exp_plus = np.exp(beta * corr_plus)
            smooth_plus = np.log(np.sum(exp_plus)) / beta
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            exp_minus = np.exp(beta * corr_minus)
            smooth_minus = np.log(np.sum(exp_minus)) / beta
            grad[i] = (smooth_plus - smooth_minus) / (2 * eps)
        return grad
    
    res = minimize(
        fun=lambda x: smooth_obj(x, beta=100),
        x0=best_h,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        jac=grad_smooth,
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    
    if res.success:
        best_h = res.x
        best_c5 = obj(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    
    return (best_h, best_c5, n_points)
