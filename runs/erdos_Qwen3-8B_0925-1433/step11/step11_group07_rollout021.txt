    else:
        h = np.zeros(n_points)

        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    current_h = h.copy()
    
    initial_c5 = best_c5
    temperature = initial_c5 * 0.15
    cooling_factor = 0.995
    max_sa_iterations = 2000
    start_time = time.time()
    
    for _ in range(max_sa_iterations):
        if time.time() - start_time > budget_s:
            break

            current_h = candidate.copy()
            if c5_candidate < best_c5:

        temperature *= cooling_factor
