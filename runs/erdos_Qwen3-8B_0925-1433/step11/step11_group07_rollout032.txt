import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate initial population
    population = []
    # Add initial_h_values
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    population.append(h)
    
    # Add shifted version
    h_shifted = np.roll(h, 1)
    if np.sum(h_shifted) > 0:
        h_shifted = h_shifted * target_sum / np.sum(h_shifted)
    else:
        h_shifted = np.zeros(n_points)
    population.append(h_shifted)
    
    # Add random pattern
    h_random = np.random.rand(n_points)
    h_random = h_random * target_sum / np.sum(h_random)
    h_random = np.clip(h_random, 0, 1)
    population.append(h_random)
    
    # Add periodic pattern
    h_periodic = np.zeros(n_points)
    for i in range(n_points):
        h_periodic[i] = 1 if (i % 5 == 0) else 0
    if np.sum(h_periodic) > 0:
        h_periodic = h_periodic * target_sum / np.sum(h_periodic)
    else:
        h_periodic = np.zeros(n_points)
    h_periodic = np.clip(h_periodic, 0, 1)
    population.append(h_periodic)
    
    # Fill remaining population with random patterns
    while len(population) < 20:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Define evaluation function
    def evaluate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    # Genetic algorithm parameters
    num_generations = 500
    population_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Run genetic algorithm
    for _ in range(num_generations):
        if time.time() - start_time >= budget_s:
            break
        # Evaluate all individuals
        fitness = [evaluate(ind) for ind in population]
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population, size=5)
            best = min(tournament, key=evaluate)
            parents.append(best)
        
        # Generate offspring
        offspring = []
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i + 1]
            if np.random.rand() < crossover_rate:
                # Arithmetic crossover
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 0:
                    child = child * target_sum / sum_child
                else:
                    child = p1.copy()
                offspring.append(child)
            else:
                # Mutation
                child = p1.copy()
                noise = np.random.normal(0, 0.01, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 0:
                    child = child * target_sum / sum_child
                else:
                    child = p1.copy()
                offspring.append(child)
        
        # Replace worst half of population
        combined = population + offspring
        fitness_combined = [evaluate(ind) for ind in combined]
        indices = np.argsort(fitness_combined)
        population = [combined[i] for i in indices[:population_size]]
    
    # Final evaluation
    fitness = [evaluate(ind) for ind in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()
    best_c5 = fitness[best_index]
    
    return (best_h, best_c5, n_points)
