        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evolve(h, std=0.01):

        new_h = np.clip(h + noise, 0, 1)

        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h
