import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = target_sum / sum_h
        return h_clipped * scale

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h = h_clipped * (target_sum / sum_h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient_for_k(h, k):

        eps = 1e-4
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)

            h_norm = h_clipped * (target_sum / sum_h)
        else:
            h_norm = np.zeros_like(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')

            h_plus = np.copy(h_norm)

            if sum_h_plus > 0:

                h_plus_norm = np.zeros_like(h_plus)
            corr_plus = np.correlate(h_plus_norm, 1 - h_plus_norm, mode='full')
            grad[i] = (corr_plus[k] - max_corr) / eps

    best_h = h.copy()
