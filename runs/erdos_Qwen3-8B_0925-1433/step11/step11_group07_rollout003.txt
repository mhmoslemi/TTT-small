        candidate = current_h + perturb
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > 0:
            candidate = candidate * target_sum / sum_candidate
        else:
            candidate = current_h.copy()
        c5_candidate = obj(candidate)
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_h = candidate.copy()
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
        temperature *= cooling_factor
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    
    return (best_h, best_c5, n_points)
