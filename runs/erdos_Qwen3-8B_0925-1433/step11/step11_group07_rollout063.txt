        ptm = ptm * target_sum / np.sum(ptm)
    else:
        ptm = np.zeros(n_points)
    candidates.append(ptm)
    
    # Generate interleaved patterns
    m1 = 8
    m2 = 16
    for i in range(n_points):
        if i % 2 == 0:
            candidates.append(np.array([1 if i % m1 < m1//2 else 0 for i in range(n_points)]))
        else:
            candidates.append(np.array([1 if i % m2 < m2//2 else 0 for i in range(n_points)]))
    candidates = list(set([tuple(c) for c in candidates]))[:5]  # deduplicate and limit
    
    best_c5 = float('inf')
    best_h = None
    
    for h_start in candidates:
        h_start = np.array(h_start)
        h_start = np.clip(h_start, 0, 1)
        if np.sum(h_start) > 0:
            h_start = h_start * target_sum / np.sum(h_start)
        else:
            h_start = np.zeros(n_points)
        
        def obj(x):
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x) if np.sum(x) > 0 else x
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        
        try:
            res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x
        except:
            pass
        
        if time.time() - time.time() > budget_s:
            break
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    
    return (best_h, best_c5, n_points)
