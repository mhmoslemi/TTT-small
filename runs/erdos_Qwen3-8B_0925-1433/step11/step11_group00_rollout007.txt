    bounds = [(1e-6, 1 - 1e-6) for _ in range(k_half)] + [(0, 1)]
    cons = [{'type': 'eq', 'fun': lambda vars: np.sum(vars[:k_half]) - 1}]
    initial_guess = initial_widths + [initial_a]

    start_time = time.time()
    res = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = np.zeros(n_points)
    best_a = 0.5
    best_widths = [1.0 / k_half] * k_half
    if res.success:
        best_widths = res.x[:-1]
        best_a = res.x[-1]
        block_start = 0.0
        blocks = []
        for i in range(k_half):
            block_end = block_start + best_widths[i]
            val = 'a' if i % 2 == 0 else 'b'
            blocks.append((block_start, block_end, val))
            block_start = block_end
        for i in range(k_half-1, -1, -1):
            block_end = block_start + best_widths[i]
            val = 'a' if i % 2 == 0 else 'b'
            blocks.append((block_start, block_end, val))
            block_start = block_end
        for j in range(n_points):
            x = j * dx
            for (start, end, val) in blocks:
                if start <= x < end:
                    if val == 'a':
                        best_h[j] = best_a
                    else:
                        best_h[j] = 1.0 - best_a
                    break
    else:
        best_h = h_initial
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    return (best_h, best_c5, n_points)
