from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 50
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.02):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    population = [perturb(best_h) for _ in range(10)]
    temp = 1.0
    cooling_rate = 0.997
    max_iter = 1000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            candidate = population[i]
            new_h = perturb(candidate, std=0.025)
            e_old = compute_c5(candidate)
            e_new = compute_c5(new_h)
            if e_new < e_old or np.random.rand() < np.exp(-(e_new - e_old)/temp):
                population[i] = new_h
                if e_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = e_new
        temp *= cooling_rate

    bounds = Bounds(0, 1)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(compute_c5, best_h, method='COBYLA', bounds=bounds, constraints=constraints, tol=1e-6)
