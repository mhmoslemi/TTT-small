            if 0 <= idx < n_points:
                h[idx] = 1
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        return h

    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h > 0:
        h_initial = h_initial * target_sum / sum_h
    else:
        h_initial = np.zeros(n_points)

    h_low = generate_low_discrepancy(n_points)

    def compute_c5(h):
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    c5_initial = compute_c5(h_initial)
    c5_low = compute_c5(h_low)
    best_h = h_initial if c5_initial < c5_low else h_low

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    start_time = time.time()

    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            best_c5 = compute_c5(best_h)
    except:
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best

    return (best_h, best_c5, n_points)
