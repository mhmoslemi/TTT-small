        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Phase 3: Symmetry-aware refinement with COBYLA
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb_std = 0.0005 * (0.9 ** _)
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final symmetry check
    symmetric_h = np.clip(best_h - np.flip(best_h) + 1.0, 0, 1)
    symmetric_h = symmetric_h * target_sum / np.sum(symmetric_h) if np.sum(symmetric_h) > 0 else best_h
    final_c5 = obj(symmetric_h)
    if final_c5 < best_c5:
        best_h, best_c5 = symmetric_h, final_c5

    return (best_h, best_c5, n_points)
