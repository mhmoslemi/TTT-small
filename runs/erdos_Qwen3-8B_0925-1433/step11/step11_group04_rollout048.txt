    def normalize(v):

    initial = np.array(initial_h_values, dtype=float)
    initial = normalize(initial)

    for _ in range(19):
        perturb = np.random.normal(0, 0.1, n_points)
        h = np.clip(initial + perturb, 0, 1)
        h = normalize(h)
        pop.append(h)

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial.copy()
    best_c5 = eval_c5(best_h)
    gen = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and gen < 500:
        gen += 1
        new_pop = []
        elites = sorted(pop, key=eval_c5)[:5]
        new_pop.extend(elites)
        for _ in range(len(pop) - 5):
            idxs = np.random.choice(len(pop), 4, replace=False)
            candidates = [pop[i] for i in idxs]
            parents = sorted(candidates, key=eval_c5)[:2]
            p1, p2 = parents
            idx = np.random.randint(1, n_points - 1)

            new_pop.append(child)
        pop = new_pop

    def obj(x):
