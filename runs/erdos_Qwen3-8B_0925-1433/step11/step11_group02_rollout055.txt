            p_new = np.zeros_like(p_clip)
        corr = np.correlate(p_new, 1 - p_new, mode='full')
        pbest_values.append(np.max(corr) * dx)
    gbest = pbest[np.argmin(pbest_values)]
    gbest_val = min(pbest_values)
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            p = particles[i]
            p_clip = np.clip(p, 0, 1)
            sum_p = np.sum(p_clip)
            if sum_p > 0:
                p_new = p_clip * target_sum / sum_p
            else:
                p_new = np.zeros_like(p_clip)
            corr = np.correlate(p_new, 1 - p_new, mode='full')
            current_val = np.max(corr) * dx
            if current_val < pbest_values[i]:
                pbest[i] = p_new.copy()
                pbest_values[i] = current_val
        current_gbest_val = min(pbest_values)
        if current_gbest_val < gbest_val:
            gbest_val = current_gbest_val
            gbest = pbest[np.argmin(pbest_values)]
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * inertia + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + vel
            p = particles[i]
            p_clip = np.clip(p, 0, 1)
            sum_p = np.sum(p_clip)
            if sum_p > 0:
                particles[i] = p_clip * target_sum / sum_p
            else:
                particles[i] = np.zeros_like(p_clip)
    best_h = gbest
    best_c5 = gbest_val
    return (best_h, best_c5, n_points)
