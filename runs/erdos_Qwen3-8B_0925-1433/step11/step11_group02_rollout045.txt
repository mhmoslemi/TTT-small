import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        h = np.array(initial_h_values)
    except NameError:
        h = np.repeat(0.5, 45)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    
    def obj(x):
        h_ = x
        sum_h = np.sum(h_)
        if sum_h == 0:
            return float('inf')
        h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    start_time = time.time()
    result_de = differential_evolution(obj, bounds, strategy='best1bin',
                                      popsize=20, tol=1e-6, mutation=(0.8, 1),
                                      recombination=0.9, maxiter=100, disp=False)
    
    if time.time() - start_time > budget_s:
        best_h = result_de.x
        best_c5 = obj(best_h)
        return (best_h, best_c5, n_points)
    
    best_h_de = result_de.x
    best_c5 = obj(best_h_de)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_slsqp = minimize(obj, best_h_de, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    
    if time.time() - start_time > budget_s:
        best_c5 = min(best_c5, obj(res_slsqp.x))
        best_h = best_h_de if best_c5 < obj(res_slsqp.x) else res_slsqp.x

    best_c5 = obj(res_slsqp.x)
    best_h = res_slsqp.x
