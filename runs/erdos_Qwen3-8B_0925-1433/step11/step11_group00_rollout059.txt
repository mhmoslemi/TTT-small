import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_cross_corr(h_):
        return np.correlate(h_, 1 - h_, mode='full')

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.max(compute_cross_corr(best_h)) * dx

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        cross_corr = compute_cross_corr(best_h)
        max_idx = np.argmax(cross_corr)
        def objective(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = compute_cross_corr(h_)
            return corr[max_idx]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_cross_corr = compute_cross_corr(new_h)
            new_c5 = np.max(new_cross_corr) * dx
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    return (best_h, best_c5, n_points)
