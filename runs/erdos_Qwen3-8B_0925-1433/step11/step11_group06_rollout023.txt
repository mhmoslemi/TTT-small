def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    h_initial = h_initial * (target_sum / m) if m > 0 else np.zeros(n_points)

    def compute_c5(h_candidate):
