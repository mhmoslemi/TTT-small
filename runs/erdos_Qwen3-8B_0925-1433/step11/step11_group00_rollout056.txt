import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        one_minus_h = 1 - h_
        corr = np.correlate(h_, one_minus_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for phase in range(3):
        if time.time() - start_time > budget_s:
            break
        if phase == 0:
            method = 'L-BFGS-B'
            maxiter = 50
            tol = 1e-5
        elif phase == 1:
            method = 'COBYLA'
            maxiter = 30
            tol = 1e-6
        else:
            method = 'SLSQP'
            maxiter = 20
            tol = 1e-8

        for i in range(maxiter):
            if time.time() - start_time > budget_s:
                break
            scale = 0.1 + 0.1 * (phase) / 3.0
            perturb = np.random.normal(0, scale, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method=method, bounds=bounds, constraints=cons, tol=tol, options={'maxiter': 1})
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
