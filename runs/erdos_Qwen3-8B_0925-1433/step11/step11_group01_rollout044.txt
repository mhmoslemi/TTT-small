        kron = max_k_idx - (n_points - 1)
        if kron > 0:
            overlap_end = int(n_points - kron - 1)
            if overlap_end >= 0:
                overlap_vals = h_[:overlap_end+1] * (1 - h_[kron:overlap_end + kron + 1])
                idx_max = np.argmax(overlap_vals)
                perturb = np.zeros_like(best_h)
                perturb[:overlap_end+1] += np.random.normal(0, 0.01, overlap_end+1)
                new_h = np.clip(best_h + perturb, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
                res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
                if res.success and res.fun < best_c5:
                    best_h, best_c5 = res.x, res.fun

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
