import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def compute_c5(h):
        h_ = np.clip(h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    current_temp = 1.0
    for step in range(150):
        if time.time() - start_time > budget_s:
            break
        # Perturb best_h
        perturb_indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        perturb = np.zeros(n_points)
        perturb[perturb_indices] += np.random.normal(0, 0.05, len(perturb_indices))
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * target_sum / new_sum
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / current_temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        current_temp *= 0.995

    # Local optimization with SLSQP
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h, best_c5 = res.x, res.fun

    # Final verification
    h_final = np.clip(best_h, 0, 1)
    sum_final = np.sum(h_final)
    if sum_final > 0:
        h_final = h_final * target_sum / sum_final
    final_c5 = compute_c5(h_final)
