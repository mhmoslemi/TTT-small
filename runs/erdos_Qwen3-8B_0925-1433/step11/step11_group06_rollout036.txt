import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)
    best_h, best_c5 = h.copy(), compute_c5(h)
    learning_rate = 0.05
    decay_rate = 0.999
    max_iterations = 500
    iteration = 0
    start_time = time.time()

    while iteration < max_iterations and time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        indices = (np.arange(n_points) + lag) % n_points
        gradient = (1 - h[indices]) * dx
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new > 0:
            h_new = h_new * target_sum / sum_new
        else:
            h_new = h.copy()
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5, best_h = new_c5, h_new.copy()
        learning_rate *= decay_rate
        h = h_new
        if iteration % 100 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            sum_perturb = np.sum(h)
            if sum_perturb > 0:
                h = h * target_sum / sum_perturb
            else:
                h = h.copy()
        iteration += 1

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    return (best_h, best_c5, n_points)
