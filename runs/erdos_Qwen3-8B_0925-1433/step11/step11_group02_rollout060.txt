import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial guess
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    
    def obj(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    start_time = time.time()
    
    # Differential evolution for global search
    try:
        result = differential_evolution(obj, bounds, strategy='best1bin',
                                       popsize=20, tol=1e-6, mutation=(0.5, 1),
                                       recombination=0.9, seed=seed,
                                       maxiter=100, dispatches=1)
        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = h_initial
        best_c5 = obj(h_initial)
    
    # Time check after DE
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (np.clip(best_h, 0, 1), best_c5, n_points)
    
    # SLSQP refinement
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_slsqp = [(0, 1)] * n_points
    
    result_slsqp = minimize(obj, best_h, method='SLSQP',
                            bounds=bounds_slsqp, constraints=cons, tol=1e-8)
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Final constraints check and clip
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * target_sum / np.sum(best_h)
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(final_corr) * dx
    
    return (best_h, c5, n_points)
