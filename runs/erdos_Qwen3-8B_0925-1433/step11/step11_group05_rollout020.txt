import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (len(h)/2) / sum_h
    else:
        n = 45
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_values):
        h = np.clip(h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    # Differential evolution for global search
    result = differential_evolution(
        objective, bounds,
        strategy='best1bin', popsize=30,
        maxiter=int((budget_s - 10) / 10), tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Local refinement with SLSQP
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_ = [(0, 1) for _ in range(n_points)]
    res = minimize(
        objective, best_h, method='SLSQP',
        bounds=bounds_, constraints=cons, tol=1e-6,
        options={'maxiter': 50}
    )
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    # Final check for validity
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    return (best_h, best_c5, n_points)
