import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()
    
    # Phase 1: Perturb and optimize
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    # Phase 2: Random restarts
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        rand_init = np.random.uniform(0, 1, n_points)
        rand_init = rand_init * target_sum / np.sum(rand_init)
        res = minimize(obj, rand_init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
