        if current_best[1] < best_c5:
            best_c5 = current_best[1]
            best_h = current_best[2].copy()

    # Local refinement
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
