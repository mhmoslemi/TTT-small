    convergence_threshold = 1e-5
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        m = np.argmax(cross_corr)
        lag = m - (n_points - 1)
        grad = np.zeros_like(h)
        if lag == 0:
            grad = 1 - 2 * h
        else:
            if lag > 0:
                for i in range(n_points - lag):
                    j = i + lag
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
            else:
                abs_lag = -lag
                for i in range(abs_lag, n_points):
                    j = i - abs_lag
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= target_sum / sum_h
        else:
            h = np.zeros_like(h) + 0.5
            h *= target_sum / np.sum(h)
        current_c5 = np.max(cross_corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if _ > 10 and np.abs(current_c5 - best_c5) < convergence_threshold:
            break
        learning_rate *= decay

    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros_like(h) + 0.5
        h = h * target_sum / np.sum(h)
    cross_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(cross_corr) * (2.0 / n_points)
    return (h, final_c5, n_points)
