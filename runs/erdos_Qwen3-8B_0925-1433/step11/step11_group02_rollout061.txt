import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        else:
            h_ = np.zeros_like(x)
        corr = np.correlate(h_, 1 - h_, mode="full")
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    
    # Differential Evolution for global search
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=20, mutation=(0.5, 1), recombination=0.9,
        tol=1e-6, maxiter=100, callback=None, 
        disp=False
    )
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # SLSQP refinement for local optimization
    def slsqp_obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        else:
            h_ = np.zeros_like(x)
        corr = np.correlate(h_, 1 - h_, mode="full")
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_slsqp = minimize(
        slsqp_obj, best_h, method='SLSQP',
        bounds=bounds, constraints=cons, tol=1e-8
    )
    
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    return (best_h, best_c5, n_points)
