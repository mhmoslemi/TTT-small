    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break

        corr_full = np.correlate(h, 1 - h, mode='full')
        corr_values = corr_full[n_points-1:2*n_points-1]
        k_max = np.argmax(corr_values)
        current_c5 = corr_values[k_max] * dx

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr_full_ = np.correlate(h_, 1 - h_, mode='full')
            overlap = corr_full_[n_points-1 + k_max] * dx
            return overlap

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_c5 = obj(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                h = new_h
        else:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                new_h = res.x
                new_c5 = obj(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    h = new_h
            else:
                res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
                if res.success:
                    new_h = res.x
                    new_c5 = obj(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
                        h = new_h

        if _ % 50 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                new_c5 = obj(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    h = new_h

    return (best_h, best_c5, n_points)
