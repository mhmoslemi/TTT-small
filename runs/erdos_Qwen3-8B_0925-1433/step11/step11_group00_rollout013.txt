        if curr_c5 >= best_c5:
            break
        best_c5 = curr_c5
        k_max = curr_k_max

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h == 0:
                return np.inf
            h_ = h_ * target_sum / sum_h
            shifted = np.roll(h_, k_max)
            overlap = np.dot(h_, 1 - shifted) * dx
            return overlap

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res_perturb = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h = res_perturb.x.copy()
                best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
