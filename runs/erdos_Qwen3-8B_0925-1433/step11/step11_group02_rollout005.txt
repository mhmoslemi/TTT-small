from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    try:

            objective_de,
            bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
            maxiter=100,
            disp=False
        )
        best_h = result_de.x
        best_c5 = result_de.fun
    except:
        pass

    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)

        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
