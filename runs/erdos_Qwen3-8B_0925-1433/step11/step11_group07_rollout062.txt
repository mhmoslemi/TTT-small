import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(h_vec):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.zeros(n_points)
    best_c5 = float('inf')

    patterns = [initial_h_values.copy()]

    # Prouhet-Thue-Morse
    pth = np.array([bin(i).count('1') % 2 for i in range(n_points)])
    patterns.append(pth)

    # Periodic with m=8
    m = 8
    periodic = np.zeros(n_points)
    half = m // 2
    for i in range(n_points):
        segment = i % m
        if segment < half:
            periodic[i] = 1
        else:
            periodic[i] = 0
    patterns.append(periodic)

    # Interleaved segments
    def gen_interleaved():
        even = np.zeros(n_points)
        odd = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                even[i] = 1
            else:
                odd[i] = 1
        return np.vstack([even, odd]).ravel()
    patterns.append(gen_interleaved())

    start_time = time.time()
    for p in patterns:
        h = np.clip(p, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
            if time.time() - start_time > budget_s:
                break

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)

    return (best_h, best_c5, n_points)
