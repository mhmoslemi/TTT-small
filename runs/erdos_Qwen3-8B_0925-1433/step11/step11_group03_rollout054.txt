import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize h
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 1000
    improvement_threshold = 1e-5

    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_idx = np.argmax(corr)
        lag = lag_idx - (n_points - 1)

        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            grad[i] = 1 - h[j]

        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Check for improvement
        current_c5 = compute_c5(h)
        if current_c5 < best_c5 - improvement_threshold:
            best_h = h.copy()
            best_c5 = current_c5
        else:
            # If no improvement, perturb
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * target_sum / np.sum(h)

        learning_rate *= decay_rate

    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
