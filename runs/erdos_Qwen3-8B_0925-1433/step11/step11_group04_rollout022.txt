            trial_value = np.correlate(trial_h, 1 - trial_h, mode='full')[k_max]
            if trial_value < original_value:
                break
            step *= 0.5
        if step < 1e-10:
            step = 1e-10
        h += step * direction
        learning_rate *= decay_rate
        if iteration % restart_interval == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            if np.sum(h) > 0:
                h *= target_sum / np.sum(h)

    def objective(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) > 0:
            h *= target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return corr[np.argmax(corr)] * dx

    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
