def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)

    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k, dx, n_points):
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + k < n_points:
                grad[m] += (1 - h[m + k]) * dx
            if m - k >= 0:
                grad[m] -= h[m - k] * dx
        return grad

    learning_rate = 0.1
    decay_rate = 0.999
    max_iterations = 2000
    tolerance = 1e-5
    line_search_max_steps = 20

    start_time = time.time()

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        current_c5 = corr[k_max] * dx

        grad = compute_gradient(h, k_max, dx, n_points)
        direction = -grad

        step = learning_rate
        for _ in range(line_search_max_steps):
            trial_h = h + step * direction
            trial_h = np.clip(trial_h, 0, 1)
            sum_trial = np.sum(trial_h)

            if corr_trial < corr[k_max]:

            step = 1e-10
