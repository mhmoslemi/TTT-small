            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                early_stop = 0
            else:
                early_stop += 1
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        h_ = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        max_k_index = np.argmax(corr)
        k = max_k_index - n_points + 1
        if k < 0:
            k = -k
        valid_indices = np.arange(0, n_points - k)
        if len(valid_indices) > 0:
            contributions = h_[valid_indices] * (1 - h_[valid_indices + k])
            top_indices = valid_indices[np.argsort(contributions)[::-1][:5]]
            perturb = np.zeros_like(best_h)
            for idx in top_indices:
                perturb[idx] += np.random.normal(0, 0.005)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
