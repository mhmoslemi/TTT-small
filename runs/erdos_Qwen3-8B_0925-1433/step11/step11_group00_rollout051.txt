    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1-h, mode='full')
        max_pos = np.argmax(corr)
        current_c5 = corr[max_pos] * dx
        if current_c5 >= best_c5:
            break

        def obj(x):
            h_ = np.clip(x, 0, 1)
            h_sum_ = np.sum(h_)
            if h_sum_ > 0:
                h_ = h_ * target_sum / h_sum_
            corr_ = np.correlate(h_, 1-h_, mode='full')
            return corr_[max_pos] * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_c5 = obj(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                h = best_h

    return (best_h, best_c5, n_points)
