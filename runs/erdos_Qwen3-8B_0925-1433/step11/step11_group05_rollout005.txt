        new_pop = sorted(new_pop, key=lambda x: compute_c5(x))[:n_pop]
        pop = new_pop
        best_c5 = min([compute_c5(p) for p in pop])
        best_h = pop[np.argmin([compute_c5(p) for p in pop])]
        time_left -= 0.1

    res = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
