import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0
    trials = 15

    for trial in range(trials):
        if time.time() - start_time > budget_s:
            break
        # Random restart
        rand_h = np.random.rand(n_points)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        rand_h = np.clip(rand_h, 0, 1)
        new_h = rand_h
        
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            early_stop = 0
        else:
            early_stop += 1
        
        # Local perturbation from best_h
        if trial % 3 == 0:
            h_ = np.clip(best_h, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr = np.correlate(h_, 1 - h_, mode='full')
            max_k = np.argmax(corr)
            k = min(max(0, max_k - 5), n_points - 1)
            
            perturb = np.zeros_like(best_h)
            for i in range(max(0, k-5), min(n_points, k+6)):
                perturb[i] += 0.05 * (1 - best_h[i]) * best_h[i + k]
                perturb[i + k] += -0.05 * best_h[i] * (1 - best_h[i + k])
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
                early_stop = 0
            else:
                early_stop += 1
        
        if early_stop >= 5:
            break

    return (best_h, best_c5, n_points)
