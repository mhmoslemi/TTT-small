def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_max_overlap(best_h, n_points, dx)

    start_time = time.time()
    max_iterations = 300  # Balance between iterations and time

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute overlaps for all lags
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlaps[k] = np.sum(h * (1 - shifted)) * dx

        max_k = np.argmax(overlaps)
        current_max = overlaps[max_k]

        # Define objective for this lag
        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            shifted = np.roll(h_, max_k)
            return np.sum(h_ * (1 - shifted)) * dx

        # Constraints
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        # Run optimization
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_c5 = compute_max_overlap(new_h, n_points, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

        # Add random perturbation every 10 iterations
        if _ % 10 == 0:
            perturb = np.random.normal(0, 0.02, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            new_c5 = compute_max_overlap(new_h, n_points, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    return (best_h, best_c5, n_points)

def compute_max_overlap(h, n_points, dx):
    overlaps = np.zeros(n_points)
    for k in range(n_points):
        shifted = np.roll(h, k)
        overlaps[k] = np.sum(h * (1 - shifted)) * dx
    return np.max(overlaps)
