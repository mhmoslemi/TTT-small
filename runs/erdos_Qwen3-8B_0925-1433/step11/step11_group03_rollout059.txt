import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.random.rand(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    learning_rate = 0.1
    decay_rate = 0.995
    max_iter = 300
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] = (1 - h[i + lag]) * dx
        h = h - learning_rate * grad

        curr_c5 = compute_c5(h)
        if curr_c5 < best_c5:
            best_h = h.copy()
            best_c5 = curr_c5
        learning_rate *= decay_rate
