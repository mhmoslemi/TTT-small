import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if initial_h_values is not None:
        h = initial_h_values * target_sum / np.sum(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def perturb(h):
        noise = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(h + noise, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    population = [h]
    for _ in range(100):
        population.append(perturb(population[-1]))

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(ind) for ind in population]
        top_idx = np.argsort(fitness)[:5]
        new_pop = [population[i] for i in top_idx]
        while len(new_pop) < 100:
            a, b = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[a], population[b]
            split = np.random.randint(1, n_points)
            child = np.concatenate([p1[:split], p2[split:]])
            child = np.clip(child + np.random.normal(0, 0.05, n_points), 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop

    best_h = min(population, key=compute_c5)
    current_c5 = compute_c5(best_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = best_h

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    result = minimize(lambda x: compute_c5(x), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
    best_h = result.x
    best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
