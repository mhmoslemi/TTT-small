import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

        pop = []
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)

        pop.append(h)

            h_rand = np.random.rand(n_points)
            h_rand = np.clip(h_rand, 0, 1)
            sum_rand = np.sum(h_rand)
            h_rand = h_rand * target_sum / sum_rand
            pop.append(h_rand)
        return pop
    
    population = generate_candidates(20)
    best_h = population[0]
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
