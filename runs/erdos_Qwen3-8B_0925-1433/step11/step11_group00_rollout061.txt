from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_max_corr(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_max_corr(best_h)

    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)

        def obj(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            corr_ = np.correlate(h_, 1 - h_, mode='full') * dx
            return corr_[max_index]

        bounds = [(0, 1)] * n_points
