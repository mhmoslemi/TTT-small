import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Targeted adjustments
    for _ in range(100):
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        # Compute gradient for max_k
        grad = np.zeros_like(h)
        for x in range(len(h)):
            if x + max_k < len(h):
                grad[x] = 1 - h[x + max_k]
        # Adjust h
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(h)

    # Local optimization
    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_c5 = res.fun

    return (best_h, best_c5, n_points)

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        grad = np.zeros_like(best_h)
        for x in range(n_points):
            if x + max_k < n_points:
                grad[x] = 1 - best_h[x + max_k]
        h_update = best_h - 0.01 * grad
        h_update = np.clip(h_update, 0, 1)
        sum_h = np.sum(h_update)
        if sum_h > 0:
            h_update = h_update * target_sum / sum_h
        else:
            h_update = best_h
        res = minimize(obj, h_update, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        grad = np.zeros_like(best_h)
        for x in range(n_points):
            if x + max_k < n_points:
                grad[x] = 1 - best_h[x + max_k]
        h_update = best_h - 0.005 * grad
        h_update = np.clip(h_update, 0, 1)
        sum_h = np.sum(h_update)
        if sum_h > 0:
            h_update = h_update * target_sum / sum_h
        else:
            h_update = best_h
        res = minimize(obj, h_update, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
