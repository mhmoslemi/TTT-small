import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial = np.array(initial_h_values, dtype=float)
    initial = initial * target_sum / np.sum(initial) if np.sum(initial) > 0 else initial

    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return eval_c5(h)

    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def local_refine(h):
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        return res.x if res.success else h

    def mutate(h, scale):
        return np.clip(h + np.random.normal(0, scale, n_points), 0, 1)

    def crossover(h1, h2):
        idx = np.random.randint(1, n_points - 1)
        return np.concatenate([h1[:idx], h2[idx:]])

    def evaluate(pop):
        return [eval_c5(h) for h in pop]

    pop = [initial.copy()]
    for _ in range(100):
        if time.time() > budget_s:
            break
        candidates = [mutate(h, 0.05) for h in pop]
        candidates.extend([np.random.uniform(0, 1, n_points) * target_sum / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(5)])
        fits = evaluate(candidates)
        best_idx = np.argmin(fits)
        pop.append(candidates[best_idx])
        if len(pop) > 15:
            pop.pop(0)

    elites = sorted(pop, key=eval_c5)[:5]
    best_h = min(elites, key=eval_c5)
    best_c5 = eval_c5(best_h)

        if current_c5 < best_c5:
            best_h, best_c5 = refined, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
