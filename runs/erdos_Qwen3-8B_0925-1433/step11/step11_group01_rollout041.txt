    def compute_gradient(h, k, n_points, dx):
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k] * dx
        return grad

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()

    early_stop = 0

    learning_rate = 0.1

def compute_gradient(h, k, n_points, dx):
    j_plus_k = np.arange(n_points) + k
    valid_plus = (j_plus_k >= 0) & (j_plus_k < n_points)
    grad = np.zeros(n_points)
    grad[valid_plus] = (1 - h[j_plus_k[valid_plus]]) * dx
    # Now for the other part
    j_minus_k = np.arange(n_points) - k
    valid_minus = (j_minus_k >= 0) & (j_minus_k < n_points)
    grad -= h[j_minus_k[valid_minus]] * dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_gradient(h, k, n_points, dx):
        j_plus_k = np.arange(n_points) + k
        valid_plus = (j_plus_k >= 0) & (j_plus_k < n_points)
        grad = np.zeros(n_points)
        grad[valid_plus] = (1 - h[j_plus_k[valid_plus]]) * dx
        j_minus_k = np.arange(n_points) - k
        valid_minus = (j_minus_k >= 0) & (j_minus_k < n_points)
        grad -= h[j_minus_k[valid_minus]] * dx
        return grad

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0

    learning_rate = 0.1
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        grad = compute_gradient(best_h, max_k, n_points, dx)
        h_new = best_h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * target_sum / sum_h
        new_c5 = obj(h_new)
        if new_c5 < best_c5:
            best_h, best_c5 = h_new, new_c5
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break
        learning_rate *= 0.95

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
