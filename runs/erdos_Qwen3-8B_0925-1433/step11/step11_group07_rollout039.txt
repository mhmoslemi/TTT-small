import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def generate_periodic(n, m):
        pattern = np.zeros(n)
        seg_len = n // m
        for i in range(m):
            if i % 2 == 0:
                pattern[i*seg_len : (i+1)*seg_len] = 1
        sum_p = np.sum(pattern)
        if sum_p > 0:
            pattern = pattern * (n / 2) / sum_p
        return np.clip(pattern, 0, 1)

    def generate_prouhet(n):
        seq = [0]
        for _ in range(n):
            seq = seq + [1 - x for x in seq]
        seq = seq[:n]
        sum_p = np.sum(seq)
        if sum_p > 0:
            seq = np.array(seq) * (n / 2) / sum_p
        return np.clip(seq, 0, 1)

    candidates = [np.array(initial_h_values)]
    for m in [5, 7, 9]:
        candidates.append(generate_periodic(n_points, m))
    candidates.append(generate_prouhet(n_points))

    best_h = np.zeros(n_points)
    best_c5 = float('inf')

    for h_start in candidates:
        h = np.clip(h_start, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)

        def obj(x):
            h_clipped = np.clip(x, 0, 1)
            sum_h_clipped = np.sum(h_clipped)
            if sum_h_clipped > 0:
                h_clipped = h_clipped * target_sum / sum_h_clipped
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
        else:
            current_c5 = obj(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h

    return (best_h, best_c5, n_points)
