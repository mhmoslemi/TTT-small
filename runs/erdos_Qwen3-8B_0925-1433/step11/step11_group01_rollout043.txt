import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    temp_initial = 1.0
    temp_decay = 0.995
    steps = 150
    for _ in range(steps):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, size=max(1, n_points // 5), replace=False)
        perturb = np.zeros_like(h)
        perturb[indices] = np.random.normal(0, 0.05, size=len(indices))
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = obj(new_h)
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0:
            h = new_h.copy()
            best_c5 = new_c5
        else:
            temp_current = temp_initial * (temp_decay ** _)
            if np.random.rand() < np.exp(-delta_c5 / temp_current):
                h = new_h.copy()
                best_c5 = new_c5

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if res.success:
        best_h = res.x

    final_h = np.clip(best_h, 0, 1)
    if np.sum(final_h) > 0:
        final_h = final_h * target_sum / np.sum(final_h)
    return (final_h, best_c5, n_points)
