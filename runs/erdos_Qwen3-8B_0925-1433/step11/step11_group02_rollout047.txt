from numpy.random import rand, normal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

            if sum_h > 0:
                h_scaled = h_clipped * target_sum / sum_h
            else:
                h_scaled = h_clipped

            c5 = np.max(corr) * dx
            if c5 < pbest_vals[j]:

        # Compute current values
        current_vals = np.array([np.max(np.correlate(np.clip(p, 0, 1), 1 - np.clip(p, 0, 1), mode='full')) * dx for p in particles])
        best_idx = np.argmin(current_vals)
        if current_vals[best_idx] < gbest_val:
