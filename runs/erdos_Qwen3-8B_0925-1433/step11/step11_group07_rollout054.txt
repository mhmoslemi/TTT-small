    initial_patterns.append(initial_h_values.copy())

    best_h = None
    best_c5 = float('inf')

    for pattern in initial_patterns:
        pattern = np.clip(pattern, 0, 1)
        sum_pattern = np.sum(pattern)
        if sum_pattern > 0:
            pattern = pattern * target_sum / sum_pattern
        else:
            pattern = np.zeros(n_points)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, pattern, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    else:
        best_h = np.zeros(n_points)

    return (best_h, best_c5, n_points)
