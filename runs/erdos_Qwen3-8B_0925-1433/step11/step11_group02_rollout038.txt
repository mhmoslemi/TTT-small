from scipy.optimize import minimize
import time

    # Velocities
    velocities = np.random.normal(0, 0.01, (n_particles, n_points))
    
    # Track bests
    pbest = particles.copy()
    pbest_values = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_value = np.inf
    
    start_time = time.time()

        if time.time() - start_time > budget_s:
            break
