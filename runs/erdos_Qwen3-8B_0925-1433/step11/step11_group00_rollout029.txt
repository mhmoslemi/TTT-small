import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_cross_corr(h_vec):
        corr = np.zeros(n_points)
        for k in range(n_points):
            total = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                total += h_vec[i] * (1 - h_vec[j])
            corr[k] = total * dx
        return corr

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = compute_cross_corr(h)[np.argmax(compute_cross_corr(h))]

    start_time = time.time()

    for iteration in range(500):
        if time.time() - start_time > budget_s:
            break

        corr = compute_cross_corr(h)
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        k_max = np.argmax(corr)
        def objective(x):
            h_ = np.clip(x, 0, 1)
            sum_h = np.sum(h_)
            if sum_h > 0:
                h_ = h_ * target_sum / sum_h
            total = 0.0
            for i in range(n_points):
                j = (i + k_max) % n_points
                total += h_[i] * (1 - h_[j])
            return total * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            h = np.clip(res.x, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
        else:
            h = best_h.copy()

        if iteration % 10 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h

    return (best_h, best_c5, n_points)
