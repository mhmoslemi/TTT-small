from scipy.optimize import minimize

def run(seed=42, budget_s=600):

        return np.max(corr) * dx

    pop_size = 100
    tournament_size = 3
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_std = 0.02
    elite_frac = 0.05

        sum_new = np.sum(new_h)
        if sum_new > 0:

        population.append(new_h)

    start_time = time.time()
