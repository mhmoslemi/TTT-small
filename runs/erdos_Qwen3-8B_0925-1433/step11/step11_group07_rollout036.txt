            x_minus = np.clip(x_minus, 0, 1)
            sum_minus = np.sum(x_minus)
            if sum_minus > 0:
                x_minus = x_minus * target_sum / sum_minus
            obj_minus = objective(x_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    current_h = h.copy()
    
    for _ in range(500):
        if time.time() - time.time() > budget_s:
            break
        x = current_h.copy()
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        grad = grad(x)
        x -= 0.1 * grad
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x > 0:
            x = x * target_sum / sum_x
        current_h = x
        c5 = objective(current_h)
        if c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = c5
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
        best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(corr)
    
    return (best_h, best_c5, n_points)
