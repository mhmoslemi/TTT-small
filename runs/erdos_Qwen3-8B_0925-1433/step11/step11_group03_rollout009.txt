import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr

    def max_overlap(h):
        corr = compute_correlation(h)
        return np.max(corr) * dx

    def get_worst_k(h):
        corr = compute_correlation(h)
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        return k

    def compute_overlap_for_k(x, k, dx, n_points):
        total = 0.0
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                total += x[i] * (1.0 - x[j])
        return total * dx

    def compute_gradient(x, k, dx, n_points):
        grad = np.zeros_like(x)
        for m in range(n_points):
            j = m + k
            if 0 <= j < n_points:
                grad[m] += (1.0 - x[j]) * dx
            i = m - k
            if 0 <= i < n_points:
                grad[m] -= x[i] * dx
        return grad

    best_h = h.copy()
    best_c5 = max_overlap(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        current_c5 = max_overlap(best_h)
        if current_c5 >= best_c5:
            k = get_worst_k(best_h)
            bounds = [(0, 1)] * n_points
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            
            def obj(x):
                return compute_overlap_for_k(x, k, dx, n_points)
            
            def grad(x):
                return compute_gradient(x, k, dx, n_points)
            
            result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad)
            if result.success:
                best_h = result.x
                best_c5 = max_overlap(best_h)
            else:
                break
        else:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
