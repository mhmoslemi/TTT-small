        new_c5 = compute_c5(h)
        if new_c5 >= compute_c5(h - alpha * grad):
            alpha *= 0.9
    
    def obj(x):
        h_ = np.clip(x, 0, 1)
        if np.sum(h_) > 0:
            h_ = h_ * target_sum / np.sum(h_)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun
    return (best_h, best_c5, n_points)
