from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.01):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    def mutate(h, max_corr):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - len(h) + 1
        k = shift
        mask = np.zeros_like(h, dtype=bool)
        for i in range(len(h)):
            j = i + k
            if 0 <= j < len(h):
                mask[i] = True
        if np.any(mask):
            h[mask] += np.random.normal(0, 0.01, np.sum(mask))
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
        return h

    population = [h.copy() for _ in range(5)]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()
    
    for _ in range(200):
        if time.time() - time_start > budget_s:
            break
        candidates = []
        for p in population:
            candidates.append(p)
            candidates.append(mutate(p))
        new_population = []
        for i in range(0, len(candidates), 2):
            p1, p2 = candidates[i], candidates[i+1]
            crossover = np.random.rand(len(p1))
            child = np.where(crossover < 0.5, p1, p2)

                child = child * target_sum / sum_child
            new_population.append(child)
        population = new_population
