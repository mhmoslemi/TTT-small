                         for i in range(1, n + 1)])
    
    if hasattr(initial_h_values, '__len__'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        num_ones = n_points // 2
        vdc = van_der_corput(num_ones, base=2)
        vdc *= 2
        indices = np.round(vdc / dx).astype(int)
        indices = np.clip(indices, 0, n_points - 1)
        h[indices] = 1
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
    
    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    
    best_h = np.clip(result.x, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    return (best_h, result.fun, n_points)
