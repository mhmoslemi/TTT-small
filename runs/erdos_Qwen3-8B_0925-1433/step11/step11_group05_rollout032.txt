from scipy.optimize import minimize
import time

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, std=0.02):
        noise = np.random.normal(0, std, len(h))
        new_h = np.clip(h + noise, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * target_sum / sum_new
        return new_h

    population = [h.copy()]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    population_size = 50
    tournament_size = 3
    crossover_rate = 0.8
    mutation_rate = 0.5
    elite_ratio = 0.05
    max_gen = 100
    std = 0.02
    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        if len(population) < population_size:

        selected = []
        for _ in range(population_size):
            candidates = np.random.choice(population, tournament_size, replace=False)
            best = min(candidates, key=compute_c5)
            selected.append(best)
        new_population = []
        for i in range(population_size):
            parent1 = selected[i]
            if np.random.rand() < crossover_rate:
                parent2 = selected[(i+1)%population_size]

                child = parent1.copy()

                child = perturb(child, std)
            sum_child = np.sum(child)
