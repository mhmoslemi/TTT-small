import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize
    if initial_h_values is not None:
        h = initial_h_values.copy()

        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    population = [h]
    best_h = h
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for gen in range(150):
        if time.time() - start_time > budget_s:
            break
        fit = np.array([compute_c5(ind) for ind in population])
        elite_idx = np.argsort(fit)[:int(len(population)*0.2)]
        elites = [population[i] for i in elite_idx]
        new_pop = elites[:]
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(len(population), 2, replace=False)
            child = np.concatenate([population[p1][:n_points//2], population[p2][n_points//2:]])
            mut = np.random.normal(0, 0.02, n_points)
            child = np.clip(child + mut, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop
        curr_best = min(population, key=compute_c5)
        curr_c5 = compute_c5(curr_best)
        if curr_c5 < best_c5:
            best_h = curr_best
            best_c5 = curr_c5

    # Local optimization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    best_k_index = np.argmax(corr)
    def objective(x):
        corr_new = np.correlate(x, 1 - x, mode='full')
        return corr_new[best_k_index] * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
