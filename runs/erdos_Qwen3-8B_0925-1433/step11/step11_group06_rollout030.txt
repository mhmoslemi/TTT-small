def van_der_corput(n, base=2):
    sequence = np.zeros(n)
    for i in range(n):
        num = i + 1
        denom = base ** (i + 1)
        d = 0
        while num > 0:
            d += (num % base) / denom
            num = num // base
            denom /= base
        sequence[i] = d
    return sequence
