import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    def _compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h):
        n = len(h)
        idxs = np.random.choice(n, size=max(1, n//5), replace=False)
        noise = np.random.normal(0, 0.07, len(idxs))
        new_h = h.copy()
        new_h[idxs] += noise
        new_h = np.clip(new_h, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * target_sum / sum_h
        return new_h

    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    steps = 300
    for _ in range(steps):
        if time.time() - start_time > budget_s:
            break
        new_h = perturb(best_h)
        new_c5 = _compute_c5(new_h, dx)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h
            best_c5 = new_c5
        T *= cooling_rate

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        return _compute_c5(h_, dx)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
