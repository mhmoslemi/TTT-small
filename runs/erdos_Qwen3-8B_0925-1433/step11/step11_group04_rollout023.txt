        direction = -grad

        step = learning_rate
        for _ in range(line_search_max_steps):
            trial_h = h + step * direction
            trial_h = clip_and_normalize(trial_h)
            corr_trial = np.correlate(trial_h, 1 - trial_h, mode='full')[k_max]
            if corr_trial < corr[k_max]:

            step = 1e-10
        h = h + step * direction

        # Decay learning rate
        learning_rate *= decay_rate

            perturb = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = clip_and_normalize(h)

    def obj(x):

        h = clip_and_normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')

            h = res.x.copy()
        else:
            break
