import time

    def obj(x):

        sum_h = np.sum(h_)

        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)
    early_stop = 0
