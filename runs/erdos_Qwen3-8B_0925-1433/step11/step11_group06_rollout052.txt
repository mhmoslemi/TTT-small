import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def compute_c5(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    learning_rate = 0.05
    decay_rate = 0.997
    max_iterations = 500
    perturb_freq = 10
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            shifted = i + k
            if 0 <= shifted < n_points:
                gradient[i] = (1 - h[shifted]) * dx

        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        if sum_h_new > 0:
            h_new = h_new * target_sum / sum_h_new
        else:
            h_new = h.copy()

        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()

        learning_rate *= decay_rate
        if _ % perturb_freq == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h

    return (best_h, best_c5, n_points)
