import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    learning_rate = 0.05
    max_iterations = 500
    decay_rate = 0.995
    perturbation_prob = 0.1
    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag_k = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus_k = i + lag_k
            if 0 <= i_plus_k < n_points:
                grad[i] += (1 - h[i_plus_k]) * dx
            i_minus_k = i - lag_k
            if 0 <= i_minus_k < n_points:
                grad[i] -= h[i_minus_k] * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros(n_points)
        if np.random.rand() < perturbation_prob:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * target_sum / sum_h
            else:
                h = np.zeros(n_points)
        new_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(new_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate

    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * target_sum / sum_best
    return (best_h, final_c5, n_points)
