import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 45
    else:
        n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = h * target_sum / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    start_time = time.time()
    max_iterations = 100
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        valid_pairs = []
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                valid_pairs.append((i, j))
        # Build Q matrix
        Q = np.zeros((n_points, n_points))
        for i, j in valid_pairs:
            Q[i, j] = 1
        # Solve QP
        h_var = cp.Variable(n_points)
        objective = cp.Maximize(h_var @ Q @ h_var)
        constraints = [cp.sum(h_var) == target_sum, 0 <= h_var, h_var <= 1]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS)
            if prob.status == 'OPTIMAL':
                new_h = h_var.value
            else:
                new_h = h.copy()
        except:
            new_h = h.copy()
        # Clip and rescale
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        h = new_h

    # Final evaluation
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return (h, c5, n_points)
