    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_index = np.argmax(corr)
        current_max = corr[max_index]

        def obj(x):
            x_clipped = np.clip(x, 0, 1)
            sum_x = np.sum(x_clipped)
            if sum_x > 0:
                x_clipped = x_clipped * target_sum / sum_x
            corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
            return corr[max_index]

        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            new_h = res.x
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5
        else:
            break

        # Random perturbation to escape local optima
        if iteration % 10 == 0:
            perturb_std = 0.01
            perturb = np.random.normal(0, perturb_std, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                new_h = res.x
                new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h, new_c5

        iteration += 1

    return (best_h, best_c5, n_points)
