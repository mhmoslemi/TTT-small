        m2 = (j + shift) < n

        return h.copy()

def nesterov_adaptive(h_init, n, t0, budget_s, max_iters=900000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.6
    K_min = 1
    K_max = 50

    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        frac = min(it / max_iters, 1.0)
        K = max(K_min, int(K_min + (K_max - K_min) * frac * frac))

        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)

        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15))**8
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(K) / K
            else:
                w /= w_sum
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx

        alpha = alpha0 / (1.0 + 0.002 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)

        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))

        if (it + 1) % 20000 == 0:
            h_avg_p = project_feasible(h_avg, n)
            c5a = compute_c5(h_avg_p)
            if c5a < c5_best:
                c5_best = c5a
                h_best = h_avg_p.copy()
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()

    return h_best, c5_best

def structured_perturb(h, n, rng, pi):
    mode = pi % 4
    x = np.arange(n) * 2.0 / n
    if mode == 0:
        f = 0.5 + 0.5 * (pi % 3)
        h_p = h + 0.02 * np.sin(2 * np.pi * f * x)
    elif mode == 1:
        f = 0.3 + 0.2 * (pi % 5)
        h_p = h + 0.015 * np.cos(4 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        block = n // 7
        delta = rng.uniform(-0.03, 0.03)
        h_p = h.copy()
        for b in range(7):
            lo_i = b * block
            hi_i = min(lo_i + block, n)
            h_p[lo_i:hi_i] += delta * rng.uniform(0.5, 1.5)
    else:
        sigma = 0.005 + 0.01 * (pi % 3)
        h_p = h + rng.normal(0, sigma, n)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse polish
    for T in [16, 8, 4]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = slsqp_smooth(h_best, n, T, 4000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Long Nesterov with adaptive K
    h_sg, c5_sg = nesterov_adaptive(h_best, n, t0, budget_s - 300, 700000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP fine polish
    for T in [2, 1]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = slsqp_smooth(h_best, n, T, 6000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Top-K SLSQP (focused on worst shifts)
    for K in [3, 5, 10]:
        if time.time() - t0 > budget_s - 120:
            break
        h_new = slsqp_topk(h_best, n, K, 4000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Structured perturbations + SLSQP
    for pi in range(30):
        if time.time() - t0 > budget_s - 30:
            break
        h_p = structured_perturb(h_best, n, rng, pi)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_topk(h_p, n, 5, 2000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try alternate resolution if time permits
    elapsed = time.time() - t0
    if elapsed < budget_s - 200:
        for n2 in [n * 2, n // 2]:
            if n2 < 60 or n2 > 1200:
                continue
            if time.time() - t0 > budget_s - 100:
                break
            if n2 > n:
                x_old = np.arange(n) * (2.0 / n)
                x_new = np.linspace(0, 2.0, n2, endpoint=False)
                h_alt = np.interp(x_new, x_old, h_best)
            else:
                x_old = np.arange(n) * (2.0 / n)
                x_new = np.linspace(0, 2.0, n2, endpoint=False)
                h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            for K2 in [5, 10]:
                if time.time() - t0 > budget_s - 50:
                    break
                h_new = slsqp_topk(h_alt, n2, K2, 3000, 1e-15)
                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2

    # Final
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
