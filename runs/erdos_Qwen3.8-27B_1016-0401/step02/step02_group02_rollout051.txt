import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def worst_lags_grad(corr, h, K, n):
    dx = 2.0 / n
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def lse_func(corr, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc) - np.max(T * (corr - mc))
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def lse_grad_func(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc) - np.max(T * (corr - mc))
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp(h, n, target, T, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_func(np.correlate(x, 1.0-x, mode="full"), T, n)
    def jac(x):
        return lse_grad_func(np.correlate(x, 1.0-x, mode="full"), x, T, n)
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        r = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, target)
    except:
        return h

def subgrad_polish(h, n, target, steps, K, lr_base, momentum=0.9):
    dx = 2.0 / n
    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(steps):
        corr = np.correlate(h, 1.0-h, mode="full")
        g = worst_lags_grad(corr, h, K, n)
        lr = lr_base / (1.0 + 0.05 * it)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
        else:
            v *= 0.5
            h_new = project_feasible(h + v, target)

    thresh = 0.5
    h_bin = (h > thresh).astype(float)

        h_bin = np.zeros(n)
        k = int(target)
        h_bin[order[:k]] = 1.0
        if k < n:
            h_bin[order[k]] = target - k
    h = h_bin
    c5_cur = compute_c5(h)
    rng_ls = np.random.default_rng(12345)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

            break

                h[b] = 0.0

            c5_cur = best_c5
            h = h.copy()
            improved = True

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_and_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    obj = (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def slsqp(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        return lse_and_grad(np.correlate(x, 1.0-x, "full"), x, T, n)[0]
    def jac(x):
        return lse_and_grad(np.correlate(x, 1.0-x, "full"), x, T, n)[1]
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        r = minimize(objf, h, method='SLSQP', jac=jac, bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, target)
    except:
        return h

def topK_grad(corr, h, K, n):
    dx = 2.0 / n
    center = n - 1
    top = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top:
        o = int(idx) - center
        if o == 0:
            g += 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += 1.0 - h[:n - o]
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
    return g * dx / K

def subgrad_momentum(h, n, target, steps, K, base_lr, mu=0.9):
    v = np.zeros(n)
    best, best_c5 = h.copy(), compute_c5(h)
    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, "full")
        g = topK_grad(corr, h, K, n)
        lr = base_lr / (1.0 + 0.03 * it)
        v = mu * v - lr * g
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best, best_c5, h = h_new.copy(), c5_new, h_new
        else:
            v *= 0.5
            h_new = project_feasible(h + v, target)
            h = h_new
    return best, best_c5

max_{p ≥ 0, ∑p_k = 1} min_h ∑_k p_k ∫ h(x)(1-h(x+k)) dx + ...

The inner minimization is a quadratic in h. The outer is over probability distributions on lags.

This suggests an iterative method:
1. Start with a distribution p over lags
2. Solve the quadratic minimization for h given p
3. Update p to concentrate on the worst lags
4. Repeat

This is basically what the log-sum-exp continuation does, but the explicit distribution update might be more efficient.

The quadratic minimization:
min_h ∑_k p_k ∫ h(x)(1-h(x+k)) dx = min_h ∑_k p_k [∫h(x)dx - ∫h(x)h(x+k)dx]
= min_h [∫h(x)dx - ∑_k p_k ∫h(x)h(x+k)dx]
= min_h [1 - ∑_k p_k (h*h)(k)]  (where * is cross-correlation)

Wait, ∫h(x)h(x+k)dx is the autocorrelation of h at lag k. So we want to maximize ∑_k p_k (h*h)(k) subject to ∫h=1, 0≤h≤1.

This is a quadratic maximization (the objective is -∑p_k (h*h)(k) + constant, and (h*h)(k) is quadratic in h). We could solve this with CVXPY or SLSQP.

Actually, the Hessian of ∑_k p_k (h*h)(k) is a matrix based on the lag weights. It's a circulant-like matrix. This is a structured quadratic program!

Let me think... (h*h)(k) = ∑_i h[i] h[i+k] (discretized). The quadratic form is:
∑_k p_k ∑_i h[i] h[i+k] = h^T A h where A is a symmetric matrix with A[i,j] = ∑_k p_k * [j = i+k or j = i-k] (roughly).

Actually, A[i,j] = p_{|i-j|} (if |i-j| < n, with appropriate boundary handling). So A is a Toeplitz matrix determined by the lag distribution p.

This means for a fixed p, the inner problem is:
min_h 1/2 h^T (-2A) h + ... subject to linear constraints

Which is a QP! We could solve it with CVXPY efficiently.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_and_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    obj = (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def slsqp(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        return lse_obj_and_grad(np.correlate(x, 1.0 - x, "full"), x, T, n)[0]
    def jac(x):
        return lse_obj_and_grad(np.correlate(x, 1.0 - x, "full"), x, T, n)[1]
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        r = minimize(objf, h, method='SLSQP', jac=jac, bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(r.x, target)
    except Exception:
        return h

def topK_subgrad(corr, h, K, n):
    dx = 2.0 / n
    center = n - 1
    top = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top:
        o = int(idx) - center
        if o == 0:
            g += 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += 1.0 - h[:n - o]
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
    return g * dx / K

def subgrad_momentum(h, n, target, steps, K, base_lr, mu=0.90):
    v = np.zeros(n)
    best, best_c5 = h.copy(), compute_c5(h)
    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, "full")
        g = topK_subgrad(corr, h, K, n)
        lr = base_lr / (1.0 + 0.02 * it)
        v = mu * v - lr * g
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best, best_c5, h = h_new.copy(), c5_new, h_new
        else:
            v *= 0.5
            h = project_feasible(h + v, target)
    return best, best_c5

def binary_local_search(h_init, n, target, time_budget, seed=77):
    order = np.argsort(h_init)[::-1]
    hb = np.zeros(n)
    kb = int(target)
    hb[order[:kb]] = 1.0
    if kb < n:
        hb[order[kb]] = target - kb
    c5 = compute_c5(hb)
    rng = np.random.default_rng(seed)
    deadline = time.time() + time_budget
    while time.time() < deadline:
        improved = False
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_try = 600
        for _ in range(n_try):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            hb[a], hb[b] = 0.0, 1.0
            c5t = compute_c5(hb)
            if c5t < c5 - 1e-14:
                c5 = c5t
                improved = True
            else:
                hb[a], hb[b] = 1.0, 0.0
        if not improved:
            n_try2 = 400
            for _ in range(n_try2):
                if rng.random() < 0.5:
                    pos = ones[rng.integers(len(ones))]
                    hb[pos] = 0.0
                else:
                    pos = zeros[rng.integers(len(zeros))]
                    hb[pos] = 1.0
                c5t = compute_c5(hb)
                if c5t < c5 - 1e-14:
                    c5 = c5t
                    improved = True
                else:
                    hb[pos] = 1.0 - hb[pos]
            if not improved:
                break
    return hb, c5

def sturmian_init(n, alpha, rho=0.0):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i * alpha + rho) % 1.0) < 0.5 else 0.0
    return h
