import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, l):
    """Gradient of f_l(h) = sum_m h[m]*(1-h[m-l]) with respect to h."""
    n = len(h)
    g = np.ones(n)
    if l == 0:
        g -= 2.0 * h
    elif l > 0:
        g[l:] -= h[:n-l]
        g[:n-l] -= h[l:]
    else:
        p = -l
        g[p:] -= h[:n-p]
        g[:n-p] -= h[p:]
    return g

def active_subgrad(h, top_k):
    """Subgradient using top-k active shifts. Returns (grad, objective)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argpartition(corr, -top_k)[-top_k:]
    center = n - 1
    grad = np.zeros(n)
    for i in idx:
        l = i - center
        grad += shift_grad(h, l)
    grad /= top_k
    obj = float(corr[idx].max() * dx)
    return grad * dx, obj

    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / len(h))

def soft_max_grad(h, T):

    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)

            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx

        g, _ = active_subgrad(y, top_k)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0

    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, l):
    n = len(h)
    g = np.ones(n)
    if l == 0:
        g -= 2.0 * h
    elif l > 0:
        g[l:] -= h[:n - l]
        g[:n - l] -= h[l:]
    else:
        p = -l
        g[p:] -= h[:n - p]
        g[:n - p] -= h[p:]
    return g

def active_subgrad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argpartition(corr, -(top_k - 1))[-top_k:] if top_k > 1 else np.array([np.argmax(corr)])
    center = n - 1
    grad = np.zeros(n)
    for i in idx:
        grad += shift_grad(h, i - center)
    grad /= top_k
    return grad * dx, float(corr[np.argmax(corr)] * dx)

def soft_max_jac(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        g = shift_grad(h, o)
        grad += w * g
    return grad * dx, float((mc + np.log(e.sum()) / T) * dx)

def fista_subgrad(h0, target, top_k, max_iter, step):
    n = len(h0)
    h = h0.copy().astype(float)
    zh = h.copy()
    t_k = 1.0

        y = h + mu * (h - zh)
        g, _ = active_subgrad(y, top_k)
        zh = project_box_sum(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-15:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        h = zh
        t_k = t_new
    return h

def lbfgsb_soft(h0, target, T, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        _, v = soft_max_jac(x, T)
        return v + pen * (x.sum() - target) ** 2
    def jac(x):
        g, _ = soft_max_jac(x, T)
        return g + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=jac,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15, 'maxls': 40})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()
