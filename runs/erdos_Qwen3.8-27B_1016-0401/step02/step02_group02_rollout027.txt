            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g /= K * (2.0 / n)
        lr = 2.0 / (1.0 + 0.1 * it)
        improved = False
        for _ in range(8):

            c5_new = compute_c5(h_new)
            if c5_new < best:
                h = h_new
                best = c5_new
                improved = True
                break
            lr *= 0.5
        if not improved and it > 50:
            break
    return h, best

def round_and_local(h, target, n_iter=3000, seed=0):
    """Round to binary then do local swap search"""
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    h_bin[order[:k]] = 1.0
    if k < n:
        h_bin[order[k]] = target - k
    best = compute_c5(h_bin)
    rng = np.random.default_rng(seed)
    for _ in range(n_iter):

        j = rng.integers(n)

        if c5 < best:
            best = c5
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, best

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    g = np.zeros(n)
    nz = np.where(w > 1e-9)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

def do_slsqp(h, n, target, T, maxiter):
    def obj(x):
        return lse_obj(get_corr(x), T, n)
    def jac(x):
        return lse_grad(get_corr(x), x, T, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

                g[p:] -= h[:n - p]
        g /= K * (2.0 / n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:

        lr = 1.5 / (1.0 + 0.08 * it)

    best = compute_c5(h)
