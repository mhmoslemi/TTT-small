    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    # Active set: within 5% of max
    active = corr > 0.95 * max_val
    n_active = np.sum(active)
    
    # Compute subgradient (average of active gradients)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        if not active[center + o]:
            continue
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if not mask.any():
                break
            tr = room[mask].sum()
            if tr < 1e-10:
                break
            frac = min(1.0, diff / tr)
            h[mask] += room[mask] * frac
        else:
            mask = h > 1e-10
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot < 1e-10:
                break
            frac = min(1.0, -diff / tot)
            h[mask] -= h[mask] * frac
    return h

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def slsqp_step(h, target, T, bounds, cons, maxiter):
    try:
        res = minimize(soft_max_obj, h, method='SLSQP',
                       jac=soft_max_jac, args=(T,),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 50:
            return fix_constraint(res.x.copy(), target)
    except Exception:
        pass
    return fix_constraint(h.copy(), target)

def subgrad_direction(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    active = corr >= 0.97 * max_val
    n_act = np.sum(active)
    if n_act == 0:
        return np.zeros(n), np.zeros(2 * n - 1)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        if not active[center + o]:
            continue
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] += -h[o:]
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] += -h[:n - p]
    grad /= n_act
    return grad, corr

def local_search(h, target, n_iter, deadline, rng):
    n = len(h)
    h_curr = h.copy()
    best = compute_c5(h_curr)
    curr = best
    step = 0.05
    for it in range(n_iter):
        if time.time() > deadline:
            break
        r = rng.random()
        h_try = h_curr.copy()
        if r < 0.35:
            grad, corr = subgrad_direction(h_curr)
            h_try -= step * grad
            h_try += rng.normal(0, 0.02, n)
        elif r < 0.55:
            grad, corr = subgrad_direction(h_curr)
            top3 = np.argsort(corr)[-3:]
            k_star = top3[rng.integers(3)]
            s = k_star - (n - 1)
            p = abs(s)
            if p == 0:
                idx = int(np.argmax(h_curr * (1 - h_curr)))
                h_try[idx] += (0.5 - h_curr[idx]) * rng.uniform(0.3, 0.9)
            else:
                if s > 0:
                    contrib = h_curr[p:] * (1 - h_curr[:n - p])
                else:
                    contrib = h_curr[:n - p] * (1 - h_curr[p:])
                idx = int(np.argmax(contrib))
                if s > 0:
                    j1, j2 = idx + p, idx
                else:
                    j1, j2 = idx, idx + p
                h_try[j1] -= rng.uniform(0.3, 0.9)
                h_try[j2] += rng.uniform(0.3, 0.9)
        elif r < 0.7:
            i, j = rng.integers(0, n, 2)
            if i != j:
                d = rng.uniform(0.2, 0.8)
                if rng.random() < 0.5:
                    h_try[i] -= d
                    h_try[j] += d
                else:
                    h_try[i] += d
                    h_try[j] -= d
        elif r < 0.82:
            length = int(rng.integers(2, max(3, n // 12)))
            start = int(rng.integers(0, n - length + 1))
            h_try[start:start + length] = 1.0 - h_try[start:start + length]
        elif r < 0.92:
            p = int(rng.integers(1, n // 5))
            idx = int(rng.integers(0, n - p))
            d = rng.uniform(0.2, 0.7)
            h_try[idx] -= d
            h_try[idx + p] += d
        else:
            length = int(rng.integers(4, max(5, n // 8)))
            start = int(rng.integers(0, n - length + 1))
            h_try[start:start + length] += rng.normal(0, 0.12, length)

        h_try = fix_constraint(h_try, target)
        c_try = compute_c5(h_try)
        if c_try < best:
            best = c_try
            h_curr = h_try
            curr = c_try
            step = max(0.01, step * 1.05)
        elif c_try < curr + 0.0001 and rng.random() < 0.5:
            curr = c_try
            h_curr = h_try
            step = max(0.01, step * 0.95)
        else:
            step = max(0.01, step * 0.8)
    return h_curr, best

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
