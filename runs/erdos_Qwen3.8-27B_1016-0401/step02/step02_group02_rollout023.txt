import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -3.0, 3.0

def corr_and_grad(corr, h, n, weights=None):
    """Compute gradient of weighted sum of corr terms w.r.t. h."""
    dx = 2.0 / n
    center = n - 1
    if weights is None:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    nz = np.where(weights > 1e-12)[0]
    for idx in nz:
        wi = weights[idx]
        o = idx - center
        if o == 0:

        elif o > 0:

        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx

def topk_subgrad_step(h, n, target, K, lr):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-K:]
    w = np.zeros(len(corr))
    w[top_idx] = 1.0 / K
    g = corr_and_grad(corr, h, n, w)
    return project_feasible(h - lr * g, target)

def lse_smooth(corr, T):
    mc = np.max(corr)
    sh = T * (corr - mc) - np.max(T * (corr - mc))
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)) - np.max(T * (corr - mc)))) / T)

        return corr_and_grad(corr, x, n, w)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def compute_grad(corr, h, n, w):
    """Compute gradient given correlation values and weights w."""
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-14)[0]
    for idx in nz:
        wi = w[idx]
        o = idx - center
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] -= wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx

def lse_objective(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    tcorr = T * (corr - mc)
    tmax = np.max(tcorr)
    return (mc + np.log(np.sum(np.exp(tcorr - tmax))) / T) * dx

def lse_gradient(x, T, n):
    dx = 2.0 / n

    mc = np.max(corr)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def compute_grad(h, n, w):
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-14)[0]
    for idx in nz:
        wi = w[idx]
        o = idx - center
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def lse_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    tc = T * (corr - mc)
    tm = np.max(tc)
    return (mc + np.log(np.sum(np.exp(tc - tm))) / T) * dx

def lse_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    tc = T * (corr - mc)
    tm = np.max(tc)
    e = np.exp(tc - tm)
    w = e / e.sum()
    return compute_grad(x, n, w)

def slsqp_phase(h, n, target, T, maxiter, tlimit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    ts = time.time()
    try:
        res = minimize(lse_obj, h, args=(T, n), method='SLSQP',
                      jac=lambda x: lse_grad(x, T, n),
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() - ts > tlimit:
            return h
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgrad_loop(h, n, target, n_iter, K, lr0, decay, tlimit):
    ts = time.time()
    c5loc = compute_c5(h)
    for it in range(n_iter):
        if time.time() - ts > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-K:]
        w = np.zeros(len(corr))
        w[top] = 1.0 / K
        g = compute_grad(h, n, w)
        lr = lr0 / (1.0 + decay * it)
        h_new = project_feasible(h - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5loc:
            c5loc = c5_new
            h = h_new
        else:
            lr *= 0.3
            h_new = project_feasible(h - lr * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5loc:
                c5loc = c5_new
                h = h_new
    return h, c5loc

def anneal_step(h, n, target, rng, temp):
    """Random swap of two coordinates to explore locally."""
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h, compute_c5(h)
    h_new = h.copy()
    di = h_new[i]
    dj = h_new[j]
    # Try swapping values (maintains sum)
    h_new[i] = dj
    h_new[j] = di
    if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
        # Try perturbation instead
        delta = rng.normal(0, 0.02)
        h_new = h.copy()
        h_new[i] = np.clip(h[i] + delta, 0, 1)
        h_new[j] = np.clip(h[j] - delta, 0, 1)
        h_new = project_feasible(h_new, target)
    c5_new = compute_c5(h_new)
    return h_new, c5_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(400) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP continuation
    T_sched = [80, 40, 25, 15, 10, 7, 5, 4, 3, 2.5, 2.0]
    for i, T in enumerate(T_sched):
        if elapsed() > budget_s - 250:
            break
        tlim = min(45.0, (budget_s - 250 - elapsed()) / max(1, len(T_sched) - i))
        h_cur = slsqp_phase(h, n, target, T, 1200, tlim)
        c5_cur = compute_c5(h_cur)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h_cur.copy()
        h = h_cur

    # Phase 2: Long K=1 subgradient (single worst lag focus)
    h_tmp = h_best.copy()
    remaining = max(10, budget_s - 300 - elapsed())
    h_tmp, _ = subgrad_loop(h_tmp, n, target, 20000, 1, 0.04, 0.002, remaining)
    c5_tmp = compute_c5(h_tmp)
    if c5_tmp < c5_best:
        c5_best = c5_tmp
        h_best = h_tmp.copy()

    # Phase 3: SLSQP at very low T
    if elapsed() < budget_s - 100:
        for T in [1.8, 1.4, 1.0]:
            if elapsed() > budget_s - 60:
                break
            h_cur = slsqp_phase(h_best, n, target, T, 2500, 35)
            c5_cur = compute_c5(h_cur)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()

    # Phase 4: Prolonged subgradient refinement
    h_tmp = h_best.copy()
    remaining = max(5, budget_s - 60 - elapsed())
    h_tmp, _ = subgrad_loop(h_tmp, n, target, 30000, 1, 0.025, 0.001, remaining)
    c5_tmp = compute_c5(h_tmp)
    if c5_tmp < c5_best:
        c5_best = c5_tmp
        h_best = h_tmp.copy()

    # Phase 5: Simulated annealing with structured moves
    if elapsed() < budget_s - 20:
        temp = c5_best * 0.001
        h_sa = h_best.copy()
        c5_sa = c5_best
        for it in range(50000):
            if elapsed() > budget_s - 5:
                break
            if it % 5000 == 0 and it > 0:
                temp *= 0.7
            h_cand, c5_cand = anneal_step(h_sa, n, target, rng, temp)
            if c5_cand < c5_sa:
                h_sa = h_cand
                c5_sa = c5_cand
                if c5_sa < c5_best:
                    c5_best = c5_sa
                    h_best = h_sa.copy()
            elif temp > 1e-6 and rng.random() < np.exp(-(c5_cand - c5_sa) / max(temp, 1e-9)):
                h_sa = h_cand
                c5_sa = c5_cand

    # Phase 6: Final micro-subgradient
    h_tmp = h_best.copy()
    h_tmp, _ = subgrad_loop(h_tmp, n, target, 5000, 1, 0.01, 0.005, 10)
    c5_tmp = compute_c5(h_tmp)
    if c5_tmp < c5_best:
        c5_best = c5_tmp
        h_best = h_tmp.copy()

    # Phase 7: Try lower resolution n=200
    if elapsed() < budget_s - 30 and n > 200:
        n_lo = 200
        t_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, t_lo)
        for T in [20, 10, 5, 3, 2]:
            if elapsed() > budget_s - 10:
                break
            h_lo = slsqp_phase(h_lo, n_lo, t_lo, T, 800, 15)
        h_lo, _ = subgrad_loop(h_lo, n_lo, t_lo, 3000, 1, 0.03, 0.003, 20)
        c5_lo = compute_c5(h_lo)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
