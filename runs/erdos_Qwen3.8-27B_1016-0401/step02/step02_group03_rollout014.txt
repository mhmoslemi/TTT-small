from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def shift_grad(h, n, k):
    j = np.arange(n)
    g = np.zeros(n)
    if k >= 0:
        m1 = (j >= k) & (j < n)
        g[m1] += (1.0 - h[j[m1] - k])
        m2 = (j >= 0) & (j < n - k)
        g[m2] -= h[j[m2] + k]
    else:
        t = -k
        m1 = (j >= 0) & (j < n - t)
        g[m1] += (1.0 - h[j[m1] + t])
        m2 = (j >= t) & (j < n)
        g[m2] -= h[j[m2] - t]
    return g

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

# Global state for SLSQP callbacks
_G_N = 0
_G_T = 1.0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    return (mc + np.log(np.exp(sh).sum()) / _G_T) * (2.0 / _G_N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    g = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-13:
            continue
        g += w * shift_grad(x, _G_N, oi - center)
    return g * (2.0 / _G_N)

def _cfun(x):
    return x.sum() - _G_N / 2.0

def _cjac(x):
    return np.ones(_G_N)

def slsqp_step(h, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = float(T)
    res = minimize(_obj, h, method='SLSQP', jac=_jac,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    h_cur = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h_cur)

        if c5_new < c5_best:

        h_cur = h_new
