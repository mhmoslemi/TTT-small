    t0 = time.time()
    _G['target'] = 0.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    _G['target'] = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    cons = _make_eq_cons()

    # Phase 1: Aggressive active-set SLSQP on top lags (45% budget)
    p1_end = t0 + budget_s * 0.45
    iteration_count = 0
    for outer in range(300):
        if time.time() > p1_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        center = n - 1
        top_k = min(10, 2*n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        # Sort by worst first
        top_idx = sorted(top_idx.tolist(), key=lambda l: -corr[l])
        for lag in top_idx:
            if time.time() > p1_end:
                break
            iteration_count += 1
            try:
                res = minimize(lag_obj, h_best, args=(lag, n), method='SLSQP',
                              jac=lambda x, ll=lag, nn=n: lag_grad(x, ll, nn),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-18})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 2: Soft-max with increasing T (25% budget)
    p2_end = t0 + budget_s * 0.70
    for T in [150, 300, 600, 1200, 2400, 4800, 9600]:
        if time.time() > p2_end:
            break
        try:
            res = minimize(softmax_obj, h_best, args=(T, n), method='SLSQP',
                          jac=lambda x, tt=T, nn=n: softmax_grad(x, tt, nn),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 10000, 'ftol': 1e-18})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Nesterov subgradient polish (10% budget)
    p3_end = t0 + budget_s * 0.80
    h_sg = h_best.copy()
    h_mom = h_best.copy()
    center = n - 1
    for it in range(8000):
        if time.time() > p3_end:
            break
        corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        worst = np.argmax(corr)
        g = lag_grad(h_sg, worst, n)
        lr = 0.005 / (1.0 + 0.005 * it)
        h_mom = 0.6 * h_mom + 0.4 * h_sg
        h_new_sg = h_mom - lr * g
        h_new_sg = fix_constraint(h_new_sg, target)
        h_sg = h_new_sg
        if it % 2000 == 0:
            c5_sg = compute_c5(h_sg)
            if c5_sg < c5_best:
                c5_best = c5_sg
                h_best = h_sg.copy()

    # Phase 4: Try interpolation to n=560 and optimize (12% budget)
    p4_end = t0 + budget_s * 0.92
    n_alt = 560
    target_alt = n_alt / 2.0
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
    h_alt = np.interp(x_new, x_old, h_best)
    h_alt = fix_constraint(h_alt, target_alt)
    c5_alt = compute_c5(h_alt)
    _G['target'] = target_alt
    cons_alt = _make_eq_cons()
    bounds_alt = [(0.0, 1.0)] * n_alt
    for outer in range(60):
        if time.time() > p4_end:
            break
        corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
        worst_a = np.argmax(corr_a)
        try:
            res = minimize(lag_obj, h_alt, args=(worst_a, n_alt), method='SLSQP',
                          jac=lambda x, ll=worst_a, nn=n_alt: lag_grad(x, ll, nn),
                          bounds=bounds_alt, constraints=cons_alt,
                          options={'maxiter': 400, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt - 1e-15:
                c5_alt = c5_new
                h_alt = h_new
        except Exception:
            pass
    c5_alt = compute_c5(h_alt)
    if c5_alt < c5_best:
        c5_best = c5_alt
        h_best = h_alt
        n = n_alt
        target = target_alt
        _G['target'] = target

    # Phase 5: Final aggressive polish (8% budget)
    p5_end = t0 + budget_s - 10.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    _G['target'] = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    cons_f = _make_eq_cons()
    for _ in range(80):
        if time.time() > p5_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = np.argmax(corr_f)
        # Also try second-worst
        second_f = np.argsort(corr_f)[-2]
        for lag_try in [worst_f, second_f]:
            if time.time() > p5_end:
                break
            try:
                res = minimize(lag_obj, h_best, args=(lag_try, n_f), method='SLSQP',
                              jac=lambda x, ll=lag_try, nn=n_f: lag_grad(x, ll, nn),
                              bounds=bounds_f, constraints=cons_f,
                              options={'maxiter': 5000, 'ftol': 1e-19})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
