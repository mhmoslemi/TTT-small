    hc = h.copy()
    best_c5 = compute_c5(hc)
    best_h = hc.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(hc, 1 - hc, mode="full")
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()
        k = int(np.argmax(corr))
        s = k - (n - 1)
        grad = np.zeros(n)
        if s == 0:
            grad = -(1.0 - 2.0 * hc) * dx
        elif s > 0:
            grad[:n-s] += (1.0 - hc[s:]) * dx
            grad[s:] -= hc[:n-s] * dx
        else:
            p = -s
            grad[p:] += (1.0 - hc[:n-p]) * dx
            grad[:n-p] -= hc[p:] * dx
        hc = fix_constraint(hc - lr * grad, target)
        lr *= decay
    return best_h, best_c5

def lsearch(h, target, n_iter, deadline, rng=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    hc = h.copy()
    stall = 0
    delta = 0.04
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 180:
            delta *= 0.65
            if delta < 0.003:
                delta = 0.04
            stall = 0
        k = int(np.argmax(corr))
        s = k - (n - 1)
        p = abs(s)
        ht = hc.copy()
        if p == 0:
            idx = int(np.argmax(hc * (1 - hc)))
            ht[idx] = np.clip(ht[idx] + (delta if ht[idx] < 0.5 else -delta), 0, 1)
        else:
            if s > 0:
                cr = hc[p:] * (1 - hc[:n-p])
            else:
                cr = hc[:n-p] * (1 - hc[p:])
            # Mix greedy and random selection
            if rng is not None and rng.random() < 0.3:
                idx = int(rng.integers(0, len(cr)))
            else:
                idx = int(np.argmax(cr))
            if s > 0:
                ht[idx+p] = np.clip(ht[idx+p] - delta, 0, 1)
                ht[idx] = np.clip(ht[idx] + delta, 0, 1)
            else:
                ht[idx] = np.clip(ht[idx] - delta, 0, 1)
                ht[idx+p] = np.clip(ht[idx+p] + delta, 0, 1)
        ht = fix_constraint(ht, target)
        cr_t = np.correlate(ht, 1 - ht, mode="full")
        c5_t = float(np.max(cr_t) * dx)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            hc = ht
            corr = cr_t
            stall = 0
        else:
            stall += 1
    return hc, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(sum_fun, target=target),
             'jac': partial(sum_jac, target=target)}]

    # Phase 1: Subgradient descent (moderate step)
    dl = t0 + budget_s * 0.12
    h_sg, c5_sg = subgrad_descent(h_best, target, 30000, 0.003, 0.9997, dl)
    if c5_sg < c5_best:
        c5_best, h_best = c5_sg, h_sg

    # Phase 2: SLSQP refinement at decreasing temperatures
    dl = t0 + budget_s * 0.45
    for T in [40, 20, 10, 5, 3, 2, 1.2]:
        if time.time() > dl:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=partial(soft_max_jac, T=T),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-16})
            if res.fun < 50:
                hn = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(hn)
                if c5 < c5_best - 1e-14:
                    c5_best, h_best = c5, hn
        except Exception:
            pass

    # Phase 3: Local search (greedy + random)
    dl = t0 + budget_s * 0.6
    h_ls, c5_ls = lsearch(h_best, target, 8000, dl, rng)
    if c5_ls < c5_best:
        c5_best, h_best = c5_ls, h_ls

    # Phase 4: Try 2x resolution
    n2 = n * 2
    target2 = n2 / 2.0
    h_up = np.interp(np.arange(n2) * (n / n2), np.arange(n), h_best)
    h_up = fix_constraint(h_up, target2)
    c5_up = compute_c5(h_up)

    dl = t0 + budget_s * 0.82
    if c5_up < c5_best + 0.002:
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': partial(sum_fun, target=target2),
                  'jac': partial(sum_jac, target=target2)}]
        hUp = h_up.copy()
        c5c = c5_up

        h_sg2, c5_sg2 = subgrad_descent(hUp, target2, 8000, 0.002, 0.999, dl)
        if c5_sg2 < c5c:
            c5c, hUp = c5_sg2, h_sg2

        for T in [10, 5, 2, 1]:
            if time.time() > dl:
                break
            try:
                res = minimize(soft_max_obj, hUp, method='SLSQP',
                              jac=partial(soft_max_jac, T=T),
                              bounds=bounds2, constraints=cons2,
                              options={'maxiter': 2500, 'ftol': 1e-15})
                if res.fun < 50:
                    hn = fix_constraint(res.x.copy(), target2)
                    c5 = compute_c5(hn)
                    if c5 < c5c - 1e-14:
                        c5c, hUp = c5, hn
            except Exception:
                pass
        h_ls2, c5_ls2 = lsearch(hUp, target2, 3000, dl, rng)
        if c5_ls2 < c5c:
            c5c, hUp = c5_ls2, h_ls2

        if c5c < c5_best:
            c5_best, h_best = c5c, hUp
            n = len(h_best)
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': partial(sum_fun, target=target),
                     'jac': partial(sum_jac, target=target)}]

    # Phase 5: Multi-start structured perturbations
    dl = t0 + budget_s * 0.94
    while time.time() < dl:
        hib = h_best
        ni = len(hib)
        ti = ni / 2.0
        bt_i = [(0.0, 1.0)] * ni
        cs_i = [{'type': 'eq', 'fun': partial(sum_fun, target=ti),
                 'jac': partial(sum_jac, target=ti)}]
        r = int(rng.integers(0, 10))
        if r < 3:
            ht = hib.copy()
            st = int(rng.integers(0, max(1, ni - 40)))
            ln = int(rng.integers(4, 40))
            ht[st:st+ln] = 1.0 - ht[st:st+ln]
        elif r < 6:
            ht = hib + rng.normal(0, 0.06, ni)
        elif r < 8:
            ht = hib.copy()
            for _ in range(int(rng.integers(2, 6))):
                st = int(rng.integers(0, max(1, ni - 15)))
                ln = int(rng.integers(2, 15))
                ht[st:st+ln] = 1.0 - ht[st:st+ln]
        else:
            ht = hib.copy()
            mask = rng.random(ni) < 0.1
            ht[mask] = 1.0 - ht[mask]
        ht = fix_constraint(ht, ti)
        c5_t = compute_c5(ht)
        if c5_t < c5_best - 1e-14:
            c5_best, h_best = c5_t, ht
            h_ls, c5_ls = lsearch(h_best, target, 800, dl, rng)
            if c5_ls < c5_best:
                c5_best, h_best = c5_ls, h_ls
        else:
            try:
                res = minimize(soft_max_obj, ht, method='SLSQP',
                              jac=partial(soft_max_jac, T=5),
                              bounds=bt_i, constraints=cs_i,
                              options={'maxiter': 600, 'ftol': 1e-14})
                if res.fun < 50:
                    hn = fix_constraint(res.x.copy(), ti)
                    c5 = compute_c5(hn)
                    if c5 < c5_best - 1e-14:
                        c5_best, h_best = c5, hn
                        h_ls, c5_ls = lsearch(h_best, target, 400, dl, rng)
                        if c5_ls < c5_best:
                            c5_best, h_best = c5_ls, h_ls
            except Exception:
                pass

    # Final polish
    if time.time() < t0 + budget_s - 5:
        h_ls, c5_ls = lsearch(h_best, target, 3000, t0 + budget_s - 2, rng)
        if c5_ls < c5_best:
            c5_best, h_best = c5_ls, h_ls

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
