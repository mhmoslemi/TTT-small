from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            idxs = np.where(mask)[0]
            order = idxs[np.argsort(-(1.0 - h[idxs]))]
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            mask = h > 0.0
            if not mask.any():
                break
            idxs = np.where(mask)[0]
            order = idxs[np.argsort(-h[idxs])]
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def c_obj(x, T):

    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
