        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5
        best_pair = None
        n_sample = min(len(ones) * len(zeros), 4000)
        for _ in range(n_sample):
            iv = ones[rng.integers(len(ones))]
            jv = zeros[rng.integers(len(zeros))]
            h[iv], h[jv] = 0.0, 1.0
            nc = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                best_pair = (iv, jv)
            h[iv], h[jv] = 1.0, 0.0
        if best_pair is not None:
            h[best_pair[0]], h[best_pair[1]] = 0.0, 1.0
            c5 = best_c5
            no_improve = 0
        else:
            found2 = False
            if len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(3000):
                    i1, i2 = rng.choice(ones, 2, replace=False)
                    j1, j2 = rng.choice(zeros, 2, replace=False)
                    h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                    nc = compute_c5(h)
                    if nc < c5 - 1e-14:
                        c5 = nc
                        found2 = True
                        break
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
            if found2:
                no_improve = 0
                continue
            no_improve += 1
            if no_improve >= 8:
                break
    return h, c5

def lse_objective(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    z = T * (corr - mc)
    z -= np.max(z)
    logsum = np.max(z) + np.log(np.sum(np.exp(z)))
    return mc + logsum / T

def lse_gradient(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    z = T * (corr - mc)
    z -= np.max(z)
    e = np.exp(z)
    w = e / e.sum()
    dx_val = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += wi * (1.0 - x[:n - o])
            grad[:n - o] -= wi * x[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - x[p:])
            grad[p:] -= wi * x[:n - p]
    return grad * dx_val

def slsqp_optimize(h, n, target, T, maxiter, bounds):
    def obj(x):
        return lse_objective(x, T, n)
    def jac(x):
        return lse_gradient(x, T, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def make_periodic(n, period, rng):
    p = period
    base = np.zeros(p)
    half = p // 2
    base[:half] = 1.0
    h = np.tile(base, n // p + 1)[:n]
    jitter = rng.normal(0, 0.3, n)
    h = h + jitter
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = (np.arange(280) % 2 == 0).astype(float)

    n0 = len(h_init)
    target0 = n0 / 2.0
    h_proj = project_feasible(h_init, target0)
    h_bin0 = round_binary(h_proj, target0)
    best_c5 = compute_c5(h_bin0)
    best_h = h_bin0.copy()

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Binary local search from initial
    if remaining() > 60:
        t_alloc = min(200, remaining() - 30)
        h_b, c5_b = binary_ls(h_bin0, t_alloc, seed=seed)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 2: SLSQP refinement
    if remaining() > 80:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        bounds = [(0.0, 1.0)] * n_cur
        h_work = project_feasible(best_h, t_cur)
        for T in [50, 20, 10, 5, 3, 2, 1.5]:
            if remaining() < 40:
                break
            h_work = slsqp_optimize(h_work, n_cur, t_cur, T, 2000, bounds)
            c5_w = compute_c5(h_work)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_work.copy()
        # Binary round and re-search
        h_b2 = round_binary(best_h, len(best_h) / 2.0)
        c5_b2 = compute_c5(h_b2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
        if remaining() > 40:
            h_b3, c5_b3 = binary_ls(h_b2, min(60, remaining() - 20), seed=seed + 100)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()

    # Phase 3: Try different starting patterns at n0
    if remaining() > 100:
        starts = []
        for period in [4, 6, 8, 10, 12, 14, 16, 20, 28, 35]:
            if period < n0:
                h_p = make_periodic(n0, period, rng)
                starts.append(round_binary(h_p, target0))
        # Also try interpolated from best
        if remaining() > 40:
            for si, h_s in enumerate(starts[:6]):
                if remaining() < 20:
                    break
                alloc = max(5, min(30, int((remaining() - 10) / max(1, len(starts) - si))))
                h_t, c5_t = binary_ls(h_s, alloc, seed=seed + 200 + si)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t.copy()

    # Phase 4: Try different n values
    if remaining() > 120:
        n_alt_list = [100, 140, 160, 200, 240, 320, 400]
        for n_alt in n_alt_list:
            if remaining() < 60:
                break
            if n_alt % 2 != 0:
                continue
            t_alt = n_alt / 2.0
            # Interpolate best to new n
            if len(best_h) != n_alt:
                x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
                x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
                h_interp = np.interp(x_new, x_old % 2.0, best_h, period=2.0)
            else:
                h_interp = best_h.copy()
            h_new = round_binary(project_feasible(h_interp, t_alt), t_alt)
            # Also try periodic starts
            periods_try = [n_alt // 8, n_alt // 6, n_alt // 5, n_alt // 4]
            periods_try = [p for p in periods_try if p >= 2 and p < n_alt]
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            # Binary LS
            alloc = min(40, int(remaining() * 0.25))
            if alloc > 5:
                h_t, c5_t = binary_ls(h_new, alloc, seed=seed + 500 + n_alt)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t.copy()
            # Quick SLSQP
            if remaining() > 30:
                bounds_alt = [(0.0, 1.0)] * n_alt
                h_s = project_feasible(h_t, t_alt)
                h_s = slsqp_optimize(h_s, n_alt, t_alt, 5, 500, bounds_alt)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                h_b4 = round_binary(h_s, t_alt)
                c5_b4 = compute_c5(h_b4)
                if c5_b4 < best_c5:
                    best_c5 = c5_b4
                    best_h = h_b4.copy()

    # Phase 5: Final polish on best
    if remaining() > 30:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        bounds_fin = [(0.0, 1.0)] * n_fin
        h_fin = project_feasible(best_h, t_fin)
        for T_fin in [2.0, 1.5, 1.0]:
            if remaining() < 15:
                break
            h_fin = slsqp_optimize(h_fin, n_fin, t_fin, T_fin, 1500, bounds_fin)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        # Binary round final
        h_bf = round_binary(best_h, t_fin)
        c5_bf = compute_c5(h_bf)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        if remaining() > 15:
            h_fp, c5_fp = binary_ls(h_bf, min(20, int(remaining() - 5)), seed=seed + 999)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Final verification
    final_c5 = min(compute_c5(best_h), best_c5)
    assert abs(best_h.sum() - len(best_h) / 2.0) < 1e-6, f"Sum constraint violated: {best_h.sum()}"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, final_c5, len(best_h))
