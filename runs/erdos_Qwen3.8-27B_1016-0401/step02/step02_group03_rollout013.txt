            if w < 1e-15:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return val, grad * (2.0 / n)

_LS_flag = [False]

def _lb_obj(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _LS_flag[0] = True
    v, _ = _sm_val_and_grad(x, n, T, K)
    return v

def _lb_grad(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _LS_flag[0] = True
    _, g = _sm_val_and_grad(x, n, T, K)
    return g

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _LS_flag[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lb_obj, h, method='L-BFGS-B', jac=_lb_grad,
                       args=(n, T, K, t0, tlim),
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.1,
                                'maxls': 30, 'maxfun': maxiter * 3})
        if _LS_flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        if time.time() - t0 > tlim:
            _LS_flag[0] = True
        v, _ = _sm_val_and_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = _sm_val_and_grad(x, n, T, K)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    _LS_flag[0] = False
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if _LS_flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def nesterov_bgd(h_init, n, t0, tlim, max_iters=600000):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    f_star = 0.375
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        f_y = compute_c5(y)
        gap = max(f_y - f_star, 1e-7)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.3 / (1.0 + 0.002 * it))
        h_new = y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 3000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
    return h_best, c5_best

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(1, 8)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.008, 0.04)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += rng.normal(0, 0.008, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine (top-K smoothing)
    schedules = [(128, 5000), (64, 5000), (32, 6000), (16, 6000), (8, 8000), (4, 8000), (2, 10000), (1.0, 12000)]
    for T, mi in schedules:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        K = min(200, 2 * n - 1)
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-14, t0, budget_s - 60)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h_sg, c5_sg = nesterov_bgd(h_best, n, t0, t0 + rem - 180, 500000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: SLSQP with equality constraint (handles sum constraint exactly)
    for T in [4, 2, 1]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = do_slsqp(h_best, n, T, 100, 4000, 1e-15, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: L-BFGS-B fine polish
    for T in [0.5, 1.0, 2.0]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_new = do_lbfgsb(h_best, n, T, 80, 15000, 1e-15, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Diverse restarts (structured + random)
    n_rest = 50
    for pi in range(n_rest):
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        if pi % 3 == 0:
            h_p = struct_perturb(h_best, n, rng)
        elif pi % 3 == 1:
            sig = 0.004 + 0.012 * (pi % 4)
            h_p = h_best + rng.normal(0, sig, n)
        else:
            # Block flip
            h_p = h_best.copy()
            bs = n // 8
            bi = int(rng.integers(0, 8))
            h_p[bi*bs:(bi+1)*bs] = 1.0 - h_p[bi*bs:(bi+1)*bs]
            h_p = h_p + rng.normal(0, 0.003, n)
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 2, 150, 3000, 1e-13, t0, budget_s - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = do_slsqp(h_best, n, 1, 50, 5000, 1e-16, t0, t0 + rem - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 7: Double resolution if time allows
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n2
            for T, mi in [(16, 4000), (4, 6000), (1, 8000)]:
                rem = budget_s - (time.time() - t0)
                if rem < 80:
                    break
                h_new = do_lbfgsb(h_best, n, T, 200, mi, 1e-14, t0, budget_s - 30)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
