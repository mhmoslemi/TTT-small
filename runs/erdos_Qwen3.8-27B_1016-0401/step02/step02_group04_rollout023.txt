    center = n - 1
    viol = np.maximum(0, corr * dx - C_target)

        if v <= 0:

        # Actually: viol = max(0, corr*dx - C_target)

            p = -o
            grad[:n-p] += w * (1.0 - h[p:])

                res = minimize(obj_lo, h_lo.copy(), method='L-BFGS-B', jac=jac_lo,
                               bounds=[(0.0, 1.0)] * n_lo,
                               options={'maxiter': 3000, 'gtol': 1e-15})
                h_lo = project_box_sum(res.x, target_lo)
            except Exception:
                pass

        h_up = np.interp(x_full, x_lo, h_lo)
        h_up = project_box_sum(h_up, target)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            c5_best = c5_up
            h_best = h_up
            n = len(h_best)
            target = n / 2.0

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
