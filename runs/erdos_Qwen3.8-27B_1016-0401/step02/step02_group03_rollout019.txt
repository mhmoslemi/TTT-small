    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()

def corr_and_grad_shift(h, n, shift):
    """Compute f_shift and its gradient."""
    j = np.arange(n)
    if shift >= 0:
        if shift == 0:
            f = np.sum(h * (1.0 - h))
            g = 1.0 - 2.0 * h
        else:
            f = np.sum(h[shift:] * (1.0 - h[:n-shift]))
            g = np.zeros(n)
            g[shift:] += (1.0 - h[:n-shift])
            g[:n-shift] -= h[shift:]
    else:
        s = -shift
        f = np.sum(h[:n-s] * (1.0 - h[s:]))
        g = np.zeros(n)
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    return f, g

def full_corr(h, n):
    """Return full correlation array (2n-1 values)."""
    return np.correlate(h, 1.0 - h, mode="full")

_N = 0
_TARGET = 0
_T = 1.0
_ACTIVE = None  # list of shift offsets from center

def _soft_obj(x):

    center = _N - 1
    if _ACTIVE is not None:
        vals = np.array([corr[center + s] for s in _ACTIVE])
    else:
        vals = corr
    mc = vals.max()
    sh = _T * (vals - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

    _TARGET = n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def corr_and_grad_shift(h, n, shift):
    if shift >= 0:
        if shift == 0:
            f = np.sum(h * (1.0 - h))
            g = 1.0 - 2.0 * h
        else:
            f = np.sum(h[shift:] * (1.0 - h[:n - shift]))
            g = np.zeros(n)
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        f = np.sum(h[:n - s] * (1.0 - h[s:]))
        g = np.zeros(n)
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return f, g

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

_N = 0
_T = 1.0
_ACTIVE = None

def _obj_fn(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = _N - 1
    vals = np.array([corr[center + s] for s in _ACTIVE])
    mc = vals.max()
    sh = _T * (vals - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def _jac_fn(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = _N - 1
    vals = np.array([corr[center + s] for s in _ACTIVE])
    mc = vals.max()
    sh = _T * (vals - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(_N)
    for i, s in enumerate(_ACTIVE):
        if sw[i] < 1e-14:
            continue
        _, g = corr_and_grad_shift(x, _N, s)
        grad += sw[i] * g
    return grad * (2.0 / _N)

def _cons_fun(x):
    return x.sum() - _N / 2.0

def _cons_jac(x):
    return np.ones(_N)

def do_slsqp(h, n, T, active, maxiter, ftol):
    global _N, _T, _ACTIVE
    _N = n
    _T = float(T)
    _ACTIVE = list(active)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fn, h, method='SLSQP', jac=_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def find_active(h, n, m):
    corr = full_corr(h)
    center = n - 1
    top = np.argsort(corr)[::-1][:m]
    active = sorted(set(int(i - center) for i in top))
    return active

def nesterov_subgrad(h_init, n, t0, time_limit, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h_init.copy()
    h_prev = h_init.copy()

        h_new = project_feasible(h_new, n)

    c5f = compute_c5(h_cur)
