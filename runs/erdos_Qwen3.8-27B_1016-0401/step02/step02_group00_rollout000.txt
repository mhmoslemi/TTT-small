        target_alt = n_alt // 2
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = project_point(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        # short SLSQP polish: 2 rounds on worst lag
        for _ in range(2):
            if elapsed() > p3_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = np.argmax(corr_a)
            cons_alt = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (target_alt,)}]
            try:
                res = minimize(lag_obj, h_alt, args=(worst_a, target_alt), method='SLSQP',
                               jac=lambda x, l=worst_a, t=target_alt: lag_grad(x, l, t),
                               bounds=bounds_alt, constraints=cons_alt,
                               options={'maxiter': 400, 'ftol': 1e-17})
                h_alt = project_point(res.x, target_alt)
            except Exception:
                pass
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt

    # Phase 4: Final Nesterov + SLSQP polish (10% budget)
    p4_end = t0 + budget_s - 5.0
    n_f = len(h_best)
    target_f = n_f // 2
    bounds_f = [(0.0, 1.0)] * n_f
    h_cur = h_best.copy()
    h_mom = h_cur.copy()
    t_k = 1.0
    for it in range(2000):
        if elapsed() > p4_end:
            break
        y = h_cur + ((t_k - 1.0) / t_k) * (h_cur - h_mom)
        y = np.clip(y, 0.0, 1.0)
        grad, _ = smax_grad(y, 3000, top_k=15)
        h_cur = project_point(y - 0.001 * grad, target_f)
        h_mom = h_best
        h_best = h_cur
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        if it % 300 == 0:
            c5_f = compute_c5(h_cur)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_cur.copy()

    # Final SLSQP on worst lag
    corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
    worst_f = np.argmax(corr_f)
    cons_f = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (target_f,)}]
    try:
        res = minimize(lag_obj, h_best, args=(worst_f, target_f), method='SLSQP',
                       jac=lambda x, l=worst_f, t=target_f: lag_grad(x, l, t),
                       bounds=bounds_f, constraints=cons_f,
                       options={'maxiter': 1500, 'ftol': 1e-18})
        h_new = project_point(res.x, target_f)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
    except Exception:
        pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
