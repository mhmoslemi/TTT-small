import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T, target, pen):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n) + pen * (x.sum() - target) ** 2

def soft_max_grad(x, T, target, pen):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] -= w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= x[:n - p]
    grad *= dx
    grad += 2.0 * pen * (x.sum() - target)
    return grad

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def calc_subgrad(h, frac):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    thr = mc * (1.0 - frac)
    active = np.where(corr >= thr)[0]

    center = n - 1
    grad = np.zeros(n)
    for idx in active:
        o = int(idx - center)
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] -= h[o:]
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    grad /= len(active)
    return grad * dx

        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()
