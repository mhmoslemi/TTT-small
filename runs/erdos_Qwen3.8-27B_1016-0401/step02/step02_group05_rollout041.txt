                    diff -= s
        else:

                s = min(-diff, h[i])

            length = int(rng.integers(5, 40))
            h_try[start:start + length] = 1.0 - h_try[start:start + length]
        else:
            h_try = h_best.copy()
            for _ in range(int(rng.integers(2, 6))):
                start = int(rng.integers(0, max(1, n - 10)))
                length = int(rng.integers(2, 10))
                h_try[start:start + length] = 1.0 - h_try[start:start + length]
        h_try = fix_constraint(h_try, target)
        # Quick SLSQP polish
        try:
            res = minimize(soft_max_obj, h_try, method='SLSQP',
                          jac=lambda x: soft_max_grad(x, 5), args=(5,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-14})
            if res.fun < 100:
                h_try = fix_constraint(res.x.copy(), target)
        except Exception:
            pass
        c5_try, _ = compute_c5(h_try)
        if c5_try < c5_best - 1e-13:
            c5_best = c5_try
            h_best = h_try
        # Brief local search
        h_best, c5_ls = two_opt_cycling(h_best, target, n_iter=2000, t0=t0,
                                         budget=budget_s - 40, n_shifts=3)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_best

    # Final polish
    h_best, c5_fin = two_opt_cycling(h_best, target, n_iter=5000, t0=t0,
                                      budget=budget_s - 2, n_shifts=5)
    if c5_fin < c5_best:
        c5_best = c5_fin
        h_best = h_best

    c5_final, _ = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
