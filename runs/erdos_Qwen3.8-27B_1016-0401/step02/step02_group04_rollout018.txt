def grad_corr_weights(h, w):
    """Compute sum_k w[k] * d(corr[k])/d(h), vectorized.
    w has length 2n-1, indexed 0..2n-2.
    corr index m corresponds to shift s = m - (n-1).
    """
    n = len(h)
    center = n - 1
    grad = np.zeros(n)
    
    # s = 0: d/dh[j] of sum h[i](1-h[i]) = (1-2h)[j]

    # s > 0: corr[center+s] = sum_{i=0}^{n-1-s} h[i+s](1-h[i])

        grad[s:] += ws * (1.0 - h[:n-s])
        # -h[j+s] for j < n-s

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_weighted(h, w):
    """Gradient of sum w[k]*corr[k] w.r.t. h, * dx."""
    n = len(h)
    c = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    w0 = w[c]
    g += w0 * (1.0 - 2.0 * h)
    for s in range(1, n):
        wp = w[c + s]
        if wp != 0.0:
            g[s:] += wp * (1.0 - h[:n-s])
            g[:n-s] -= wp * h[s:]
        wn = w[c - s]
        if wn != 0.0:
            g[:n-s] += wn * (1.0 - h[s:])
            g[s:] -= wn * h[:n-s]
    return g * dx

def soft_w(corr, T):
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    return e / e.sum()

def topk_w(corr, k):
    n = len(corr)
    idx = np.argsort(corr)[::-1][:k]
    w = np.zeros(n)
    w[idx] = 1.0 / k
    return w

def proj_box_sum(h, target):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0, 1)

def lbfgs(h0, n, target, obj_fn, grad_fn, maxiter, bounds):
    try:
        res = minimize(obj_fn, h0.copy(), method='L-BFGS-B', jac=grad_fn,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14, 'maxls': 50})
        return res.x
    except:
        return h0.copy()

def make_obj_grad(h_init, T, k, alpha, pen):
    n = len(h_init)
    target = n / 2.0
    ones = np.ones(n)
    def obj(x):
        corr = get_corr(x)
        sw = soft_w(corr, T)
        sm = np.max(corr) + np.log(np.sum(np.exp(T*(corr - np.max(corr))))) / T
        tw = topk_w(corr, k)
        topk_mean = np.sum(tw * corr)
        val = alpha * sm + (1.0 - alpha) * topk_mean
        val *= (2.0 / n)
        val += pen * (x.sum() - target) ** 2
        return float(val)
    def grad(x):
        corr = get_corr(x)
        sw = soft_w(corr, T)
        tw = topk_w(corr, k)
        w = alpha * sw + (1.0 - alpha) * tw
        g = grad_weighted(x, w) + 2.0 * pen * (x.sum() - target) * ones
        return g
    return obj, grad

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_weighted(h, w):
    n = len(h)
    c = n - 1
    dx = 2.0 / n
    g = w[c] * (1.0 - 2.0 * h)
    for s in range(1, n):
        wp = w[c + s]
        if abs(wp) > 1e-15:
            g = g.copy()
            g[s:] += wp * (1.0 - h[:n - s])
            g[:n - s] -= wp * h[s:]
        wn = w[c - s]
        if abs(wn) > 1e-15:
            g[:n - s] += wn * (1.0 - h[s:])
            g[s:] -= wn * h[:n - s]
    return g * dx

def soft_weights(corr, T):
    mc = np.max(corr)
    s = T * (corr - mc)
    e = np.exp(s)
    return e / e.sum()

def topk_weights(corr, k):
    k = min(k, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / k
    return w

def proj_box_sum(h, target):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

    pass

def obj_fn(x, p):
    corr = get_corr(x)
    mc = np.max(corr)
    s = p.T * (corr - mc)
    sm = mc + np.log(np.sum(np.exp(s))) / p.T
    tw = topk_weights(corr, p.k)
    tk = np.dot(tw, corr)
    val = (p.alpha * sm + (1.0 - p.alpha) * tk) * (2.0 / p.n)
    val += p.pen * (x.sum() - p.target) ** 2
    return float(val)

def grad_fn(x, p):
    corr = get_corr(x)
    sw = soft_weights(corr, p.T)
    tw = topk_weights(corr, p.k)
    w = p.alpha * sw + (1.0 - p.alpha) * tw
    g = grad_weighted(x, w)
    g += 2.0 * p.pen * (x.sum() - p.target)
    return g

        h_n = proj_box_sum(h_n, t_p)
        c5_t = compute_c5(h_n)
        if c5_t

 < c5_best:
            c5_best = c5_t
            h_best = h_n

    c5_final = compute_c5(h_best)
    return (h_best, min(c5_final, c5_best), len(h_best))
