import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    dx = 2.0 / n
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    grad += sw[center] * (1.0 - 2.0 * h)
    for o in range(1, n):
        w = sw[center + o]
        if w > 1e-12:
            grad[:n-o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n-o]
        w2 = sw[center - o]
        if w2 > 1e-12:
            grad[:n-o] += w2 * (1.0 - h[o:])
            grad[o:] -= w2 * h[:n-o]
    return grad * dx

def topk_grad(h, k=5):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    dx = 2.0 / n
    # Get top-k indices
    top_idx = np.argsort(corr)[-k:]
    # Average gradient over top-k lags
    grad = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n-o] += (1.0 - h[o:])
            grad[o:] -= h[:n-o]
        else:
            p = -o
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] -= h[:n-p]
    grad /= k
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgs(h0, n, target, T, pen, maxiter, use_topk=False):
    bounds = [(0.0, 1.0)] * n
    total = n / 2.0
    def obj(x):
        v = softmax_obj(x, T)
        return v + pen * (x.sum() - total) ** 2
    def grad(x):
        g = softmax_grad(x, T) if not use_topk else topk_grad(x, 5)
        return g + 2.0 * pen * (x.sum() - total)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_box_sum(res.x, target)
    except:
        return h0.copy()

def softmax_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")

    dx = 2.0 / n

    e = np.exp(shifted)

            grad[o:] -= w1 * h[:n-o]

            grad[o:] += w2 * (1.0 - h[:n-o])
            grad[:n-o] -= w2 * h[o:]

    n = len(h)

        # But for lag=0: both terms apply, giving (1-h[i]) - h[i] = 1-2h[i]. Correct!

        i_minus = i - lags  # array of indices

        valid_minus = (i_minus >= 0) & (i_minus < n)

        h_plus = np.zeros(2*n-1)
        h_minus = np.zeros(2*n-1)
        h_plus[valid_plus] = h[i_plus[valid_plus]]
        h_minus[valid_minus] = h[i_minus[valid_minus]]

        c[valid_plus] += (1.0 - h_plus[valid_plus])
        c[valid_minus] -= h_minus[valid_minus]

        # The function is convex piecewise linear.
        # Find where the slope changes from negative to positive.
        
        # Simple approach: ternary search on [0,1] (since f is convex)
        # Or: evaluate at v=0, v=1, and the breakpoints.

        # Too many breakpoints. Use the fact that we only need the minimum.

            else:

                hi = mid
        v_new = (lo + hi) / 2.0

        # Update h and corr
        delta = v_new - old
        if abs(delta) > 1e-15:
            h[i] = v_new
            corr += delta * c

    h = project_box_sum(h, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_og(h, T):
    """Softmax objective and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    dx = 2.0 / n
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    c = n - 1
    g = np.zeros(n)
    if sw[c] > 1e-13:
        g += sw[c] * (1.0 - 2.0 * h)
    for o in range(1, n):
        w1 = sw[c + o]
        if w1 > 1e-13:
            g[:n-o] += w1 * (1.0 - h[o:])
            g[o:] -= w1 * h[:n-o]
        w2 = sw[c - o]
        if w2 > 1e-13:
            g[o:] += w2 * (1.0 - h[:n-o])
            g[:n-o] -= w2 * h[o:]
    return obj, g * dx

def proj(h, tgt):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgs(h0, tgt, T, pen, maxit):
    n = len(h0)
    tot = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def fo(x):
        v, _ = sm_og(x, T)
        return v + pen * (x.sum() - tot) ** 2
    def go(x):
        _, g = sm_og(x, T)
        return g + 2.0 * pen * (x.sum() - tot)
    try:
        r = minimize(fo, h0.copy(), method='L-BFGS-B', jac=go,
                     bounds=bnds, options={'maxiter': maxit, 'gtol': 1e-12})
        return proj(r.x, tgt)
    except:
        return h0.copy()

def coord_sweep(h):
    """One coordinate descent sweep (box constraints only)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = n - 1  # center
    lags = np.arange(2*n-1) - c
    
    for i in range(n):
        old = h[i]
        # Compute slope c[k] for all lags
        slopes = np.zeros(2*n-1)
        for k in range(2*n-1):
            p = lags[k]
            if p == 0:
                slopes[k] = 1.0 - 2.0 * old
            elif p > 0:
                s = 0.0
                if i >= p and i - p < n:
                    s += 1.0 - h[i-p]
                if 0 <= i + p < n:
                    s -= h[i+p]
                slopes[k] = s
            else:
                q = -p
                s = 0.0
                if 0 <= i + q < n:
                    s += 1.0 - h[i+q]
                if q <= i and i - q < n:
                    s -= h[i-q]
                slopes[k] = s
        
        # 1D min-max: min_v max_k (corr[k] + slopes[k]*(v - old))
        b = corr - slopes * old
        lo, hi = 0.0, 1.0
        for _ in range(30):
            mid = (lo + hi) / 2.0
            vals = b + slopes * mid
            fv = vals.max()
            act = np.where(np.abs(vals - fv) < 1e-12)[0]
            ms = slopes[act].max()
            ns = slopes[act].min()
            if ns >= 0:

            else:
                lo = hi = mid

        v_new = (lo + hi) / 2.0
        v_new = np.clip(v_new, 0.0, 1.0)
        delta = v_new - old
        if abs(delta) > 1e-15:
            h[i] = v_new
            corr += delta * slopes
    return h, corr
