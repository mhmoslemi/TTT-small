import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += diff * room / total
        else:
            avail = h
            total = avail.sum()
            if total < 1e-14:
                break
            h += diff * avail / total
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] -= w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    return grad * dx

def sum_eq_fun(x, target):
    return x.sum() - target

    return np.ones(len(x))

def minimize_1d(c, d):

    d_max = np.max(d)
    active0 = np.where(d >= d_max - 1e-12)[0]
    if np.min(c[active0]) >= -1e-10:
        return 0.0

    active1 = np.where(c + d >= cd_max - 1e-12)[0]
    if np.max(c[active1]) <= 1e-10:

        elif max_s <= 1e-10:

        else:

def coord_descent_pass(h, corr, dx, lam, n):
    """One pass of coordinate descent with Lagrangian for sum constraint."""
    for j in range(n):
        a = np.zeros(2 * n - 1)
        lo1 = n - 1 - j
        hi1 = 2 * n - 2 - j
        idx1 = np.arange(lo1, hi1 + 1)
        a[idx1] += (1.0 - h[idx1 - (n - 1) + j])

        hi2 = n - 1 + j

from scipy.optimize import minimize
from functools import partial

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += diff * room / tot
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += diff * h / tot
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] -= w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    return grad * dx

    return x.sum() - target

        return 1.0
    lo, hi = 0.0, 1.0

        ms, xs = np.min(c[act]), np.max(c[act])
        if ms >= -1e-10:
            hi = mid
        elif xs <= 1e-10:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def coord_pass(h, corr, dx, lam, n):
    for j in range(n):
        a = np.zeros(2 * n - 1)
        lo1, hi1 = n - 1 - j, 2 * n - 2 - j
        idx1 = np.arange(lo1, hi1 + 1)
        a[idx1] += (1.0 - h[idx1 - (n - 1) + j])
        lo2, hi2 = j, n - 1 + j
        idx2 = np.arange(lo2, hi2 + 1)
        a[idx2] -= h[idx2 - (n - 1) + j]
        c = a * dx + lam
        d = (corr - a * h[j]) * dx
        t_opt = minimize_1d(c, d)
        delta = t_opt - h[j]
        if abs(delta) > 1e-14:
            h[j] = t_opt
            corr += delta * a
    return h, corr
