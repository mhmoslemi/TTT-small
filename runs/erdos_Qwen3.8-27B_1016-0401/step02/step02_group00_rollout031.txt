def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_at_lag(h, lag_idx):
    n = len(h)
    center = n - 1
    o = lag_idx - center
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * h)
    elif o > 0:
        g[o:] += (1.0 - h[:n-o])
        g[:n-o] += -h[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - h[p:])
        g[p:] += -h[:n-p]
    return g

def project_feasible(v, target):
    """Project v onto {sum(h)=target, 0<=h<=1}"""
    n = len(v)
    # Bisection on mu: find mu such that sum(clamp(v - mu, 0, 1)) = target
    lo = v.min() - target

        if s > target:
            lo = mid
        else:
            hi = mid

            else:

    n = len(h)
    corr = compute_corr(h)
    # Get top_k lags

    for idx in top_idx:
        g += grad_at_lag(h, idx)

    g *= (2.0 / n)  # scale by dx

    o = lag_idx - center

  The second sum is a quadratic form: h^T A h where A has 1s at positions (i+k, i) for i=0..n-1-k.
  This is NOT convex (A is not PSD in general).

So we can't use standard convex optimization for a single lag. SLSQP is the practical choice.

Let me focus on making the SLSQP calls more effective:
1. Better initial guess (warm start)
2. More iterations
3. Handling multiple lags simultaneously

For handling multiple lags, I'll use the log-sum-exp smoothing with a very high T (e.g., T=5000), which effectively makes the smooth objective very close to the max.

- This penalizes any lag that exceeds the current worst

For the step size, I'll use:

import numpy as np
from scipy.optimize import minimize
import time

_GT = 0.0
_GN = 0
_GL = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_at_lag(h, lag_idx):
    n = len(h)
    center = n - 1
    o = lag_idx - center
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * h)
    elif o > 0:
        g[o:] += (1.0 - h[:n - o])
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += (1.0 - h[p:])
        g[p:] += -h[:n - p]
    return g

def project_feasible(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    diff -= add
            else:
                sub = min(-diff, h[i])
                if sub > 0:
                    h[i] -= sub
                    diff += sub
            if abs(diff) < 1e-15:
                break
    return h

def obj_lag(x):
    dx = 2.0 / _GN
    return float(np.correlate(x, 1.0 - x, mode="full")[_GL] * dx)

def obj_lag_grad(x):
    return grad_at_lag(x, _GL) * (2.0 / _GN)

def eq_fun(x):
    return x.sum() - _GT

def eq_jac(x):
    return np.ones(_GN)

def smooth_obj(x, T):
    n = _GN
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def smooth_obj_grad(x, T):
    n = _GN
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    e = np.exp(shifted)
    sw = e / np.sum(e)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if sw[idx] > 1e-13:
            g += sw[idx] * grad_at_lag(x, idx)
    return g * (2.0 / n)

def nesterov_loop(h0, target, n_iter, top_k, step_init, step_decay):
    n = len(h0)
    dx = 2.0 / n
    h_curr = h0.copy()
    h_mom = h0.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    prev_g = np.zeros(n)
    for it in range(n_iter):
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        h_y = h_mom + (t_k - 1.0) / t_k * (h_mom - h_curr)
        h_y = np.clip(h_y, 0.0, 1.0)
        h_y = project_feasible(h_y, target)
        corr = compute_corr(h_y)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            g += grad_at_lag(h_y, idx)
        g /= top_k
        g *= dx
        gg = float(np.dot(g, g))
        if gg < 1e-30:
            break
        pg = float(np.dot(h_curr, prev_g))
        sg = float(np.dot(prev_g, prev_g))
        if sg > 1e-20:
            bb_step = max(0.005, min(2.0, sg / (gg + 1e-20)))
        else:
            bb_step = step_init
        step = bb_step * step_decay ** (it // 100)
        for s_mult in [0.25, 0.5, 1.0, 2.0, 4.0]:
            s = step * s_mult
            h_cand = h_y - s * g
            h_cand = project_feasible(h_cand, target)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
                h_new = h_cand
                break
        else:
            h_new = h_y - step * g
            h_new = project_feasible(h_new, target)
        prev_g = g
        h_mom = h_new
        h_curr = h_new
    return best_h, best_c5

def slsqp_polish(h0, target, bounds, max_outer, max_iter):
    global _GT, _GN, _GL
    n = len(h0)
    _GT = target
    _GN = n
    h_curr = h0.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    dx = 2.0 / n
    for outer in range(max_outer):
        corr = compute_corr(h_curr)
        top_idx = np.argsort(corr)[-2:]
        for lag in top_idx:
            _GL = int(lag)
            cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
            try:
                res = minimize(obj_lag, h_curr, jac=obj_lag_grad, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    best_h = h_new
                    h_curr = h_new
            except Exception:
                pass
    return best_h, best_c5

def smooth_polish(h0, target, bounds, T, n_iter):
    global _GT, _GN
    n = len(h0)
    _GT = target
    _GN = n
    h_curr = h0.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    dx = 2.0 / n
    for it in range(n_iter):
        corr = compute_corr(h_curr)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted -= shifted.max()
        e = np.exp(shifted)
        sw = e / np.sum(e)
        g = np.zeros(n)
        for idx in range(2 * n - 1):
            if sw[idx] > 1e-12:
                g += sw[idx] * grad_at_lag(h_curr, idx)
        g *= dx
        step = 0.08 / (1.0 + 0.003 * it)
        h_new = h_curr - step * g
        h_new = project_feasible(h_new, target)
        h_curr = h_new
        if it % 100 == 0:
            c5 = compute_c5(h_curr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _GT, _GN
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    _GT = target
    _GN = n

    # Phase 1: Nesterov subgradient with line search (30%)
    p1_end = t0 + budget_s * 0.30
    for rd in range(4):
        if elapsed() > p1_end:
            break
        pk = 3 if rd < 2 else 5
        h_best, c5r = nesterov_loop(h_best, target, 2500, pk, 0.15 / (rd + 1), 0.97)
        if c5r < c5_best:
            c5_best = c5r

    # Phase 2: SLSQP on top-2 lags (25%)
    p2_end = t0 + budget_s * 0.55
    bounds = [(0.0, 1.0)] * n
    h_best, c5s = slsqp_polish(h_best, target, bounds, 40, 800)
    if c5s < c5_best:
        c5_best = c5s
    if elapsed() < p2_end:
        h_best, c5s = slsqp_polish(h_best, target, bounds, 30, 1200)
        if c5s < c5_best:
            c5_best = c5s

    # Phase 3: Smooth max optimization (15%)
    p3_end = t0 + budget_s * 0.70
    for T in [300, 800, 2000]:
        if elapsed() > p3_end:
            break
        h_best, c5m = smooth_polish(h_best, target, bounds, T, 600)
        if c5m < c5_best:
            c5_best = c5m
            h_best = fix_constraint(h_best, target)

    # Phase 4: Multi-n exploration (15%)
    p4_end = t0 + budget_s * 0.85
    for n_alt in [252, 308, 350, 420]:
        if elapsed() > p4_end:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        h_alt, c5_alt = nesterov_loop(h_alt, target_alt, 1500, 3, 0.12, 0.98)
        _GT = target_alt
        _GN = n_alt
        h_alt, c5_alt = slsqp_polish(h_alt, target_alt, bounds_alt, 15, 600)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = target_alt
            bounds = bounds_alt

    # Phase 5: Final SLSQP polish (12%)
    p5_end = t0 + budget_s - 5.0
    _GT = target
    _GN = len(h_best)
    bounds_f = [(0.0, 1.0)] * len(h_best)
    h_best, c5f = slsqp_polish(h_best, target, bounds_f, 50, 2000)
    if c5f < c5_best:
        c5_best = c5f

    # Phase 6: Tiny nesterov cleanup (remaining)
    if elapsed() < p5_end:
        h_best, c5t = nesterov_loop(h_best, target, 2000, 3, 0.05, 0.99)
        if c5t < c5_best:
            c5_best = c5t

    h_best = fix_constraint(np.clip(h_best, 0.0, 1.0), len(h_best) / 2.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
