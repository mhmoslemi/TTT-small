            # Block flip
            block_size = rng.integers(3, min(15, n // 4))
            start = rng.integers(0, n - block_size)
            h_pert = h_best.copy()
            h_pert[start:start + block_size] = 1.0 - h_pert[start:start + block_size]
        h_pert = project_box_sum(h_pert, target)
        T = rng.integers(100, 1500)
        h_new = lbfgs_call(h_pert, T, target, n, 1500, pen=10.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: FISTA local search with small perturbations
    while elapsed() < budget_s - 80:
        h_pert = h_best + rng.normal(0, 0.03, n)
        h_pert = project_box_sum(h_pert, target)
        T = rng.integers(300, 2000)
        step = min(0.15, 2.0 / (T + 2.0))
        h_new = fista_opt(h_pert, T, target, n, 3000, step)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try different resolutions
    for n_try in [140, 200, 100]:
        if elapsed() > budget_s - 40:
            break
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        best_local = h_try.copy()
        best_try_c5 = c5_try
        for T in [200, 800]:
            if elapsed() > budget_s - 15:
                break
            h_t = lbfgs_call(best_local, T, t_try, n_try, 2000, pen=10.0)
            c5_t = compute_c5(h_t)
            if c5_t < best_try_c5:
                best_try_c5 = c5_t
                best_local = h_t
        if best_try_c5 < c5_best:
            c5_best = best_try_c5
            h_best = best_local
            n = n_try
            target = t_try

    # Phase 7: Final high-T polish
    if elapsed() < budget_s - 5:
        h_new = lbfgs_call(h_best, 5000, target, n, 3000, pen=5.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        # One more FISTA polish
        step = 0.1
        h_new2 = fista_opt(h_best, 3000, target, n, 2000, step)
        c5 = compute_c5(h_new2)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new2

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
