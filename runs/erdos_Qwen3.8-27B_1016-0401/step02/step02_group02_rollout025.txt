import time
from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def corr_grad(h, s_list, n):
    """Gradient of corr[k] w.r.t. h for given shifts k (full correlation indices)."""
    center = n - 1
    g = np.zeros((len(s_list), n))
    for idx, k in enumerate(s_list):
        s = k - center
        gi = g[idx]
        if s == 0:
            gi[:] = 1.0 - 2.0 * h
        elif s > 0:
            gi[:n-s] += (1.0 - h[s:])
            gi[s:] -= h[:n-s]
        else:
            p = -s
            gi[p:] += (1.0 - h[:n-p])
            gi[:n-p] -= h[p:]
    return g  # shape (len(s_list), n)

def active_set_optimize(h, n, target, max_iter=20, K=15):
    """Iterative active-set LP method."""
    h_cur = h.copy()
    for it in range(max_iter):
        corr = get_corr(h_cur)
        mc = np.max(corr)
        # Find active set: shifts with corr in top-K
        k_active = np.argsort(corr)[-K:]
        # Linearize: corr_k(h) ≈ corr_k(h_cur) + grad_k . (h - h_cur)
        grads = corr_grad(h_cur, k_active, n)  # (K, n)
        c_vals = corr[k_active]  # (K,)
        
        # LP: minimize t
        # s.t. c_k + grad_k . (h - h_cur) <= t  for k in active
        #      0 <= h_i <= 1
        #      sum(h) = target
        #
        # Variables: [h (n), t (1)]
        # Objective: minimize t → c = [0,...,0, 1]
        # Constraints (ineq: A_ub @ x <= b_ub):
        #   c_k + grad_k . h - grad_k . h_cur - t <= 0  →  [grad_k, -1] . [h,t] <= -c_k + grad_k.h_cur
        # Equality: sum(h) = target → A_eq @ x = b_eq
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        # Inequality constraints: for each active k
        # c_k + grad_k . (h - h_cur) <= t
        # c_k + grad_k . h - grad_k . h_cur - t <= 0
        # [grad_k, -1] . [h; t] <= -c_k + grad_k . h_cur
        A_ub = np.zeros((len(k_active), n + 1))
        b_ub = np.zeros(len(k_active))
        for i in range(len(k_active)):
            A_ub[i, :n] = grads[i]
            A_ub[i, n] = -1.0
            b_ub[i] = -c_vals[i] + grads[i] @ h_cur
        
        # Also add 0 <= h_i (as -h_i <= 0) and h_i <= 1
        # Linprog handles bounds separately for h, but t has no bounds
        # Actually, let's just use bounds
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
            if res.status == 0:
                h_new = res.x[:n]
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
            else:
                break
        except Exception:
            break
        
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h_cur)
        # Line search: take a step toward h_new
        alpha = 1.0
        for _ in range(10):

            if compute_c5(h_try) < c5_old or alpha < 0.01:
                h_cur = h_try
                break
            alpha *= 0.5
    
    return h_cur

import numpy as np
import time
from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def corr_grad_shift(h, s, n):
    """Gradient of corr[n-1+s] w.r.t. h."""
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def active_set_lp_opt(h, n, target, K=15, max_iter=30):
    """Iterative active-set linearized optimization."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for it in range(max_iter):
        corr = get_corr(h_cur)
        k_active = np.argsort(corr)[-K:]
        c_vals = corr[k_active]
        max_c = c_vals.max()
        # Build LP: min t s.t. corr_k(h) ~ c_k + g_k.(h-h_cur) <= t
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        gsum = np.zeros(n)
        for i in range(K):
            g = corr_grad_shift(h_cur, k_active[i] - (n - 1), n)
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = -c_vals[i] + g @ h_cur
            gsum += g
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            h_lp = np.clip(res.x[:n], 0.0, 1.0)
            h_lp = project_feasible(h_lp, target)
        except Exception:
            break
        # Line search
        best_alpha = 0.0
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
            h_try = h_cur + alpha * (h_lp - h_cur)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                best_alpha = alpha
                h_cur = h_try
                c5_cur = c5_try
                break
    return h_cur, c5_cur

def binary_anneal(h_init, n, target, n_iters, rng, t_start=1.0):
    """Simulated annealing on binary solutions via swaps."""
    h_bin = np.zeros(n)
    order = np.argsort(h_init)[::-1]
    h_bin[order[:int(target)]] = 1.0
    c5_cur = compute_c5(h_bin)
    corr_cur = get_corr(h_bin)
    max_e = n - 1
    best_h = h_bin.copy()
    best_c5 = c5_cur
    ones = np.where(h_bin == 1.0)[0].tolist()
    zeros = np.where(h_bin == 0.0)[0].tolist()
    t_end = t_start * 1e-5
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        a = int(rng.integers(len(ones)))
        b = int(rng.integers(len(zeros)))
        ia, ib = ones[a], zeros[b]
        # Compute delta in max corr
        delta_corr = np.zeros(2 * n - 1)
        for s in range(-(n - 1), n):
            # Change at shift s for positions ia (1->0) and ib (0->1)
            d = 0.0
            # ia goes 1->0: removes h[ia]*(1-h[ia+s]) and h[ia-s]*(1-h[ia])
            if 0 <= ia < n and 0 <= ia + s < n:
                d -= (1.0 - h_bin[ia + s])
            if 0 <= ia - s < n and 0 <= ia < n:
                d -= h_bin[ia - s]
            # ib goes 0->1: adds h[ib]*(1-h[ib+s]) and h[ib-s]*(1-h[ib])
            if 0 <= ib < n and 0 <= ib + s < n:
                d += (1.0 - h_bin[ib + s])
            if 0 <= ib - s < n and 0 <= ib < n:
                d += h_bin[ib - s]
            delta_corr[n - 1 + s] = d
        corr_new = corr_cur + delta_corr
        max_new = corr_new.max()
        max_old = corr_cur.max()
        delta = max_new - max_old
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            h_bin[ia], h_bin[ib] = 0.0, 1.0
            ones[a] = ib
            zeros[b] = ia
            corr_cur = corr_new
            c5_cur = max_new * (2.0 / n)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_bin.copy()
    return best_h, best_c5

def slsqp_refine(h, n, target, T, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = get_corr(x)
        mc = c.max()
        s = T * (c - mc)
        s = s - s.max()
        return (mc + np.log(np.sum(np.exp(s))) / T) * dx
    def jac(x):
        c = get_corr(x)
        mc = c.max()
        s = T * (c - mc)
        s = s - s.max()
        e = np.exp(s)
        w = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for idx in nz:
            o = int(idx) - center
            wi = w[idx]
            gg = corr_grad_shift(x, o, n)
            g += wi * gg
        return g * dx
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgrad_polish(h, n, target, n_steps, K=5):
    dx = 2.0 / n
    c5_local = compute_c5(h)
    for it in range(n_steps):
        corr = get_corr(h)
        top_idx = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for idx in top_idx:
            g += corr_grad_shift(h, int(idx) - (n - 1), n)
        g /= K
        for lr in [0.08, 0.04, 0.02, 0.01]:
            h_cand = project_feasible(h - lr * g, target)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_local:
                h = h_cand
                c5_local = c5_cand
                break
        else:
            h = project_feasible(h - 0.005 * g, target)
            c5_local = compute_c5(h)
    return h, c5_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
