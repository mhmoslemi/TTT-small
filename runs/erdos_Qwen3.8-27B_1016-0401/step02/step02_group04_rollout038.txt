def adam_opt(h0, target, T, mi, tk=None, lr=0.02):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, ep = 0.8, 0.999, 1e-8
    for t in range(mi):
        g = sm_grad(h, T, tk)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** (t + 1))
        vh = v / (1 - b2 ** (t + 1))
        h = project_box_sum(h - lr * mh / (np.sqrt(vh) + ep), target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def el():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Progressive L-BFGS-B with increasing T
    for T in [200, 600, 2000, 6000, 15000]:
        if el() > budget_s - 350:
            break
        h_new = lbfgsb_opt(h_best, target, T, 10.0, 4000, tk=min(15, n))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Adam with high T for fine tuning
    for T, tk in [(1000, 6), (4000, 4), (10000, 4), (25000, 3)]:
        if el() > budget_s - 300:
            break
        h_new = adam_opt(h_best, target, T, 4000, tk=tk, lr=0.025)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
        # Second Adam pass with slightly adjusted lr
        h_new = adam_opt(h_best, target, T, 2000, tk=tk, lr=0.01)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 3: Perturbation-based local search
    cnt = 0
    best_since = 0
    while el() < budget_s - 130:
        cnt += 1
        scale = max(0.015, 0.09 / (1.0 + 0.04 * cnt))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T = 800 + int(rng.integers(0, 5000))
        tk = int(rng.integers(3, 12))
        h_new = lbfgsb_opt(h_pert, target, T, 20.0, 2000, tk=tk)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best, best_since = c5, h_new, cnt
        if cnt - best_since > 20 and cnt % 20 == 0:
            scale = 0.15

    # Phase 4: Very high T final polish (multiple passes)
    for T in [8000, 20000, 50000]:
        if el() > budget_s - 20:
            break
        h_new = lbfgsb_opt(h_best, target, T, 50.0, 3000, tk=5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 5: Try different resolutions
    if el() < budget_s - 50:
        n_cands = [n // 2, n * 2, max(40, n - 20), n + 20]
        n_cands = [x for x in n_cands if 20 <= x <= 560]
        for n_try in n_cands:
            if el() > budget_s - 25:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, h_best)
            h_try = project_box_sum(h_try, n_try / 2.0)
            c5_try = compute_c5(h_try)
            bl, bc = h_try.copy(), c5_try
            for T in [500, 3000, 10000]:
                if el() > budget_s - 8:
                    break
                ht = lbfgsb_opt(bl, n_try / 2.0, T, 10.0, 2000, tk=min(10, n_try))
                ct = compute_c5(ht)
                if ct < bc:
                    bc, bl = ct, ht
            if bc < c5_best:
                c5_best, h_best = bc, bl
                n, target = n_try, n_try / 2.0

    # Final micro-polish
    if el() < budget_s - 3:
        h_new = lbfgsb_opt(h_best, target, 2000, 10.0, 2000, tk=8)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
        h_new = lbfgsb_opt(h_best, target, 50000, 50.0, 1500, tk=3)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    return (h_best, min(compute_c5(h_best), c5_best), len(h_best))
