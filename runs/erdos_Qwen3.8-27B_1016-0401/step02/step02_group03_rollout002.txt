from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def f_shift(h, n, s):
    """Compute f_s = sum_{i=0}^{n-1-s} h[i](1-h[i+s]) for s >= 0"""
    m = n - s
    return float(np.sum(h[:m] * (1.0 - h[s:s+m])))

def grad_f_shift(h, n, s):
    """Gradient of f_s w.r.t. h"""
    g = np.zeros(n)
    m = n - s
    # linear part: +1 for i in [0, m-1]
    g[:m] += 1.0
    # quadratic part: -h[i+s] for i in [0, m-1] → g[i] -= h[i+s]
    g[:m] -= h[s:s+m]  # g[i] -= h[i+s] for i in [0, m-1]
    # quadratic part: -h[i-s] for i in [s, n-1] → g[i] -= h[i-s]
    if s > 0:
        g[s:] -= h[:m]  # g[i] -= h[i-s] for i in [s, n-1], h[i-s] = h[0:m]
    return g

def smooth_obj(x, n, T):
    """Smoothed max of all f_k * dx using softmax"""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    if T < 1e-6:
        return mc
    sh = T * (corr - mc)
    e = np.exp(sh)
    return mc + np.log(e.sum()) / T

def smooth_grad(x, n, T):
    """Gradient of smoothed max"""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    # grad = sum_k sw[k] * grad_f_k * dx
    # But computing each grad_f_k is O(n), and we have 2n-1 of them
    # Total: O(n^2) which is fine for n=280
    center = n - 1
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        s = oi - center
        if s >= 0:
            g = grad_f_shift(x, n, s)
        else:
            g = grad_f_shift(x, n, -s)  # by symmetry f_{-s} = f_s
        grad += w * g
    return grad * dx

def cons_fun(x, n):
    return x.sum() - n / 2.0

def cons_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: cons_fun(x, n), 'jac': lambda x: cons_jac(x, n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, T), method='SLSQP', 
                       jac=lambda x: smooth_grad(x, n, T),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_f_shift(h, n, s):
    g = np.zeros(n)
    m = n - s
    g[:m] += 1.0
    g[:m] -= h[s:s + m]
    if s > 0:
        g[s:] -= h[:m]
    return g

def smooth_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    if T < 1e-6:
        return mc
    sh = T * (corr - mc)
    e = np.exp(sh)
    return mc + np.log(e.sum()) / T

def smooth_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    if T < 1e-6:
        idx = int(np.argmax(corr))
        s = abs(idx - (n - 1))
        return grad_f_shift(x, n, s) * dx
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-9:
            continue
        s = abs(oi - center)
        grad += w * grad_f_shift(x, n, s)
    return grad * dx

def _cf(x, n):
    return x.sum() - n / 2.0

def _cj(x, n):
    return np.ones(n)

def do_slsqp(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _cf(x, n), 'jac': lambda x: _cj(x, n)}]
    try:
        res = minimize(lambda x: smooth_obj(x, n, T), h, method='SLSQP',
                       jac=lambda x: smooth_grad(x, n, T),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_refine(h_init, n, t_start, deadline, max_iters=300000):
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    a0 = 0.35
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = float(it) / (it + 1.5)
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        s = abs(idx - (n - 1))
        g = grad_f_shift(h_y, n, s) * dx
        alpha = a0 / (1.0 + 0.004 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 1500 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    return h_best, c5_best

def multi_restart_slsqp(h_best, n, t_start, deadline, rng):
    c5_best = compute_c5(h_best)
    attempts = 0
    while time.time() < deadline and attempts < 50:
        attempts += 1
        sigma = 0.004 + 0.006 * (attempts % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 1.0, 1200, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    return h_best, c5_best

def try_resolution(h_best, n, n_try, t_start, deadline):
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
    h_hi = np.interp(x_new, x_old, h_best)
    h_hi = project_feasible(h_hi, n_try)
    c5_hi = compute_c5(h_hi)
    for T in [4.0, 1.0]:
        if time.time() > deadline:
            break
        h_new = do_slsqp(h_hi, n_try, T, 2000, 1e-15)
        h_new = project_feasible(h_new, n_try)
        c5 = compute_c5(h_new)
        if c5 < c5_hi:
            c5_hi = c5
            h_hi = h_new
    return h_hi, c5_hi

    # Phase 2: Nesterov subgradient

    # Phase 4: Multi-restart local search

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
