    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: Soft-max projected gradient continuation
    for T in [100, 50, 25, 10, 5, 3, 2, 1.5, 1.0]:
        limit = budget_s - 250
        if time.time() - t0 > limit:
            break
        h, ccur = proj_grad(h, target, T, 400, 5.0 / (n * T), t0, limit)
        if ccur < c5_best:
            c5_best = ccur
            h_best = h.copy()

    # Phase 2: Additional gradient at very low T
    for T in [0.8, 0.5, 0.3]:
        limit = budget_s - 200
        if time.time() - t0 > limit:
            break
        h, ccur = proj_grad(h_best, target, T, 200, 2.0 / (n * T), t0, limit)
        if ccur < c5_best:
            c5_best = ccur
            h_best = h.copy()

    # Phase 3: Directed local search (main fine-tuning)
    limit_ls = budget_s - 150
    n_ls = 60000
    h_best, c5_ls = directed_local(h_best, target, rng, t0, limit_ls, n_ls)
    if c5_ls < c5_best:
        c5_best = c5_ls

    # Phase 4: Fine random local search
    limit_fl = budget_s - 40
    n_fl = 80000
    h_best, c5_fl = fine_local(h_best, target, rng, t0, limit_fl, n_fl)
    if c5_fl < c5_best:
        c5_best = c5_fl

    # Phase 5: One more gradient pass from current best
    if time.time() - t0 < budget_s - 30:
        h_try, c_try = proj_grad(h_best, target, 0.5, 150, 1.0 / n, t0, budget_s - 20)
        if c_try < c5_best:
            c5_best = c_try
            h_best = h_try.copy()

    # Phase 6: Try alternate resolutions
    for n_alt in [200, 150, 100, 350]:
        if time.time() - t0 > budget_s - 15:
            break
        h_best, c5_best = try_alternate_n(h_best, c5_best, n_alt, t0, budget_s - 10)

    # Final verification
    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)
    return (h_best, c5_best, len(h_best))
