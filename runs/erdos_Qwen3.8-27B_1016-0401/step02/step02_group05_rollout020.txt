def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] -= w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n-p]
    return grad * dx

def project_sum_box(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(-(1.0 - h))
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(-h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def projected_gradient_step(h, target, T, step_size, t0, budget, n_steps):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    h_prev = h.copy()
    
    for it in range(n_steps):
        if time.time() - t0 > budget:
            break
        
        # Nesterov momentum
        momentum = 0.5
        h_nest = h_curr + momentum * (h_curr - h_prev)
        h_nest = project_sum_box(h_nest, target)
        
        grad = soft_max_jac(h_nest, T)
        h_new = h_nest - step_size * grad
        h_new = project_sum_box(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h_prev = h_curr.copy()
            h_curr = h_new
        else:
            # Halve step and retry
            h_new = h_nest - step_size * 0.5 * grad
            h_new = project_sum_box(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_new.copy()
                h_prev = h_curr.copy()
                h_curr = h_new
                step_size *= 0.7
            else:
                step_size *= 0.5
                h_curr = h_new  # stay put
                h_prev = h_curr.copy()
    
    return best, best_c5

    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    
    for outer in range(max_iter):
        corr = np.correlate(h_curr, 1 - h_curr, mode="full")
        max_val = np.max(corr)
        
        # Find top lags
        top_idx = np.argsort(-corr)[:max_active]
        active_lags = top_idx - (n-1)
        
        # Build QP: minimize sum of active correlations
        # f(h) = sum_j sum_i h[i]*(1-h[i+k_j]) * dx
        # = dx * (sum_j (n-|k_j|) - sum_j autocorr_h(k_j))
        # Minimize f = -sum_j autocorr_h(k_j) + const
        # = -h^T Q h + const where Q = sum_j (E_j + E_j^T)/2
        
        # Gradient: -2*Q*h (for the h^T Q h part)
        # = -2 * sum_j (shifted h)
        
        def qp_obj(h_vec):
            total = 0.0
            for k in active_lags:
                if k == 0:
                    total += np.sum(h_vec * (1 - h_vec))
                elif k > 0:
                    total += np.sum(h_vec[k:] * (1 - h_vec[:n-k]))
                else:
                    p = -k
                    total += np.sum(h_vec[:n-p] * (1 - h_vec[p:]))
            return total * dx

            g = np.zeros(n)
            for k in active_lags:
                if k == 0:
                    g[:] += (1 - 2*h_vec)
                elif k > 0:
                    g[k:] += (1 - h_vec[:n-k])
                    g[:n-k] -= h_vec[k:]
                else:
                    p = -k
                    g[:n-p] += (1 - h_vec[p:])
                    g[p:] -= h_vec[:n-p]
            return g * dx
        
        bounds = [(0,1)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        
        res = minimize(qp_obj, h_curr, method='SLSQP', jac=qp_grad,
                      bounds=bounds, constraints=cons,

        if res.fun < 50:

            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_curr) - 1e-14:

        # Check convergence
        corr_new = np.correlate(h_curr, 1-h_curr, mode="full")
        if np.max(corr_new) - np.median(corr_new) < 1e-8:

            break
        if stall > 200:
            delta *= 0.6
            stall = 0
            if delta < 0.001:
                break
        k_star = np.argmax(corr)
        s = k_star - (n-1)
        p = abs(s)
        h_try = h_curr.copy()
        if p == 0:
            vals = h_curr * (1.0 - h_curr)
            idx = np.argmax(vals)
            h_try[idx] += -delta if h_curr[idx] > 0.5 else delta
        else:
            if s > 0:
                contrib = h_curr[p:] * (1.0 - h_curr[:n-p])
            else:
                contrib = h_curr[:n-p] * (1.0 - h_curr[p:])
            idx = np.argmax(contrib)
            if s > 0:
                h_try[idx+p] -= delta
                h_try[idx] += delta
            else:
                h_try[idx] -= delta
                h_try[idx+p] += delta
        h_try = fix_constraint(np.clip(h_try, 0, 1), target)
        corr_try = np.correlate(h_try, 1 - h_try, mode="full")
        c5_try = float(np.max(corr_try) * dx)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            h_curr = h_try
            corr = corr_try
            stall = 0
        else:
            stall += 1
    return h_curr, best_c5
