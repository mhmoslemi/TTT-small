def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

def smooth_max(corr, T):

    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()

    else:
        lo = int(rng.integers(0, n // 3))
        hi = int(rng.integers(2 * n // 3, n))
        h_p[lo:hi] = 1.0 - h_p[lo:hi]
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(280)
        h[::2] = 1.0
    n = len(h)
    tgt = n / 2.0
    h = project_sum(h, tgt)
    c5_best = compute_c5(h)

    # Phase 1: Multi-temperature PG at native resolution
    t1_end = t0 + budget * 0.22
    for T0, T1, ni, lr in [(500, 5, 5000, 0.12), (200, 2, 5000, 0.07), (60, 1, 6000, 0.04)]:
        if time.time() > t1_end:
            break
        h, c5 = pg_loop(h, tgt, T0, T1, ni, lr, t1_end)
        if c5 < c5_best:
            c5_best = c5

    # Phase 2: Upsample to 840 for more DOF
    n2 = 840
    tgt2 = n2 / 2.0
    x_old = np.arange(n) * (2.0 / n)
    x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x2, x_old, h)
    h2 = project_sum(h2, tgt2)
    t2_end = t0 + budget * 0.52
    for T0, T1, ni, lr, tk in [(400, 5, 6000, 0.09, False), (120, 2, 6000, 0.05, True), (50, 1, 6000, 0.03, False)]:
        if time.time() > t2_end:
            break
        h2, c5_2 = pg_loop(h2, tgt2, T0, T1, ni, lr, t2_end, use_topk=tk)
        if c5_2 < c5_best:
            c5_best = c5_2
            h = h2.copy()
            n = n2
            tgt = tgt2

    # Phase 3: Diverse local search
    t3_end = t0 + budget * 0.82
    n_cur = len(h)
    tgt_cur = n_cur / 2.0
    trial_end = 0.0
    for r in range(80):
        if time.time() > t3_end:
            break
        t_it = r % 8
        h_p = perturb(h, rng, n_cur, t_it)
        h_p = project_sum(h_p, tgt_cur)
        c5_p_init = compute_c5(h_p)
        if c5_p_init > c5_best * 1.05:
            continue
        t_it_end = min(t3_end, time.time() + 30)
        h_p, c5_p = pg_loop(h_p, tgt_cur, 80, 2, 1400, 0.05, t_it_end)
        if c5_p < c5_best:
            c5_best = c5_p
            h = h_p.copy()

    # Phase 4: Top-k focused refinement on best
    if time.time() < t0 + budget * 0.92:
        nf = len(h)
        h, c5_f = pg_loop(h, nf / 2.0, 100, 1, 5000, 0.04, t0 + budget * 0.92, use_topk=True)
        if c5_f < c5_best:
            c5_best = c5_f
            h = h

    # Phase 5: Final micro-polish
    if time.time() < t0 + budget - 5:
        nf = len(h)
        h, c5_f = pg_loop(h, nf / 2.0, 200, 1, 4000, 0.03, t0 + budget - 2)
        if c5_f < c5_best:
            c5_best = c5_f
            h = h

    c5_final = compute_c5(h)
    c5_best = min(c5_best, c5_final)
    # Return at <=1000 points
    if len(h) > 1000:
        n_out = 840
        x_f = np.arange(len(h)) * (2.0 / len(h))
        x_out = np.linspace(0, 2, n_out, endpoint=False)
        h_out = np.interp(x_out, x_f, h)
        h_out = project_sum(h_out, n_out / 2.0)
        c5_out = compute_c5(h_out)
        if c5_out <= c5_best + 1e-5:
            return (h_out, c5_out, n_out)
    return (h, c5_best, len(h))
