    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():

        if m2.any():

        if m2.any():
            g[m2] -= h[j[m2] - s]
    return g

def lbfgsb_smooth(h, n, T, penalty, maxiter):
    """L-BFGS-B with smooth objective and penalty for sum constraint."""
    dx = 2.0 / n
    center = n - 1
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx + penalty * (x.sum() - target)**2

        sh = T * (corr - mc)

    return x.sum() - _WS['target']

def _cons_jac(x):
    return _WS['ones']

def do_slsqp(h, n, T, maxiter):
    _WS['target'] = n / 2.0
    _WS['ones'] = np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(
            lambda x: soft

_obj(x, T, n),
            h, method='SLSQP',
            jac=lambda x: soft_obj_jac(x, T, n),
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h.copy()

def active_set_gradient_descent(h_init, n, t0, time_limit, k=5, max_iters=200000):
    dx = 2.0 / n
    center = n - 1
    h_curr = project_feasible(h_init.copy(), n)
    c5_best = compute_c5(h_curr)
    h_best = h_curr.copy()
    
    step = 0.02
    vel = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        top = np.argsort(corr)[-k:]
        
        g = np.zeros(n)
        for oi in top:
            g += shift_grad(h_curr, n, oi - center)
        g *= dx / k
        
        # Nesterov
        beta = it / (it + 1.5)
        y = np.clip(h_curr + beta * vel, 0.0, 1.0)
        
        # Gradient at y
        corr_y = np.correlate(y, 1.0 - y, mode="full")
        top_y = np.argsort(corr_y)[-k:]
        g_y = np.zeros(n)
        for oi in top_y:
            g_y += shift_grad(y, n, oi - center)
        g_y *= dx / k
        
        h_new = project_feasible(y - step * g_y, n)
        vel = h_new - h_curr
        h_curr = h_new
        
        c5_new = compute_c5(h_curr)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_curr.copy()
            no_improve = 0
            step = min(step * 1.05, 0.1)
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                step *= 0.85
            if no_improve > 2000:
                step *= 0.5
                vel[:] = 0
                no_improve = 0
    
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: SLSQP coarse-to-fine
    for T in [64, 32, 16, 8, 4, 2, 1]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = do_slsqp(h_best, n, T, 2000)

        c5 = compute_c5(h_new)

    # Phase 2: Active-set Nesterov gradient descent

    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    
    # Phase 3: SLSQP final polish
    for T in [1, 2]:
        if time.time() - t0 > budget_s - 60:
            break
        h_new = do_slsqp(h_best, n, T, 5000)
        h_new = project_feasible(h_new, n)
