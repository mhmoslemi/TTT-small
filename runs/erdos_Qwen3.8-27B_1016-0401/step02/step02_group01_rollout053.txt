def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_obj_grad(h, idx, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr[idx]
    mc = float(vals.max())
    sh = T * (vals - mc)
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for ii, j in enumerate(idx):
        w = sw[ii]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad * dx

    nr = len(corr)
    if k >= nr:
        return np.arange(nr)
    return np.sort(np.argpartition(corr, -k)[-k:])

            obj, g = topk_obj_grad(h, idx, T)
            gn = np.linalg.norm(g)
            if gn < 1e-14:

            ok = False
            lt = lr
            for _ in range(25):
                hn = project_constraint(h - lt * g, target)
                o2, _ = topk_obj_grad(hn, idx, T)
                if o2 < obj - 1e-16:
                    ok = True
                    break
                lt *= 0.5
            if not ok:

            h = hn
            lr = min(lt * 1.05, lr0)
        idx = get_topk(h, k_use)
        c5 = compute_c5(h)
        if c5 < best_c5:

    c5 = compute_c5(h)

    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_obj_grad(h, idx, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr[idx]
    mc = float(vals.max())
    sh = T * (vals - mc)
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for ii, j in enumerate(idx):
        w = sw[ii]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad * dx

def get_topk(h, k):
    corr = np.correlate(h, 1 - h, mode="full")
    nr = len(corr)
    if k >= nr:
        return np.arange(nr)
    return np.sort(np.argpartition(corr, -k)[-k:])

def pg_opt(h0, target, t0, deadline, lr0=0.04):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    nr = 2 * n - 1
    sched = [
        (300, 60, 1200), (150, 40, 1800), (80, 30, 2500),
        (40, 20, 3000), (25, 15, 3000), (15, 12, 4000),
        (10, 10, 5000), (7, 8, 5000), (5, 7, 6000),
        (3, 6, 7000), (2, 5, 8000), (1.5, 5, 10000),
        (1.0, 4, 12000), (0.7, 3, 12000), (0.5, 3, 15000),
        (0.3, 2, 15000),
    ]
    for T, k, ni in sched:
        if time.time() > deadline - 3:
            break
        k_use = min(k, nr)
        idx = get_topk(h, k_use)
        lr = lr0
        for _ in range(ni):
            if time.time() > deadline - 3:
                break
            obj, g = topk_obj_grad(h, idx, T)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            ok = False
            lt = lr
            for _ in range(25):
                hn = project_constraint(h - lt * g, target)
                o2, _ = topk_obj_grad(hn, idx, T)
                if o2 < obj - 1e-16:
                    ok = True
                    break
                lt *= 0.5
            if not ok:
                break
            h = hn
            lr = min(lt * 1.05, lr0)
        idx = get_topk(h, k_use)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def subgrad_refine(h0, target, t0, deadline):
    h = h0.copy()
    n = len(h)
    c = n - 1
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.002
    for _ in range(10000):
        if time.time() > deadline - 2:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        # Use top 2 with equal weight
        top2 = np.argpartition(corr, -2)[-2:]
        top2.sort()
        g = np.zeros(n)
        for j in top2:
            k = j - c
            if k == 0:
                g += (1.0 - 2.0 * h)
            elif k > 0:
                g[k:] += (1.0 - h[:n - k])
                g[:n - k] -= h[k:]
            else:
                p = -k
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g *= dx / 2.0
        hn = project_constraint(h - lr * g, target)
        c5n = compute_c5(hn)
        if c5n < best_c5 - 1e-12:
            best_c5, best_h = c5n, hn.copy()
            h = hn
            lr = min(lr * 1.005, 0.004)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return best_h, best_c5

def perturb(h, n, rng):
    r = rng.random()
    h = h.copy()
    if r < 0.25:
        i = int(rng.integers(0, max(1, n - 15)))
        s = int(rng.integers(3, min(15, n - 1)))
        h[i:i+s] = 1.0 - h[i:i+s]
    elif r < 0.5:
        h += rng.normal(0, 0.04, n)
    elif r < 0.75:
        h = (h > rng.uniform(0.2, 0.8)).astype(float)
    else:
        h[:] = np.roll(h, int(rng.integers(1, min(12, n - 1))))
    return np.clip(h, 0, 1)
