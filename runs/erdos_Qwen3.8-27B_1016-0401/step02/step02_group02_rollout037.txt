    for _ in range(60):

        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

    elif n_ones < target:

def continuous_polish(h, n, target, t_end):

            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.85
            if lr < 0.002:
                break
    return best_h, best_c5
