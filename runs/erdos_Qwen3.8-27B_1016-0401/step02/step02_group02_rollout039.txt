```python
    # o ranges from -(n-1) to n-1
    # d_o = (1-h[p-o]) * 1{0≤p-o<n} - (1-h[q-o]) * 1{0≤q-o<n} - h[p+o]*1{0≤p+o<n} + h[q+o]*1{0≤q+o<n}
    
    o_vals = np.arange(-(n-1), n)  # 2n-1 values
    # For each o, compute d_o
    # p-o valid: 0 ≤ p-o < n → o is in [p-n+1, p]
    # q-o valid: 0 ≤ q-o < n → o is in [q-n+1, q]
    # p+o valid: 0 ≤ p+o < n → o is in [-p, n-1-p]
    # q+o valid: 0 ≤ q+o < n → o is in [-q, n-1-q]
    
    d = np.zeros(2*n-1)

        if 0 <= q - o < n: val -= (1 - h[q-o])

        elif d_min > 0:
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * (2.0/n))

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo+hi)/2
        if np.clip(x-mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(x-(lo+hi)/2, 0, 1)

def _lse_val(corr, T):
    m = corr.max()
    s = T*(corr-m); s -= s.max()
    return m + np.log(np.exp(s).sum())/T

def _lse_grad_vec(corr, T):
    m = corr.max()
    s = T*(corr-m); s -= s.max()
    e = np.exp(s); return e/e.sum()

class _OptHelper:
    def __init__(self, T, n, target):
        self.T, self.n, self.target = T, n, target
    def f(self, x):
        c = np.correlate(x, 1.0-x, mode="full")
        return _lse_val(c, self.T) * (2.0/self.n)
    def j(self, x):
        c = np.correlate(x, 1.0-x, mode="full")
        w = _lse_grad_vec(c, self.T)
        ctr = self.n-1; n = self.n
        g = np.zeros(n)
        for idx in np.where(w>1e-10)[0]:
            o = int(idx)-ctr; wi = w[idx]
            if o==0: g += wi*(1.0-2.0*x)
            elif o>0:

        return g*(2.0/n)
    def cf(self, x): return x.sum()-self.target
    def cj(self, x): return np.ones(self.n)

            options={'maxiter':maxiter, 'ftol':1e-14})
        out = project_feasible(res.x, target)
        return out
    except:

def compute_d(h, p, q):
    """Compute the d vector for pair swap (p,q)."""
    n = len(h)
    o_vals = np.arange(-(n-1), n)
    d = np.zeros(2*n-1)
    # (1-h[p-o]) for 0<=p-o<n
    lo, hi = max(p-n+1, -(n-1)), min(p, n-1)
    if lo <= hi:
        o = np.arange(lo, hi+1)
        d[o+(n-1)] += (1.0-h[p-o])
    # -(1-h[q-o]) for 0<=q-o<n
    lo, hi = max(q-n+1, -(n-1)), min(q, n-1)
    if lo <= hi:
        o = np.arange(lo, hi+1)
        d[o+(n-1)] -= (1.0-h[q-o])
    # -h[p+o] for 0<=p+o<n
    lo, hi = max(-p, -(n-1)), min(n-1-p, n-1)
    if lo <= hi:
        o = np.arange(lo, hi+1)
        d[o+(n-1)] -= h[p+o]
    # h[q+o] for 0<=q+o<n
    lo, hi = max(-q, -(n-1)), min(n-1-q, n-1)
    if lo <= hi:
        o = np.arange(lo, hi+1)
        d[o+(n-1)] += h[q+o]
    return d

    """Find optimal t for h[p]+=t, h[q]-=t. Returns (t, d, new_c5)."""
    n = len(h)
    d = compute_d(h, p, q)
    t_lo = max(-h[p], h[q]-1) + 1e-12
    t_hi = min(1-h[p], h[q]) - 1e-12
    if t_lo >= t_hi:
        return 0.0, d, compute_c5(h)

    for _ in range(45):

        else: t_lo = t_mid

    t_end = time.time()+budget
```

```python
def compute_d(h, p, q):
    n = len(h)
    d = np.zeros(2*n-1)
    # Term 1: (1-h[p+o]) for -p <= o <= n-1-p
    lo1, hi1 = max(-p, -(n-1)), min(n-1-p, n-1)
    if lo1 <= hi1:
        o = np.arange(lo1, hi1+1)
        d[o+(n-1)] += (1.0 - h[p+o])
    # Term 2: -(1-h[q+o]) for -q <= o <= n-1-q
    lo2, hi2 = max(-q, -(n-1)), min(n-1-q, n-1)
    if lo2 <= hi2:
        o = np.arange(lo2, hi2+1)
        d[o+(n-1)] -= (1.0 - h[q+o])
    # Term 3: -h[p-o] for 0 <= p-o < n, i.e., p-n+1 <= o <= p
    lo3, hi3 = max(p-n+1, -(n-1)), min(p, n-1)
    if lo3 <= hi3:
        o = np.arange(lo3, hi3+1)
        d[o+(n-1)] -= h[p-o]
    # Term 4: h[q-o] for 0 <= q-o < n, i.e., q-n+1 <= o <= q
    lo4, hi4 = max(q-n+1, -(n-1)), min(q, n-1)
    if lo4 <= hi4:
        o = np.arange(lo4, hi4+1)
        d[o+(n-1)] += h[q-o]
    return d
```
