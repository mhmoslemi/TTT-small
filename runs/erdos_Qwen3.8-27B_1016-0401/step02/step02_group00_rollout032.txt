Actually, I realize I should think more carefully about the LP linearization. The issue is:

corr_k(h) = sum_i h[i+o]*(1-h[i]) (for appropriate range)

This is a concave function of h (it's a sum of products h[a]*(1-h[b]), each of which is bilinear, and the Hessian is indefinite). So the maximum of concave functions is not convex, and minimizing it is nonconvex.

The linearization: corr_k(h) ≈ corr_k(h0) + grad_k(h0)·(h-h0)

Since corr_k is quadratic (sum of bilinear terms), the linearization is the first-order Taylor approximation. The error is:
corr_k(h) - [corr_k(h0) + grad_k(h0)·(h-h0)] = (1/2)(h-h0)^T H_k (h-h0)

where H_k is the Hessian of corr_k (which is a matrix with -1 entries at (i,j) positions that are coupled).

This error is bounded by (1/2)||H_k|| ||h-h0||_2^2. Since ||H_k|| ≤ 1 (the Hessian entries are at most 1 in magnitude), the error is at most (1/2)||h-h0||_2^2.

So if I solve the LP to get a new h, and ||h-h0|| is small, the linearization error is small. This suggests the method should converge quadratically in the neighborhood of the optimum.

def solve_linearized_lp(h0, active_lags, n, target):
    """Solve LP with linearized correlation constraints"""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for lag in active_lags:
        c0 = np.correlate(h0, 1.0 - h0, mode="full")[lag]
        g = corr_grad(h0, lag, n)
        # Linearized: c0 + g·(h - h0) <= t/dx
        # i.e., g·h <= t/dx - c0 + g·h0
        constraints.append(g @ h_var <= t_var / dx - c0 + g @ h0)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            prob.solve(solver=cp.SCS)
    
    if h_var.value is None:
        return None, None
    return np.array(h_var.value).flatten(), t_var.value

def coord_descent(h, n, target, passes=3):
    """Coordinate descent: for each i, find best h[i] in [0,1]"""
    center = n - 1
    for p in range(passes):
        for i in range(n):
            # Compute current correlations
            # For each lag, corr as function of x = h[i]:
            # corr_k = A_k + B_k * x (at most, since h[i] appears linearly in most terms)
            # Actually h[i] can appear in both h and (1-h) slots, giving quadratic
            # For simplicity, do a 1D search
            
            best_x = h[i]
            best_c = None
            corr_base = np.correlate(h, 1.0 - h, mode="full")
            base_max = np.max(corr_base)
            
            # Quick: try 0, 1, current
            candidates = [0.0, 1.0]
            if h[i] not in (0.0, 1.0):
                candidates.append(h[i])
            # Add some interpolated values
            for x in candidates:
                if x == h[i]:
                    c = base_max
                else:
                    h[i] = x
                    c = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if best_c is None or c < best_c:
                    best_c = c
                    best_x = x
            h[i] = best_x
    
    # Fix constraint after coordinate descent
    return fix_constraint(h, target)

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

_TARGET = 0.0
_N_DIM = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def corr_grad(h, lag, n):
    center = n - 1
    o = lag - center
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n - o]
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - h[p:]
        g[p:] += -h[:n - p]
    return g

def solve_lp_active(h0, active_lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    corr_full = np.correlate(h0, 1.0 - h0, mode="full")
    for lag in active_lags:
        c0 = corr_full[lag]
        g = corr_grad(h0, lag, n)
        cons.append(g @ h_var <= t_var - c0 + g @ h0)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None, None
    if h_var.value is None:
        return None, None
    return np.array(h_var.value).flatten(), float(t_var.value)

    return x.sum() - _TARGET

    return np.ones(_N_DIM)

def _sm_obj(x):

    dx = 2.0 / n
    T = _SM_T
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def _sm_grad(x):
    n = _N_DIM
    dx = 2.0 / n
    T = _SM_T
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n - o])
            g[:n - o] += -w * x[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - x[p:])
            g[p:] += -w * x[:n - p]
    return g * dx
