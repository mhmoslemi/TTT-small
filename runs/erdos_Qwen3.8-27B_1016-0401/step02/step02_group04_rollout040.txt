        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def structured_perturb(h, rng, n):
    h_p = h.copy()
    n_freq = rng.integers(1, 7)
    for _ in range(n_freq):
        f = rng.integers(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.06)
        x = np.arange(n) * (2.0 / n)
        h_p += amp * np.sin(f * np.pi * x + phase)
    return project_box_sum(h_p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Adam broad optimization (top-k=40)
    if elapsed() < budget_s - 300:
        h_new, c5 = adam_optimize(h_best, target, 40, 0.02, 4000, 50.0, 6000)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Adam medium focus (top-k=15)
    if elapsed() < budget_s - 250:
        h_new, c5 = adam_optimize(h_best, target, 15, 0.01, 5000, 80.0, 8000)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Adam fine focus (top-k=5)
    if elapsed() < budget_s - 200:
        h_new, c5 = adam_optimize(h_best, target, 5, 0.005, 8000, 150.0, 10000)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: L-BFGS-B polish
    if elapsed() < budget_s - 150:
        for T in [80, 200, 500]:
            if elapsed() > budget_s - 100:
                break
            h_new = lbfgs_optimize(h_best, n, target, T, 3000)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 5: Multi-restart with Adam
    n_restarts = 0
    while elapsed() < budget_s - 80:
        n_restarts += 1
        if n_restarts < 8:
            h_pert = h_best.copy()
            h_pert = project_box_sum(h_pert + rng.normal(0, 0.03, n), target)
        else:
            h_pert = structured_perturb(h_best, rng, n)
        
        top_k = max(3, 20 - n_restarts * 2)
        lr = 0.008 / (1 + 0.1 * n_restarts)
        h_new, c5 = adam_optimize(h_pert, target, top_k, lr, 3000, 100.0, 4000)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 6: L-BFGS-B final polish at various T
    if elapsed() < budget_s - 40:
        for T in [30, 100, 300, 800]:
            if elapsed() > budget_s - 20:
                break
            h_new = lbfgs_optimize(h_best, n, target, T, 5000)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Ultra-fine Adam on top-2 lags
    if elapsed() < budget_s - 10:
        h_new, c5 = adam_optimize(h_best, target, 2, 0.002, 10000, 300.0, 2000)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final verification
    h_best = project_box_sum(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    if compute_c5(h_best) < c5_final:
        c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
