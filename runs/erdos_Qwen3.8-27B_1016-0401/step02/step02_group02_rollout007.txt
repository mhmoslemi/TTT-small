import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def subgrad_topk(corr, h, K, n):
    """Compute subgradient from top K correlation values."""
    center = n - 1
    # Get top K indices
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n-o])
            g[:n-o] -= h[o:]
        else:
            p = -o
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
    return g / K

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    def elapsed():
        return time.time() - t0
    
    # Get initial values

        h_init = np.array(initial_h_values, dtype=float).copy()
    else:

    n = len(h_init)
    target = n / 2.0

    c5_best = compute_c5(h)
    h_best = h.copy()
    
    # Phase 1: Projected subgradient on true max with top-K
    # Use multiple K values and decreasing step sizes
    for K in [1, 2, 3, 4, 5, 3, 2, 1]:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_topk(corr, h, K, n):
    center = n - 1
    top_idx = np.argpartition(corr, -K)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g / K

def coord_descent(h, n, target, n_sweeps):
    h = h.copy()
    for _ in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        for i in range(n):
            old = h[i]
            effect = np.zeros(2 * n - 1)
            # h[i] as second factor: shift m, i-m in [0,n-1], i.e. m in [i-(n-1), i]
            lo_m = max(i - (n - 1), -(n - 1))
            hi_m = min(i, n - 1)
            for m in range(lo_m, hi_m + 1):
                effect[m + n - 1] += (1.0 - h[i - m])
            # h[i] as first factor: shift m, i+m in [0,n-1], i.e. m in [-i, n-1-i]
            lo_m2 = max(-i, -(n - 1))
            hi_m2 = min(n - 1 - i, n - 1)
            for m in range(lo_m2, hi_m2 + 1):
                effect[m + n - 1] -= h[i + m]
            # minimize max(corr + dv*effect) over dv in [-old, 1-old]
            best_dv = 0.0
            best_max = np.max(corr)
            for dv in np.linspace(-old, 1.0 - old, 15):
                tm = np.max(corr + dv * effect)
                if tm < best_max:
                    best_max = tm
                    best_dv = dv
            if best_dv != 0.0:
                h[i] = old + best_dv
                corr += best_dv * effect
        h = project_feasible(h, target)
    return h

def subgrad_optimize(h, n, target, n_steps, K, lr0, lr_dec, momentum=0.85):
    c5_local = compute_c5(h)
    h_local = h.copy()
    v = np.zeros(n)
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_cur = float(np.max(corr) * (2.0 / n))
        g = subgrad_topk(corr, h, K, n)
        lr = lr0 / (1.0 + it * lr_dec)
        v = momentum * v + lr * g
        h_new = project_feasible(h - v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            if c5_new < c5_local:
                c5_local = c5_new
                h_local = h.copy()
        else:
            v *= 0.4
            h_new = project_feasible(h - v, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h = h_new
                if c5_new < c5_local:
                    c5_local = c5_new
                    h_local = h.copy()
    return h_local, c5_local

def slsqp_polish(h, n, target, T, maxiter):
    from scipy.optimize import minimize
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= np.max(sh)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= np.max(sh)
        e = np.exp(sh)
        w = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for idx in nz:
            o = idx - center
            wi = w[idx]
            if o == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif o > 0:
                grad[o:] += wi * (1.0 - x[:n - o])
                grad[:n - o] -= wi * x[o:]
            else:
                p = -o
                grad[:n - p] += wi * (1.0 - x[p:])
                grad[p:] -= wi * x[:n - p]
        return grad * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.ones(280) * 0.5
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    def elapsed():
        return time.time() - t0

    # Phase 1: Coordinate descent (exact per-coordinate)
    for ns in [20, 15, 10, 8, 5, 5]:
        if elapsed() > budget_s - 400:
            break
        h_cd = coord_descent(h_best, n, target, ns)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd.copy()

    # Phase 2: Subgradient with various K
    configs = [
        (4000, 1, 0.04, 0.0004, 0.8),
        (4000, 2, 0.03, 0.0004, 0.85),
        (4000, 3, 0.02, 0.0003, 0.85),
        (5000, 1, 0.02, 0.0002, 0.9),
        (5000, 2, 0.01, 0.0002, 0.9),
        (6000, 1, 0.008, 0.0001, 0.92),
    ]
    for (st, K, lr0, lrdec, mu) in configs:
        if elapsed() > budget_s - 300:
            break
        h_sg, c5_sg = subgrad_optimize(h_best.copy(), n, target, st, K, lr0, lrdec, mu)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg.copy()
        # Follow up with coord descent
        if elapsed() < budget_s - 250:
            h_cd = coord_descent(h_best, n, target, 5)
            c5_cd = compute_c5(h_cd)
            if c5_cd < c5_best:
                c5_best = c5_cd
                h_best = h_cd.copy()

    # Phase 3: SLSQP polish at low T
    if elapsed() < budget_s - 150:
        for T_val in [5.0, 3.0, 2.0, 1.5]:
            if elapsed() > budget_s - 100:
                break
            h_sl = slsqp_polish(h_best.copy(), n, target, T_val, 2000)
            c5_sl = compute_c5(h_sl)
            if c5_sl < c5_best:
                c5_best = c5_sl
                h_best = h_sl.copy()
            # Coord descent after SLSQP
            if elapsed() < budget_s - 80:
                h_cd = coord_descent(h_best, n, target, 10)
                c5_cd = compute_c5(h_cd)
                if c5_cd < c5_best:
                    c5_best = c5_cd
                    h_best = h_cd.copy()

    # Phase 4: Very fine subgradient + coord descent interleaved
    if elapsed() < budget_s - 80:
        for _ in range(3):
            if elapsed() > budget_s - 40:
                break
            h_sg, c5_sg = subgrad_optimize(h_best.copy(), n, target, 3000, 1, 0.005, 0.0002, 0.95)
            if c5_sg < c5_best:
                c5_best = c5_sg
                h_best = h_sg.copy()
            h_cd = coord_descent(h_best, n, target, 15)
            c5_cd = compute_c5(h_cd)
            if c5_cd < c5_best:
                c5_best = c5_cd
                h_best = h_cd.copy()

    # Phase 5: Try alternative starting points
    if elapsed() < budget_s - 60:
        x = np.linspace(0, 2.0, n, endpoint=False)
        alt_starts = [
            np.maximum(0.0, 1.0 - np.abs(x - 1.0)),
            np.maximum(0.0, 1.3 - 0.65 * np.abs(x - 1.0)),
            0.5 + 0.5 * np.cos(np.pi * x),
        ]
        for s in alt_starts:
            if elapsed() > budget_s - 20:
                break
            h_try = project_feasible(s, target)
            h_try = coord_descent(h_try, n, target, 15)
            h_try, c5_try = subgrad_optimize(h_try, n, target, 3000, 2, 0.02, 0.001, 0.85)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try.copy()
                h_best = coord_descent(h_best, n, target, 10)
                c5_try2 = compute_c5(h_best)
                if c5_try2 < c5_best:
                    c5_best = c5_try2

    # Phase 6: Final random perturbations
    if elapsed() < budget_s - 15:
        for _ in range(15):
            if elapsed() > budget_s - 5:
                break
            scale = 0.03
            h_p = project_feasible(h_best + rng.normal(0, scale, n), target)
            h_p = coord_descent(h_p, n, target, 8)
            c5_p = compute_c5(h_p)
            if c5_p < c5_best:
                c5_best = c5_p
                h_best = h_p.copy()

    # Final projection and evaluation
    h_best = project_feasible(h_best, target)
    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)
    return (h_best, c5_best, len(h_best))
