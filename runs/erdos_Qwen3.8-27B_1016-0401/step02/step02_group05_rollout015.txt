        if tr > 1e-15:
            h += room * (err / tr)
    else:
        tr = h.sum()
        if tr > 1e-15:
            h -= h * (-err / tr)
    h = np.clip(h, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-13:
        order = np.argsort(1.0 - h) if err > 0 else np.argsort(h)
        for i in order:
            if abs(err) < 1e-13:
                break
            if err > 0:
                s = min(err, 1.0 - h[i]); h[i] += s; err -= s
            else:
                s = min(-err, h[i]); h[i] -= s; err += s
    return h

def proj_grad_step(h, target, T, lr_init, n_iters, t0, budget):
    n = len(h)
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    lr = lr_init
    for it in range(n_iters):
        if time.time() - t0 > budget:
            break
        grad = soft_max_grad(h_curr, T)
        h_try = h_curr - lr * grad
        h_try = fix_sum(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            h_curr = h_try
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h_curr, best_c5

def coord_descent(h, target, grid, t0, budget, sa_temp=0.0, rng=None):
    n = len(h)
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    sa = sa_temp
    for _ in range(100000):
        if time.time() - t0 > budget:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget:
                break
            save = h_curr[j]
            best_c = save
            best_val = compute_c5(h_curr)
            for c in grid:
                if abs(c - save) < 1e-8:
                    continue
                h_curr[j] = c
                h_curr = fix_sum(h_curr, target)
                c5 = compute_c5(h_curr)
                accept = False
                if c5 < best_val - 1e-15:
                    accept = True
                elif sa > 0 and rng is not None:
                    if rng.random() < np.exp(-(c5 - best_val) / (sa * 0.001)):
                        accept = True
                if accept:
                    if c5 < best_val:
                        best_val = c5
                        best_c = c
                    else:
                        best_c = c
                    improved = True
            h_curr[j] = best_c
            h_curr = fix_sum(h_curr, target)
            sa *= 0.999
        final_c5 = compute_c5(h_curr)
        if final_c5 < best_c5:
            best_c5 = final_c5
        if not improved and sa < 1e-4:
            break
    return h_curr, best_c5

def block_perturb(h, target, rng, block_sizes):
    n = len(h)
    h_new = h.copy()
    for _ in range(rng.integers(2, 5)):
        bs = int(rng.choice(block_sizes))
        start = int(rng.integers(0, max(1, n - bs)))
        h_new[start:start + bs] = 1.0 - h_new[start:start + bs]
    h_new = fix_sum(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    def _sum_fun(x):
        return x.sum() - target
    def _sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]

    # Phase 1: Projected gradient with soft-max (fast, multiple temperatures)
    for T in [50, 30, 15, 10, 7, 5]:
        if elapsed() > budget_s - 450:
            break
        h_pg, c5_pg = proj_grad_step(h_best, target, T, 0.03, 5000, t0, budget_s - 440)
        if c5_pg < c5_best - 1e-12:
            c5_best = c5_pg
            h_best = h_pg

    # Phase 2: SLSQP at moderate temperature
    for T in [20, 10]:
        if elapsed() > budget_s - 350:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_grad, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2500, 'ftol': 1e-16})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: Coordinate descent with simulated annealing (coarse grid)
    grid_coarse = np.linspace(0.0, 1.0, 11)
    h_cd, c5_cd = coord_descent(h_best, target, grid_coarse, t0, budget_s - 250,
                                sa_temp=0.5, rng=rng)
    if c5_cd < c5_best - 1e-12:
        c5_best = c5_cd
        h_best = h_cd

    # Phase 4: Block perturb + coord descent for global escape
    block_sizes = np.array([3, 5, 7, 10])
    n_attempts = 0
    while elapsed() < budget_s - 150:
        n_attempts += 1
        if n_attempts % 4 == 0:
            h_try = block_perturb(h_best, target, rng, block_sizes)
        elif n_attempts % 4 == 1:
            h_try = h_best + rng.normal(0, 0.06, n)
            h_try = fix_sum(h_try, target)
        elif n_attempts % 4 == 2:
            h_try = h_best + rng.normal(0, 0.03, n)
            h_try = fix_sum(h_try, target)
        else:
            h_try = h_best.copy()
            start = int(rng.integers(0, max(1, n - 20)))
            length = int(rng.integers(5, 20))
            h_try[start:start + length] += 0.5
            h_try = fix_sum(h_try, target)
        h_pg2, c5_pg2 = proj_grad_step(h_try, target, 10, 0.02, 2000, t0, budget_s - 140)
        grid_local = np.linspace(max(0, np.min(h_pg2) - 0.2), min(1, np.max(h_pg2) + 0.2), 15)
        h_cd2, c5_cd2 = coord_descent(h_pg2, target, grid_local, t0, budget_s - 140,
                                       sa_temp=0.2, rng=rng)
        if c5_cd2 < c5_best - 1e-12:
            c5_best = c5_cd2
            h_best = h_cd2

    # Phase 5: Fine coordinate descent (local grid)
    grid_fine = np.linspace(0.0, 1.0, 25)
    h_cd3, c5_cd3 = coord_descent(h_best, target, grid_fine, t0, budget_s - 60,
                                   sa_temp=0.1, rng=rng)
    if c5_cd3 < c5_best - 1e-12:
        c5_best = c5_cd3
        h_best = h_cd3

    # Phase 6: SLSQP polish at low temperature
    for T in [4, 2, 1.5]:
        if elapsed() > budget_s - 10:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_grad, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-17})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Final fine coord descent
    h_final, c5_final_cd = coord_descent(h_best, target,
                                          np.linspace(0, 1, 21), t0, budget_s - 1,
                                          sa_temp=0.0, rng=rng)
    if c5_final_cd < c5_best:
        c5_best = c5_final_cd
        h_best = h_final

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_sum(h_best, target)
    return (h_best, c5_final, n)
