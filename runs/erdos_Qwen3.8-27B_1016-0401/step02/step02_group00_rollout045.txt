def slsqp_wave(h, target, lag_idx, maxiter=1000):

    center = n - 1
    o = lag_idx - center

    def eq_jac(x):
        return np.ones(n)

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_box_sum(np.clip(res.x, 0, 1), target)
        return h_new
    except:

def sm_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    total = e.sum()
    obj = (mc + np.log(total) / T) * dx
    w = e / total
    g = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        o = k - center
        if o == 0:
            g += wk * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wk * (1.0 - h[:n-o])
            g[:n-o] += -wk * h[o:]
        else:
            p = -o
            g[:n-p] += wk * (1.0 - h[p:])
            g[p:] += -wk * h[:n-p]
    g *= dx
    return obj, g
