import numpy as np
import time
from scipy.optimize import minimize

_target = 0.0
_n_glob = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = 0.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def round_to_binary(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    k_full = int(np.floor(target))
    h_bin = np.zeros(n)
    h_bin[order[:k_full]] = 1.0
    frac = target - k_full
    if frac > 1e-10 and k_full < n:
        h_bin[order[k_full]] = frac
    return h_bin

def swap_search(h, target, max_rounds, rng, sa_temps=None):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        best_val = c5_cur
        bi, bj = -1, -1
        T = 0.0
        if sa_temps is not None:
            T = sa_temps[min(rnd, len(sa_temps) - 1)]
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c5_t = compute_c5(h)
                if c5_t < best_val:
                    d = best_val - c5_t
                    if d > 0 or (T > 0 and rng.random() < np.exp(d / T)):
                        best_val = c5_t
                        bi, bj = i, j
                h[i], h[j] = h[j], h[i]
        if bi >= 0:
            h[bi], h[bj] = h[bj], h[bi]
            c5_cur = best_val
        else:
            break
    return h, c5_cur

def _sum_fun(x):
    return x.sum() - _target

def _ones_jac(x):
    return np.ones(len(x))

def lse_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += wi * (1.0 - x[:n - o])
            grad[:n - o] -= wi * x[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - x[p:])
            grad[p:] -= wi * x[:n - p]
    return grad * dx

def slsqp_optimize(h, n, target, T, maxiter):
    global _target, _n_glob
    _target = target
    _n_glob = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _ones_jac}]

    def obj(x):
        return lse_obj(x, T, n)

    def jac(x):
        return lse_grad(x, T, n)

    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def make_fractional_seeds(h_bin, target, n_fractions, rng):
    """Create continuous seeds from binary by adding fractional values at boundary positions."""
    h = h_bin.copy()
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    n_pool = min(n_fractions, len(ones), len(zeros))
    if n_pool == 0:
        return h
    pick_i = rng.choice(ones, size=n_pool, replace=False)
    pick_j = rng.choice(zeros, size=n_pool, replace=False)
    for a, b in zip(pick_i, pick_j):
        h[a] = 0.3
        h[b] = 0.7
    return project_feasible(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    global _target, _n_glob
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = (np.arange(200) % 2 == 0).astype(float)

    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = project_feasible(h0, target0)

    h_best = h0.copy()
    c5_best = compute_c5(h0)

    # Phase 1: SLSQP from initial
    for T in [80, 20, 5]:
        if time_left() < 80:
            break
        h_tmp = slsqp_optimize(h0, n0, target0, T, 600)
        c5_t = compute_c5(h_tmp)
        if c5_t < c5_best:
            c5_best = c5_t
            h_best = h_tmp.copy()

    # Phase 2: Binary + swap at n0
    h_bin = round_to_binary(h_best, target0)
    c5_b = compute_c5(h_bin)
    if c5_b < c5_best:
        c5_best = c5_b
        h_best = h_bin.copy()

    h_sw, c5_sw = swap_search(h_best, target0, 25, rng)
    if c5_sw < c5_best:
        c5_best = c5_sw
        h_best = h_sw.copy()

    # Phase 3: Multi-resolution exploration
    res_list = [70, 100, 140, 180, 210, 250, 280, 320, 350]
    for n_try in res_list:
        if time_left() < 50:
            break
        target_try = n_try / 2.0
        # Interpolate from current best
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, target_try)

        # Also try a fresh structured init
        h_struct = np.zeros(n_try)
        block = n_try // 3
        h_struct[:block] = 1.0
        h_struct[2 * block:3 * block] = 1.0
        h_struct = project_feasible(h_struct, target_try)

        for h_start in [h_try, h_struct]:
            if time_left() < 40:
                break
            # Binary + swap
            h_b = round_to_binary(h_start, target_try)
            c5_tb = compute_c5(h_b)
            if c5_tb < c5_best:
                c5_best = c5_tb
                h_best = h_b.copy()
            h_s, c5_ts = swap_search(h_b, target_try, 12, rng)
            if c5_ts < c5_best:
                c5_best = c5_ts
                h_best = h_s.copy()

            # Fractional seeds + SLSQP
            if time_left() > 30:
                h_frac = make_fractional_seeds(h_s, target_try, 6, rng)
                h_pol = slsqp_optimize(h_frac, n_try, target_try, 5, 800)
                c5_tp = compute_c5(h_pol)
                if c5_tp < c5_best:
                    c5_best = c5_tp
                    h_best = h_pol.copy()

    # Phase 4: SA swap on best binary structure
    if time_left() > 150:
        n_sa = len(h_best)
        t_sa = n_sa / 2.0
        h_sa = round_to_binary(h_best, t_sa)
        sa_temps = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
        h_sa, c5_sa = swap_search(h_sa, t_sa, 50, rng, sa_temps=sa_temps)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa.copy()

    # Phase 5: Alternating binary/continuous refinement
    for it in range(4):
        if time_left() < 60:
            break
        n_alt = len(h_best)
        t_alt = n_alt / 2.0
        # Binary from continuous
        h_b = round_to_binary(h_best, t_alt)
        c5_b = compute_c5(h_b)
        if c5_b < c5_best:
            c5_best = c5_b
            h_best = h_b.copy()
        # Swap
        h_s, c5_s = swap_search(h_b, t_alt, 15, rng)
        if c5_s < c5_best:
            c5_best = c5_s
            h_best = h_s.copy()
        # Continuum with fractional seeds
        if time_left() > 40:
            h_f = make_fractional_seeds(h_s, t_alt, 8, rng)
            h_p = slsqp_optimize(h_f, n_alt, t_alt, 3, 1500)
            c5_p = compute_c5(h_p)
            if c5_p < c5_best:
                c5_best = c5_p
                h_best = h_p.copy()

    # Phase 6: Final fine SLSQP
    if time_left() > 30:
        n_fin = len(h_best)
        t_fin = n_fin / 2.0
        for T in [2, 1, 0.5]:
            if time_left() < 15:
                break
            h_f = slsqp_optimize(h_best, n_fin, t_fin, T, 3000)
            c5_f = compute_c5(h_f)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f.copy()

    # Phase 7: Final binary check
    h_br = round_to_binary(h_best, len(h_best) / 2.0)
    c5_br = compute_c5(h_br)
    if c5_br < c5_best:
        c5_best = c5_br
        h_best = h_br.copy()

    # Ensure feasibility
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)

    return (h_best, c5_final, len(h_best))
