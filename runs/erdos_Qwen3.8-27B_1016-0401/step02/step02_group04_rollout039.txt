import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    s = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(s))) / T) * (2.0 / n)

def sm_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    s = T * (corr - mc)
    e = np.exp(s)
    sw = e / np.sum(e)
    c = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[c + o]
        if w < 1e-14: continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n-o])
            g[:n-o] -= w * h[o:]
        else:
            p = -o
            g[:n-p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n-p]
    return g * dx

def lbfgs(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return sm_obj(x, T) + pen * (x.sum() - target)**2
    def jac(x):
        return sm_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        r = minimize(obj, h0.copy(), method='L-BFGS-B', jac=jac,
                     bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_box_sum(r.x, target)
    except:
        return h0.copy()

def fista(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    tk = 1.0
    step = step_init
    best_f = sm_obj(h, T)
    best_h = h.copy()
    for _ in range(max_iter):
        y = h + ((tk - 1.0) / (tk + 2.0)) * (h - zh)
        g = sm_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        f = sm_obj(zh, T)
        if f < best_f:
            best_f = f
            best_h = zh.copy()
        if np.linalg.norm(zh - h) < 1e-13:
            break
        tnew = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        h = zh
        tk = tnew
    return best_h

def sa_move(h, target, rng, move_type):
    n = len(h)
    h_new = h.copy()
    if move_type == 0:  # Gaussian perturbation on subset
        idx = rng.choice(n, size=max(1, n//10), replace=False)
        h_new[idx] += rng.normal(0, 0.15, len(idx))
    elif move_type == 1:  # Flip segment
        i = rng.integers(0, n//2)
        length = rng.integers(2, max(3, n//5))
        j = min(i + length, n)
        h_new[i:j] = 1.0 - h[i:j]
    elif move_type == 2:  # Swap segments
        length = rng.integers(2, max(3, n//6))
        i = rng.integers(0, n - 2*length)
        j = rng.integers(0, n - 2*length)
        # Make non-overlapping
        if j < i:
            i, j = j, i
        if j < i + length:
            j = i + length
        h_new[i:i+length], h_new[j:j+length] = h_new[j:j+length].copy(), h_new[i:i+length].copy()
    else:  # Shift segment

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

def project_box_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sm_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / n)

def sm_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    e = np.exp(T * (corr - np.max(corr)))
    sw = e / np.sum(e)
    c = n - 1
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[c + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * (2.0 / n)

class SmWr:
    def __init__(self, T, target, pen):
        self.T = T
        self.target = target
        self.pen = pen
    def f(self, x):
        return sm_obj(x, self.T) + self.pen * (x.sum() - self.target) ** 2
    def g(self, x):
        n = len(x)
        return sm_grad(x, self.T) + 2.0 * self.pen * (x.sum() - self.target)

def lbfgs_solve(h0, n, target, T, pen, maxiter):
    wr = SmWr(T, target, pen)
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(wr.f, h0.copy(), method='L-BFGS-B', jac=wr.g,
                     bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_box_sum(r.x, target)
    except Exception:
        return h0.copy()

def fista_solve(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    tk = 1.0
    step = step_init
    best_h = h.copy()
    best_f = sm_obj(h, T)
    for _ in range(max_iter):
        y = h + ((tk - 1.0) / (tk + 2.0)) * (h - zh)
        zh = project_box_sum(y - step * sm_grad(y, T), target)
        f = sm_obj(zh, T)
        if f < best_f:
            best_f = f
            best_h = zh.copy()
        if np.linalg.norm(zh - h) < 1e-13:
            break
        tk = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        h = zh
    return best_h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

def project_box_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sm_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / n)

def sm_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    e = np.exp(T * (corr - np.max(corr)))
    sw = e / np.sum(e)
    c = n - 1
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[c + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * (2.0 / n)

class SmWr:
    def __init__(self, T, target, pen):
        self.T = T
        self.target = target
        self.pen = pen
    def f(self, x):
        return sm_obj(x, self.T) + self.pen * (x.sum() - self.target) ** 2
    def g(self, x):
        n = len(x)
        return sm_grad(x, self.T) + 2.0 * self.pen * (x.sum() - self.target)

def lbfgs_solve(h0, n, target, T, pen, maxiter):
    wr = SmWr(T, target, pen)
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(wr.f, h0.copy(), method='L-BFGS-B', jac=wr.g,
                     bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_box_sum(r.x, target)
    except Exception:
        return h0.copy()

def fista_solve(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    tk = 1.0
    step = step_init
    best_h = h.copy()
    best_f = sm_obj(h, T)
    for _ in range(max_iter):
        y = h + ((tk - 1.0) / (tk + 2.0)) * (h - zh)
        zh = project_box_sum(y - step * sm_grad(y, T), target)
        f = sm_obj(zh, T)
        if f < best_f:
            best_f = f
            best_h = zh.copy()
        if np.linalg.norm(zh - h) < 1e-13:
            break
        tk = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        h = zh
    return best_h
