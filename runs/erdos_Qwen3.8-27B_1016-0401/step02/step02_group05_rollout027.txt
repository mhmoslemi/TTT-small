                h_new = project(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-14:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass
        h_l2, c5_l2 = local_search(h_best, target, 600, t0, budget_s - 120)
        if c5_l2 < c5_best:
            c5_best = c5_l2
            h_best = h_l2
    # Phase 4: Fine SLSQP polish
    for T in [4, 2, 1, 0.5, 0.3]:
        if time.time() - t0 > budget_s - 25:
            break
        try:
            res = minimize(softmax_obj, h_best, method='SLSQP',
                           jac=softmax_grad, args=(T,),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            if res.fun < 100:
                h_new = project(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-15:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass
    # Phase 5: Final local search
    h_l3, c5_l3 = local_search(h_best, target, 4000, t0, budget_s - 1)
    if c5_l3 < c5_best:
        c5_best = c5_l3
        h_best = h_l3
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
