import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def topk_gradient(corr, h, K, n):
    dx = 2.0 / n
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def projected_grad_descent(h, n, target, n_iters, K=10):
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = topk_gradient(corr, h, K, n)
        # Backtracking line search
        lr = 0.5
        for _ in range(20):
            h_new = project_feasible(h - lr * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-10:
                h = h_new
                c5_best = c5_new
                h_best = h.copy()
                break
            lr *= 0.5
        else:
            # No improvement; try smaller K or break
            if lr < 1e-6:
                break
    return h_best, c5_best

    return h_bin

        for _ in range(n_attempts):
            i = zeros[rng.integers(len(zeros))]
            j = ones[rng.integers(len(ones))]

            h_trial[i] = 1.0

        if best_i >= 0:

            h_best[best_j] = 0.0
            c5_best = best_flip_c5
    return h_best, c5_best

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def topk_grad(corr, h, K, n):
    dx = 2.0 / n
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def proj_grad_descent(h, n, target, n_iters, K_start=15, K_end=3):
    c5_best = compute_c5(h)
    h_best = h.copy()
    stall = 0
    for it in range(n_iters):
        K = max(K_end, int(K_start * (1.0 - it / n_iters)))
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = topk_grad(corr, h, K, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        lr = 2.0 / g_norm
        improved = False
        for _ in range(15):
            h_new = project_feasible(h - lr * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-12:
                h = h_new
                c5_best = c5_new
                h_best = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    return h_best, c5_best

def to_binary(h, target):
    n = len(h)
    k_full = int(target)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def binary_ls(h, target, n_trials, rng, batch=200):
    n = len(h)
    c5_best = compute_c5(h)
    h_best = h.copy()
    for _ in range(n_trials):
        zeros = np.where(h_best < 0.3)[0]
        ones = np.where(h_best > 0.7)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_c5 = c5_best
        best_pair = None
        n_try = min(batch, len(zeros) * len(ones))
        for _ in range(n_try):
            i = zeros[rng.integers(len(zeros))]
            j = ones[rng.integers(len(ones))]
            h_t = h_best.copy()
            h_t[i] = 1.0
            h_t[j] = 0.0
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_pair = (i, j)
        if best_pair is not None:
            h_best[best_pair[0]] = 1.0
            h_best[best_pair[1]] = 0.0
            c5_best = best_c5
    return h_best, c5_best

def thue_morse(n):
    seq = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c += v & 1
            v >>= 1
        seq[i] = c % 2
    return seq

def rudin_shapiro(n):
    seq = np.zeros(n)
    r, s = np.array([1.0]), np.array([1.0])
    while len(r) < n:
        nr = np.concatenate([r, -r[::-1]])
        ns = np.concatenate([s, s[::-1]])
        r, s = nr, ns
    return r[:n]

def make_start(n, rng, method):
    if method == 'tm':
        h = thue_morse(n).astype(float)
    elif method == 'rs':
        h = ((rudin_shapiro(n) + 1) / 2)
    elif method == 'rand':
        h = rng.random(n)
    elif method == 'rand2':
        h = (rng.random(n) > 0.5).astype(float)
    else:
        h = rng.random(n)
    # Ensure sum is correct for binary starts
    if method in ('tm', 'rs', 'rand2'):
        k = int(n / 2)
        order = np.argsort(h)[::-1]
        h = np.zeros(n)
        h[order[:k]] = 1.0
        if k < n and n % 2 != 0:
            h[order[k]] = 0.5
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    # Get primary starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(280)
        h0[::2] = 1.0
    n = len(h0)
    target = n / 2.0
    h0 = project_feasible(h0, target)
    c5_best = compute_c5(h0)
    h_best = h0.copy()

    # Phase 1: Gradient descent from initial
    h1, c5_1 = proj_grad_descent(h0, n, target, 4000, 15, 3)
    if c5_1 < c5_best:
        c5_best = c5_1
        h_best = h1.copy()

    if elapsed() > budget_s - 200:
        return (h_best, c5_best, n)

    # Phase 2: Binary local search from best
    hb = to_binary(h_best, target)
    hb2, c5_b2 = binary_ls(hb, target, 3000, rng, batch=300)
    if c5_b2 < c5_best:
        c5_best = c5_b2
        h_best = hb2.copy()

    # Phase 3: Gradient from binary
    h3, c5_3 = proj_grad_descent(h_best, n, target, 3000, 8, 2)
    if c5_3 < c5_best:
        c5_best = c5_3
        h_best = h3.copy()

    # Phase 4: Multi-start with structured sequences
    if elapsed() < budget_s - 100:
        for method in ['tm', 'rs', 'rand', 'rand2']:
            if elapsed() > budget_s - 50:
                break
            hs = make_start(n, rng, method)
            hg, c5g = proj_grad_descent(hs, n, target, 2000, 12, 3)
            if c5g < c5_best:
                c5_best = c5g
                h_best = hg.copy()
            hsb = to_binary(hg, target)
            hsb2, c5sb = binary_ls(hsb, target, 2000, rng, batch=200)
            if c5sb < c5_best:
                c5_best = c5sb
                h_best = hsb2.copy()
            hsf, c5sf = proj_grad_descent(h_best, n, target, 1500, 6, 2)
            if c5sf < c5_best:
                c5_best = c5sf
                h_best = hsf.copy()

    # Phase 5: Alternating refinement
    if elapsed() < budget_s - 40:
        for _ in range(4):
            if elapsed() > budget_s - 15:
                break
            hp, c5p = proj_grad_descent(h_best, n, target, 2000, 5, 2)
            if c5p < c5_best:
                c5_best = c5p
                h_best = hp.copy()
            hb2, c5b2 = binary_ls(to_binary(h_best, target), target, 3000, rng, batch=250)
            if c5b2 < c5_best:
                c5_best = c5b2
                h_best = hb2.copy()

    # Phase 6: Try lower resolution if time permits
    if elapsed() < budget_s - 60 and n > 100:
        for n_lo in [n // 2, n // 3]:
            if elapsed() > budget_s - 20:
                break
            target_lo = n_lo / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
            h_lo = np.interp(x_new, x_old, h_best)
            h_lo = project_feasible(h_lo, target_lo)
            h_lo2, c5lo = proj_grad_descent(h_lo, n_lo, target_lo, 3000, 10, 3)
            if c5lo < c5_best:
                c5_best = c5lo
                h_best = h_lo2.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
