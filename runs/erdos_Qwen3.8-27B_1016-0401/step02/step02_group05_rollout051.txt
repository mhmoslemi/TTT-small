    # Project v onto {x: 0<=x<=1, sum(x)=target}
    # This is the Euclidean projection onto the intersection of box and hyperplane
    # Use iterative water-filling
    x = v.copy()
    for _ in range(30):
        x = np.clip(x, 0, 1)
        d = target - x.sum()
        if abs(d) < 1e-14: break
        # Add d uniformly to those not at bounds
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if not free.any(): 
            # can't fix, just clip
            return np.clip(x, 0, 1)
        x[free] += d / free.sum()
    x = np.clip(x, 0, 1)
    # Final fix
    d = target - x.sum()
    if abs(d) > 1e-14:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += d / free.sum()
            x = np.clip(x, 0, 1)
    return x

        h_new = project_onto_box_sum(h_new, target)

        else:

            h_prev = hc
        if t % 100 == 0:

    return hc, best

    h_final, c5_final = local_search(h_best, target0, n_iter=3000, deadline=deadline_final, sa_temp=0.001)
    if c5_final < c5_best:
        c5_best = c5_final; h_best = h_final

    h_final2, c5_final2 = targeted_refine(h_best, target0, n_iter=2000, deadline=deadline_final)
    if c5_final2 < c5_best:
        c5_best = c5_final2; h_best = h_final2

    h_best = fix_constraint(h_best, target0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n0)
