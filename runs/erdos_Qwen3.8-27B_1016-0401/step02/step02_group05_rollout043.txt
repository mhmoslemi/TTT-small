    h_c = h.copy()
    best_h = h.copy()
    curr_c5 = compute_c5(h_c)
    best_c5 = curr_c5
    corr = compute_corr(h_c)
    center = n - 1
    T0, T1 = 0.8, 0.005
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        frac = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** frac
        h_t = h_c.copy()
        mt = rng.integers(0, 5)
        if mt == 0:
            k_star = np.argmax(corr)
            s = k_star - center
            p = abs(s)
            if p == 0:
                vals = h_c * (1 - h_c)
                idx = rng.choice(np.where(vals > 0.02)[0], size=1,
                                 p=vals / vault_sum if (vault_sum := vals.sum()) > 0 else None)
                idx = np.argmax(vals) if idx is None else idx
                d = rng.uniform(0.03, 0.15)
                if h_c[idx] > 0.5:
                    h_t[idx] -= d
                else:
                    h_t[idx] += d
            else:
                if s > 0:
                    contrib = h_c[p:] * (1 - h_c[:n - p])
                else:
                    contrib = h_c[:n - p] * (1 - h_c[p:])
                pos = np.where(contrib > 0.03)[0]
                if len(pos) == 0:
                    continue
                idx = pos[rng.integers(0, len(pos))]
                d = rng.uniform(0.03, 0.15)
                if s > 0:
                    h_t[idx + p] -= d
                    h_t[idx] += d
                else:
                    h_t[idx] -= d
                    h_t[idx + p] += d
        elif mt == 1:
            ln = rng.integers(2, min(12, n // 4))
            st = rng.integers(0, n - ln)
            h_t[st:st + ln] = 1.0 - h_t[st:st + ln]
        elif mt == 2:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 != i2:
                h_t[[i1, i2]] = h_c[[i2, i1]]
        elif mt == 3:
            mask = rng.random(n) < 0.25
            h_t[mask] += rng.normal(0, 0.08, mask.sum())
        else:
            st = rng.integers(0, max(1, n - 6))
            ln = rng.integers(3, min(8, n - st))
            h_t[st:st + ln] = rng.uniform(0, 1, ln)
        h_t = project(h_t, target)
        c5_t = compute_c5(h_t)
        if c5_t <= curr_c5 or rng.random() < np.exp(-(c5_t - curr_c5) / max(T, 1e-12)):
            curr_c5 = c5_t
            h_c = h_t
            corr = compute_corr(h_c)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_c.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def left():
        return max(budget_s - (time.time() - t0), 0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: PGD with decreasing temperatures
    temps = [200, 100, 50, 30, 20, 12, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5]
    for T in temps:
        if left() < 120:
            break
        h_try, c5_try = pgd_solve(h_best, target, T, 3000, t0, time.time() + left() - 110)
        if c5_try < c5_best - 1e-14:
            c5_best = c5_try
            h_best = h_try

    # Phase 2: Simulated Annealing (multiple runs)
    for sr in range(5):
        if left() < 60:
            break
        sa_budget = min(left() - 50, 120)
        h_sa, c5_sa = sa_search(h_best, target, n_iter=8000,
                                t0=t0, budget=time.time() + sa_budget,
                                seed=seed + sr * 7)
        if c5_sa < c5_best - 1e-14:
            c5_best = c5_sa
            h_best = h_sa

    # Phase 3: Short PGD polish after SA
    for T in [3, 1.5, 0.8, 0.4]:
        if left() < 30:
            break
        h_try, c5_try = pgd_solve(h_best, target, T, 2000, t0, time.time() + left() - 25)
        if c5_try < c5_best - 1e-14:
            c5_best = c5_try
            h_best = h_try

    # Phase 4: Final SA with small noise
    if left() > 30:
        h_sa, c5_sa = sa_search(h_best, target, n_iter=6000,
                                t0=t0, budget=time.time() + left() - 5,
                                seed=seed + 999)
        if c5_sa < c5_best - 1e-14:
            c5_best = c5_sa
            h_best = h_sa

    # Final PGD
    if left() > 10:
        h_try, c5_try = pgd_solve(h_best, target, 0.5, 2000, t0, time.time() + left() - 2)
        if c5_try < c5_best - 1e-14:
            c5_best = c5_try
            h_best = h_try

    h_best = project(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
