    dx = 2.0 / n

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, target):
    """Convert to binary with exactly target ones"""
    idx = np.argsort(-h)
    h_b = np.zeros(len(h))
    h_b[idx[:int(target)]] = 1.0
    return h_b

def sm_obj_and_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    # gradient
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= (2.0 / n)
    return obj, grad

            lr *= 0.5

            lr *= 0.5

        lr = min(lr * 1.1, lr * 3)

        if time.time() - t0 > budget:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        top_idx = np.argsort(-corr)[:top_k]
        grad = np.zeros(n)
        for j in top_idx:
            k = j - c
            if k == 0:
                grad += (1.0 - 2.0 * h)
            elif k > 0:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
            else:
                p = -k
                grad[:n - p] += (1.0 - h[p:])
                grad[p:] -= h[:n - p]
        grad *= dx / top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step_new = step
        for _ in range(20):
            h_new = project_constraint(h - step_new * grad / gn, target)
            if compute_c5(h_new) < compute_c5(h) - 1e-15:
                break
            step_new *= 0.5
        else:
            step *= 0.7
            continue
        h = h_new
        step = min(step_new * 1.05, lr)
    return h, compute_c5(h)

def binary_swap_search(h_b, n_trials, rng, t0, budget):
    n = len(h_b)
    h = h_b.copy()
    best = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    stalls = 0
    for _ in range(n_trials):
        if time.time() - t0 > budget:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best - 1e-15:
            best = c5_new
            stalls = 0
            ia = ones.index(a)
            ib = zeros.index(b)
            ones[ia] = b
            zeros[ib] = a
            ones.sort()
            zeros.sort()
        else:
            h[a] = 1.0
            h[b] = 0.0
            stalls += 1
            if stalls > 5000:
                # Occasionally do a random restart perturbation
                idx = rng.integers(0, n, size=10)
                for i in idx:
                    h[i] = 1.0 - h[i]
                h = project_constraint(h, n / 2.0)
                best = compute_c5(h)
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
                stalls = 0
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(560)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Multi-temperature softmin from initial
    for T in [60, 120, 250, 500]:
        if time.time() - t0 > budget - 400:
            break
        h_best, c5 = softmin_optimize(h_best, target, T, 0.05, 800, t0, budget - 300)
        if c5 < c5_best:
            c5_best = c5

    # Phase 2: Subgradient refinement
    if time.time() - t0 < budget - 300:
        h_best, c5 = subgrad_optimize(h_best, target, 0.03, 500, t0, budget - 250, top_k=3)
        if c5 < c5_best:
            c5_best = c5

    # Phase 3: Extract binary backbone and search
    h_bin = to_binary(h_best, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin <= c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    h_best, c5_best = binary_swap_search(h_best, 300000, rng, t0, budget - 250)

    # Phase 4: Soft refinement from best binary
    for T in [100, 300, 600]:
        if time.time() - t0 > budget - 150:
            break
        h_r, c5 = softmin_optimize(h_best, target, T, 0.04, 1000, t0, budget - 100)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_r

    # Phase 5: Another binary cycle
    h_b2 = to_binary(h_best, target)
    c5_b2 = compute_c5(h_b2)
    if c5_b2 < c5_best:
        c5_best = c5_b2
        h_best = h_b2.copy()
    if time.time() - t0 < budget - 100:
        h_best, c5_best = binary_swap_search(h_best, 300000, rng, t0, budget - 80)

    # Phase 6: Final soft refinement + subgradient
    for T in [200, 400]:
        if time.time() - t0 > budget - 60:
            break
        h_r, c5 = softmin_optimize(h_best, target, T, 0.03, 1200, t0, budget - 40)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_r
    if time.time() - t0 < budget - 40:
        h_r, c5 = subgrad_optimize(h_best, target, 0.02, 400, t0, budget - 20, top_k=4)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_r

    # Phase 7: Fourier perturbations for diversity
    for r in range(8):
        if time.time() - t0 > budget - 30:
            break
        h_p = h_best.copy()
        for mf in range(rng.integers(2, 6)):
            f = rng.integers(1, n // 4)
            p = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.03, 0.12)
            x = np.arange(n)
            h_p += a * np.sin(2.0 * np.pi * f * x / n + p)
        h_p = project_constraint(h_p, target)
        h_p, c5_p = softmin_optimize(h_p, target, 300, 0.04, 500, t0, budget - 15)
        h_pb = to_binary(h_p, target)
        c5_pb = compute_c5(h_pb)
        if c5_pb < c5_best:
            c5_best = c5_pb
            h_best = h_pb
        elif c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p

    # Phase 8: Final binary check + quick swaps
    h_fin = to_binary(h_best, target)
    c5_fin = compute_c5(h_fin)
    if c5_fin < c5_best:
        c5_best = c5_fin
        h_best = h_fin
    if time.time() - t0 < budget - 5:
        h_best, c5_best = binary_swap_search(h_best, 100000, rng, t0, budget - 2)

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
