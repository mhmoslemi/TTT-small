        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sm_eval(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def grad_from_sw(h, sw):
    n = len(h)
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] -= w * h[k:]
        else:
            p = -k
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * (2.0 / n)

def pg_run(h0, target, T, n_iter, t0, deadline, lr0, beta=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, sw = sm_eval(h, T)
        grad = grad_from_sw(h, sw)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        v = beta * v - lr * grad
        h_try = project_sum(h + v, target)
        if it % 8 == 7:
            obj_try, _ = sm_eval(h_try, T)
            if obj_try > obj + 1e-11:
                lr *= 0.6
                v *= 0.4
                h_try = h
            else:
                lr = min(lr * 1.06, lr0 * 2.0)
            c5 = compute_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
        h = h_try
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def slsqp_refine(h0, target, T, maxiter, t0, deadline):
    import scipy.optimize as so
    n = len(h0)
    _T, _TR, _NN = T, target, n

    def fun(x):
        return sm_eval(x, _T)[0]

    def jac(x):
        _, sw = sm_eval(x, _T)
        return grad_from_sw(x, sw)

    def cf(x):
        return x.sum() - _TR

    def cjac(x):
        return np.ones(_NN)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cjac}]
    try:
        res = so.minimize(fun, h0, method='SLSQP', jac=jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() < deadline and res.fun < 100:
            h = project_sum(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(560)
        h[::2] = 1.0

    n = len(h)
    target = n / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Coarse-to-fine T schedule with momentum
    sched1 = [(15, 600, 0.06), (40, 700, 0.045), (80, 700, 0.035),
              (160, 600, 0.03), (320, 500, 0.025), (600, 400, 0.02)]
    for T, nit, lr0 in sched1:
        if time.time() > deadline - 350:
            break
        h, c5 = pg_run(h, target, T, nit, t0, deadline - 120, lr0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP fine-tune at moderate T
    for T in [80, 200, 500]:
        if time.time() > deadline - 200:
            break
        h_f, c5_f = slsqp_refine(best_h, target, T, 200, t0, deadline - 60)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 3: Very high-T polishing (near-true max gradient)
    for T in [1000, 2000, 5000]:
        if time.time() > deadline - 60:
            break
        h_p, c5_p = pg_run(best_h, target, T, 300, t0, deadline - 20, 0.008, beta=0.7)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: Structured perturbations + short re-optimize
    for r in range(80):
        if time.time() > deadline - 10:
            break
        h_p = best_h.copy()
        tp = r % 6
        if tp == 0:
            st = int(rng.integers(0, n))
            sz = int(rng.integers(3, 20))
            for i in range(sz):
                h_p[(st + i) % n] = 1.0 - h_p[(st + i) % n]
        elif tp == 1:
            h_p = np.clip(h_p + rng.normal(0, 0.02, n), 0, 1)
        elif tp == 2:
            h_p = np.roll(h_p, int(rng.integers(1, 8)))
        elif tp == 3:
            fr = rng.uniform(3.0, 20.0)
            h_p = np.clip(h_p + 0.015 * np.sin(2 * np.pi * fr * np.arange(n) / n), 0, 1)
        elif tp == 4:
            h_p = (h_p > rng.uniform(0.35, 0.65)).astype(float)
        else:
            # swap two small blocks
            a = int(rng.integers(0, n - 20))
            b = int(rng.integers(0, n - 20))
            sz = int(rng.integers(3, 12))
            pa = h_p[a:a+sz].copy()
            pb = h_p[b:b+sz].copy()
            h_p[a:a+sz] = pb
            h_p[b:b+sz] = pa
        h_p = project_sum(h_p, target)
        T_p = 60 + (r % 6) * 50
        h_p, c5_p = pg_run(h_p, target, T_p, 300, t0, deadline - 5, 0.02, 0.8)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final polish
    if time.time() < deadline - 3:
        h_f, c5_f = pg_run(best_h, target, 3000, 200, t0, deadline - 1, 0.005, 0.6)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    return (best_h, best_c5, n)
