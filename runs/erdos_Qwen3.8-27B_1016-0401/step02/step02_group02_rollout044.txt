    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)

        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)

        return h

    kf = int(target)
    h_bin[order[:kf]] = 1.0
    if kf < len(h):
        h_bin[order[kf]] = target - kf
    return h_bin

    if rng is None:
        rng = np.random.default_rng(0)
    c5 = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_imp = 0
        best_i, best_j = -1, -1
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = 1.0, 0.0
            c5_new = compute_c5(h)
            imp = c5 - c5_new
            if imp > best_imp:
                best_imp = imp
                best_i, best_j = i, j
            h[i], h[j] = 0.0, 1.0  # undo
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            c5 -= best_imp
            improved = True
        if not improved:

    return h, c5

def subgrad_step(h, n, target, lr, K=1):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argsort(corr)[-K:]

    for idx in top_idx:
        o = int(idx) - center
        if o == 0:

            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    g = g * (2.0 / n) / K

             'jac': lambda x: np.ones(n)}]

    hb[order[:kf]] = 1.0
    if kf < len(h):

    return hb

def binary_greedy(h, n, target, max_rounds, rng, max_pairs=None):
    """Greedy binary swap local search."""
    h = h.copy()
    c5 = compute_c5(h)
    for r in range(max_rounds):
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_pair = None
        if max_pairs is None or len(ones) * len(zeros) <= max_pairs:
            pairs = [(i, j) for i in ones for j in zeros]
        else:
            pairs = []
            for _ in range(max_pairs):

                j = zeros[rng.integers(len(zeros))]
                pairs.append((i, j))

def binary_sa(h, n, target, max_rounds, n_tries, rng, t0=0.5, t1=0.05):
    """Simulated annealing on binary swaps."""
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for r in range(max_rounds):
        temp = t0 * (t1 / t0) ** (r / max(1, max_rounds - 1))
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5_new = compute_c5(h)
            delta = c5_new - c5
            if delta < 0 or rng.random() < np.exp(-delta / (temp * 0.005)):
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    return best_h, best_c5

def subgrad_polish(h, n, target, steps, K, lr0, rng):
    h = h.copy()
    c5 = compute_c5(h)
    lr = lr0
    for it in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        center = n - 1
        top = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for idx in top:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g = g * (2.0 / n) / K
        h_new = project_feasible(h - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            lr = min(lr * 1.02, 0.12)
        else:
            lr *= 0.7
    return h, c5

def make_init(n, mode, rng):
    h = np.zeros(n)
    if mode == 0:
        h[np.arange(0, n, 2)] = 1.0
    elif mode == 1:
        xc = np.cos(2 * np.pi * np.arange(n) / (n // 3))
        order = np.argsort(xc)[::-1]
        h[order[:n // 2]] = 1.0
    else:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    return h
