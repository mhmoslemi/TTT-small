import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        step = diff / len(h)
        h += step
        h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0: continue
                h[i] += add; diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0: continue
                h[i] -= sub; diff += sub
            if abs(diff) < 1e-15: break
    return h

def soft_obj(x, T, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n) + mu * (x.sum() - target) ** 2

def soft_grad(x, T, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    ctr = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[ctr + o]
        if w < 1e-14: continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    grad *= dx
    grad += 2.0 * mu * (x.sum() - target)
    return grad

def lag_pen_obj(x, lag_idx, n, target, mu):
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n) + mu * (x.sum() - target) ** 2

def lag_pen_grad(x, lag_idx, n, target, mu):
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    g *= dx
    g += 2.0 * mu * (x.sum() - target)
    return g

    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
