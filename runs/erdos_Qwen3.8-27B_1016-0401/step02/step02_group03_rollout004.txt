def project_feasible(y, n):

    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):

        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def all_shifts_grad(h, n):
    """Compute gradients of all shift correlations. Returns (2n-1, n) array."""
    grads = np.zeros((2*n-1, n))
    j = np.arange(n)
    for k in range(2*n-1):
        s = k - (n-1)
        g = np.zeros(n)
        if s >= 0:
            m1 = j >= s
            m2 = (j + s) < n
            g[m1] += (1.0 - h[j[m1] - s])
            g[m2] -= h[j[m2] + s]
        else:
            ss = -s
            m1 = (j + ss) < n
            m2 = j >= ss
            g[m1] += (1.0 - h[j[m1] + ss])
            g[m2] -= h[j[m2] - ss]
        grads[k] = g
    return grads

def solve_lp_local(h0, n, max_iter=50):

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_grads(h, n):
    m = 2 * n - 1
    grads = np.zeros((m, n))
    j = np.arange(n)
    for k in range(m):
        s = k - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            idx1 = np.arange(s, n)
            g[idx1] += (1.0 - h[idx1 - s])
            idx2 = np.arange(0, n - s)
            g[idx2] -= h[idx2 + s]
        else:
            ss = -s
            idx1 = np.arange(0, n - ss)
            g[idx1] += (1.0 - h[idx1 + ss])
            idx2 = np.arange(ss, n)
            g[idx2] -= h[idx2 - ss]
        grads[k] = g
    return grads

def lp_local_optimize(h0, n, max_iter=80, topk=None):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = len(corr)
        if topk and topk < m:
            # Use only top-k shifts
            idx = np.argpartition(corr, -topk)[-topk:]
        else:
            idx = np.arange(m)
        grads_k = compute_all_grads(h, n)[idx]
        corr_k = corr[idx]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = grads_k
        A_ub[:, -1] = -1.0
        b_ub = corr_k - grads_k @ h
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                break
            h_new = res.x[:n]
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if np.max(np.abs(h_new - h)) < 1e-10:
                break
            h = h_new
        except Exception:
            break
    return best_h, best_c5

def soft_max_slsqp(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-12:
                continue
            s = oi - center
            if s >= 0:
                idx1 = np.arange(s, n)
                grad[idx1] += w * (1.0 - x[idx1 - s])
                idx2 = np.arange(0, n - s)
                grad[idx2] -= w * x[idx2 + s]
            else:
                ss = -s
                idx1 = np.arange(0, n - ss)
                grad[idx1] += w * (1.0 - x[idx1 + ss])
                idx2 = np.arange(ss, n)
                grad[idx2] -= w * x[idx2 - ss]
        return grad * dx
    def cons_f(x):
        return x.sum() - target
    def cons_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_f, 'jac': cons_j}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def subgradient_multi(h_init, n, t0, deadline, max_iters=200000):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.3
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        # Use top-3 active shifts for gradient
        top_k = min(3, len(corr))
        active = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(n)
        for ao in active:
            s = ao - (n - 1)
            if s >= 0:
                idx1 = np.arange(s, n)
                g[idx1] += (1.0 - h_y[idx1 - s])
                idx2 = np.arange(0, n - s)
                g[idx2] -= h_y[idx2 + s]
            else:
                ss = -s
                idx1 = np.arange(0, n - ss)
                g[idx1] += (1.0 - h_y[idx1 + ss])
                idx2 = np.arange(ss, n)
                g[idx2] -= h_y[idx2 - ss]
        g *= dx / len(active)
        alpha = alpha0 / (1.0 + 0.005 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 50000 == 0:
            for candidate in [project_feasible(h_avg, n), h_sg]:
                c = compute_c5(candidate)
                if c < c5_best:
                    c5_best = c
                    h_best = candidate.copy()
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)
        g[idx2] -= h[idx2 + s]
    else:
        ss = -s
        idx1 = np.arange(0, n - ss)
        g[idx1] += (1.0 - h[idx1 + ss])
        idx2 = np.arange(ss, n)
        g[idx2] -= h[idx2 - ss]
    return g

def compute_all_grads(h, n):
    m = 2 * n - 1
    grads = np.zeros((m, n))
    for k in range(m):
        grads[k] = shift_grad(h, n, k - (n - 1))
    return grads

def lp_local_optimize(h0, n, max_iter=80, topk=None, damp=1.0):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = len(corr)
        if topk is not None and topk < m:
            sel = np.argpartition(corr, -topk)[-topk:]
        else:
            sel = np.arange(m)
        grads_sel = compute_all_grads(h, n)[sel]
        corr_sel = corr[sel]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((len(sel), n + 1))
        A_ub[:, :n] = grads_sel
        A_ub[:, -1] = -1.0
        b_ub = corr_sel - grads_sel @ h
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                break
            h_new = res.x[:n]
            h_new = (1.0 - damp) * h + damp * h_new
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if np.max(np.abs(h_new - h)) < 1e-11:
                break
            h = h_new
        except Exception:
            break
    return best_h, best_c5

def soft_max_slsqp(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-12:
                continue
            s = oi - center
            if s >= 0:
                idx1 = np.arange(s, n)
                grad[idx1] += w * (1.0 - x[idx1 - s])
                idx2 = np.arange(0, n - s)
                grad[idx2] -= w * x[idx2 + s]
            else:
                ss = -s
                idx1 = np.arange(0, n - ss)
                grad[idx1] += w * (1.0 - x[idx1 + ss])
                idx2 = np.arange(ss, n)
                grad[idx2] -= w * x[idx2 - ss]
        return grad * dx
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def subgradient_multi(h_init, n, t0, deadline, max_iters=300000):

    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.35
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_k = min(3, len(corr))
        active = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(n)
        for ao in active:
            g += shift_grad(h_y, n, ao - (n - 1))
        g *= dx / len(active)
        alpha = alpha0 / (1.0 + 0.006 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 40000 == 0:
            for cand in [project_feasible(h_avg, n), h_sg]:
                c = compute_c5(cand)
                if c < c5_best:
                    c5_best = c
                    h_best = cand.copy()
    return h_best, c5_best
