        mx = vals.max()

        return slopes.min(), slopes.max()

    if sg_lo[0] >= 0:  # subgradient positive at lo

    sg_hi = subgrad_range(hi)

        elif smax < 0:

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n))

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc) - T * (corr - mc).max()
    e = np.exp(sh)
    se = e.sum()
    return (mc + np.log(se) / T) * (2.0 / n), e / se

def sm_grad(h, sw):
    n = len(h)
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] -= w * h[k:]
        else:
            p = -k
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * (2.0 / n)

def nesterov_pd(h0, target, T0, T1, n_iter, lr0, momentum, t0, tlimit):
    h = h0.copy()
    v = np.zeros_like(h)
    best_c5, best_h = compute_c5(h), h.copy()
    for it in range(n_iter):
        if time.time() - t0 > tlimit:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iter - 1))
        obj, sw = sm_values(h, T)
        grad = sm_grad(h, sw)
        v = momentum * v + (1 - momentum) * grad
        # Backtracking line search
        lr = lr0
        for _ in range(30):
            h_new = project_constraint(h - lr * v, target)
            obj_new, _ = sm_values(h_new, T)
            if obj_new < obj - 1e-15:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        if it % 20 == 19:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def coord_polish(h, target, n_rounds, t0, tlimit):
    """Coordinate-wise optimization of max correlation."""
    n = len(h)
    h = h.copy()
    for r in range(n_rounds):
        if time.time() - t0 > tlimit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Compute effect of changing h[i]
            # corr[j] changes by g_i[j] * (new - old)
            # Just try new values: 0, 1, and a few in between
            best_val, best_h = compute_c5(h), old
            for trial in [0.0, 1.0, 0.25, 0.5, 0.75]:
                h[i] = trial
                c5_try = compute_c5(h)
                if c5_try < best_val - 1e-12:
                    best_val, best_h = c5_try, trial
            h[i] = best_h
            if best_h != old:
                improved = True
        if not improved:
            break
    # Re-project to maintain constraint
    h = project_constraint(h, target)
    return h, compute_c5(h)

def binary_swaps(h, target, n_steps, t0, tlimit, rng):
    """Local search via random swaps on near-binary solution."""
    n = len(h)
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_cur = h.copy()
    for _ in range(n_steps):
        if time.time() - t0 > tlimit:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or abs(h_cur[i] - h_cur[j]) < 1e-6:
            continue
        # Swap
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5 = compute_c5(h_cur)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]  # revert
        # Re-project
        h_cur = project_constraint(h_cur, target)
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):

    budget = budget_s
    rng = np.random.default_rng(seed)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Nesterov PD with geometric cooling
    for (T0, T1, niter, lr0, mom) in [(50, 500, 2000, 0.04, 0.8),
                                         (100, 800, 1500, 0.03, 0.85),
                                         (200, 1500, 1000, 0.02, 0.9)]:
        if time.time() - t0 > budget - 300:
            break
        h, c5 = nesterov_pd(h_best, target, T0, T1, niter, lr0, mom, t0, budget - 200)
        if c5 < c5_best:
            c5_best, h_best = c5, h
    
    # Phase 2: Coordinate polish
    if time.time() - t0 < budget - 150:
        h, c5 = coord_polish(h_best, target, 3, t0, budget - 100)
        if c5 < c5_best:
            c5_best, h_best = c5, h
    
    # Phase 3: Binary swap local search
    if time.time() - t0 < budget - 100:
        # Make it more binary
        h_bin = project_constraint((h_best > 0.4).astype(float), target)
        h, c5 = binary_swaps(h_bin, target, 50000, t0, budget - 80, rng)
        if c5 < c5_best:
            c5_best, h_best = c5, h
        # Also try on the original (fractional)
        h, c5 = binary_swaps(h_best, target, 30000, t0
