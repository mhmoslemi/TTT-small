import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def overlaps(h):
    """Return all overlap values (2n-1 of them)"""
    n = len(h)
    return np.correlate(h, 1 - h, mode="full") * (2.0 / n)

def smooth_max_and_weights(corr, T):
    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T)
    w = e / se
    return obj, w

def grad_overlaps(h, w):
    """Gradient of weighted sum of overlaps: sum_m w[m] * corr[m] * dx"""
    n = len(h)
    dx = 2.0 / n
    c = n - 1
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wg = w[j]
        if wg < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += wg * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wg * (1.0 - h[:n - k])
            grad[:n - k] -= wg * h[k:]
        else:
            p = -k
            grad[:n - p] += wg * (1.0 - h[p:])
            grad[p:] -= wg * h[:n - p]
    return grad * dx

def pg_step(h, target, T, n_iter, lr0, t0, budget, check_every=200):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        corr = overlaps(h)
        obj, w = smooth_max_and_weights(corr, T)
        grad = grad_overlaps(h, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        # Backtracking line search
        for _ in range(50):
            h_new = project_constraint(h - lr * grad, target)
            corr_new = overlaps(h_new)
            obj_new, _ = smooth_max_and_weights(corr_new, T)
            if obj_new < obj - 1e-16:

    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def project_constraint(h, target):
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0

        else:

def soft_max_obj_w(corr, T):
    mx = corr.max()
    sh = T * (corr - mx)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    return (mx + np.log(se) / T), e / se

def grad_from_w(h, w):
    n = len(h)
    dx = 2.0 / n

        else:
            p = -k
