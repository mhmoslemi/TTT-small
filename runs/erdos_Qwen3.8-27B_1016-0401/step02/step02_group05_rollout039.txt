    n_corr = len(corr)
    center = n - 1
    # Find top_k lags by correlation value
    top_indices = np.argsort(corr)[::-1][:top_k]

    elif s > 0:

def subgrad_descent(h, n_iter, t0=None, budget=None, top_k=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    corr = compute_corr(h_curr)
    stall = 0
    alpha = 0.5
    
    for it in range(n_iter):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        if stall > 100:
            alpha *= 0.5
            stall = 0
            if alpha < 0.001:
                break
        
        # Find top lags
        top_idx = np.argsort(corr)[::-1][:top_k]
        max_c = corr[top_idx[0]]
        
        # Weighted gradient
        grad = np.zeros(n)
        for idx in top_idx:
            w = 1.0 - (max_c - corr[idx]) / (max_c + 1e-10)
            w = max(0.0, w)
            s = idx - center
            grad += w * corr_grad_single(h_curr, s)

            best_c5 = c5_new

            corr = corr_new

    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0.0, 1.0)

def corr_grad_single(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        v = 1.0 - h
        grad[s:] += v[:n - s]
        grad[:n - s] -= h[s:]
    else:
        p = -s
        v = 1.0 - h
        grad[:n - p] += v[p:]
        grad[p:] -= h[:n - p]
    return grad

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        s = o
        if s == 0:
            grad += w * (1.0 - 2.0 * x)
        elif s > 0:
            grad[s:] += w * (1.0 - x[:n - s])
            grad[:n - s] += -w * x[s:]
        else:
            p = -s
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def sum_con_fun(x, target):
    return x.sum() - target

def sum_con_jac(x, target):
    return np.ones(len(x))

def subgrad_loop(h, n_iter, t0, budget, top_k=3, alpha_max=0.5, alpha_min=0.01):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    corr = compute_corr(h_curr)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        frac = min(1.0, it / max(1, n_iter - 1))
        alpha = alpha_max * (alpha_min / alpha_max) ** frac
        top_idx = np.argsort(corr)[::-1][:top_k]
        max_c = corr[top_idx[0]]
        grad = np.zeros(n)
        for idx in top_idx:
            w = max(0.0, 1.0 - (max_c - corr[idx]) / (max_c + 1e-10))
            s = idx - center
            grad += w * corr_grad_single(h_curr, s)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = project_box_sum(h_curr - alpha * grad, target)
        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new) * dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h_curr = h_new
            corr = corr_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
    return h_curr, best_c5

def pair_descend(h, n_iter, t0, budget, delta=0.03):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    corr = compute_corr(h_curr)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 200:
            delta *= 0.5
            stall = 0
            if delta < 0.001:
                break
        k_star = int(np.argmax(corr))
        s = k_star - center
        p = abs(s)
        h_old = h_curr.copy()
        if p == 0:
            vals = h_curr * (1 - h_curr)
            idx = int(np.argmax(vals))
            h_curr[idx] += -delta if h_curr[idx] > 0.5 else delta
        else:
            if s > 0:
                contrib = h_curr[p:] * (1 - h_curr[:n - p])
            else:
                contrib = h_curr[:n - p] * (1 - h_curr[p:])
            idx = int(np.argmax(contrib))
            j1 = idx if s < 0 else idx + p
            j2 = idx + p if s < 0 else idx
            d = min(delta, h_curr[j1], 1.0 - h_curr[j2])
            if d < 1e-8:
                stall += 50
                h_curr = h_old
                continue
            h_curr[j1] -= d
            h_curr[j2] += d
        h_curr = np.clip(h_curr, 0, 1)
        h_curr = project_box_sum(h_curr, target)
        corr_new = compute_corr(h_curr)
        c5_new = float(np.max(corr_new) * dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            corr = corr_new
            stall = 0
        else:
            h_curr = h_old
            stall += 1
    return h_curr, best_c5

    if c5_2 < c5_best:

    # Phase 3: SLSQP soft-max annealing
    for T in [15, 8, 5, 3, 2, 1.5, 1]:
        time_limit = t0 + budget_s * 0.6
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (target,)}],
                          options={'maxiter': 2000, 'ftol': 1e-15})
            if res.fun < 100:
                h_new = project_box_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12:
                    c5_best, h_best = c5, h_new
        except Exception:
            pass
    
    # Phase 4: Interleaved refinement
    for _ in range(15):
        if time.time() - t0 > t0 + budget_s * 0.85:
            break
        h4a, c5_4a = subgrad_loop(h_best, 400, t0, time.time() + 20, top_k=5, alpha_max=0.1, alpha_min=0.005)
        if c5_4a < c5_best:
            h_best, c5_best = h4a, c5_4a
        h4b, c5_4b = pair_descend(h_best, 400, t0, time.time() + 15, delta=0.02)
        if c5_4b < c5_best:
            h_best, c5_best = h4b, c5_4b
    
    # Phase 5: Sinusoidal perturbation restarts
    for r in range(25):
        if time.time() - t0 > t0 + budget_s * 0.93:
            break
        freq = int(rng.integers(1, 12))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.03, 0.15))
        x = np.arange(n, dtype=float) / n
        h_try = h_best + amp * np.sin(2 * np.pi * freq * x + phase)
        h_try = project_box_sum(h_try, target)
        h5a, c5_5a = subgrad_loop(h_try, 150, t0, time.time() + 12, top_k=3, alpha_max=0.15, alpha_min=0.01)
        h5b, c5_5b = pair_descend(h5a, 200, t0, time.time() + 8, delta=0.015)
        if c5_5b < c5_best - 1e-12:
            h_best, c5_best = h5b, c5_5b
    
    # Final polish
    hf, c5f = pair_descend(h_best, 2000, t0, t0 + budget_s - 2, delta=0.01)
    if c5f < c5_best:
        h_best, c5_best = hf, c5f
    
    c5_final = compute_c5(h_best)
    return (h_best, min(c5_final, c5_best), n)
