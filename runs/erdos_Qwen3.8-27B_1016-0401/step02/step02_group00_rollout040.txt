        g[:n - p] += (1.0 - x[p:])
        g[p:] += -x[:n - p]
    return g * dx

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def sm_wrap(x, T):
    return sm_obj(x, T)

def sm_wrap_grad(x, T):
    return sm_grad(x, T)

def lag_wrap(x, lag_idx):
    return lag_obj(x, lag_idx)

def lag_wrap_grad(x, lag_idx):
    return lag_grad(x, lag_idx)

def _slsqp_optimize(obj_fn, jac_fn, x0, bounds, cons, maxiter, ftol):
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return x0

def _slsqp_optimize_kw(obj_fn, jac_fn, x0, bounds, cons, maxiter, ftol, arg):
    try:
        res = minimize(obj_fn, x0, args=(arg,), method='SLSQP', jac=jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return x0

def sm_obj_arg(x, T):
    return sm_obj(x, T)

def sm_grad_arg(x, T):
    return sm_grad(x, T)

def lag_obj_arg(x, lag_idx):
    return lag_obj(x, lag_idx)

def lag_grad_arg(x, lag_idx):
    return lag_grad(x, lag_idx)

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    # Multi-start: try a few structured initial guesses
    starts = [h_best.copy()]
    hb = np.zeros(n)
    hb[:n // 4] = 1.0
    hb[n // 2:3 * n // 4] = 1.0
    starts.append(fix_constraint(hb, target))
    hs = 0.5 + 0.5 * np.sin(2.0 * np.pi * np.arange(n) * 2.0 / n)
    starts.append(fix_constraint(hs, target))
    for s in starts:
        cs = compute_c5(s)
        if cs < c5_best:
            c5_best = cs
            h_best = s.copy()

    # Phase 1: Soft-max warmup at increasing T (25% budget)
    p1_end = t0 + budget_s * 0.25
    for T in [20, 40, 80, 160, 320, 640, 1280]:
        if elapsed() > p1_end:
            break
        x_new = _slsqp_optimize_kw(sm_obj_arg, sm_grad_arg, h_best, bounds, cons,
                                    3000, 1e-16, T)
        h_new = fix_constraint(np.clip(x_new, 0, 1), target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new

    # Phase 2: Active-set cycling on top lags (40% budget)
    p2_end = t0 + budget_s * 0.65
    n_total_calls = 0
    max_calls_p2 = 200
    while elapsed() < p2_end and n_total_calls < max_calls_p2:
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        center = n - 1
        top_k = 5
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for lag in top_idx:
            if elapsed() > p2_end or n_total_calls >= max_calls_p2:
                break
            x_new = _slsqp_optimize_kw(lag_obj_arg, lag_grad_arg, h_best,
                                       bounds, cons, 1000, 1e-16, lag)
            h_new = fix_constraint(np.clip(x_new, 0, 1), target)
            c5_new = compute_c5(h_new)
            n_total_calls += 1
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
                improved = True
        if not improved and n_total_calls > 50:
            # Stagnation: try a perturbation
            h_pert = h_best + rng.normal(0, 0.005, n)
            h_pert = fix_constraint(h_pert, target)
            cp = compute_c5(h_pert)
            if cp < c5_best:
                c5_best = cp
                h_best = h_pert

    # Phase 3: High-T soft-max for global balancing (15% budget)
    p3_end = t0 + budget_s * 0.80
    for T in [1000, 2000, 4000, 8000]:
        if elapsed() > p3_end:
            break
        x_new = _slsqp_optimize_kw(sm_obj_arg, sm_grad_arg, h_best, bounds, cons,
                                    4000, 1e-16, T)
        h_new = fix_constraint(np.clip(x_new, 0, 1), target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new

    # Phase 4: Final single-worst-lag polish (15% budget)
    p4_end = t0 + budget_s - 15.0
    n_calls_p4 = 0
    while elapsed() < p4_end and n_calls_p4 < 40:
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        x_new = _slsqp_optimize_kw(lag_obj_arg, lag_grad_arg, h_best,
                                   bounds, cons, 2000, 1e-17, worst_f)
        h_new = fix_constraint(np.clip(x_new, 0, 1), target)
        c5_new = compute_c5(h_new)
        n_calls_p4 += 1
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 5: Try a quick interpolation to n*1.5 and back (if time)
    if elapsed() < budget_s - 30:
        n2 = n + n // 2
        x_old = np.arange(n) * (2.0 / n)
        x_new_grid = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new_grid, x_old, h_best)
        target2 = n2 / 2.0
        h_hi = fix_constraint(h_hi, target2)
        c5_hi = compute_c5(h_hi)
        _TARGET = target2
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        for _ in range(3):
            if elapsed() > budget_s - 20:
                break
            corr_h = np.correlate(h_hi, 1.0 - h_hi, mode="full")
            worst_h = int(np.argmax(corr_h))
            x_new = _slsqp_optimize_kw(lag_obj_arg, lag_grad_arg, h_hi,
                                       bounds2, cons2, 500, 1e-15, worst_h)
            h_hi = fix_constraint(np.clip(x_new, 0, 1), target2)
            c5_hi = compute_c5(h_hi)
        # Interpolate back to n
        x_new_back = np.arange(n) * (2.0 / n)
        h_back = np.interp(x_new_back, x_new_grid, h_hi)
        h_back = fix_constraint(h_back, target)
        c5_back = compute_c5(h_back)
        if c5_back < c5_best:
            c5_best = c5_back
            h_best = h_back
        elif c5_hi < c5_best:
            # Keep the higher-n solution
            c5_best = c5_hi
            h_best = h_hi
            _TARGET = target2

    # Restore target for best solution
    n_final = len(h_best)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n_final)
