        # Quick active-set on alt
        for lag_pass in range(5):
            if elapsed() > p4_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = int(np.argmax(corr_a))
            try:
                res = minimize(lag_obj, h_alt, args=(worst_a,), method='SLSQP',
                              jac=lag_grad, bounds=bounds_alt, constraints=cons_alt,
                              options={'maxiter': 1000, 'ftol': 1e-15})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            bounds = bounds_alt
            _TARGET = target_alt

    # Phase 5: Final high-T polish (5% budget)
    p5_end = t0 + budget_s - 5.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    _TARGET = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    cons_f = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for T_f in [100000, 500000]:
        if elapsed() > p5_end:
            break
        try:
            res = minimize(sm_obj, h_best, args=(T_f,), method='SLSQP',
                          jac=sm_grad, bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 8000, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Final active-set polish
    for _ in range(15):
        if elapsed() > p5_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            res = minimize(lag_obj, h_best, args=(worst_f,), method='SLSQP',
                          jac=lag_grad, bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 5000, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_constraint(h_best, len(h_best) / 2.0)
    return (h_best, c5_final, len(h_best))
