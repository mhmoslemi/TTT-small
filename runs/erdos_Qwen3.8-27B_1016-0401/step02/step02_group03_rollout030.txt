        h_new = do_slsqp(h_p, n, T_use, 3000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Very fine local directions
    if time.time() - t0 < budget_s - 15:
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        mx = np.max(corr)
        active = np.where(corr > mx * 0.998)[0]
        center = n - 1
        dx_local = 2.0 / n
        for trial in range(25):
            if time.time() - t0 > budget_s - 5:
                break
            direction = np.zeros(n)
            for ai in active:
                direction -= 0.5 * shift_grad(h_best, n, ai - center)
            norm_d = np.linalg.norm(direction)
            if norm_d < 1e-15:
                break
            direction /= norm_d
            improved = False
            for step in [0.008, 0.004, 0.002, 0.001, 0.0005]:
                h_try = project_feasible(h_best + step * direction, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-14:
                    c5_best = c5_try
                    h_best = h_try
                    improved = True
                    break
            if not improved:
                direction = -direction
                for step in [0.004, 0.002, 0.001]:
                    h_try = project_feasible(h_best + step * direction, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_best - 1e-14:
                        c5_best = c5_try
                        h_best = h_try
                        improved = True
                        break
            if not improved:
                # Random micro-perturbation
                h_try = project_feasible(h_best + rng.normal(0, 0.001, n), n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best:
                    c5_best = c5_try
                    h_best = h_try

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
