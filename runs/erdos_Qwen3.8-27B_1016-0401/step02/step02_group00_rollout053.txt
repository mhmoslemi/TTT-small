def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-15:
                break
    return h

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(1, n):
        w_pos = sw[center + o]
        w_neg = sw[center - o]
        if w_pos > 1e-12:
            grad[o:] += w_pos * (1.0 - x[:n-o])
            grad[:n-o] -= w_pos * x[o:]
        if w_neg > 1e-12:
            grad[:n-o] += w_neg * (1.0 - x[o:])
            grad[o:] -= w_neg * x[:n-o]
    w0 = sw[center]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * x)
    return grad * dx

        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
        n_sweeps += 1
        if n_sweeps > 200:
            break

    # Phase 2: L-BFGS-B with penalty + SLSQP soft-max (20%)
    p2_end = t0 + budget_s * 0.55
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    for T, mu in [(200, 10.0), (500, 5.0), (1000, 2.0), (2000, 1.0)]:
        if time.time() - t0 > p2_end:
            break
        try:
            res = minimize(pen_sm_obj, h_best, args=(T, mu, target), method='L-BFGS-B',
                          jac=lambda x, t=T, m=mu: pen_sm_grad(x, t, m, target),
                          bounds=bounds, options={'maxiter': 2000, 'factr': 1e10})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    if time.time() - t0 < p2_end:
        for T in [500, 1500, 5000]:
            if time.time() - t0 > p2_end:
                break
            try:
                res = minimize(sm_obj, h_best, args=(T,), method='SLSQP',
                              jac=lambda x, t=T: sm_grad(x, t),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 3: Active-set top lags (20%)
    p3_end = t0 + budget_s * 0.75
    center = n - 1
    for outer in range(50):
        if time.time() - t0 > p3_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        for lag in top_idx:
            if time.time() - t0 > p3_end:
                break
            try:
                res = minimize(lag_obj, h_best, args=(lag,), method='SLSQP',
                              jac=lambda x, l=lag: lag_grad(x, l),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 1500, 'ftol': 1e-17})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 4: Random perturbations + re-optimize (15%)
    p4_end = t0 + budget_s * 0.90
    for attempt in range(15):
        if time.time() - t0 > p4_end:
            break
        h_pert = h_best.copy()
        n_pert = rng.integers(3, 10)
        indices = rng.choice(n, size=n_pert, replace=False)
        for idx in indices:
            h_pert[idx] = np.clip(h_pert[idx] + rng.uniform(-0.1, 0.1), 0, 1)
        h_pert = fix_constraint(h_pert, target)
        # Quick CD sweep
        h_pert = coord_cd_sweep(h_pert, grid, 5.0)
        h_pert = fix_constraint(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert < c5_best:
            # Polish with SLSQP
            try:
                res = minimize(sm_obj, h_pert, args=(2000,), method='SLSQP',
                              jac=lambda x: sm_grad(x, 2000),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                c5_best = c5_pert
                h_best = h_pert

    # Phase 5: Try n=350 (10% remaining)
    p5_end = t0 + budget_s - 15.0
    for n_alt in [350, 420]:
        if time.time() - t0 > p5_end:
            break
        target_alt = n_alt / 2.0
        _TARGET = target_alt
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.arange(n_alt) * (2.0 / n_alt)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        cons_alt = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        grid_alt = np.linspace(0.0, 1.0, 301)
        while time.time() - t0 < p5_end and (p5_end - (t0 + budget_s * 0.90)) > 5:
            if time.time() - t0 > p5_end:
                break
            h_alt = coord_cd_sweep(h_alt, grid_alt, p5_end - (t0 + budget_s*0.90) * 0.5)
            h_alt = fix_constraint(h_alt, target_alt)
            c5_new = compute_c5(h_alt)
            if c5_new < c5_alt:
                c5_alt = c5_new
        try:
            res = minimize(sm_obj, h_alt, args=(3000,), method='SLSQP',
                          jac=lambda x: sm_grad(x, 3000),
                          bounds=bounds_alt, constraints=cons_alt,
                          options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt = h_new
        except Exception:
            pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt

    _TARGET = len(h_best) / 2.0
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
