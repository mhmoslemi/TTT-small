def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1 - h, mode="full")

def soft_max_obj(h, T):
    n = len(h)
    corr = corr_full(h)
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = corr_full(h)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted - np.max(shifted))  # numerical stability
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(sw)):
        w = sw[idx]
        if w < 1e-14:

        o = idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx
