        if time.time() - t0 > budget_s * 0.80:
            break
        h_new = do_slsqp(h_best, n, T, 4000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Diverse perturbations + quick polish
    for pi in range(35):
        if time.time() - t0 > budget_s - 30:
            break
        r = rng.random()
        if r < 0.30:
            sg = 0.002 + 0.008 * rng.random()
            h_p = h_best + rng.normal(0, sg, n)
        elif r < 0.55:
            w = rng.integers(5, 40)
            idx = rng.integers(0, max(1, n - w))
            h_p = h_best.copy()
            h_p[idx:idx+w] += rng.normal(0, 0.04, w)
        elif r < 0.75:
            freq = rng.uniform(0.5, 12.0)
            ph = rng.uniform(0, 2.0 * np.pi)
            amp = 0.02 + 0.03 * rng.random()
            h_p = h_best + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + ph)
        else:
            # Swap two blocks
            b1 = int(n * 0.2)
            b2 = int(n * 0.35)
            i1 = rng.integers(0, n - b1)
            i2 = rng.integers(0, n - b2)
            h_p = h_best.copy()
            h_p[i1:i1+b1], h_p[i2:i2+b2] = h_p[i2:i2+b2].copy(), h_p[i1:i1+b1].copy()
        h_p = project_feasible(h_p, n)
        T_p = 2.0
        h_new = do_slsqp(h_p, n, T_p, 1200, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Final aggressive SLSQP at very low T
    if time.time() - t0 < budget_s - 15:
        h_new = do_slsqp(h_best, n, 0.05, 6000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new2 = do_slsqp(h_best, n, 0.1, 4000, 1e-19)
        h_new2 = project_feasible(h_new2, n)
        c52 = compute_c5(h_new2)
        if c52 < c5_best:
            c5_best = c52
            h_best = h_new2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
