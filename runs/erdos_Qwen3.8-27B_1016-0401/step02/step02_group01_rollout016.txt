def project_constraint(h, target):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_from_shifts(h, shifts, dx):
    """Compute subgradient at given shifts (indices into full correlation)"""
    n = len(h)
    c = n - 1
    grad = np.zeros(n)
    for j in shifts:
        k = j - c
        if k >= 0:
            # r = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
            # d/dh[i] for i in [k, n-1]: (1-h[i-k])
            # d/dh[i] for i in [0, n-1-k]: -h[i+k]
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            p = -k

            # = sum_{i=0}^{n-1-p} h[i]*(1-h[i-p]) ... wait

            # d/dh[i] for i in [0, n-1-p]: -h[i+p]
            grad[p:] += (1.0 - h[:n-p])
            grad[:n-p] -= h[p:]

def perturb(h, target, rng, mode):

    n = len(h)
    if mode == 0:
        # Block flip
        idx = rng.integers(0, n)
        sz = rng.integers(5, 40)

        h2[mask] = 1.0 - h2[mask]
    elif mode == 1:
        # Gaussian noise

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, k, dx):
    """Gradient of overlap at shift k (correlation index)"""
    n = len(h)
    c = n - 1
    k = k - c
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        p = -k
        grad[p:] += (1.0 - h[:n-p])
        grad[:n-p] -= h[p:]
    return grad * dx

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_descent(h, target, t0, budget, max_iter, step0, k_top):
    n = len(h)
    dx = 2.0 / n
    c = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for it in range(max_iter):
        if time.time() - t0 > budget - 3:
            break
        corr = compute_corr(h)
        if k_top < len(corr):
            top_k = np.argpartition(corr, -k_top)[-k_top:]
        else:
            top_k = np.arange(len(corr))
        top_vals = corr[top_k].astype(float)
        ss = np.dot(top_vals, top_vals)
        if ss < 1e-30:
            break
        inv_ss = 1.0 / np.sqrt(ss)
        grad = np.zeros(n)
        for idx in range(len(top_k)):
            w = top_vals[idx] * inv_ss
            if abs(w) < 1e-14:
                continue
            j = top_k[idx]
            k = j - c
            if k >= 0:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
            else:
                p = -k
                grad[p:] += w * (1.0 - h[:n - p])
                grad[:n - p] -= w * h[p:]
        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        improved = False
        for _ in range(25):
            h_new = project_constraint(h - step * grad, target)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.3
            continue
        step = min(step * 1.02, step0)
    return best_h, best_c5

def perturb(h, target, rng, mode):
    n = len(h)
    if mode == 0:
        idx = int(rng.integers(0, n))
        sz = int(rng.integers(4, 35))
        h2 = h.copy()
        for i in range(sz):
            h2[(idx + i) % n] = 1.0 - h2[(idx + i) % n]
    elif mode == 1:
        h2 = np.clip(h + rng.normal(0, 0.06, n), 0, 1)
    elif mode == 2:
        h2 = (h + rng.normal(0, 0.08, n) > rng.uniform(0.3, 0.7)).astype(float)
    elif mode == 3:
        h2 = np.roll(h, int(rng.integers(2, 15)))
    elif mode == 4:
        h2 = h.copy()
        n_f = int(rng.integers(5, 20))
        idxs = rng.choice(n, size=n_f, replace=False)
        h2[idxs] = 1.0 - h2[idxs]
    else:
        h2 = h.copy()
        start = int(rng.integers(0, n - 60))
        w = 60
        kernel = np.ones(w) / w
        h2[start:start + w] = np.convolve(h2[start:start + w], kernel, 'same')
    return project_constraint(h2, target)

def binarize_step(h, target, t0, budget):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid_idx = np.where((h > 0.2) & (h < 0.8))[0]
    rng_local = np.random.default_rng(123)
    rng_local.shuffle(mid_idx)
    for i in mid_idx:
        if time.time() - t0 > budget - 3:
            break
        for tv in [0.0, 1.0]:
            h_trial = best_h.copy()
            h_trial[i] = tv
            h_trial = project_constraint(h_trial, target)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_trial
    return best_h, best_c5
