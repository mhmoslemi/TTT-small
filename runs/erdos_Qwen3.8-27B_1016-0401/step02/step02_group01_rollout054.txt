```python
import numpy as np
from scipy.optimize import minimize, minimize_nlbfgs
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmin_obj_and_weights(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc) - sh.max()  # numerical stability
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    w = e / se
    return obj, w, corr, mc

def softmin_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-15:
            continue
        k = j - c
        if k == 0:
            g += wj * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += wj * (1.0 - h[:n-k])
            g[:n-k] -= wj * h[k:]
        else:
            p = -k
            g[:n-p] += wj * (1.0 - h[p:])
            g[p:] -= wj * h[:n-p]
    return g * (2.0 / n)
```

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid

    n = len(h)

    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sm_eval(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    return (mc + np.log(e.sum()) / T) * 2.0 / n, e / e.sum(), corr

def sm_grad(h, w, thresh=1e-10):
    n = len(h)
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < thresh:
            continue
        k = j - c
        if k == 0:
            g += wj * (1.0 - 2.0 * h)
        elif k > 0:
            g[:n-k] += wj * (1.0 - h[k:])
            g[k:] -= wj * h[:n-k]
        else:
            p = -k
            g[p:] += wj * (1.0 - h[:n-p])
            g[:n-p] -= wj * h[p:]
    return g * 2.0 / n

_G = {}

def _f(x):
    return sm_eval(x, _G['T'])[0]

def _g(x):
    _, w, _ = sm_eval(x, _G['T'])
    return sm_grad(x, w)

def _cf(x):
    return x.sum() - _G['target']

def _cj(x):
    return np.ones(len(x))

def mom_pg(h0, target, T, n_iter, deadline, lr0, beta):
    h = h0.copy()
    v = np.zeros_like(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, w, _ = sm_eval(h, T)
        grad = sm_grad(h, w)
        lr = lr0
        v = beta * v - lr * grad
        h_new = project_constraint(h + v, target)
        obj_new, _, _ = sm_eval(h_new, T)
        tries = 0
        while obj_new > obj + 1e-13 and tries < 25:
            lr *= 0.5
            v = beta * v - lr * grad
            h_new = project_constraint(h + v, target)
            obj_new, _, _ = sm_eval(h_new, T)
            tries += 1
        if tries >= 25:
            v *= 0.3
            lr = lr0
            continue
        h = h_new
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def slsqp_solve(h0, target, T, maxiter, deadline):
    _G['T'] = T
    _G['target'] = target
    n = len(h0)
    try:
        res = minimize(_f, h0, jac=_g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, float)
    else:
        h = np.zeros(560)
        h[::2] = 1.0
    n = len(h)
    tgt = n / 2.0
    h = project_constraint(h, tgt)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    d1 = t0 + budget_s * 0.50
    for T in [5, 15, 50, 150, 500, 1500, 4000]:
        if time.time() > d1 - 3:
            break
        h, c5 = mom_pg(h, tgt, T, 200000, d1, 0.05, 0.85)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # SLSQP fine-tune
    d2 = t0 + budget_s * 0.62
    for T in [300, 1000]:
        if time.time() > d2 - 2:
            break
        h, c5 = slsqp_solve(h, tgt, T, 150, d2)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    d3 = t0 + budget_s * 0.80
    if time.time() < d3 - 20:
        n2 = 280
        tgt2 = n2 * 0.5
        x2 = np.linspace(0, 2, n2, endpoint=False)
        xo = np.arange(len(h)) * 2.0 / len(h)
        h2 = project_constraint(np.interp(x2, xo, h), tgt2)
        for T in [10, 80, 500]:
            if time.time() > d3 - 2:

            h2, c5 = mom_pg(h2, tgt2, T, 200000, d3, 0.06, 0.87)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()
        if time.time() < d3 - 15:
            h2, c5 = slsqp_solve(h2, tgt2, 400, 100, d3)
            if c5 < best_c5:
                best_c5, best_h = c5, h2

    # Perturbation restarts
    d4 = t0 + budget_s * 0.97
    nb = len(best_h)
    tgtb = nb * 0.5
    for r in range(80):
        if time.time() > d4 - 1:
            break
        hp = best_h.copy()
        m = r % 5
        if m == 0:
            hp += rng.normal(0, 0.03, nb)
        elif m == 1:
            pos = (np.arange(int(rng.integers(0, nb)), int(rng.integers(0, nb)) + 12)) % nb
            hp[pos] = 1.0 - hp[pos]
        elif m == 2:
            hp = np.clip(hp + rng.normal(0, 0.08, nb), 0, 1)
        elif m == 3:
            hp = np.roll(hp, int(rng.integers(1, 15)))
        else:
            hp = (hp > rng.uniform(0.2, 0.8)).astype(float)
        hp = project_constraint(hp, tgtb)
        hp, c5 = mom_pg(hp, tgtb, 200 + r * 15, 4000, d4, 0.03, 0.85)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Final micro-polish
    if time.time() < t0 + budget_s - 3:

        best_c5 = min(best_c5, c5)

    return best_h, min(compute_c5(best_h), best_c5), len(best_h)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sm_eval(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    return (mc + np.log(e.sum()) / T) * 2.0 / n, e / e.sum(), corr

def sm_grad(h, w, thresh=1e-10):
    n = len(h)

        else:
            p = -k
            g[p:] += wj * (1.0 - h[:n - p])
            g[:n - p] -= wj * h[p:]
    return g * 2.0 / n
