    # Compute a_k and b_k for all shifts k
    corr = get_corrs(h)
    dx = 2.0 / n
    center = n - 1
    j = np.arange(n)
    
    a = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        s = k - center
        if s >= 0:
            # dc[k]/dh[i] = (1-h[i-s])*I(i>=s) - h[i+s]*I(i+s<n)
            val = 0.0
            if i >= s:
                val += (1.0 - h[i - s])
            if i + s < n:
                val -= h[i + s]
            a[k] = val
        else:
            ss = -s
            val = 0.0
            if i + ss < n:
                val += (1.0 - h[i + ss])
            if i >= ss:
                val -= h[i - ss]
            a[k] = val
    
    b = corr - h[i] * a  # constant terms: corr[k] = a[k]*h[i] + b[k]

        # minimum is more likely near 0
        lo, hi = 0.0, 1.0
    else:
        lo, hi = 0.0, 1.0  # we'll bisect regardless
    
    for _ in range(n_bisect):
        mid = (lo + hi) / 2.0
        f_mid = np.max(a * mid + b)
        # Subgradient: min and max of a[k] where a[k]*mid+b[k] = f_mid

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    dx = 2.0 / n
    center = n - 1
    s = shift_idx - center
    j = np.arange(n)
    grad = np.zeros(n)
    if s >= 0:
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] -= h[j[m1] + s]
        grad[m2] += (1.0 - h[j[m2] - s])
    else:
        ss = -s
        m1 = (j + ss) < n
        m2 = j >= ss
        grad[m1] += (1.0 - h[j[m1] + ss])
        grad[m2] -= h[j[m2] - ss]
    return grad * dx

_G_N = 0
_G_T = 1.0

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    return (mc + np.log(np.exp(sh).sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-13:
            continue
        grad += w * shift_grad(x, _G_N, oi)
    return grad * 1.0

def _g_cons_fun(x):
    return x.sum() - _G_N / 2.0

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol, t0, budget_s, margin=30):
    global _G_N, _G_T
    _G_N = n
    _G_T = float(T)
    if time.time() - t0 > budget_s - margin:
        return h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coord_descent_pass(h, n, n_bisect=25, order=None, t0=None, budget_s=None):
    center = n - 1
    if order is None:
        order = np.random.permutation(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_arr = np.arange(2 * n - 1) - center
    for i in order:
        if t0 is not None and time.time() - t0 > budget_s - 10:
            break
        pos = i - s_arr
        neg = i + s_arr
        a = np.zeros(2 * n - 1)
        m_pos = (pos >= 0) & (pos < n)
        m_neg = (neg >= 0) & (neg < n)
        a[m_pos] += (1.0 - h[pos[m_pos]])
        a[m_neg] -= h[neg[m_neg]]
        b = corr - h[i] * a
        lo_t, hi_t = 0.0, 1.0
        for _ in range(n_bisect):
            mid = (lo_t + hi_t) / 2.0
            vals = a * mid + b
            f_mid = vals.max()
            active = np.abs(vals - f_mid) < 1e-10
            if np.any(active):
                min_a = a[active].min()
                max_a = a[active].max()
                if min_a > 1e-12:
                    hi_t = mid
                elif max_a < -1e-12:
                    lo_t = mid
                else:
                    break
            else:
                hi_t = mid
        new_val = (lo_t + hi_t) / 2.0
        h[i] = new_val
        corr = b + new_val * a
    return h

def nesterov_subgrad(h_init, n, t0, budget_s, max_iters=200000):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.5
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = (it / (it + 1.5)) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        g = shift_grad(h_y, n, idx) * dx
        alpha = alpha0 / (1.0 + 0.005 * it)
        h_new = h_y - alpha * g
        if (it + 1) % 20 == 0:
            h_new = project_feasible(h_new, n)
        else:

            if abs(diff) > 1e-10:
                h_new = np.clip(h_new + diff / n, 0.0, 1.0)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 50000 == 0:

            if c5a < c5_best:
                c5_best = c5a
                h_best = project_feasible(h_avg, n).copy()
    return h_best, c5_best
