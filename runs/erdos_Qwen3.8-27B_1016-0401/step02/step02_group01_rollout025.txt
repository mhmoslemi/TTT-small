    return g + 200.0 * (x.sum() - _g_target)

def lbfgs_or_step(h0, target, T, t0, budget):
    global _g_T, _g_target, _g_n
    _g_T = T
    _g_target = target
    _g_n = len(h0)
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        if time.time() - t0 > budget - 30:
            return h0, compute_c5(h0)
        res = minimize(_lbfgs_fun, h0, method='L-BFGS-B', jac=_lbfgs_grad,
                       bounds=bounds, options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12})
        h = project_constraint(res.x, target)
        return h, compute_c5(h)
    except Exception:
        return h0, compute_c5(h0)
