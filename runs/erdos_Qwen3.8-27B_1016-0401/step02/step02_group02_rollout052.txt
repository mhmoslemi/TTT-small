    mc = np.max(corr)
    shifted = T * (corr - mc) - np.max(T * (corr - mc))
    return (mc + np.log(np.sum(np.exp(shifted))) / T)

- The "worst case" is when there are many such (1,0) pairs at the same shift

The minimum possible max of these counts over all shifts... for a "random" binary sequence with 140 ones, the expected count at shift s is approximately 140*(140/280)*(n-s)/n = 70*(n-s)/n. The max over s would be around 70 (at small s). But we can do better with structured sequences.

Actually, the ideal would be a "difference set" type construction where all autocorrelation values are as equal as possible.

For a binary sequence of length N with K ones, the autocorrelation at lag s is the number of positions i where x[i]=x[i+s]=1. We want this to be as constant as possible (≈ K*(K-1)/(N-1) for random).

But our metric is slightly different: we're looking at h[i]*(1-h[i+s]), which is the number of (1,0) pairs at distance s. This equals (number of 1s in positions [0, N-1-s]) - (autocorrelation at lag s).

For a perfectly periodic sequence (alternating 1,0,1,0,...), the autocorrelation at even lags is 0 and at odd lags is K-1 (or similar). This gives oscillating values.

For an ideal "balanced" sequence, we'd want the (1,0) count at each shift to be as close to K*(N-K)/N * (N-s)/(N-1) as possible.

This is related to the theory of "perfect sequences" or "Barker sequences." For our purposes, let me just focus on the optimization.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(x, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0, 1)

def topk_subgrad(corr, h, K, n):
    dx = 2.0 / n
    top = np.argsort(corr)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top:
        o = int(idx) - center
        wi = 1.0 / K
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n-o])
            g[:n-o] -= wi * h[o:]
        else:
            p = -o
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
    return g * dx

def slsqp_smooth(h0, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    center_corr = n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        mc = c.max()
        s = T*(c - mc)
        s -= s.max()
        return (mc + np.log(np.sum(np.exp(s)))/T) * (2.0/n)
    
    def jac(x):
        c = np.correlate(x, 1.0-x, mode='full')
        mc = c.max()
        s = T*(c - mc)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for idx in np.where(w > 1e-12)[0]:
            o = int(idx) - center_corr
            wi = w[idx]
            if o == 0:
                g += wi*(1-2*x)
            elif o > 0:
                g[o:] += wi*(1-x[:n-o])
                g[:n-o] -= wi*x[o:]
            else:
                p = -o
                g[:n-p] += wi*(1-x[p:])
                g[p:] -= wi*x[:n-p]
        return g * (2.0/n)
    
    def cfun(x): return x.sum() - target
    def cjac(x): return np.ones(n)
    
    res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=[{'type':'eq','fun':cfun,'jac':cjac}],
                   options={'maxiter':maxiter, 'ftol':1e-16})
    return project_sum(res.x, target)

Actually, let me try the **outer approximation / linearization approach**:

At each iteration:
1. Compute current corr values
2. The max is at some lag k*. Linearize corr_{k*} at the current point
3. Solve: min sum of top K linearized correlations, s.t. 0≤h≤1, sum(h)=target
4. This is an LP!
5. Update h, repeat

This is the "cutting plane" or "outer approximation" method for bilevel problems.

Wait, but the correlations are quadratic, not linear. However, if I linearize around the current point, I get:
corr_k(h + d) ≈ corr_k(h) + ∇corr_k(h) · d

And ∇corr_k is easy to compute. So the LP at each step is:
min t
s.t. corr_k(h) + ∇corr_k(h) · (h' - h) ≤ t for k in top-K
     0 ≤ h' ≤ 1
     sum(h') = target

This is an LP with n+1 variables and K+1+2n constraints. Very solvable!

import numpy as np
import time
from scipy.optimize import linprog, minimize

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(x, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0, 1)

def corr_and_grad(h, n):
    """Compute correlation and gradient w.r.t. h for all lags."""
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    # grad[k, i] = d(corr[k])/d(h[i])
    # corr[k] = sum_j h[j] * v[j + k - n + 1]
    # d/dh[i]: appears in h[i]*v[i+k-n+1] and h[i-k+n-1]*v[i] (if valid)
    # = v[i+k-n+1] (if i+k-n+1 in [0,n)) + (-h[i-k+n-1]) (if i-k+n+1 in [0,n))
    # Wait, d/dh[i] of h[i]*v[j] where j = i+k-n+1: v[j] = 1-h[j], so d/dh[i] = 1-h[j] if j≠i, or 1-2h[i] if j=i
    # d/dh[i] of h[j]*v[i] where j = i+k-n+1... wait I'm confusing indices.
    
    # Let me use a different approach: for each lag index idx in corr (0 to 2n-2),
    # shift s = idx - (n-1)
    # corr[idx] = sum_{i} h[i] * (1-h[i+s]) where i in [0,n) and i+s in [0,n)
    # d(corr[idx])/d(h[j]) = (1 - h[j+s]) if j+s in [0,n) and j in [0,n)
    #                      + (-h[j-s]) if j-s in [0,n) and j in [0,n)  [from the other term where h[j] is the shifted one]
    # Wait, more carefully:
    # corr[idx] = sum_i h[i] * (1-h[i+s]) for valid i
    # = sum_i h[i] - sum_i h[i]*h[i+s]
    # d/dh[j]: 
    #   From sum h[i]: 1 if j in range
    #   From -sum h[i]*h[i+s]: 
    #     -h[j+s] if j+s in [0,n) [j appears as first factor]
    #     -h[j-s] if j-s in [0,n) [j appears as second factor, i = j-s]
    #     +2h[j] if j+s=j and j-s=j, i.e., s=0 [both terms involve h[j]^2]
    #   
    # Actually for s≠0: d/dh[j] of -h[i]*h[i+s] summed over i:
    #   When i=j: contributes -h[j+s]
    #   When i+s=j, i.e., i=j-s: contributes -h[j-s]
    # For s=0: d/dh[j] of -h[j]^2 = -2h[j]
    
    return corr  # I'll compute grad numerically for simplicity

    def cjac(x): return np.ones(n)

                   options={'maxiter':maxiter, 'ftol':1e-16})
    return project_sum(res.x, target)

def lp_Initialize(h, n, target, K):
    """Solve LP: min t s.t. corr_k_linearized <= t, using top-K lags."""
    corr = np.correlate(h, 1-h, mode='full')
    top = np.argsort(corr)[-K:]
    center = n - 1
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A @ x <= b
    # For each top lag idx: grad_k @ h + h0 - h0_grad... 
    # Linearize: corr_k(h') ≈ corr_k(h) + g_k @ (h' - h) = corr_k(h) - g_k @ h + g_k @ h'
    # We want: corr_k(h) - g_k @ h + g_k @ h' <= t
    # => g_k @ h' - t <= g_k @ h - corr_k(h)  [wrong sign]
    # Actually: corr_k(h') ≈ corr_k(h) + g_k @ (h'-h)
    # corr_k(h) + g_k @ h' - g_k @ h <= t
    # g_k @ h' - t <= g_k @ h - corr_k(h)
    
    A_rows = []
    b_rows = []
    for idx in top:
        o = int(idx) - center
        g_k = np.zeros(n)
        if o == 0:
            g_k[:] = 1.0 - 2.0 * h
        elif o > 0:
            g_k[o:] = 1.0 - h[:n-o]
            g_k[:n-o] -= h[o:]
        else:
            p = -o
            g_k[:n-p] = 1.0 - h[p:]
            g_k[p:] -= h[:n-p]
        # corr_k(h) + g_k @ (h' - h) <= t
        # g_k @ h' - t <= g_k @ h - corr_k(h)
        row = np.zeros(n+1)
        row[:n] = g_k
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(g_k @ h - corr[idx])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h') = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds_lp = [(0, 1)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return res.x[:n]
    return h

import numpy as np
import time
from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(x, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0, 1)

def topk_subgrad(corr, h, K, n):
    dx = 2.0 / n
    top = np.argsort(corr)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top:
        o = int(idx) - center
        wi = 1.0 / K
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

def lp_step(h, n, target, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-K:]
    center = n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows, b_rows = [], []
    for idx in top:
        o = int(idx) - center
        g_k = np.zeros(n)
        if o == 0:
            g_k[:] = 1.0 - 2.0 * h
        elif o > 0:
            g_k[o:] = 1.0 - h[:n - o]
            g_k[:n - o] -= h[o:]
        else:
            p = -o
            g_k[:n - p] = 1.0 - h[p:]
            g_k[p:] -= h[:n - p]
        row = np.zeros(n + 1)
        row[:n] = g_k
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(g_k @ h - corr[idx])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds_lp = [(0, 1)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                  A_eq=A_eq, b_eq=np.array([target]), bounds=bounds_lp, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0, 1)
    return h.copy()

def slsqp_smooth(h0, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    center = n - 1
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = c.max()
        s = T * (c - mc)
        s -= s.max()
        return (mc + np.log(np.sum(np.exp(s))) / T) * dx
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = c.max()
        s = T * (c - mc)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for idx in np.where(w > 1e-12)[0]:
            o = int(idx) - center
            wi = w[idx]
            if o == 0:
                g += wi * (1 - 2 * x)
            elif o > 0:
                g[o:] += wi * (1 - x[:n - o])
                g[:n - o] -= wi * x[o:]
            else:
                p = -o
                g[:n - p] += wi * (1 - x[p:])
                g[p:] -= wi * x[:n - p]
        return g * dx
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def swap_local_search(h_bin, n, target, rng, max_iters, time_limit):
    t_start = time.time()
    c5_cur = compute_c5(h_bin)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for _ in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_try = h_bin.copy()
        h_try[i], h_try[j] = 0.0, 1.0
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-12:
            h_bin = h_try
            c5_cur = c5_try
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        elif rng.random() < 0.15 * np.exp(-(c5_try - c5_cur) * 800):
            h_bin = h_try
            c5_cur = c5_try
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
    return h_bin, c5_cur
