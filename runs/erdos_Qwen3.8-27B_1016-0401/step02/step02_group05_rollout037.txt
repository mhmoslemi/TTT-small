    dx = 2.0 / n

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def fix_constraint(h, target):

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            order = np.argsort(-(1.0 - h))

                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(-h)
            for i in order:

                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

    h_new = h - step * grad
    h_new = fix_constraint(h_new, target)

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

    h = np.clip(h, 0.0, 1.0).copy()

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(-(1.0 - h))
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(-h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def soft_max_jac(x, T):
