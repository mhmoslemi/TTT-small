Actually, for the problem with the constraint sum(h) = n/2 and h ∈ [0,1], the objective is:
f(h) = max_k sum_i h[i](1-h[i+k]) * dx

For a fixed k, the function h ↦ sum_i h[i](1-h[i+k]) is:
= sum_i h[i] - sum_i h[i]h[i+k]

The Hessian of sum_i h[i]h[i+k] w.r.t. h is a banded matrix with 1's on the k-th sub/super-diagonal. So the Hessian of -sum_i h[i]h[i+k] has -1's on those diagonals, making the function concave (negative semidefinite Hessian). The Hessian of sum_i h[i] is 0.

So each f_k is concave! And the max of concave functions is not necessarily concave. So the problem is NOT convex.

However, each individual f_k is concave, which means minimizing f_k over the polytope has the property that a local minimum is at a vertex (0-1 point). The global minimum of max_k f_k might not be at a vertex, but it's worth trying vertices.

For n=280, trying all 0-1 patterns is impossible (2^280), but we can do local search:
- Start from a 0-1 pattern
- Try flipping one bit at a time (changing 0 to 1 and 1 to 0, maintaining sum = n/2)
- Accept if C5 improves
- This is like a 2-opt local search

With n=280, there are 140 ones and 140 zeros. A "move" is: pick a 1 and a 0, swap them (i.e., flip both). That's 140*140 = 19,600 possible moves per iteration. Each evaluation is O(n) for the correlation update (if we update incrementally) or O(n) for full recompute.

Actually, full recompute of n-point correlation is O(n^2) = 78,400 operations. With 19,600 moves, that's 1.5 billion operations per full sweep. Too slow in Python.

But we can be smarter: when we flip positions i and j, only the correlations involving i or j change. For a flip at position i, the correlation at lag k changes by:
- For each lag k where i and i+k are both valid: the change in corr(k) depends on the new values.

This is still O(n) per flip, so 19,600 * 280 = 5.5M per sweep. Doable but might be slow in Python.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project_box_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smax_obj(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    v = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    v += pen * (h.sum() - n * 0.5) ** 2
    return v

def smax_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / np.sum(e)
    center = n - 1
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        wi = w[center + o]
        if wi < 1e-13:
            continue
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    g *= dx
    if pen > 0:
        g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def lbfgsb_run(h0, n, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smax_obj(x, T, pen)
    def g(x):
        return smax_grad(x, T, pen)
    try:
        res = minimize(f, h0.copy(), jac=g, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-13})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        return h0.copy()

def fista_run(h0, n, T, pen, max_iter, step):
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        gd = smax_grad(y, T, pen)
        zh = project_box_sum(y - step * gd, n / 2.0)
        if np.linalg.norm(zh - h) < 1e-14:
            break
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        h = zh
    return h

def fourier_perturb(h, rng, n_freqs=5):
    n = len(h)
    x = np.arange(n)
    noise = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        noise += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return h + noise

def block_perturb(h, rng):
    n = len(h)
    h2 = h.copy()
    bsize = max(3, n // 20)
    for _ in range(rng.integers(2, 6)):
        start = rng.integers(0, n - bsize)
        end = start + bsize
        shift = rng.integers(1, bsize)
        h2[start:end], h2[end - shift:end], h2[start:end - shift] = \
            h2[start:end - shift], h2[start:end], h2[end - shift:end]
    return project_box_sum(h2, n / 2.0)

def try_resolution(h_best, n_try, target_c5, elapsed, budget_s, rng):
    n_old = len(h_best)
    if n_try == n_old:
        return h_best, target_c5
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
    h_try = np.interp(x_new, x_old, h_best)
    t_try = n_try / 2.0
    h_try = project_box_sum(h_try, t_try)
    c5_try = compute_c5(h_try)
    best_h, best_c5 = h_try, c5_try
    for T in [50, 200, 800, 2000]:
        if elapsed() > budget_s - 15:
            break
        h_t = lbfgsb_run(best_h, n_try, T, 5.0, 1500)
        h_t = project_box_sum(h_t, t_try)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
    if best_c5 < target_c5:
        return best_h, best_c5
    return h_best, target_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_box_sum(h_best, n / 2.0)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine at current n with progressive T
    T_schedule = [100, 200, 400, 800, 1600, 3000, 5000]
    for T in T_schedule:
        if elapsed() > budget_s - 350:
            break
        for pen in [5.0, 20.0]:
            h_new = lbfgsb_run(h_best, n, T, pen, 2500)
            h_new = project_box_sum(h_new, n / 2.0)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 2: FISTA refinement
    for T in [300, 1000, 3000]:
        if elapsed() > budget_s - 250:

        step = min(0.3, 2.5 / (T + 1.0))
        h_new = fista_run(h_best, n, T, 10.0, 5000, step)
        h_new = project_box_sum(h_new, n / 2.0)
        c5 = compute_c5(h_new)

            h_best = h_new

    # Phase 3: Diversified perturbations + re-optimization
    n_pert = 0
    while elapsed() < budget_s - 200:

        r = rng.random()
        if r < 0.3:

            h_pert = h_best + rng.normal(0, scale, n)

        else:

        for T in [2000, 5000, 10000]:
            if elapsed() > budget_s - 3:

            h_new = lbfgsb_run(h_best, n, T, 1.0, 3000)
            h_new = project_box_sum(h_new, n / 2.0)
            c5 = compute_c5(h_new)
            if c5 < c5_best:

        # Fix count
        n_ones = int(h_bin.sum())
        target_ones = n // 2
        if n_ones > target_ones:
            idx_frac = np.where(h_bin == 1.0)[0]
            order = np.argsort(h_best[idx_frac])
            for i in range(n_ones - target_ones):
                h_bin[idx_frac[order[i]]] = 0.0
        elif n_ones < target_ones:

            order = np.argsort(-h_best[idx_frac])
            for i in range(target_ones - n_ones):
                h_bin[idx_frac[order[i]]] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
