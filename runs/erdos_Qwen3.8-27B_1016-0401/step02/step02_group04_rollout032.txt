import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    step = step_init
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = soft_max_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def subgrad_cycling(h, target, max_iter, n_top=5):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argpartition(corr, -n_top)[-n_top:]
        grad = np.zeros(n)
        for o_idx in k_max:
            o = o_idx - (n - 1)
            if o == 0:
                grad += (1.0 - 2.0 * h)
            elif o > 0:
                grad[o:] += (1.0 - h[:n - o])
                grad[:n - o] -= h[o:]
            else:
                p = -o
                grad[:n - p] += (1.0 - h[p:])
                grad[p:] -= h[:n - p]
        grad *= dx / max(1, n_top)
        step = 0.8 / (it + 20)
        h_new = project_box_sum(h - step * grad, target)
        if np.linalg.norm(h_new - h) < 1e-14:
            break
        h = h_new
    return h

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def local_swap_search(h, target, max_rounds=200, rng=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for i in order:
            neighbor = (i + 1) % n
            if h[i] < h[neighbor]:
                old = (h[i], h[neighbor])
                h[i], h[neighbor] = h[neighbor], h[i]
                c5_new = compute_c5(h)
                if c5_new < c5:
                    c5 = c5_new
                    improved = True
                else:
                    h[i], h[neighbor] = old
        if not improved:
            break
    return h, c5

def flip_search(h, target, n_trials=5000, rng=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(n_trials):
        i = rng.integers(0, n)
        old_v = h[i]
        h[i] = 1.0 if old_v < 0.5 else 0.0
        h = project_box_sum(h, target)
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            h[i] = old_v
            h = project_box_sum(h, target)
    return h, c5

    n4 = 70
    h4 = np.zeros(n4)
    for i in range(n4):
        x = 0.0
        d = 1.0
        v = i + 1
        while v > 0:
            d /= 2.0
            x += d * (v & 1)
            v >>= 1
        h4[i] = 1.0 if x < 0.5 else 0.0
    starts.append(project_box_sum(h4, n4 / 2.0))
    h_rand = (rng.random(70) < 0.5).astype(float)
    starts.append(project_box_sum(h_rand, 35.0))
    x6 = np.linspace(0, 2, 70, endpoint=False)
    h6 = (np.cos(2 * np.pi * x6 * 3) > 0).astype(float)
    starts.append(project_box_sum(h6, 35.0))
    return starts

def optimize_one(h_s, target, t0, budget_s, rng):
    n = len(h_s)
    h = h_s.copy()
    c5 = compute_c5(h)
    remaining = budget_s - (time.time() - t0)
    if remaining < 20:
        return h, c5
    t_schedule = [400, 150, 60, 25, 10]
    for T in t_schedule:
        if time.time() - t0 > budget_s - 30:
            break
        max_it = max(500, int(800 * (400 / T + 1)))
        h_new = lbfgsb_optimize(h, n, target, T, 5.0, min(max_it, 3000))
        h_new = project_box_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
    if time.time() - t0 < budget_s - 20:
        h_sg = subgrad_cycling(h, target, 800, n_top=5)
        c5_sg = compute_c5(h_sg)
        if c5_sg < c5:
            h, c5 = h_sg, c5_sg
    if time.time() - t0 < budget_s - 10:
        h_f = fista_optimize(h, target, 200, 1500, 0.1)
        c5_f = compute_c5(h_f)
        if c5_f < c5:
            h, c5 = h_f, c5_f
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):

    have_parent = 'initial_h_values' in globals() and initial_h_values is not None

    # Final L-BFGS-B polish
    if time.time() - t0 < budget_s - 5:
        for T in [15, 8, 4]:
            if time.time() - t0 > budget_s - 3:

            h_new = lbfgsb_optimize(h_best, n_b, tgt_b, T, 1.0, 2000)

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
