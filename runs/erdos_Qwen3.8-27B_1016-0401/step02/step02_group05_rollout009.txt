import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 0:
                break
            frac = min(1.0, diff / s)
            h += room * frac
        else:
            s = h.sum()
            if s <= 0:
                break
            frac = min(1.0, -diff / s)
            h -= h * frac
    return np.clip(h, 0, 1)

def lse_obj(x, T, mu, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    se = np.sum(e)
    f = 1.0 - 2.0 * x
    grad_corr = np.zeros(n)
    for j in range(n):
        grad_corr[j] = np.dot(e[j:j+n], f)
    grad_lse = grad_corr / (T * se)
    cdiff = x.sum() - target
    obj = (mc + np.log(se) / T) * dx + mu * cdiff * cdiff
    grad = grad_lse * dx + 2.0 * mu * cdiff
    return obj, grad

                   method='L-BFGS-B', jac=lambda x: lse_obj(x, T, mu, target)[1],
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-14})
    return res.x

def treewidth_descent(h, target, n_steps, step_size, t0, budget):
    n = len(h)
    h_c = h.copy()
    corr = get_corr(h_c)
    best_c5 = float(np.max(corr) * 2.0 / n)
    for _ in range(n_steps):
        if time.time() - t0 > budget:
            break
        k_star = int(np.argmax(corr))
        f = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                f[j] = 1.0 - 2.0 * h_c[kj]
        f = f - f.mean()
        h_try = h_c - step_size * f
        h_try = fix_sum(h_try, target)
        corr_try = get_corr(h_try)
        c5_try = float(np.max(corr_try) * 2.0 / n)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            h_c = h_try
            corr = corr_try
    return h_c, best_c5

def local_swap(h, target, n_iter, t0, budget):
    n = len(h)
    h_c = h.copy()
    corr = get_corr(h_c)
    best_c5 = float(np.max(corr) * 2.0 / n)
    stall = 0
    delta = 0.1
    for _ in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 200:
            delta *= 0.6
            stall = 0
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        if p == 0:
            prod = h_c * (1.0 - h_c)
            idx = int(np.argmax(prod))
            h_c[idx] += -delta if h_c[idx] > 0.5 else delta
        else:
            if s > 0:
                prod = h_c[p:] * (1.0 - h_c[:n-p])
                idx = int(np.argmax(prod))
                h_c[idx+p] -= delta
                h_c[idx] += delta
            else:
                prod = h_c[:n-p] * (1.0 - h_c[p:])
                idx = int(np.argmax(prod))
                h_c[idx] -= delta
                h_c[idx+p] += delta
        h_c = np.clip(h_c, 0, 1)
        h_c = fix_sum(h_c, target)
        corr = get_corr(h_c)
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
    return h_c, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with penalty, annealing T
    temps_mus = [(80,50),(50,50),(30,100),(20,100),(12,200),(8,300),(5,500),(3,500),(2,800)]
    for T, mu in temps_mus:
        if time.time() - t0 > budget_s - 350:
            break
        h_ref = lbfgs_step(h_best, target, T, mu, max_iter=200)
        h_ref = fix_sum(h_ref, target)
        c5 = compute_c5(h_ref)
        if c5 < c5_best - 1e-13:
            c5_best = c5
            h_best = h_ref

    # Phase 2: Treewidth descent
    if time.time() - t0 < budget_s - 250:
        h_tw, c5_tw = treewidth_descent(h_best, target, 800, 0.015, t0, budget_s - 240)
        if c5_tw < c5_best:
            c5_best = c5_tw
            h_best = h_tw

    # Phase 3: Local swap search
    if time.time() - t0 < budget_s - 180:
        h_ls, c5_ls = local_swap(h_best, target, 5000, t0, budget_s - 170)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls

    # Phase 4: Iterated local search with perturbations
    round_t0 = time.time()
    rcount = 0
    while time.time() - t0 < budget_s - 120:
        rcount += 1
        h_pert = h_best.copy()
        strategy = rcount % 5
        if strategy == 0:
            idx = rng.choice(n, size=max(1, n//10), replace=False)
            h_pert[idx] = rng.random(len(idx))
        elif strategy == 1:
            h_pert = h_best + rng.normal(0, 0.06, n)
        elif strategy == 2:
            start = int(rng.integers(0, n - 20))
            ln = int(rng.integers(10, 30))
            h_pert[start:start+ln] = 1.0 - h_best[start:start+ln]
        elif strategy == 3:
            for _ in range(int(rng.integers(2, 5))):
                st = int(rng.integers(0, n - 8))
                h_pert[st:st+8] = 1.0 - h_best[st:st+8]
        else:
            h_pert = h_best + rng.normal(0, 0.03, n)
        h_pert = fix_sum(h_pert, target)
        h_ref = lbfgs_step(h_pert, target, 15, 300, max_iter=150)
        h_ref = fix_sum(h_ref, target)
        h_ref2, c5r2 = treewidth_descent(h_ref, target, 200, 0.01, t0, budget_s - 110)
        h_ls2, c5_ls2 = local_swap(h_ref2, target, 1500, t0, budget_s - 110)
        if c5_ls2 < c5_best - 1e-13:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 5: Final fine-tuning
    for T, mu in [(4, 800), (2, 1500), (1.5, 2000), (1, 3000)]:
        if time.time() - t0 > budget_s - 40:
            break
        h_ref = lbfgs_step(h_best, target, T, mu, max_iter=400)
        h_ref = fix_sum(h_ref, target)
        c5 = compute_c5(h_ref)
        if c5 < c5_best - 1e-14:
            c5_best = c5
            h_best = h_ref

    # Final local search
    if time.time() - t0 < budget_s - 3:
        h_ls3, c5_ls3 = local_swap(h_best, target, 3000, t0, budget_s - 2)
        if c5_ls3 < c5_best:
            c5_best = c5_ls3
            h_best = h_ls3

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
