import numpy as np
from scipy.optimize import minimize
import time

_G_T = 100.0
_G_TARGET = 0.0
_G_N = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_eval(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-12:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def pg_optimize(h, target, T, n_iter, lr0, deadline, momentum=0.0):
    h = h.copy()
    n = len(h)
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, sw = sm_eval(h, T)
        grad = sm_grad(h, sw)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        eff_grad = momentum * v + (1.0 - momentum) * grad
        v = eff_grad
        found = False
        lr_try = lr
        for _ in range(50):
            h_new = project_constraint(h - lr_try * eff_grad, target)
            obj_new, _ = sm_eval(h_new, T)
            if obj_new < obj - 1e-16:
                found = True
                break
            lr_try *= 0.5
        if not found:
            lr = min(lr * 0.8, lr0)
            continue
        h = h_new
        lr = min(lr_try * 1.06, lr0)
        if it % 300 == 299:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _scl_fun(x):
    return x.sum() - _G_TARGET

def _scl_jac(x):
    return np.ones(_G_N)

def _scl_obj(x):
    return sm_eval(x, _G_T)[0]

def _scl_grad(x):
    _, sw = sm_eval(x, _G_T)
    return sm_grad(x, sw)

def slsqp_optimize(h0, target, T, maxiter, deadline):
    global _G_T, _G_TARGET, _G_N
    _G_T = T
    _G_TARGET = target
    _G_N = len(h0)
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _scl_fun, 'jac': _scl_jac}]
    try:
        res = minimize(_scl_obj, h0, method='SLSQP', jac=_scl_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n0 = len(h_best)
    target0 = n0 / 2.0
    h_best = project_constraint(h_best, target0)
    c5_best = compute_c5(h_best)

    # Phase 1: Anneal at original resolution
    for T in [30, 80, 200, 500, 1200]:
        if time.time() > budget - 450:
            break
        h_best, c5 = pg_optimize(h_best, target0, T, 3000, 0.05, budget - 400)
        if c5 < c5_best:
            c5_best = c5

    # Phase 2: SLSQP at original resolution
    for T in [800, 2000]:
        if time.time() > budget - 300:
            break
        h_t, c5 = slsqp_optimize(h_best, target0, T, 350, budget - 250)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_t

    # Phase 3: Upsample to 840
    n1 = 840
    target1 = n1 / 2.0
    x_old = np.arange(n0) * (2.0 / n0)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    h1 = np.interp(x1, x_old, h_best)
    h1 = project_constraint(h1, target1)
    c5_1 = compute_c5(h1)
    if c5_1 < c5_best:
        c5_best = c5_1
        h_best = h1.copy()
    else:
        h_best = h1.copy()

    # Phase 4: Anneal at n=840 with momentum
    T_schedule = [50, 120, 300, 700, 1500, 3000, 6000]
    for i, T in enumerate(T_schedule):
        if time.time() > budget - 80:
            break
        n_it = 3000 if T < 500 else 2000
        h1, c5 = pg_optimize(h_best, target1, T, n_it, 0.04, budget - 50,
                             momentum=0.1 if T > 500 else 0.0)
        if c5 < c5_best:
            c5_best = c5
            h_best = h1.copy()

    # Phase 5: Restart with perturbations
    for r in range(14):
        if time.time() > budget - 60:
            break
        h_p = h_best.copy()
        kind = r % 5
        if kind == 0:
            idx = rng.integers(0, n1)
            sz = rng.integers(5, 20)
            for i in range(sz):
                h_p[(idx + i) % n1] = 1.0 - h_p[(idx + i) % n1]
        elif kind == 1:
            h_p = np.clip(h_best + rng.normal(0, 0.03, n1), 0, 1)
        elif kind == 2:
            h_p = (h_best > rng.uniform(0.35, 0.65)).astype(float)
        elif kind == 3:
            shift = rng.integers(2, 15)
            h_p = np.roll(h_best, shift)
        else:
            h_p = np.clip(h_best + 0.02 * rng.standard_t(4, n1), 0, 1)
        h_p = project_constraint(h_p, target1)
        T_p = 300 + r * 80
        h_p, c5_p = pg_optimize(h_p, target1, T_p, 1000, 0.03, budget - 20,
                                momentum=0.1)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()

    # Phase 6: Very high-T fine refinement
    if time.time() < budget - 25:
        h_best, c5 = pg_optimize(h_best, target1, 12000, 3000, 0.015,
                                 budget - 10, momentum=0.15)
        if c5 < c5_best:
            c5_best = c5

    # Phase 7: Final SLSQP
    if time.time() < budget - 8:
        h_t, c5 = slsqp_optimize(h_best, target1, 5000, 250, budget - 2)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_t

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
