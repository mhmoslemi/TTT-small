        else:
            m = h > 1e-12

    dx = 2.0 / n
    corr = get_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    max_s = np.max(shifted)
    e = np.exp(shifted - max_s)
    return (mc - max_s / T + np.log(np.sum(e)) / T) * dx

def sm_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = get_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    max_s = np.max(shifted)
    e = np.exp(shifted - max_s)
    s = np.sum(e)
    sw = e / s
    grad = np.zeros(n)
    center = n - 1

        if time.time() - t0 > budget_s - 500:
            break
        h_p, c5_p = pgd_momentum(h_best, target, T, 4000, t0, budget_s - 500, 0.02)
        if c5_p < c5_best - 1e-13:
            c5_best = c5_p
            h_best = h_p

    # Phase 2: SLSQP soft-max at various T
    for T in [40, 25, 15, 10, 7, 5, 3.5, 2.5, 2, 1.5, 1.2]:
        if time.time() - t0 > budget_s - 350:
            break
        h_s, c5_s = sla_slsqp(h_best, target, T, bounds, t0, budget_s - 350)
        if c5_s < c5_best - 1e-13:
            c5_best = c5_s
            h_best = h_s

    # Phase 3: Alternating multi-shift and 2-opt local search
    for _ in range(6):
        if time.time() - t0 > budget_s - 200:
            break
        h_m, c5_m = multi_opt(h_best, target, 1500, t0, budget_s - 200, 3)
        if c5_m < c5_best:
            c5_best = c5_m
            h_best = h_m
        h_2, c5_2 = two_opt(h_best, target, 1500, t0, budget_s - 200)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h_2

    # Phase 4: Simulated annealing with diverse moves
    if time.time() - t0 < budget_s - 120:
        temp = 0.05
        h_sa = h_best.copy()
        c5_sa = c5_best
        n_inner = n
        for it in range(3000):
            if time.time() - t0 > budget_s - 120:
                break
            temp = max(0.001, temp * 0.998)
            r = rng.random()
            h_try = h_sa.copy()
            if r < 0.4:
                # Pair move
                j1 = int(rng.integers(n_inner))
                j2 = int(rng.integers(n_inner))
                if j1 != j2:
                    d = float(rng.uniform(0.02, 0.12))
                    if h_sa[j1] > 0.5:
                        h_try[j1] -= d
                        h_try[j2] += d
                    else:
                        h_try[j2] -= d
                        h_try[j1] += d
            elif r < 0.7:
                # Triple move
                d = float(rng.uniform(0.02, 0.08))
                for _ in range(3):
                    j1 = int(rng.integers(n_inner))
                    j2 = int(rng.integers(n_inner))
                    if j1 != j2 and h_try[j1] > 0.5:
                        h_try[j1] -= d
                        h_try[j2] += d
            elif r < 0.85:
                # Block flip
                ln = int(rng.integers(3, max(4, n_inner // 15)))
                st = int(rng.integers(0, n_inner - ln))
                h_try[st:st + ln] = 1.0 - h_sa[st:st + ln]
            else:
                # Gaussian noise
                h_try = h_sa + rng.normal(0, temp, n_inner)
            h_try = repair(np.clip(h_try, 0, 1), target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_sa + temp * 0.01:
                h_sa = h_try
                c5_sa = c5_try
                if c5_sa < c5_best:
                    c5_best = c5_sa
                    h_best = h_sa.copy()

    # Phase 5: Final SLSQP refinement
    for T in [1.5, 1.0, 0.7, 0.5]:
        if time.time() - t0 > budget_s - 20:
            break
        h_s, c5_s = sla_slsqp(h_best, target, T, bounds, t0, budget_s - 20)
        if c5_s < c5_best - 1e-14:
            c5_best = c5_s
            h_best = h_s

    # Final local search
    h_f, c5_f = two_opt(h_best, target, 1000, t0, budget_s - 3)
    if c5_f < c5_best:
        c5_best = c5_f
        h_best = h_f
    h_m2, c5_m2 = multi_opt(h_best, target, 1000, t0, budget_s - 1, 4)
    if c5_m2 < c5_best:
        c5_best = c5_m2
        h_best = h_m2

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
