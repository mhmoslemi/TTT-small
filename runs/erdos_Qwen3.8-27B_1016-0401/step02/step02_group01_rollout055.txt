    c5 = float(np.max(corr) * dx)
    cmax = np.max(corr)
    eps = max(1e-10, cmax * 1e-6)
    active = np.where(corr >= cmax - eps)[0]
    c_center = n - 1
    grad = np.zeros(n)
    for j in active:
        k = j - c_center
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            p = -k
            g = np.zeros(n)
            g[:n-p] -= h[p:]
            g[p:] += (1.0 - h[:n-p])
        grad += g
    grad /= len(active)
    grad *= dx
    return c5, grad

        else:

    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)

    obj = (mc + np.log(se) / T) * (2.0 / n)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    cmax = np.max(corr)
    eps = max(1e-10, cmax * 1e-6)
    active = np.where(corr >= cmax - eps)[0]
    c_center = n - 1
    grad = np.zeros(n)
    for j in active:
        k = j - c_center
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            p = -k
            g = np.zeros(n)
            g[:n - p] -= h[p:]
            g[p:] += (1.0 - h[:n - p])
        grad += g
    grad /= len(active)
    grad *= dx
    return c5, grad

def project_constraint(h, target):
    lo, hi = -1.0, 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

        else:

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)

    return obj, grad

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    cmax = np.max(corr)
    eps = max(1e-10, cmax * 1e-6)
    active = np.where(corr >= cmax - eps)[0]
    cc = n - 1
    grad = np.zeros(n)
    for j in active:
        k = j - cc
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g = np.zeros(n)
            g[k:] += 1.0 - h[:n-k]
            g[:n-k] -= h[k:]
        else:
            p = -k
            g = np.zeros(n)
            g[:n-p] -= h[p:]
            g[p:] += 1.0 - h[:n-p]
        grad += g
    grad /= len(active)
    grad *= dx
    return c5, grad

def project_constraint(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    grad = np.zeros(n)

            grad[:n-k] -= w * h[k:]
        else:
            p = -k
            grad[:n-p] -= w * h[p:]
            grad[p:] += w * (1.0 - h[:n-p])
    grad *= (2.0 / n)
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    cmax = np.max(corr)
    eps = max(1e-10, cmax * 1e-6)
    active = np.where(corr >= cmax - eps)[0]
    cc = n - 1
    grad = np.zeros(n)
    for j in active:
        k = j - cc
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g = np.zeros(n)
            g[k:] += 1.0 - h[:n-k]
            g[:n-k] -= h[k:]
        else:
            p = -k
            g = np.zeros(n)
            g[:n-p] -= h[p:]
            g[p:] += 1.0 - h[:n-p]
        grad += g
    grad /= len(active)
    grad *= dx
    return c5, grad

def project_constraint(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    grad = np.zeros(n)
    cc = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - cc
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            p = -k
            grad[:n-p] -= w * h[p:]
            grad[p:] += w * (1.0 - h[:n-p])
    grad *= (2.0 / n)
    return obj, grad

            break
        ok = False

            cv = compute_c5(c)

                ok = True
                break

                h = c

                ok = True

        if lr < 1e-8:
            break
    cv = compute_c5(h)

    return bh, bc
