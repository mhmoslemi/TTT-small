            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 1e-15:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 1e-15:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def corr_all(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_obj(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = corr_all(h) * dx
    mc = c.max()
    s = T * (c - mc)
    e = np.exp(s)
    val = mc + np.log(e.sum()) / T
    se = h.sum() - n/2.0
    val += pen * se * se
    return val

def soft_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = corr_all(h) * dx
    mc = c.max()
    s = T * (c - mc)
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o]
        if wi < 1e-15:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] += -wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad *= dx
    se = h.sum() - n/2.0
    grad += 2.0 * pen * se
    return grad

def single_lag_obj(h, lag, pen):
    n = len(h)
    dx = 2.0 / n
    c = corr_all(h)[lag] * dx
    se = h.sum() - n/2.0
    return c + pen * se * se

def single_lag_grad(h, lag, pen):
    n = len(h)
    dx = 2.0 / n
    o = lag - (n-1)
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * h)
    elif o > 0:
        g[o:] += (1.0 - h[:n-o])
        g[:n-o] += -h[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - h[p:])
        g[p:] += -h[:n-p]
    g *= dx
    se = h.sum() - n/2.0
    g += 2.0 * pen * se
    return g

def pair_exchange(h, T, n_rounds=5, step_frac=0.1):
    """Pair-exchange coordinate descent using softmax-weighted gradient."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    center = n - 1
    
    for rnd in range(n_rounds):
        c = corr_all(h) * dx
        mc = c.max()
        s = T * (c - mc)
        e = np.exp(s)
        w = e / e.sum()
        
        grad = np.zeros(n)
        for o in range(-(n-1), n):
            wi = w[center + o]
            if wi < 1e-14:
                continue
            if o == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif o > 0:
                grad[o:] += wi * (1.0 - h[:n-o])
                grad[:n-o] += -wi * h[o:]
            else:
                p = -o
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
        grad *= dx
        
        # Sort by gradient
        order = np.argsort(grad)
        # sources: high gradient (want to decrease)
        # sinks: low gradient (want to increase)
        n_move = max(1, n // 20)
        step = step_frac * dx  # small step
        
        for _ in range(n_move):
            for src in order[-n_move:]:  # highest gradient
                if h[src] < 1e-12:
                    continue
                for sink in order[:n_move]:  # lowest gradient
                    if h[sink] > 1.0 - 1e-12:
                        continue
                    # Try moving from src to sink
                    delta = min(step, h[src], 1.0 - h[sink])
                    if delta < 1e-14:
                        continue
                    h[src] -= delta
                    h[sink] += delta
                    break
                # After each move, recompute would be expensive; 
                # just continue with same gradient
    return h

            h_pe = pair_exchange(h_best, T=300, n_rounds=30, step_frac=0.02)
            c5_pe = compute_c5(h_pe)
            if c5_pe < c5_best - 1e-14:
                h_best = h_pe
                c5_best = c5_pe
                improved = True
            if not improved and outer > 20:
                break

    # Phase 3: Adaptive pair exchange refinement (15% budget)
    p3_end = t0 + budget_s * 0.70
    T_prog = [50, 200, 500, 1000, 2000]
    step_prog = [0.05, 0.02, 0.01, 0.005, 0.002]
    for Ti, si in zip(T_prog, step_prog):
        if elapsed() > p3_end:
            break
        for _ in range(3):
            if elapsed() > p3_end:
                break
            h_pe = pair_exchange(h_best, T=Ti, n_rounds=80, step_frac=si)
            c5_pe = compute_c5(h_pe)
            if c5_pe < c5_best - 1e-15:
                h_best = h_pe
                c5_best = c5_pe
            else:
                break

    # Phase 4: Multi-n exploration (15% budget)
    p4_end = t0 + budget_s * 0.85
    for n_alt in [252, 308, 364, 420, 560]:
        if elapsed() > p4_end:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bnd_alt = [(0.0, 1.0)] * n_alt
        _TGT = target_alt
        for T in [100, 400]:
            if elapsed() > p4_end:
                break
            _GT = T
            _PEN = 2000.0
            try:
                res = minimize(soft_obj, h_alt, jac=soft_grad, method='L-BFGS-B',
                              bounds=bnd_alt, options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    h_alt = h_new
                    c5_alt = c5_new
            except Exception:
                pass
        h_alt = pair_exchange(h_alt, T=300, n_rounds=40, step_frac=0.02)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = target_alt
            bounds = bnd_alt
        _TGT = target

    # Phase 5: Final ultra-fine polish (10% budget)
    p5_end = t0 + budget_s - 5.0
    n_f = len(h_best)
    _TGT = n_f / 2.0
    bounds_f = [(0.0, 1.0)] * n_f
    for T in [5000, 10000, 20000]:
        if elapsed() > p5_end:
            break
        _GT = T
        _PEN = 1e5
        try:
            res = minimize(soft_obj, h_best, jac=soft_grad, method='L-BFGS-B',
                          bounds=bounds_f, options={'maxiter': 10000, 'ftol': 1e-22, 'gtol': 1e-18, 'maxfun': 150000})
            h_new = fix_constraint(np.clip(res.x, 0, 1), n_f/2.0)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        except Exception:
            pass

    h_pe = pair_exchange(h_best, T=1000, n_rounds=200, step_frac=0.001)
    c5_pe = compute_c5(h_pe)
    if c5_pe < c5_best:
        h_best = h_pe
        c5_best = c5_pe

    h_final = fix_constraint(h_best, len(h_best)/2.0)
    c5_final = compute_c5(h_final)
    c5_final = min(c5_final, c5_best)
    return (h_final, c5_final, len(h_final))
