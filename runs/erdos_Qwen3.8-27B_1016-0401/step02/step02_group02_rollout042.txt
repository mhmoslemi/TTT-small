def project_feasible(x, target):

import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(idx, h, n):
    o = int(idx) - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += (1.0 - h[:n - o])
        g[:n - o] -= h[o:]
    else:
        p = -o
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def slsqp_opt(h, n, target, T, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = sh - np.max(sh)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = sh - np.max(sh)
        e = np.exp(sh)
        w = e / e.sum()
        g = np.zeros(n)
        center = n - 1
        nz = np.where(w > 1e-9)[0]
        for idx in nz:
            o = int(idx) - center
            if o == 0:
                g += w[idx] * (1.0 - 2.0 * x)
            elif o > 0:
                g[o:] += w[idx] * (1.0 - x[:n - o])
                g[:n - o] -= w[idx] * x[o:]
            else:
                p = -o
                g[:n - p] += w[idx] * (1.0 - x[p:])
                g[p:] -= w[idx] * x[:n - p]
        return g * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def momentum_loop(h, n, target, n_iters, K, lr0, mom):
    best = h.copy()
    best_c5 = compute_c5(h)
    h_c = h.copy()
    vel = np.zeros(n)
    for it in range(n_iters):
        h_look = project_feasible(h_c + mom * vel, target)
        corr = compute_corr(h_look)
        top = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for idx in top:
            g += grad_lag(idx, h_look, n)
        g /= K
        vel = mom * vel - lr0 * g
        h_new = project_feasible(h_c + vel, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        h_c = h_new
    return best, best_c5
