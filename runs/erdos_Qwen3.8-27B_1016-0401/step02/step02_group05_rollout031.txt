import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def grad_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        p = -s
        g[p:] += 1.0 - h[:n-p]
        g[:n-p] -= h[p:]
    return g

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def multi_lag_gradient(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Get top k lags
    top_idx = np.argsort(corr)[-top_k:]
    # Weights: softmax-like
    top_vals = corr[top_idx]
    weights = np.exp(top_vals / (top_vals.max() + 1e-10))
    weights /= weights.sum()
    
    grad_total = np.zeros(n)
    center = n - 1
    for w, ki in zip(weights, top_idx):
        s = ki - center
        grad_total += w * grad_lag(h, s)
    return grad_total * dx, float(corr.max() * dx)

    best_c5, _ = compute_c5(h_curr)

        g, c5_curr = multi_lag_gradient(h_curr, top_k)

        step = lr
        improved = False

            best_h = h_curr.copy()
            lr = min(lr * 1.05, 0.1)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T, target, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    pen = lam * (x.sum() - target) ** 2
    return val * dx + pen

def soft_max_jac(x, T, target, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] -= w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    grad += 2.0 * lam * (x.sum() - target)
    return grad * dx

    for _ in range(200):

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:

        if not mask.any():
            break
        step = diff / mask.sum()
        # But we can't exceed bounds
        if diff > 0:
            avail = (1.0 - h[mask])
            max_step = avail.max()
            if step > max_step:
                # Distribute to those with most room
                order = np.argsort(-avail)
                remaining = diff
                for idx in order:
                    s = min(remaining, 1.0 - h[idx])
                    h[idx] += s
                    remaining -= s
                    if remaining < 1e-13:
                        break
                continue
        else:
            avail = h[mask]
            max_step = avail.max()
            if -step > max_step:
                order = np.argsort(-avail)
                remaining = -diff
                for idx in order:
                    s = min(remaining, h[idx])
                    h[idx] -= s
                    remaining -= s
                    if remaining < 1e-13:
                        break
                continue
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

        if stall > 500:

        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        
        h_try = h_curr.copy()
        
        if p == 0:
            # h[i]*(1-h[i]) is maximized at h[i]=0.5
            vals = h_curr * (1 - h_curr)
            # Find where to increase/decrease
            if h_curr[vals.argmax()] > 0.5:
                h_try[vals.argmax()] -= 0.05
            else:
                h_try[vals.argmax()] += 0.05
        else:
            if s > 0:

            else:
                contrib = h_curr[:n - p] * (1 - h_curr[p:])
