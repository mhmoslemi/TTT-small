```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad_h(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] -= wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx

def slsqp_optimize(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj(np.correlate(x, 1.0-x, mode="full"), T, n)
    def jac(x):
        return lse_grad_h(np.correlate(x, 1.0-x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def binary_best_swap(h):
    """One pass of best-improvement binary swap. Returns (h, improved)."""
    n = len(h)
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0

        zeros = np.where(h == 0)[0]
        diff = s - n // 2
        if diff > 0:
            for i in range(int(diff)):
                h[ones[i]] = 0
        elif diff < 0:
            for i in range(int(-diff)):
                h[zeros[i]] = 1
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    c5_c = np.max(np.correlate(h, 1.0-h, mode="full"))
    
    best_improve = 0

            h_trial[i] = 0.0
            h_trial[j] = 1.0
            c_trial = np.max(np.correlate(h_trial, 1.0-h_trial, mode="full"))
            diff = c5_c - c_trial
            if diff > best_improve:

                best_pair = (i, j)

        h[i] = 0.0
        h[j] = 1.0

def subgrad_topk(h, n, target, n_steps, K=3):
    """Subgradient descent focusing on top-K correlations."""
    dx = 2.0 / n
    c5_local = compute_c5(h)
    for it in range(n_steps):

        top_idx = np.argsort(corr)[-K:]
        center = n - 1
        g = np.zeros(n)

                g += (1.0 - 2.0*h)

                g[:n-o] -= h[o:]
            else:
                p = -o
                g[:n-p] += (1.0 - h[p:])

        c_cand = compute_c5(h_cand)
        if c_cand < c5_local:

            h = h_cand
        else:

def coord_descent_binary(h, n, max_sweeps):
    """Coordinate descent for binary h via pairwise swaps."""
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Fix count
    s = int(h.sum())
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        h[ones[s-target:s]] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target-s]] = 1
    
    c5_best = np.max(np.correlate(h, 1.0-h, mode="full"))
    
    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        c5_cur = np.max(np.correlate(h, 1.0-h, mode="full"))
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = np.max(np.correlate(h, 1.0-h, mode="full"))
                if c < c5_cur - 1e-12:
                    c5_cur = c
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        
        if improved and c5_cur < c5_best:
            c5_best = c5_cur
        if not improved:
            break
    
    return h, c5_best * (2.0/n)
```

        return project_feasible(res.x, target)
    except Exception:
        return h

def proj_grad_optimize(h, n, target, T, n_iter):
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = lse_gradient(corr, h, T, n)
        lr = 5.0 / (1.0 + 0.05 * it)
        for _bt in range(20):
            h_new = project_feasible(h - lr * g, target)
            f_new = lse_objective(np.correlate(h_new, 1.0 - h_new, mode="full"), T, n)
            f_old = lse_objective(corr, T, n)
            if f_new <= f_old + 1e-18:
                h = h_new
                break
            lr *= 0.5
        else:
            h = project_feasible(h - lr * g, target)
    return h

def to_binary(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def swap_best_improve(h, n, max_sweeps=20):
    h = h.copy()
    h = to_binary(h, n)
    target = n // 2
    s = int(h.sum())
    if s > target:
        ones = np.where(h == 1)[0]
        h[ones[s - target:s]] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - s]] = 1.0
    c5_cur = np.max(np.correlate(h, 1.0 - h, mode="full"))
    for _ in range(max_sweeps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_diff = 0.0
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = np.max(np.correlate(h, 1.0 - h, mode="full"))
                d = c5_cur - c
                if d > best_diff:
                    best_diff = d
                    best_ij = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_ij is None:
            break
        i, j = best_ij
        h[i] = 0.0
        h[j] = 1.0
        c5_cur = c5_cur - best_diff
    return h, float(c5_cur * (2.0 / n))

def swap_first_improve(h, n, max_sweeps=30):
    h = h.copy()
    h = to_binary(h, n)
    target = n // 2
    s = int(h.sum())
    if s > target:
        ones = np.where(h == 1)[0]
        h[ones[s - target:s]] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - s]] = 1.0
    c5_cur = np.max(np.correlate(h, 1.0 - h, mode="full"))
    for _ in range(max_sweeps):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        c_local = c5_cur
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = np.max(np.correlate(h, 1.0 - h, mode="full"))
                if c < c_local:
                    c_local = c
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if improved:
            c5_cur = c_local
        else:
            break
    return h, float(c5_cur * (2.0 / n))

def interp_down(h, n_hi, n_lo):
    x_hi = np.arange(n_hi) * (2.0 / n_hi)
    x_lo = np.arange(n_lo) * (2.0 / n_lo) + 1.0 / n_lo
    h_lo = np.interp(x_lo, x_hi, h)
    target_lo = n_lo / 2.0
    return project_feasible(h_lo, target_lo)

def interp_up(h, n_lo, n_hi):
    x_lo = np.arange(n_lo) * (2.0 / n_lo) + 1.0 / n_lo
    x_hi = np.arange(n_hi) * (2.0 / n_hi) + 1.0 / n_hi
    h_hi = np.interp(x_hi, x_lo, h)
    target_hi = n_hi / 2.0
    return project_feasible(h_hi, target_hi)

def subgrad_polish(h, n, target, n_steps, K=3):
    h = h.copy()
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-K:]
        center = n - 1
        g = np.zeros(n)
        for idx in top_idx:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g *= (2.0 / n) / K
        lr = 0.05 / (1.0 + 0.1 * it)
        h_new = project_feasible(h - lr * g, target)
        c_new = compute_c5(h_new)
        c_old = compute_c5(h)
        if c_new < c_old:
            h = h_new
        else:
            lr *= 0.5
            h = project_feasible(h - lr * g, target)
    return h

def make_init(n, pattern, rng):
    target = n / 2.0
    if pattern == 0:
        h = (np.arange(n) % 2 == 0).astype(float)
    elif pattern == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif pattern == 2:
        h = np.tile(np.array([1.0, 1.0, 0.0, 0.0]), n // 4 + 1)[:n]
    elif pattern == 3:
        h = rng.random(n)
    elif pattern == 4:
        h = np.tile(np.array([1.0, 0.0, 1.0, 1.0, 0.0]), n // 5 + 1)[:n]
    else:
        h = rng.integers(0, 2, n).astype(float) * (n / (n + 1))
    return project_feasible(h, target)

def optimize_one_resolution(h, n, target, t_start, t_end, seed):
    rng = np.random.default_rng(seed)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    # SLSQP phase
    if time.time() < t_end - 15:
        for T in [50, 20, 10, 5, 3, 2, 1.5]:
            if time.time() > t_end - 15:
                break
            h_cur = slsqp_optimize(h_cur, n, target, T, 400)
            c5_new = compute_c5(h_cur)
            if c5_new < c5_cur:
                c5_cur = c5_new
    # Subgradient polish
    if time.time() < t_end - 10:
        h_cur = subgrad_polish(h_cur, n, target, 100, K=3)
        c5_new = compute_c5(h_cur)
        if c5_new < c5_cur:
            c5_cur = c5_new
    # Binary + swap
    h_bin, c5_bin = swap_best_improve(h_cur, n, max_sweeps=10)
    if c5_bin < c5_cur:
        c5_cur = c5_bin
        h_cur = h_bin
    h_bin2, c5_bin2 = swap_first_improve(h_cur, n, max_sweeps=20)
    if c5_bin2 < c5_cur:
        c5_cur = c5_bin2
        h_cur = h_bin2
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    n_main = 280
    target_main = n_main / 2.0
    h_best = None
    c5_best = 1.0

    # Phase 0: Start from initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        target0 = n0 / 2.0
        h0 = project_feasible(h0, target0)
    else:
        h0 = make_init(n_main, 3, rng)
        n0 = n_main
        target0 = target_main

    # Phase 1: Optimize at original resolution (200s budget)
    t_end1 = t0 + 200
    h1, c5_1 = optimize_one_resolution(h0, n0, target0, t0, t_end1, seed=1)
    if c5_1 < c5_best:
        c5_best = c5_1
        h_best = h1
        n_best = n0

    # Phase 2: Multi-resolution exploration
    res_list = [200, 160, 140, 120, 100, 80, 60, 40, 30, 20]
    for n_r in res_list:
        t_start = time.time()
        if t_start > t0 + budget_s - 120:
            break
        t_end_r = t_start + 80
        target_r = n_r / 2.0
        # Try 2 initializations
        for pat in [0, 3]:
            if time.time() > t_end_r - 10:
                break
            h_r = make_init(n_r, pat, rng)
            # Also try interpolated version
            if h_best is not None and pat == 0:
                h_r2 = interp_down(h_best, n_best if n_best != n_r else n_r + 1, n_r)
                h_r = project_feasible((h_r + h_r2) / 2, target_r)
            h_r, c5_r = optimize_one_resolution(h_r, n_r, target_r, t_start, t_end_r, seed=seed + n_r + pat)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r
                n_best = n_r

    # Phase 3: Upsample best low-res to 280 and polish
    if h_best is not None and n_best != n_main:
        t_end3 = t0 + budget_s - 80
        if time.time() < t_end3:
            h_up = interp_up(h_best, n_best, n_main)
            h_up, c5_up = optimize_one_resolution(h_up, n_main, target_main, time.time(), t_end3, seed=999)
            if c5_up < c5_best:
                c5_best = c5_up
                h_best = h_up
                n_best = n_main

    # Phase 4: Final polish at best resolution
    t_end4 = t0 + budget_s - 30
    if time.time() < t_end4 and h_best is not None:
        n_f = n_best
        target_f = n_f / 2.0
        # SLSQP at very low T
        for T in [1.0, 0.7, 0.5]:
            if time.time() > t_end4 - 20:
                break
            h_f = slsqp_optimize(h_best, n_f, target_f, T, 3000)
            c5_f = compute_c5(h_f)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f
        # Subgradient
        if time.time() < t_end4 - 10:
            h_f = subgrad_polish(h_best, n_f, target_f, 200, K=2)
            c5_f = compute_c5(h_f)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f
        # Binary swap
        h_b, c5_b = swap_best_improve(h_best, n_f, max_sweeps=15)
        if c5_b < c5_best:
            c5_best = c5_b
            h_best = h_b
        h_b2, c5_b2 = swap_first_improve(h_best, n_f, max_sweeps=30)
        if c5_b2 < c5_best:
            c5_best = c5_b2
            h_best = h_b2

    # Phase 5: Noise perturbation + quick local search
    t_end5 = t0 + budget_s - 10
    if time.time() < t_end5 and h_best is not None:
        n_p = len(h_best)
        target_p = n_p / 2.0
        for it in range(200):
            if time.time() > t_end5:
                break
            scale = 0.02 + 0.02 * min(it, 50) * 0.1
            noise = rng.normal(0, scale, n_p)
            h_try = project_feasible(h_best + noise, target_p)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try
                h_try2, c5_try2 = swap_best_improve(h_best, n_p, max_sweeps=5)
                if c5_try2 < c5_best:
                    c5_best = c5_try2
                    h_best = h_try2

    # Final validation
    c5_final = min(c5_best, compute_c5(h_best))
    return (h_best, c5_final, len(h_best))
