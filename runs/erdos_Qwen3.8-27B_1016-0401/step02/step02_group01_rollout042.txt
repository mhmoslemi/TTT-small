def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

def project_step(v, target):
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def sm_obj_grad(h, T, ntop=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    if ntop and ntop < len(corr):
        idx = np.argpartition(corr, -ntop)[-ntop:]
    else:
        idx = np.arange(len(corr))
    mc = corr[idx].max()
    sh = corr[idx] - mc
    sh -= sh.max()
    e = np.exp(T * sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    c = n - 1
    for j in range(len(idx)):
        w = sw[j]
        if w < 1e-16:
            continue
        k = idx[j] - c

            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad * dx
