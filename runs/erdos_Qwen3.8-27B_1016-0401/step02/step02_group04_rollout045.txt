def soft_max_obj(h, T, top_k=None):

    corr = correlate_vec(h, 1.0 - h)
    if top_k is not None and top_k < len(corr):

        mc = np.max(corr)
        # Zero out non-top-k entries in the soft-max
        mask = np.zeros(len(corr), dtype=bool)
        idx = np.argsort(corr)[-top_k:]
        mask[idx] = True

    else:

        corr_masked = corr
    shifted = T * (corr_masked - mc)
    valid = np.isfinite(shifted)
    if not np.any(valid):
        return mc * (2.0 / n)
    val = mc + np.log(np.sum(np.exp(shifted[valid]))) / T
    return val * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_top_k_corr(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    # Get top k lags
    if top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(corr)[-top_k:]
    return corr, idx

def topk_softmax_obj_and_grad(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    N = 2 * n - 1
    if top_k >= N:
        top_idx = np.arange(N)
    else:
        top_idx = np.argsort(corr)[-top_k:]
    mc = np.max(corr)
    shifted = T * (corr[top_idx] - mc)
    e = np.exp(shifted - np.max(shifted))
    w = e / np.sum(e)
    obj = (mc + np.log(np.sum(e)) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for i, ci in enumerate(top_idx):
        o = ci - center
        wi = w[i]
        if wi < 1e-14:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def adam_projected(h0, target, T, top_k, lr, max_iter, beta1=0.9, beta2=0.999):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    for t in range(1, max_iter + 1):
        obj, grad = topk_softmax_obj_and_grad(h, T, top_k)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target)
        h = h_new
    return h

    bounds = [(0.0, 1.0)] * n

    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def topk_obj_grad(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    N = 2 * n - 1
    center = n - 1
    if top_k >= N:
        idx = np.arange(N)
    else:
        idx = np.argpartition(corr, -top_k)[-top_k:]
    mc = np.max(corr)
    shifted = T * (corr[idx] - mc)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    obj = (mc + np.log(np.sum(e)) / T) * dx
    grad = np.zeros(n)
    for i in range(len(idx)):
        o = idx[i] - center
        w = sw[i]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad * dx

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def adam_opt(h0, target, T, top_k, lr, max_iter):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for t in range(1, max_iter + 1):
        _, g = topk_obj_grad(h, T, top_k)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_box_sum(h, target)
    return h

def lbfgs_opt(h0, n, target, T, top_k, maxiter, pen):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def objf(x):
        o, _ = topk_obj_grad(x, T, top_k)
        return o + pen * (x.sum() - target) ** 2
    def gradf(x):
        _, g = topk_obj_grad(x, T, top_k)
        return g + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(objf, h0.copy(), method='L-BFGS-B', jac=gradf,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-15, 'maxls': 40})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

        scale = max(0.02, 0.12 / (1.0 + 0.04 * restart_count))
        h_pert = h_best + rng.normal(0, scale, n)
        # Also try structured perturbation based on active lags
        if restart_count % 3 == 0:
            corr = np.correlate(h_best, 1.0 - h_best, mode="full")
            center = n - 1
            top_lags = np.argsort(corr)[-5:]
            for tl in top_lags:
                o = tl - center
                if o > 0:
                    shift = np.roll(h_best, -o)
                    h_pert[:n-o] += 0.02 * (shift[:n-o] - h_pert[:n-o])
                elif o < 0:
                    p = -o
                    shift = np.roll(h_best, p)
                    h_pert[p:] += 0.02 * (shift[p:] - h_pert[p:])
        h_pert = project_box_sum(h_pert, target)
        T = 300 + rng.integers(0, 500)
        tk = 15 + rng.integers(0, 20)
        h_new = adam_opt(h_pert, target, T, tk, 0.2, 8000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Final polish
    if elapsed() < budget_s - 10:
        h_new = adam_opt(h_best, target, 2000, 5, 0.05, 15000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new2 = lbfgs_opt(h_best, n, target, 500, 10, 2000, 50.0)
        c5_2 = compute_c5(h_new2)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h_new2

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
