import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

            hi = mid

def sm_values(h, T):

    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def max_grad(h, K=20):
    """Direct gradient from top-K active lags."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    # Top K lags
    idx = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    c = n - 1
    for j in idx:
        k = j - c
        w = (corr[j] - mc + corr.max() + 1e-10)  # weight
        w = max(w, 0.01)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def pg_momentum(h0, target, T, n_iter, lr0, t0, budget):
    h = h0.copy().astype(float)
    v = np.zeros_like(h)
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mom = 0.82
    for it in range(n_iter):
        if time.time() - t0 > budget - 2:
            break
        obj, sw = sm_values(h, T)
        grad = sm_grad_from_sw(h, sw)
        v = mom * v - lr * grad
        h_new = project_constraint(h + v, target)
        obj_new, _ = sm_values(h_new, T)
        if obj_new > obj + 1e-13:
            ok = False
            for _ in range(25):
                lr *= 0.4
                v = mom * v - lr * grad
                h_new = project_constraint(h + v, target)
                obj_new, _ = sm_values(h_new, T)
                if obj_new <= obj + 1e-13:
                    ok = True
                    break
            if not ok:
                lr *= 8.0
                v *= 0.1
                continue
        h = h_new
        lr = min(lr * 1.04, lr0)
        mom = max(0.55, mom - 0.0001 * it)
        if it % 400 == 399:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def swap_search(h, target, n_trials, t0, budget):
    n = len(h)
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng_s = np.random.default_rng(777)
    high = np.where(h > 0.7)[0]
    low = np.where(h < 0.3)[0]
    mid = np.where((h >= 0.3) & (h <= 0.7))[0]
    for it in range(n_trials):
        if time.time() - t0 > budget - 2:
            break
        r = rng_s.random()
        if r < 0.4 and len(high) > 0 and len(low) > 0:
            i = high[rng_s.integers(0, len(high))]
            j = low[rng_s.integers(0, len(low))]
            d = 0.03 + 0.04 * rng_s.random()
        elif r < 0.7 and len(mid) > 1:
            i = mid[rng_s.integers(0, len(mid))]
            j = mid[rng_s.integers(0, len(mid))]
            if i == j:
                continue
            d = 0.02 + 0.03 * rng_s.random()
        else:
            i = rng_s.integers(0, n)
            j = rng_s.integers(0, n)
            if i == j:
                continue
            d = 0.01 + 0.03 * rng_s.random()
        hi_o, hj_o = h[i], h[j]
        h[i] = np.clip(hi_o + d, 0, 1)

        hp = project_constraint(h, target)

            best_c5 = c5
            best_h = hp.copy()
            h = hp.copy()
        else:
            h[i], h[j] = hi_o, hj_o
    return best_h, best_c5

def coord_refine(h, target, n_passes, t0, budget):

    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

        order = rng_c.permutation(n)
        for i in order:

                h2 = h.copy()
                h2[i] = np.clip(h2[i] + delta, 0, 1)
                h2 = project_constraint(h2, target)
                c5 = compute_c5(h2)
                if c5 < best_c5 - 3e-7:
                    h = h2
                    best_c5 = c5
                    best_h = h.copy()
                    break
    return best_h, best_c5

    # Phase 3: Diverse multi-start

        h_p = h_best.copy()
        style = r % 5

        h_p = project_constraint(h_p, target0)
