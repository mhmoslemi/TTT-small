import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_binary_sum(h, target):
    """Project to exact sum with binary constraint (sort and threshold)"""
    n = len(h)
    # For binary: sort values, set top 'target' to 1, rest to 0
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:int(target)]] = 1.0
    return result

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def c5_of_binary(b):
    """Compute C5 for a binary array"""
    n = len(b)
    corr = np.correlate(b, 1 - b, mode="full")
    return float(np.max(corr) * 2.0 / n)

    b = b.copy()
    # Ensure exactly target ones

        b = (np.zeros(n))
        b[idx[:target]] = 1.0
    
    best_c5 = c5_of_binary(b)
    best_b = b.copy()
    temp = 0.01
    min_temp = 1e-6
    
    for it in range(1000000):
        if time.time() - t0 > time_limit:
            break
        if it % 100 == 0:
            temp = max(min_temp, temp * 0.999)
        
        # Generate perturbation
        r = rng.random()
        b_new = b.copy()
        
        if r < 0.3:
            # Swap two bits (one 1 with one 0)
            ones = np.where(b_new == 1)[0]
            zeros = np.where(b_new == 0)[0]
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            b_new[i] = 0
            b_new[j] = 1
        elif r < 0.5:
            # Block flip
            start = rng.integers(0, n // 4)
            length = rng.integers(3, max(4, n // 10))
            end = min(start + length, n)
            b_new[start:end] = 1 - b_new[start:end]
            # Fix sum
            if b_new.sum() > target:
                excess = int(b_new.sum() - target)
                b0_idx = np.where(b_new == 0)[0]
                b1_idx = np.where(b_new == 1)[0]
                for _ in range(min(excess, len(b0_idx))):
                    j = b1_idx[rng.integers(len(b1_idx))]
                    i = b0_idx[rng.integers(len(b0_idx))]
                    b_new[j] = 0
                    b_new[i] = 1
            elif b_new.sum() < target:
                deficit = int(target - b_new.sum())
                b0_idx = np.where(b_new == 0)[0]
                b1_idx = np.where(b_new == 1)[0]
                for _ in range(min(deficit, len(b1_idx))):
                    j = b0_idx[rng.integers(len(b0_idx))]
                    i = b1_idx[rng.integers(len(b1_idx))]
                    b_new[j] = 1
                    b_new[i] = 0
        elif r < 0.7:
            # Directed: find worst lag, swap misaligned pair
            corr = np.correlate(b_new, 1 - b_new, mode="full")
            worst_idx = np.argmax(corr)
            k = worst_idx - n + 1

                if len(candidates) > 0:

                    b_new[i] = 1
                    b_new[i+k] = 0
                else:

                    candidates = np.where((b_new[:-k] == 1) & (b_new[k:] == 0))[0]
                    if len(candidates) > 0:
                        i = candidates[rng.integers(len(candidates))]
                        b_new[i] = 0
                        b_new[i+k] = 1
        elif r < 0.85:

            start = rng.integers(0, n // 3)
            length = rng.integers(4, max(5, n // 6))
            end = min(start + length, n)
            b_new[start:end] = b_new[start:end][::-1]
            # Fix sum (reversal preserves sum, so no fix needed)
        else:
            # Random restart (rare)
            b_new = (rng.random(n) < 0.5).astype(int)
            # Fix sum
            ones = np.where(b_new == 1)[0]
            target = n // 2
            if len(ones) > target:
                extra_ones = rng.choice(ones, len(ones) - target, replace=False)
                b_new[extra_ones] = 0
            elif len(ones) < target:
                zeros = np.where(b_new == 0)[0]
                need = target - len(ones)
                extra_zeros = rng.choice(zeros, need, replace=False)
                b_new[extra_zeros] = 1
        
        c5_new = c5_of_binary(b_new)
        delta = c5_new - best_c5
        if c5_new < best_c5 or rng.random() < np.exp(-delta / max(temp, 1e-10)):

                best_c5 = c5_new
                best_b = b_new.copy()
                temp = max(min_temp, temp * 0.95)
    
    return best_b, best_c5

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

    lo, hi = h.min() - 1.0, h.max() + 1.0

        if s > target:

        else:
            hi = mid

def active_set_cg(h, target, M, n_iter, t0, budget, lr0=0.05):

    dx = 2.0 / n
    h = h.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    lr = lr0
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 1:
            break
        
        corr = np.correlate(h, 1-h, mode="full")
        mc = float(np.max(corr))
        c5_curr = mc * dx
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        
        # Top M lags
        top_idx = np.argsort(-corr)[:M]
        thresh = corr[top_idx[-1]]
        
        # Gradient of mean top-M correlation
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - n + 1
            if k == 0:
                grad += (1.0 - 2.0*h)
            elif k > 0:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
            else:
                p = -k
                grad[:n-p] += (1.0 - h[p:])
                grad[p:] -= h[:n-p]
        grad /= M
        grad *= dx
        
        # Handle box constraints: gradient at boundaries
        grad[(h <= 0) & (grad < 0)] = 0
        grad[(h >= 1) & (grad > 0)] = 0
        
        # Line search

            lr *= 0.5
        else:
            break
        
        h = h_new
        lr = min(lr * 1.1, lr0)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def c5_binary(b):
    n = len(b)
    c = (b == 1).astype(np.float64)
    corr = np.correlate(c, 1.0 - c, mode="full")
    return float(np.max(corr) * 2.0 / n)

def binarize(h, target):
    n = len(h)
    idx = np.argsort(-h)

def soft_max_opt(h, target, T, n_iter, t0, budget, lr0=0.03):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget - 1.0:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = float(np.max(corr))
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        se = e.sum()
        sw = e / se
        c5_curr = mc * dx
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        grad = np.zeros(n)
        c = n - 1
        for j in range(2 * n - 1):
            w = sw[j]
            if w < 1e-15:
                continue
            k = j - c
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            elif k > 0:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
            else:
                p = -k
                grad[:n-p] += w * (1.0 - h[p:])
                grad[p:] -= w * h[:n-p]
        grad *= dx
        grad[(h <= 1e-8) & (grad < 0)] = 0.0
