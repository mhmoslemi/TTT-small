        amp = rng.uniform(0.02, 0.07)
        x = np.arange(n) * 2.0 / n
        ph = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(np.pi * freq * x + ph)
    elif ptype == 3:
        h_p += rng.uniform(0.005, 0.03) * rng.normal(0, 1, n)
    elif ptype == 4:
        b1 = rng.integers(5, 30)
        s1 = rng.integers(0, max(1, n - b1))
        s2 = rng.integers(0, max(1, n - b1))
        tmp = h_p[s1:s1 + b1].copy()
        h_p[s1:s1 + b1] = h_p[s2:s2 + b1]
        h_p[s2:s2 + b1] = tmp
    else:
        mask = rng.random(n) < 0.15
        h_p[mask] = rng.random(int(mask.sum()))
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine SLSQP with adaptive T and k
    schedule = [(128, 999, 4000), (64, 500, 4000), (32, 200, 4000),
                (16, 100, 5000), (8, 50, 5000), (4, 30, 5000),
                (2, 15, 5000), (1, 8, 5000), (0.5, 5, 6000)]
    for T, k, mi in schedule:
        rem = budget_s - (time.time() - t0)
        if rem < 150:
            break
        h_new = slsqp_opt(h_best, n, T, k, min(mi, int(rem * 2)), 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA with top-3 active gradient
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h_f, c5_f = fist_opt(h_best, n, t0 + rem * 0.4)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: SLSQP fine polish (very focused)
    for T, k in [(1.0, 5), (0.5, 5), (0.25, 3)]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_new = slsqp_opt(h_best, n, T, k, 6000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Structured perturbations + SLSQP
    rem = budget_s - (time.time() - t0)
    n_perts = max(0, int(rem / 18))
    for pi in range(n_perts):
        if time.time() - t0 > budget_s - 10:
            break
        h_p = perturb_structured(h_best, n, rng)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_opt(h_p, n, 4, 20, 2500, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try different resolutions
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        for n_try in [n * 2, 200]:
            if n_try == n or n_try < 50:
                continue
            if time.time() - t0 > budget_s - 80:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            c5_hi = compute_c5(h_hi)
            for T, k in [(16, 50), (4, 20), (1, 5)]:
                if time.time() - t0 > budget_s - 20:
                    break
                h_new = slsqp_opt(h_hi, n_try, T, k, 4000, 1e-17)
                h_new = project_feasible(h_new, n_try)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    n = n_try

    # Final check
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
