def build_cons(target, n):
    return [{'type': 'eq', 'fun': lambda x, t=target: sum_eq(x, t),
             'jac': lambda x, t=target: sum_jac(x, t)}]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Projected gradient with adaptive soft-max (50%)
    p1_end = t0 + budget_s * 0.50
    temperatures = [40, 80, 160, 320, 640, 1280, 2560]
    for T in temperatures:
        if time.time() > p1_end:
            break
        alpha = 0.05
        stall = 0
        for it in range(20000):
            if time.time() > p1_end:
                break
            val, g = softmax_grad(h_best, T)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            g_dir = -g / gnorm
            a = min(alpha, 2.0 / (T * gnorm + 1.0))
            improved = False
            for _ in range(15):
                h_try = fix_constraint(h_best + a * g_dir, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-15:
                    h_best = h_try
                    c5_best = c5_try
                    improved = True
                    break
                a *= 0.5
            if improved:
                stall = 0
                alpha = min(alpha * 1.05, 0.2)
            else:
                stall += 1
                alpha *= 0.7
                if stall > 300:
                    break

    # Phase 2: SLSQP active-set cycling (25%)
    p2_end = t0 + budget_s * 0.75
    bounds = [(0.0, 1.0)] * n
    cons = build_cons(target, n)
    for outer in range(200):
        if time.time() > p2_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_idx = np.argsort(corr)[-4:]
        for lag in top_idx:
            if time.time() > p2_end:
                break
            try:
                res = minimize(lag_objective, h_best.copy(), args=(lag,),
                              method='SLSQP', jac=lag_jac,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-18})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 3: Multi-n exploration (15%)
    p3_end = t0 + budget_s * 0.90
    x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
    for n_alt in [252, 300, 336, 360, 420]:
        if time.time() > p3_end:
            break
        target_alt = n_alt / 2.0
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        cons_alt = build_cons(target_alt, n_alt)
        for outer in range(12):
            if time.time() > p3_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst = int(np.argmax(corr_a))
            try:
                res = minimize(lag_objective, h_alt.copy(), args=(worst,),
                              method='SLSQP', jac=lag_jac,
                              bounds=bounds_alt, constraints=cons_alt,
                              options={'maxiter': 250, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-16:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt

    # Phase 4: Final dense SLSQP polish (15%)
    p4_end = t0 + budget_s - 15.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    h_best = fix_constraint(h_best, target_f)
    bounds_f = [(0.0, 1.0)] * n_f
    cons_f = build_cons(target_f, n_f)
    for outer in range(60):
        if time.time() > p4_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_f = np.argsort(corr_f)[-3:]
        for lag in top_f:
            if time.time() > p4_end:
                break
            try:
                res = minimize(lag_objective, h_best.copy(), args=(lag,),
                              method='SLSQP', jac=lag_jac,
                              bounds=bounds_f, constraints=cons_f,
                              options={'maxiter': 1000, 'ftol': 1e-19})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-17:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
