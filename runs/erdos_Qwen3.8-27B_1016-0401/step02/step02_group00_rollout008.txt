import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < (1.0 - 1e-12)
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

    elif o > 0:

        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += (1.0 - x[p:])
        g[p:] += -x[:n - p]
    return g * (2.0 / n)

    return g * dx

def eq_con_fun(x, target):
    return x.sum() - target

def eq_con_jac(x, target):
    return np.ones(len(x))

def active_set_slsqp(h, target, bounds, n_lags, max_iter, ftol, time_limit):
    t_start = time.time()
    n = len(h)
    for outer in range(n_lags):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-min(4, 2 * n - 1):]
        for lag in idx:
            if time.time() - t_start > time_limit:
                break
            try:
                cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (target,)}]
                res = minimize(lag_obj, h, args=(lag,), method='SLSQP',
                              jac=lag_jac, bounds=bounds, constraints=cons,
                              options={'maxiter': max_iter, 'ftol': ftol})
                h_new = project_feasible(np.clip(res.x, 0, 1), target)
                if compute_c5(h_new) < compute_c5(h):
                    h = h_new
            except Exception:
                pass
    return h

def softmax_slsqp(h, target, bounds, T, max_iter, time_limit):
    t_start = time.time()
    try:
        cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (target,)}]
        res = minimize(sm_obj, h, args=(T,), method='SLSQP',

            if compute_c5(h_new) < compute_c5(h):
                return h_new
    except Exception:
        pass
    return h

def nesterov_polish(h, target, n_iters, time_limit):
    t_start = time.time()
    n = len(h)
    h_x = h.copy()
    h_m = h.copy()
    L = 5.0
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        worst = np.argmax(corr)
        g = lag_jac(h_x, worst)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        h_new = h_m - (L / t_k) * g
        h_new = project_feasible(h_new, target)
        h_m = h_new + ((L - 1.0) / t_k) * (h_new - h_x)
        h_m = np.clip(h_m, 0.0, 1.0)
        h_x = h_new
        L += 1.0
        if it % 300 == 0 and compute_c5(h_x) < compute_c5(h):
            h = h_x.copy()
    return h
