            idx_z = rng.integers(len(zeros))

        if c5_mk2 < c5_best:
            c5_best = c5_mk2
            h_best = h_mk2.copy()

    # Phase 5: Block-level perturbations
    if elapsed() < budget_s - 200:
        for _ in range(30):
            if elapsed() > budget_s - 180:
                break
            h_b = h_best.copy()
            h_b = block_swap(h_b, rng)
            h_b = block_swap(h_b, rng)
            h_b, c5_b = steepest_descent(h_b, 12, 1500, rng)
            if c5_b < c5_best:
                c5_best = c5_b
                h_best = h_b.copy()

    # Phase 6: Second anneal with different params
    if elapsed() < budget_s - 150:
        h_a2, c5_a2 = simulated_anneal(h_best.copy(), 0.012, 0.0002, 12000, rng)
        if c5_a2 < c5_best:
            c5_best = c5_a2
            h_best = h_a2.copy()
        h_bd3, c5_bd3 = steepest_descent(h_best.copy(), 30, 4000, rng)
        if c5_bd3 < c5_best:
            c5_best = c5_bd3
            h_best = h_bd3.copy()

    # Phase 7: Try alternate resolutions
    for n_try in [n - 20, n + 20, n + 40, max(80, n // 2)]:
        if elapsed() > budget_s - 60:
            break
        if n_try < 60 or n_try > 400:
            continue
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try / 2.0)
        h_try_b = to_binary(h_try, n_try / 2.0)
        c5_try = compute_c5(h_try_b)
        if c5_try < c5_best:
            h_try_b, c5_try = steepest_descent(h_try_b.copy(), 20, 2000, rng)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try_b.copy()

    # Phase 8: Final SLSQP polish
    if elapsed() < budget_s - 20:
        n_cur = len(h_best)
        target_cur = n_cur / 2.0
        bounds_cur = [(0.0, 1.0)] * n_cur
        for T in [2.0, 1.0]:
            if elapsed() > budget_s - 8:
                break
            h_f = slsqp_opt(h_best.copy(), n_cur, target_cur, bounds_cur, T, 600)
            c5_f = compute_c5(h_f)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f.copy()

    # Final binary check
    h_bf = to_binary(h_best, len(h_best) / 2.0)
    c5_bf = compute_c5(h_bf)
    if c5_bf < c5_best:
        c5_best = c5_bf
        h_best = h_bf.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
