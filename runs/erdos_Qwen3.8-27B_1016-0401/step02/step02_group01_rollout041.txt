import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_corr(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k]) * dx
            grad[:n - k] -= w * h[k:] * dx
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:]) * dx
            grad[p:] -= w * h[:n - p] * dx
    return obj, grad

def fista_solve(h0, target, T, n_iter, lr0, t0, budget):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    step = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h + (t_k - 1.0) / t_new * (h - h_prev)
        obj, grad = smooth_max_corr(y, T)
        # Adaptive step via Barzilai-Borwein-like
        x_new = project_constraint(y - step * grad, target)
        h_prev = h
        h = x_new
        t_k = t_new
        # Slowly decay step
        step = lr0 / (1.0 + 0.0002 * it)
        if it % 150 == 149:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h
    return best_h, best_c5

def proj_grad_solve(h0, target, T, n_iter, lr0, t0, budget):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        obj, grad = smooth_max_corr(h, T)
        tried = 0
        for _ in range(30):
            h_new = project_constraint(h - lr * grad, target)
            obj_new, _ = smooth_max_corr(h_new, T)
            if obj_new < obj - 1e-16:
                break
            lr *= 0.5
            tried += 1
        else:
            break
        h = h_new
        lr = min(lr * 1.06, lr0)
        if it % 100 == 99:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h
    return best_h, best_c5

def slsqp_solve(h0, target, T, maxiter, t0, budget):
    n = len(h0)
    def obj_f(x):
        return smooth_max_corr(x, T)[0]
    def grad_f(x):
        return smooth_max_corr(x, T)[1]
    def con_f(x):
        return x.sum() - target
    def con_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj_f, h0, method='SLSQP', jac=grad_f,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def periodic_try(n, p, T, n_iter, lr, t0, budget, rng):
    target_p = p / 2.0
    h_p = rng.uniform(0, 1, p)
    h_p = project_constraint(h_p, target_p)
    best_c5 = 1.0
    best_h_p = h_p.copy()
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        h_full = np.tile(h_p, n // p)
        obj, grad_full = smooth_max_corr(h_full, T)
        grad_p = np.zeros(p)
        for i in range(p):
            grad_p[i] = np.mean(grad_full[i::p])
        h_p = project_constraint(h_p - lr * grad_p, target_p)
        if it % 80 == 79:
            h_full = np.tile(h_p, n // p)
            c5 = compute_c5(h_full)
            if c5 < best_c5:
                best_c5 = c5
                best_h_p = h_p.copy()
    h_full = np.tile(best_h_p, n // p)
    return h_full, best_c5

def fourier_init(n, rng, K=6):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, K + 1):
        amp = rng.uniform(0.05, 0.3) / (1.0 + 0.4 * m)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * m * x + ph)
    return project_constraint(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)
