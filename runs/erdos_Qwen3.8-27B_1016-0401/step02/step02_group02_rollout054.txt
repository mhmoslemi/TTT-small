    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    s = _gT * (c - m)
    s -= s.max()
    return (m + np.log(np.exp(s).sum()) / _gT) * (2.0 / len(x))

def _jac(x):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    s = _gT * (c - m)
    s -= s.max()
    e = np.exp(s); w = e / e.sum()
    ctr = n - 1; g = np.zeros(n)
    for i in np.where(w > 1e-8)[0]:
        o = int(i) - ctr; wi = w[i]
        if o == 0:
            g += wi * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += wi * (1.0 - x[:n-o])
            g[:n-o] -= wi * x[o:]
        else:
            p = -o
            g[:n-p] += wi * (1.0 - x[p:])
            g[p:] -= wi * x[:n-p]
    return g * dx

def _cf(x):
    return x.sum() - _gTg

def _cj(x):
    return np.ones(len(x))

def slsqp(h, T, mi=2000):
    global _gT, _gTg
    _gT, _gTg = float(T), len(h) / 2.0
    try:
        r = minimize(_obj, h, method='SLSQP', jac=_jac,
                     bounds=[(0.0, 1.0)] * len(h),
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        return project_feasible(r.x, _gTg)
    except Exception:
        return h

def to_binary(h, target):
    n = len(h)
    hb = np.zeros(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    hb[order[:k]] = 1.0
    if k < n and target > k:
        hb[order[k]] = target - k
    return hb

def bin_best_improve(h, max_passes=80, t0=None, tlit=None):
    n = len(h)

    for p in range(max_passes):

            break
        best_c5 = c5
        best_ij = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                cn = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if cn < best_c5 - 1e-15:
                    best_c5 = cn
                    best_ij = (i, j)
        if best_ij is None:
            break
        h[best_ij[0]], h[best_ij[1]] = 0.0, 1.0
        c5 = best_c5
    return h, c5

where w[k] = exp(T*(corr[k]-max_k)) / sum_j exp(T*(corr[j]-max_k))

For the full correlation, corr[m] = sum_i h[i] * (1-h[i+m-(n-1)]) for valid indices.

- h[i] appears in corr[m] as a factor: if i is in the sum, corr[m] += h[i]*(1-h[i+m-(n-1)])

This is the cross-correlation structure. My gradient computation handles this by iterating over the significant w[k] values and computing the subgradient contribution.
