import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

_S = {}

def _obj(x):
    n, T = _S['n'], _S['T']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    return (mc + sh.max() + np.log(np.exp(sh).sum())) / T * (2.0 / n)

def _jac(x):
    n, T = _S['n'], _S['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for oi, w in enumerate(sw):
        if w < 1e-9:
            continue
        k = oi - center
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - x)
            else:
                grad[k:] += w * (1.0 - x[:n - k])
                grad[:n - k] -= w * x[k:]
        else:
            s = -k
            grad[:n - s] += w * (1.0 - x[s:])
            grad[s:] -= w * x[:-s]
    return grad * dx

def _cf(x):
    return x.sum() - _S['n'] / 2.0

def _cj(x):
    return np.ones(_S['n'])

def slsqp(h0, n, T, maxit, ftol):

                     options={'maxiter': maxit, 'ftol': ftol})
        return r.x
    except Exception:
        return h0.copy()
