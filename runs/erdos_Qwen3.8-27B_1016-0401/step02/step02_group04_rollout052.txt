= -(sum(h_t) - sum(h_t²)) + sum(h_t) - 2*sum(h_t²)
= -sum(h_t) + sum(h_t²) + sum(h_t) - 2*sum(h_t²)
= -sum(h_t²)

→ (1-2h_t)·h - C ≤ -sum(h_t²)
→ (1-2h_t)·h ≤ C - sum(h_t²)

import numpy as np
from scipy.optimize import linprog, minimize
import time

_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def all_lag_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    center = n - 1
    for o in range(-(n - 1), n):
        if o == 0:
            grads[center] = 1.0 - 2.0 * h
        elif o > 0:
            grads[center + o, :n - o] = 1.0 - h[o:]
            grads[center + o, o:] = -h[:n - o]
        else:
            p = -o
            grads[center - o, p:] = 1.0 - h[:n - p]
            grads[center - o, :n - p] = -h[p:]
    return grads

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lm_obj(x):
    T = _state['T']
    target = _state['target']
    pen = _state['pen']
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = float(np.max(corr))
    shifted = T * (corr - mc)
    return (mc + float(np.log(np.sum(np.exp(shifted)))) / T) * (2.0 / n) + pen * (x.sum() - target) ** 2

def _lm_jac(x):
    T = _state['T']
    target = _state['target']
    pen = _state['pen']
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = float(np.max(corr))
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / float(np.sum(e))
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[:n - o] += w * (1.0 - x[o:])
            grad[o:] -= w * x[:n - o]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    return grad * dx + 2.0 * pen * (x.sum() - target)

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    _state['T'] = T
    _state['target'] = target
    _state['pen'] = pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lm_obj, h0.copy(), method='L-BFGS-B', jac=_lm_jac,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def lp_single_step(h_cur, target, A_box, b_box_base, c_obj):
    n = len(h_cur)
    m = 2 * n - 1
    corr = np.correlate(h_cur, 1 - h_cur, mode="full")
    c_vals = corr * (2.0 / n)
    G = all_lag_grads(h_cur)
    A_ub = np.zeros((m + 2 * n, n + 1))
    A_ub[:m, :n] = G
    A_ub[:m, n] = -1.0
    b_ub = np.zeros(m + 2 * n)
    b_ub[:m] = -c_vals + G @ h_cur
    A_ub[m:, :n] = A_box
    b_ub[m:] = b_box_base
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(-1, 1)], method='highs')
    except Exception:
        return None
    return res.x[:n] if res.success else None

def lp_optimize(h, target, max_iters, time_limit):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    A_box = np.vstack([np.eye(n), -np.eye(n)])
    b_box_base = np.concatenate([np.ones(n), np.zeros(n)])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    stuck = 0
    t_start = time.time()
    for _ in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        h_lp = lp_single_step(h_cur, target, A_box, b_box_base, c_obj)
        if h_lp is None:
            stuck += 1
            if stuck > 30:
                break
            continue
        direction = h_lp - h_cur
        nrm = np.linalg.norm(direction)
        if nrm < 1e-10:
            stuck += 1
            if stuck > 30:
                break
            continue
        best_alpha = 0.0
        best_c5 = c5_cur
        max_a = min(1.0, 2.0 / (nrm + 1.0))
        for alpha in [max_a, max_a * 0.5, max_a * 0.3, max_a * 0.15, max_a * 0.05]:
            h_try = np.clip(h_cur + alpha * direction, 0, 1)
            h_try = project_box_sum(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_alpha = alpha
        if best_alpha > 0:
            h_cur = project_box_sum(h_cur + best_alpha * direction, target)
            c5_cur = compute_c5(h_cur)
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
    return h_cur, c5_cur

def boundary_push(h, target, rng, max_sweeps=5):
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_sweeps):
        idx = np.where((h_cur > 0.05) & (h_cur < 0.95))[0]
        if len(idx) == 0:
            break
        idx = rng.choice(idx, size=min(len(idx), 20), replace=False)
        for i in idx:
            for val in [0.0, 1.0]:
                h_try = h_cur.copy()
                h_try[i] = val
                h_try = project_box_sum(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h_cur = h_try
                    c5_cur = c5_try
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: LP-based optimization at n=280
    t1_limit = min(budget_s - 300, 400)
    h_new, c5_new = lp_optimize(h_best, target, 800, t1_limit)
    if c5_new < c5_best:
        c5_best = c5_new
        h_best = h_new

    # Phase 2: Boundary push
    if time.time() - t0 < budget_s - 200:
        h_new, c5_new = boundary_push(h_best, target, rng, 3)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 3: L-BFGS-B polish with progressive temperature
    for T in [30, 80, 200, 500]:
        if time.time() - t0 > budget_s - 180:
            break
        for pen in [5.0, 20.0]:
            if time.time() - t0 > budget_s - 180:
                break

            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Perturbation + LP cycles
    count = 0
    while time.time() - t0 < budget_s - 120:
        count += 1
        scale = max(0.03, 0.12 / (1.0 + 0.08 * count))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        t_rem = min(25.0, budget_s - (time.time() - t0) - 100)
        if t_rem < 5:
            break
        h_new, c5_new = lp_optimize(h_pert, target, 200, t_rem)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 5: Try lower resolution n=140
    if time.time() - t0 < budget_s - 40:

        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n140, endpoint=False)

        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(c5_best, compute_c5(h_best))
    return (h_best, c5_final, len(h_best))
