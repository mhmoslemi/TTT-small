                if 0 <= j + k <= n - 1:
                    a -= h[j + k]
                alpha[center + k] = a
            beta = corr - alpha * old
            sum_rest = h.sum() - old
            def f1d(x):
                c = beta + alpha * x
                m = np.max(c)
                s = T * (c - m)
                sm = m + np.log(np.sum(np.exp(s))) / T
                p = pen * (sum_rest + x - target) ** 2
                return sm * dx + p
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            h[j] = float(res.x)
        h = project_box_sum(h, target)
    return h

def fista_step(h, T, target, max_iter=3000, step=0.1):
    n = len(h)
    zh = h.copy()
    t_k = 1.0
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = sm_grad_fn(y, T, 5.0, target)
        zh = project_box_sum(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with progressive T at current resolution
    for T in [200, 500, 1500, 5000, 20000]:
        if elapsed() > budget_s - 400:
            break
        for pen in [5.0, 50.0, 500.0]:
            h_new = lbfgsb_opt(h_best, n, target, T, pen, 2000)
            h_new = project_box_sum(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 2: Coordinate descent (new method)
    for T in [500, 2000, 10000]:
        if elapsed() > budget_s - 300:
            break
        h_new = coord_descent(h_best, T, target, n_sweeps=20, pen=200.0)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: FISTA with small step for polishing
    for T in [1000, 5000]:
        if elapsed() > budget_s - 250:
            break
        h_new = fista_step(h_best, T, target, 2000, 0.05)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Perturbation + coordinate descent escaping
    n_pert = 0
    while elapsed() < budget_s - 150:
        n_pert += 1
        scale = max(0.02, 0.10 / (1.0 + 0.03 * n_pert))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T_cd = 1000 + rng.integers(0, 5000)
        h_new = coord_descent(h_pert, T_cd, target, n_sweeps=10, pen=200.0)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if elapsed() > budget_s - 150:
            break

    # Phase 5: Try higher resolution n=560
    if elapsed() < budget_s - 80:
        n2 = min(560, max(2 * n, 280))
        target2 = n2 / 2.0
        h_try = interpolate_h(h_best, n2)
        c5_try = compute_c5(h_try)
        for T in [500, 2000, 10000]:
            if elapsed() > budget_s - 30:
                break
            h_t = lbfgsb_opt(h_try, n2, target2, T, 50.0, 1500)
            h_t = project_box_sum(h_t, target2)
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        if elapsed() < budget_s - 20:
            h_try = coord_descent(h_try, 3000, target2, n_sweeps=5, pen=200.0)
            h_try = project_box_sum(h_try, target2)
            c5_try = compute_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n = n2
            target = target2

    # Phase 6: Try lower resolution n=140 (sometimes simpler structure wins)
    if elapsed() < budget_s - 30:
        n3 = min(140, n // 2 if n > 140 else 140)
        target3 = n3 / 2.0
        h_try3 = interpolate_h(h_best, n3)
        c5_try3 = compute_c5(h_try3)
        for T in [1000, 5000]:
            if elapsed() > budget_s - 10:
                break
            h_t3 = lbfgsb_opt(h_try3, n3, target3, T, 50.0, 1000)
            h_t3 = project_box_sum(h_t3, target3)
            c5_t3 = compute_c5(h_t3)
            if c5_t3 < c5_try3:
                c5_try3 = c5_t3
                h_try3 = h_t3
        if c5_try3 < c5_best:
            c5_best = c5_try3
            h_best = h_try3
            n = n3
            target = target3

    # Final: quick coordinate descent polish
    if elapsed() < budget_s - 5:
        h_new = coord_descent(h_best, 5000, target, n_sweeps=5, pen=500.0)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
