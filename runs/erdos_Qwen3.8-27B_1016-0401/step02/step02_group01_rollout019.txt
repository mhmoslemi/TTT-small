import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj_and_sw(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def binary_swap_search(h, target, t0, deadline, rounds=100, trials=2500):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(rounds):
        if time.time() > deadline:
            break
        rng_l = np.random.default_rng(rnd * 104729 + 42)
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break

        improved = False

                break

            b = ones[oi]

                improved = True
