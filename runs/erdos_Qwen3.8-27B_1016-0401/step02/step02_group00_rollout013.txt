from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        if not m.any():
            break
        h[m] = np.clip(h[m] + d / m.sum(), 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, lag):
    n = len(h)
    c = n - 1
    o = lag - c
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] += -h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]
    return g * dx

    corr = get_corr(h)
    mc = corr.max()
    s = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(s))) / T) * (2.0 / len(h))
