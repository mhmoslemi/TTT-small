        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def corr_grad_single(h, o, dx):
    n = len(h)
    grad = np.zeros(n)
    if o == 0:
        grad[:] = (1.0 - 2.0 * h)
    elif o > 0:
        grad[o:] += (1.0 - h[:n - o])
        grad[:n - o] -= h[o:]
    else:
        p = -o
        grad[:n - p] += (1.0 - h[p:])
        grad[p:] -= h[:n - p]
    return grad * dx

def obj_topk(x, k_act, target, n, dx):
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    s = 0.0
    for k in k_act:
        s += c[k]
    return s / len(k_act) + 5.0 * (x.sum() - target) ** 2

def grad_topk(x, k_act, target, n, dx):
    g = np.zeros(n)
    for k in k_act:
        o = k - (n - 1)
        g += corr_grad_single(x, o, dx)
    g /= len(k_act)
    g += 10.0 * (x.sum() - target)
    return g

def obj_hinge(x, T_val, target, n, dx):
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    excess = np.maximum(0.0, c - T_val)
    return float(np.sum(excess ** 2) + 1e-4 * T_val + 5.0 * (x.sum() - target) ** 2)

def grad_hinge(x, T_val, target, n, dx):
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    g = np.zeros(n)
    center = n - 1
    for idx in range(len(c)):
        ex = c[idx] - T_val
        if ex < 1e-10:
            continue
        w = 2.0 * ex
        o = idx - center
        g += w * corr_grad_single(x, o, dx)
    g += 10.0 * (x.sum() - target)
    return g

def smooth_init(n, mode):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if mode == 0:
        h = 1.0 + np.cos(np.pi * x)
    elif mode == 1:
        h = np.abs(np.cos(np.pi * x / 2.0))
    else:
        h = 1.0 - 0.5 * np.cos(2.0 * np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = smooth_init(140, 0)

    n = len(h_best)
    target = n / 2.0
    dx = 2.0 / n
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Top-K chase (progressively larger K)
    for K in [2, 3, 5, 8, 12]:
        if elapsed() > budget_s - 400:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full") * dx
        k_act = tuple(np.argsort(corr)[-K:])
        res = minimize(obj_topk, h_best.copy(), method='L-BFGS-B',
                       jac=grad_topk, args=(k_act, target, n, dx),
                       bounds=bounds, options={'maxiter': 2000, 'gtol': 1e-13})
        h_new = project_box_sum(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 2: Hinge-loss with decreasing target
    for frac in np.linspace(1.0, 0.995, 8):
        if elapsed() > budget_s - 350:
            break
        T_val = c5_best * frac
        res = minimize(obj_hinge, h_best.copy(), method='L-BFGS-B',
                       jac=grad_hinge, args=(T_val, target, n, dx),
                       bounds=bounds, options={'maxiter': 1500, 'gtol': 1e-13})
        h_new = project_box_sum(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 3: Basin hopping with adaptive perturbation
    no_improve = 0
    while elapsed() < budget_s - 200 and no_improve < 60:
        scale = 0.02 + 0.04 * min(no_improve, 20) / 20.0
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        corr_p = np.correlate(h_pert, 1.0 - h_pert, mode="full") * dx
        thr = np.max(corr_p) * (1.0 - 0.05 * min(no_improve, 10) / 10.0)
        k_act = tuple(np.where(corr_p > thr)[0])
        if len(k_act) == 0:
            k_act = (int(np.argmax(corr_p)),)
        res = minimize(obj_topk, h_pert.copy(), method='L-BFGS-B',
                       jac=grad_topk, args=(k_act, target, n, dx),
                       bounds=bounds, options={'maxiter': 800, 'gtol': 1e-12})
        h_new = project_box_sum(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
            no_improve = 0
        else:
            no_improve += 1

    # Phase 4: Try smooth initializations at lower resolution, then upscale
    for mode in range(3):
        if elapsed() > budget_s - 100:
            break
        n_s = 140
        h_s = smooth_init(n_s, mode)
        h_s = project_box_sum(h_s, n_s / 2.0)
        bounds_s = [(0.0, 1.0)] * n_s
        dx_s = 2.0 / n_s
        corr_s = np.correlate(h_s, 1.0 - h_s, mode="full") * dx_s
        k_s = tuple(np.argsort(corr_s)[-5:])
        res = minimize(obj_topk, h_s.copy(), method='L-BFGS-B',
                       jac=grad_topk, args=(k_s, n_s / 2.0, n_s, dx_s),
                       bounds=bounds_s, options={'maxiter': 3000, 'gtol': 1e-13})
        h_s = project_box_sum(res.x, n_s / 2.0)
        # Upscale to current n
        h_up = interpolate_h(h_s, n_s, n)
        h_up = project_box_sum(h_up, target)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            k_up = tuple(np.where(np.correlate(h_up, 1.0-h_up, mode="full")*dx >
                                  c5_up*0.85)[0])
            res2 = minimize(obj_topk, h_up.copy(), method='L-BFGS-B',
                            jac=grad_topk, args=(k_up, target, n, dx),
                            bounds=bounds, options={'maxiter': 2000, 'gtol': 1e-13})
            h_up = project_box_sum(res2.x, target)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best:
                c5_best = c5_up
                h_best = h_up

    # Phase 5: Resolution change to n=420 for finer resolution
    if elapsed() < budget_s - 80:
        n_new = 420
        h_new_res = interpolate_h(h_best, n, n_new)
        t_new = n_new / 2.0
        h_new_res = project_box_sum(h_new_res, t_new)
        dx_new = 2.0 / n_new
        bounds_new = [(0.0, 1.0)] * n_new
        corr_nr = np.correlate(h_new_res, 1.0 - h_new_res, mode="full") * dx_new
        k_nr = tuple(np.argsort(corr_nr)[-5:])
        res = minimize(obj_topk, h_new_res.copy(), method='L-BFGS-B',
                       jac=grad_topk, args=(k_nr, t_new, n_new, dx_new),
                       bounds=bounds_new, options={'maxiter': 2000, 'gtol': 1e-13})
        h_new_res = project_box_sum(res.x, t_new)
        c5_nr = compute_c5(h_new_res)
        if c5_nr < c5_best:
            c5_best = c5_nr
            h_best = h_new_res
            n = n_new
            target = n / 2.0
            dx = 2.0 / n
            bounds = [(0.0, 1.0)] * n

    # Phase 6: Final intensive polish
    if elapsed() < budget_s - 10:
        for K in [3, 5, 8]:
            if elapsed() > budget_s - 2:
                break
            corr = np.correlate(h_best, 1.0 - h_best, mode="full") * dx
            k_act = tuple(np.argsort(corr)[-K:])
            res = minimize(obj_topk, h_best.copy(), method='L-BFGS-B',
                           jac=grad_topk, args=(k_act, target, n, dx),
                           bounds=bounds, options={'maxiter': 3000, 'gtol': 1e-14})
            h_new = project_box_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        T_val = c5_best * 1.0
        res = minimize(obj_hinge, h_best.copy(), method='L-BFGS-B',
                       jac=grad_hinge, args=(T_val, target, n, dx),
                       bounds=bounds, options={'maxiter': 2000, 'gtol': 1e-14})
        h_new = project_box_sum(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
