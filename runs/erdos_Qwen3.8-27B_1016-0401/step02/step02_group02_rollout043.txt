import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc) - np.max(T * (corr - mc))
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc) - np.max(T * (corr - mc))
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_phase(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h

def nesterov_subgrad(h, n, target, n_steps, K=3, lr_init=0.05, lr_decay=0.999):
    dx = 2.0 / n
    c5_local = compute_c5(h)
    v = np.zeros(n)
    lr = lr_init
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-K:]
        center = n - 1
        g = np.zeros(n)
        for idx in top_idx:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g *= dx / K
        v = 0.9 * v + 0.1 * g
        h_cand = project_feasible(h - lr * v, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_local:
            c5_local = c5_cand
            h = h_cand
        else:
            lr *= 0.9
            h_cand = project_feasible(h - lr * v, target)
            h = h_cand
        lr = max(lr_init * (lr_decay ** it), 0.001)
    return h, c5_local

def pair_local_search(h, n, target, n_iters, rng):
    c5_cur = compute_c5(h)
    for it in range(n_iters):
        i, j = rng.integers(0, n, 2)

            continue

                c5_t = compute_c5(h2)
                if c5_t < c5_cur:

from scipy.optimize import minimize

def lse_obj_grad(corr, h, T, n):

    mc = np.max(corr)
    sh = T * (corr - mc)
    m = np.max(sh)
    sh = sh - m
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + (m + np.log(se)) / T) * dx
    w = e / se
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]

            grad[p:] -= wi * h[:n - p]

        return x.sum() - target

            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h2 = h.copy()
                h2[i] = hi
                h2[j] = hj
                c5_t = compute_c5(h2)
                if c5_t < c5_cur:
                    c5_cur = c5_t
                    h = h2
    return h, c5_cur

    k_full = int(target)
    h_bin[order[:k_full]] = 1.0

    return h_bin

import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    m = np.max(sh)
    sh = sh - m
    return (mc + (m + np.log(np.sum(np.exp(sh)))) / T) * dx

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    m = np.max(sh)
    sh = sh - m
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_phase(h, n, target, T, maxiter):

        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):

        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
