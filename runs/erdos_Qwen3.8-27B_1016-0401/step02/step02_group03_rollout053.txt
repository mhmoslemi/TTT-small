```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if np.any(m2):
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        if np.any(m2):
            grad[m2] -= h[j[m2] - s]
    return grad

def compute_gradient_full(h, n, T):
    """Compute gradient of smoothed max objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh_max = np.max(sh)  # should be 0
    e = np.exp(sh - sh_max)
    p = e / e.sum()
    
    # Only compute gradient for active lags (p > threshold)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    active = p > 1e-10
    for k in range(len(p)):
        if not active[k]:
            continue
        shift = k - center
        grad += p[k] * shift_grad(h, n, shift)
    return grad * dx

def adam_smoothed_max(h_init, n, t0, time_budget, T_schedule=None):
    """Adam optimizer for the smoothed max problem."""
    h = h_init.copy()
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    
    if T_schedule is None:
        T_schedule = [32, 16, 8, 4, 2, 1, 1, 0.5]
    
    for T_idx, T in enumerate(T_schedule):
        # Steps per T level
        per_T_budget = time_budget / len(T_schedule)
        t_start = time.time()
        it = 0
        while time.time() - t_start < per_T_budget:
            if time.time() - t0 > time_budget:
                return h_best, c5_best
            it += 1
            
            # Decay learning rate
            lr_cur = lr / (1.0 + 0.0001 * it)
            
            # Compute gradient
            grad = compute_gradient_full(h, n, T)
            
            # Adam update
            m_v = beta1 * m_v + (1 - beta1) * grad
            v_v = beta2 * v_v + (1 - beta2) * grad * grad
            m_hat = m_v / (1 - beta1 ** it)
            v_hat = v_v / (1 - beta2 ** it)
            h = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project
            if it % 5 == 0:
                h = project_feasible(h, n)
            else:
                h = np.clip(h, 0.0, 1.0)
                diff = n / 2.0 - h.sum()
                if abs(diff) > 1e-10:
                    h = np.clip(h + diff / n, 0.0, 1.0)
            
            # Evaluate
            if it % 500 == 0:
                c5_cur = compute_c5(project_feasible(h, n))
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    h_best = project_feasible(h, n).copy()
    
    c5_cur = compute_c5(project_feasible(h, n))
    if c5_cur < c5_best:
        c5_best = c5_cur
        h_best = project_feasible(h, n).copy()
    
    return h_best, c5_best
```

```python
def compute_c5(h):

    dx = 2.0 / n

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
```

```python
def compute_grad_smoothed(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh_max = np.max(sh)

    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    sh_max = np.max(sh)

    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

    return x.sum() - _G_TARGET

    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):

    _G_T = float(T)
    _G_TARGET = n / 2.0

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
```
