def fast_correlate(a, b):
    n = len(a)
    size = 2 * n - 1
    # Zero-pad to next power of 2 or 2*size
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    # Circular correlation: use conjugate
    corr_circ = np.fft.irfft(fa * np.conj(fb), N)
    # Extract the linear part
    return corr_circ[:size]  # full mode correlation

def max_subgrad(h, n_active_max=10, rel_tol=1e-6):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    center = n - 1
    max_val = corr.max()
    active = np.where(corr >= max_val - rel_tol * max_val)[0]
    if len(active) > n_active_max:

        active = active[order[:n_active_max]]
    grad = np.zeros(n)

        w = corr[idx] / corr[active].sum()  # weight by value

        if k == 0:

        if s > target:

        else:

        j = rng.choice(ones)
        i = rng.choice(zeros)
        h[j] = 0.0
        h[i] = 1.0
        c5_new = compute_c5(h)
        if c5_new > c5_cur + 0.001:
            h[j] = 1.0
            h[i] = 0.0
        else:
            c5_cur = c5_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    time_budget = budget_s
    t_limit = t0 + time_budget

    # Phase 1: L-BFGS-B with aggressive T cooling
    for T in [200, 80, 30, 15, 8]:
        if time.time() > t_limit - 300:
            break
        h_new = lbfgsb_softmax(h_best, n, target, T, 4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Nesterov with true max subgradient
    for step in [3.0, 2.0, 1.0]:
        if time.time() > t_limit - 250:
            break
        h_new = nesterov_subgrad(h_best, target, 5000, step=step)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 3: Coordinate descent polish
    if time.time() < t_limit - 200:
        h_new, c5 = coord_descent(h_best, target, n_rounds=3, rng=rng)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Perturbation + L-BFGS-B restarts
    h_best, c5_best = annealed_perturb_opt(h_best, c5_best, n, target, rng, t_limit - 150)

    # Phase 5: More Nesterov + coord descent
    if time.time() > t_limit - 100:
        pass
    else:
        h_new = nesterov_subgrad(h_best, target, 8000, step=1.5)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
        if time.time() < t_limit - 50:
            h_new, c5 = coord_descent(h_best, target, n_rounds=5, rng=rng)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new

    # Phase 6: Try higher resolution
    n2 = n * 2
    if time.time() < t_limit - 80:
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h_best)
        h2 = project_box_sum(h2, n2 / 2.0)
        h2 = lbfgsb_softmax(h2, n2, n2 / 2.0, 50, 2000)
        c5_2 = compute_c5(h2)
        if c5_2 < c5_best:
            h_best, c5_best = h2, c5_2
            n, target = n2, n2 / 2.0

    # Phase 7: Try lower resolution (can sometimes be better)
    if time.time() < t_limit - 40 and n > 70:
        n3 = max(70, n // 2)
        x_arr = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new3 = np.linspace(0.0, 2.0, n3, endpoint=False)
        h3 = np.interp(x_new3, x_arr, h_best)
        h3 = project_box_sum(h3, n3 / 2.0)
        h3 = lbfgsb_softmax(h3, n3, n3 / 2.0, 30, 2000)
        c5_3 = compute_c5(h3)
        if c5_3 < c5_best:
            h_best, c5_best = h3, c5_3
            n, target = n3, n3 / 2.0

    # Final coordinate descent
    if time.time() < t_limit - 5:
        h_new, c5 = coord_descent(h_best, target, n_rounds=3, rng=rng)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Final L-BFGS-B polish
    if time.time() < t_limit - 2:
        h_new = lbfgsb_softmax(h_best, n, target, 10, 2000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    c5_final = min(compute_c5(h_best), c5_best)
    h_final = project_box_sum(h_best, len(h_best) / 2.0)
    return (h_final, c5_final, len(h_final))
