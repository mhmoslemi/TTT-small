    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def subgrad_topk(corr, h, n, K):
    dx = 2.0 / n
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]

def softplus_obj(h, T, n):
    corr = compute_corr(h)
    dx = 2.0 / n
    mc = np.max(corr)
    arg = T * (corr - mc)
    arg -= np.max(arg)
    e = np.exp(arg)
    return (mc + np.log(e.sum()) / T) * dx

def softplus_grad(h, T, n):
    corr = compute_corr(h)
    dx = 2.0 / n
    mc = np.max(corr)
    arg = T * (corr - mc)
    arg -= np.max(arg)
    e = np.exp(arg)
    w = e / e.sum()
    center = n - 1
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        o = idx - center
        wi = w[idx]
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

    if abs(h_b.sum() - target) > 0.5:
