def fix_sum(h, target):
    """Fix sum to target while keeping [0,1]."""
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to add mass; prioritize where h is below 1
            headroom = 1.0 - h
            order = np.argsort(-headroom)  # most headroom first
            for idx in order:
                if diff <= 0:
                    break
                add = min(diff, headroom[idx])
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(-h)  # most mass first
            for idx in order:
                if diff >= 0:
                    break
                rem = min(-diff, h[idx])
                h[idx] -= rem
                diff += rem
    return h

def slsqp_step(h, target, T, maxiter=3000, ftol=1e-15):

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,

def sa_search(h, target, n_iter, t0, deadline, rng, T_init=0.001, cool=0.97):
    """Simulated annealing on 2-point mass transfers."""
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    corr_curr = np.correlate(h_curr, 1 - h_curr, mode="full")
    c5_curr = float(np.max(corr_curr) * dx)
    best_h = h_curr.copy()
    best_c5 = c5_curr
    T = T_init
    delta = 0.05
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick two distinct random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        
        # Transfer delta from i to j (or vice versa)
        if h_curr[i] >= h_curr[j]:
            move = (-delta, delta)  # take from i, give to j
        else:
            move = (delta, -delta)  # take from j, give to i
        
        h_try = h_curr.copy()
        h_try[i] += move[0]
        h_try[j] += move[1]
        h_try = np.clip(h_try, 0, 1)
        h_try = fix_sum(h_try, target)
        
        # Evaluate
        corr_try = np.correlate(h_try, 1 - h_try, mode="full")
        c5_try = float(np.max(corr_try) * dx)

                best_h = h_curr.copy()

        # Occasionally refresh delta
        if it % 200 == 199:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            headroom = 1.0 - h
            idx = np.where(headroom > 1e-10)[0]
            if len(idx) == 0:
                break
            per = diff / len(idx)
            h[idx] = np.minimum(1.0, h[idx] + per)
        else:
            idx = np.where(h > 1e-10)[0]
            if len(idx) == 0:
                break
            per = -diff / len(idx)
            h[idx] = np.maximum(0.0, h[idx] - per)
    return h

def sm_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def sm_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    g = np.zeros(n)
    c = n - 1
    for s in range(-(n-1), n):
        w = sw[c + s]
        if w < 1e-14:
            continue
        if s >= 0:
            g[s:] += w * (1.0 - x[:n-s])
            g[:n-s] += -w * x[s:]
        else:
            p = -s
            g[p:] += w * (1.0 - x[:n-p])
            g[:n-p] += -w * x[p:]
    return g * dx

def c5_con_fun(x, tgt):
    return x.sum() - tgt

def c5_con_jac(x, tgt):
    return np.ones(len(x))

def do_slsqp(h, target, T, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c5_con_fun, 'jac': c5_con_jac, 'args': (target,)}]
    try:
        res = minimize(sm_obj, h, method='SLSQP', jac=sm_jac, args=(T,),
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < 10:
            return fix_sum(res.x, target)
    except Exception:
        pass
    return h.copy()

def sa_local(h, target, n_iter, deadline, rng, T_start, cool, d0):
    n = len(h)
    dx = 2.0 / n
    hc = h.copy()
    cc = np.correlate(hc, 1 - hc, mode="full")
    c5c = float(np.max(cc) * dx)
    best_h = hc.copy()
    best_c5 = c5c
    T = T_start
    d = d0
    for it in range(n_iter):

            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:

        if hc[i] < hc[j]:

        amt = min(d, hc[i])
        if amt < 1e-8:

        ht = hc.copy()
        ht[i] -= amt
        ht[j] += amt
        ht = np.clip(ht, 0, 1)
        ht = fix_sum(ht, target)
        ct = np.correlate(ht, 1 - ht, mode="full")
        c5t = float(np.max(ct) * dx)
        dc = c5t - c5c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-10)):

            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
        T *= cool
        if it % 500 == 499:
            d *= 0.9
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
            if len(idx) == 0:
                break
            h[idx] = np.minimum(1.0, h[idx] + diff / len(idx))
        else:
            idx = np.where(h > 1e-10)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return h

def sm_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def sm_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    g = np.zeros(n)
    c = n - 1
    for s in range(-(n - 1), n):
        w = sw[c + s]
        if w < 1e-14:
            continue
        if s >= 0:
            g[s:] += w * (1.0 - x[:n - s])
            g[:n - s] += -w * x[s:]
        else:
            p = -s
            g[p:] += w * (1.0 - x[:n - p])
            g[:n - p] += -w * x[p:]
    return g * dx

def eq_fun(x, tgt):
    return x.sum() - tgt

def eq_jac(x, tgt):
    return np.ones(len(x))

def do_slsqp(h, target, T, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (target,)}]
    try:
        res = minimize(sm_obj, h, method='SLSQP', jac=sm_jac, args=(T,),
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < 10:
            return fix_sum(res.x, target)
    except Exception:
        pass
    return h.copy()

def sa_local(h, target, n_iter, deadline, rng, T_start, cool, d0, d_decay_period, d_decay):
    n = len(h)
    dx = 2.0 / n
    hc = h.copy()
    cc = np.correlate(hc, 1 - hc, mode="full")
    c5c = float(np.max(cc) * dx)
    best_h = hc.copy()
    best_c5 = c5c
    T = T_start
    d = d0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if hc[i] < hc[j]:
            i, j = j, i
        amt = min(d, hc[i])
        if amt < 1e-9:
            continue
        ht = hc.copy()
        ht[i] -= amt
        ht[j] += amt
        ht = np.clip(ht, 0, 1)
        ht = fix_sum(ht, target)
        ct = np.correlate(ht, 1 - ht, mode="full")
        c5t = float(np.max(ct) * dx)
        dc = c5t - c5c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-12)):
            hc, cc, c5c = ht, ct, c5t
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
        T *= cool
        if (it + 1) % d_decay_period == 0:
            d *= d_decay
    return best_h, best_c5

def greedy_2opt(h, target, n_iter, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    hc = h.copy()
    cc = np.correlate(hc, 1 - hc, mode="full")
    best_c5 = float(np.max(cc) * dx)
    search_c5 = best_c5
    stall = 0
    delta = 0.04
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            delta *= 0.6
            stall = 0
            if delta < 0.001:
                break
        k_star = np.argmax(cc)
        s = k_star - (n - 1)
        p = abs(s)
        hc_try = hc.copy()
        if p == 0:
            vals = hc * (1.0 - hc)
            idx = np.argmax(vals)
            if hc[idx] > 0.5:
                hc_try[idx] -= delta
            else:
                hc_try[idx] += delta
        else:
            if s > 0:
                contrib = hc[p:] * (1.0 - hc[:n - p])
                idx = np.argmax(contrib)
                hc_try[idx + p] -= delta
                hc_try[idx] += delta
            else:
                contrib = hc[:n - p] * (1.0 - hc[p:])
                idx = np.argmax(contrib)
                hc_try[idx] -= delta
                hc_try[idx + p] += delta
        hc_try = np.clip(hc_try, 0, 1)
        hc_try = fix_sum(hc_try, target)
        cc_try = np.correlate(hc_try, 1 - hc_try, mode="full")
        c5_try = float(np.max(cc_try) * dx)
        if c5_try < search_c5 - 1e-15:
            search_c5 = c5_try
            hc = hc_try
            cc = cc_try
            stall = 0
            if search_c5 < best_c5:
                best_c5 = search_c5
        else:
            stall += 1
    return hc, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP annealing
    for T in [150, 80, 50, 30, 20, 12, 8, 5, 3, 2, 1.5, 1]:
        if time.time() > deadline - 500:
            break
        h_new = do_slsqp(h_best, target, T, 4000, 1e-16)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-15:
            c5_best = c5
            h_best = h_new

    # Phase 2: SA medium
    h_sa, c5_sa = sa_local(h_best, target, 40000, deadline - 350, rng,
                           0.002, 0.99, 0.1, 300, 0.92)
    if c5_sa < c5_best:
        c5_best = c5_sa
        h_best = h_sa

    # Phase 3: SLSQP refinement
    for T in [10, 5, 3, 2, 1.5, 1]:
        if time.time() > deadline - 250:
            break
        h_new = do_slsqp(h_best, target, T, 5000, 1e-17)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-15:
            c5_best = c5
            h_best = h_new

    # Phase 4: Greedy 2-opt
    h_g, c5_g = greedy_2opt(h_best, target, 5000, deadline - 200, rng)
    if c5_g < c5_best:
        c5_best = c5_g
        h_best = h_g

    # Phase 5: SA fine
    h_sa2, c5_sa2 = sa_local(h_best, target, 30000, deadline - 150, rng,

    # Phase 6: SLSQP final

        h_new = do_slsqp(h_best, target, 1, 8000, 1e-18)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 7: Greedy 2-opt final + tiny SA
    h_g2, c5_g2 = greedy_2opt(h_best, target, 3000, deadline - 10, rng)

        if c5_sa3 < c5_best:
            c5_best = c5_sa3
            h_best = h_sa3

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
