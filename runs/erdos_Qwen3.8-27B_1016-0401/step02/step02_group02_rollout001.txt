import numpy as np
import time
from scipy.optimize import minimize, fmin_lsbfgs

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def smooth_max_val(corr, T):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return mc + np.log(np.sum(np.exp(sh))) / T

def smooth_max_grad(corr, h, T):
    n = len(h)
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_optimize(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val(np.correlate(x, 1.0-x, mode="full"), T) * (2.0/n)
    def jac(x):
        return smooth_max_grad(np.correlate(x, 1.0-x, mode="full"), x, T)
    def cfun(x):
        return x.sum() - n/2.0
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n/2.0)
    except:
        return h

def nesterov_pg(h, n, target, T, n_iter, lr0, momentum0=0.3):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    velocity = np.zeros(n)
    
    for it in range(n_iter):
        lr = lr0 / (1.0 + 0.05 * it)

        grad = smooth_max_grad(corr, h_curr, T)

        h_new = project_feasible(h_curr + velocity, target)

        best_h = h_curr.copy()
    return best_h, best_c5

def flat_max_obj(h, n, T, K):
    """Objective: smooth max + penalty on top K raw correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_val(corr, T)
    topk = np.sort(corr)[-K:]
    # Penalty: make top K more uniform
    penalty = np.var(topk) * 0.1
    return (sm + penalty) * (2.0/n)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def smax_val(corr, T):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return mc + np.log(np.sum(np.exp(sh))) / T

def smax_grad(corr, h, T):
    n = len(h)
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_run(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smax_val(np.correlate(x, 1.0 - x, mode="full"), T) * (2.0 / n)
    def jac(x):
        return smax_grad(np.correlate(x, 1.0 - x, mode="full"), x, T)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def nesterov_pg(h, n, target, T, n_iter, lr0):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_curr + beta * (h_curr - h_prev)
        corr = np.correlate(y, 1.0 - y, mode="full")
        grad = smax_grad(corr, y, T)
        lr = lr0 / (1.0 + 0.02 * it)
        h_next = project_feasible(y - lr * grad, target)
        h_prev = h_curr
        h_curr = h_next
        t = t_new
        if it % 200 == 0 or it == n_iter - 1:
            c5_v = compute_c5(h_curr)
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h_curr.copy()
    return best_h, best_c5

def fine_subgrad(h, n, target, n_steps, K):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-K:]
        center = n - 1
        dx = 2.0 / n
        g = np.zeros(n)
        for idx in top_idx:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g *= dx / K
        lr = 0.03 / (1.0 + 0.05 * it)
        h_cand = project_feasible(h - lr * g, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            h = h_cand
        else:
            lr *= 0.5
            h = project_feasible(h - lr * g, target)
    return best_h, best_c5

    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    h_best = h.copy()
    c5_best = compute_c5(h)

    # Phase 1: Continuation SLSQP
    schedule = [(200, 150), (100, 200), (50, 300), (25, 400),
                (15, 500), (10, 700), (7, 900), (5, 1200),
                (3, 1800), (2, 2500), (1.5, 3500), (1.0, 5000)]
    for T, mit in schedule:
        if elapsed() > budget_s - 180:

        h = slsqp_run(h, n, target, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:

            h_best = h.copy()

    # Phase 2: Nesterov PG (new)
    if elapsed() < budget_s - 100:

                c5_best = c5_pg
                h_best = h_pg.copy()

    # Phase 3: Fine subgradient
    for K in [5, 3, 2]:
        if elapsed() > budget_s - 80:

        h_sg, c5_sg = fine_subgrad(h_best, n, target, 300, K)

    # Phase 4: Final SLSQP polish at very low T

            scale = 0.005 + 0.003 * trial

                c5_best = c5_sg2
                h_best = h_sg2.copy()

    # Phase 7: Try slightly different n

        for n_try in [n - 4, n + 4]:
            if n_try < 40:
                continue
            if elapsed() > budget_s - 20:
                break
            target_try = n_try / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, h_best)
            h_try = project_feasible(h_try, target_try)
            for T_t in [20, 10, 5, 2]:
                if elapsed() > budget_s - 10:

                h_try = slsqp_run(h_try, n_try, target_try, T_t, 1500)
            c5_try = compute_c5(h_try)
            h_sg3, c5_sg3 = fine_subgrad(h_try, n_try, target_try, 200, 3)
            if c5_sg3 < c5_try:
                c5_try = c5_sg3
                h_try = h_sg3
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
