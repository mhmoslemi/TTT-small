    try:
        res = minimize(sm_obj, h_best, method='SLSQP', jac=sm_grad,
                       bounds=bounds, constraints=[lin_cons],
                       options={'maxiter': 3000, 'ftol': 1e-20})
        if res.fun < 200:
            h_new = project_feasible(res.x.copy(), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-13:
                c5_best = c5_new
                h_best = h_new
    except Exception:
        pass

    h_fin, c5_fin = multi_lag_search(h_best, target, t0 + budget_s - 3,
                                       n_iter=3000, seed=seed + 99)
    if c5_fin < c5_best:
        c5_best = c5_fin
        h_best = h_fin

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
