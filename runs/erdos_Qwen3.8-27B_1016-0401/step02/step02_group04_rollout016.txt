```python
def top_lag_objective(h, top_k, T):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1-h, mode="full")
    # Find top K lags
    idx = np.argpartition(-corr, top_k)[:top_k]
    idx = idx[np.argsort(-corr[idx])]
    weights = np.exp(T * (corr[idx] - corr[idx[0]]))
    weights /= weights.sum()
    # Compute weighted sum
    val = np.sum(weights * corr[idx]) * dx
    # Compute gradient
    grad = np.zeros(n)
    for k, i in enumerate(idx):
        o = i - center
        w = weights[k] * dx
        if o == 0:
            grad += w * (1 - 2*h)
        elif o > 0:
            grad[:n-o] += w * (1 - h[o:])
            grad[o:] -= w * h[:n-o]
        else:
            p = -o
            grad[p:] += w * (1 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return val, grad
```

```python
    t = cp.Variable()

    constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1, t >= 0]
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def corr_at(h, o):
    n = len(h)
    if o == 0:
        return np.sum(h * (1 - h))
    elif o > 0:
        return np.sum(h[:n-o] * (1 - h[o:]))
    else:
        p = -o
        return np.sum(h[p:] * (1 - h[:n-p]))

    n = len(h)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1 - 2*h
    elif o > 0:
        g[:n-o] += (1 - h[o:])
        g[o:] -= h[:n-o]
    else:
        p = -o
        g[p:] += (1 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1-h, mode="full")
    # Top K indices
    top_idx = np.argpartition(-corr, K-1)[:K]
    top_idx = top_idx[np.argsort(-corr[top_idx])]
    # Soft weights
    log_w = T * (corr[top_idx] - corr[top_idx[0]])
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    val = np.sum(w * corr[top_idx]) * dx
    grad = np.zeros(n)

        wk = w[k] * dx

            grad += wk * (1 - 2*h)
        elif o > 0:
            grad[:n-o] += wk * (1 - h[o:])
            grad[o:] -= wk * h[:n-o]
        else:
            p = -o
            grad[p:] += wk * (1 - h[:n-p])
            grad[:n-p] -= wk * h[p:]
    return val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_grad(h, o):
    n = len(h)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1 - 2 * h
    elif o > 0:
        g[:n - o] += (1 - h[o:])
        g[o:] -= h[:n - o]
    else:
        p = -o
        g[p:] += (1 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def top_lag_obj_grad(h, K, T):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argpartition(-corr, K - 1)[:K]
    top_idx = top_idx[np.argsort(-corr[top_idx])]
    log_w = T * (corr[top_idx] - corr[top_idx[0]])
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    val = np.sum(w * corr[top_idx]) * dx
    grad = np.zeros(n)
    for k in range(K):
        o = top_idx[k] - center
        wk = w[k] * dx
        if o == 0:
            grad += wk * (1 - 2 * h)
        elif o > 0:
            grad[:n - o] += wk * (1 - h[o:])
            grad[o:] -= wk * h[:n - o]
        else:
            p = -o
            grad[p:] += wk * (1 - h[:n - p])
            grad[:n - p] -= wk * h[p:]
    return val, grad

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_top_lag(h0, target, K, T, max_iter, step):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        val, g = top_lag_obj_grad(y, K, T)
        zh = project_box_sum(y - step * g, target)
        zv = compute_c5(zh)
        if zv < best_val:
            best_val = zv
            best_h = zh.copy()
        if np.linalg.norm(zh - h) < 1e-14:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        h = zh
        t_k = t_new
    return best_h

class TopLagFunc:
    def __init__(self, K, T, target, pen):
        self.K = K
        self.T = T
        self.target = target
        self.pen = pen
    def __call__(self, x):
        v, _ = top_lag_obj_grad(x, self.K, self.T)
        return v + self.pen * (x.sum() - self.target) ** 2
    def grad(self, x):
        _, g = top_lag_obj_grad(x, self.K, self.T)
        return g + 2 * self.pen * (x.sum() - self.target) * np.ones(len(x))

def lbfgsb_wrap(h0, K, T, target, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = TopLagFunc(K, T, target, pen)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=obj.grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-13})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def lp_alternate(h, n, max_rounds, time_limit):
    dx = 2.0 / n
    t_start = time.time()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) * dx == 1]
        for o in range(-(n - 1), n):
            if o == 0:
                lhs = dx * (h_var @ g)
            elif o > 0:
                lhs = dx * (h_var[:n - o] @ g[o:])
            else:
                p = -o
                lhs = dx * (h_var[:n - p] @ g[p:])
            constraints.append(lhs <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_box_sum(h_new, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-10:
            h = h_new
            c5_cur = c5_new
        else:

                h_mix = project_box_sum(h_mix, n / 2.0)

                if c5_mix < c5_cur:
                    h = h_mix
                    c5_cur = c5_mix
                    break
            else:
                break
    return h
```
