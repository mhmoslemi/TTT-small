import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lag_grad(h, k, n):
    """Contribution of one lag k to gradient of c[k]."""
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-12:
            continue
        grad += w * _lag_grad(h, j - c, n)
    return obj, grad * (2.0 / n)

def wmax_grad(h, m=5, sharp=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    order = np.argsort(corr)[::-1][:m]
    w = np.exp(sharp * (corr[order] - mc))
    w /= w.sum()
    grad = np.zeros(n)
    c = n - 1
    for idx in range(m):
        grad += w[idx] * _lag_grad(h, order[idx] - c, n)
    return grad * dx, mc * dx

def pg_optimize(h0, target, t0, budget, T_list, lr_base, tag="sm"):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for T in T_list:
        if time.time() - t0 > budget - 10:
            break
        lr = lr_base
        if tag == "sm":
            obj_cur, _ = sm_obj_grad(h, T)
        else:
            _, obj_cur = wmax_grad(h)
        for it in range(3000):
            if time.time() - t0 > budget - 10:
                break
            if tag == "sm":
                obj_cur, grad = sm_obj_grad(h, T)
            else:
                grad, obj_cur = wmax_grad(h)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            for _ in range(50):
                h_new = project_constraint(h - lr * grad, target)
                if tag == "sm":
                    obj_new, _ = sm_obj_grad(h_new, T)
                else:
                    _, obj_new = wmax_grad(h_new)
                if obj_new < obj_cur - 1e-16:
                    break
                lr *= 0.5
            else:
                lr = 0.01
                h_new = project_constraint(h - lr * grad, target)
            h = h_new
            lr = min(lr * 1.06, lr_base)
            if it % 300 == 299:
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-11:
                    best_c5 = c5
                    best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _scl_f(x, target, n):
    return x.sum() - target

def _scl_jac(x):
    return np.ones(len(x))

def _scl_obj(x, T):
    return sm_obj_grad(x, T)[0]

def _scl_grad(x, T):
    return sm_obj_grad(x, T)[1]

def scl_refine(h0, target, T, maxiter, t0, budget):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': _scl_jac}]
    try:
        def f(x):
            return sm_obj_grad(x, T)[0]
        def g(x):
            return sm_obj_grad(x, T)[1]
        res = minimize(f, h0, method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n0 = len(h_best)
    target0 = n0 / 2.0
    h_best = project_constraint(h_best, target0)
    c5_best = compute_c5(h_best)

    # Phase 1: Soft-max descent with decreasing T (smooth → sharp)
    if time.time() - t0 < budget - 200:
        h_best, c5 = pg_optimize(h_best, target0, t0, budget - 200,
                                 [400, 150, 60, 25, 12], 0.04, "sm")
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 2: Weighted-max subgradient (top-5 lags)
    if time.time() - t0 < budget - 100:
        h_best, c5 = pg_optimize(h_best, target0, t0, budget - 100,
                                 [10, 10, 10], 0.025, "wm")
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 3: SLSQP fine-tune
    if time.time() - t0 < budget - 60:
        h_s, c5_s = scl_refine(h_best, target0, 80, 300, t0, budget - 30)
        if c5_s < c5_best:
            c5_best, h_best = c5_s, h_s

    # Phase 4: Diverse restarts
    for r in range(30):
        if time.time() - t0 > budget - 30:
            break
        hp = h_best.copy()
        m = r % 6
        if m == 0:
            idx = rng.integers(0, n0)
            sz = rng.integers(6, np.max([7, n0 // 10]))
            for i in range(sz):
                hp[(idx + i) % n0] = 1.0 - hp[(idx + i) % n0]
        elif m == 1:
            hp = np.clip(h_best + rng.normal(0, 0.04, n0), 0, 1)
        elif m == 2:
            thr = rng.uniform(0.3, 0.7)
            hp = (h_best > thr).astype(float)
        elif m == 3:
            s = rng.integers(2, 20)
            hp = np.roll(h_best, s)
        elif m == 4:
            # Block swap
            a = rng.integers(0, n0 - 20)
            b = rng.integers(a + 20, n0)
            ln = rng.integers(5, 15)
            tmp = hp[a:a+ln].copy()
            hp[a:a+ln] = hp[b:b+ln]
            hp[b:b+ln] = tmp
        else:
            # Local inversion
            idx = rng.integers(0, n0 - 10)
            ln = rng.integers(4, 12)
            hp[idx:idx+ln] = 1.0 - hp[idx:idx+ln]
        hp = project_constraint(hp, target0)
        # Short optimize from this perturbation
        T_r = 30 + (r % 4) * 20
        hr, c5_r = pg_optimize(hp, target0, t0, budget - 15,
                               [T_r, 15], 0.035, "sm")
        if c5_r < c5_best - 1e-7:
            c5_best, h_best = c5_r, hr
            # Follow up with weighted-max
            if time.time() - t0 < budget - 20:
                h_best, c5_r2 = pg_optimize(h_best, target0, t0,
                                            budget - 10, [10], 0.02, "wm")
                if c5_r2 < c5_best:
                    c5_best, h_best = c5_r2, h_best

    # Phase 5: Final polish - very low T soft-max then weighted-max
    if time.time() - t0 < budget - 15:
        h_best, c5 = pg_optimize(h_best, target0, t0, budget - 5,
                                 [8, 5, 3], 0.015, "sm")
        if c5 < c5_best:
            c5_best, h_best = c5, h

    if time.time() - t0 < budget - 5:
        h_best, c5 = pg_optimize(h_best, target0, t0, budget - 1,
                                 [5], 0.01, "wm")
        if c5 < c5_best:
            c5_best, h_best = c5, h

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
