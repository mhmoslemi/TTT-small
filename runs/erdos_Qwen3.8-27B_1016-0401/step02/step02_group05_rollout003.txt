    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

```python
def project_box_sum(h, target, lb=0.0, ub=1.0):
    """Project h onto {x ∈ [lb,ub]^n : sum(x) = target}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -100.0, 100.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h + mid, lb, ub)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + (lo + hi) / 2, lb, ub)
    # Tiny residual fix
    diff = target - x.sum()
    if abs(diff) < 1e-10:
        return x
    # Distribute remaining
    free = (x > lb + 1e-12) & (x < ub - 1e-12)
    if free.sum() > 0:
        x[free] += diff / free.sum()
    else:
        x += diff / n
    return np.clip(x, lb, ub)
```

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def project_onto_box_sum(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h + mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.sum() > 0:
            x[free] += diff / free.sum()
        else:
            x += diff / len(x)
    return np.clip(x, 0.0, 1.0)

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def enhanced_local_search(h, target, n_iter, t0, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    h_curr = h.copy()
    stall = 0
    base_delta = 0.08

    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            base_delta *= 0.7
            base_delta = max(base_delta, 0.01)
            stall = 0
        delta = base_delta
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        best_c5_try = best_c5
        best_h_try = None
        n_cand = 8

        for ci in range(n_cand):
            h_try = h_curr.copy()
            d = delta * (1.0 + 0.3 * rng.random())
            if p == 0:
                vals = h_curr * (1.0 - h_curr)
                if ci < 4:
                    top = np.argsort(vals)[-4:]
                    idx = top[ci]
                else:
                    idx = int(rng.integers(n))
                    j2 = (idx + 1 + ci) % n
                j2 = (idx + 1 + ci) % n
                if h_curr[idx] > 0.5:
                    h_try[idx] = max(0.0, h_curr[idx] - d)
                    h_try[j2] = min(1.0, h_curr[j2] + d)
                else:
                    h_try[idx] = min(1.0, h_curr[idx] + d)
                    h_try[j2] = max(0.0, h_curr[j2] - d)
            else:
                if s > 0:
                    contrib = h_curr[p:] * (1.0 - h_curr[:n - p])
                else:
                    contrib = h_curr[:n - p] * (1.0 - h_curr[p:])
                if ci < 5:
                    top = np.argsort(contrib)[-5:]
                    idx = top[ci]
                else:
                    idx = int(rng.integers(len(contrib)))
                if s > 0:
                    j1, j2 = idx + p, idx
                else:
                    j1, j2 = idx, idx + p
                h_try[j1] = max(0.0, h_curr[j1] - d)
                h_try[j2] = min(1.0, h_curr[j2] + d)
            h_try = fix_constraint(h_try, target)
            corr_try = np.correlate(h_try, 1 - h_try, mode="full")
            c5_try = float(np.max(corr_try) * dx)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_h_try = h_try

        if best_h_try is not None and best_c5_try < best_c5 - 1e-14:
            best_c5 = best_c5_try
            h_curr = best_h_try
            corr = np.correlate(h_curr, 1 - h_curr, mode="full")
            stall = 0
        else:
            stall += 1
    return h_curr, best_c5

def pgd_phase(h, target, t0, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    c5_best = compute_c5(h_curr)
    h_best_sol = h_curr.copy()
    lr = 0.02
    lmbda = 5.0
    stall = 0
    while time.time() < deadline:
        grad = soft_max_jac(h_curr, lmbda)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        alpha = lr / max(gnorm, 1e-8) * 0.5
        h_new = h_curr - alpha * grad
        h_new = project_onto_box_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-13:
            c5_best = c5_new
            h_best_sol = h_new.copy()
            h_curr = h_new
            stall = 0
            lr = min(lr * 1.1, 0.05)
        else:
            h_curr = h_best_sol + rng.normal(0, 0.01, n)
            h_curr = fix_constraint(h_curr, target)
            stall += 1
            lr *= 0.8
            if stall > 50:
                lmbda *= 0.7
                lr = 0.03
                stall = 0
    return h_best_sol, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    bounds = [(0.0, 1.0)] * n

    temps = [60, 30, 15, 8, 5]

        try:

            if res.fun < 50:
