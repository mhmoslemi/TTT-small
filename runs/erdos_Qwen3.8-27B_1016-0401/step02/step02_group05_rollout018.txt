def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

        if abs(diff) < 1e-13:
            break
        if diff > 0:

                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s

            order = np.argsort(-h)
            for i in order:

                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s

    h_new = np.clip(h_new, 0, 1)

            break
        grad = grad / gnorm
        alpha = step
        f0 = compute_c5(h_c)
        found = False
        for _ in range(15):
            h_try = h_c - alpha * grad
            h_try = np.clip(h_try, 0, 1)
            h_try = fix_constraint(h_try, target)
            f1 = compute_c5(h_try)
            if f1 < f0 - 1e-16:
                h_c = h_try
                if f1 < best:
                    best = f1
                found = True
                break
            alpha *= 0.5
        if found:
            step = min(step * 1.04, 0.5)
            stall = 0
        else:
            step *= 0.55
            stall += 1
            if step < 1e-10 or stall > 200:
                break
    return h_c, best

def two_opt(h, target, n_iter, t0, budget, delta0=0.05):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    corr = np.correlate(h_c, 1 - h_c, mode="full")
    best = float(np.max(corr) * dx)
    delta = delta0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 120:
            delta *= 0.5
            if delta < 3e-4:
                break
            stall = 0
        k = np.argmax(corr)
        s = k - (n - 1)
        p = abs(s)
        h_t = h_c.copy()
        if p == 0:
            v = h_c * (1 - h_c)
            i = int(np.argmax(v))
            h_t[i] -= delta * (2 * h_c[i] - 1)
        else:
            if s > 0:
                c = h_c[p:] * (1 - h_c[:n - p])
            else:
                c = h_c[:n - p] * (1 - h_c[p:])
            i = int(np.argmax(c))
            if s > 0:
                h_t[i + p] -= delta
                h_t[i] += delta
            else:
                h_t[i] -= delta
                h_t[i + p] += delta
        h_t = np.clip(h_t, 0, 1)
        h_t = fix_constraint(h_t, target)
        ct = np.correlate(h_t, 1 - h_t, mode="full")
        c5t = float(np.max(ct) * dx)
        if c5t < best - 1e-14:
            best = c5t
            h_c = h_t
            corr = ct
            stall = 0
        else:
            stall += 1
    return h_c, best

def swap_search(h, target, n_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    corr = np.correlate(h_c, 1 - h_c, mode="full")
    best = float(np.max(corr) * dx)
    rng = np.random.default_rng(12345)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        k = int(np.argmax(corr))
        s = k - (n - 1)
        p = abs(s)
        if p == 0:
            i = int(np.argmax(h_c * (1 - h_c)))
            h_t = h_c.copy()
            h_t[i] = 1.0 - h_t[i]
        else:
            i1 = int(rng.integers(0, max(1, n - 2)))
            i2 = int(rng.integers(0, max(1, n - 2)))
            h_t = h_c.copy()
            h_t[i1], h_t[i2] = h_t[i2], h_t[i1]
        h_t = np.clip(h_t, 0, 1)
        h_t = fix_constraint(h_t, target)
        ct = np.correlate(h_t, 1 - h_t, mode="full")
        c5t = float(np.max(ct) * dx)
        if c5t < best - 1e-14:
            best = c5t
            h_c = h_t
            corr = ct
    return h_c, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(sum_eq_fun, target=target),
             'jac': partial(sum_eq_jac, target=target)}]

    # Phase 1: Gradient flow with temperature annealing
    for T in [40, 20, 10, 5, 2.5, 1.2, 0.6]:
        if time.time() - t0 > budget_s - 550:
            break
        h_g, c5_g = gradient_flow(h_best, target, T, 600, 0.04, t0, budget_s - 530)
        if c5_g < c5_best - 1e-13:
            c5_best = c5_g
            h_best = h_g

    # Phase 2: 2-opt local search
    h_2, c5_2 = two_opt(h_best, target, 4000, t0, budget_s - 420)
    if c5_2 < c5_best:
        c5_best = c5_2
        h_best = h_2

    # Phase 3: SLSQP refinement
    for T in [6, 3, 1.5]:
        if time.time() - t0 > budget_s - 250:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                           jac=soft_max_jac, args=(T,),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-17})
            if res.fun < 100:
                h_n = fix_constraint(res.x.copy(), target)
                c = compute_c5(h_n)
                if c < c5_best - 1e-13:
                    c5_best = c
                    h_best = h_n
        except Exception:
            pass

    # Phase 4: 2-opt + swap search
    h_2b, c5_2b = two_opt(h_best, target, 3000, t0, budget_s - 200)
    if c5_2b < c5_best:
        c5_best = c5_2b
        h_best = h_2b
    h_sw, c5_sw = swap_search(h_best, target, 1000, t0, budget_s - 170)
    if c5_sw < c5_best:
        c5_best = c5_sw
        h_best = h_sw

    # Phase 5: Multi-restart refinement
    nr = 0
    while time.time() - t0 < budget_s - 100:
        nr += 1
        sig = max(0.008, 0.035 * np.exp(-nr * 0.35))
        h_r = h_best + rng.normal(0, sig, n)
        h_r = fix_constraint(h_r, target)
        h_g2, c5_g2 = gradient_flow(h_r, target, 5, 250, 0.015, t0, budget_s - 80)
        if c5_g2 < c5_best - 1e-13:
            c5_best = c5_g2
            h_best = h_g2
        h_2c, c5_2c = two_opt(h_best, target, 600, t0, budget_s - 50)
        if c5_2c < c5_best:
            c5_best = c5_2c
            h_best = h_2c

    # Phase 6: Final fine gradient polish
    h_gp, c5_gp = gradient_flow(h_best, target, 0.8, 800, 0.004, t0, budget_s - 15)
    if c5_gp < c5_best - 1e-14:
        c5_best = c5_gp
        h_best = h_gp

    # Phase 7: Final 2-opt
    h_2f, c5_2f = two_opt(h_best, target, 800, t0, max(t0 + 1, budget_s - 1), delta0=0.02)
    if c5_2f < c5_best:
        c5_best = c5_2f
        h_best = h_2f

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
