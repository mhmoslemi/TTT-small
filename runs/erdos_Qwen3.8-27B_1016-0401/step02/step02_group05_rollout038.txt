import time

    corr = np.correlate(h, 1-h, mode="full")

    n = len(h)
    g = np.zeros(n)
    if s > 0:
        g[:s] = -h[s:2*s]
        if n > 2*s:
            g[s:n-s] = (1-h[:n-2*s]) - h[s:n-s]
        g[n-s:] = 1-h[:s]
    elif s < 0:

        g[:p] = 1-h[p:2*p]
        if n > 2*p:

        g[n-p:] = -h[:p]
    else:
        g[:] = 1-2*h
    return g

    n = len(x)

    corr = np.correlate(x, 1-x, mode="full")

    shifted = T * (corr - mc)

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1-x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase sum: add to elements with most room
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, room[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(-h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def sum_eq_fun(x, target):
    return x.sum() - target

def proj_subgrad(h, target, n_iter=10000, t0=None, budget=None):
    """Projected subgradient descent for the max correlation."""
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    lr = 0.5 / n  # initial learning rate (scaled by 1/n for the correlation)
    lr_decay = 0.9999
    moment = 0.9
    
    for it in range(n_iter):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget:
                break
        
        corr = np.correlate(h_curr, 1-h_curr, mode="full")
        m = np.argmax(corr)
        s = m - (n-1)
        g = corr_grad(h_curr, s) * dx
        
        # Step
        h_new = h_curr - lr * g
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_constraint(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_curr = h_new
        lr *= lr_decay
    
    return best_h, best_c5

def block_flip_search(h, target, t0=None, budget=None, n_iter=5000):
    """Local search with block flips and pair moves."""
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    rng = np.random.default_rng(123)
    stall = 0
    
    for it in range(n_iter):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget:
                break
        
        if stall > 500:
            # Random restart within nearby solutions
            h_curr = best_h + rng.normal(0, 0.05, n)
            h_curr = fix_constraint(h_curr, target)
            stall = 0
        
        move_type = it % 3
        if move_type == 0:
            # Block flip
            length = int(rng.integers(2, min(15, n//4)))
            start = int(rng.integers(0, n - length))
            h_try = h_curr.copy()
            h_try[start:start+length] = 1.0 - h_try[start:start+length]
        elif move_type == 1:
            # Pair move: shift mass from i to j
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = 0.1 * rng.random() + 0.05
            h_try = h_curr.copy()
            h_try[i] -= delta
            h_try[j] += delta
        else:
            # Gradient-based move at the max lag
            corr = np.correlate(h_curr, 1-h_curr, mode="full")
            m = np.argmax(corr)
            s = m - (n-1)
            g = corr_grad(h_curr, s)
            # Move in -gradient direction
            step = 0.05
            h_try = h_curr - step * g
            h_try = np.clip(h_try, 0, 1)
        
        h_try = fix_constraint(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_curr = h_try
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5
