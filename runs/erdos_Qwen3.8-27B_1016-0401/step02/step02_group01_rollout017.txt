        grad += w * lag_gradient(h, j - c)
    return obj, grad * dx

def nesterov_optimize(h0, target, T, n_iter, lr0, t0, budget, beta=0.9):
    n = len(h)
    h = h0.copy()
    y = h.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        t_it = it + 1
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * t_it * t_it)) / 2.0
        y = h + (beta_t - 1.0) / (beta_t + 1.0) * (h - y) if it > 0 else h.copy()
        obj, grad = sm_obj_grad(y, T)
        lr_t = lr / (1.0 + 0.01 * it)
        step = h - lr_t * grad
        for trial in range(25):
            h_new = project_constraint(step, target)
            obj_new, _ = sm_obj_grad(h_new, T)
            if obj_new < obj - 1e-14:
                break
            lr_t *= 0.5
            step = h - lr_t * grad
        else:
            h_new = project_constraint(h - 0.3 * lr_t * grad, target)
        h = h_new
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def plain_optimize(h0, target, T, n_iter, lr0, t0, budget):
    n = len(h)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        obj, grad = sm_obj_grad(h, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        for trial in range(30):
            h_new = project_constraint(h - lr * grad, target)
            obj_new, _ = sm_obj_grad(h_new, T)
            if obj_new < obj - 1e-15:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        lr = min(lr * 1.05, lr0)
        if it % 100 == 99:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def binary_round(h, target):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    diff = int(round(h_bin.sum() - target))
    if diff > 0:
        idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[idx])
        for i in range(diff):
            h_bin[idx[order[i]]] = 0.0
    elif diff < 0:
        idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[idx])
        for i in range(-diff):
            h_bin[idx[order[i]]] = 1.0
    return h_bin

def sorry_slsqp(h0, target, T, maxiter, t0, budget):
    n = len(h0)
    gT, gN = T, n
    def _f(x):
        return sm_obj_grad(x, gT)[0]
    def _g(x):
        return sm_obj_grad(x, gT)[1]
    def _c(x):
        return x.sum() - target
    def _cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c, 'jac': _cj}]
    try:
        res = minimize(_f, h0, method='SLSQP', jac=_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 50:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(560)
        h_best[::2] = 1.0
    n0 = len(h_best)
    target0 = n0 / 2.0
    h_best = project_constraint(h_best, target0)
    c5_best = compute_c5(h_best)

    # Phase 1: Curriculum smooth-max with Nesterov
    for T in [15, 40, 100, 250, 500, 1000]:
        if time.time() - t0 > budget - 350:
            break
        h_best, c5 = nesterov_optimize(h_best, target0, T, 1500, 0.025, t0, budget - 300)
        if c5 < c5_best:
            c5_best = c5

    # Phase 2: Binary rounding + reoptimize
    for r in range(15):
        if time.time() - t0 > budget - 250:
            break
        noise = rng.normal(0, 0.08, n0)
        h_b = binary_round(h_best + noise, target0)
        h_b, c5 = plain_optimize(h_b, target0, 300, 1000, 0.02, t0, budget - 200)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_b.copy()

    # Phase 3: Diverse perturbations
    for r in range(50):
        if time.time() - t0 > budget - 120:
            break
        h_p = h_best.copy()
        s = r % 6
        if s == 0:
            idx = int(rng.integers(0, n0))
            sz = int(rng.integers(5, 50))
            h_p[(idx + np.arange(sz)) % n0] = 1.0 - h_best[(idx + np.arange(sz)) % n0]
        elif s == 1:
            h_p = np.clip(h_best + rng.normal(0, 0.04, n0), 0, 1)
        elif s == 2:
            h_p = (h_best > rng.uniform(0.35, 0.65)).astype(float)
        elif s == 3:
            sh = int(rng.integers(1, 25))
            h_p = np.roll(h_best, sh)
        elif s == 4:
            swap_n = int(rng.integers(3, 15))
            idx = rng.permutation(n0)[:swap_n]
            for a, b in zip(idx[0::2], idx[1::2]):
                h_p[a], h_p[b] = h_best[b], h_best[a]
        else:
            h_p = 1.0 - h_best[::-1].copy()
        h_p = project_constraint(h_p, target0)
        T_p = 80 + (r % 20) * 30
        h_p, c5_p = nesterov_optimize(h_p, target0, T_p, 600, 0.02, t0, budget - 80)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()

    # Phase 4: SLSQP polish
    if time.time() - t0 < budget - 60:
        h_s, c5_s = sorry_slsqp(h_best, target0, 500, 200, t0, budget - 30)
        if c5_s < c5_best:
            c5_best = c5_s
            h_best = h_s

    # Phase 5: Coarse resolution n=280
    if time.time() - t0 < budget - 80:
        nc = 280
        tc = nc / 2.0
        x_old = np.arange(n0) * (2.0 / n0)
        x_c = np.linspace(0.0, 2.0, nc, endpoint=False)
        h_c = np.interp(x_c, x_old, h_best)
        h_c = project_constraint(h_c, tc)
        for T_c in [50, 200, 600]:
            if time.time() - t0 > budget - 30:
                break
            h_c, c5_c = nesterov_optimize(h_c, tc, T_c, 1200, 0.03, t0, budget - 20)
            if c5_c < c5_best:
                c5_best = c5_c
                h_best = h_c

    # Phase 6: Final high-T micro-polish
    nf = len(h_best)
    tf = nf / 2.0
    if time.time() - t0 < budget - 15:
        h_best, c5_f = plain_optimize(h_best, tf, 2000, 500, 0.003, t0, budget - 5)
        if c5_f < c5_best:
            c5_best = c5_f

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
