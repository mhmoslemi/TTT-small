import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def update_corr(corr, h, i, delta):
    n = len(h)
    o = np.arange(-(n - 1), n)
    valid1 = (i + o >= 0) & (i + o < n)
    valid2 = (i - o >= 0) & (i - o < n)
    idx1 = i + o[valid1]
    idx2 = i - o[valid2]
    corr[valid1] += delta * (1.0 - h[idx1])
    corr[valid2] += delta * (-h[idx2])

def coord_optimize_point(h, i, corr, max_bi=60):
    n = len(h)
    center = n - 1
    o = np.arange(-(n - 1), n)
    valid1 = (i + o >= 0) & (i + o < n)
    valid2 = (i - o >= 0) & (i - o < n)
    idx1 = i + o[valid1]
    idx2 = i - o[valid2]
    grad = np.zeros(2 * n - 1)
    grad[valid1] += (1.0 - h[idx1])
    grad[valid2] += (-h[idx2])
    b = corr - grad * h[i]
    a = grad
    vals_0 = np.max(b)
    vals_1 = np.max(a + b)
    best_t = 0.0 if vals_0 < vals_1 else 1.0
    best_val = min(vals_0, vals_1)
    lo, hi = 0.0, 1.0
    for _ in range(max_bi):
        mid = 0.5 * (lo + hi)
        val = np.max(a * mid + b)
        if val < best_val:
            best_val = val
            best_t = mid
        kmax = np.argmax(a * mid + b)
        if a[kmax] > 1e-15:
            hi = mid
        elif a[kmax] < -1e-15:
            lo = mid
        else:
            break
    return best_t

def lag_obj(x, lag_idx):
    n = len(x)
    return np.dot(x, 1.0 - np.roll(x, lag_idx - (n - 1))) * (2.0 / n)

def lag_obj2(x, lag_idx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    return corr[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += (1.0 - x[:n - o])
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += (1.0 - x[p:])
        g[p:] += -x[:n - p]
    return g * dx

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def eq_fun(x, target):
    return x.sum() - target

def eq_jac(x, target):
    return np.ones(len(x))
