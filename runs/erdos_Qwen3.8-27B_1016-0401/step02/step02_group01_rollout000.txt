import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _sm_eval(x, T, target, penalty):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    base = (mc + np.log(se) / T) * (2.0 / n)
    s = x.sum() - target
    return base + penalty * s * s, sw if (sw := e / se) is not None else e / se

def sm_obj(x, T, target, penalty):
    return _sm_eval(x, T, target, penalty)[0]

def sm_grad(x, T, target, penalty):
    n = len(x)
    _, sw = _sm_eval(x, T, target, penalty)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            grad[k:] += w * (1.0 - x[:n - k])
            grad[:n - k] -= w * x[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    grad *= (2.0 / n)
    s = x.sum() - target
    grad += 2.0 * penalty * s
    return grad

def lbfgs_solve(h0, target, T, penalty, maxiter, t0, budget):
    n = len(h0)
    def fun(x):
        return sm_obj(x, T, target, penalty)
    def jac(x):
        return sm_grad(x, T, target, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': maxiter * 3})
        if res.fun < 1e10:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def pg_solve(h0, target, T, n_iter, lr, t0, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n = len(h)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        _, sw = _sm_eval(h, T, target, 0.0)
        grad = np.zeros(n)
        c = n - 1
        for j in range(2 * n - 1):
            w = sw[j]
            if w < 1e-15:
                continue
            k = j - c
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            elif k > 0:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
            else:
                p = -k
                grad[:n - p] += w * (1.0 - h[p:])
                grad[p:] -= w * x[:n - p] if False else w * h[:n - p]
        grad *= (2.0 / n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        obj_old = sm_obj(h, T, target, 0.0)
        step = lr
        h_new = h
        for _ in range(35):
            h_new = project_constraint(h - step * grad, target)
            obj_new = sm_obj(h_new, T, target, 0.0)
            if obj_new < obj_old - 1e-15 or step < 1e-12:
                break
            step *= 0.5
        h = h_new
        lr = min(step * 1.05, 0.1)
        if it % 60 == 59:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def binary_ls(h0, target, n_trials, t0, budget, rng):
    h = (h0 > 0.5).astype(float)
    h = project_constraint(h, target)
    best_c5 = compute_c5(h)
    n = len(h)
    for _ in range(n_trials):
        if time.time() - t0 > budget:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        i, j = ones[i], zeros[j]
        h[i], h[j] = 1.0, 0.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
        else:
            h[i], h[j] = 0.0, 1.0
    return h, best_c5

def upsample(h0, n_new):
    n_old = len(h0)
    target = n_new / 2.0
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h0, assume_sorted=True)
    return project_constraint(h_new, target)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h
