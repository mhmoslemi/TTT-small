    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def subgradient_c5(h):
    """Compute subgradient of max_k corr(k) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    # Find active constraints (within tolerance)
    tol = 1e-6 * max(1, mc)
    active = np.where(corr >= mc - tol)[0]
    center = n - 1
    grad = np.zeros(n)
    for idx in active:
        o = idx - center
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n-o])
            grad[:n-o] += (-h[o:])
        else:
            p = -o
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] += (-h[:n-p])
    return grad * dx, mc * dx

def fix_sum(h, target):
    """Project h onto {sum(h) = target} ∩ [0,1]^n"""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: add to cells with room
            room = 1.0 - h
            order = np.argsort(-room)  # most room first
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, room[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(h)  # smallest first
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def subgrad_at_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    center = n - 1
    tol = 1e-7 * max(1.0, abs(mc))
    active = np.where(corr >= mc - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        o = int(idx - center)
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] += (-h[o:])
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] += (-h[:n - p])
    return grad * dx, mc * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            idx = np.argsort(h)
            for i in idx:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def momentum_descent(h, target, t0, budget, n_iter=5000):
    n = len(h)
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    m = np.zeros(n)
    beta = 0.8
    lr = 0.03
    rng = np.random.default_rng(77)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        grad, c5 = subgrad_at_max(h_curr)
        m = beta * m + (1 - beta) * grad
        accepted = False
        for _ in range(15):
            step = -lr * m
            h_try = h_curr + step
            h_try = fix_sum(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                h_curr = h_try
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            lr = max(0.001, lr * 2.0)
            if it % 200 == 100:
                h_curr = fix_sum(h_curr + 0.005 * rng.standard_normal(n), target)
    return h_curr, best_c5

def local_search(h, target, n_iter=6000, t0=None, budget=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    h_curr = h.copy()
    stall = 0
    delta = 0.08
    rng = np.random.default_rng(42)
    for it in range(n_iter):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        if stall > 300:
            delta = max(0.003, delta * 0.6)
            stall = 0
        k_star = np.argmax(corr)
        s = int(k_star - (n - 1))
        p = abs(s)
        best_local = None
        best_local_c5 = best_c5
        if p > 0:
            if s > 0:
                contrib = h_curr[p:] * (1 - h_curr[:n - p])
            else:
                contrib = h_curr[:n - p] * (1 - h_curr[p:])
            top_k = min(8, n - p)
            top = np.argsort(-contrib)[:top_k]
            for idx in top:
                if s > 0:
                    j1, j2 = idx + p, idx
                else:
                    j1, j2 = idx, idx + p
                for d_frac in (1.0, 0.5, 0.25):
                    d = delta * d_frac
                    h_try = h_curr.copy()
                    h_try[j1] -= d
                    h_try[j2] += d
                    h_try = np.clip(h_try, 0, 1)
                    h_try = fix_sum(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_local_c5 - 1e-14:
                        best_local_c5 = c5_try
                        best_local = h_try
            if best_local is None and len(top) >= 2:
                for a in range(min(3, len(top))):
                    for b in range(a + 1, min(5, len(top))):
                        if s > 0:
                            ja1, ja2 = top[a] + p, top[a]
                            jb1, jb2 = top[b] + p, top[b]
                        else:
                            ja1, ja2 = top[a], top[a] + p
                            jb1, jb2 = top[b], top[b] + p
                        d = delta * 0.4
                        h_try = h_curr.copy()
                        h_try[ja1] -= d
                        h_try[ja2] += d
                        h_try[jb1] -= d
                        h_try[jb2] += d
                        h_try = np.clip(h_try, 0, 1)
                        h_try = fix_sum(h_try, target)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_local_c5 - 1e-14:
                            best_local_c5 = c5_try
                            best_local = h_try
        if best_local is not None:
            best_c5 = best_local_c5
            h_curr = best_local
            corr = np.correlate(h_curr, 1 - h_curr, mode="full")
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                h_try = h_curr.copy()
                for _ in range(rng.integers(2, 5)):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    h_try[i] -= delta * 0.2
                    h_try[j] += delta * 0.2
                h_try = np.clip(h_try, 0, 1)
                h_try = fix_sum(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 + 0.0005:
                    h_curr = h_try
                    corr = np.correlate(h_curr, 1 - h_curr, mode="full")
    return h_curr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(sum_eq_fun, target=target),
             'jac': partial(sum_eq_jac, target=target)}]

    # Phase 1: Momentum gradient descent (80s)
    h_gd, c5_gd = momentum_descent(h_best, target, t0, 80, n_iter=3000)
    if c5_gd < c5_best:
        c5_best = c5_gd
        h_best = h_gd

    # Phase 2: SLSQP temperature annealing (250s)
    temps = [80, 40, 20, 10, 6, 4, 2.5, 1.5, 1.0, 0.7, 0.4]
    for T in temps:
        if elapsed() > budget_s - 450:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2500, 'ftol': 1e-15})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: Local search (200s)
    h_ls, c5_ls = local_search(h_best, target, n_iter=10000, t0=t0, budget=budget_s - 350)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 4: Another momentum round (80s)
    h_gd2, c5_gd2 = momentum_descent(h_best, target, t0, budget_s - 250, n_iter=2000)
    if c5_gd2 < c5_best:
        c5_best = c5_gd2
        h_best = h_gd2

    # Phase 5: Random restarts with short optimization
    n_restarts = 0
    while elapsed() < budget_s - 120:
        n_restarts += 1
        r = n_restarts
        if r % 4 == 1:
            h_init = h_best + rng.normal(0, 0.02, n)
        elif r % 4 == 2:
            h_init = h_best.copy()
            seg = int(rng.integers(5, 25))
            st = int(rng.integers(0, max(1, n - seg)))
            h_init[st:st + seg] = 1.0 - h_init[st:st + seg]
        elif r % 4 == 3:
            h_init = h_best.copy()
            for _ in range(int(rng.integers(2, 5))):
                seg = int(rng.integers(3, 12))
                st = int(rng.integers(0, max(1, n - seg)))
                h_init[st:st + seg] = 1.0 - h_init[st:st + seg]
        else:
            h_init = (rng.random(n) >= 0.5).astype(float)
        h_init = fix_sum(h_init, target)
        T = 8
        try:
            res = minimize(soft_max_obj, h_init, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-13})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass
        h_ls2, c5_ls2 = local_search(h_best, target, n_iter=400, t0=t0, budget=budget_s - 100)
        if c5_ls2 < c5_best:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 6: Final SLSQP polish
    for T in [1.5, 0.8, 0.3]:
        if elapsed() > budget_s - 15:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-17})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Final local search
    h_ls3, c5_ls3 = local_search(h_best, target, n_iter=3000, t0=t0, budget=budget_s - 2)
    if c5_ls3 < c5_best:
        c5_best = c5_ls3
        h_best = h_ls3

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
