import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    s = np.sum(np.exp(T * (corr - mc)))
    return (mc + np.log(s) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for k in range(len(sw)):
        w = sw[k]
        if w < 1e-15:
            continue
        o = k - center
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < o < n:
            grad[:n - o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n - o]
        elif -n < o < 0:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def lbfgs_opt(h, target, T, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

        return softmax_obj(x, T) + 300.0 * (x.sum() - target) ** 2
    def grad(x):
        return softmax_grad(x, T) + 600.0 * (x.sum() - target)
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-15, 'maxls': 50})
        return project_box_sum(res.x, target)
    except Exception:
        return h

def fista_opt(h, target, T, max_iter, step):
    n = len(h)
    hc = h.copy()
    z = h.copy()
    t_k = 1.0
    for _ in range(max_iter):
        y = hc + ((t_k - 1.0) / (t_k + 2.0)) * (hc - z)
        g = softmax_grad(y, T)
        z = project_box_sum(y - step * g, target)
        if np.linalg.norm(z - hc) < 1e-14:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        hc = z
        t_k = t_new
    c1, c2 = compute_c5(hc), compute_c5(z)
    return z if c2 < c1 else hc

def coord_refine(h, target, n_trials=200):
    n = len(h)
    best = h.copy()

    frac = np.where((h > 0.01) & (h < 0.99))[0]

    rng = np.random.default_rng(12345)

                best = h_try
                frac = np.where((best > 0.01) & (best < 0.99))[0]

                    return best
    return best
