def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    # Numerical stability
    if shifted.max() > 700:
        shifted = shifted - shifted.max()
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    if shifted.max() > 700:
        shifted = shifted - (shifted.max() - np.log(np.sum(np.exp(shifted))))
    e = np.exp(shifted - shifted.max())
    sw = e / e.sum()

        if w < 1e-10:

        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n-p]
    return grad * dx

def sm_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    a = shifted - shifted.max()
    logsum = shifted.max() + np.log(np.exp(a).sum())
    val = mc + logsum / T
    return val * dx

def sm_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    a = shifted - shifted.max()
    e = np.exp(a)
    sw = e / e.sum()

        w = sw[c + o]
        if w < 1e-10:
            continue

            grad += w * (1.0 - 2.0 * x)

            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] -= w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fast_fix(h_best, target)
    return (h_best, c5_final, len(h_best))
