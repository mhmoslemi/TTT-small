This is related to the "minimum maximum autocorrelation" problem for binary sequences. Known constructions (like Sobel sequences, Barker sequences) achieve low autocorrelation, but we're looking at a different quantity (cross-correlation with the complement).

Actually, let me reconsider. For a binary sequence h with n/2 ones:
corr(o) = Σ_{i=0}^{n-1-o} h[i](1-h[i+o]) for o ≥ 0

This equals (# of 1s in positions 0..n-1-o) - (# of 1s in both positions i and i+o for i=0..n-1-o).

Let A(o) = # of 1s in positions 0..n-1-o and B(o) = # of pairs (i, i+o) both being 1.
Then corr(o) = A(o) - B(o).

To minimize the max of corr(o), we want:
1. A(o) to be small → 1s should be in later positions (but this doesn't help for small o)
2. B(o) to be large → 1s should be clustered (so that h[i]=1 implies h[i+o]=1)

But these two are somewhat in tension. The optimal arrangement is non-trivial.

Actually, one more important insight: the gradient of the max is NOT well-defined when multiple lags achieve the max. In that case, the subdifferential is the convex hull of the individual gradients. import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def grad_lags(h, n, lags):
    g = np.zeros(n)
    for o in lags:
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g

def subgrad_line_search(h, n, target, iters, time_budget, K_init=4, K_max=20):
    t_s = time.time()
    h_c = h.copy()
    c5_c = compute_c5(h_c)
    h_b = h_c.copy()
    c5_b = c5_c
    center = n - 1
    v = np.zeros(n)
    beta = 0.85
    for it in range(iters):
        if time.time() - t_s > time_budget:
            break
        corr = compute_corr(h_c)
        max_c = corr.max()
        # Find lags within 1% of max (active set)
        thresh = max_c * 0.99
        active = np.where(corr >= thresh)[0]
        K = max(K_init, min(K_max, len(active), 3 + it // 500))
        top = active[np.argsort(corr[active])[-K:]]
        lags = [int(i) - center for i in top]
        g = grad_lags(h_c, n, lags) / K
        dx = 2.0 / n
        # Momentum
        v = beta * v + g
        lr = 0.04 / (1.0 + 0.003 * it)
        improved = False
        for _ in range(10):
            h_t = project_feasible(h_c - lr * v, target)
            c5_t = compute_c5(h_t)
            if c5_t < c5_c - 1e-14:
                h_c = h_t
                c5_c = c5_t
                improved = True
                if c5_c < c5_b:
                    c5_b = c5_c
                    h_b = h_c.copy()
                break
            lr *= 0.4
        if not improved and it % 500 == 0:
            beta *= 0.9
            v *= 0.5
    return h_b, c5_b

def slsqp_smoothed(h, n, target, T, maxiter):
    dx = 2.0 / n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(c)
        s = T * (c - mc)
        s -= np.max(s)
        return (mc + np.log(np.sum(np.exp(s))) / T) * dx

    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return _grad_lse(c, x, T, n)

    def _cf(x):
        return x.sum() - target

    def _cj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def _grad_lse(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= np.max(s)
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    g = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        o = idx - center
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

def binary_swap(h, n, target, iters, rng, time_budget):
    t_s = time.time()
    order = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    k = int(target)
    h_b[order[:k]] = 1.0
    frac = target - k
    if frac > 1e-6:
        h_b[order[k]] = frac
    c5_b = compute_c5(h_b)
    ones = np.where(h_b >= 0.5)[0]
    zeros = np.where(h_b < 0.5)[0]
    for it in range(iters):
        if time.time() - t_s > time_budget:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        saved = (h_b[i], h_b[j])
        h_b[i], h_b[j] = saved[1], saved[0]
        c5_n = compute_c5(h_b)
        if c5_n < c5_b:
            c5_b = c5_n
            ones = np.where(h_b >= 0.5)[0]
            zeros = np.where(h_b < 0.5)[0]
        else:
            h_b[i], h_b[j] = saved
    return h_b, c5_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_tmp = 280
        h = np.zeros(n_tmp)
        h[np.arange(0, n_tmp, 2)] = 1.0
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP continuation
    for T, mit in [(150, 200), (50, 300), (20, 400), (10, 500), (5, 700)]:
        if elapsed() > budget_s - 400:
            break
        h = slsqp_smoothed(h, n, target, T, mit)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 2: Aggressive subgradient with line search
    if elapsed() < budget_s - 300:
        tr = budget_s - elapsed() - 300
        h, c5 = subgrad_line_search(h_best, n, target, 200000, tr, K_init=4, K_max=15)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 3: SLSQP at very low T
    if elapsed() < budget_s - 200:
        for T, mit in [(3, 800), (2, 1200)]:
            if elapsed() > budget_s - 100:
                break
            h = slsqp_smoothed(h_best, n, target, T, mit)
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()

    # Phase 4: Focused subgradient (fewer active lags)
    if elapsed() < budget_s - 120:
        tr = budget_s - elapsed() - 120
        h, c5 = subgrad_line_search(h_best, n, target, 100000, tr, K_init=3, K_max=8)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 5: Binary local search
    if elapsed() < budget_s - 60:
        tr = min(budget_s - elapsed() - 40, 80)
        h_b, c5_b = binary_swap(h_best, n, target, 200000, rng, tr)
        if c5_b < c5_best:
            c5_best = c5_b
            h_best = h_b.copy()

    # Phase 6: Un-round polish after binary search
    if elapsed() < budget_s - 40:
        h = slsqp_smoothed(h_best, n, target, 2, 800)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
        tr = max(0, budget_s - elapsed() - 10)
        if tr > 5:
            h, c5 = subgrad_line_search(h_best, n, target, 50000, tr, K_init=3, K_max=10)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()

    # Phase 7: Try n/2 resolution if time remains
    if elapsed() < budget_s - 80 and n > 100:
        n_lo = n // 2
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        c5_lo = compute_c5(h_lo)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()
        if elapsed() < budget_s - 30:
            for T, mit in [(10, 300), (3, 500)]:
                if elapsed() > budget_s - 15:
                    break
                h_lo = slsqp_smoothed(h_lo, n_lo, target_lo, T, mit)
                c5_lo = compute_c5(h_lo)
                if c5_lo < c5_best:
                    c5_best = c5_lo
                    h_best = h_lo.copy()

    # Final validation
    h_best = project_feasible(h_best, target)
    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)
    return (h_best, c5_best, len(h_best))
