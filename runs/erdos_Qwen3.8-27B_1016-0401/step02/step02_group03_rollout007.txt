```python
    return (mc + np.log(e.sum()) / T)

def lse_grad(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    mc = corr.max()
    e = np.exp(T * (corr - mc))
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for i, wi in enumerate(w):
        if wi < 1e-10:
            continue
        k = i - center
        s = subgrad_shift(x, n, k)
        grad += wi * s
    return grad
```

Hmm, the lse_grad is O(n²) per call due to the loop over all shifts. For n=280, that's 559 shifts * O(n) each = O(n²) ≈ 157K operations. That's fine.

Actually wait, I can vectorize the gradient computation. Let me think...

The soft-max gradient involves a weighted sum of subgradients. Each subgradient at shift k is:
- For k ≥ 0: grad[i] += (1-h[i+k]) for i < n-k, and grad[i+k] -= h[i] for i < n-k
- For k < 0: similar with signs flipped

The weighted sum: ∑_k w_k * subgrad_shift(h, n, k)

This can be computed as:
For each pair (i, j) where j = i + k for some k, the contribution is w_k * [(1-h[j]) for the first term, -h[i] for the second].

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_shift(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        idx = np.arange(n - k)
        grad[idx] += 1.0 - h[idx + k]
        grad[idx + k] -= h[idx]
    else:
        idx = np.arange(-k, n)
        grad[idx] += 1.0 - h[idx + k]
        grad[idx + k] -= h[idx]
    return grad * dx

def lse_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    mc = corr.max()
    e = np.exp(T * (corr - mc))
    return mc + np.log(e.sum()) / T

def lse_grad(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    mc = corr.max()
    e = np.exp(T * (corr - mc))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for oi in range(2 * n - 1):
        wi = w[oi]
        if wi < 1e-12:
            continue
        k = oi - center
        g = subgrad_shift(x, n, k)
        grad += wi * g
    return grad

def slsqp_opt(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lse_obj, h, args=(n, T), method='SLSQP', jac=lambda x: lse_grad(x, n, T),
                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_accel(h_init, n, t_stop, max_iters=500000):
    target = n / 2.0
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.35
    t0 = time.time()
    for it in range(max_iters):
        if time.time() > t_stop:
            break
        lam = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + lam * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        g = subgrad_shift(h_y, n, idx - (n - 1))
        alpha = alpha0 / (1.0 + 0.005 * it)
        h_new = h_y - alpha * g
        if (it + 1) % 20 == 0:
            h_new = project_feasible(h_new, n)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - h_new.sum()
            if abs(diff) > 1e-10:
                h_new = np.clip(h_new + diff / n, 0.0, 1.0)
        h_prev = h_sg.copy()
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 20000 == 0:
            h_check = project_feasible(h_avg, n)
            c5a = compute_c5(h_check)
            c5s = compute_c5(h_sg)
            if c5a < c5_best:
                c5_best = c5a
                h_best = h_check
            if c5s < c5_best:
                c5_best = c5s
                h_best = h_sg.copy()
    h_avg_p = project_feasible(h_avg, n)
    c5a = compute_c5(h_avg_p)
    if c5a < c5_best:
        c5_best = c5a
        h_best = h_avg_p
    return h_best, c5_best

def freq_perturb(h, rng, n, sigma):
    hf = np.fft.rfft(h)
    mask = np.ones(len(hf), dtype=bool)
    mask[0] = False
    noise = rng.normal(0, sigma, size=len(hf))
    noise[:2] = 0
    hf_pert = hf + noise
    hf_pert = np.real(hf_pert)
    h_pert = np.fft.irfft(hf_pert, n)
    return project_feasible(h_pert, n)

def block_perturb(h, rng, n, sigma):
    h_pert = h.copy()
    nb = max(1, n // 10)
    blocks = np.array_split(np.arange(n), nb)
    for b in blocks:
        if rng.random() < 0.5:
            h_pert[b] += rng.normal(0, sigma, len(b))
    return project_feasible(h_pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_try = 280
        h_best = np.ones(n_try) * 0.5
        h_best = project_feasible(h_best, n_try)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP multi-temperature
    for T in [32, 16, 8, 4, 2, 1.5, 1.0, 0.7, 0.5]:
        if time.time() - t0 > budget_s - 350:
            break
        h_new = slsqp_opt(h_best, n, T, 2000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov subgradient (long)
    t_sg = time.time() + min(250, budget_s - 400)
    h_sg, c5_sg = nesterov_accel(h_best, n, t_sg)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP polish after subgradient
    for T in [1.0, 0.7, 0.5, 0.3]:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = slsqp_opt(h_best, n, T, 3000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Structured perturbation + SLSQP restarts
    n_restarts = 50
    for pi in range(n_restarts):
        if time.time() - t0 > budget_s - 30:
            break
        ptype = pi % 3
        sigma = 0.005 + 0.012 * (pi % 6) * 0.5
        if ptype == 0:
            h_p = h_best + rng.normal(0, sigma, n)
            h_p = project_feasible(h_p, n)
        elif ptype == 1:
            h_p = freq_perturb(h_best, rng, n, sigma)
        else:
            h_p = block_perturb(h_best, rng, n, sigma)
        T_use = 2.0 if pi < 20 else 1.0
        h_new = slsqp_opt(h_p, n, T_use, 1500, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try coarser n=140
    if time.time() - t0 < budget_s - 120:
        n_c = 140
        x_old = np.arange(n) * (2.0 / n)
        x_c = np.linspace(0, 2.0, n_c, endpoint=False)
        h_c = np.interp(x_c, x_old, h_best)
        h_c = project_feasible(h_c, n_c)
        c5_c = compute_c5(h_c)
        for T in [8, 2, 1]:
            if time.time() - t0 > budget_s - 30:
                break
            h_new = slsqp_opt(h_c, n_c, T, 2000, 1e-15)
            h_new = project_feasible(h_new, n_c)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                n = n_c

    # Phase 6: Try finer n=560
    if time.time() - t0 < budget_s - 120 and n < 560:
        n_f = 560
        x_old = np.arange(n) * (2.0 / n)
        x_f = np.linspace(0, 2.0, n_f, endpoint=False)
        h_f = np.interp(x_f, x_old, h_best)
        h_f = project_feasible(h_f, n_f)

            h_new = project_feasible(h_new, n_f)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    if time.time() - t0 < budget_s - 10:
        h_new = slsqp_opt(h_best, n, 0.5, 2000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
```
