    if topk >= 2 * n - 1:

        n_init = 140
        x = np.linspace(0.0, 2.0, n_init, endpoint=False)
        h_best = 0.5 + 0.25 * np.cos(np.pi * x) + 0.1 * np.cos(2.0 * np.pi * x)
        h_best = project_box_sum(h_best, n_init / 2.0)

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse FISTA (fast subgradient, many iterations)
    h_new = fista_run(h_best, target, topk=3, step=0.35 * n, max_iter=60000, check_every=200)
    c5 = compute_c5(h_new)
    if c5 < c5_best:
        c5_best, h_best = c5, h_new

    if elapsed() > budget_s - 500:
        pass  # Skip to later phases

    # Phase 2: Medium FISTA with larger topk
    if elapsed() < budget_s - 450:
        h_new = fista_run(h_best, target, topk=8, step=0.25 * n, max_iter=40000, check_every=300)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    if elapsed() < budget_s - 400:
        h_new = fista_run(h_best, target, topk=15, step=0.2 * n, max_iter=25000, check_every=300)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 3: L-BFGS-B softmax polish
    if elapsed() < budget_s - 350:
        for T_val in [80, 200, 500]:
            if elapsed() > budget_s - 250:
                break
            h_try = lbfgs_polish(h_best, n, T_val, maxiter=1200)
            c5_t = compute_c5(h_try)
            if c5_t < c5_best:
                c5_best, h_best = c5_t, h_try

    # Phase 4: Random restarts + FISTA
    n_rest = 0
    while elapsed() < budget_s - 180:
        n_rest += 1
        scale = max(0.03, 0.12 / (1.0 + 0.07 * n_rest))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        # Also try cyclic shifts as structure perturbation
        if n_rest % 4 == 0:
            shift_amt = rng.integers(1, max(2, n // 8))
            h_pert = np.roll(h_best, shift_amt)
            h_pert = project_box_sum(h_pert, target)
        step_r = rng.uniform(0.15, 0.4) * n
        h_new = fista_run(h_pert, target, topk=5, step=step_r, max_iter=12000, check_every=250)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
            # Quick L-BFGS polish after improvement
            if elapsed() < budget_s - 180:
                h_p = lbfgs_polish(h_best, n, 100, maxiter=500)
                c5p = compute_c5(h_p)
                if c5p < c5_best:
                    c5_best, h_best = c5p, h_p

    # Phase 5: Resolution adaptation
    for n_try in [140, 280, 560]:
        if elapsed() > budget_s - 60 or n_try == n:
            continue
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new_g = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new_g, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        step_t = 0.25 * n_try
        h_try = fista_run(h_try, t_try, topk=5, step=step_t, max_iter=20000, check_every=300)
        c5_t = compute_c5(h_try)
        if c5_t < c5_best:
            c5_best, h_best, n, target = c5_t, h_try, n_try, t_try

    # Final polish
    if elapsed() < budget_s - 10:
        h_final = fista_run(h_best, target, topk=10, step=0.15 * n, max_iter=15000, check_every=400)
        c5_f = compute_c5(h_final)
        if c5_f < c5_best:
            c5_best, h_best = c5_f, h_final
        h_final2 = lbfgs_polish(h_best, n, 50, maxiter=800)
        c5_f2 = compute_c5(h_final2)
        if c5_f2 < c5_best:
            c5_best, h_best = c5_f2, h_final2

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
