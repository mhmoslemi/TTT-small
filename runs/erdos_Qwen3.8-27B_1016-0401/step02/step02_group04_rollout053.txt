import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def smax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    center = n - 1
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o >= 0:
            g[:n-o] += w * (1.0 - h[o:])
            g[o:] -= w * h[:n-o]
        else:
            p = -o
            g[p:] += w * (1.0 - h[:n-p])
            g[:n-p] -= w * h[p:]
    obj = (mc + np.log(e.sum()) / T) * dx
    return obj, g * dx

def project_box_sum(h, target):
    if abs(h.sum() - target) < 1e-10 and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgsb_opt(h0, target, T, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        c, _ = smax_obj_grad(x, T)
        return c + pen * (x.sum() - target) ** 2
    def grad(x):
        _, g = smax_obj_grad(x, T)
        return g + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15, 'ftol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def fista_opt(h0, target, T, max_iter, step):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        _, g = smax_obj_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-14:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def top_shift_descent(h, target, top_k, n_iter, step):
    n = len(h)
    dx = 2.0 / n
    for _ in range(n_iter):
        corr = get_corr(h)
        idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for ix in idx:
            o = int(ix) - (n - 1)
            if o >= 0:
                g[:n-o] += (1.0 - h[o:])
                g[o:] -= h[:n-o]
            else:
                p = -o
                g[p:] += (1.0 - h[:n-p])
                g[:n-p] -= h[p:]
        g /= float(top_k)
        h = project_box_sum(h - step * dx * g, target)
    return h
