    import cvxpy as cp
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    
    # Compute gradients of each lag's overlap w.r.t. h
    # grad[k, j] = ∂C(k)/∂h[j]
    # C(k) for k≥0: sum_{i=0}^{n-1-k} h[i+k](1-h[i])
    # ∂C(k)/∂h[j] = (1-h[j-k])*[k≤j≤n-1] - h[j+k]*[0≤j≤n-1-k]
    
    eps = cp.Variable(n)
    t = cp.Variable()

    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h0, target, T, max_iter, step):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    best_obj = soft_max_obj(h, T)
    for i in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = soft_max_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        obj = soft_max_obj(zh, T)
        if obj < best_obj:
            best_obj = obj
        if np.linalg.norm(zh - h) < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15, 'ftol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def active_subgrad_step(h, target, top_m=10, lr=0.001):
    """One step of subgradient method on the max of top-m lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Get top-m lags
    top_idx = np.argsort(corr)[-top_m:]
    # Subgradient: average of gradients at top lags
    grad = np.zeros(n)
    for idx in top_idx:
        o = idx - (n - 1)
        if o == 0:
            g = 1.0 - 2.0 * h
        elif o > 0:
            g = np.zeros(n)
            g[o:] += (1.0 - h[:n-o])
            g[:n-o] -= h[o:]
        else:
            p = -o
            g = np.zeros(n)
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
        grad += g
    grad /= len(top_idx)
    grad *= dx
    h_new = project_box_sum(h - lr * grad, target)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:

        if o == 0:
            grad += w * (1.0 - 2.0 * h)

        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

    n = len(h)

    except Exception:

    """Polish using subgradient of top-m lags with decreasing step size."""
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    for i in range(iterations):

        h = h_new

    try:

    except ImportError:
        return h

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def soft_max_obj(h, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / len(h))

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    c = n - 1
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[c+o]
        if w < 1e-13: continue
        if o == 0: g += w*(1-2*h)
        elif o > 0:
            g[o:] += w*(1-h[:n-o]); g[:n-o] -= w*h[o:]
        else:
            p=-o; g[:n-p] += w*(1-h[p:]); g[p:] -= w*h[:n-p]
    return g * (2.0/n)

def project(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        m = (lo+hi)/2
        if np.clip(h-m, 0, 1).sum() > target: lo = m
        else: hi = m
    return np.clip(h-(lo+hi)/2, 0, 1)

def fista(h0, target, T, iters, step):
    n = len(h0); h = h0.copy(); zh = h.copy(); tk = 1.0
    for _ in range(iters):

        z = project(y - step*soft_max_grad(y, T), target)
        if np.linalg.norm(z-h) < 1e-14*max(1, np.linalg.norm(h)): break
        tk = (1+np.sqrt(1+4*tk*tk))/2; h, zh = z, z
    return h

    def ob(x): return soft_max_obj(x,T) + pen*(x.sum()-target)**2

    try:
        r = minimize(ob, h0.copy(), method='L-BFGS-B', jac=gr, bounds=b,
                     options={'maxiter':maxit,'gtol':1e-15})
        return project(r.x, target)
    except: return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def soft_max_obj(h, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / len(h))

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    c = n - 1
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[c+o]
        if w < 1e-13: continue
        if o == 0: g += w*(1-2*h)
        elif o > 0:
            g[o:] += w*(1-h[:n-o])
            g[:n-o] -= w*h[o:]
        else:
            p = -o
            g[:n-p] += w*(1-h[p:])
            g[p:] -= w*h[:n-p]
    return g * (2.0/n)
