                h_try += amp * np.sin(2 * np.pi * fr * np.arange(n) / n + ph)
        else:
            h_try = (rng.random(n) > 0.5).astype(float)
        h_try = fix_constraint(h_try, target)

        Tr = [25, 12, 6][cnt % 3]
        try:
            res = minimize(soft_max_obj, h_try, method='SLSQP',
                         jac=soft_max_jac, args=(Tr,),
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 1200, 'ftol': 1e-14})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best, h_best = c5, h_new
        except Exception:
            pass
        h_ls2, c5_ls2 = targeted_search(h_best, target, 900, t0, budget_s - 130)
        if c5_ls2 < c5_best:
            c5_best, h_best = c5_ls2, h_ls2

    # Phase 4: Final polish
    for T in [3, 1.5, 1]:
        if time.time() - t0 > budget_s - 20:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                         jac=soft_max_jac, args=(T,),
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-17})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best, h_best = c5, h_new
        except Exception:
            pass

    # Final targeted search
    h_fin, c5_fin = targeted_search(h_best, target, 3000, t0, budget_s - 2)
    if c5_fin < c5_best:
        c5_best, h_best = c5_fin, h_fin

    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)
    return (h_best, c5_best, len(h_best))
