        else:

    """Compute combined gradient of top-K correlations."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Find top K lags
    top_idx = np.argsort(corr)[::-1][:K]
    c_max = corr[top_idx[0]]
    # Weight by proximity to max
    weights = np.exp(corr[top_idx] / max(c_max * 0.01, 1e-10) * 20)

def project_sum_box(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_and_grad(h, K=20):
    """Return corr vector, top-K indices, and combined gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:K]
    # Weights: exponential of corr values (normalized)
    cmax = corr[top_idx[0]]
    weights = np.exp(20.0 * (corr[top_idx] - cmax) / max(cmax, 1e-10))
    sw = weights.sum()
    if sw < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= sw
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            grad[:m] += w
            grad[:m] -= w * h[k:]
            grad[k:] -= w * h[:m]
        else:
            p = -k
            m = n - p
            grad[p:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[p:]
    return corr * dx, grad * dx, top_idx

def subgrad_solve(h0, target, n_iter, t0, budget, K=20, lr0=0.01, momentum=0.9):
    """Nesterov-accelerated projected subgradient."""
    n = len(h)
    h = h0.copy()
    h_prev = h.copy()
    y = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        _, grad, _ = corr_and_grad(y, K)
        # Nesterov step
        h_new = project_sum_box(y - lr0 * grad, target)
        # Update momentum
        t_k = (1 + np.sqrt(1 + 4 * momentum * momentum)) / 2
        y = h_new + ((it * momentum + 1) / (it + t_k)) * (h_new - h)
        y = np.clip(y, 0, 1)
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()
        h_prev, h = h, h_new
    
    return h_best, c5_best

Let me be careful about the coordinate descent. The 1D subproblem:

Given current h (with h[j] = v), for each lag k in top_K:
- C(k) = sum_i h[i]*(1-h[i+k]) (for k≥0, appropriate range)
- As h[j] changes to x: C(k) changes by dC(k)/dh[j] * (x - v)
- So C(k; x) = C(k; v) + B_k * (x - v) where B_k = dC(k)/dh[j]

We want: min_x max_k C(k; x) * dx, x ∈ [0, 1]

This is a 1D convex minimization. Golden section search with ~40 iterations gives very high precision. Each evaluation is O(K).

But we also need to handle the sum constraint. After changing h[j], we need to project. The projection changes all h values slightly. This complicates things.

Alternative: do the coordinate descent WITHOUT the sum constraint (just 0 ≤ h ≤ 1), and project after each sweep. The projection is a small adjustment (the average of h should be 0.5, and small deviations mean small adjustments).

Actually, the projection changes the objective. So we should include it in the evaluation. But for the coordinate step, we can ignore it and project at the end of the sweep.
