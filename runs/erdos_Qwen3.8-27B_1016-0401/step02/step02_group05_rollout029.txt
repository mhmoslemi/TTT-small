def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")

    shifted = T * (corr - mc)

    dx = 2.0 / len(x)
    c = np.correlate(x, 1 - x, mode="full")
    m = c.max()
    return (m + np.log(np.exp(T * (c - m)).sum()) / T) * dx

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode="full")

    n = len(h)
    dx = 2.0 / n
    ctr = n - 1
    hc = h.copy()
    c = np.correlate(hc, 1 - hc, mode="full") * dx
    best = float(c.max())
    for it in range(n_it):
        if time.time() > dl:
            break
        # Top lags
        top_lags = np.argsort(-c)[:K_lags]
        # For each top lag, find best coordinate to decrease
        candidates = []
        for ki in top_lags:
            o = int(ki) - ctr
            if o == 0:
                contrib = hc * (1 - hc)
                idxs = np.argsort(-contrib)[:K_cands]
                for ti in idxs:
                    candidates.append((int(ti), 'dec'))
            elif o > 0:
                # c(o) = sum_{m=0}^{n-1-o} h[m] * (1-h[m+o])
                # To reduce: decrease h[m] where h[m] is high and h[m+o] is low
                # Or increase h[m+o] where h[m] is high
                m_range = np.arange(n - o)
                contrib_dec = hc[m_range] * (1 - hc[m_range + o])
                top_m = m_range[np.argsort(-contrib_dec)[:K_cands]]
                for tm in top_m:
                    candidates.append((int(tm), 'dec'))  # decrease h[tm]
                # Also: increase h[tm+o]
                for tm in top_m:
                    candidates.append((int(tm + o), 'inc'))
            else:
                p = -o
                m_range = np.arange(n - p)
                contrib_dec = hc[m_range + p] * (1 - hc[m_range])
                top_m = m_range[np.argsort(-contrib_dec)[:K_cands]]
                for tm in top_m:

                    candidates.append((int(tm), 'inc'))

def sm_obj(x, T):
    dx = 2.0 / len(x)
    c = np.correlate(x, 1 - x, mode="full")
    m = c.max()
    return (m + np.log(np.exp(T * (c - m)).sum()) / T) * dx
(13)

def sm_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode="full")
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    ctr = n - 1
    for o in range(-(n - 1), n):
        wt = w[ctr + o]
        if wt < 1e-14:
            continue
        if o == 0:
            g += wt * (1 - 2 * x)
        elif o > 0:
            g[o:] += wt * (1 - x[:n-o])
            g[:n-o] -= wt * x[o:]
        else:
            p = -o
            g[:n-p] += wt * (1 - x[p:])
            g[p:] -= wt * x[:n-p]
    return g * dx
