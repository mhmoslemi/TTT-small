    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

        return h_src.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    global _G, _G_LAG_IDX
    t0 = time.time()
    elapsed = time.time
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_par = np.array(initial_h_values, dtype=float)
    else:
        h_par = np.zeros(280)
        h_par[::2] = 1.0

    h_best = h_par.copy()
    c5_best = compute_c5(h_best)
    n_par = len(h_par)

    n_list = [n_par]
    if n_par < 500:
        n_list.append(2 * n_par)

    time_limit_frac = 0.88

    for n in n_list:
        if elapsed() - t0 > budget_s * time_limit_frac:
            break
        target = n / 2.0
        _G['n'] = n
        _G['target'] = target
        bounds = [(0.0, 1.0)] * n

        h = _interp_to_n(h_par, n)
        h = fix_constraint(h, target)
        c5_cur = compute_c5(h)

        # Phase A: L-BFGS-B soft-max, increasing T
        T_schedule = [20, 40, 80, 160, 320, 640, 1280, 2560]
        for T in T_schedule:
            if elapsed() - t0 > budget_s * time_limit_frac:
                break
            _G['T'] = T
            _G['penalty'] = 1e5
            try:
                res = minimize(_smooth_obj, h, jac=_smooth_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 4000, 'ftol': 1e-15, 'gtol': 1e-7})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    h = h_new
            except Exception:
                pass

        # Phase B: Active-set SLSQP (top 3 lags, multiple passes)
        for outer in range(8):
            if elapsed() - t0 > budget_s * time_limit_frac:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_idx = np.argsort(corr)[-3:]
            for lag in top_idx:
                if elapsed() - t0 > budget_s * time_limit_frac:
                    break
                _G_LAG_IDX = lag
                cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
                try:
                    res = minimize(_lag_obj, h, jac=_lag_grad, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 1500, 'ftol': 1e-16})
                    h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        c5_cur = c5_new
                        h = h_new
                except Exception:
                    pass

        # Phase C: Perturbation restarts (escape local minima)
        for perm in range(4):
            if elapsed() - t0 > budget_s * time_limit_frac:
                break
            scale = 0.015 * (perm + 1)
            h_pert = h + rng.uniform(-scale, scale, n)
            h_pert = fix_constraint(h_pert, target)
            _G['T'] = 320
            _G['penalty'] = 1e5
            try:
                res = minimize(_smooth_obj, h_pert, jac=_smooth_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-6})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    h = h_new
            except Exception:
                pass
            # Quick active-set on perturbed
            corr_p = np.correlate(h, 1.0 - h, mode="full")
            worst_p = np.argmax(corr_p)
            _G_LAG_IDX = worst_p
            cons_p = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
            try:
                res = minimize(_lag_obj, h, jac=_lag_grad, method='SLSQP',
                               bounds=bounds, constraints=cons_p,
                               options={'maxiter': 800, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    h = h_new
            except Exception:
                pass

        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h

    # Final polish on best solution
    n_f = len(h_best)
    target_f = n_f / 2.0
    _G['n'] = n_f
    _G['target'] = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    end_t = t0 + budget_s - 5.0
    for _ in range(12):
        if elapsed() > end_t:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        _G_LAG_IDX = worst_f
        cons_f = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        try:
            res = minimize(_lag_obj, h_best, jac=_lag_grad, method='SLSQP',
                           bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 2500, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
