import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc) - T * corr.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n-k] -= w * h[k:]
        else:
            p = -k
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * (2.0 / n)

def pg_solve(h0, target, T, n_iter, lr0, t0, budget):
    h = h0.copy()
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        obj, sw = sm_values(h, T)
        grad = sm_grad(h, sw)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        for _ in range(50):
            h_new = project_constraint(h - lr * grad, target)
            obj_new, _ = sm_values(h_new, T)
            if obj_new < obj - 1e-16:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        lr = min(lr * 1.01, lr0)
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def pair_flip(h, target, t0, budget, n_iter, seed):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    ds = np.array([0.005, 0.01, 0.02, 0.05, 0.1])
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = ds[rng.integers(len(ds))]
        for s in (1, -1):
            ni = h[i] + s * d
            nj = h[j] - s * d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    h[i], h[j] = best_h[i], best_h[j]
                break
    return best_h, best_c5

def block_search(h, target, t0, budget, n_trials, seed):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    for _ in range(n_trials):
        if time.time() - t0 > budget - 3:
            break
        sz = rng.integers(2, 12)
        start = rng.integers(0, max(1, n - sz))
        trial = h.copy()
        seg = trial[start:start+sz]
        trial[start:start+sz] = np.clip(seg + rng.normal(0, 0.1, sz), 0, 1)
        trial = project_constraint(trial, target)
        c5 = compute_c5(trial)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = trial.copy()
    return best_h, best_c5

    x_old = np.arange(n0) * (2.0 / n0)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    h1 = np.interp(x1, x_old, h_best)
    h1 = project_constraint(h1, target1)
    c5_1 = compute_c5(h1)
    if c5_1 < c5_best:
        c5_best, h_best = c5_1, h1.copy()

    # Phase 3: Gradient at 1120 with cooling
    for T in [20, 60, 150, 400, 1000]:
        if time.time() - t0 > budget - 250:
            break
        h1, c5 = pg_solve(h1, target1, T, 3000, 0.025, t0, budget - 80)
        if c5 < c5_best:
            c5_best = c5
            h_best = h1.copy()

    # Phase 4: Pair flip at 1120
    if time.time() - t0 < budget - 80:
        h1, c5 = pair_flip(h1, target1, t0, budget - 40, 15000, seed)
        if c5 < c5_best:
            c5_best = c5
            h_best = h1.copy()

    # Phase 5: Block search
    if time.time() - t0 < budget - 50:
        h1, c5 = block_search(h1, target1, t0, budget - 20, 5000, seed + 7)
        if c5 < c5_best:
            c5_best = c5
            h_best = h1.copy()

    # Phase 6: Final gradient + flip at 560 (downsample if needed)
    n2 = 560
    target2 = n2 / 2.0
    nc = len(h_best)
    xc = np.arange(nc) * (2.0 / nc)
    x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x2, xc, h_best)
    h2 = project_constraint(h2, target2)
    if time.time() - t0 < budget - 30:
        h2, c5 = pg_solve(h2, target2, 500, 2000, 0.03, t0, budget - 10)
        if c5 < c5_best:
            c5_best = c5

            if c5 < c5_best:
                c5_best = c5
                h_best = h2.copy()

    return (h_best, c5_final, len(h_best))
