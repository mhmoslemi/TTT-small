```python
def slp_lp_step(h, n, tau_mat):

    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Variables: [h_0, ..., h_{n-1}, M]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # A_ub rows:
    # 1) For each m: tau_mat[:,m] @ h - M <= -corr[m] + tau_mat[:,m] @ h
    #    → [tau_mat[:,m] | -1] as row, rhs = -corr[m] + tau_mat[:,m] @ h

    A1[:, n] = -1.0

    # Note: -corr[m] + tau_mat[:,m]@h = -(corr[m] - tau_mat[:,m]@h)
    # And corr[m] - tau_mat[:,m]@h = corr[m] - Σ_i tau_i[m]*h[i]
    # This is NOT zero in general (corr is quadratic)
    
    rhs1 = -corr + tau_mat @ h
    
    # 2) h_i >= 0: -h_i <= 0
    # 3) h_i <= 1: h_i <= 1
    A2 = -np.eye(n)  # -h_i <= 0
    b2 = np.zeros(n)
    A3 = np.eye(n)   # h_i <= 1
    b3 = np.ones(n)
    
    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([rhs1, b2, b3])
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    return res
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_tau_vec(i, h, n):
    center = n - 1
    tau = np.zeros(2 * n - 1)
    tau[center - i : center + n - i] += (1.0 - h)
    tau[center + i - n + 1 : center + 1] += -h[::-1]
    return tau

def solve_1d_minimax(tau, corr, old_h, lam):
    a, b = 0.0, 1.0
    for _ in range(120):
        mid = (a + b) / 2.0
        u = mid - old_h
        vals = tau * u + corr
        mx = vals.max()
        near = vals >= mx - 1e-14
        sg_min = tau[near].min() + lam
        sg_max = tau[near].max() + lam
        if sg_min > 1e-11:
            b = mid
        elif sg_max < -1e-11:
            a = mid
        else:
            break
    return (a + b) / 2.0

def coord_descent(h, n, t0, budget, max_passes=200):
    target = n / 2.0
    h = h.copy()
    h = project_feasible(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_best = float(np.max(corr)) * (2.0 / n)
    h_best = h.copy()
    lam = 0.0
    no_improve = 0
    for p in range(max_passes):
        if time.time() - t0 > budget - 80:
            break
        S = h.sum()
        lam = 0.9 * lam + 0.1 * 1.5 * (S - target)
        for i in range(n):
            tau = compute_tau_vec(i, h, n)
            old_h = h[i]
            new_h = solve_1d_minimax(tau, corr, old_h, lam)
            delta = new_h - old_h
            if abs(delta) > 1e-15:
                h[i] = new_h
                corr += delta * tau
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr)) * (2.0 / n)
        if c5 < c5_best - 1e-11:
            c5_best = c5
            h_best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break
    return h_best, c5_best

def slp_optimize(h, n, t0, budget, max_rounds=30):
    target = n / 2.0
    h = project_feasible(h.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_best = float(np.max(corr)) * (2.0 / n)
    h_best = h.copy()
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 100:
            break
        tau_mat = np.zeros((n, 2 * n - 1))
        for i in range(n):
            tau_mat[i] = compute_tau_vec(i, h, n)
        active = np.argsort(-corr)[:40]
        h_lp = h.copy()
        for _ in range(8):
            m_act = len(active)
            A1 = np.zeros((m_act, n + 1))
            A1[:, :n] = tau_mat[:, active].T
            A1[:, n] = -1.0
            b1 = -corr[active] + tau_mat[:, active] @ h
            A_ub = np.vstack([A1, -np.eye(n), np.eye(n)])
            b_ub = np.concatenate([b1, np.zeros(n), np.ones(n)])
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            c_vec = np.zeros(n + 1)
            c_vec[-1] = 1.0
            bounds = [(0, 1)] * n + [(None, None)]
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[target],
                          bounds=bounds, method='highs')
            if not res.success:
                break
            h_lp = res.x[:n]
            M_lp = res.x[n]
            lin_corr = corr + tau_mat @ (h_lp - h)
            viol = np.where(lin_corr > M_lp + 1e-7)[0]
            if len(viol) == 0:
                break
            active = np.unique(np.concatenate([active, viol]))
            if len(active) > 120:
                active = np.argsort(-lin_corr)[:120]
        step = 1.0
        h_next = project_feasible(h + step * (h_lp - h), n)
        c5_new = compute_c5(h_next)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h = h_next
            h_best = h_next.copy()
            corr = np.correlate(h, 1.0 - h, mode="full")
        else:
            improved = False
            for s in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h + s * (h_lp - h), n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-12:
                    c5_best = c5_try
                    h = h_try
                    h_best = h_try.copy()
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    improved = True
                    break
            if not improved:
                break
    return h_best, c5_best

_N = 0
_TGT = 0.0
_T = 1.0

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _N - 1
    dx = 2.0 / _N
    grad = np.zeros(_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-12:
            continue
        grad += sw[oi] * compute_tau_vec(oi - center + center, x, _N) if False else 0
    # Use direct gradient computation
    grad = np.zeros(_N)
    for oi in range(2 * _N - 1):
        if sw[oi] < 1e-12:
            continue
        shift = oi - center
        g = compute_tau_vec_all(x, _N, shift)
        grad += sw[oi] * g
    return grad * dx

def compute_tau_vec_all(h, n, shift):
    """Gradient of corr at a given shift w.r.t. all h[i]."""
    center = n - 1
    grad = np.zeros(n)
    if shift >= 0:
        m1 = np.arange(0, n - shift)
        grad[m1] += (1.0 - h[m1 + shift])
    else:
        s = -shift
        m1 = np.arange(s, n)
        grad[m1] += (1.0 - h[m1 - s])
    if shift >= 0:
        m2 = np.arange(shift, n)
        grad[m2] -= h[m2 - shift]
    else:
        m2 = np.arange(0, n - (-shift))
        grad[m2] -= h[m2 + (-shift)]
    return grad

def do_slsqp(h, n, T, maxiter, ftol):
    global _N, _TGT, _T
    _N = n
    _TGT = n / 2.0
    _T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - _TGT, 'jac': lambda x: np.ones(_N)}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLP
    h1, c5_1 = slp_optimize(h_best, n, t0, budget_s - 400, max_rounds=30)
    if c5_1 < c5_best:
        c5_best = c5_1
        h_best = h1

    # Phase 2: Coordinate descent
    h2, c5_2 = coord_descent(h_best, n, t0, budget_s - 300, max_passes=150)
    if c5_2 < c5_best:
        c5_best = c5_2
        h_best = h2

    # Phase 3: SLSQP polish
    for T in [1, 2, 4, 8]:
        if time.time() - t0 > budget_s - 120:
            break
        h3 = do_slsqp(h_best, n, T, 3000, 1e-16)
        h3 = project_feasible(h3, n)
        c5_3 = compute_c5(h3)
        if c5_3 < c5_best:
            c5_best = c5_3
            h_best = h3

    # Phase 4: More coordinate descent
    h4, c5_4 = coord_descent(h_best, n, t0, budget_s - 80, max_passes=100)
    if c5_4 < c5_best:
        c5_best = c5_4
        h_best = h4

    # Phase 5: Perturbation restarts
    for pi in range(30):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.005 + 0.01 * (pi % 6)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new, c5_new = coord_descent(h_p, n, t0, budget_s - 15, max_passes=60)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s - 30:
        h_fin = do_slsqp(h_best, n, 1, 2000, 1e-17)
        h_fin = project_feasible(h_fin, n)
        c5_fin = compute_c5(h_fin)
        if c5_fin < c5_best:
            c5_best = c5_fin
            h_best = h_fin

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
```
