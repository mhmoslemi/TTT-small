    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

    return lse_obj_val(np.correlate(x, 1.0 - x, mode="full"), T, n)

    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_eval(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_deriv(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def subgrad_topk(corr, h, K, n):
    dx = 2.0 / n
    top_idx = np.argsort(corr)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def slsqp_phase(h, n, target, bounds, T, maxiter):
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: lse_eval(x, T, n), h, method='SLSQP',
                       jac=lambda x: lse_deriv(np.correlate(x, 1.0 - x, mode="full"), x, T, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgrad_bt(h, n, target, n_iter, K, lr0):
    c5 = compute_c5(h)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = subgrad_topk(corr, h, K, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        lr = lr0 / (1.0 + 0.03 * it)
        improved = False
        for _ in range(12):
            h_new = project_feasible(h - lr * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return h, c5

def binary_round(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    h_bin[order[:k]] = 1.0
    if k < n:
        h_bin[order[k]] = target - k
    return h_bin

def swap_search(h, n, target, n_iter, rng):
    c5 = compute_c5(h)
    ones = np.where(h >= 0.99)[0]
    zeros = np.where(h <= 0.01)[0]
    stalls = 0
    for _ in range(n_iter):
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
            stalls = 0
            ones = np.where(h >= 0.99)[0]
            zeros = np.where(h <= 0.01)[0]
        else:
            h[i], h[j] = 1.0, 0.0
            stalls += 1
            if stalls > 500:
                break
    return h, c5

def optimize_single(h0, n, t0, budget_s, rng, n_slq, n_sg, n_swap):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = project_feasible(h0.copy(), target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    sched = [(300, 100), (100, 150), (40, 200), (15, 300), (7, 400), (3, 600), (1.5, 900)]
    for T, mit in sched:
        if time.time() - t0 > budget_s - 200:
            break
        h = slsqp_phase(h, n, target, bounds, T, mit)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    h = h_best.copy()
    for K in [10, 6, 4, 3, 2]:
        if time.time() - t0 > budget_s - 100:
            break
        h, c5 = subgrad_bt(h, n, target, n_sg, K, 0.1)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    h_bin = binary_round(h_best, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    h_sw, c5_sw = swap_search(h_bin, n, target, n_swap, rng)
    if c5_sw < c5_best:
        c5_best = c5_sw
        h_best = h_sw.copy()

    if time.time() - t0 < budget_s - 40:
        h = slsqp_phase(h_best, n, target, bounds, 1.0, 1500)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    h, c5 = subgrad_bt(h_best, n, target, n_sg // 2, 3, 0.06)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()

    h_bin2 = binary_round(h_best, target)
    c5_b2 = compute_c5(h_bin2)
    if c5_b2 < c5_best:
        c5_best = c5_b2
        h_best = h_bin2.copy()
    h_sw2, c5_sw2 = swap_search(h_best, n, target, n_swap // 2, rng)
    if c5_sw2 < c5_best:
        c5_best = c5_sw2
        h_best = h_sw2.copy()

    return h_best, c5_best, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = (np.arange(200) % 2 == 0).astype(float)

    n0 = len(h_init)
    best_h = None
    best_c5 = 1e9
    best_n = n0

    # Primary: optimize at native resolution
    h1, c5_1, n1 = optimize_single(h_init, n0, t0, budget_s, rng, 200, 300, 4000)
    best_h, best_c5, best_n = h1, c5_1, n1

    # Try a different resolution
    if time.time() - t0 < budget_s - 350:
        n2 = n0 // 2 if n0 > 160 else n0 - 40
        x_old = np.arange(n0) * (2.0 / n0)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2_init = np.interp(x_new, x_old, best_h)
        h2, c5_2, _ = optimize_single(h2_init, n2, t0, budget_s, rng, 150, 200, 2000)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Another resolution
    if time.time() - t0 < budget_s - 250:
        n3 = min(n0 + 20, 400)
        x_old = np.arange(best_n) * (2.0 / best_n)
        x_new = np.linspace(0, 2.0, n3, endpoint=False)
        h3_init = np.interp(x_new, x_old, best_h)
        h3, c5_3, _ = optimize_single(h3_init, n3, t0, budget_s, rng, 150, 200, 2000)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n3

    # Random restart at best_n
    if time.time() - t0 < budget_s - 120:
        for r in range(3):
            if time.time() - t0 > budget_s - 30:
                break
            h_rand = rng.random(best_n)
            h_rand = project_feasible(h_rand, best_n / 2.0)
            hr, cr, _ = optimize_single(h_rand, best_n, t0, budget_s, rng, 80, 150, 1500)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr

    # Final micro-polish: very short subgradient
    if time.time() - t0 < budget_s - 5:
        h_f, c5_f = subgrad_bt(best_h, best_n, best_n / 2.0, 100, 2, 0.04)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_b = binary_round(best_h, best_n / 2.0)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        h_s, c5_s = swap_search(best_h, best_n, best_n / 2.0, 3000, rng)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    c5_final = min(compute_c5(best_h), best_c5)
    return (best_h, c5_final, len(best_h))
