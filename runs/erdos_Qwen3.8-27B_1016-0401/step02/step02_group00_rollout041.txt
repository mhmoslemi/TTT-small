import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(g, target):
    """Project g onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(g)
    # Water filling
    lo, hi = g.min() - 1, g.max() + 1
    for _ in range(100):
        lam = (lo + hi) / 2
        h = np.clip(g - lam, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(g - (lo+hi)/2, 0, 1)

def lag_gradient(x, lag_idx, n):
    """Gradient of the i-th lag correlation w.r.t. x"""
    o = lag_idx - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * x)
    elif o > 0:
        valid = n - o
        g[o:] += (1.0 - x[:valid])
        g[:valid] += -x[o:]
    else:
        p = -o
        valid = n - p
        g[:valid] += (1.0 - x[p:])
        g[p:] += -x[:valid]
    return g

def softmax_obj_grad(x, T, n):
    """Softmax approximation of max of correlations, with gradient"""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        grad += w * lag_gradient(x, center + o, n)
    return mc * dx + (1.0/T) * np.log(np.sum(e)) * dx, grad * dx

def sgcd_polish(h, target, n, t_end, T_start=100, max_iter=20000):
    """Projective subgradient with Nesterov acceleration on smoothed max"""
    dx = 2.0 / n
    center = n - 1
    t0 = time.time()
    L = 1.0
    h_mom = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = T_start
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Softmax weights
        mc = np.max(corr)
        shifted = T * (corr - mc)
        e = np.exp(shifted)
        sw = e / np.sum(e)
        
        # Weighted gradient
        g = np.zeros(n)
        for o in range(-(n-1), n):
            w = sw[center + o]
            if w < 1e-12:
                continue
            g += w * lag_gradient(h, center + o, n)
        g *= dx
        
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        step = L / t_k * 2.0  # scaled step
        h_new = h_mom - step * g
        h_new = project_sum_box(h_new, target)
        h_mom = h_new + ((L - 1.0) / t_k) * (h_new - h)
        h_mom = np.clip(h_mom, 0.0, 1.0)
        h = h_new
        L += 1.0
        
        # Periodic evaluation
        if it % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # Adapt T
            T = min(T * 1.05, 5000.0)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(g, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        lam = (lo + hi) / 2.0
        h = np.clip(g - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def lag_grad(x, o, n):
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        v = n - o
        g[o:] += 1.0 - x[:v]
        g[:v] -= x[o:]
    else:
        p = -o
        v = n - p
        g[:v] += 1.0 - x[p:]
        g[p:] -= x[:v]
    return g

def softmax_grad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    g = np.zeros(n)
    for idx in range(len(sw)):
        w = sw[idx]
        if w < 1e-13:
            continue
        o = idx - center
        g += w * lag_grad(x, o, n)
    return g * dx

def sgsolve(wrong_key, h, target, n, bounds, t_end, T, max_it, step_init):
    dx = 2.0 / n
    center = n - 1
    h_ng = h.copy()
    h_mom = h.copy()
    step = step_init
    best_local = compute_c5(h_ng)
    best_h = h_ng.copy()
    no_imp = 0
    for it in range(max_it):
        if time.time() > t_end:
            break
        g = softmax_grad(h_ng, T, n)
        t_k = it + 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = step / t_new

        h_new = project_sum_box(h_new, target)

        h_ng = h_new
        if (it + 1) % 400 == 0:
            c5_cur = compute_c5(h_ng)
            if c5_cur < best_local - 1e-9:
                best_local = c5_cur
                best_h = h_ng.copy()
                no_imp = 0
                step = min(step * 1.3, 8.0)
            else:
                no_imp += 1
                if no_imp > 8:
                    step = max(step * 0.6, 0.05)
                    no_imp = 0
    return best_h, best_local

def slsqp_lag(h, lag, n, bounds, cons, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    o = lag - center
    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full")[lag] * dx
    def jac(x):
        return lag_grad(x, o, n) * dx
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum_box(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h

def make_eq_cons(target):
    def eq_fun(x):
        return x.sum() - target
    def eq_jac(x):
        return np.ones(len(x))
    return [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
