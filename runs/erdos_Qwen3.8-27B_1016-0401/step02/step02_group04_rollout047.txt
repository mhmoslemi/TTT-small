    lo, hi = -3.0, 3.0

        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n-o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n-o]
        else:
            p = -o
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad * dx

def fista_optimize(h0, target, T, max_iter, step_init, tol=1e-13):

    zh = h.copy()
    t_k = 1.0

        diff = np.linalg.norm(zh - h)
        if diff < tol * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n-o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n-o]
        else:
            p = -o
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad * dx

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def fista_optimize(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    step = step_init
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = soft_max_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def simulated_annealing(h0, target, n_iter, T_start, T_end, rng, perturb_scale):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    T = T_start
    cool = (T_end / T_start) ** (1.0 / max(n_iter, 1))
    h_best = h.copy()
    c5_best = c5_cur
    for it in range(n_iter):
        T *= cool
        # Random perturbation of a few coordinates
        n_pert = int(rng.integers(1, 4))
        idx = rng.integers(0, n, n_pert)
        h_new = h.copy()
        h_new[idx] += rng.normal(0, perturb_scale, n_pert)
        h_new = project_box_sum(h_new, target)
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
    return h_best, c5_best

def coord_descent(h, target, n_passes, deltas):
    n = len(h)
    c5_cur = compute_c5(h)
    for p in range(n_passes):
        improved = False
        for i in range(n):
            for d in deltas:
                for s in [1.0, -1.0]:

                    h_alt[i] += s * d
                    h_alt = project_box_sum(h_alt, target)

                    c5_alt = compute_c5(h_alt)
                    if c5_alt < c5_cur - 1e-14:
                        h = h_alt
                        c5_cur = c5_alt
                        improved = True
        if not improved:
            break
    return h, c5_cur

    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: High-T FISTA for thorough local optimization
    for T in [500, 1000, 2000, 4000]:
        if elapsed() > budget_s * 0.12:
            break
        step = min(0.5, 5.0 / (T + 10.0))
        h_new = fista_optimize(h_best, target, T, 15000, step)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-15:
            c5_best = c5
            h_best = h_new

    # Phase 2: L-BFGS-B with various T
    for T in [500, 1500, 3000]:
        if elapsed() > budget_s * 0.25:
            break
        for pen in (5.0, 20.0):
            h_new = lbfgsb_optimize(h_best, n, target, T, pen, 5000)
            c5 = compute_c5(h_new)
            if c5 < c5_best - 1e-15:
                c5_best = c5
                h_best = h_new

    # Phase 3: Simulated annealing (key differentiator)
    if elapsed() < budget_s * 0.5:
        h_sa, c5_sa = simulated_annealing(h_best, target, 300000, 0.002, 1e-9, rng, 0.04)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa
        # Refine SA result
        if elapsed() < budget_s * 0.55:
            h_new = fista_optimize(h_best, target, 2000, 10000, 0.2)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Random restarts + FISTA
    n_restart = 0
    while elapsed() < budget_s * 0.72:
        n_restart += 1
        scale = max(0.008, 0.04 / (1.0 + 0.05 * n_restart))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T = 400 + rng.integers(0, 2000)
        h_new = fista_optimize(h_pert, target, T, 8000, min(0.3, 4.0 / (T + 10)))
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-15:
            c5_best = c5
            h_best = h_new

    # Phase 5: Second SA round with smaller steps
    if elapsed() < budget_s * 0.82:
        h_sa2, c5_sa2 = simulated_annealing(h_best, target, 150000, 0.0005, 1e-10, rng, 0.015)
        if c5_sa2 < c5_best:
            c5_best = c5_sa2
            h_best = h_sa2
        if elapsed() < budget_s * 0.85:
            h_new = fista_optimize(h_best, target, 3000, 8000, 0.15)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 6: Try nearby resolutions
    for n_try in (270, 290, 250, 310, 240):
        if elapsed() > budget_s * 0.92:
            break
        if n_try == n:
            continue
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        for T in (800, 2000):
            if elapsed() > budget_s * 0.94:
                break
            h_t = fista_optimize(h_try, t_try, T, 5000, min(0.25, 4.0 / (T + 10)))
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        if c5_try < c5_best - 1e-15:
            c5_best = c5_try
            h_best = h_try
            n = n_try
            target = t_try

    # Phase 7: Coordinate descent polish
    if elapsed() < budget_s - 20:
        h_best, c5_cd = coord_descent_polish(h_best, target, 3, (0.004, 0.002, 0.001, 0.0005))
        if c5_cd < c5_best:
            c5_best = c5_cd

    # Phase 8: Final FISTA + LBFGS polish
    if elapsed() < budget_s - 5:
        h_new = fista_optimize(h_best, target, 5000, 10000, 0.1)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    if elapsed() < budget_s - 2:
        h_new = lbfgsb_optimize(h_best, n, target, 3000, 10.0, 3000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
