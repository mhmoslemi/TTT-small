import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi
            grad[:n - o] -= wi * h[o:]
            if 2 * o <= n - 1:
                grad[o:n - o] -= wi * h[:n - 2 * o]
        else:
            p = -o
            grad[:n - p] += wi
            grad[:n - p] -= wi * h[p:]
            if 2 * p <= n - 1:
                grad[p:n - p] -= wi * h[:n - 2 * p]
    return grad * dx

def subgrad_topk(corr, h, K, n):
    dx = 2.0 / n
    top_idx = np.argsort(corr)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        wi = 1.0 / K
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi
            g[:n - o] -= wi * h[o:]
            if 2 * o <= n - 1:
                g[o:n - o] -= wi * h[:n - 2 * o]
        else:
            p = -o
            g[:n - p] += wi
            g[:n - p] -= wi * h[p:]
            if 2 * p <= n - 1:
                g[p:n - p] -= wi * h[:n - 2 * p]
    return g

def slsqp_run(h, n, target, T, maxiter, ftol=1e-14):
    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def subgrad_polish(h, n, target, n_steps, K=4, t_end=None):
    dx = 2.0 / n
    c5_local = compute_c5(h)
    for it in range(n_steps):
        if t_end is not None and time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = subgrad_topk(corr, h, K, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        lr = 0.3 / (1.0 + it * 0.05)
        h_cand = project_feasible(h - lr * g, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_local:
            c5_local = c5_cand
            h = h_cand
        else:
            lr *= 0.5
            h_cand = project_feasible(h - lr * g, target)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_local:
                c5_local = c5_cand
                h = h_cand
    return h, c5_local

def make_periodic_h(n, period, d_fill):
    pattern = np.zeros(period)
    fill_count = int(round(d_fill * period))
    pattern[:fill_count] = 1.0
    pattern[fill_count] = d_fill * period - fill_count
    reps = n // period
    remainder = n - reps * period
    h = np.tile(pattern, reps + 1)[:n]
    return h
