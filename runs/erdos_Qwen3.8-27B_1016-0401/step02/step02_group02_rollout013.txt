    sh = T * (corr - mc) - np.max(T * (corr - mc))
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

    vb = (j - o >= 0) & (j - o < n)
    d[vb] += (1.0 - h[j - o[vb]])
    vc = (i + o >= 0) & (i + o < n)
    d[vc] += h[i + o[vc]]
    vd = (j + o >= 0) & (j + o < n)
    d[vd] -= h[j + o[vd]]
    d[center] += -2.0
    ip = j - i + center
    if 0 <= ip < total:
        d[ip] += 1.0
    ineg = i - j + center
    if 0 <= ineg < total:
        d[ineg] += 1.0
    return d

def binary_local_search(h_init, n, target, rng, remaining_frac):
    h = h_init.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    c5 = float(np.max(corr)) * dx
    best_c5 = c5
    best_h = h.copy()
    total = 2 * n - 1
    stall = 0
    t_start = time.time()
    max_time = remaining_frac

    while True:
        if time.time() - t_start > max_time:
            break
        n_on, n_zer = len(ones), len(zeros)
        npair = min(300, n_on * n_zer)
        idxs = rng.integers(0, n_on * n_zer, size=npair)
        curr_max = np.max(corr)
        found = False
        best_improve = None

        for s in range(npair):
            idx = int(idxs[s])
            ii = idx // n_zer
            jj = idx % n_zer
            ip = ones[ii]
            jp = zeros[jj]
            d = delta_corr(h, ip, jp, n)
            new_max = np.max(corr + d)
            if new_max < curr_max - 1e-12:
                if best_improve is None or new_max < best_improve[0]:
                    best_improve = (new_max, ii, jj, ip, jp, d)

        if best_improve is not None:
            nm, ii, jj, ip, jp, d = best_improve
            h[ip] = 0.0
            h[jp] = 1.0
            ones[ii] = jp
            zeros[jj] = ip
            corr = corr + d
            c5 = nm * dx
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
            found = True
        else:
            stall += 1
            if stall >= 80:
                ii = int(rng.integers(n_on))
                jj = int(rng.integers(n_zer))
                ip, jp = ones[ii], zeros[jj]
                h[ip] = 0.0
                h[jp] = 1.0
                ones[ii] = jp
                zeros[jj] = ip
                corr = np.correlate(h, 1.0 - h, mode="full")
                stall = 0
    return best_h, best_c5

def fractional_refine(h, n, target, T_slsqp, time_left):
    h_c = h.astype(float).copy()
    t0 = time.time()
    h_c = slsqp_phase(h_c, n, target, T_slsqp, 2000)
    c5 = compute_c5(h_c)
    if time.time() - t0 < time_left:
        h_c = slsqp_phase(h_c, n, target, max(1.0, T_slsqp * 0.5), 2000)
        c5 = compute_c5(h_c)
    return h_c, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed_frac():
        return (time.time() - t0) / budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(100) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP refinement (short)
    for T, mit in [(50, 200), (20, 300), (10, 400), (5, 500)]:
        if elapsed_frac() > 0.15:
            break
        h = slsqp_phase(h, n, target, T, mit)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 2: Binary round + local search on n
    h_bin = to_binary(h_best, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()

    h_bin2, c5_bin2 = binary_local_search(h_bin, n, target, rng,
                                          max(5.0, budget_s * 0.2 - (time.time() - t0)))
    if c5_bin2 < c5_best:
        c5_best = c5_bin2
        h_best = h_bin2.copy()

    # Phase 3: Try different n values
    for n_try in [n - 20, n + 20, n - 40, n + 40, 200, 240]:
        if elapsed_frac() > 0.55 or n_try < 40:
            break
        target_t = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_t = np.interp(x_new, x_old, h_best)
        h_t = project_feasible(h_t, target_t)
        h_t = slsqp_phase(h_t, n_try, target_t, 30, 300)
        h_t = slsqp_phase(h_t, n_try, target_t, 5, 500)
        h_bt = to_binary(h_t, target_t)
        c5_t = compute_c5(h_bt)
        h_bt2, c5_t2 = binary_local_search(h_bt, n_try, target_t, rng,
                                           max(3.0, budget_s * 0.1 - (time.time() - t0)))
        if c5_t2 < c5_best:
            c5_best = c5_t2
            h_best = h_bt2.copy()
            n = n_try
            target = target_t
        elif c5_t < c5_best:
            c5_best = c5_t
            h_best = h_bt.copy()
            n = n_try
            target = target_t

    # Phase 4: Fractional refinement on best
    if elapsed_frac() < 0.75:
        h_f, c5_f = fractional_refine(h_best, n, target, 5.0, budget_s * 0.08)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f.copy()
        h_f2, c5_f2 = fractional_refine(h_best, n, target, 2.0, budget_s * 0.05)
        if c5_f2 < c5_best:
            c5_best = c5_f2
            h_best = h_f2.copy()

    # Phase 5: More binary local search from best (possibly fractional)
    if elapsed_frac() < 0.90:
        h_b5 = to_binary(h_best, target)
        c5_b5 = compute_c5(h_b5)
        if c5_b5 < c5_best:
            c5_best = c5_b5
            h_best = h_b5.copy()
        h_b5b, c5_b5b = binary_local_search(h_b5, n, target, rng,
                                            max(2.0, budget_s * 0.08 - (time.time() - t0)))
        if c5_b5b < c5_best:
            c5_best = c5_b5b
            h_best = h_b5b.copy()

    # Phase 6: Final polish - subgradient on continuous
    if elapsed_frac() < 0.95:
        dx = 2.0 / n
        for _ in range(500):
            if elapsed_frac() > 0.95:
                break
            corr = np.correlate(h_best, 1.0 - h_best, mode="full")
            top_k = 3
            top_idx = np.argsort(corr)[-top_k:]
            center = n - 1
            g = np.zeros(n)
            for idx in top_idx:
                o = int(idx) - center
                if o == 0:
                    g += (1.0 - 2.0 * h_best)
                elif o > 0:
                    g[o:] += (1.0 - h_best[:n - o])
                    g[:n - o] -= h_best[o:]
                else:
                    p = -o
                    g[:n - p] += (1.0 - h_best[p:])
                    g[p:] -= h_best[:n - p]
            g *= dx / top_k
            for lr in [0.01, 0.005, 0.002]:
                h_c = project_feasible(h_best - lr * g, target)
                c5_c = compute_c5(h_c)
                if c5_c < c5_best:
                    c5_best = c5_c
                    h_best = h_c.copy()
                    break

    h_final = project_feasible(h_best, target)
    c5_final = min(compute_c5(h_final), c5_best)
    return (h_final, c5_final, len(h_final))
