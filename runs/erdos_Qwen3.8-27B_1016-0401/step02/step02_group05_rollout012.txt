    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_obj(x, T):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / len(x))

def sm_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    g = np.zeros(n)
    c = n - 1
    for o in range(-(n-1), n):
        w = sw[c + o]
        if w < 1e-14: continue
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n-o])
            g[:n-o] -= w * x[o:]
        else:
            p = -o
            g[:n-p] += w * (1.0 - x[p:])
            g[p:] -= w * x[:n-p]
    return g * dx

    for _ in range(80):

        if abs(d) < 1e-14: break
        if d > 0:
            for i in np.argsort(-(1-h)):
                if d <= 0: break

        else:
            for i in np.argsort(-h):
                if d >= 0: break
                s = min(-d, h[i]); h[i] -= s; d += s
    return h

def eq_fun(x, tgt): return x.sum() - tgt
def eq_jac(x, tgt): return np.ones(len(x))
