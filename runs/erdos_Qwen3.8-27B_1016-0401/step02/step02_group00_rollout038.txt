def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-15:
                break
    return h

def lag_obj(x, lag_idx):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        idx = np.argsort(1.0 - h if diff > 0 else h)[::-1]
        for i in idx:
            if diff > 0:
                d = min(diff, 1.0 - h[i])
                if d <= 0: continue
                h[i] += d; diff -= d
            else:
                d = min(-diff, h[i])
                if d <= 0: continue
                h[i] -= d; diff += d
            if abs(diff) < 1e-15: break
    return h

def lag_obj(x, li):
    return np.correlate(x, 1.0-x, mode="full")[li] * (2.0/len(x))

def lag_grad(x, li):
    n, o, dx = len(x), li - len(x) + 1, 2.0/len(x)
    g = np.zeros(n)
    if o == 0: g[:] = 1.0 - 2.0*x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] -= x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] -= x[:n-p]
    return g * dx

    c = np.correlate(x, 1.0-x, mode="full")
    mc = c.max()
    return (mc + np.log(np.sum(np.exp(T*(c-mc))))/T) * dx
