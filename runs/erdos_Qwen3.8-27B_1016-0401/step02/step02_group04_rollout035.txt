import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_lag(h, o, n):
    """Gradient of correlation at lag o, without dx factor."""
    g = np.zeros(n)
    if o == 0:
        g = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += (1.0 - h[:n-o])
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def top_k_obj(h, K, T_sm=100):
    """Top-K mean + small soft-max for smoothness."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_K = np.sort(corr)[-K:]
    base = np.mean(top_K)
    mc = np.max(corr)
    shifted = T_sm * (corr - mc)
    smax = mc + np.log(np.sum(np.exp(shifted))) / T_sm
    return (base + 0.1 * smax) * dx

def top_k_grad(h, K, T_sm=100):
    """Gradient of top-K mean + small soft-max."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-K:]
    center = n - 1
    
    # Top-K mean gradient
    g_k = np.zeros(n)
    for k_idx in top_idx:
        o = k_idx - center
        g_k += grad_lag(h, o, n)
    g_k /= K
    
    # Soft-max gradient (small weight)
    mc = np.max(corr)
    shifted = T_sm * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    g_sm = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        g_sm += w * grad_lag(h, o, n)

        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid

def lbfgsb(h0, n, target, obj_func, grad_func, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return obj_func(x) + 100.0 * (x.sum() - target) ** 2
    def grad(x):
        return grad_func(x) + 200.0 * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12,
                       'maxfun': maxiter * 3})
        return project_box_sum(res.x, target)
    except Exception:
        return project_box_sum(h0, target)

def fista(h0, target, grad_func, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    hp = h.copy()
    t_k = 1.0
    step = step_init
    for _ in range(max_iter):

        y = np.clip(y, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_lag(h, o, n):
    g = np.zeros(n)
    if o == 0:
        g = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += (1.0 - h[:n-o])
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def soft_max_obj(x, T, n, target):
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n) + 50.0 * (x.sum() - target) ** 2

def soft_max_grad(x, T, n, target):
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        g += w * grad_lag(x, o, n)
    return g * dx + 100.0 * (x.sum() - target) * np.ones(n)

def top_k_obj(x, K, Tsm, n, target):
    corr = compute_corr(x)
    topK = np.sort(corr)[-K:].mean()
    mc = np.max(corr)
    sh = Tsm * (corr - mc)
    smax = mc + np.log(np.sum(np.exp(sh))) / Tsm
    return (topK + 0.05 * smax) * (2.0 / n) + 50.0 * (x.sum() - target) ** 2

def top_k_grad(x, K, Tsm, n, target):
    corr = compute_corr(x)
    top_idx = np.argsort(corr)[-K:]
    center = n - 1
    dx = 2.0 / n
    gk = np.zeros(n)
    for ki in top_idx:
        gk += grad_lag(x, ki - center, n)
    gk /= K
    mc = np.max(corr)
    sh = Tsm * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    gs = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        gs += w * grad_lag(x, o, n)
    return (gk + 0.05 * gs) * dx + 100.0 * (x.sum() - target) * np.ones(n)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_lbfgsb(h0, n, target, objf, gradf, args, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf, h0.copy(), args=args, method='L-BFGS-B',
                       jac=gradf, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-13,
                                'maxfun': maxiter * 4})
        return project_box_sum(res.x, target)
    except Exception:
        return project_box_sum(h0, target)

def fista_solve(h0, target, gradf, args, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    hp = h.copy()
    t_k = 1.0
    step = step_init
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - hp)
        y = np.clip(y, 0.0, 1.0)
        g = gradf(y, *args)
        hp = h
        h = project_box_sum(y - step * g, target)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
    return h

def binary_search_pass(h, target, n_evals, rng):
    n = len(h)
    c5 = compute_c5(h)
    for _ in range(n_evals):
        ones = np.where(h > 0.65)[0]
        zeros = np.where(h < 0.35)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h_try = h.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        h_try = project_box_sum(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5 - 1e-10:
            h = h_try
            c5 = c5_try
    return h, c5

    if 'initial_h_values' in globals() and initial_h_values is not None:
