        for (i1, i2, d) in cands:
            if i1 == i2: continue
            if d < 0:
                md = min(h[i1], 1.0 - h[i2])
            else:
                md = min(1.0 - h[i1], h[i2])
            if md < 1e-9: continue
            for fr in fracs:
                dd = md * fr
                ht = h.copy()
                ht[i1] += d * dd
                ht[i2] -= d * dd
                ht = fix_sum(ht, target)
                ct = float(np.max(np.correlate(ht, 1 - ht, mode="full")) * dx)
                if ct < best_val:
                    best_val = ct; best_move = ht
                elif ct < best_sa_val and stall > 300:
                    prob = np.exp(-(ct - best) / sa_temp)
                    if rng.random() < prob:
                        best_sa_val = ct; sa_acc = ht

        if best_move is not None:
            h = best_move
            corr = np.correlate(h, 1 - h, mode="full")
            best = best_val; stall = 0
        elif sa_acc is not None:
            h = sa_acc
            corr = np.correlate(h, 1 - h, mode="full")
            best = max(best, float(np.max(corr) * dx))
            stall = 0
        else:
            stall += 1
            if stall > 800:
                sa_temp *= 0.8
                stall = 0

    return h, best

def seg_flip(h, tidx, tlen, target):
    h2 = h.copy()
    h2[tidx:tidx + tlen] = 1.0 - h2[tidx:tidx + tlen]
    return fix_sum(h2, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70); h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': partial(_eq_fun, target),
             'jac': partial(_eq_jac, target)}]

    # Phase 1: SLSQP temperature annealing
    for T in [100, 50, 25, 12, 7, 4, 2.5, 1.5, 1]:
        if time.time() - t0 > budget_s * 0.28: break
        try:
            res = minimize(sm_obj, h_best, method='SLSQP',
                           jac=sm_jac, args=(T,),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-15})
            if res.fun < 50:
                hn = fix_sum(res.x.copy(), target)
                c5 = compute_c5(hn)
                if c5 < c5_best - 1e-13:
                    c5_best = c5; h_best = hn
        except Exception:
            pass

    # Phase 2: Line-search 2-opt with SA
    h2 = ls_2opt(h_best, target, 12000, t0, budget_s * 0.55, sa_temp=0.003, rng_seed=111)
    if h2[1] < c5_best:
        c5_best = h2[1]; h_best = h2[0]

    # Phase 3: SLSQP from LS result
    for T in [6, 3, 2, 1.3, 0.8, 0.5]:
        if time.time() - t0 > budget_s * 0.72: break
        try:
            res = minimize(sm_obj, h_best, method='SLSQP',
                           jac=sm_jac, args=(T,),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            if res.fun < 50:
                hn = fix_sum(res.x.copy(), target)
                c5 = compute_c5(hn)
                if c5 < c5_best - 1e-13:
                    c5_best = c5; h_best = hn
        except Exception:
            pass

    # Phase 4: More LS with different SA seed
    h4 = ls_2opt(h_best, target, 12000, t0, budget_s * 0.85, sa_temp=0.0015, rng_seed=222)
    if h4[1] < c5_best:
        c5_best = h4[1]; h_best = h4[0]

    # Phase 5: Segment flip perturbations + SLSQP polish
    for flip_r in range(6):
        if time.time() - t0 > budget_s - 60: break
        st = int(rng.integers(0, max(1, n - 20)))
        ln = int(rng.integers(3, min(15, n - st)))
        hf = seg_flip(h_best, st, ln, target)
        c5f = compute_c5(hf)
        if c5f < c5_best + 0.005:
            try:
                res = minimize(sm_obj, hf, method='SLSQP',
                               jac=sm_jac, args=(3.0,),
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                if res.fun < 50:
                    hn = fix_sum(res.x.copy(), target)
                    c5 = compute_c5(hn)
                    if c5 < c5_best - 1e-13:
                        c5_best = c5; h_best = hn
            except Exception:
                pass

    # Final SLSQP + tiny LS
    for T in [1.0, 0.6, 0.3]:
        if time.time() - t0 > budget_s - 20: break
        try:
            res = minimize(sm_obj, h_best, method='SLSQP',
                           jac=sm_jac, args=(T,),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-17})
            if res.fun < 50:
                hn = fix_sum(res.x.copy(), target)
                c5 = compute_c5(hn)
                if c5 < c5_best - 1e-13:
                    c5_best = c5; h_best = hn
        except Exception:
            pass

    h5 = ls_2opt(h_best, target, 4000, t0, budget_s - 2, sa_temp=0.0005, rng_seed=333)
    if h5[1] < c5_best:
        c5_best = h5[1]; h_best = h5[0]

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
