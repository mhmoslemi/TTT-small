                    best_t = t

def coord_descent(h, n, elapsed, t_limit, rng, top_k=20):

        top_idx = np.argsort(corr)[-top_k:]
        order = rng.permutation(n)
        for count, m in enumerate(order):
            if elapsed() > t_limit:
                break

            sl_top = sl[top_idx]
            b_vals = corr[top_idx] - h[m] * sl_top
            t_opt = solve_1d_mm(sl_top, b_vals)

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        idx = np.argsort(h) if diff > 0 else np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 1e-15: continue
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 1e-15: continue
                h[i] -= a; diff += a
            if abs(diff) < 1e-13: break
    return h

def get_slopes(h, m, n):
    c = n - 1
    offs = np.arange(2 * n - 1) - c
    s = np.zeros(2 * n - 1)
    v1 = (m - offs >= 0) & (m - offs < n)
    s[v1] += 1.0 - h[m - offs[v1]]
    v2 = (m + offs >= 0) & (m + offs < n)
    s[v2] -= h[m + offs[v2]]
    return s

    K = len(a)
    best_t, best_v = 0.0, np.max(b)
    v1 = np.max(a + b)
    if v1 < best_v: best_v, best_t = v1, 1.0

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        idx = np.argsort(h) if diff > 0 else np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 1e-15: continue
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 1e-15: continue
                h[i] -= a; diff += a
            if abs(diff) < 1e-13: break
    return h

def get_slopes(h, m, n):
    c = n - 1
    offs = np.arange(2 * n - 1) - c
    s = np.zeros(2 * n - 1)
    v1 = (m - offs >= 0) & (m - offs < n)
    s[v1] += 1.0 - h[m - offs[v1]]
    v2 = (m + offs >= 0) & (m + offs < n)
    s[v2] -= h[m + offs[v2]]
    return s

    K = len(a)
    best_t, best_v = 0.0, float(np.max(b))
    v1 = float(np.max(a + b))
    if v1 < best_v: best_v, best_t = v1, 1.0
    for i in range(K):
        ai = a[i]
        for j in range(i + 1, K):
            da = ai - a[j]
            if abs(da) < 1e-14: continue
            t = (b[j] - b[i]) / da
            if 0.0 <= t <= 1.0:
                v = float(np.max(a * t + b))
                if v < best_v: best_v, best_t = v, t
    return best_t

    tgt = n / 2.0
    h = h.copy()
    prev_max = np.inf

            sl = get_slopes(h, m, n)[ti]
            bv = corr[ti] - h[m] * sl
            to = solve_1d_mm(sl, bv)
            d = to - h[m]
            if abs(d) < 1e-14: continue
            h[m] = to
            corr[ti] += d * sl

            stall += 1
        else:
            stall = 0

    return h
