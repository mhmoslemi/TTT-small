        c5_cur = corr.max() * (2.0/n)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1

    nz = np.where(w > 1e-8)[0]
    for idx in nz:

        return np.ones(n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def round_to_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def lse_obj_val(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

    def obj(x):

        return lse_grad(compute_corr(x), x, T, n)

        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    try:

                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
