import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 0: continue
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 0: continue
                h[i] -= a; diff += a
            if abs(diff) < 1e-15: break
    return h

def lag_grad(h, lag):
    n = len(h)
    o = lag - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] -= h[:n-p]
    return g * dx

def lag_obj(h, lag):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")[lag] * (2.0 / n)

def sm_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / n)

def sm_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    w = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o]
        if wi < 1e-12: continue
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n-o])
            g[:n-o] -= wi * h[o:]
        else:
            p = -o
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
    return g * dx

def sum_fun(h):
    return h.sum() - _TARGET

def sum_jac(h):
    return np.ones(len(h))

def nesterov_phase(h, target, budget_end, rng):
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    c5_local = compute_c5(h_cur)
    for it in range(50000):
        if time.time() > budget_end: break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        worst = int(np.argmax(corr))
        g = lag_grad(h_cur, worst)
        g -= g.sum() / n
        L = it + 50
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        alpha = 2.0 / (L + 3.0)
        momentum = (L - 1.0) / t_k
        h_cand = h_prev + momentum * (h_prev - h_cur) - alpha * g
        h_cand = fix_constraint(h_cand, target)
        h_cur, h_prev = h_cand, h_cur
        if it % 300 == 0:
            c5_check = compute_c5(h_cur)
            if c5_check < c5_local - 1e-12:
                c5_local = c5_check
    return h_cur, c5_local

def slsqp_multi_lag(h, target, budget_end, rng, n_lags=4):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    c5_local = compute_c5(h)
    outer = 0
    while time.time() < budget_end:
        outer += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        center = n - 1
        top = np.argsort(corr)[-n_lags:]
        rand_lags = rng.integers(0, 2*n-1, size=2)
        lags = np.unique(np.concatenate([top.astype(int), rand_lags]))
        for lag in lags:
            if time.time() > budget_end: break
            _TARGET = target
            cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
            try:
                res = minimize(lag_obj, h, args=(lag,), method='SLSQP',
                              jac=lag_grad, bounds=bounds, constraints=cons,
                              options={'maxiter': 800, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_local - 1e-14:
                    c5_local = c5_new
                    h = h_new
            except Exception:
                pass
            if time.time() > budget_end: break
    return h, c5_local

def optimize_at_n(h_src, n_target, target, budget_end, rng):
    if n_target == len(h_src):
        h = h_src.copy()
    else:
        x_old = np.linspace(0, 2.0, len(h_src), endpoint=False)
        x_new = np.linspace(0, 2.0, n_target, endpoint=False)
        h = np.interp(x_new, x_old, h_src)
    h = fix_constraint(h, target)
    # Nesterov
    h, c5 = nesterov_phase(h, target, budget_end, rng)
    if time.time() > budget_end:
        return h, c5
    # SLSQP multi-lag
    h, c5 = slsqp_multi_lag(h, target, budget_end, rng)
    # Soft-max polish
    if time.time() < budget_end - 5:
        n = len(h)
        bounds = [(0.0, 1.0)] * n
        _TARGET = target
        cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
        try:
            res = minimize(sm_obj, h, args=(500,), method='SLSQP',
                          jac=sm_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h, c5 = h_new, c5_new
        except Exception:
            pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(280)
        h_init[::2] = 1.0
    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = fix_constraint(h_init, target0)
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()

    # Phase 1: Nesterov on initial (20% budget)
    p1_end = t0 + budget_s * 0.20
    h1, c5_1 = nesterov_phase(h_init, target0, p1_end, rng)
    if c5_1 < c5_best:
        c5_best, h_best = c5_1, h1

    # Phase 2: Multi-lag SLSQP on initial (25% budget)
    p2_end = t0 + budget_s * 0.45
    h2, c5_2 = slsqp_multi_lag(h_best, target0, p2_end, rng, n_lags=5)
    if c5_2 < c5_best:
        c5_best, h_best = c5_2, h2

    # Phase 3: Multi-n exploration (35% budget)
    p3_end = t0 + budget_s * 0.80
    n_options = [280, 320, 350, 400, 420, 480]
    for n_alt in n_options:
        if time.time() > p3_end: break
        if n_alt == len(h_best): continue
        target_alt = n_alt / 2.0
        x_old = np.linspace(0, 2.0, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        h_alt, c5_alt = optimize_at_n(h_alt, n_alt, target_alt, p3_end, rng)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt

    # Phase 4: Tight SLSQP polish on best (12% budget)
    p4_end = t0 + budget_s * 0.92
    n_f = len(h_best)
    target_f = n_f / 2.0
    bounds_f = [(0.0, 1.0)] * n_f
    for _ in range(30):
        if time.time() > p4_end: break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        # Also try second and third worst
        top3 = np.argsort(corr_f)[-3:]
        for lag in top3:
            if time.time() > p4_end: break
            _TARGET = target_f
            cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
            try:
                res = minimize(lag_obj, h_best, args=(lag,), method='SLSQP',
                              jac=lag_grad, bounds=bounds_f, constraints=cons,

                h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)

                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:

    # Phase 5: Short Nesterov + single-lag SLSQP final (8% budget)
    p5_end = t0 + budget_s - 5.0
    n_f2 = len(h_best)
    target_f2 = n_f2 / 2.0
    h_f, _ = nesterov_phase(h_best, target_f2, min(t0 + budget_s * 0.2, p5_end), rng)
    if compute_c5(h_f) < c5_best:
        c5_best = compute_c5(h_f)

            c5_best = c5_new
            h_best = h_new
    except Exception:

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
