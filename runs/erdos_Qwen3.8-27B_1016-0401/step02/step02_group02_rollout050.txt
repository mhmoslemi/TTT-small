import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_grad(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * (2.0 / n)
    # Gradient
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    grad *= 2.0 / n
    return obj, grad

def slsqp_step(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: lse_obj_grad(x, T, n)[0], h,
                      method='SLSQP', jac=lambda x: lse_obj_grad(x, T, n)[1],
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h

def swap_search(h, n, target, max_rounds=2000, top_k=5):
    """Local search: try transferring mass between pairs of coordinates."""
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    center = n - 1
    
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_cur = float(np.max(corr)) * dx
        
        # Find active lags
        max_val = corr.max()
        active = np.where(corr >= max_val * 0.999)[0]
        
        # Compute directional gradient
        grad = np.zeros(n)
        for idx in active[:top_k]:
            o = int(idx) - center
            if o == 0:
                grad += (1.0 - 2.0 * h)
            elif o > 0:
                grad[o:] += (1.0 - h[:n - o])
                grad[:n - o] -= h[o:]
            else:
                p = -o
                grad[:n - p] += (1.0 - h[p:])
                grad[p:] -= h[:n - p]
        
        # Want to decrease: move from high-grad to low-grad
        # Top donors (high grad → decrease h), top recipients (low grad → increase h)
        sorted_idx = np.argsort(grad)
        n_donors = min(8, n // 4)
        donors = sorted_idx[n - n_donors:]  # highest grad
        recipients = sorted_idx[:n_donors]  # lowest grad
        
        best_c5 = c5_cur
        best_move = None
        for a in recipients:
            for b in donors:
                if a == b:

                # Transfer δ from b to a
                max_d = min(1.0 - h[a], h[b])
                if max_d < 0.005:

                for delta in [max_d * 0.5, max_d * 0.8, max_d * 0.3]:

                    new_h[a] += delta

                    c5_test = compute_c5(new_h)
                    if c5_test < best_c5:

        max_c = corr.max()

        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    g = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            g += wi * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += wi * (1.0 - x[:n - o])
            g[:n - o] -= wi * x[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - x[p:])
            g[p:] -= wi * x[:n - p]
    return g * (2.0 / n)

_S_T = 1.0
_S_N = 100
_S_tgt = 50.0

def _sf(x):
    return lse_obj(x, _S_T, _S_N)

def _sj(x):
    return lse_grad(x, _S_T, _S_N)

def _sc(x):
    return x.sum() - _S_tgt

def _sct(x):
    return np.ones(_S_N)

def do_slsqp(h, n, target, T, maxiter):
    global _S_T, _S_N, _S_tgt
    _S_T, _S_N, _S_tgt = T, n, target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _sct}]
    try:
        res = minimize(_sf, h, method='SLSQP', jac=_sj, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h

def swap_search(h, n, target, max_rounds=600, n_don=8, n_rec=8):
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_c = corr.max()
        active = np.where(corr >= max_c - 0.5)[0]
        center = n - 1
        g = np.zeros(n)
        for idx in active:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        si = np.argsort(g)
        donors = si[n - n_don:n]
        recips = si[:n_rec]
        best_c5, best_a, best_b, best_d = c5_cur, -1, -1, -1
        for a in recips:
            for b in donors:
                if a == b:
                    continue
                md = min(1.0 - h[a], h[b])
                if md < 0.005:
                    continue
                for fr in [0.3, 0.5, 0.7, 0.95]:
                    d = md * fr
                    h[a] += d
                    h[b] -= d
                    c = compute_c5(h)
                    if c < best_c5:
                        best_c5, best_a, best_b, best_d = c, a, b, d
                    h[a] -= d
                    h[b] += d
        if best_a >= 0:
            h[best_a] += best_d
            h[best_b] -= best_d
            c5_cur = best_c5
        else:
            break
    return h, c5_cur

def proj_grad(h, n, target, T, n_steps, lr0=0.4):
    for it in range(n_steps):
        g = lse_grad(h, T, n)
        lr = lr0 / (1.0 + 0.05 * it)
        obj_cur = lse_obj(h, T, n)
        for _ in range(25):
            h_new = project_feasible(h - lr * g, target)
            if lse_obj(h_new, T, n) < obj_cur - 1e-15:
                h = h_new
                break
            lr *= 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(100) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP continuation
    sched1 = [(50, 200), (20, 300), (10, 400), (5, 500), (3, 700),
              (2, 1000), (1.5, 1200), (1.0, 1500), (0.7, 2000)]
    for T, mit in sched1:
        if elapsed() > budget_s - 400:
            break
        h = do_slsqp(h, n, target, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 2: Swap local search (aggressive)
    if elapsed() < budget_s - 250:
        h2 = h_best.copy()
        h2, c5_sw = swap_search(h2, n, target, max_rounds=1200, n_don=10, n_rec=10)
        if c5_sw < c5_best:
            c5_best = c5_sw
            h_best = h2.copy()

    # Phase 3: SLSQP polish from swap result
    if elapsed() < budget_s - 200:
        for T in [0.8, 0.6, 0.5]:
            if elapsed() > budget_s - 160:
                break
            h3 = do_slsqp(h_best, n, target, T, 3000)
            c5_cur = compute_c5(h3)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h3.copy()

    # Phase 4: Second swap search with wider neighborhood
    if elapsed() < budget_s - 150:
        h4 = h_best.copy()
        h4, c5_sw2 = swap_search(h4, n, target, max_rounds=800, n_don=12, n_rec=12)
        if c5_sw2 < c5_best:
            c5_best = c5_sw2
            h_best = h4.copy()

    # Phase 5: Projected gradient at very low T
    if elapsed() < budget_s - 100:
        h5 = h_best.copy()
        h5 = proj_grad(h5, n, target, T=0.8, n_steps=400, lr0=0.2)
        c5_cur = compute_c5(h5)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h5.copy()

    # Phase 6: Final SLSQP + swap
    if elapsed() < budget_s - 80:
        h6 = do_slsqp(h_best, n, target, 0.5, 2000)
        c5_cur = compute_c5(h6)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h6.copy()
        if elapsed() < budget_s - 30:
            h6b = h_best.copy()
            h6b, c5_sw3 = swap_search(h6b, n, target, max_rounds=400, n_don=10, n_rec=10)
            if c5_sw3 < c5_best:
                c5_best = c5_sw3
                h_best = h6b.copy()

    # Phase 7: Small random perturbations with polish
    if elapsed() < budget_s - 30:
        for it in range(200):
            if elapsed() > budget_s - 10:
                break
            scale = 0.002 + 0.003 * (it % 5) * 0.4
            noise = rng.normal(0, scale, n)
            h_p = project_feasible(h_best + noise, target)
            c5_p = compute_c5(h_p)
            if c5_p < c5_best:
                c5_best = c5_p
                h_best = h_p.copy()
                if elapsed() < budget_s - 40:
                    h_p = do_slsqp(h_best, n, target, 0.5, 800)
                    c5_p2 = compute_c5(h_p)
                    if c5_p2 < c5_best:
                        c5_best = c5_p2
                        h_best = h_p.copy()

    # Phase 8: Try reduced resolution
    if elapsed() < budget_s - 50 and n > 60:
        for frac in [0.5, 2.0/3]:
            if elapsed() > budget_s - 20:
                break
            n_t = int(n * frac)
            if n_t < 30 or n_t > 500:
                continue
            tgt_t = n_t / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_t, endpoint=False)
            h_t = np.interp(x_new, x_old, h_best)
            h_t = project_feasible(h_t, tgt_t)
            for T in [20, 5, 1, 0.5]:
                if elapsed() > budget_s - 15:
                    break
                h_t = do_slsqp(h_t, n_t, tgt_t, T, 600)
            h_t, _ = swap_search(h_t, n_t, tgt_t, max_rounds=200, n_don=6, n_rec=6)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best:
                c5_best = c5_t
                h_best = h_t.copy()
                n = n_t
                target = tgt_t

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
