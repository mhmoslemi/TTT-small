from scipy.optimize import minimize
import time
import cvxpy as cp
import cvxpy.settings as cp_settings

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def build_lp_matrix(h_star, n):
    """Build constraint matrix A and constant vector for linearized minimax LP."""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    for p in range(n_shifts):
        k = p - (n - 1)  # shift
        if k >= 0:
            # sum over i from k to n-1
            for i in range(k, n):
                # coefficient of h[i] in linearized f_k
                A[p, i] += (1.0 - h_star[i - k])
                # coefficient of h[i-k] (which is h[j] where j = i-k)
                if i - k < n:
                    A[p, i - k] += (-h_star[i])
        else:
            s = -k
            # sum over i from 0 to n-1-s
            for i in range(0, n - s):
                A[p, i] += (1.0 - h_star[i + s])
                if i + s < n:
                    A[p, i + s] += (-h_star[i])
    A *= dx
    return A

def solve_lp_minimax(h_star, n):
    """Solve linearized minimax LP around h_star."""
    A = build_lp_matrix(h_star, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    # A @ h <= t
    constraints.append(A @ h_var <= t_var)
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CBC, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                return h_star.copy(), None
    if h_var.value is None:
        return h_star.copy(), None
    h_new = np.clip(h_var.value, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, float(t_var.value) if t_var.value is not None else None

    """Iterate linearized minimax until convergence or time budget."""
    h = h_init.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_budget:

        h_new, t_lp = solve_lp_minimax(h, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5:

            c5 = c5_new

                c5_blend = compute_c5(h_blend)
                if c5_blend < c5:
                    h = h_blend
                    c5 = c5_blend
                    break
            else:
                # Converged or stuck
                break
    return h, c5

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):

        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def build_lp_data(h_star, n):
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    for k in range(0, n):
        p = k + n - 1
        A[p, k:] += (1.0 - h_star[:n-k])
        A[p, :n-k] += (-h_star[k:])
        b[p] = h_star[k:].dot(h_star[:n-k])

        p = n - 1 - s
        A[p, :n-s] += (1.0 - h_star[s:])
        A[p, s:] += (-h_star[:n-s])
        b[p] = h_star[:n-s].dot(h_star[s:])
    A *= dx
    b *= dx
    return A, b

def solve_lp(h_star, n, timeout=20):
    A, b = build_lp_data(h_star, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0]
    cons.append(A @ hv - tv <= -b)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.CBC, cp.GLPK, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, max_time=timeout)
            if hv.value is not None:
                h_new = np.clip(np.asarray(hv.value).flatten(), 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                return h_new
        except Exception:
            continue
    return h_star.copy()

    h = h_init.copy()
    c5 = compute_c5(h)
    for it in range(max_it):
        if time.time() - t_start > time_limit:
            break
        h_lp = solve_lp(h, n)
        h_blend = damp * h_lp + (1 - damp) * h
        h_blend = project_feasible(h_blend, n)
        c5_blend = compute_c5(h_blend)

            c5 = c5_blend

            c5_try = compute_c5(h_try)
            if c5_try < c5 - 1e-10:
                h = h_try
                c5 = c5_try
            else:
                break

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def soft_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def soft_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(h, n, oi - center)
    return grad * dx

def slsqp_polish(h, n, T, maxiter, ftol, t_start, time_limit):
    hx = h.copy()
    dx = 2.0 / n
    def obj(x):
        if time.time() - t_start > time_limit:
            raise ValueError("timeout")
        return soft_obj(x, n, T)
    def jac(x):
        return soft_grad(x, n, T)
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, hx, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

        m2 = (j + sh) < n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def build_lp(hs, n):
    dx = 2.0 / n
    ns = 2 * n - 1
    A = np.zeros((ns, n))
    b = np.zeros(ns)
    for k in range(n):
        p = k + n - 1
        A[p, k:] += 1.0 - hs[:n - k]
        A[p, :n - k] -= hs[k:]
        b[p] = hs[k:].dot(hs[:n - k])
    for s in range(1, n):
        p = n - 1 - s
        A[p, :n - s] += 1.0 - hs[s:]
        A[p, s:] -= hs[:n - s]
        b[p] = hs[:n - s].dot(hs[s:])
    A *= dx
    b *= dx
    return A, b

def solve_lp(hs, n, M=0, tout=25):
    A, b = build_lp(hs, n)
    ns = 2 * n - 1
    if 0 < M < ns:
        cr = np.correlate(hs, 1.0 - hs, mode="full")
        ix = np.argsort(cr)[ns - M:]
        A, b = A[ix, :], b[ix]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0, A @ hv - tv <= -b]
    pb = cp.Problem(cp.Minimize(tv), cons)
    for sv in [cp.CBC, cp.GLPK, cp.CVXOPT]:
        try:
            pb.solve(solver=sv, max_time=tout)
            if hv.value is not None:
                hw = np.clip(np.asarray(hv.value).ravel(), 0.0, 1.0)
                return project_feasible(hw, n)
        except Exception:
            pass
    return hs.copy()

    d = dp

            break

            ct = compute_c5(ht)
            if ct < c5 - 1e-12:
                h, c5, ok = ht, ct, True
        if ok:
            d = min(d * 1.03, 0.97)
        else:
            break
    return h, c5

_GN = 0
_GT = 1.0

def _gobj(x):
    cr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(cr)
    sh = _GT * (cr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _GT) * (2.0 / _GN)

def _sg(x, n, sh):
    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        m1 = j >= sh
        m2 = (j + sh) < n
        g[m1] += 1.0 - x[j[m1] - sh]
        g[m2] -= x[j[m2] + sh]
    else:
        s = -sh
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += 1.0 - x[j[m1] + s]
        g[m2] -= x[j[m2] - s]
    return g

def _gjac(x):
    cr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(cr)
    sh = _GT * (cr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    c = _GN - 1
    g = np.zeros(_GN)
    for oi in range(len(sw)):
        if sw[oi] > 1e-12:
            g += sw[oi] * _sg(x, _GN, oi - c)
    return g * (2.0 / _GN)

def _gcf(x):
    return x.sum() - _GN / 2.0

def _gcj(x):
    return np.ones(_GN)

def slsqp(h, n, T, mi, ft):
    global _GN, _GT
    _GN, _GT = n, float(T)
    try:
        r = minimize(_gobj, h, method='SLSQP', jac=_gjac,
                     bounds=[(0.0, 1.0)] * n,

                     options={'maxiter': mi, 'ftol': ft})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)


def build_lp(hs, n):
    dx = 2.0 / n
    ns = 2 * n - 1
    A = np.zeros((ns, n))
    b = np.zeros(ns)
    for k in range(n):
        p = k + n - 1
        A[p, k:] += 1.0 - hs[:n-k]
        A[p, :n-k] -= hs[k:]
        b[p] = hs[k:].dot(hs[:n-k])
    for s in range(1, n):
        p = n - 1 - s
        A[p, :n-s] += 1.0 - hs[s:]
        A[p, s:] -= hs[:n-s]
        b[p] = hs[:n-s].dot(hs[s:])
    A *= dx
    b *= dx
    return A, b


def solve_lp(hs, n, M=0, tout=25):
    A, b = build_lp(hs, n)
    ns = 2 * n - 1
    if 0 < M < ns:
        cr = np.correlate(hs, 1.0 - hs, mode="full")
        ix = np.argsort(cr)[ns - M:]
        A, b = A[ix, :], b[ix]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0, A @ hv - tv <= -b]
    pb = cp.Problem(cp.Minimize(tv), cons)
    for sv in [cp.CBC, cp.GLPK, cp.CVXOPT]:
        try:
            pb.solve(solver=sv, max_time=tout)
            if hv.value is not None:
                hw = np.clip(np.asarray(hv.value).ravel(), 0.0, 1.0)
                return project_feasible(hw, n)
        except Exception:
            pass
    return hs.copy()


def lp_it(hi, n, t0, tlim, mi=35, dp=0.5, M=0):
    h = hi.copy()
    c5 = compute_c5(h)
    d = dp
    for _ in range(mi):
        if time.time() - t0 > tlim:
            break
        hl = solve_lp(h, n, M=M)
        hb = d * hl + (1.0 - d) * h
        hb = project_feasible(hb, n)
        cb = compute_c5(hb)
        ok = False
        if cb < c5 - 1e-12:
            h, c5, ok = hb, cb, True
        else:
            ht = project_feasible(hl, n)
            ct = compute_c5(ht)
            if ct < c5 - 1e-12:
                h, c5, ok = ht, ct, True
        if ok:
            d = min(d * 1.03, 0.97)
        else:
            break
    return h, c5


_GN = 0
_GT = 1.0


def _gobj(x):
    cr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(cr)
    sh = _GT * (cr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _GT) * (2.0 / _GN)


def _sg(x, n, sh):
    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        m1 = j >= sh
        m2 = (j + sh) < n
        g[m1] += 1.0 - x[j[m1] - sh]
        g[m2] -= x[j[m2] + sh]
    else:
        s = -sh
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += 1.0 - x[j[m1] + s]
        g[m2] -= x[j[m2] - s]
    return g


def _gjac(x):
    cr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(cr)
    sh = _GT * (cr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    c = _GN - 1
    g = np.zeros(_GN)
    for oi in range(len(sw)):
        if sw[oi] > 1e-12:
            g += sw[oi] * _sg(x, _GN, oi - c)
    return g * (2.0 / _GN)

    return x.sum() - _GN / 2.0


def _gcj(x):
    return np.ones(_GN)
