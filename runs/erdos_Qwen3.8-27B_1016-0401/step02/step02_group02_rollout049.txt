            if c5_l < c5_best:
                c5_best = c5_l
                h_best = h_work.copy()

    # Phase 5: Lower resolution search
    if elapsed() < budget_s - 120 and n >= 200:
        n_lo = n // 2
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        c5_lo = compute_c5(h_lo)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()
        for T in [10, 5, 2, 1.0, 0.5]:
            if elapsed() > budget_s - 60:
                break
            h_lo = slsqp_optimize(h_lo, n_lo, target_lo, T, 400, mu)
            h_lo, _ = subgrad_polish(h_lo, n_lo, target_lo, 150, 1, 0.015)
            c5_lo = compute_c5(h_lo)
            if c5_lo < c5_best:
                c5_best = c5_lo
                h_best = h_lo.copy()

    # Phase 6: Final aggressive polish
    h_fin = h_best.copy()
    nf = len(h_fin)
    tf = nf / 2.0
    if elapsed() < budget_s - 40:
        h_fin = slsqp_optimize(h_fin, nf, tf, 0.3, 1500, mu)
        h_fin, cf = subgrad_polish(h_fin, nf, tf, 300, 1, 0.004)
        if cf < c5_best:
            c5_best = cf
            h_best = h_fin.copy()
        # Try a few more subgrad with K=1
        if elapsed() < budget_s - 20:
            corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
            for _ in range(50):
                g = subgrad_step(h_best, corr_f, tf, nf, 1)
                lr = 0.003
                h_cand = project_feasible(h_best - lr * g, tf)
                c_cand = compute_c5(h_cand)
                if c_cand < c5_best:
                    c5_best = c_cand
                    h_best = h_cand
                    corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
                else:
                    lr *= 0.5
                    h_cand = project_feasible(h_best - lr * g, tf)
                    c_cand = compute_c5(h_cand)
                    if c_cand < c5_best:
                        c5_best = c_cand
                        h_best = h_cand

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
