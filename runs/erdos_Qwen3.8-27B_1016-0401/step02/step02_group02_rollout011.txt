def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def grad_topk(h, K):
    """Gradient of sum of top-K correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in top:
        o = int(idx) - center
        if o == 0:
            g += 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += 1.0 - h[:n-o]
            g[:n-o] -= h[o:]
        else:
            p = -o
            g[:n-p] += 1.0 - h[p:]
            g[p:] -= h[:n-p]
    return g / K, np.max(corr)

def lse_objective(h, T, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc) - T * mc
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * 2.0 / n

    mc = np.max(corr)
    sh = T * (corr - mc)

    w = np.exp(sh) / np.exp(sh).sum()
    center = n - 1

        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

    best_h = h.copy()
    best_c5 = c5_cur
    t_start = time.time()
    for it in range(n_steps):
        if time.time() - t_start > max_time:
            break
        K = K_seq[it % len(K_seq)]
        lr = lr0 / (1.0 + 0.004 * it)
        g = grad_topk(h, K)
        h_new = project_feasible(h - lr * g * dx, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            acc = False
            if T_met > 1e-12:
                acc = rng.random() < np.exp(-max(0.0, (c5_new - c5_cur) / T_met))
            if acc:
                h = h_new
                c5_cur = c5_new
            elif it % 300 == 299:
                noise = rng.normal(0, 0.004, n)
                h_try = project_feasible(h + noise, target)
                ct = compute_c5(h_try)
                if ct < c5_cur:
                    h = h_try
                    c5_cur = ct
    return best_h, best_c5

def make_binary(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def binary_2opt(h_bin, n_swaps, rng):
    n = len(h_bin)
    c5_cur = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5_cur
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for it in range(n_swaps):
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i], h_bin[j] = 0.0, 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_bin.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
    return best_h, best_c5

def beatty_start(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        if (i * alpha) % 1.0 < 0.5:
            h[i] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def _el():
        return time.time() - t0

    # Collect candidates
    candidates = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        candidates.append(h0)

    if len(candidates) == 0:
        candidates.append(beatty_start(280, 0.6180339887))
    else:
        n0 = len(candidates[0])
        candidates.append(beatty_start(n0, 0.6180339887 + 0.1))
        h_blk = np.zeros(n0)
        h_blk[: n0 // 2] = 1.0
        h_blk = h_blk[rng.permutation(n0)]
        candidates.append(h_blk)

    best_h = candidates[0].copy()
    n = len(best_h)
    target = n / 2.0
    best_c5 = float('inf')

    # Optimize each candidate briefly, keep best
    for h_cand in candidates:
        n_c = len(h_cand)
        t_c = n_c / 2.0
        h_cand = project_feasible(h_cand, t_c)
        # Quick SLSQP
        if _el() > budget_s - 300:
            break
        for T, mit in [(50, 300), (15, 500), (5, 800)]:
            if _el() > budget_s - 250:
                break
            h_cand = do_slsqp(h_cand, n_c, t_c, T, mit)
        c5_c = compute_c5(h_cand)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_cand.copy()
            n = n_c
            target = t_c

    # Phase 2: Intensive SGA with cycling K
    h_w = best_h.copy()
    K_seq = [1, 2, 3, 2, 1, 4, 2, 3, 1, 2, 1, 5]
    h_w, c5s = sga_loop(h_w, target, 30000, K_seq, 0.08, 0.003, 0.002, rng, 180)
    if c5s < best_c5:
        best_c5 = c5s
        best_h = h_w.copy()

    # Phase 3: Second SGA with different pattern
    K_seq2 = [1, 1, 2, 1, 1, 3, 1, 2, 1, 1, 4, 1]
    h_w = best_h.copy()
    h_w, c5s2 = sga_loop(h_w, target, 20000, K_seq2, 0.04, 0.002, 0.001, rng, 150)
    if c5s2 < best_c5:
        best_c5 = c5s2
        best_h = h_w.copy()

    # Phase 4: Binary rounding + 2-opt
    h_bin = make_binary(best_h, target)
    h_bin, c5b = binary_2opt(h_bin, 10000, rng)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = h_bin.copy()

    # Phase 5: SGA from binary
    if _el() < budget_s - 150:
        h_w = best_h.copy()
        h_w, c5s3 = sga_loop(h_w, target, 15000, K_seq, 0.06, 0.003, 0.001, rng, 100)
        if c5s3 < best_c5:
            best_c5 = c5s3
            best_h = h_w.copy()

    # Phase 6: SLSQP polish
    if _el() < budget_s - 80:
        for T, mit in [(3, 2000), (2, 2000), (1.5, 3000)]:
            if _el() > budget_s - 20:
                break
            h_p = do_slsqp(best_h, n, target, T, mit)
            cp = compute_c5(h_p)
            if cp < best_c5:
                best_c5 = cp
                best_h = h_p.copy()

    # Phase 7: Tiny perturbations
    for it in range(500):
        if _el() > budget_s - 5:
            break
        scale = 0.002 + 0.002 * (it // 100)
        noise = rng.normal(0, scale, n)
        h_try = project_feasible(best_h + noise, target)
        ct = compute_c5(h_try)
        if ct < best_c5:
            best_c5 = ct
            best_h = h_try.copy()

    # Phase 8: Try lower resolution
    if _el() < budget_s - 30 and n > 140:
        n2 = max(140, n // 2)
        t2 = n2 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, t2)
        h2, c5L = sga_loop(h2, t2, 15000, K_seq, 0.08, 0.003, 0.002, rng, 20)
        if c5L < best_c5:
            best_c5 = c5L
            best_h = h2.copy()

    # Final: one more round of SGA on the absolute best
    if _el() < budget_s - 10:
        h_w = best_h.copy()
        n_f = len(h_w)
        t_f = n_f / 2.0
        h_w, c5f = sga_loop(h_w, t_f, 10000, K_seq, 0.03, 0.002, 0.0005, rng, max(5, budget_s - _el() - 10))
        if c5f < best_c5:
            best_c5 = c5f
            best_h = h_w.copy()

    c5_final = min(compute_c5(best_h), best_c5)
    return (best_h, c5_final, len(best_h))
