    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Progressive T L-BFGS-B (moderate T for landscape shaping)
    for T in [100, 300, 800]:
        if elapsed() > budget_s - 500:
            break
        h_new = lbfgsb_step(h_best, target, T, 4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Very high T for precise max targeting
    for T in [2000, 5000, 10000]:
        if elapsed() > budget_s - 400:
            break
        h_new = lbfgsb_step(h_best, target, T, 6000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Direct subgradient on active max lags
    if elapsed() < budget_s - 350:
        step_sz = 0.005
        for _ in range(30000):
            if elapsed() > budget_s - 350:
                break
            h_new = subgrad_step_direct(h_best, target, step_sz)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                step_sz = min(0.008, step_sz * 1.02)
            else:
                step_sz = max(0.0005, step_sz * 0.97)

    # Phase 4: Structured perturbations + high-T reoptimization
    n_perturb = 0
    while elapsed() < budget_s - 180:
        n_perturb += 1
        ptype = n_perturb % 5
        if ptype == 0:
            h_pert = h_best + rng.normal(0, 0.015, n)
        elif ptype == 1:
            shift = int(rng.integers(1, max(2, n // 14)))
            h_pert = np.roll(h_best, shift).copy()
        elif ptype == 2:
            h_hat = np.fft.rfft(h_best)
            h_hat[1:] += rng.normal(0, 0.008, len(h_hat[1:]))
            h_pert = np.fft.irfft(h_hat, n)
        elif ptype == 3:
            h_pert = h_best.copy()
            idx = int(rng.integers(0, n - 10))
            w = int(rng.integers(2, max(3, n // 10)))
            h_pert[idx:min(idx + w, n)] = 1.0 - h_pert[idx:min(idx + w, n)]
        else:
            scale = max(0.01, 0.04 / (1.0 + 0.02 * n_perturb))
            h_pert = h_best + rng.normal(0, scale, n)

        h_pert = project_box_sum(h_pert, target)
        T = 3000 + int(rng.integers(0, 5000))
        h_new = lbfgsb_step(h_pert, target, T, 1500)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try higher resolutions
    for n_try in [420, 560]:
        if elapsed() > budget_s - 80:
            break
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        for T in [2000, 8000]:
            if elapsed() > budget_s - 40:
                break
            h_t = lbfgsb_step(h_try, t_try, T, 4000)
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n = n_try
            target = t_try

    # Phase 6: Final ultra-high-T polish
    if elapsed() < budget_s - 3:
        h_new = lbfgsb_step(h_best, target, 20000, 5000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
