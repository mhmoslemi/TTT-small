                if add <= 0: continue

                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0: continue

                diff += sub

import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0
_CUR_LAG = 0
_CUR_T = 100.0
_CUR_P = 8

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        idx = np.argsort(1.0 - h)[::-1] if d > 0 else np.argsort(h)[::-1]
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0: continue
                h[i] += a; d -= a
            else:
                s = min(-d, h[i])
                if s <= 0: continue
                h[i] -= s; d += s
            if abs(d) < 1e-14: break
    return h

def _eq_fun(x):
    return x.sum() - _TARGET

def _eq_jac(x):
    return np.ones(len(x))

def _lag_jac(x):
    return lag_grad(x, _CUR_LAG)

def _sm_jac(x):
    return sm_grad(x, _CUR_T)

def _pn_jac(x):
    return pn_grad(x, _CUR_P)

def lag_obj(x, li):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[li] * (2.0 / n)

def lag_grad(x, li):
    n = len(x)
    o = li - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        f = -o
        g[:n - f] += 1.0 - x[f:]
        g[f:] += -x[:n - f]
    return g * dx

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    ctr = n - 1
    dx = 2.0 / n
    g = np.zeros(n)

        w = sw[ctr + o]

        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n - o])
            g[:n - o] += -w * x[o:]
        else:
            f = -o
            g[:n - f] += w * (1.0 - x[f:])
            g[f:] += -w * x[:n - f]
    return g * dx

def pn_obj(x, p):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.sum(corr ** p) ** (1.0 / p) * (2.0 / n)

def pn_grad(x, p):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    ctr = n - 1
    dx = 2.0 / n
    sp = np.sum(corr ** p)
    if sp < 1e-30: return np.zeros(n)
    pref = sp ** (1.0 / p - 1.0)
    wt = corr ** (p - 1)
    g = np.zeros(n)

            f = -o

def do_slsqp_lag(h, lag, bnds, maxiter, ftol):

                     jac=_lag_jac, bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return fix_constraint(np.clip(r.x, 0, 1), _TARGET)
    except Exception:
        return h.copy()

                     options={'maxiter': maxiter, 'ftol': ftol})

        return h.copy()
