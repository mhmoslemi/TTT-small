import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    obj = (mc + np.log(e.sum()) / T) * dx
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-14)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx] * dx
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad

def slsqp_phase(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj_grad(x, T)[0]
    def jac(x):
        return lse_obj_grad(x, T)[1]
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h

def topk_subgrad(h, K, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argpartition(corr, -K)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        w = dx / K
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g

def subgrad_polish(h, n, target, n_steps, K, lr_init):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_steps):
        g = topk_subgrad(h, K, n)
        h_new = project_feasible(h - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.6
            h = project_feasible(h - lr * g, target)
        lr = min(lr * 1.005, 0.08)
    return best_h, compute_c5(best_h)

def pairwise_local_search(h, target, n_trials, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = rng.uniform(0.005, 0.08)
        if best_h[i] + d > 1.0 or best_h[j] - d < 0.0:
            continue
        trial = best_h.copy()
        trial[i] += d
        trial[j] -= d
        c5 = compute_c5(trial)
        if c5 < best_c5:
            best_c5 = c5
            best_h = trial
    return best_h, best_c5

def round_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 280
        target = n / 2.0
        h = np.zeros(n)
        h[::2] = 1.0

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: LSE continuation SLSQP
    schedule = [(80, 150), (40, 200), (20, 300), (12, 400),
                (8, 500), (5, 600), (3.5, 800), (2.5, 1000),
                (1.8, 1200), (1.2, 1500), (0.9, 2000)]
    for T, mit in schedule:
        if elapsed() > budget_s - 350:
            break
        h = slsqp_phase(h, n, target, T, mit)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 2: Very low T SLSQP from best
    if elapsed() < budget_s - 250:
        for T in [0.7, 0.5, 0.4]:
            if elapsed() > budget_s - 200:
                break
            h = slsqp_phase(h_best.copy(), n, target, T, 2500)
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()

    # Phase 3: Subgradient polish with multiple K
    h_cur = h_best.copy()
    for K_val in [10, 7, 5, 4, 3, 2]:
        if elapsed() > budget_s - 120:
            break
        h_cur, c5 = subgrad_polish(h_cur, n, target, 250, K_val, lr_init=0.025)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_cur.copy()

    # Phase 4: Pairwise local search
    if elapsed() < budget_s - 40:
        h_best, c5 = pairwise_local_search(h_best, target, 1500, rng)
        if c5 < c5_best:
            c5_best = c5

    # Phase 5: Random restart perturbations
    for restart in range(4):
        if elapsed() > budget_s - 50:
            break
        scale = 0.03 + restart * 0.02
        h_p = project_feasible(h_best + rng.normal(0, scale, n), target)
        # Quick SLSQP polish
        h_p = slsqp_phase(h_p, n, target, 3.0, 400)
        h_p = slsqp_phase(h_p, n, target, 1.5, 600)
        c5_p = compute_c5(h_p)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()

    # Phase 6: Lower resolution attempt
    if elapsed() < budget_s - 80:
        n_lo = 200
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        for T in [30, 15, 8, 4, 2.5, 1.5, 1.0]:
            if elapsed() > budget_s - 30:
                break
            h_lo = slsqp_phase(h_lo, n_lo, target_lo, T, 600)
        c5_lo = compute_c5(h_lo)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()
            n = n_lo
            target = target_lo

    # Phase 7: Binary rounding and local search on binary
    h_bin = round_binary(h_best, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    if elapsed() < budget_s - 10:
        h_best, c5 = pairwise_local_search(h_best, target, 800, rng)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_bin if c5_bin < c5 else h_best

    c5_final = compute_c5(h_best)
    return (h_best, min(c5_final, c5_best), len(h_best))
