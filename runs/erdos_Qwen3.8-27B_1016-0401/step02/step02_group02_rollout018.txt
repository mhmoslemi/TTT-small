            h_p = np.tile(pat, n_tiles)
            h_p = project_feasible(h_p, target)

import numpy as np
import time
from scipy.optimize import minimize, differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_step(h_in, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h_in, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h_in

def binary_round(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def directed_binary_search(h_bin, target, rng, max_iters=2000, deadline=1e18):
    n = len(h_bin)
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    stuck = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = np.max(corr)
        c5_cur = mc * (2.0 / n)
        
        h_ones = np.where(h > 0.5)[0]
        h_zeros = np.where(h < 0.5)[0]
        if len(h_ones) == 0 or len(h_zeros) == 0:
            break
        
        top_k = min(5, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        center = n - 1
        g = np.zeros(n)
        for idx in top_idx:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        
        ns = 5
        one_scores = g[h_ones]
        zero_scores = g[h_zeros]
        top_ones = h_ones[np.argsort(one_scores)[:ns]]
        top_zeros = h_zeros[np.argsort(zero_scores)[-ns:]]
        
        found = False
        local_best = c5_cur

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue

                h[i] = 0.0

                h[j] = 0.0
                h[i] = 1.0

                    local_best = c5_test
                    local_swap = (i, j)

            h[i] = 0.0

                h[i] = 0.0
