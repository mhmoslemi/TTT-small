        g[p:] -= h[:n - p]
    return g

def lse_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def lse_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_run(h, n, target, bounds, T, maxiter):
    def obj(x):
        return lse_obj(x, T, n)
    def jac(x):
        return lse_grad(x, T, n)
    def cfin(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfin, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def structured_init(n, rng):
    h = np.zeros(n)
    block = max(2, n // 50)
    fills = 0
    i = 0
    while fills < n // 2 and i < n:
        for j in range(block):
            if i + j < n and fills < n // 2:
                h[i + j] = 1.0
                fills += 1
        i += 2 * block
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = structured_init(280, rng)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Fast projected subgradient with decreasing K
    # Alternating top-K and single-top for diversity
    schedules = [
        (20, 8, 400, 0.025, 0.002),
        (8, 4, 400, 0.018, 0.0015),
        (4, 2, 500, 0.012, 0.001),
        (2, 1, 600, 0.008, 0.0008),
        (1, 1, 800, 0.004, 0.0004),
    ]
    for Ks, Ke, nit, lr0, lrd in schedules:
        if elapsed() > budget_s - 300:
            break
        for it in range(nit):
            if elapsed() > budget_s - 250:
                break
            K = max(1, int(round(Ks + (Ke - Ks) * it / max(1, nit - 1))))
            lr = lr0 / (1.0 + lrd * it)
            corr = np.correlate(h, 1.0 - h, mode="full")
            if K == 1:
                g = make_subgrad_one(h, np.argmax(corr), n)
            else:
                top_idx = np.argpartition(corr, -K)[-K:]
                g = np.zeros(n)
                for idx in top_idx:
                    g += make_subgrad_one(h, idx, n)
                g /= K
            h = project_feasible(h - lr * g, target)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 2: SLSQP continuation with decreasing T
    T_schedule = [(100, 600), (50, 800), (25, 1000), (12, 1200),
                  (6, 1500), (3, 2000), (1.5, 2500), (1.0, 3000)]
    for T, mit in T_schedule:
        if elapsed() > budget_s - 150:
            break
        h = slsqp_run(h, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 3: Fine top-1 subgradient with adaptive LR
    h = h_best.copy()
    lr = 0.003
    no_improve = 0
    for it in range(5000):
        if elapsed() > budget_s - 80 or no_improve > 300:
            break
        c5_cur = compute_c5(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        g = make_subgrad_one(h, idx, n)
        h_cand = project_feasible(h - lr * g, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_cur - 1e-11:
            h = h_cand
            no_improve = 0
            if no_improve == 0 and lr < 0.01:
                lr *= 1.005
        else:
            no_improve += 1
            lr *= 0.97
        if c5_cand < c5_best:
            c5_best = c5_cand
            h_best = h_cand.copy()

    # Phase 4: Random perturbations + mini SLSQP
    for it in range(300):
        if elapsed() > budget_s - 30:
            break
        scale = 0.003 + 0.008 * (it % 7) * 0.2
        noise = rng.normal(0, scale, n)
        h_p = project_feasible(h_best + noise, target)
        c5_p = compute_c5(h_p)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()
            if elapsed() < budget_s - 40:
                h_s = slsqp_run(h_best, n, target, bounds, 5, 500)
                c5_s = compute_c5(h_s)
                if c5_s < c5_best:
                    c5_best = c5_s
                    h_best = h_s.copy()

    # Phase 5: Try coarser resolution
    if elapsed() < budget_s - 60 and n > 160:
        n_lo = n // 2
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        c5_lo = compute_c5(h_lo)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()
        bounds_lo = [(0.0, 1.0)] * n_lo
        for T in [30, 10, 5, 3]:
            if elapsed() > budget_s - 25:
                break
            h_lo = slsqp_run(h_lo, n_lo, target_lo, bounds_lo, T, 800)
            c5_lo = compute_c5(h_lo)
            if c5_lo < c5_best:
                c5_best = c5_lo
                h_best = h_lo.copy()

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, len(h_best))
