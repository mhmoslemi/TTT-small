After swap: corr'[o] = corr[o] + delta

The terms affected:

- t = j-o: h[t+o] = h[j] changes from 0 to 1, so term changes from 0 to (1-h[j-o]). Delta: +(1-h[j-o])
- t = i: (1-h[t]) changes from 0 to 1, so term changes from h[i+o]*0 to h[i+o]*1. Delta: +h[i+o]
- t = j: (1-h[t]) changes from 1 to 0, so term changes from h[j+o]*1 to h[j+o]*0. Delta: -h[j+o]

    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    """Compute subgradient of sum of top-K correlations (approximation to max)."""
    dx = 2.0 / n
    top_idx = np.argsort(corr)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]

    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=[(0,1)]*n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h

def make_periodic_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    blocks = 4
    block_size = n // blocks
    ones_per_block = half // blocks
    for b in range(blocks):
        start = b * block_size
        positions = rng.permutation(block_size)[:ones_per_block]
        h[start + positions] = 1.0
    if h.sum() != half:
        diff = half - int(h.sum())
        idx = np.where(h == 0)[0]
        h[idx[:diff]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = 999.0
    best_n = 0

    def update(h, c5):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    h_starts = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
        target = n / 2.0
        h_init = project_feasible(h_init, target)
        h_starts.append(('init', h_init, n, target))
        h_starts.append(('comp', 1.0 - h_init, n, target))
        h_starts.append(('rev', h_init[::-1], n, target))
    else:
        n = 280
        target = n / 2.0
        h_starts.append(('zero', np.zeros(n), n, target))

    # Phase 1: Continuation PG on initial solutions
    T_sched = [200, 80, 40, 20, 10, 5, 3, 2, 1.5, 1.0]
    for name, h0, n, target in h_starts[:3]:
        if elapsed() > budget_s - 300:
            break
        h, c5 = pg_continuation(h0, target, T_sched, 150, rng)
        update(h, c5)
        if elapsed() > budget_s - 150:
            break
        h2 = slqp_polish(h, n, target, 1.0, 2000)
        c5_2, _ = compute_c5(h2)
        update(h2, c5_2)

    # Phase 2: Binary local search
    for name, h0, n, target in h_starts[:4]:
        if elapsed() > budget_s - 200:
            break
        h_bin = binary_round(best_h if best_h is not None else h0, target)
        h_b, c5_b = binary_local_search(h_bin, 300, 12, rng)
        update(h_b, c5_b)
        h_b2, c5_b2 = binary_local_search(h_b, 300, 8, rng)
        update(h_b2, c5_b2)

    # Phase 3: Different resolutions
    for n_try in [200, 320, 400]:
        if elapsed() > budget_s - 100 or best_h is None:
            break
        if n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, target_try)
        h_t, c5_t = pg_continuation(h_try, target_try, [50, 20, 10, 5, 3, 2, 1.5], 100, rng)
        update(h_t, c5_t)
        if elapsed() > budget_s - 60:
            break
        h_tb = binary_round(h_t, target_try)
        h_tb, c5_tb = binary_local_search(h_tb, 200, 10, rng)
        update(h_tb, c5_tb)

    # Phase 4: Multi-start with structured inits
    if elapsed() < budget_s - 120:
        n_s = len(best_h) if best_h is not None else 280
        target_s = n_s / 2.0
        for s in range(6):
            if elapsed() > budget_s - 60:
                break
            h_p = make_periodic_init(n_s, np.random.default_rng(seed + 100 + s))
            h_s, c5_s = pg_continuation(h_p, target_s, [100, 50, 20, 10, 5, 3, 2], 100, rng)
            update(h_s, c5_s)
            h_sb = binary_round(h_s, target_s)
            h_sb, c5_sb = binary_local_search(h_sb, 200, 10, rng)
            update(h_sb, c5_sb)

    # Phase 5: Final SLSQP polish on best
    if best_h is not None and elapsed() < budget_s - 30:
        n_f = len(best_h)
        target_f = n_f / 2.0
        for T_f in [0.8, 0.5, 0.3]:
            if elapsed() > budget_s - 15:
                break
            h_f = slqp_polish(best_h, n_f, target_f, T_f, 3000)
            c5_f, _ = compute_c5(h_f)
            update(h_f, c5_f)

    # Phase 6: Final binary polish
    if best_h is not None and elapsed() < budget_s - 10:
        n_f = len(best_h)
        target_f = n_f / 2.0
        h_fb = binary_round(best_h, target_f)
        h_fb, c5_fb = binary_local_search(h_fb, 500, 15, rng)
        update(h_fb, c5_fb)

    c5_final, _ = compute_c5(best_h)
    return (best_h, min(c5_final, best_c5), len(best_h))
