def softmax_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    # Soft max approximation
    sh = T * (corr - mc)
    # For numerical stability
    sh = sh - sh.max()
    logsum = np.log(np.exp(sh).sum())
    return (mc + logsum / T) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def corr_grad(h, n, s):
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[s:] += (1.0 - h[:m])
        grad[:m] -= h[s:]
    else:
        t = -s
        m = n - t
        grad[:m] += (1.0 - h[t:])
        grad[t:] -= h[:m]
    return grad

def nesterov_topk(h_init, n, t0, deadline, top_k=3, max_iters=600000):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    center = n - 1
    a0 = 0.25
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it == 0:
            beta = 0.0
        else:
            beta = it / (it + 1.8)
        h_y = np.clip(h_curr + beta * (h_curr - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        max_c = corr[top_idx].max()
        wts = np.exp(6.0 * (corr[top_idx] - max_c))
        wts /= wts.sum()
        g = np.zeros(n)
        for i in range(top_k):
            g += wts[i] * corr_grad(h_y, n, top_idx[i] - center)
        g *= dx
        alpha = a0 / np.sqrt(float(it + 1))
        h_new = np.clip(h_y - alpha * g, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        h_prev = h_curr
        h_curr = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 10000 == 0:
            c5a = compute_c5(h_avg)
            if c5a < best_c5:
                best_c5 = c5a
                best_h = h_avg.copy()
            c5c = compute_c5(h_curr)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_curr.copy()
    return best_h, best_c5

_N = 0
_T = 1.0

def _sm_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    sh = sh - sh.max()
    return (mc + np.log(np.exp(sh).sum()) / _T) * (2.0 / _N)

def _sm_grad(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    sh = sh - sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    center = _N - 1
    dx = 2.0 / _N
    grad = np.zeros(_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-13:
            continue
        grad += sw[oi] * corr_grad(x, _N, oi - center)
    return grad * dx

def _c_fun(x):
    return x.sum() - _N / 2.0

def _c_jac(x):
    return np.ones(_N)

def slsqp_phase(h, n, T, maxiter, ftol):
    global _N, _T
    _N = n
    _T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP', jac=_sm_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
