            p = -o
            g[:n-p] = (1.0 - h[p:]) * dx
            g[p:] -= h[:n-p] * dx
        h = project_box_sum(h - lr * g, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.zeros(n)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B progressive cooling
    for T in [50, 200, 600, 2000, 6000]:
        if elapsed() > budget_s - 350:
            break
        h_new = lbfgsb_run(h_best, target, T, 5000, 1.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Adam at high T (different dynamics than L-BFGS)
    for T in [2000, 8000]:
        if elapsed() > budget_s - 250:
            break
        h_new = adam_optimize(h_best, target, T, 8000, lr=0.01)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Chase the max (subgradient on argmax)
    if elapsed() < budget_s - 150:
        h_new = chase_max(h_best, target, steps=500, lr=0.003)
        h_new = lbfgsb_run(h_new, target, 3000, 3000, 1.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Multi-start perturbation
    n_perturb = 0
    while elapsed() < budget_s - 120:
        n_perturb += 1
        scale = max(0.02, 0.10 / (1.0 + 0.1 * n_perturb))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T = 1000 + int(rng.integers(0, 5000))
        h_new = lbfgsb_run(h_pert, target, T, 2000, 5.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            scale = max(0.02, 0.10 / (1.0 + 0.1 * 0.5 * n_perturb))
        if n_perturb > 60:
            break

    # Phase 5: Momentum fine-tune
    if elapsed() < budget_s - 80:
        h_new = momentum_optimize(h_best, target, 10000, 5000, lr=0.002, mu=0.95)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try resolution 350 and 210
    for n_try in [350, 210, 420]:
        if elapsed() > budget_s - 40:
            break
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        best_local = h_try
        best_try = c5_try
        for T in [2000, 8000]:
            if elapsed() > budget_s - 10:
                break
            h_t = lbfgsb_run(best_local, t_try, T, 3000, 5.0)
            c5_t = compute_c5(h_t)
            if c5_t < best_try:
                best_try = c5_t
                best_local = h_t
        if best_try < c5_best:
            c5_best = best_try
            h_best = best_local
            n = n_try
            target = t_try

    # Final polish
    if elapsed() < budget_s - 5:
        h_new = lbfgsb_run(h_best, target, 20000, 5000, 1.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new = chase_max(h_best, target, 300, 0.002)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
