import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2

def _sm_grad(x, T, target, pen):
    n = len(x)
    ones = np.ones(n)
    return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones

                       bounds=bounds, args=(T, target, pen),
                       options={'maxiter': maxiter, 'gtol': 1e-14})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

    base = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    center = n - 1
    pn = 0.0

        idx = center + lag
        if 0 <= idx < len(corr):
            pn += lam * (corr[idx] * dx) ** 2
    return base + pn + pen * (x.sum() - target) ** 2

def _act_grad(x, T, target, pen, lags, lam, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)

    return grad

def coord_descent(h, target, n_sweeps):
    n = len(h)
    h = h.copy()

            best_c = compute_c5(h)

                    best_val = h
                else:

                    h = project_box_sum(h, target)
            h[i] = best_val if best_c < compute_c5(h) else orig
