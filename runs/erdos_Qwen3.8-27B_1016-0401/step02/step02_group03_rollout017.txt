def project_feasible(y, n):
    target = n / 2.0

            break

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[shift:] += (1.0 - h[:n - shift])
            grad[:n - shift] -= h[shift:]
    else:

        if s < n:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
