def compute_c5_all(h):

    corr = np.correlate(h, 1 - h, mode="full")

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()

    sw = e / se
    return obj, sw

    grad = np.zeros(n)

        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T, k_top=0):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    if k_top > 0 and k_top < len(corr):
        top_idx = np.argsort(corr)[-k_top:]
        top_vals = corr[top_idx]
        mc = top_vals.max()
        sh = T * (top_vals - mc)
        sh -= sh.max()
        e = np.exp(sh)
        se = e.sum()
        obj = (mc + np.log(se) / T) * (2.0 / n)
        sw = np.zeros(2 * n - 1)
        sw[top_idx] = e / se
    else:
        mc = corr.max()
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        se = e.sum()
        obj = (mc + np.log(se) / T) * (2.0 / n)
        sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def pg_solve(h0, target, T, n_iter, lr0, t0, budget, k_top=0, antisym=False):

def nudge_search(h0, target, t0, budget, max_tries, seed_v=99):

    best_c5 = compute_c5(h)
    rng = np.random.default_rng(seed_v)
    tries = 0
    while tries < max_tries and time.time() - t0 < budget - 3:
        boundary = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(boundary) > 0:
            i = int(boundary[rng.integers(len(boundary))])
            delta = rng.uniform(0.02, 0.15)
            if h[i] < 0.5:
                delta = abs(delta)

                delta = -abs(delta)

            h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)

        else:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:

            h = h_new
        tries += 1
