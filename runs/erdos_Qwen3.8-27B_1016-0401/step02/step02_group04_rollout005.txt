                    c5_best = c5
                    h_best = h_new

    # Phase 3: Multi-scale coarse-to-fine
    for n_c in [70, 140]:
        if n_c >= n or elapsed() > budget_s - 400:
            continue
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_c, endpoint=False)
        h_c = np.interp(x_new, x_old, h_best)
        h_c = project_box_sum(h_c, n_c / 2.0)
        for T in [40, 120, 300]:
            if elapsed() > budget_s - 350:
                break
            h_c2 = lbfgs(h_c, n_c, n_c / 2.0, T, 10.0, 1500)
            h_c2 = project_box_sum(h_c2, n_c / 2.0)
            h_up = np.interp(x_old, x_new, h_c2)
            h_up = project_box_sum(h_up, target)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best:
                c5_best = c5_up
                h_best = h_up

    # Phase 4: FISTA with top-k focus
    for T in [60, 150, 350]:
        if elapsed() > budget_s - 200:
            break
        step = min(0.25, 1.8 / (T + 2.0))
        h_new = fista(h_best, target, T, 4000, step, topk=True, k=5, alpha=5.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Structured perturbations (block flips + blends)
    n_iter = 0
    while elapsed() < budget_s - 100:
        n_iter += 1
        mod = n_iter % 4
        if mod == 0:
            # Block flip perturbation
            h_pert = h_best.copy()
            n_flips = rng.integers(2, 6)
            for _ in range(n_flips):
                start = int(rng.integers(0, n - 4))
                ln = int(rng.integers(2, 7))
                h_pert[start:start+ln] = 1.0 - h_pert[start:start+ln]
            h_pert = project_box_sum(h_pert, target)
        elif mod == 1:
            # Small Gaussian
            sc = max(0.02, 0.10 / (1.0 + 0.04 * n_iter))
            h_pert = h_best + rng.normal(0, sc, n)
            h_pert = project_box_sum(h_pert, target)
        elif mod == 2:
            # Blend with structured pattern
            h_pat = make_pattern(n, int(rng.integers(0, 5)), rng)
            h_pert = 0.25 * h_pat + 0.75 * h_best
            h_pert = project_box_sum(h_pert, target)
        else:
            # Medium Gaussian with random T
            h_pert = h_best + rng.normal(0, 0.05, n)
            h_pert = project_box_sum(h_pert, target)

        T = 50 + int(rng.integers(0, 300))
        use_k = (n_iter % 3 != 0)
        h_new = lbfgs(h_pert, n, target, T, 8.0, 800, topk=use_k, k=5, alpha=4.0)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            n_iter = max(0, n_iter - 20)  # reset perturbation counter

    # Phase 6: Fine FISTA polish
    while elapsed() < budget_s - 30:
        h_pert = h_best + rng.normal(0, 0.015, n)
        h_pert = project_box_sum(h_pert, target)
        T = 250 + int(rng.integers(0, 300))
        h_new = fista(h_pert, target, T, 2000, 0.12, topk=True, k=4, alpha=6.0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final L-BFGS-B polish
    if elapsed() < budget_s - 3:
        h_new = lbfgs(h_best, n, target, 25, 1.0, 1500)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
