def active_set_step(h, n, target, M, eps, bounds):
    """Linearized active-set step"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    active = np.argsort(corr)[-M:]
    center = n - 1
    
    # Build gradient matrix for active lags
    # ∂corr[o]/∂h[i] for each active lag o
    G = np.zeros((M, n))  # G[m, i] = ∂corr[active[m]]/∂h[i]
    
    for m, idx in enumerate(active):
        o = int(idx) - center
        if o == 0:
            G[m] = 1.0 - 2.0 * h
        elif o > 0:
            G[m, o:] += (1.0 - h[:n - o])
            G[m, :n - o] -= h[o:]
        else:
            p = -o
            G[m, :n - p] += (1.0 - h[p:])
            G[m, p:] -= h[:n - p]
    
    # Use scipy to solve the LP/QP
    # min t s.t. G δ + corr[active] ≤ t, ||δ||_2 ≤ eps, sum(δ)=0, 0≤h_i+δ_i≤1
    # Variable: [δ_0, ..., δ_{n-1}, t], dimension n+1
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # G δ + corr_active ≤ t  →  G δ - t ≤ -corr_active
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -corr[active]
    
    # sum(δ) = 0 → handle as equality
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # 0 ≤ h_i + δ_i ≤ 1 → -h_i ≤ δ_i ≤ 1 - h_i
    bounds_lp = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(corr[active].min(), None)]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_step(h, n, target, bounds, T, maxiter):
    def obj(x):
        return lse_obj(compute_corr(x), T, n)
    def jac(x):
        return lse_grad(compute_corr(x), x, T, n)
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except:
        return h

def build_grad_matrix(corr_active_idx, h, n):
    """Build gradient matrix for specified correlation indices"""
    center = n - 1
    G = np.zeros((len(corr_active_idx), n))
    for m, idx in enumerate(corr_active_idx):
        o = int(idx) - center
        if o == 0:
            G[m] = 1.0 - 2.0 * h
        elif o > 0:
            G[m, o:] += (1.0 - h[:n - o])
            G[m, :n - o] -= h[o:]
        else:
            p = -o
            G[m, :n - p] += (1.0 - h[p:])
            G[m, p:] -= h[:n - p]
    return G

def lp_step(h, n, target, M, step_scale):
    """LP-based active set step"""
    corr = compute_corr(h)
    active = np.argsort(corr)[-M:]
    G = build_grad_matrix(active, h, n)
    
    # LP: min t s.t. G@delta - t <= -corr[active], sum(delta)=0, -h<=delta<=1-h
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -corr[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bnds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(-np.inf, np.inf)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bnds, method='highs')
    if res.status != 0:
        return h
    delta = res.x[:n]
    # Line search
    best_h = h
    best_c5 = compute_c5(h)
    for alpha in [step_scale, step_scale*0.5, step_scale*0.25, step_scale*0.1]:
        h_try = project_feasible(h + alpha * delta, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            break
    return best_h

def coord_descent(h, n, target, n_iters, rng):
    c5_cur = compute_c5(h)
    for it in range(n_iters):
        i = rng.integers(n)
        j = rng.integers(n)

            h_trial[j] += delta
            c5_t = compute_c5(h_trial)
            if c5_t < c5_cur:
                h = h_trial
                c5_cur = c5_t
                break
    return h, c5_cur

    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad_func(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

_slq = {}
def _s_obj(x):
    return lse_obj_func(compute_corr(x), _slq['T'], _slq['n'])
def _s_jac(x):
    return lse_grad_func(compute_corr(x), x, _slq['T'], _slq['n'])
def _s_cfun(x):
    return x.sum() - _slq['target']
def _s_cjac(x):
    return np.ones(_slq['n'])

def slsqp_step(h, n, target, bounds, T, maxiter):
    _slq['T'] = T
    _slq['n'] = n
    _slq['target'] = target
    cons = [{'type': 'eq', 'fun': _s_cfun, 'jac': _s_cjac}]
    try:
        res = minimize(_s_obj, h, method='SLSQP', jac=_s_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgrad_polish(h, n, target, n_steps, K):
    c5_local = compute_c5(h)
    dx = 2.0 / n
    for it in range(n_steps):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-K:]
        G = build_grad_matrix(top_idx, h, n)
        g = G.sum(axis=0) / K
        for lr in [0.04 / (1 + it * 0.03), 0.015, 0.005]:
            h_cand = project_feasible(h - lr * g, target)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_local:
                h = h_cand
                c5_local = c5_cand
                break
    return h, c5_local

def coord_descent(h, n, target, n_iters, rng):
    c5_cur = compute_c5(h)
    for it in range(n_iters):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or h[i] < 0.01 or h[j] > 0.99:
            continue
        for delta in [0.02, 0.05, 0.1, 0.15]:
            if h[i] < delta or h[j] > 1.0 - delta:
                continue
            h_trial = h.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            c5_t = compute_c5(h_trial)
            if c5_t < c5_cur:
                h = h_trial
                c5_cur = c5_t
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(280) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Quick SLSQP to get close
    for T, mit in [(30, 100), (10, 150), (5, 200), (2, 300)]:
        if elapsed() > budget_s - 500:
            break
        h = slsqp_step(h, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 2: LP active-set iterations (main optimization)
    h = h_best.copy()
    for phase in range(8):
        if elapsed() > budget_s - 300:
            break
        M = [5, 8, 10, 15, 20, 30, 40, 50][phase]
        step = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15][phase]
        n_steps = [80, 60, 60, 50, 40, 40, 30, 30][phase]
        for _ in range(n_steps):
            if elapsed() > budget_s - 280:
                break
            h_new = lp_step(h, n, target, M, step)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()
            h = h_new

    # Phase 3: SLSQP at very low T
    if elapsed() < budget_s - 200:
        for T in [1.5, 1.0, 0.7]:
            if elapsed() > budget_s - 120:
                break
            h = slsqp_step(h_best, n, target, bounds, T, 2500)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 4: More LP from best
    if elapsed() < budget_s - 120:
        h = h_best.copy()
        for M in [10, 20, 30, 50, 80]:
            if elapsed() > budget_s - 80:
                break
            for _ in range(30):
                h_new = lp_step(h, n, target, M, 0.2)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new.copy()
                h = h_new

    # Phase 5: Subgradient polish
    if elapsed() < budget_s - 80:
        h = h_best.copy()
        for K in [5, 3, 2]:
            if elapsed() > budget_s - 50:
                break
            h, c5_local = subgrad_polish(h, n, target, 150, K)
            if c5_local < c5_best:
                c5_best = c5_local
                h_best = h.copy()

    # Phase 6: Coordinate descent
    if elapsed() < budget_s - 40:
        h_cd, c5_cd = coord_descent(h_best, n, target, 5000, rng)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd.copy()

    # Phase 7: Random restarts
    if elapsed() < budget_s - 30:
        for trial in range(20):
            if elapsed() > budget_s - 15:
                break
            noise_scale = 0.03 + 0.02 * (trial % 3)
            h_r = project_feasible(h_best + rng.normal(0, noise_scale, n), target)
            h_r = lp_step(h_r, n, target, 15, 0.5)
            h_r = lp_step(h_r, n, target, 30, 0.3)
            c5_r = compute_c5(h_r)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r.copy()

    # Phase 8: Lower resolution
    if elapsed() < budget_s - 50 and n >= 200:
        n_lo = n // 2
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        bounds_lo = [(0.0, 1.0)] * n_lo
        h_lo = slsqp_step(h_lo, n_lo, target_lo, bounds_lo, 5, 300)
        for M in [10, 20, 40]:
            if elapsed() > budget_s - 15:
                break
            for _ in range(40):
                h_lo = lp_step(h_lo, n_lo, target_lo, M, 0.3)
                c5_lo = compute_c5(h_lo)
                if c5_lo < c5_best:
                    c5_best = c5_lo
                    h_best = h_lo.copy()

    # Phase 9: Final LP polish
    if elapsed() < budget_s - 5:
        h = h_best.copy()
        for _ in range(20):
            if elapsed() > budget_s - 2:
                break
            h = lp_step(h, len(h), len(h) / 2.0, 30, 0.15)
            c5_f = compute_c5(h)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h.copy()

    h_best = project_feasible(np.clip(h_best, 0.0, 1.0), len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
