def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

Actually, for a specific lag o:
- If o=0: f_0 includes h[j]*(1-h[j])*dx = (h[j]-h[j]²)*dx, which is quadratic in h[j]
- If o>0: f_o includes h[j]*(1-h[j+o])*dx [if j<n-o] and h[j-o]*(1-h[j])*dx [if j>=o]
  = (h[j] - h[j]*h[j+o] + h[j-o] - h[j]*h[j-o])*dx (if both terms valid)
  = (h[j] - h[j]*(h[j+o] + h[j-o]) + h[j-o])*dx
  = (h[j]*(1 - h[j+o] - h[j-o]) + h[j-o])*dx
  This is quadratic in h[j] (the term -h[j]*h[j-o] is linear in h[j] if h[j-o] is fixed, but h[j]*h[j+o] is also linear)
  
Wait: h[j]*(1-h[j-o]) is linear in h[j]. And h[j-o]*(1-h[j]) is also linear in h[j] (it's h[j-o] - h[j]*h[j-o]).
So f_o is linear in h[j] for o≠0. For o=0, it's h[j]*(1-h[j])*dx which is quadratic.

So the objective max_k f_k(h with h[j]=x) is the max of:
- A quadratic (from lag 0)
- Several linear functions (from other lags)

The minimum of this max over x ∈ [0,1] can be found by:
- If the quadratic is the max: minimize the quadratic (vertex)
- If a linear is the max: the best x is at a boundary or intersection

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad

def project_box_sum(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_backtrack(h0, target, T, max_iter):
    n = len(h0)
    h = h0.copy()
    prev = h.copy()
    t_k = 1.0
    base_step = 2.0 / (T + 10.0)
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - prev)
        g = soft_max_grad(y, T)
        step = base_step
        for _ls in range(40):
            cand = project_box_sum(y - step * g, target)
            fy = soft_max_value(y, T)
            fc = soft_max_value(cand, T)
            if fc <= fy - 0.5 * step * np.dot(g, cand - y):
                break
            step *= 0.5
        prev = h
        h = cand
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        if np.linalg.norm(h - prev) < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
    return h

def worst_lag_grad(h):
    n = len(h)

def focused_lbfgs(h0, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return soft_max_value(x, T)
