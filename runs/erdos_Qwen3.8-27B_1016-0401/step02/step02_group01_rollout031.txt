import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary(h, target):
    """Round h to binary while maintaining sum = target."""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    s = h_bin.sum()
    # Need to adjust: if s > target, change some 1s to 0s; if s < target, change some 0s to 1s
    if s > target:
        ones = np.where(h_bin == 1)[0]
        # Sort by h value (lowest h among ones most likely to be 0)
        order = ones[np.argsort(h[ones])]
        for idx in order:
            if s == target:
                break
            h_bin[idx] = 0
            s -= 1
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for idx in order:
            if s == target:
                break
            h_bin[idx] = 1
            s += 1
    return h_bin

def binary_local_search(h_bin, target, max_rounds=10, c5_threshold=0, t0=None, budget=1000, rng=None):

    n = len(h_bin)
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones = set(np.where(h == 1)[0].tolist())

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

    cold = np.where(h == 0)[0].copy()

    temp_end = 0.001

    for rnd in range(max_rounds):
        if time.time() - t0 > budget:
            break
        temp = temp_start * (temp_end / temp_start) ** (rnd / max(min(max_rounds, 1) - 1, 1))
        improved = False
        for _ in range(n_swaps):
            i = hot[rng.integers(len(hot))]
            j = cold[rng.integers(len(cold))]

                cold = np.where(h == 0)[0]

                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = 1, 0
        if not improved:
            break
    return best_h, best_c5

    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()

    sw = e / se

    c = n - 1
    for j in range(2 * n - 1):

        if w < 1e-15:

        k = j - c
        if k == 0:

        elif k > 0:

            grad[:n - k] -= w * h[k:]
        else:

            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]

            stuck += 1
            if stuck > 50000:
                break
        n_done += 1
    return best_h, best
