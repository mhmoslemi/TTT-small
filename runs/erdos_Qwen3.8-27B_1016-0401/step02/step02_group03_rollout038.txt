import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

# Global state for SLSQP callbacks
_NS = {}

def _obj(x):
    n = _NS['n']
    T = _NS['T']
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _jac(x):
    n = _NS['n']
    T = _NS['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    w = e / e.sum()
    # Gradient: dx * sum_m w[m] * grad_f_m
    # grad_f_m[j] = (1-h[j+k_m])*I(0<=j+k_m<n) - h[j-k_m]*I(0<=j-k_m<n)
    # where k_m = m - (n-1)
    grad = np.zeros(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - center
        if k >= 0:
            grad[:n - k] += wm * (1.0 - x[k:n])
            grad[k:] -= wm * x[:n - k]
        else:
            s = -k
            grad[s:] += wm * (1.0 - x[:n - s])
            grad[:n - s] -= wm * x[s:]
    return grad * dx

def _cons_fun(x):
    return x.sum() - _NS['target']

def _cons_jac(x):
    return np.ones(_NS['n'])

def slsqp_opt(h, n, T, maxiter, ftol):
    _NS['n'] = n
    _NS['T'] = float(T)
    _NS['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def fourier_perturb(h, n, rng):
    """Perturb with low-frequency Fourier modes"""
    freqs = rng.choice(1, size=min(3, n//2))  # low frequencies
    h_p = h.copy()
    for f in freqs:
        amp = rng.uniform(0.005, 0.03)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h_p += amp * np.sin(2 * np.pi * f * x / n + phase)
    return h_p

def accelerated_gradient(h_init, n, T, t0, time_limit, max_iters=200000):
    """FISTA-like accelerated gradient with projection"""
    dx = 2.0 / n
    h = h_init.copy()
    h_prev = h.copy()
    h_avg = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    # Estimate L (Lipschitz constant)
    L = 4.0 / dx * 0.5  # conservative: max eigenvalue of Hessian of f_k is ~2, times dx, times 2 for safety
    # Actually L of the smooth max is bounded by 2 (from Hessian of individual f_k)
    # The log-sum-exp adds positive definiteness, so L ≈ 2 + T * var(T*f)
    # Let's use L = 2.0 as a starting point
    L = 2.0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        t_k = (1 + np.sqrt(1 + 4 * (it if it > 0 else 0) ** 2)) / 2
        beta = (t_k - 1) / (t_k + 2) if it > 0 else 0.0
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient step
        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        w = e / e.sum()
        
        grad = np.zeros(n)
        center = n - 1
        for m in range(2 * n - 1):
            wm = w[m]
            if wm < 1e-14:
                continue
            k = m - center
            if k >= 0:
                grad[:n - k] += wm * (1.0 - y[k:n])
                grad[k:] -= wm * y[:n - k]
            else:
                s = -k
                grad[s:] += wm * (1.0 - y[:n - s])
                grad[:n - s] -= wm * y[s:]
        grad *= dx
        
        # Line search for L
        step = 1.0 / L
        h_new = y - step * grad
        h_new = project_feasible(h_new, n)
        
        # Update average
        h_avg += (h_new - h_avg) / (it + 1)
        h_prev = h
        h = h_new
        
        # Check improvement periodically
        if (it + 1) % 5000 == 0:
            c5_check = compute_c5(project_feasible(h_avg, n))
            if c5_check < c5_best:
                c5_best = c5_check
                h_best = project_feasible(h_avg, n).copy()
    
    c5_final = compute_c5(project_feasible(h_avg, n))
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = project_feasible(h_avg, n).copy()
    return h_best, c5_best

    return (h_best, c5_final, n)

The issue is that for any periodic pattern, there will be shifts where the autocorrelation is 0 or very small.

The key to achieving C₅ < 1 is to have a pattern that's aperiodic and has "flat" autocorrelation. This is exactly what Barker sequences and related constructions do!

For a binary sequence of length N, the ideal autocorrelation would be:
R(0) = N, R(k) = -1 for k≠0 (Barker-like), or R(k) = 0 for k≠0 (impossible for N>3).

In our problem, we want MINIMUM autocorrelation to be as large as possible (not zero, not negative). So we want all R(k) to be close to each other.

(R(0) + sum of others) / (number of shifts) ≈ R_avg

_G = {}  # Global state for SLSQP

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _slsqp_obj(x):
    T = _G['T']
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / _G['n'])

def _slsqp_jac(x):
    n = _G['n']
    T = _G['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - center
        if k >= 0:
            grad[:n-k] += wm * (1.0 - x[k:n])
            grad[k:] -= wm * x[:n-k]
        else:
            s = -k
            grad[s:] += wm * (1.0 - x[:n-s])
            grad[:n-s] -= wm * x[s:]
    return grad * dx

    return np.ones(_G['n'])
