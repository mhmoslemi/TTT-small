- j in [o, n-1-o]: both terms, = 1-h[j+o] - h[j-o]
- j in [n-o, n-1]: only second term, = -h[j-o]

For o < 0 (let p = -o > 0):
corr[k] = sum_{i=p}^{n-1} h[i] * (1 - h[i-p])

∂corr[k]/∂h[j]:
- If j is in [p, n-1]: h[j] appears as h[i] with i=j, contributing (1-h[j-p])
- If j+p is in [p, n-1], i.e., j is in [0, n-1-p]: h[j] appears as h[i-p] with i=j+p, contributing -h[j+p]

So for o = -p:
- j in [0, p-1]: only second term, = -h[j+p]
- j in [p, n-1-p]: both terms, = 1-h[j-p] - h[j+p]
- j in [n-p, n-1]: only first term, = 1-h[j-p]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def sm_objective(x, T, pen, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = (corr - mc) / T
    se = np.sum(np.exp(shifted))
    val = (mc + T * np.log(se)) * dx
    if pen > 0:
        val += pen * (x.sum() - target) ** 2
    return val


def sm_gradient(x, T, pen, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = (corr - mc) / T
    e = np.exp(shifted)
    sw = e / np.sum(e)
    g = np.zeros(n)
    c = n - 1
    for o in range(-(n - 1), n):
        w = sw[c + o]
        if w < 1e-12:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n - o])
            g[:n - o] -= w * x[o:]
        else:
            p2 = -o
            g[:n - p2] += w * (1.0 - x[p2:])
            g[p2:] -= w * x[:n - p2]
    g *= dx
    if pen > 0:
        g += 2.0 * pen * (x.sum() - target)
    return g


def eq_fun(x, target):
    return x.sum() - target


def eq_jac(x, target):
    return np.ones(len(x))


def snap_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def local_search(h, target, t0, deadline, n_iter=5000):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5_best = float(np.max(corr) * dx)
    h_cur = h.copy()
    stall = 0
    delta = 0.05
    rng_local = np.random.default_rng(12345)
    
    for it in range(n_iter):
        if time.time() - t0 > deadline:
            break
        if stall > 200:
            delta *= 0.7
            stall = 0
            if delta < 0.001:
                break
        
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        
        if p == 0:
            vals = h_cur * (1 - h_cur)
            idx = int(np.argmax(vals))
            h_try = h_cur.copy()
            h_try[idx] += -delta if h_cur[idx] > 0.5 else delta
        else:
            if s > 0:
                contrib = h_cur[p:] * (1 - h_cur[:n - p])
            else:
                contrib = h_cur[:n - p] * (1 - h_cur[p:])
            idx = int(np.argmax(contrib))
            if s > 0:

            else:
                j1, j2 = idx, idx + p
            h_try = h_cur.copy()
            h_try[j1] -= delta
            h_try[j2] += delta

            c5_best = c5_try
            h_cur = h_try
            corr = corr_try
