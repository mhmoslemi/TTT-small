import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0
_N = 0
_LAG = 0
_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(600):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff -= add
            if abs(diff) < 1e-13:
                break
    return h

def obj_sm(x):
    n = _N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = _T * (corr - mc)
    shifted = np.clip(shifted, -600, 600)
    return (mc + np.log(np.sum(np.exp(shifted))) / _T) * (2.0 / n)

def grad_sm(x):
    n = _N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = _T * (corr - mc)
    shifted = np.clip(shifted, -600, 600)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o_idx in np.where(sw > 1e-15)[0]:
        w = sw[o_idx]
        o = o_idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] -= w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n-p]
    return grad * dx

def obj_single(x):
    n = _N
    return np.correlate(x, 1.0 - x, mode="full")[_LAG] * (2.0 / n)

def grad_single(x):
    n = _N
    o = _LAG - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] -= x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] -= x[:n-p]
    return g * dx

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(_N)

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET, _N, _LAG, _T
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    _N = n
    target = n / 2.0
    _TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    # Phase 1: Smooth max at multiple temperatures (45% budget)
    p1_end = t0 + budget_s * 0.45
    T_list = [60, 120, 240, 480, 960, 1920]
    for T in T_list:
        if time.time() > p1_end:
            break
        _T = T
        for restart in range(2):
            if time.time() > p1_end:
                break
            if restart == 0:
                x0 = h_best.copy()
            else:
                x0 = fix_constraint(h_best + rng.normal(0, 0.004, n), target)
            try:
                res = minimize(obj_sm, x0, method='SLSQP', jac=grad_sm,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 4000, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 2: Active-set cycling top 5 lags (25% budget)
    p2_end = t0 + budget_s * 0.70
    for outer in range(200):
        if time.time() > p2_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_k = min(5, 2*n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for lag in top_idx:
            if time.time() > p2_end:
                break
            _LAG = lag
            try:
                res = minimize(obj_single, h_best, method='SLSQP', jac=grad_single,
                              bounds=bounds, constraints=cons,

                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
                    improved = True
            except Exception:
                pass
        if not improved and outer > 80:
            sc = 0.003 * (0.92 ** ((outer - 80) // 20))
            h_pert = fix_constraint(h_best + rng.normal(0, sc, n), target)
            if compute_c5(h_pert) < c5_best:
                h_best = h_pert
                c5_best = compute_c5(h_best)

    # Phase 3: Multi-perturbation + re-optimization (20% budget)
    p3_end = t0 + budget_s * 0.90
    for attempt in range(40):
        if time.time() > p3_end:
            break
        sc = 0.004 * (0.85 ** (attempt // 15))
        h_pert = fix_constraint(h_best + rng.normal(0, sc, n), target)
        c5_pert = compute_c5(h_pert)
        if c5_pert >= c5_best:
            continue
        _T = 200
        try:
            res = minimize(obj_sm, h_pert, method='SLSQP', jac=grad_sm,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-17, 'gtol': 1e-14})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        _LAG = int(np.argmax(corr))
        try:
            res = minimize(obj_single, h_best, method='SLSQP', jac=grad_single,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-15})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Final tight polish (10% budget)
    for _ in range(40):
        if time.time() > deadline:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        sorted_lags = np.argsort(corr)[-3:]
        for lag in sorted_lags:
            if time.time() > deadline:
                break
            _LAG = lag
            try:
                res = minimize(obj_single, h_best, method='SLSQP', jac=grad_single,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 6000, 'ftol': 1e-19, 'gtol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
