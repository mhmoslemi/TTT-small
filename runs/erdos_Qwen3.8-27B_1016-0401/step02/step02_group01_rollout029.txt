import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def project(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        p = -k
        g[p:] += 1.0 - h[:n - p]
        g[:n - p] -= h[p:]
    return g

_G = {'T': 100.0, 'target': 0.0, 'n': 0}

def _fun(x):
    n, T = _G['n'], _G['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    return (mc + np.log(e.sum()) / T) * dx

def _grad(x):
    n, T = _G['n'], _G['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    sw = e / e.sum()
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w > 1e-14:
            g += w * grad_k(x, j - c)
    return g * dx

def _c_fun(x):
    return x.sum() - _G['target']

def _c_jac(x):
    return np.ones(_G['n'])

def scl_solve(h0, target, T, maxiter, deadline):
    _G['T'], _G['target'], _G['n'] = T, target, len(h0)
    try:
        res = minimize(_fun, h0, method='SLSQP', jac=_grad,
                       bounds=[(0.0, 1.0)] * _G['n'],
                       constraints=[{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 100:
            h = project(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def nest_pg(h0, target, T, n_iter, lr0, deadline):
    n, dx = len(h0), 2.0 / len(h0)
    h, hp = h0.copy(), h0.copy()
    bc5, bh = compute_c5(h), h.copy()
    bt = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        bk = (1 + np.sqrt(1 + 4 * bt * bt)) / 2
        y = project(h + (bt - 1) / bk * (h - hp), target)
        corr = np.correlate(y, 1 - y, mode="full")
        mc = corr.max()
        s = T * (corr - mc)
        s -= s.max()
        e = np.exp(s)
        sw = e / e.sum()
        g = np.zeros(n)
        c = n - 1
        for j in range(2 * n - 1):
            if sw[j] > 1e-14:
                g += sw[j] * grad_k(y, j - c)
        lr = lr0 / (1 + 0.005 * it)
        h, hp = project(y - lr * g * dx, target), h
        bt = bk
        if it % 400 == 399:
            c5 = compute_c5(h)
            if c5 < bc5:
                bc5, bh = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < bc5:
        bc5, bh = c5, h.copy()
    return bh, bc5
