def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    # Numerical stability
    shifted = T * (corr - mc)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se
    obj = (mc + np.log(se) / T) * dx
    
    # Gradient computation (vectorized loop over lags)
    center = n - 1
    g = np.zeros(n)
    for idx in range(len(sw)):
        w = sw[idx] * dx
        if w < 1e-14:
            continue
        o = idx - center  # lag
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return obj, g

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def exact_max_grad(h):
    """Compute subgradient of max(corr*dx) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)

            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    g /= len(active)

def compute_c5(h):
    n = len(h)

def correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_at_lags(h, lags, weights):
    """Compute gradient contribution from specified lags with weights."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    for idx, w in zip(lags, weights):
        o = idx - center
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
