            f_sub = np.max(corr[None, :] + d_sub[:, None] * delta_s[None, :], axis=1)
            best_d = d_sub[int(np.argmin(f_sub))]

        new_c5 = float(np.max(corr + best_d * delta_s) * dx)
        accept = new_c5 < c5 - 1e-15
        if not accept and accept_worse and rng.random() < 0.01:
            # Annealing: accept small worse moves
            if new_c5 < c5 + 0.001:
                accept = True

        if accept and abs(best_d) > 1e-14:
            h[a] += best_d
            h[b] -= best_d
            corr += best_d * delta_s
            c5 = new_c5
            drift += best_d  # track drift (should be ~0)

    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Pair coordinate descent (main optimization)
    budget_p1 = budget_s * 0.55
    h1, c5_1 = pair_coord_desc(h_best, target, t0, t0 + budget_p1, rng, accept_worse=True)
    if c5_1 < c5_best - 1e-14:
        c5_best = c5_1
        h_best = h1

    # Phase 2: SLSQP polish
    def smo(x, T):
        return softmax_obj(x, T, target, n)

    def smo_jac(x, T):
        dx = 2.0 / n
        corr = np.correlate(x, 1 - x, mode="full")
        mc = np.max(corr)
        e = np.exp(T * (corr - mc))
        sw = e / np.sum(e)
        grad = np.zeros(n)
        center = n - 1
        for o in range(-(n - 1), n):
            w = sw[center + o]
            if w < 1e-13:
                continue
            if o == 0:
                grad += w * (1.0 - 2.0 * x)
            elif o > 0:
                grad[o:] += w * (1.0 - x[:n - o])
                grad[:n - o] += -w * x[o:]
            else:
                p = -o
                grad[:n - p] += w * (1.0 - x[p:])
                grad[p:] += -w * x[:n - p]
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda: np.ones(n)}]

    for T in [50, 20, 10, 5]:
        if time.time() - t0 > budget_s * 0.72:
            break
        try:
            res = minimize(smo, h_best, method='SLSQP', jac=smo_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-16})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-14:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: More pair coordinate descent
    budget_p3 = budget_s * 0.88
    h3, c5_3 = pair_coord_desc(h_best, target, t0, t0 + budget_p3, rng, accept_worse=True)
    if c5_3 < c5_best - 1e-14:
        c5_best = c5_3
        h_best = h3

    # Phase 4: Final SLSQP + pair descent
    for T in [3, 1.5]:
        if time.time() - t0 > budget_s - 80:
            break
        try:
            res = minimize(smo, h_best, method='SLSQP', jac=smo_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-17})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-14:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    if time.time() - t0 < budget_s - 20:
        h4, c5_4 = pair_coord_desc(h_best, target, t0, t0 + budget_s - 5, rng, accept_worse=False)
        if c5_4 < c5_best - 1e-14:
            c5_best = c5_4
            h_best = h4

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
