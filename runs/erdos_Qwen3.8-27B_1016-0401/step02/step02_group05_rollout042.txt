    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    h_curr = h.copy()
    cur_c5 = best_c5
    T0, T1 = 0.8, 0.002
    cool = (T1/T0)**(1.0/max(1, n_iter-1))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * cool**it
        h_try = h_curr.copy()
        mt = rng.integers(3)
        d = 0.12
        if mt == 0:
            idx = int(rng.integers(n))
            h_try[idx] += rng.normal(0, d)
        elif mt == 1:
            i1, i2 = int(rng.integers(n)), int(rng.integers(n))
            a = rng.normal(0, d)
            h_try[i1] -= a
            h_try[i2] += a
        else:
            bl = int(rng.integers(2, min(7, n//5)))
            st = int(rng.integers(0, max(1, n-bl)))
            h_try[st:st+bl] = 1.0 - h_curr[st:st+bl]
        h_try = fix_constraint(h_try, target)
        ct = np.correlate(h_try, 1 - h_try, mode="full")
        c5t = float(np.max(ct) * dx)
        dc = c5t - cur_c5
        if dc < 0 or rng.random() < np.exp(-dc/(max(T,1e-12)*dx)):
            h_curr = h_try
            cur_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
    return h_curr, best_c5

def slsqp_polish(h, target, T, maxiter, t0, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type':'eq', 'fun': partial(sum_eq_fun, target=target),
             'jac': partial(sum_eq_jac, target=target)}]
    try:
        res = minimize(soft_max_obj, h, method='SLSQP', jac=soft_max_jac,
                      args=(T,), bounds=bounds, constraints=cons,
                      options={'maxiter':maxiter, 'ftol':1e-15})
        if res.fun < 80 and time.time() < deadline:
            return fix_constraint(res.x.copy(), target)
    except Exception:
        pass
    return h

def downsample_target(h, n_new):
    n = len(h)
    if n_new >= n:
        return h
    idx = np.linspace(0, n-1, n_new).astype(int)
    hd = h[idx].copy()
    t = n_new / 2.0
    return fix_constraint(hd, t)

def upsample_target(h, n_new):
    n = len(h)
    if n_new <= n:
        return h
    idx_old = np.linspace(0, n_new-1, n).astype(int)
    hu = np.zeros(n_new)
    hu[idx_old] = h
    for i in range(n_new-1):
        oi = int(np.floor(i*(n-1)/(n_new-1)))
        on = int(np.ceil(i*(n-1)/(n_new-1)))
        if oi != on:
            hu[i] = 0.5*(h[oi]+h[on])
    t = n_new / 2.0
    return fix_constraint(hu, t)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP medium temp
    if time.time() < deadline - 350:
        h_best = slsqp_polish(h_best, target, 25, 2500, t0, deadline - 300)
        c5_best = compute_c5(h_best)

    # Phase 2: Aggressive multi-move local search
    if time.time() < deadline - 250:
        h_ls, c5_ls = multi_move_local(h_best, target, 6000, t0, deadline - 230, rng, delta_start=0.10)
        if c5_ls < c5_best:
            c5_best, h_best = c5_ls, h_ls

    # Phase 3: Simulated annealing
    if time.time() < deadline - 150:
        h_sa, c5_sa = sa_search(h_best, target, 4000, t0, deadline - 130, rng)
        if c5_sa < c5_best:
            c5_best, h_best = c5_sa, h_sa

    # Phase 4: SLSQP low temp polish
    if time.time() < deadline - 80:
        h_best = slsqp_polish(h_best, target, 6, 3000, t0, deadline - 50)
        c5_best = min(c5_best, compute_c5(h_best))

    # Phase 5: Another round of local search with smaller delta
    if time.time() < deadline - 30:
        h_ls2, c5_ls2 = multi_move_local(h_best, target, 4000, t0, deadline - 15, rng, delta_start=0.05)
        if c5_ls2 < c5_best:
            c5_best, h_best = c5_ls2, h_ls2

    # Phase 6: Try downsample to n//2, optimize, upsample back
    if time.time() < deadline - 10 and n > 80:
        n2 = n // 2
        h_d = downsample_target(h_best, n2)
        t2 = n2 / 2.0
        if time.time() < deadline - 8:
            h_d = slsqp_polish(h_d, t2, 12, 1500, t0, deadline - 5)
            h_d = fix_constraint(h_d, t2)
            c5_d = compute_c5(h_d)
            h_u = upsample_target(h_d, n)
            c5_u = compute_c5(h_u)
            # Local search on upsampled
            if time.time() < deadline - 3:
                h_u2, c5_u2 = multi_move_local(h_u, target, 1500, t0, deadline - 1, rng, delta_start=0.04)
                if c5_u2 < c5_u:
                    h_u, c5_u = h_u2, c5_u2
            if c5_u < c5_best:
                c5_best, h_best = c5_u, h_u

    # Phase 7: Final SLSQP
    if time.time() < deadline - 2:
        h_best = slsqp_polish(h_best, target, 3, 2000, t0, deadline - 1)
        c5_best = min(c5_best, compute_c5(h_best))

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
