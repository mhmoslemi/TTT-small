                                fft_iv=400, fft_keep=30)

    # Phase 3: High-T SLSQP fine-tune
    dl3 = t0 + budget_s - 150
    for T in [300, 600]:
        if time.time() > dl3: break
        h, c5 = slsqp(h_best, T, 300, dl3)
        if c5 < c5_best: c5_best, h_best = c5, h

    # Phase 4: Structured perturbations + re-optimize
    dl4 = t0 + budget_s - 40
    n_pert = 12
    for r in range(n_pert):
        if time.time() > dl4: break
        if r < 4:
            h_p = perturb_fourier(h_best, rng, 5 + r * 3, 0.4 + r * 0.15)
        elif r < 8:
            h_p = h_best.copy()
            st = rng.integers(0, n - 30)
            ln = rng.integers(8, 35)
            h_p[st:st + ln] = 1.0 - h_p[st:st + ln]
            h_p = project_constraint(h_p, target)
        else:
            h_p = project_constraint(h_best + rng.normal(0, 0.02 + r * 0.005, n), target)
        T_p = 80 + r * 30
        h_p, c5_p = pg_solve(h_p, target, T_p, T_p * 3, 1200, 0.04, t0, dl4, fft_iv=0)
        if c5_p < c5_best:
            c5_best, h_best = c5_p, h_p

    # Phase 5: Final concentrated refinement on best
    if time.time() < t0 + budget_s - 8:
        dl5 = t0 + budget_s - 3
        h_best, c5_f = pg_solve(h_best, target, 100, 1200, 800, 0.02, t0, dl5, fft_iv=0)
        if c5_f < c5_best: c5_best = c5_f
        if time.time() < dl5 - 5:
            h_fl, c5_fl = slsqp(h_best, 500, 200, dl5)
            if c5_fl < c5_best: c5_best = c5_fl; h_best = h_fl

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = project_constraint(h_best, target)
    return (h_best, c5_final, len(h_best))
