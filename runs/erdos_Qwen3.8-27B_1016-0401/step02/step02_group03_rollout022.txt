def do_lbfgs(h, n, T, maxiter=3000):
    global _g_n, _g_T
    _g_n = n
    _g_T = float(T)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h, jac=_sgrad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': 20000})
        return res.x
    except Exception:
        return h.copy()

def subgrad_ctc(h_init, n, t_end, max_iter=500000):
    """Subgradient on actual max with Chebyshev-type step."""
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_cur)
    c5_min = c5_best  # track min seen
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] += (1.0 - h_cur[:n - s])
            g[:n - s] -= h_cur[s:]
        else:
            ss = -s
            g[ss:] += (1.0 - h_cur[:n - ss])
            g[:n - ss] -= h_cur[ss:]
        g *= dx
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        alpha = 0.8 / np.sqrt(1.0 + 0.05 * it)
        h_new = h_cur - alpha * g
        h_new = project_feasible(h_new, n)
        h_cur = h_new
        if (it + 1) % 5000 == 0:
            c5 = compute_c5(h_cur)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_cur.copy()
    c5_final = compute_c5(h_cur)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h_cur.copy()
    return h_best, c5_best

            t[i] = 1.0 - t[i >> 1]

            t[i] = t[i >> 1]
