import numpy as np
from scipy.optimize import minimize
import time

_G = 0.0
_L = 0
_T = 1.0

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            h[idx] = np.minimum(h[idx] + d / len(idx), 1.0)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(h[idx] - (-d) / len(idx), 0.0)
    return h

def _g_lag(x):

    if o == 0:

    elif o > 0:

        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g * (2.0 / n)

def _f_lag(x):
