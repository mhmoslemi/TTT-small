import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            for i in np.argsort(-room):
                if d <= 0: break
                s = min(d, room[i])
                if s > 1e-14: h[i] += s; d -= s
        else:
            for i in np.argsort(h):
                if d >= 0: break
                s = min(-d, h[i])
                if s > 1e-14: h[i] -= s; d += s
    return h

def sm_obj(h, T):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(c); sh = T * (c - mc)
    return (mc + np.log(np.sum(np.exp(sh)))) * dx

def sm_jac(h, T):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(c); sh = T * (c - mc)
    e = np.exp(sh); sw = e / np.sum(e)
    g = np.zeros(n); ctr = n - 1
    for o in range(-(n-1), n):
        w = sw[ctr + o]
        if w < 1e-14: continue
        if o == 0: g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n-o]); g[:n-o] -= w * h[o:]
        else:
            p = -o; g[:n-p] += w * (1.0 - h[p:]); g[p:] -= w * h[:n-p]
    return g * dx

def _cfun(x, t): return x.sum() - t
def _cjac(x, t): return np.ones(len(x))

def slsqp(h, tgt, T, mx=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (tgt,)}]
    r = minimize(sm_obj, h, method='SLSQP', jac=sm_jac, args=(T,),
                 bounds=bnds, constraints=cons, options={'maxiter': mx, 'ftol': 1e-16})
    if r.fun < 100: return fix_sum(r.x.copy(), tgt)
    return h.copy()

def do_2opt_move(h, corr, n, k_star, delta):
    ctr = n - 1
    s = int(k_star - ctr); p = abs(s)
    if p == 0:
        vals = h * (1.0 - h)
        idx = int(np.argmax(vals))
        h_new = h.copy()
        h_new[idx] += 0.04 if h[idx] < 0.5 else -0.04
        return h_new
    if s > 0:
        contrib = h[p:] * (1.0 - h[:n-p])
    else:
        contrib = h[:n-p] * (1.0 - h[p:])
    # Try top-5 candidates
    top5 = np.argsort(-contrib)[:5]
    for idx in top5:
        h_try = h.copy()
        if s > 0:
            j1, j2 = idx + p, idx
        else:
            j1, j2 = idx, idx + p
        h_try[j1] -= delta
        h_try[j2] += delta
        h_try = np.clip(h_try, 0, 1)
        h_try = fix_sum(h_try, np.sum(h))
        c_new = float(np.max(get_corr(h_try)))
        if c_new < float(np.max(corr)) - 1e-15:
            return h_try
    return None

def do_4opt_move(h, corr, n, k1, k2, delta):
    ctr = n - 1
    s1 = int(k1 - ctr); p1 = abs(s1)
    s2 = int(k2 - ctr); p2 = abs(s2)
    if p1 == 0 or p2 == 0 or p1 == p2: return None
    if s1 > 0: c1 = h[p1:] * (1.0 - h[:n-p1])
    else: c1 = h[:n-p1] * (1.0 - h[p1:])
    if s2 > 0: c2 = h[p2:] * (1.0 - h[:n-p2])
    else: c2 = h[:n-p2] * (1.0 - h[p2:])
    i1 = int(np.argmax(c1)); i2 = int(np.argmax(c2))
    a1, a2 = (i1+p1, i1) if s1 > 0 else (i1, i1+p1)
    b1, b2 = (i2+p2, i2) if s2 > 0 else (i2, i2+p2)
    d4 = delta * 0.5
    h_try = h.copy()
    h_try[a1] -= d4; h_try[a2] += d4
    h_try[b1] -= d4; h_try[b2] += d4
    h_try = np.clip(h_try, 0, 1)
    h_try = fix_sum(h_try, np.sum(h))
    c_new = float(np.max(get_corr(h_try)))
    if c_new < float(np.max(corr)) - 1e-15:
        return h_try
    return None

def local_search(h, tgt, deadline, max_iter=500000):
    n = len(h)
    corr = get_corr(h)
    best = float(np.max(corr))
    cur = h.copy()
    stall = 0
    delta = 0.08
    kick_rng = np.random.default_rng(777)
    for it in range(max_iter):
        if time.time() > deadline: break
        if stall > 400:
            delta *= 0.85
            if delta < 0.003:
                idx_k = kick_rng.integers(0, n, size=6)
                cur[idx_k] += kick_rng.uniform(-0.04, 0.04, 6)
                cur = fix_sum(cur, tgt)
                corr = get_corr(cur)
                best = float(np.max(corr))
                stall = 0
                delta = 0.03
        top3 = np.argsort(-corr)[:3]
        k_star = top3[0]
        accepted = False
        # Try 2-opt with current delta
        h_try = do_2opt_move(cur, corr, n, k_star, delta)
        if h_try is None and delta > 0.01:
            h_try = do_2opt_move(cur, corr, n, k_star, delta * 0.5)
        if h_try is not None:
            c_new = float(np.max(get_corr(h_try)))
            if c_new < best - 1e-14:
                best = c_new; cur = h_try; corr = get_corr(cur); stall = 0; accepted = True
        # Try 4-opt
        if not accepted and stall % 4 == 2 and len(top3) >= 2:
            h_try4 = do_4opt_move(cur, corr, n, k_star, top3[1], delta)
            if h_try4 is not None:
                c4 = float(np.max(get_corr(h_try4)))
                if c4 < best - 1e-14:
                    best = c4; cur = h_try4; corr = get_corr(cur); stall = 0; accepted = True
        if not accepted:
            # Try 2-opt on second lag
            k2 = top3[1]
            if k2 != k_star:
                h_try2 = do_2opt_move(cur, corr, n, k2, delta * 0.7)
                if h_try2 is not None:
                    c2n = float(np.max(get_corr(h_try2)))
                    if c2n < best - 1e-14:
                        best = c2n; cur = h_try2; corr = get_corr(cur); stall = 0; accepted = True
        if not accepted:
            stall += 1
    return cur, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70); h_best[::2] = 1.0

    n = len(h_best)
    tgt = n / 2.0
    h_best = fix_sum(h_best, tgt)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP (20% time)
    ph1_end = t0 + 0.20 * budget_s
    for T in [120, 60, 30, 15, 8, 4, 2.5, 1.5, 1]:
        if time.time() > ph1_end: break
        h_new = slsqp(h_best, tgt, T, mx=500)
        c = compute_c5(h_new)
        if c < c5_best - 1e-13:
            c5_best = c; h_best = h_new

    # Phase 2: Intensive local search (55% time)
    ph2_end = t0 + 0.75 * budget_s
    h_best, c5_best = local_search(h_best, tgt, ph2_end, max_iter=800000)

    # Phase 3: SLSQP polish (13% time)
    ph3_end = t0 + 0.88 * budget_s
    for T in [6, 3, 2, 1.2, 0.8]:
        if time.time() > ph3_end: break
        h_new = slsqp(h_best, tgt, T, mx=800)
        c = compute_c5(h_new)
        if c < c5_best - 1e-13:
            c5_best = c; h_best = h_new

    # Phase 4: Final local search (7% time minus 10s buffer)
    ph4_end = t0 + budget_s - 10
    h_best, c5_best = local_search(h_best, tgt, ph4_end, max_iter=300000)

    # Verify and return
    c5_final = min(compute_c5(h_best), c5_best)
    c5_chk = compute_c5(h_best)
    c5_final = min(c5_final, c5_chk)
    return (h_best, c5_final, n)
