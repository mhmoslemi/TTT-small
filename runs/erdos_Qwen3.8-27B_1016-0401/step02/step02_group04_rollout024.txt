def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0*h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])

        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx

        if s > target:
            lo = mid
        else:
            hi = mid

    n = len(h0)

        # Nesterov step
        h = h - beta * m

            h_new = project_box_sum(h_new - 0.1 * alpha * g_max, target)
        m = h_new - h
        h = h_new
    return h

**Idea: "Trading" optimization**
The max is achieved at certain lags. If we can make the top few lags all equal (equioscillation), that's the optimal (by minmax theory). So:
1. Find the top M lags
2. Try to make their correlation values equal
3. This is like a Chebyshev approximation problem

**Idea: Dual variable / Lagrangian approach**
For the problem min_h max_k f_k(h):
- Introduce dual variables λ_k ≥ 0, sum λ_k = 1
- Minimize sum λ_k f_k(h) over h (this is convex for fixed λ)
- Update λ to put more weight on the active constraints
- This is the classical minimax duality

The update: λ_k ← λ_k * exp(T * f_k(h) / max) (softmax update to the dual)

This is actually equivalent to the softmax approach I'm already using! The temperature T controls the sharpness of the dual update.

**Idea: Alternating optimization**
- Step A: Given weights λ on lags, minimize sum λ_k f_k(h) (weighted sum, convex-ish)
- Step B: Update λ based on current f_k values (exponentiate)
- This decouples the problem and might allow better optimization in Step A

For Step A, the weighted sum objective:
sum_k λ_k * (sum_i h[i](1-h[i+k])) * dx

This is a quadratic in h. The gradient is linear in h. I can solve this more efficiently.

Actually, the gradient of sum_k λ_k corr_k(h) is:
sum_k λ_k * (∂corr_k/∂h)

Which is what my softmax_grad computes (with λ_k = sw[center+k]).

So the iteration is:
1. Compute g = gradient of weighted sum
2. Do a few L-BFGS-B steps on this quadratic-ish objective
3. Update weights from current correlations
4. Repeat

This might converge faster than pure softmax because the weights update less frequently.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = weights[center + o]
        if w < 1e-15:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize_weighted(h0, weights, target, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = get_corr(x)
        return float(np.dot(weights, corr))
    def grad(x):
        return weighted_grad(x, weights)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-14})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def block_minmax(h0, target, T_start, T_end, n_blocks, inner_iter, t0, limit):
    h = h0.copy()
    n = len(h)
    L = 2 * n - 1
    for b in range(n_blocks):
        if time.time() - t0 > limit:
            break
        T = T_start + (T_end - T_start) * (b / max(1, n_blocks - 1))
        corr = get_corr(h)
        mc = float(np.max(corr))
        shifted = T * (corr - mc)
        msh = float(np.max(shifted))
        weights = np.exp(shifted - msh)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros(L)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= ws
        h = optimize_weighted(h, weights, target, inner_iter)
    return h

def lbfgs_softmax(h0, target, T, maxiter):
    n = len(h0)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = get_corr(x)
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    def grad(x):
        corr = get_corr(x)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        e = np.exp(shifted)
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        for o in range(-(n-1), n):
            w = sw[center + o]
            if w < 1e-14:
                continue
            if o == 0:
                g += w * (1.0 - 2.0 * x)
            elif o > 0:
                g[o:] += w * (1.0 - x[:n-o])
                g[:n-o] -= w * x[o:]
            else:
                p = -o
                g[:n-p] += w * (1.0 - x[p:])
                g[p:] -= w * x[:n-p]
        return g * dx
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15, 'ftol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.ones(200) * 0.5

    n = len(h_best)
    target = float(n) / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Block minmax - alternating weight update + quadratic optimization
    if elapsed() < budget_s - 400:
        h_new = block_minmax(h_best, target, 30.0, 3000.0, 35, 200, t0, budget_s - 400)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: L-BFGS-B softmax at increasing temperatures
    for T in [100, 400, 1000, 3000, 8000]:
        if elapsed() > budget_s - 300:
            break
        h_new = lbfgs_softmax(h_best, target, T, 4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Second round of block minmax at higher T
    if elapsed() < budget_s - 250:
        h_new = block_minmax(h_best, target, 500.0, 10000.0, 20, 300, t0, budget_s - 250)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Perturbation restarts with varied strategies
    cnt = 0
    while elapsed() < budget_s - 140:
        cnt += 1
        strat = cnt % 3
        if strat == 0:
            h_pert = h_best + rng.normal(0, 0.04, n)
        elif strat == 1:
            h_pert = h_best.copy()
            for _ in range(max(1, n // 15)):
                s = int(rng.integers(0, max(1, n - 8)))
                ln = int(rng.integers(3, min(9, n - s)))
                h_pert[s:s+ln] += rng.normal(0, 0.12)
        else:
            h_pert = h_best.copy()
            for _ in range(max(1, n // 6)):
                i, j = rng.integers(0, n, 2)
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
        h_pert = project_box_sum(h_pert, target)
        T = 200 + int(rng.integers(0, 3000))
        h_new = lbfgs_softmax(h_pert, target, T, 1500)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            break

    # Phase 5: Resolution changes - upsample and optimize
    for n_try in [200, 240, 320, 360, 400, 280]:
        if elapsed() > budget_s - 40:
            break
        if n_try == n:
            continue
        x_old = np.arange(n, dtype=float) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        t_try = float(n_try) / 2.0
        h_try = project_box_sum(h_try, t_try)
        best_l = h_try.copy()
        best_t = compute_c5(h_try)
        for T in [100, 500, 2000]:
            if elapsed() > budget_s - 10:
                break
            h_t = lbfgs_softmax(best_l, t_try, T, 2000)
            c5_t = compute_c5(h_t)
            if c5_t < best_t:
                best_t = c5_t
                best_l = h_t.copy()
        if best_t < c5_best:
            c5_best = best_t
            h_best = best_l
            n = n_try
            target = t_try

    # Phase 6: Final high-T polish
    if elapsed() < budget_s - 3:
        h_new = lbfgs_softmax(h_best, target, 6000, 3000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(c5_best, compute_c5(h_best))
    return (h_best, c5_final, len(h_best))
