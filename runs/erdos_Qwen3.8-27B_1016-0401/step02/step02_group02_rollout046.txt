def project_feasible(x, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]

        wi = w[idx]

            grad += wi * (1.0 - 2.0 * h)

        return np.ones(n)

            h_try = slsqp_phase(h_try, n_try, tgt, bnd, 1.5, 1000)
            c5_t = compute_c5(h_try)
            if c5_t < c5_best:
                c5_best = c5_t
                h_best = h_try.copy()
                n = n_try

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
