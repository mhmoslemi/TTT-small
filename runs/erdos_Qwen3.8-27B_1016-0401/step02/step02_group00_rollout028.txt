def fix_constraint(h, target):

    e = np.exp(sh)
    ss = np.sum(e)
    obj = dx * (mc + np.log(ss) / T)
    sw = e / ss
    center = n - 1
    g = np.zeros(n)
    for idx_o in range(2 * n - 1):
        o = idx_o - center
        w = sw[idx_o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] += -w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] += -w * h[:n - p]
    return obj, g * dx

def _sqp_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.sum(corr[_G_LAGS]) * dx

def _sqp_grad(x):
    n = len(x)
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    for lag in _G_LAGS:
        o = lag - center
        if o == 0:
            g += (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += (1.0 - x[:n - o])
            g[:n - o] += -x[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - x[p:])
            g[p:] += -x[:n - p]
    return g * dx

def _sqp_eq(x):
    return x.sum() - _G_TARGET

def _sqp_eq_jac(x):
    return np.ones(len(x))

def coord_descend(h, n_passes, n_grid, t0, t_end):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    grid = np.linspace(0.0, 1.0, n_grid)
    nlags = 2 * n - 1
    for _pass in range(n_passes):
        if time.time() > t_end:
            break
        for i in range(n):
            old_val = h[i]
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = old_val
            A = corr1 - corr0
            C = corr0.copy()
            vals = A[None, :] * grid[:, None] + C[None, :]
            vals[:, center] -= grid ** 2
            c5_vals = dx * np.max(vals, axis=1)
            bi = int(np.argmin(c5_vals))
            best_t = grid[bi]
            if 0 < bi < n_grid - 1:
                t_lo = grid[bi - 1]
                t_hi = grid[bi + 1]
                fg = np.linspace(t_lo, t_hi, 60)
                vf = A[None, :] * fg[:, None] + C[None, :]
                vf[:, center] -= fg ** 2
                c5f = dx * np.max(vf, axis=1)
                best_t = fg[int(np.argmin(c5f))]
            if abs(best_t - old_val) > 1e-12:
                h[i] = best_t
        h = fix_constraint(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _G_LAGS, _G_TARGET
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sqp_eq, 'jac': _sqp_eq_jac}]

    # Phase 1: Nesterov smoothed gradient descent (35%)
    p1_end = t0 + budget_s * 0.35
    h_sm = h_best.copy()
    h_mom = h_best.copy()
    T = 120.0
    L = 1.0
    for it in range(100000):
        if time.time() > p1_end:
            break
        _, grad = smooth_obj_and_grad(h_sm, T)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        step = L / t_k
        h_new = h_mom - step * grad
        h_new = fix_constraint(h_new, target)
        h_mom = h_new + ((L - 1.0) / t_k) * (h_new - h_sm)
        h_mom = np.clip(h_mom, 0.0, 1.0)
        h_sm = h_new
        L += 1.0
        if it % 1500 == 1499:
            T *= 1.5
        if it % 200 == 0:
            c5_sm = compute_c5(h_sm)
            if c5_sm < c5_best:
                c5_best = c5_sm
                h_best = h_sm.copy()

    # Phase 2: Multi-lag SLSQP (25%)
    p2_end = t0 + budget_s * 0.60
    for outer in range(50):
        if time.time() > p2_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_k = min(3, 2 * n - 1)
        _G_LAGS = np.argsort(corr)[-top_k:]
        _G_TARGET = target
        try:
            res = minimize(_sqp_obj, h_best, method='SLSQP', jac=_sqp_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Coordinate descent on max (20%)
    p3_end = t0 + budget_s * 0.80
    cd_h = h_best.copy()
    while time.time() < p3_end:
        cd_h = coord_descend(cd_h, 5, 120, t0, p3_end)
        c5_cd = compute_c5(cd_h)
        if c5_cd < c5_best - 1e-14:
            c5_best = c5_cd
            h_best = cd_h.copy()
        else:
            # Try one more round with finer grid
            cd_h = coord_descend(cd_h, 3, 250, t0, p3_end)
            c5_cd = compute_c5(cd_h)
            if c5_cd < c5_best - 1e-14:
                c5_best = c5_cd
                h_best = cd_h.copy()
            else:
                break

    # Phase 4: Alternate n values + polish (20%)
    p4_end = t0 + budget_s - 15.0
    n_opts = [252, 308, 336, 392, 420, 448]
    for n_alt in n_opts:
        if time.time() > p4_end:
            break
        t_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, t_alt)
        c5_alt = compute_c5(h_alt)
        if c5_alt > c5_best + 0.003:
            continue
        bounds_alt = [(0.0, 1.0)] * n_alt
        cons_alt = [{'type': 'eq', 'fun': _sqp_eq, 'jac': _sqp_eq_jac}]
        for outer in range(12):
            if time.time() > p4_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            _G_LAGS = np.argsort(corr_a)[-2:]
            _G_TARGET = t_alt
            try:
                res = minimize(_sqp_obj, h_alt, method='SLSQP', jac=_sqp_grad,
                              bounds=bounds_alt, constraints=cons_alt,
                              options={'maxiter': 1500, 'ftol': 1e-17})
                h_new = fix_constraint(np.clip(res.x, 0, 1), t_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-15:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        h_alt = coord_descend(h_alt, 5, 150, t0, p4_end)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt

    # Final coordinate descent polish
    h_final = h_best.copy()
    h_final = coord_descend(h_final, 4, 200, t0, t0 + budget_s - 2.0)
    c5_final = compute_c5(h_final)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h_final

    c5_out = min(compute_c5(h_best), c5_best)
    return (h_best, c5_out, len(h_best))
