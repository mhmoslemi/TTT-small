def project_constraint(h, target):
    """Project h to satisfy 0<=h<=1 and sum(h)=target using bisection on offset."""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj_and_grad(h, T, top_k=None):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

        idx = np.argpartition(corr, -top_k)[-top_k:]
        corr_f = corr[idx]
    else:
        corr_f = corr
    mc = corr_f.max()
    shifted = T * (corr_f - mc)
    shifted -= shifted.max()
    e = np.exp(shifted)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if K < len(corr):
        idx = np.argpartition(corr, -K)[-K:]

    mc = ctk.max()
    sh = T * (ctk - mc) - T * ctk.max()

    se = e.sum()

        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def pg_solve(h0, target, T, K, n_iter, lr0, t_stop):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    stall = 0
    for it in range(n_iter):
        if time.time() > t_stop:
            break
        obj = topk_obj(h, T, K)
        grad = topk_grad(h, T, K)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        accepted = False
        for _ in range(50):
            h_new = project_constraint(h - lr * grad, target)
            obj_new = topk_obj(h_new, T, K)
            if obj_new < obj - 1e-16:
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
            h = h_new
            lr = min(lr * 1.12, lr0)
            if it % 40 == 39:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def binary_ls(h0, target, rng, max_iter, t_stop):
    n = len(h0)
    h = (h0 >= 0.5).astype(float)
    nc = n // 2
    cnt = int(h.sum())
    if cnt > nc:
        idx = np.where(h == 1.0)[0]
        flip = rng.choice(idx, cnt - nc, replace=False)
        h[flip] = 0.0
    elif cnt < nc:
        idx = np.where(h == 0.0)[0]
        flip = rng.choice(idx, nc - cnt, replace=False)
        h[flip] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for _ in range(max_iter):
        if time.time() > t_stop:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        for _ in range(120):
            if time.time() > t_stop:
                break
            p = ones[rng.integers(len(ones))]
            q = zeros[rng.integers(len(zeros))]
            h[p], h[q] = 0.0, 1.0
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5, best_h = c5, h.copy()
                improved = True
                stall = 0
                break
            h[p], h[q] = 1.0, 0.0
        if not improved:
            stall += 1
            if stall > 25:
                p = ones[rng.integers(len(ones))]
                q = zeros[rng.integers(len(zeros))]
                h[p], h[q] = 0.0, 1.0
                stall = 0
    return best_h, best_c5

def _scl_f(x):
    return topk_obj(x, _G_T, _G_K)

def _scl_jac(x):
    return topk_grad(x, _G_T, _G_K)

def _scl_cf(x):
    return x.sum() - _G_TARGET

def _scl_cj(x):
    return np.ones(_G_N)

def scl_solve(h0, target, T, K, maxiter):
    global _G_T, _G_K, _G_TARGET, _G_N
    _G_T = T
    _G_K = K
    _G_TARGET = target
    _G_N = len(h0)
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scl_cf, 'jac': _scl_cj}]
    try:
        res = minimize(_scl_f, h0.copy(), method='SLSQP', jac=_scl_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(560)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Projected gradient with top-K, varied T
    t1 = t0 + min(220, budget_s * 0.20)
    for T, K in [(25, 12), (80, 8), (200, 6), (500, 4), (1000, 3)]:
        if time.time() > t1:
            break
        h, c5 = pg_solve(h_best, target, T, K, 3000, 0.06, t1)
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 2: Binary local search
    t2 = t0 + min(580, budget_s * 0.52)
    h_b, c5_b = binary_ls(h_best, target, rng, 6000, t2)
    if c5_b < c5_best:
        c5_best, h_best = c5_b, h_b

    # Phase 3: Continuous refinement from (near-)binary
    t3 = t0 + min(840, budget_s * 0.75)
    for T, K in [(40, 10), (100, 6), (250, 4), (600, 3)]:
        if time.time() > t3:
            break
        h, c5 = pg_solve(h_best, target, T, K, 2500, 0.04, t3)
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 4: SLSQP fine polish
    if time.time() < t0 + budget_s * 0.84:
        h_s, c5_s = scl_solve(h_best, target, 300, 4, 35)
        if c5_s < c5_best:
            c5_best, h_best = c5_s, h_s

    # Phase 5: Alternate binary + continuous
    t5 = t0 + budget_s * 0.93
    for _ in range(5):
        if time.time() > t5:
            break
        h_b, c5_b = binary_ls(h_best, target, rng, 2000, t5)
        if c5_b < c5_best:
            c5_best, h_best = c5_b, h_b
        h, c5 = pg_solve(h_best, target, 120, 5, 1500, 0.03, t5)
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 6: Final micro-refinement with very high T (sharp focus)
    t6 = t0 + budget_s - 10
    h, c5 = pg_solve(h_best, target, 2000, 3, 1000, 0.008, t6)
    if c5 < c5_best:
        c5_best, h_best = c5, h
    h, c5 = pg_solve(h_best, target, 50, 8, 500, 0.02, t6)
    if c5 < c5_best:
        c5_best, h_best = c5, h

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
