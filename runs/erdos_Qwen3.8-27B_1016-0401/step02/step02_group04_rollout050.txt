    lo, hi = -3.0, 3.0

        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid

    """Compute correlation and gradient of max(corr) w.r.t. h"""

def blocked_obj(h, n_active, power):
    """Minimize sum of top n_active^power correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    sorted_c = np.sort(corr)[::-1]
    top = sorted_c[:n_active]
    return np.sum(top ** power) * dx

def blocked_grad(h, n_active, power, T):
    """Gradient of blocked objective with soft weighting"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    # Soft weights
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    w = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o] * dx
        if wi < 1e-14:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)

            grad[o:] += wi * (1.0 - h[:n-o])

        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])

    return grad

    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)

        c = np.correlate(x, 1-x, mode="full")
        mc = np.max(c)
        shifted = T * (c - mc)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0/n) + 0.5 * (x.sum() - target)**2

        c = np.correlate(x, 1-x, mode="full")
        mc = np.max(c)
        shifted = T * (c - mc)
        e = np.exp(shifted)
        sw = e / np.sum(e)
        center = n - 1
        g = np.zeros(n)

            wi = sw[center+o] * (2.0/n)

                g[:n-o] -= wi * x[o:]
            else:
                p = -o
                g[:n-p] += wi * (1.0 - x[p:])
                g[p:] -= wi * x[:n-p]

        return g

def structured_perturb(h, rng):
    """Apply structured perturbations"""
    n = len(h)
    h = h.copy()
    choice = rng.integers(0, 5)
    if choice == 0:
        # Block flip
        blen = rng.integers(3, max(4, n//10))
        start = rng.integers(0, n - blen)
        h[start:start+blen] = 1.0 - h[start:start+blen]

        s2 = rng.integers(0, n - b2)
        tmp = h[s1:s1+b1].copy()
        h[s1:s1+b1] = h[s2:s2+b2]
        h[s2:s2+b2] = tmp
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    s = np.sum(e)
    sw = e / s
    obj = (mc + np.log(s) / T) * dx
    center = n - 1
    g = np.zeros(n)
    for o in range(-(n-1), n):
        wi = sw[center + o] * dx
        if wi < 1e-15:
            continue
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n-o])
            g[:n-o] -= wi * h[o:]
        else:
            p = -o
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
    return obj, g

def lbfgs(h0, target, T, maxit):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return softmax_obj_and_grad(x, T)[0]
    
    def grad(x):
        return softmax_obj_and_grad(x, T)[1]
    
    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxit, 'gtol': 1e-16})
        return project_box_sum(res.x, target)
    except:
        return h0.copy()

def active_set_step(h, target, n_top, alpha):
    """Take a step along the gradient of the average of top n_top correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_top]
    center = n - 1
    g = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n-o])
            g[:n-o] -= h[o:]
        else:
            p = -o
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
    g *= (alpha * dx) / n_top
    return project_box_sum(h - g, target)

def interp_h(h, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def project_box_sum(h, target):

            lo = mid

def sm_grad(h, T):
    """Soft-max objective gradient"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = c.max()
    sh = T * (c - mc)
    e = np.exp(sh)
    w = e / e.sum()
    ctr = n - 1
    g = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[ctr + o] * dx
        if wi < 1e-15: continue

            p = -o
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
    return g

    c = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = c.max()
    sh = T * (c - mc)
    e = np.exp(sh)
    w = e / e.sum()
    ctr = n - 1
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        wi = w[ctr + o] * dx
        if wi < 1e-15:
            continue
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g

def sm_obj(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = c.max()
    e = np.exp(T * (c - mc))
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def run_lbfgs(h0, target, T, nit):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(sm_obj, h0.copy(), method='L-BFGS-B', jac=lambda x: sm_grad(x, T),
                     bounds=bnds, options={'maxiter': nit, 'gtol': 1e-16})
        return project_box_sum(r.x, target)
    except Exception:
        return h0.copy()

def as_step(h, target, kt, a):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    ti = np.argsort(c)[::-1][:kt]
    ctr = n - 1
    g = np.zeros(n)
    for idx in ti:
        o = idx - ctr
        if o == 0:
            g += 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += 1.0 - h[:n - o]
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
    return project_box_sum(h - a * dx * g / kt, target)

def interp_h(h, no, nn):
    xo = np.arange(no) * (2.0 / no)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(xn, xo, h)

def fista_step(h, zh, t_k, target, T, step):
    n = len(h)
    y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
    g = sm_grad(y, T)
    zh2 = project_box_sum(y - step * g, target)
    t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
    return zh2, t_new

def run_fista(h0, target, T, niter, step):
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0

        if np.linalg.norm(zh - h) < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
        h = zh
    return h
