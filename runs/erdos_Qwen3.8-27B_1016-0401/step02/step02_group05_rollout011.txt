            h_try = subgradient_step(h_best, target, step0, T_w)
            c5_t = compute_c5(h_try)
            if c5_t < c5_best - 1e-14:
                c5_best = c5_t
                h_best = h_try
            else:
                step0 *= 0.8

    # Phase 3: Local search
    h_ls, c5_ls = local_search(h_best, target, n_iter=5000, t0=t0, budget=budget_s - 250)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 4: SLSQP refinement at very low T
    for T in [3, 2, 1.5, 1.0, 0.6, 0.4]:
        if time.time() - t0 > budget_s - 80:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 8000, 'ftol': 1e-17})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-14:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 5: Random perturbations + SLSQP + local search
    rcount = 0
    while time.time() - t0 < budget_s - 120:
        rcount += 1
        if rcount % 7 == 1:
            h_init = h_best + rng.normal(0, 0.05, n)
        elif rcount % 7 == 2:
            h_init = h_best.copy()
            start = int(rng.integers(0, max(1, n - 40)))
            length = int(rng.integers(4, 35))
            h_init[start:start + length] = 1.0 - h_init[start:start + length]
        elif rcount % 7 == 3:
            h_init = h_best.copy()
            for _ in range(int(rng.integers(2, 6))):
                start = int(rng.integers(0, max(1, n - 12)))
                length = int(rng.integers(2, 10))
                h_init[start:start + length] = 1.0 - h_init[start:start + length]
        elif rcount % 7 == 4:
            h_init = h_best + rng.normal(0, 0.02, n)
        elif rcount % 7 == 5:
            # Swap-based perturbation
            h_init = h_best.copy()
            idx = rng.permutation(n)
            for i in range(0, 60, 2):
                a, b = idx[i], idx[i + 1]
                h_init[a], h_init[b] = h_init[b], h_init[a]
        elif rcount % 7 == 6:
            h_init = h_best.copy()
            mask = rng.random(n) < 0.15
            h_init[mask] = 1.0 - h_init[mask]
        else:
            h_init = (rng.random(n) >= 0.5).astype(float)
        h_init = fix_sum(h_init, target)
        T = 20
        try:
            res = minimize(soft_max_obj, h_init, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-14})
            if res.fun < 50:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-14:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass
        h_ls2, c5_ls2 = local_search(h_best, target, n_iter=1500, t0=t0, budget=budget_s - 100)
        if c5_ls2 < c5_best:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 6: Final subgradient + local search polish
    for stage in range(3):
        if time.time() - t0 > budget_s - 10:
            break
        step0 = 0.02 / (stage + 1)
        for it in range(200):
            if time.time() - t0 > budget_s - 10:
                break
            h_try = subgradient_step(h_best, target, step0, 6.0)
            c5_t = compute_c5(h_try)
            if c5_t < c5_best - 1e-14:
                c5_best = c5_t
                h_best = h_try
            else:
                step0 *= 0.7

    h_ls3, c5_ls3 = local_search(h_best, target, n_iter=3000, t0=t0, budget=budget_s - 1)
    if c5_ls3 < c5_best:
        c5_best = c5_ls3
        h_best = h_ls3

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_sum(h_best, target)
    return (h_best, c5_final, len(h_best))
