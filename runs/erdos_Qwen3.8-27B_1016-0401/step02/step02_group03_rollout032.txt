    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _obj(x):
    dx = 2.0 / _N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * dx

def _jac(x):
    dx = 2.0 / _N
    n = _N
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    for oi in range(2 * n - 1):
        w = sw[oi]
        if w < 1e-10:
            continue
        shift = oi - center
        j = np.arange(n)
        if shift >= 0:
            m1 = j >= shift
            m2 = (j + shift) < n
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s = -shift
            m1 = (j + s) < n
            m2 = j >= s
            grad[m1] += w * (1.0 - x[j[m1] + s])
            grad[m2] -= w * x[j[m2] - s]
    return grad * dx

def _cons_fun(x):
    return x.sum() - _N / 2.0

def _cons_jac(x):
    return np.ones(_N)

def slsqp_solve(h, n, T, maxiter, ftol):
    global _N, _T
    _N = n
    _T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coordinate_descent(h, n, passes, t0, budget_s):
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    for p in range(passes):
        if time.time() - t0 > budget_s:
            break
        for i in range(n):
            old_v = h[i]
            best_v = old_v
            best_c = c5_cur
            grid = np.linspace(0, 1, 11)
            for v in grid:
                h[i] = v
                c5_t = compute_c5(h)
                if c5_t < best_c:
                    best_c = c5_t
                    best_v = v
            if best_v != old_v:
                lo = max(0.0, best_v - 0.1)
                hi = min(1.0, best_v + 0.1)
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    h[i] = lo
                    c_lo = compute_c5(h)
                    h[i] = mid
                    c_mid = compute_c5(h)
                    if c_lo < c_mid:
                        hi = mid
                    else:
                        lo = mid
                h[i] = (lo + hi) / 2.0
                c5_t = compute_c5(h)
                if c5_t < best_c:
                    best_c = c5_t
            h[i] = best_v
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
    return h_best, c5_best

def grad_subgradient(h, n, t0, budget_s, max_iters=200000):
    dx = 2.0 / n
    h_sg = h.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    alpha0 = 0.25
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        j = np.arange(n)
        g = np.zeros(n)
        if shift >= 0:
            m1 = j >= shift
            m2 = (j + shift) < n
            g[m1] += (1.0 - h_y[j[m1] - shift])
            g[m2] -= h_y[j[m2] + shift]
        else:
            s = -shift
            m1 = (j + s) < n
            m2 = j >= s
            g[m1] += (1.0 - h_y[j[m1] + s])
            g[m2] -= h_y[j[m2] - s]
        g *= dx
        alpha = alpha0 / (1.0 + 0.015 * it)
        h_new = h_y - alpha * g
        if (it + 1) % 20 == 0:
            h_new = project_feasible(h_new, n)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = n / 2.0 - h_new.sum()
            if abs(diff) > 1e-10:
                h_new = np.clip(h_new + diff / n, 0.0, 1.0)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 20000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    for T in [64, 32, 16, 8, 4, 2, 1]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = slsqp_solve(h_best, n, T, 2000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Coordinate descent (multiple passes)
    h_cd, c5_cd = coordinate_descent(h_best, n, 8, t0, budget_s - 300)
    c5_cd = min(c5_cd, compute_c5(h_cd))
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = h_cd

    # Phase 3: SLSQP fine polish
    for T in [2, 1, 4]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = slsqp_solve(h_best, n, T, 3000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: More coordinate descent
    h_cd2, c5_cd2 = coordinate_descent(h_best, n, 10, t0, budget_s - 150)
    c5_cd2 = min(c5_cd2, compute_c5(h_cd2))
    if c5_cd2 < c5_best:
        c5_best = c5_cd2
        h_best = h_cd2

    # Phase 5: Subgradient gotover + coordinate descent
    h_sg, c5_sg = grad_subgradient(h_best, n, t0, budget_s - 100, 100000)
    c5_sg = min(c5_sg, compute_c5(h_sg))
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_cd3, c5_cd3 = coordinate_descent(h_best, n, 6, t0, budget_s - 60)
    if c5_cd3 < c5_best:
        c5_best = c5_cd3
        h_best = h_cd3

    # Phase 6: Perturbation restarts with coordinate descent
    for pi in range(30):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.005 + 0.010 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new, c5_pw = coordinate_descent(h_p, n, 3, t0, budget_s - 18)
        if c5_pw < c5_best:
            c5_best = c5_pw
            h_best = h_new

    # Phase 7: Final SLSQP + coordinate descent polish
    if time.time() - t0 < budget_s - 15:
        h_new = slsqp_solve(h_best, n, 1, 2000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_cd4, c5_cd4 = coordinate_descent(h_best, n, 4, t0, budget_s - 5)
        if c5_cd4 < c5_best:
            c5_best = c5_cd4
            h_best = h_cd4

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
