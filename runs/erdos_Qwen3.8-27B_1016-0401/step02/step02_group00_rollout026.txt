        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]

    corr = get_corr(h)
    mc = np.max(corr)
    s = T * (corr - mc)
    obj = (mc + np.log(np.sum(np.exp(s)))) / T * dx

                g[:] = 1.0 - 2.0 * h_sg
            elif o > 0:
                g[o:] += 1.0 - h_sg[:n-o]
                g[:n-o] -= h_sg[o:]
            else:
                p = -o
                g[:n-p] += 1.0 - h_sg[p:]
                g[p:] -= h_sg[:n-p]
            g *= dx
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
            h_new_sg = h_mom - (L / t_k) * g
            h_new_sg = fix_constraint(h_new_sg, target)
            h_mom = h_new_sg + ((L - 1.0) / t_k) * (h_new_sg - h_sg)
            h_mom = np.clip(h_mom, 0.0, 1.0)
            h_sg = h_new_sg
            L += 1.0
            if it % 3000 == 0:
                c5_sg = compute_c5(h_sg)
                if c5_sg < c5_best:
                    c5_best = c5_sg
                    h_best = h_sg.copy()

    # Phase 4: Final dense L-BFGS-B polish (15% budget)
    p4_end = t0 + budget_s - 5.0
    T_final = [3000, 5000, 8000, 15000]
    for Tf in T_final:
        if elapsed() > p4_end:
            break
        h_cur = h_best.copy()
        mu = 200.0
        for _ in range(3):
            if elapsed() > p4_end:
                break
            mu *= 3.0
            try:
                res = minimize(sm_og, h_cur, args=(Tf, target, mu),
                              jac=lambda x, T=Tf, tt=target, mm=mu: sm_og(x, T, tt, mm)[1],
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12})
                h_cur = fix_constraint(np.clip(res.x, 0, 1), target)
            except Exception:
                break
        c5_new = compute_c5(h_cur)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_cur.copy()

    # Quick try with slightly different n
    if time.time() - t0 < budget_s - 30:
        for n_alt in [n - 2, n + 2, n + 4]:
            if n_alt < 100 or n_alt > 600:
                continue
            if elapsed() > p4_end:
                break
            target_alt = n_alt / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = fix_constraint(h_alt, target_alt)
            c5_alt = compute_c5(h_alt)
            bounds_alt = [(0.0, 1.0)] * n_alt
            target_global = target_alt
            for T_a in [200, 800, 3000]:
                if elapsed() > p4_end:
                    break
                try:
                    res = minimize(sm_og, h_alt, args=(T_a, target_alt, 500.0),
                                  jac=lambda x, T=T_a, tt=target_alt, mm=500.0: sm_og(x, T, tt, mm)[1],
                                  method='L-BFGS-B', bounds=bounds_alt,
                                  options={'maxiter': 3000, 'ftol': 1e-17, 'gtol': 1e-10})
                    h_alt = fix_constraint(np.clip(res.x, 0, 1), target_alt)
                except Exception:
                    pass
            c5_alt = compute_c5(h_alt)
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_alt
                n = n_alt
                target = target_alt

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_constraint(h_best, len(h_best) / 2.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
