import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = mc + np.log(se) / T
    sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def pg_momentum(h0, target, T, n_iter, lr0, t0, deadline, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta = momentum
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, sw = sm_values(h, T)
        grad = sm_grad_from_sw(h, sw)
        v = beta * v - lr * grad
        h_new = project_constraint(h + v, target)
        # Line search

        if obj_new > obj:

        h = h_new
        if it % 200 == 199:

                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def lp_polish(h, target, K=15, r=0.05, n_iter=50, t0=None, deadline=None):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        C = corr.max()
        # Top K indices
        top_idx = np.argsort(corr)[-K:]
        c_top = corr[top_idx]
        
        # Gradients of each correlation w.r.t. h
        # c_k = sum_i h[i] * (1-h[i+k]) for appropriate i
        # For the "full" correlation, index j corresponds to lag k = j - (n-1)
        grads = []
        for j in top_idx:
            k = j - (n - 1)
            g = np.zeros(n)
            if k == 0:
                g = 1.0 - 2.0 * h
            elif k > 0:
                g[k:] += (1.0 - h[:n-k])
                g[:n-k] -= h[k:]
            else:
                p = -k
                g[:n-p] += (1.0 - h[p:])
                g[p:] -= h[:n-p]
            grads.append(g)
        
        G = np.array(grads)  # K x n
        # LP: minimize C_new
        # s.t. G @ delta <= C_new - c_top  (K constraints)
        #      lb <= delta <= ub  (2n constraints)
        #      sum(delta) = 0  (1 constraint)
        
        lb = np.maximum(-r, -h)  # h + delta >= 0
        ub = np.minimum(r, 1.0 - h)  # h + delta <= 1
        # Also need sum(h+delta) = target → sum(delta) = 0
        
        # Variables: x = [delta_0, ..., delta_{n-1}, C_new] (n+1 vars)
        # Objective: minimize C_new → c = [0,...,0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # G @ delta <= C_new - c_top
        # G @ delta - C_new <= -c_top
        A_ub = np.hstack([G, -np.ones((K, 1))])
        b_ub = -c_top
        
        # Bounds: lb_i <= delta_i <= ub_i, and C_new is free (or >= 0)
        bounds = [(lb[i], ub[i]) for i in range(n)] + [(0, None)]
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            delta = res.x[:n]
            h_new = h + delta
            h_new = project_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            else:
                # Shrink trust region
                r *= 0.7
                if r < 1e-5:
                    break
        else:
            r *= 0.5
            if r < 1e-5:
                break
    
    return best_h, best_c5

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def all_grads(h):
    n = len(h)
    c = n - 1
    G = np.zeros((2 * n - 1, n))
    G[c] = 1.0 - 2.0 * h
    for offset in range(1, n):
        k = offset
        G[c + k, k:] += (1.0 - h[:n - k])
        G[c + k, :n - k] -= h[k:]
        G[c - k, :n - k] += (1.0 - h[k:])
        G[c - k, k:] -= h[:n - k]
    return G

def lp_polish(h, target, K=40, r_init=0.06, n_iter=300, deadline=None):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    r = r_init
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        C = np.max(corr)
        top_idx = np.argsort(corr)[-K:]
        c_t = corr[top_idx]
        G = all_grads(h)[top_idx]
        lb = np.maximum(-r, -h)
        ub = np.minimum(r, 1.0 - h)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((K, 1))])
        b_ub = -c_t
        bounds = list(zip(lb, ub)) + [(None, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            r *= 0.5
            if r < 1e-6:
                break
            continue
        delta = res.x[:n]
        h_new = project_constraint(h + delta, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-11:
            best_c5, best_h = c5_new, h_new.copy()
            h = h_new.copy()
            r = min(r * 1.15, r_init)
        else:
            r *= 0.6
            if r < 1e-6:
                break
    return best_h, best_c5

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = mc + np.log(se) / T
    sw = e / se
    return obj, sw

def sm_grad(h, sw):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def pg_solve(h0, target, T, n_iter, lr0, deadline):
    h = h0.copy()
    v = np.zeros_like(h)
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta = 0.82
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, sw = sm_values(h, T)
        grad = sm_grad(h, sw)
        v = beta * v - lr * grad
        h_new = project_constraint(h + v, target)
        obj_new = sm_values(h_new, T)[0]
        if obj_new > obj + 1e-13:
            for _ in range(30):
                v *= 0.5
                h_new = project_constraint(h + v, target)
                obj_new = sm_values(h_new, T)[0]
                if obj_new <= obj:
                    break
            else:
                v[:] = 0.0
                h_new = h.copy()
        h = h_new
        if it % 250 == 249:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def perturb(h, r, rng):
    n = len(h)
    hp = h.copy()
    mod = r % 5
    if mod == 0:
        idx = rng.integers(0, n)
        sz = rng.integers(3, 18)
        for i in range(sz):
            hp[(idx + i) % n] = 1.0 - hp[(idx + i) % n]
    elif mod == 1:
        hp = np.clip(h + rng.normal(0.0, 0.025, n), 0.0, 1.0)
    elif mod == 2:
        t = rng.uniform(0.35, 0.65)
        hp = (h > t).astype(float)
        hp += rng.uniform(0, 0.15, n) * (1.0 - hp)
    elif mod == 3:
        hp = np.roll(h, int(rng.integers(1, 12)))
    else:
        idx = rng.integers(0, n)
        sz = rng.integers(2, 8)
        perm = rng.permutation(sz)
        for i in range(sz):
            hp[(idx + i) % n] = h[(idx + perm[i]) % n]
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(560)
        h[::2] = 1.0
    n = len(h)
    target = n / 2.0
    h = project_constraint(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SM gradient with momentum
    for T in [30, 80, 200]:
        if time.time() > deadline - 350:
            break
        h, c5 = pg_solve(h, target, T, 2500, 0.035, deadline - 250)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: LP trust-region polishing (key differentiator)
    if time.time() < deadline - 120:
        h, c5 = lp_polish(h, target, K=50, r_init=0.06, n_iter=400, deadline=deadline - 100)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: Diversified perturb + re-optimize
    for r in range(30):
        if time.time() > deadline - 80:
            break
        hp = perturb(h, r, rng)
        hp = project_constraint(hp, target)
        tl = min(deadline - 80, time.time() + 25)
        hp, c5w = pg_solve(hp, target, 60, 600, 0.03, tl)
        hp, c5p = lp_polish(hp, target, K=35, r_init=0.04, n_iter=150, deadline=deadline - 80)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()
            h = hp.copy()

    # Phase 4: Second round of LP polish with more iterations
    if time.time() < deadline - 20:
        h2, c52 = lp_polish(best_h, target, K=60, r_init=0.03, n_iter=500, deadline=deadline - 8)
        if c52 < best_c5:
            best_c5, best_h = c52, h2.copy()

    # Phase 5: Final micro LP
    if time.time() < deadline - 5:
        h3, c53 = lp_polish(best_h, target, K=80, r_init=0.015, n_iter=200, deadline=deadline - 2)
        if c53 < best_c5:
            best_c5, best_h = c53, h3.copy()

    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    return (best_h, best_c5, n)
