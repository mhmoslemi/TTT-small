def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def binary_hill_climb(h, n, target, max_iters, rng):
    """Hill climbing on binary sequences using swaps."""
    h = h.copy()
    c5_cur = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        # Pick random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            # Update tracking
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1
            h[j] = 0
            # Increase diversity: try again with different approach
            if it % 1000 == 0:
                # Try all swaps from a random one-position
                i_test = ones[rng.integers(len(ones))]
                best_j = -1
                best_c5 = c5_cur
                for jj in zeros:
                    h[i_test] = 0
                    h[jj] = 1

                    h[i_test] = 1

                    h[best_j] = 1
                    c5_cur = best_c5

                    zeros = np.where(h == 0)[0]

def compute_c5(h):

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def make_binary(x, n_ones):
    order = np.argsort(x)[::-1]
    h = np.zeros(len(x))
    h[order[:n_ones]] = 1.0
    return h

def binary_sa(h, seed, max_time, T0=0.003, cool_rate=0.999997):
    """Simulated annealing on binary sequences (maintaining n/2 ones)."""
    rng = np.random.default_rng(seed)
    t_start = time.time()
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    
    while time.time() - t_start < max_time:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        d = c5_new - c5_cur
        if d < 0 or (T > 1e-10 and rng.random() < np.exp(-d / T)):
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            ones, zeros = np.where(h == 1), np.where(h == 0)
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool_rate
    return h_best, c5_best

def topk_grad(corr, h, K, n):
    """Subgradient of max(corr) using top-K smoothing."""
    dx = 2.0 / n
    top_idx = np.argpartition(corr, -K)[-K:]
    center = n - 1
    g = np.zeros(n)
    for k in top_idx:
        o = int(k) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def cont_polish(h, target, max_time):
    """Projected gradient descent for continuous h."""
    n = len(h)
    t_start = time.time()
    lr = 0.05
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnant = 0
    
    while time.time() - t_start < max_time and stagnant < 2000:
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = topk_grad(corr, h, 5, n)
        h_new = project_feasible(h - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
        elif c5_new < compute_c5(h):
            h = h_new
            stagnant += 1
        else:
            stagnant += 1
        if stagnant % 200 == 0:
            lr *= 0.8
    return best_h, best_c5

def slsqp_refine(h, target, max_time, T=3.0):
    """SLSQP with log-sum-exp objective."""
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        sh = T * (corr - mc)
        sh -= sh.max()
        return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        w = e / e.sum()
        g = np.zeros(n)
        center = n - 1
        for idx in np.where(w > 1e-8)[0]:
            o = int(idx) - center
            wi = w[idx]
            if o == 0:
                g += wi * (1.0 - 2.0 * x)
            elif o > 0:
                g[o:] += wi * (1.0 - x[:n - o])
                g[:n - o] -= wi * x[o:]
            else:
                p = -o
                g[:n - p] += wi * (1.0 - x[p:])
                g[p:] -= wi * x[:n - p]
        return g * (2.0 / n)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    t_start = time.time()
    maxiter = 2000
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14,
                               'time_limit': max_time})
        h_out = project_feasible(res.x, target)
        return h_out, compute_c5(h_out)
    except Exception:
        return h, compute_c5(h)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def make_binary(x, n_ones):
    order = np.argsort(x)[::-1]
    h = np.zeros(len(x))
    h[order[:n_ones]] = 1.0
    return h

def binary_sa(h, seed, max_time, T0=0.004, cool=0.999997):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    while time.time() - t_start < max_time:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        d = c5_new - c5_cur
        if d < 0 or (T > 1e-10 and rng.random() < np.exp(-d / T)):
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            ones, zeros = np.where(h == 1), np.where(h == 0)
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return h_best, c5_best

def topk_grad(corr, h, K, n):
    dx = 2.0 / n
    idx = np.argpartition(corr, -K)[-K:]
    center = n - 1
    g = np.zeros(n)
    for k in idx:
        o = int(k) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def cont_polish(h, target, max_time):
    n = len(h)
    t_start = time.time()
    lr = 0.08
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnant = 0
    while time.time() - t_start < max_time and stagnant < 3000:
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = topk_grad(corr, h, 6, n)
        h_new = project_feasible(h - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 100 == 0:
                lr *= 0.85
        if lr < 1e-6:
            lr = 0.05
    return best_h, best_c5

def slsqp_refine(h, target, max_time, T=4.0):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = float(corr.max())
        sh = T * (corr - mc)
        sh -= sh.max()
        return (mc + float(np.log(np.sum(np.exp(sh)))) / T) * (2.0 / n)
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = float(corr.max())
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        w = e / e.sum()
        g = np.zeros(n)
        center = n - 1
        for idx in np.where(w > 1e-8)[0]:
            o = int(idx) - center
            wi = w[idx]
            if o == 0:
                g += wi * (1.0 - 2.0 * x)
            elif o > 0:
                g[o:] += wi * (1.0 - x[:n - o])
                g[:n - o] -= wi * x[o:]
            else:
                p = -o
                g[:n - p] += wi * (1.0 - x[p:])
                g[p:] -= wi * x[:n - p]
        return g * (2.0 / n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                      constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target)
        return h_out, compute_c5(h_out)
    except Exception:
        return h, compute_c5(h)

def make_periodic_pattern(n, q):
    h = np.zeros(n)
    half = q // 2
    for i in range(n):
        if (i % q) < half:
            h[i] = 1.0
    # Adjust to get exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[target - n_ones // 2:]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1.0
    return make_binary(h, target)
