    if c5 < bc:
        bc, bh = c5, h.copy()
    return bh, bc

def slsqp_opt(h0, tgt, T, maxit, t0, tlim):
    global _G_T, _G_TGT, _G_N
    _G_TGT, _G_N, _G_T = tgt, len(h0), T
    try:
        r = minimize(_vo, h0, method='SLSQP', jac=_vg,
                     bounds=[(0.0, 1.0)] * _G_N,
                     constraints=[{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}],
                     options={'maxiter': maxit, 'ftol': 1e-14})
        if r.fun < 100:
            h = project_constraint(r.x, tgt)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def active_refine(h, tgt, t0, tlim, niter=4000):
    n = len(h)
    h = h.copy()
    bc = compute_c5(h)
    bh = h.copy()
    lr = 0.008
    stall = 0
    for it in range(niter):
        if time.time() - t0 > tlim:
            break
        c5 = compute_c5(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = np.max(corr)
        T = 2000.0
        w = np.exp(T * (corr - mc))
        w /= w.sum()
        g = sm_grad(h, w)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        ok = False
        for _ in range(25):
            hn = project_constraint(h - lr * g, tgt)
            c5n = compute_c5(hn)
            if c5n < c5 - 1e-11:
                ok = True
                break
            lr *= 0.65
        if not ok:
            stall += 1
            lr = max(lr * 0.7, 1e-8)
            if stall > 30:
                break
            continue
        stall = 0
        h = hn
        lr = min(lr * 1.06, 0.015)
        if c5 < bc - 1e-10:
            bc, bh = c5, h.copy()
    return bh, bc

def local_search(h, tgt, t0, tlim, ntrials=4000, s=42):
    n = len(h)
    rng = np.random.default_rng(s)
    bc = compute_c5(h)
    bh = h.copy()
    for _ in range(ntrials):
        if time.time() - t0 > tlim:
            break
        ht = bh.copy()
        r = rng.random()
        if r < 0.2:
            s2 = rng.integers(0, n - 3)
            l = rng.integers(1, 7)
            ht[s2:s2+l] = 1.0 - ht[s2:s2+l]
        elif r < 0.45:
            for _ in range(rng.integers(1, 4)):
                ht[rng.integers(0, n)] = rng.uniform(0, 1)
        elif r < 0.65:
            i, j = rng.choice(n, 2, replace=False)
            ht[[i, j]] = ht[[j, i]]
        elif r < 0.82:
            c = rng.integers(5, n - 5)
            w = rng.integers(2, 14)
            ht[max(0, c-w):min(n, c+w)] = np.clip(
                ht[max(0, c-w):min(n, c+w)] + rng.normal(0, 0.07, min(2*w, n)), 0, 1)
        else:
            s2 = rng.integers(0, n - 10)
            l = rng.integers(3, 18)
            ht[s2:s2+l] = np.roll(ht[s2:s2+l], rng.integers(1, l))
        ht = project_constraint(ht, tgt)
        c5t = compute_c5(ht)
        if c5t < bc - 1e-12:
            bc, bh = c5t, ht.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_b = np.array(initial_h_values, dtype=float)
    else:
        h_b = np.zeros(560)
        h_b[::2] = 1.0
    n = len(h_b)
    tgt = n / 2.0
    h_b = project_constraint(h_b, tgt)
    c5b = compute_c5(h_b)

    # Phase 1: SLSQP warm-up
    for T in [60, 200]:
        if time.time() - t0 > B - 500:
            break
        h, c5 = slsqp_opt(h_b, tgt, T, 250, t0, B - 350)
        if c5 < c5b:
            c5b, h_b = c5, h

    # Phase 2: PG with escalating T
    for T in [25, 60, 150, 350, 800, 2000]:
        if time.time() - t0 > B - 180:
            break
        ni = 4000 if T < 100 else 2500
        h_b, c5 = pg_opt(h_b, tgt, T, ni, 0.04 if T < 100 else 0.018, t0, B - 60)
        if c5 < c5b:
            c5b = c5

    # Phase 3: SLSQP at high T
    for T in [400, 1000]:
        if time.time() - t0 > B - 90:
            break
        h, c5 = slsqp_opt(h_b, tgt, T, 350, t0, B - 50)
        if c5 < c5b:
            c5b, h_b = c5, h

    # Phase 4: Active-lag direct refinement
    if time.time() - t0 < B - 50:
        h_b, c5 = active_refine(h_b, tgt, t0, B - 15, 5000)
        if c5 < c5b:
            c5b = c5

    # Phase 5: Local combinatorial search
    if time.time() - t0 < B - 10:
        h_b, c5 = local_search(h_b, tgt, t0, B - 5, 3500, seed)
        if c5 < c5b:
            c5b = c5

    # Phase 6: Restart with perturbation + re-optimize
    if time.time() - t0 < B - 15:
        hp = h_b.copy()
        hp += rng.normal(0, 0.03, n)
        hp = project_constraint(np.clip(hp, 0, 1), tgt)
        hp, c5 = pg_opt(hp, tgt, 500, 1500, 0.02, t0, B - 8)
        if c5 < c5b:
            c5b, h_b = c5, hp
        else:
            h_b, c5 = active_refine(h_b, tgt, t0, B - 3, 2000)
            if c5 < c5b:
                c5b = c5

    # Final
    c5f = min(compute_c5(h_b), c5b)
    return (h_b, c5f, len(h_b))
