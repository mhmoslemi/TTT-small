def soft_max_obj_grad(h, T, pen, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -500, 500)
    e = np.exp(shifted)
    se = np.sum(e)

    obj = (mc + np.log(se) / T) * dx
    if pen > 0:
        obj += pen * (h.sum() - target) ** 2
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
