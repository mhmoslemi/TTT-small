def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def active_set_grad(h, top_k, T=0):
    """Gradient using top-k active lags with optional soft smoothing."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    if T > 0:
        mc = np.max(corr)
        shifted = T * (corr - mc)
        e = np.exp(shifted)
        sw = e / np.sum(e)
        # Zero out weights below top_k threshold
        thresh = np.sort(e)[-top_k]
        sw[e < thresh * 0.5] = 0.0
        sw /= sw.sum()

    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0

        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 5: Final polish at very high T (near-exact max)
    if elapsed() < budget_s - 10:
        for T_fin in [3000, 5000]:
            if elapsed() > budget_s - 5:
                break
            h_new = lbfgsb_run(h_best, n, target, T_fin, 3, 1000)
            h_new = project_box_sum(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Final FISTA polish
    if elapsed() < budget_s - 2:
        h_f = fista_active(h_best, target, 3, 3000, 0.1, 0.0002)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_box_sum(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
