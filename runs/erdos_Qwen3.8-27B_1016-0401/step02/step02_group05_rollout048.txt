def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(g, target):
    """Project g onto {h: 0<=h<=1, sum(h)=target} using bisection on lambda."""
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2, 0.0, 1.0)
    # Final correction for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Spread the difference to elements not at bounds
        interior = (h > 1e-10) & (h < 1 - 1e-10)
        if interior.sum() > 0:
            adj = diff / interior.sum()
            h[interior] += adj
            h = np.clip(h, 0, 1)
    return h

import cvxpy as cp

def solve_qp(p, n):
    """Solve min sum_k p_k * corr_k(h) s.t. 0<=h<=1, sum(h)=n/2"""
    h = cp.Variable(n)
    # corr_k(h) = sum_i h[i] * (1-h[i+k]) for valid indices
    # = sum_i h[i] - sum_i h[i]*h[i+k]
    # For lag 0: sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
    # For lag s>0: sum_{i=0}^{n-1-s} h[i+s]*(1-h[i]) = sum_{i=0}^{n-1-s} h[i+s] - sum_{i=0}^{n-1-s} h[i+s]*h[i]
    
    # This is quadratic. Let's build it.
    # objective = sum_k p_k * (linear_k - quadratic_k)
    # = sum_k p_k * linear_k - sum_k p_k * quadratic_k
    
    # For the quadratic part: sum_k p_k * sum_i h[i]*h[i+k] (with appropriate indices)
    # This can be written as h^T Q h where Q is built from the p_k weights
    
    # Actually, let's just build it term by term using cvxpy
    obj = 0
    for k in range(-(n-1), n):
        if p[k + (n-1)] < 1e-12:  # skip if weight is negligible
            continue
        s = k  # shift
        p_k = p[k + (n-1)]
        if s == 0:
            obj += p_k * (cp.sum(h) - cp.sum_squares(h))
        elif s > 0:
            # sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # = sum_{i=0}^{n-1-s} h[i+s] - sum_{i=0}^{n-1-s} h[i+s]*h[i]
            obj += p_k * (cp.sum(h[s:]) - cp.sum(cp.multiply(h[s:], h[:n-s])))
        else:
            p2 = -s
            # sum_{i=0}^{n-1-p2} h[i]*(1-h[i+p2])
            obj += p_k * (cp.sum(h[:n-p2]) - cp.sum(cp.multiply(h[:n-p2], h[p2:])))
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    prob.solve(solver=cp.SCS)  # or ECOS, CVXOPT
    return h.value

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(g, target):
    n = len(g)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = interior.sum()
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0, 1)
    return h

def lag_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    p = abs(s)
    if p == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        grad[p:] += 1.0 - h[:n - p]
        grad[:n - p] -= h[p:]
    else:
        p = -s
        grad[:n - p] += 1.0 - h[p:]
        grad[p:] -= h[:n - p]
    return grad

def weighted_gradient(h, p, n):
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk > 1e-14:
            grad += pk * lag_gradient(h, k)
    return grad

def solve_inner(h_init, p, n, target, max_iter=500, t0=None, budget=None):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    step = 0.5
    for it in range(max_iter):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        grad = weighted_gradient(h, p, n)
        h_new = project_sum_box(h - step * grad, target)
        corr = np.correlate(h_new, 1 - h_new, mode="full")
        c5_new = float(np.max(corr) * (2.0 / n))
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
            step = min(step * 1.05, 2.0)
        else:
            step *= 0.8
            if step < 0.005:
                break
    return h, best_c5

def local_search(h, target, n_iter=5000, t0=None, budget=None):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    corr = np.correlate(h_curr, 1 - h_curr, mode="full")
    best_c5 = float(np.max(corr) * dx)
    delta = 0.04
    stall = 0
    for it in range(n_iter):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        if stall > 300:
            delta *= 0.65
            if delta < 0.002:
                break
            stall = 0
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        h_try = h_curr.copy()
        if p == 0:
            vals = h_curr * (1.0 - h_curr)
            idx = int(np.argmax(vals))
            h_try[idx] += -delta if h_curr[idx] > 0.5 else delta
        else:
            if s > 0:
                contrib = h_curr[p:] * (1.0 - h_curr[:n - p])
            else:
                contrib = h_curr[:n - p] * (1.0 - h_curr[p:])
            idx = int(np.argmax(contrib))
            if s > 0:
                h_try[idx + p] -= delta
                h_try[idx] += delta
            else:
                h_try[idx] -= delta
                h_try[idx + p] += delta
        h_try = project_sum_box(h_try, target)
        corr_try = np.correlate(h_try, 1 - h_try, mode="full")
        c5_try = float(np.max(corr_try) * dx)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            h_curr = h_try
            corr = corr_try
            stall = 0
        else:
            stall += 1
    return h_curr, best_c5

def minimax_loop(h, n, t0, budget, max_rounds=150):
    target = n / 2.0
    num_lags = 2 * n - 1
    p = np.ones(num_lags) / num_lags
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    for rnd in range(max_rounds):
        if time.time() - t0 > budget:
            break
        inner_budget = min(time.time() + 60, budget)
        h_qp, c5_qp = solve_inner(h_curr, p, n, target, max_iter=300, t0=t0, budget=inner_budget)
        if c5_qp < best_c5:
            best_c5 = c5_qp
            h_curr = h_qp
        corr = np.correlate(h_curr, 1 - h_curr, mode="full") * (2.0 / n)
        T_p = 8.0 / (rnd + 1.0)
        p_new = np.exp(-corr / max(T_p, 0.5))
        p_new = p_new / (p_new.sum() + 1e-30)
        p = 0.4 * p + 0.6 * p_new
    return h_curr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_sum_box(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Minimax iteration (main optimization)
    h_mm, c5_mm = minimax_loop(h_best, n, t0, budget_s - 350, max_rounds=120)
    if c5_mm < c5_best:
        c5_best = c5_mm
        h_best = h_mm

    # Phase 2: Local search
    h_ls, c5_ls = local_search(h_best, target, n_iter=8000, t0=t0, budget=budget_s - 250)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 3: Fine minimax with concentrated p
    h_mm2, c5_mm2 = minimax_loop(h_best, n, t0, budget_s - 150, max_rounds=60)
    if c5_mm2 < c5_best:
        c5_best = c5_mm2
        h_best = h_mm2

    # Phase 4: SLSQP polish at low temperature
    dx = 2.0 / n
    def slsqp_obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(c)
        T = 1.5
        return (mc + np.log(np.sum(np.exp((c - mc) * T)) / T)) * dx

    def slsqp_jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(c)
        T = 1.5
        e = np.exp((c - mc) * T)
        sw = e / e.sum()
        grad = np.zeros(n)
        center = n - 1
        for o in range(-(n - 1), n):
            w = sw[center + o]
            if w < 1e-14:
                continue
            grad += w * lag_gradient(x, o)
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(slsqp_obj, h_best, method='SLSQP', jac=slsqp_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-18})
        if res.fun < 5.0:
            h_new = project_sum_box(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
    except Exception:
        pass

    # Phase 5: Final local search
    h_ls2, c5_ls2 = local_search(h_best, target, n_iter=4000, t0=t0, budget=budget_s - 5)
    if c5_ls2 < c5_best:
        c5_best = c5_ls2
        h_best = h_ls2

    # Final projection and verification
    h_final = project_sum_box(h_best, target)
    c5_final = compute_c5(h_final)
    if c5_final < c5_best:
        c5_best = c5_final
    return (h_final, c5_best, n)
