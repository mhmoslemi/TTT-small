    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    sh = T * (corr - mc) - np.max(T * (corr - mc))
    e = np.exp(sh)

        h_bin, c5_bin = binary_local_search(h_best, target, 150, t0, budget - 80, seed=777)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin

    # Phase 3: Continue gradient on best (possibly binary)
    if time.time() - t0 < budget - 100:
        h_best, c5 = run_multistart(h_best, target, n, t0, budget - 60,
                                    [300, 800, 1500, 3000], 2000, 0.02)
        if c5 < c5_best:
            c5_best = c5

    # Phase 4: Another local search round
    if time.time() - t0 < budget - 50:
        h_bin, c5_bin = binary_local_search(h_best, target, 80, t0, budget - 20, seed=444)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin

    # Phase 5: Even higher T refinement + local search
    if time.time() - t0 < budget - 40:
        h_best, c5 = pg_solve(h_best, target, 5000, 2000, 0.01, t0, budget - 15)
        if c5 < c5_best:
            c5_best = c5
        if time.time() - t0 < budget - 10:
            h_bin, c5_bin = binary_local_search(h_best, target, 40, t0, budget - 3, seed=222)
            if c5_bin < c5_best:
                c5_best = c5_bin
                h_best = h_bin

    # Phase 6: Try n=840 if time allows
    if time.time() - t0 < budget - 150:
        n2 = 840
        target2 = n2 / 2.0
        xc = np.arange(n) * (2.0 / n)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x2, xc, h_best)
        h2 = project_constraint(h2, target2)
        h2, c5_2 = run_multistart(h2, target2, n2, t0, budget - 50,
                                  [50, 150, 400, 900], 1500, 0.025)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2
        if time.time() - t0 < budget - 25:
            h_bin2, c5_bin2 = binary_local_search(h_best, target2, 60, t0, budget - 8, seed=555)
            if c5_bin2 < c5_best:
                c5_best = c5_bin2
                h_best = h_bin2

    # Final: micro-refinement
    nf = len(h_best)
    tf = nf / 2.0
    if time.time() - t0 < budget - 12:
        h_best, c5_f = pg_solve(h_best, tf, 2000, 800, 0.01, t0, budget - 4)
        if c5_f < c5_best:
            c5_best = c5_f
        h_bin, c5_bin = binary_local_search(h_best, tf, 20, t0, budget - 1, seed=111)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = np.clip(h_best, 0.0, 1.0)
    return (h_best, c5_final, len(h_best))
