        g[o:] += -x[:n-o]
    else:
        p = -o
        g[p:] += (1.0 - x[:n-p])
        g[:n-p] += -x[p:]
    return g * dx

def _eq_fun(x):
    return x.sum() - _G_TARGET

def _eq_jac(x):
    return np.ones(_G_N)

def _sm_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = _G_T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / _G_T) * dx

def _sm_grad(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = _G_T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o] * dx
        if w < 1e-14:
            continue
        if o == 0:
            g += (1.0 - 2.0 * x) * w
        elif o > 0:
            g[:n-o] += (1.0 - x[o:]) * w
            g[o:] += -x[:n-o] * w
        else:
            p = -o
            g[p:] += (1.0 - x[:n-p]) * w
            g[:n-p] += -x[p:] * w
    return g

def _cons_list():
    return [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

def run(seed=42, budget_s=1100, **kwargs):
    global _G_LAG, _G_TARGET, _G_T, _G_N
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _G_N = n
    _G_TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Softmax L-BFGS-B sweep (22%)
    p1_end = t0 + budget_s * 0.22
    for T in [60, 120, 250, 500, 1000, 2000, 4000]:
        if elapsed() > p1_end:
            break
        _G_T = T
        try:
            res = minimize(_sm_obj, h_best, method='L-BFGS-B',
                          jac=_sm_grad, bounds=bounds,
                          options={'maxiter': 8000, 'gtol': 1e-12, 'ftol': 1e-15})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 2: Active-set SLSQP cycling (30%)
    p2_end = t0 + budget_s * 0.52
    for outer in range(150):
        if elapsed() > p2_end:
            break
        _G_N = n
        _G_TARGET = target
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        center = n - 1
        top_k = min(7, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        improved = False
        for lag in top_idx:
            if elapsed() > p2_end:
                break
            _G_LAG = lag
            try:
                res = minimize(_lag_obj, h_best, method='SLSQP',
                              jac=_lag_grad, bounds=bounds,
                              constraints=_cons_list(),
                              options={'maxiter': 2000, 'ftol': 1e-17})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-17:
                    c5_best = c5_new
                    h_best = h_new
                    improved = True
            except Exception:
                pass
        if not improved and outer % 15 == 14:
            _G_T = 3000 + 500 * (outer // 15)
            try:
                res = minimize(_sm_obj, h_best, method='L-BFGS-B',
                              jac=_sm_grad, bounds=bounds,
                              options={'maxiter': 5000, 'gtol': 1e-12})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 3: Perturb + re-optimize to escape local minima (10%)
    p3_end = t0 + budget_s * 0.62
    for trial in range(5):
        if elapsed() > p3_end:
            break
        noise = rng.uniform(-0.02, 0.02, size=n)
        h_pert = fix_constraint(h_best + noise, target)
        c5_pert = compute_c5(h_pert)
        _G_N = n
        _G_TARGET = target
        _G_T = 2000
        try:
            res = minimize(_sm_obj, h_pert, method='L-BFGS-B',
                          jac=_sm_grad, bounds=bounds,
                          options={'maxiter': 4000, 'gtol': 1e-12})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
        corr_p = np.correlate(h_best, 1.0 - h_best, mode="full")
        for lag in np.argsort(corr_p)[-3:]:
            if elapsed() > p3_end:
                break
            _G_LAG = lag
            try:
                res = minimize(_lag_obj, h_best, method='SLSQP',
                              jac=_lag_grad, bounds=bounds,
                              constraints=_cons_list(),
                              options={'maxiter': 1000, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 4: Subgradient Nesterov (12%)
    p4_end = t0 + budget_s * 0.74
    h_sg = h_best.copy()
    h_mom = h_best.copy()
    center = n - 1
    for it in range(40000):
        if elapsed() > p4_end:
            break
        corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        worst = np.argmax(corr)
        o = worst - center
        dx = 2.0 / n
        g = np.zeros(n)
        if o == 0:
            g = (1.0 - 2.0 * h_sg) * dx
        elif o > 0:
            g[:n-o] += (1.0 - h_sg[o:]) * dx
            g[o:] += -h_sg[:n-o] * dx
        else:
            p = -o
            g[p:] += (1.0 - h_sg[:n-p]) * dx
            g[:n-p] += -h_sg[p:] * dx
        step = 1.0 / (np.linalg.norm(g) * np.sqrt(it + 1) + 1e-5)
        step = min(step, 0.5)
        h_new_sg = fix_constraint(h_mom - step * g, target)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1))) / 2.0
        h_mom = h_new_sg + (t_k - 1.0) / t_k * (h_new_sg - h_sg)
        h_mom = np.clip(h_mom, 0.0, 1.0)
        h_sg = h_new_sg
        if it % 3000 == 0 and it > 0:
            c5_sg = compute_c5(h_sg)
            if c5_sg < c5_best:
                c5_best = c5_sg
                h_best = h_sg.copy()

    # Phase 5: Alternate n (10%)
    p5_end = t0 + budget_s * 0.84
    for n_alt in [252, 308, 350]:
        if elapsed() > p5_end or n_alt == len(h_best):
            continue
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        _G_N = n_alt
        _G_TARGET = target_alt
        for outer in range(25):
            if elapsed() > p5_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            _G_LAG = np.argmax(corr_a)
            try:
                res = minimize(_lag_obj, h_alt, method='SLSQP',
                              jac=_lag_grad, bounds=bounds_alt,
                              constraints=_cons_list(),
                              options={'maxiter': 600, 'ftol': 1e-15})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-16:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt

    # Phase 6: Final intensive polish (14%)
    p6_end = t0 + budget_s - 10.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    _G_N = n_f
    _G_TARGET = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    center_f = n_f - 1

    for outer in range(60):
        if elapsed() > p6_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        top3 = np.argsort(corr_f)[-3:]
        top3 = top3[np.argsort(corr_f[top3])[::-1]]
        for lag in top3:
            if elapsed() > p6_end:
                break
            _G_LAG = lag
            _G_N = n_f
            _G_TARGET = target_f
            try:
                res = minimize(_lag_obj, h_best, method='SLSQP',
                              jac=_lag_grad, bounds=bounds_f,
                              constraints=_cons_list(),
                              options={'maxiter': 2500, 'ftol': 1e-18})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass
        if outer % 5 == 4:
            _G_T = 6000 + 2000 * (outer // 5)
            _G_N = n_f
            try:
                res = minimize(_sm_obj, h_best, method='L-BFGS-B',
                              jac=_sm_grad, bounds=bounds_f,
                              options={'maxiter': 3000, 'gtol': 1e-13})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Final cleanup
    h_best = fix_constraint(np.clip(h_best, 0, 1), len(h_best) / 2.0)
    c5_final = min(c5_best, compute_c5(h_best))
    return (h_best, c5_final, len(h_best))
