def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        if abs(diff) < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

_cn = 0
_ct = 0.0
_cl = 0
_cT = 100.0

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def bin_c5(hb):
    n = len(hb)
    f = hb.astype(float)
    corr = np.correlate(f, 1.0 - f, mode="full")
    return np.max(corr) * (2.0 / n)

def _lag_obj(x):
    return np.correlate(x, 1.0 - x, mode="full")[_cl] * (2.0 / _cn)

def _lag_grad(x):
    n = _cn
    o = _cl - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] += -x[:n-p]
    return g * dx

def _eq_fun(x):
    return x.sum() - _ct

def _eq_jac(x):
    return np.ones(len(x))

def _sm_obj(x):
    n = _cn
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _cT * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    return (mc + np.log(np.sum(e)) / _cT) * (2.0 / n)

def _sm_grad(x):
    n = _cn
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _cT * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    sw = e / np.sum(e)
    c = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[c + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n-o])
            g[:n-o] += -w * x[o:]
        else:
            p = -o
            g[:n-p] += w * (1.0 - x[p:])
            g[p:] += -w * x[:n-p]
    return g * dx

def sa_swap(hb, n, rng, T_init, cool, max_it, t_end):
    tgt = n // 2
    best = hb.copy()
    bc = bin_c5(hb)
    cc = bc
    T = T_init
    for it in range(max_it):
        if time.time() > t_end or T < 1e-9:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.integers(0, len(ones))]
        i0 = zeros[rng.integers(0, len(zeros))]
        hb[i1] = 0
        hb[i0] = 1
        nc = bin_c5(hb)
        d = nc - cc
        if d < 0 or rng.random() < np.exp(-d / T):
            cc = nc
            if nc < bc:
                bc = nc
                best = hb.copy()
        else:
            hb[i1] = 1
            hb[i0] = 0
        if it % 300 == 0:
            T *= cool
    return best, bc

def sd_binary(hb, n, max_rnd=300):
    tgt = n // 2
    cc = bin_c5(hb)
    for _ in range(max_rnd):
        best_d = 0.0
        best_i = -1
        for i in range(n):
            if hb[i] == 1:
                j = np.where(hb == 0)[0]
                if len(j) == 0:
                    continue
            else:
                j = np.where(hb == 1)[0]
                if len(j) == 0:
                    continue
            jv = j[0]
            hb[hb[i]] = hb[i]
            hb[i] = 1 - hb[i]
            hb[jv] = 1 - hb[jv]
            tc = bin_c5(hb)
            if cc - tc > best_d:
                best_d = cc - tc
                best_i = i
                best_j = jv
            hb[i] = 1 - hb[i]

            break
        hb[best_i] = 1 - hb[best_i]
        hb[best_j] = 1 - hb[best_j]
        cc = bin_c5(hb)
    return hb, cc
