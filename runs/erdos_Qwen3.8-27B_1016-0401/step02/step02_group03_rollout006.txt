from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, n, T, penalty=1000.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    e = np.exp(sh)
    lse = mc + np.log(e.sum()) / T
    sw = e / e.sum()
    
    # Penalty for constraint violation
    s = h.sum() - n / 2.0
    obj = lse * dx + penalty * s * s
    
    # Gradient
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(sw)):
        w = sw[idx]
        if w < 1e-14:
            continue
        sh_val = idx - center
        if sh_val >= 0:
            ns = n - sh_val
            if ns > 0:
                grad[:ns] += w * (1.0 - h[sh_val:])
            if sh_val > 0:
                grad[sh_val:] -= w * h[:ns]
        else:
            t = -sh_val
            ns = n - t
            if ns > 0:
                grad[:ns] += w * (1.0 - h[t:])
            if t > 0:
                grad[t:] -= w * h[:ns]
    grad *= dx
    grad += 2.0 * penalty * s * np.ones(n)
    
    return obj, grad

        return lse_obj_and_grad(x, n, T, penalty)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 1.0
_G_PEN = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(y - mid, 0.0, 1.0)

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    lse = mc + np.log(e.sum()) / _G_T
    sv = float(x.sum()) - _G_N / 2.0
    return lse * (2.0 / _G_N) + _G_PEN * sv * sv

def _grad(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(sw)):
        w = sw[idx]
        if w < 1e-14:
            continue
        s = idx - center
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += w * (1.0 - x[s:])
            if s > 0:
                grad[s:] -= w * x[:ns]
        else:
            t = -s
            ns = n - t
            if ns > 0:
                grad[:ns] += w * (1.0 - x[t:])
            if t > 0:
                grad[t:] -= w * x[:ns]
    grad *= dx
    sv = float(x.sum()) - n / 2.0
    grad += 2.0 * _G_PEN * sv
    return grad

def run_lbfgsb(h, n, T, maxiter, ftol, pen=500.0):
    global _G_N, _G_T, _G_PEN
    _G_N = n
    _G_T = float(T)
    _G_PEN = float(pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='L-BFGS-B', jac=_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return res.x
    except Exception:
        return h.copy()

def nesterov_subgrad(h_init, n, t0, deadline):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_best_local = h_init.copy()
    c5_best_local = compute_c5(h_init)
    alpha = 0.3
    for it in range(500000):

            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            ns = n - s
            if ns > 0:
