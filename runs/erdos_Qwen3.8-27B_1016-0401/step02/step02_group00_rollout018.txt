def project_box_eq(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def lag_grad_raw(h, lag_idx):
    n = len(h)
    center = n - 1
    o = lag_idx - center
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n - o]
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - h[p:]
        g[p:] += -h[:n - p]
    return g

    e = np.exp(shifted)
    sw = e / e.sum()
    top_k = min(K, 2 * n - 1)
