def coord_descent(h, n, target, max_sweeps=200):
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    gr = (np.sqrt(5) - 1) / 2
    
    for sweep in range(max_sweeps):
        for i in range(n):
            old_v = h[i]
            # Compute b[c] = ∂corr[c]/∂h[i] for all c
            b = np.zeros(2 * n - 1)
            # For lag o > 0 (c = center + o, o = 1..n-1):
            #   b[center+o] includes (1-h[i-o]) if i >= o, and -h[i+o] if i < n-o
            for o in range(1, n):
                c1 = center + o
                if i >= o:
                    b[c1] += (1.0 - h[i - o])
                if i < n - o:
                    b[c1] -= h[i + o]
                c2 = center - o
                if i < n - o:
                    b[c2] += (1.0 - h[i + o])
                if i >= o:
                    b[c2] -= h[i - o]

            b[center] = 1.0 - 2.0 * old_v

    b[center] = 1.0 - 2.0 * h[i]

    o = np.arange(1, n)
    # Positive lag part
    mask_pos1 = (i >= o)  # i >= o

    b[center - o[mask_pos2]] += (1.0 - h[i + o[mask_pos2]])

    return b

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    # Objective
    obj = (mc + np.log(np.sum(e)) / T) * dx
    # Gradient
    grad = np.zeros(n)
    center = n - 1
    for o_idx in range(len(sw)):
        w = sw[o_idx]
        if w < 1e-13:
            continue
        o = o_idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return obj, grad * dx

            mask_lo = (i < n - o_all)
            b[center + o_all[mask_hi]] += (1.0 - h[i - o_all[mask_hi]])
            b[center + o_all[mask_lo]] -= h[i + o_all[mask_lo]]
            b[center - o_all[mask_lo]] += (1.0 - h[i + o_all[mask_lo]])
            b[center - o_all[mask_hi]] -= h[i - o_all[mask_hi]]
            
            # a[c] = corr[c] - old_v * b[c]
            a = corr - old_v * b
            # Golden section to find v minimizing max(a + b*v) on [0,1]
            lo, hi = 0.0, 1.0
            c1 = hi - gr * (hi - lo)
            d1 = lo + gr * (hi - lo)
            fc = np.max(a + b * c1)
            fd = np.max(a + b * d1)
            for _ in range(35):
                if fc < fd:
                    hi = d1; d1 = c1; fd = fc
                    c1 = hi - gr * (hi - lo)
                    fc = np.max(a + b * c1)
                else:
                    lo = c1; c1 = d1; fc = fd
                    d1 = lo + gr * (hi - lo)
                    fd = np.max(a + b * d1)
            new_v = (lo + hi) / 2.0
            # Clamp
            new_v = max(0.0, min(1.0, new_v))
            delta = new_v - old_v
            if abs(delta) > 1e-12:
                max_change = max(max_change, abs(delta))
                h[i] = new_v
                corr += delta * b
        
        # Re-project
        h_new = project_box_sum(h, target)
        if max_change < 1e-10:
            h = h_new
            break
        h = h_new
        corr = np.correlate(h, 1 - h, mode="full")
    
    return h
