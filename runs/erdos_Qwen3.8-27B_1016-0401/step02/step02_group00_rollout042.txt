def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-15:
                break
    return h

def sm_grad(x, T):
