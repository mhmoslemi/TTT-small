import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj_and_sw(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    return obj, sw, dx

def sm_grad(h, sw, dx):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def momentum_pg(h0, target, T, n_iter, lr0, beta, t0, budget):

    v = np.zeros_like(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        obj, sw, dx = sm_obj_and_sw(h, T)
        grad = sm_grad(h, sw, dx)
        v = beta * v - lr0 * grad
        h_new = project_constraint(h + v, target)
        obj_new, _, _ = sm_obj_and_sw(h_new, T)
        lam = lr0
        for _ in range(25):
            if obj_new < obj - 1e-16:
                break
            lam *= 0.5
            h_new = project_constraint(h - lam * grad + beta * v * lam / lr0, target)
            obj_new, _, _ = sm_obj_and_sw(h_new, T)
        else:
            h_new = project_constraint(h - lam * grad, target)
        v = beta * v - lam * grad
        h = h_new
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def subgrad_topk(h0, target, topk, n_iter, lr0, t0, budget):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    dx = 2.0 / n

        mc = np.max(corr)
        idx = np.argsort(corr)[-topk:]
        w = np.exp(20.0 * (corr[idx] - mc))
        w /= w.sum()
        grad = np.zeros(n)
        c = n - 1

            k = j - c

def _sg_fun(x, target):
    return x.sum() - target

def _sg_jac(x):
    return np.ones(len(x))

def slsqp_solve(h0, target, T, maxiter, t0, budget):
    n = len(h0)
    dx = 2.0 / n
    def fobj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= sh.max()
        return (mc + np.log(np.exp(sh).sum()) / T) * dx
    def fgrad(x):
        obj, sw, _ = sm_obj_and_sw(x, T)
        return sm_grad(x, sw, dx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fobj, h0, method='SLSQP', jac=fgrad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)
