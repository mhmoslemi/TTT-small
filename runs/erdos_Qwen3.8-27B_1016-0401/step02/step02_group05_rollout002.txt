import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, room[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(-h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def grad_topk_corr(h, k_top):
    n = len(h)
    corr = compute_corr(h)
    center = n - 1
    topk = np.argpartition(corr, -k_top)[-k_top:]
    grad = np.zeros(n)
    for o_idx in topk:
        o = int(o_idx) - center
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n - o] += (1.0 - h[o:])
            grad[o:] -= h[:n - o]
        else:
            p = -o
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] -= h[p:]
    return grad / float(k_top)

def softmax_obj_fn(x, n, T):
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def softmax_jac_fn(x, n, T):
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[:n - o] += w * (1.0 - x[o:])
            grad[o:] -= w * x[:n - o]
        else:
            p = -o
            grad[p:] += w * (1.0 - x[:n - p])
            grad[:n - p] -= w * x[p:]
    return grad * (2.0 / n)

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def pg_search(h, target, dx, n_iter=5000, t0=None, budget=None, k_top=12):

    best_c5 = compute_c5(h_curr)

            stall = 0

        h_try = h_curr - alpha * grad

        if c5_try < best_c5 - 1e-15:

            stall = 0
            alpha = min(alpha * 1.08, 2.5)
        else:
            stall += 1
    return h_curr, best_c5
