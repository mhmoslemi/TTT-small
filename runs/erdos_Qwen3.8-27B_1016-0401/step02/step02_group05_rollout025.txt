            grad[p:] += dx * (1.0 - h_curr[:n - p])
            grad[:n - p] += -dx * h_curr[p:]
        else:
            pp = -s
            grad[:n - pp] += dx * (1.0 - h_curr[pp:])
            grad[pp:] += -dx * h_curr[:n - pp]
        
        # Projected gradient with momentum
        v = mu * v - lr * grad
        h_new = h_curr + v
        h_new = constrained_project(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
            lr *= 1.01  # slight increase
        else:
            h_curr = h_new
            lr *= 0.95
            v *= 0.5
        
        if lr < 1e-8:
            break
    
    return h_curr, best_c5

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def topm_val(x, m):
    n = len(x)
    dx = 2.0 / n
    corr = get_corr(x)
    idx = np.argpartition(corr, -m)[-m:]
    return corr[idx].mean() * dx

def topm_grad(x, m):
    n = len(x)
    dx = 2.0 / n
    corr = get_corr(x)
    idx = np.argpartition(corr, -m)[-m:]
    mask = np.zeros(len(corr), dtype=bool)
    mask[idx] = True
    center = n - 1
    grad = np.zeros(n)
    w = dx / m
    for o in range(-(n - 1), n):
        if not mask[center + o]:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fix_sum(h, target):
    return project_sum(h, target)

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def enhanced_2opt(h, target, n_iter=20000, t0=None, budget=None):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    best_c5 = float(np.max(corr) * dx)
    h_curr = h.copy()
    delta = 0.15
    stall = 0
    best_corr = corr.copy()

    for it in range(n_iter):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        if stall > 500:
            delta *= 0.65
            if delta < 0.005:
                break
            stall = 0

        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        best_move_c5 = best_c5
        best_move_h = None
        best_move_corr = None

        candidates = []
        if p > 0:
            if s > 0:
                contrib = h_curr[p:] * (1.0 - h_curr[:n - p])
            else:
                contrib = h_curr[:n - p] * (1.0 - h_curr[p:])
            top_k = min(8, len(contrib))
            top_idx = np.argpartition(contrib, -top_k)[-top_k:]
            for ti in top_idx:
                if s > 0:
                    candidates.append((int(ti) + p, int(ti)))
                else:
                    candidates.append((int(ti), int(ti) + p))
            # Also try second-worst shift
            k2 = np.argsort(corr)[-2]
            if k2[0] != k_star:
                s2 = int(k2[0]) - (n - 1)
                p2 = abs(s2)
                if p2 > 0:
                    if s2 > 0:
                        c2 = h_curr[p2:] * (1.0 - h_curr[:n - p2])
                    else:
                        c2 = h_curr[:n - p2] * (1.0 - h_curr[p2:])
                    cidx = int(np.argmax(c2))
                    if s2 > 0:
                        candidates.append((cidx + p2, cidx))
                    else:
                        candidates.append((cidx, cidx + p2))
        else:
            vals = h_curr * (1.0 - h_curr)
            top_k = min(8, n)
            top_idx = np.argpartition(vals, -top_k)[-top_k:]
            for ti in top_idx:

        for (a, b) in candidates:

                        best_move_h = h_try
                        best_move_corr = ct
            else:

                    h_try = h_curr.copy()

                    h_try[b] += t
                    h_try = fix_sum(h_try, target)

                    if c5t < best_move_c5:

                        best_move_h = h_try
                        best_move_corr = ct
                # Swap
                if h_curr[a] != h_curr[b]:
                    h_try = h_curr.copy()
                    h_try[a], h_try[b] = h_curr[b], h_curr[a]
                    ct = get_corr(h_try)

                    h_try = h_curr.copy()
                    h_try[a] += t
                    h_try[b] -= t
                    h_try = fix_sum(h_try, target)
                    ct = get_corr(h_try)
                    c5t = float(np.max(ct) * dx)
                    if c5t < best_move_c5:
                        best_move_c5 = c5t
                        best_move_h = h_try
                        best_move_corr = ct

        if best_move_h is not None:
            h_curr = best_move_h
            corr = best_move_corr
            best_c5 = best_move_c5
            stall = 0
        else:
            stall += 1

    return h_curr, best_c5

def subgrad_polish(h, target, n_iter=5000, t0=None, budget=None):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    lr = 0.02 / n
    nu = 0.85
    v = np.zeros(n)

    for it in range(n_iter):
        if t0 is not None and budget is not None and time.time() - t0 > budget:
            break
        corr = get_corr(h_curr)
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        grad = np.zeros(n)
        if p == 0:
            grad = dx * (1.0 - 2.0 * h_curr)
        elif s > 0:
            grad[p:] += dx * (1.0 - h_curr[:n - p])
            grad[:n - p] -= dx * h_curr[p:]
        else:
            pp = -s
            grad[:n - pp] += dx * (1.0 - h_curr[pp:])
            grad[pp:] -= dx * h_curr[:n - pp]
        v = nu * v - lr * grad
        h_new = project_sum(h_curr + v, target)
        c5_new = compute_c5(h_new)
        h_curr = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 0.1 / n)
        else:
            lr *= 0.93
            if lr < 1e-9:
                break
    return best_h, best_c5
