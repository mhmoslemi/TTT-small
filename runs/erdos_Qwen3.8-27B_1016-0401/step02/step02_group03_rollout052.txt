import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def _sm_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    e = np.exp((corr - mc) * T)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _sm_jac(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    e = np.exp((corr - mc) * T)
    w = e / e.sum()
    center = n - 1
    g = np.zeros(n)
    dx = 2.0 / n
    for oi in range(len(w)):
        if w[oi] < 1e-10:
            continue
        g += w[oi] * shift_grad(x, n, oi - center)
    return g * dx

def _sum_cons_fun(x, target):
    return x.sum() - target

def _sum_cons_jac(x):
    return np.ones_like(x)

def do_slsqp(h, n, T, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(_sm_obj, h, args=(n, T), method='SLSQP',
                   jac=lambda x: _sm_jac(x, n, T),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or (hasattr(res, 'x') and res.x is not None):
        return res.x
    return h.copy()

def do_lbfgsb(h, n, T, pen, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        e = np.exp((corr - mc) * T)
        sm = (mc + np.log(e.sum()) / T) * (2.0 / n)
        s = x.sum() - target
        return sm + pen * s * s
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        e = np.exp((corr - mc) * T)
        w = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        dx = 2.0 / n
        for oi in range(len(w)):
            if w[oi] < 1e-10:
                continue
            g += w[oi] * shift_grad(x, n, oi - center)
        g *= dx
        s = x.sum() - target
        g += 2.0 * pen * s * np.ones(n)
        return g
    res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def nesterov_topk(h_init, n, t0, budget_s, K, max_iters=200000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    f_best = c5_best
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        # Top-K indices
        k_idx = np.argpartition(corr, -K)[-K:]
        g = np.zeros(n)
        for oi in k_idx:
            g += shift_grad(h_y, n, oi - center)
        g *= dx / K
        # Adaptive step: alpha = f_best / ||g||^2
        gn = np.dot(g, g)
        if gn < 1e-16:
            break
        alpha = min(0.5 / (it + 1) * 50.0, f_best * 2.0 / gn)
        alpha = max(alpha, 1e-4)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 5000 == 0:
            c5a = compute_c5(h_sg)
            if c5a < c5_best:
                c5_best = c5a
                h_best = h_sg.copy()
    c5f = compute_c5(h_sg)
    if c5f < c5_best:
        c5_best = c5f
        h_best = h_sg.copy()
    return h_best, c5_best

_G_T = 1.0
_G_PEN = 0.0
_G_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def _obj_func(x):
    n = _G_N
    T = _G_T

                       bounds=bounds, constraints=cons,

def nesterov_topk(h_init, n, t0, remaining_s, K, max_iters=150000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
