    dx = 2.0 / n

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

                h[a] = 1.0
                h[b] = 0.0

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    return obj, e / se

    hb[idx[:target]] = 1.0
    return hb

def sa_search(h, rng, deadline, T0, n_iter, T_min=0.00003):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T0
    cool = (T_min / T0) ** (1.0 / max(n_iter, 2))
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i1 = rng.integers(n)
        i0 = rng.integers(n)
        ok = False
        for _ in range(12):
            if h[i1] >= 0.5 and h[i0] < 0.5 and i1 != i0:
                ok = True
                break
            i1 = rng.integers(n)
            i0 = rng.integers(n)
        if not ok:
            continue
        h[i1] = 0.0
        h[i0] = 1.0
        c5n = compute_c5(h)
        d = c5n - best_c5
        if c5n <= best_c5:
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        elif T > T_min and d < 5.0 * T:
            if rng.random() > np.exp(-d / T):
                h[i1] = 1.0
                h[i0] = 0.0
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        T *= cool
    fc = compute_c5(h)
    if fc < best_c5:
        best_c5 = fc
        best_h = h.copy()
    return best_h, best_c5

def steepest(h, deadline, max_passes=8):
    n = len(h)
    bc5 = compute_c5(h)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        bd = 0.0
        bp = None
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for a in ones:
            if time.time() > deadline:
                break
            for b in zeros:
                if a == b:
                    continue
                h[a] = 0.0
                h[b] = 1.0
                cx = compute_c5(h)
                if bc5 - cx > bd:
                    bd = bc5 - cx
                    bp = (a, b)
                h[a] = 1.0
                h[b] = 0.0
        if bp is None:
            break
        h[bp[0]] = 0.0
        h[bp[1]] = 1.0
        bc5 = compute_c5(h)
    return h, bc5

def softmin_vals(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    return obj, e / se

def softmin_grad(h, sw):
    n = len(h)
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] -= w * h[k:]
        else:
            p = -k
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * (2.0 / n)

def pg_refine(h, target, T, max_iter, deadline, lr0=0.02):
    n = len(h)
    bh = h.copy()
    bc5 = compute_c5(h)
    h = h.copy()
    lr = lr0
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        _, sw = softmin_vals(h, T)
        g = softmin_grad(h, sw)
        hn = project_constraint(h - lr * g, target)
        cn = compute_c5(hn)
        if cn < bc5 - 1e-14:
            h = hn
            bc5 = cn
            bh = hn.copy()
            lr = min(lr * 1.05, lr0)
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if stall > 50 or lr < 1e-7:
                break
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = budget_s
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        ho = np.array(initial_h_values, dtype=float)
    else:
        ho = np.zeros(560)
        ho[::2] = 1.0
    n = len(ho)
    tgt = n / 2.0
    ho = project_constraint(ho, tgt)

    # Phase 1: Initial SA from parent (0-18%)
    hb = make_balanced_binary(ho, rng)
    best_h, best_c5 = sa_search(hb, rng, t0 + B * 0.18, 0.005, 3000000)

    # Phase 2: ILS restarts (18-45%)
    for r in range(20):
        if time.time() > t0 + B * 0.45:
            break
        hp = best_h.copy()
        nflip = int(rng.integers(2 + r % 4, 5 + r % 6))
        for _ in range(nflip):
            o = np.where(hp >= 0.5)[0]
            z = np.where(hp < 0.5)[0]
            if len(o) == 0 or len(z) == 0:
                break
            hp[o[rng.integers(len(o))]] = 0.0
            hp[z[rng.integers(len(z))]] = 1.0
        dl = min(t0 + B * 0.45, time.time() + B * 0.018)
        T_r = 0.002 + 0.001 * (1 - r / 20.0)
        hp, c5r = sa_search(hp, rng, dl, T_r, 600000)
        if c5r < best_c5:
            best_c5, best_h = c5r, hp.copy()

    # Phase 3: Steepest descent + SA polish (45-58%)
    if time.time() < t0 + B * 0.58:
        h3 = make_balanced_binary(best_h, rng)
        h3, c53 = steepest(h3, t0 + B * 0.53, 8)
        if c53 < best_c5:
            best_c5, best_h = c53, h3
        if time.time() < t0 + B * 0.58:
            h3, c53b = sa_search(h3, rng, t0 + B * 0.58, 0.001, 500000, T_min=0.00002)
            if c53b < best_c5:
                best_c5, best_h = c53b, h3

    # Phase 4: Fractional PG (58-75%)
    if time.time() < t0 + B * 0.75:
        hf = best_h + rng.normal(0, 0.06, n)
        hf = project_constraint(hf, tgt)
        for T in [12, 30, 60, 120, 250]:
            if time.time() > t0 + B * 0.75:
                break
            hf, cf = pg_refine(hf, tgt, T, 400, t0 + B * 0.75, 0.018)
            if cf < best_c5:
                best_c5, best_h = cf, hf.copy()
            hf = best_h + rng.normal(0, 0.02, n)
            hf = project_constraint(hf, tgt)

    # Phase 5: n=280 exploration (75-88%)
    if time.time() < t0 + B * 0.88:
        ns = 280
        ts = ns / 2.0
        xb = np.arange(n) * (2.0 / n)
        xs = np.linspace(0.0, 2.0, ns, endpoint=False)
        hs = np.interp(xs, xb, best_h)
        hsb = make_balanced_binary(hs, rng)
        hsb, c5s = sa_search(hsb, rng, t0 + B * 0.85, 0.004, 1200000)
        hsb, c5s2 = steepest(hsb, t0 + B * 0.88, 4)
        if c5s2 < best_c5:
            best_c5, best_h = c5s2, hsb

    # Phase 6: Final micro-refinement (88-99%)
    if time.time() < t0 + B - 10:
        nf = len(best_h)
        tf = nf / 2.0
        ht = best_h.copy()
        for T in [40, 80]:
            if time.time() > t0 + B - 5:
                break
            ht, ct = pg_refine(ht, tf, T, 300, t0 + B - 5, 0.012)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    cf = min(compute_c5(best_h), best_c5)
    return (best_h, cf, len(best_h))
