    corr = np.correlate(h, 1 - h, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = float(np.sum(e))
    obj = (mc + np.log(se) / T) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_compute(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = float(np.sum(e))
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def active_weighted_grad(h, m, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    ov = corr * dx
    top_idx = np.argsort(ov)[::-1][:m]
    tau = ov[top_idx[-1]]
    wts = np.maximum(ov - tau + 1e-6, 0.0)
    sh = T * wts
    sh -= sh.max()
    e = np.exp(sh)
    se = float(e.sum())
    if se < 1e-30:
        sw = np.zeros(len(corr))
        sw[top_idx[0]] = 1.0
    else:
        sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])

        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
