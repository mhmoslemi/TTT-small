**Key idea: Active shift identification**

The C₅ is the maximum over all shifts. At the optimum, typically 2-5 shifts are "active" (close to the max). If I can identify these active shifts and specifically try to reduce the overlap at those shifts, I can make more targeted progress.

Algorithm:
1. Compute C₅ and identify the top-3 shifts
2. For each active shift k, the overlap is ∫h(x)(1-h(x+k))dx
3. To reduce this, I need to either decrease h where h(x+k)=0, or increase h where h(x+k)=1
4. This gives a specific direction in the h-space

```python
def targeted_descent(h, n, t0, deadline, n_iters=5000):
    """Reduce C5 by focusing on active shifts."""
    c5_cur = compute_c5(h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        # Find top 3 shifts
        top_idx = np.argsort(corr)[-3:]
        # For each active shift, compute gradient
        center = n - 1
        grad = np.zeros(n)
        for ti in top_idx:
            k = ti - center
            if k == 0:
                grad += (1.0 - 2.0 * h)
            elif k > 0:
                g = np.zeros(n)
                g[k:] += 1.0 - h[:n-k]
                g[:n-k] -= h[k:]
                grad += g
            else:
                s = -k
                g = np.zeros(n)
                g[:n-s] += 1.0 - h[s:]
                g[s:] -= h[:n-s]
                grad += g
        grad *= dx / 3.0  # Average over 3 shifts
        # Gradient descent with projection
        alpha = 0.1 / (1.0 + it * 0.001)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
        else:
            alpha *= 0.5  # Reduce step
```

Actually, I want to also add one more idea: **alternating optimization**. 

The overlap at shift k is ∫h(x)(1-h(x+k))dx. If I fix all h values except position i, the overlap becomes a linear function of h[i]. So I can do coordinate descent: for each i, optimize h[i] given all others fixed.

For a fixed h[-i], the overlap at shift k is:
  ∫_{x≠i} h(x)(1-h(x+k))dx + h[i](1-h[i+k]) - h[i](1-h[i+k]) + ... 

Actually, it's more complex because h[i] appears in multiple terms (as h(x) and as h(x+k) for different k). But the total C₅ is max over k, so it's the max of multiple linear functions of h[i]. To minimize the max, I can use a bisection: find the h[i] that minimizes the maximum.

This is a 1D optimization per coordinate. With n=280 coordinates, a full sweep takes 280 1D optimizations. If each takes ~50 evaluations, that's 14K evaluations per sweep. Each evaluation is O(n) for the correlation. So 14K * 280 ≈ 4M per sweep. A few sweeps in a few seconds.

Actually, the 1D optimization for coordinate i is:
  min_{h[i] ∈ [0,1]} max_k f_k(h[i])
where f_k is the overlap at shift k, which is linear in h[i].

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_T = 1.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _obj(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _g_T * (corr - mc)
    msh = np.max(sh)
    e = np.exp(sh - msh)
    val = (mc + (msh + np.log(e.sum())) / _g_T) * dx
    val += 500.0 * (x.sum() - _g_target) ** 2
    return val

def _grad(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _g_T * (corr - mc)
    msh = np.max(sh)
    e = np.exp(sh - msh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-14:
            continue
        k = oi - center
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            tmp = np.zeros(n)
            tmp[k:] += 1.0 - x[:n - k]
            tmp[:n - k] -= x[k:]
            grad += w * tmp
        else:
            s = -k
            tmp = np.zeros(n)
            tmp[:n - s] += 1.0 - x[s:]
            tmp[s:] -= x[:n - s]
            grad += w * tmp
    grad *= dx
    grad += 1000.0 * (x.sum() - _g_target) / n
    return grad

def _cfn(x):
    return x.sum() - _g_target

def _cjc(x):
    return np.ones(_g_n)

def _setup(n, T):
    global _g_n, _g_T, _g_target
    _g_n = n
    _g_T = float(T)
    _g_target = n / 2.0

def do_lgbfsm(h, n, T, maxiter, gtol=1e-12):
    _setup(n, T)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='L-BFGS-B', jac=_grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': gtol, 'maxfun': max(1000, maxiter * 4)})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, maxiter):
    _setup(n, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfn, 'jac': _cjc}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def lgbfsm_sweep(h, n, T_list, mi, t0, deadline):
    for T in T_list:
        if time.time() > deadline:
            break
        h = do_lgbfsm(h, n, T, mi)
        h = project_feasible(h, n)
    return h

def targeted_descent(h, n, t0, deadline, niters=3000):
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    center = n - 1
    alpha = 0.06
    for it in range(niters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-3:]
        grad = np.zeros(n)
        for ti in top:
            k = ti - center
            if k == 0:
                grad += 1.0 - 2.0 * h
            elif k > 0:
                tmp = np.zeros(n)
                tmp[k:] += 1.0 - h[:n - k]
                tmp[:n - k] -= h[k:]
                grad += tmp
            else:
                s = -k
                tmp = np.zeros(n)
                tmp[:n - s] += 1.0 - h[s:]
                tmp[s:] -= h[:n - s]
                grad += tmp
        grad *= dx / len(top)
        a = alpha / (1.0 + 0.002 * it)
        h_new = project_feasible(h - a * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h, c5_cur = h_new, c5_new
        else:
            alpha *= 0.65
    return h, c5_cur

def make_pert(h, n, strat, rng):
    if strat == 0:
        return h + rng.normal(0.0, 0.012, n)
    if strat == 1:
        hp = h + rng.normal(0.0, 0.03, n)
        k9 = np.ones(9) / 9.0
        return np.convolve(hp, k9, mode='same')
    if strat == 2:
        hp = h.copy()
        bs = int(rng.integers(5, 25))
        b0 = int(rng.integers(0, max(1, n - bs)))
        hp[b0:b0 + bs] = 1.0 - hp[b0:b0 + bs]
        return hp
    if strat == 3:
        hp = h.copy()
        f = int(rng.integers(1, n // 3 + 1))
        p = rng.uniform(0.0, 2.0 * np.pi)
        hp += rng.uniform(0.015, 0.06) * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
        return hp
    hp = h.copy()
    ctr = int(rng.integers(0, n))
    wid = float(int(rng.integers(8, 35)))
    amp = rng.uniform(0.02, 0.08) * rng.choice([-1.0, 1.0])
    d = np.minimum(np.abs(np.arange(n) - ctr), n - np.abs(np.arange(n) - ctr)).astype(float)
    hp += amp * np.exp(-d * d / (2.0 * wid * wid))
    return hp

def opt_chain(h, n, t0, deadline):
    h = project_feasible(h, n)
    if time.time() > deadline:
        return h
    h = project_feasible(do_lgbfsm(h, n, 16, 200), n)
    if time.time() > deadline:
        return h
    h = project_feasible(do_lgbfsm(h, n, 2, 400), n)
    if time.time() > deadline:
        return h
    h = project_feasible(do_lgbfsm(h, n, 0.25, 400, gtol=1e-14), n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
```
