import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    # Binary search for threshold that gives sum = target after clipping
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - mid, 0.0, 1.0)
    # Fine-tune: adjust to hit target exactly
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference to elements not at boundary
        free = (result > 1e-10) & (result < 1 - 1e-10)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def shift_grad_raw(h, n, shift):
    """Gradient of corr[shift] with respect to h."""
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(np.clip(sh, -500, 0))
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(np.clip(sh, -500, 0))
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_subgradient(h_init, n, t0, budget_s, max_iters=500000):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best_local = compute_c5(h_init)
    h_best_local = h_init.copy()
    # Adaptive step size
    f_star_est = c5_best_local - 0.02  # rough lower bound estimate
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad_raw(h_y, n, shift) * dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        # Polyak-like step
        gap = mc * dx - f_star_est
        if gap < 1e-8:
            alpha = 0.01
        else:
            alpha = min(0.5, gap / (g_norm * g_norm + 1e-10))
        alpha = max(alpha, 0.001)
        h_new = h_y - alpha * g
        # Project
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 5000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best_local:
                c5_best_local = c5c
                h_best_local = h_sg.copy()
            # Update f_star estimate
            f_star_est = max(c5_best_local - 0.01, 0.1)
    c5c = compute_c5(h_sg)
    if c5c < c5_best_local:
        c5_best_local = c5c
        h_best_local = h_sg.copy()
    return h_best_local, c5_best_local

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - mid, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-10) & (result < 1 - 1e-10)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(np.clip(sh, -500, 0))
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(np.clip(sh, -500, 0))
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_sub(h_init, n, t0, budget_s, max_iters=600000):
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    f_lo = c5_best - 0.05
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad_raw(h_y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        gap = mc * dx - f_lo
        if gap < 1e-7:
            alpha = 0.005 / (1.0 + 0.002 * it)
        else:
            alpha = min(0.8, gap / (gn * gn + 1e-10))
            alpha = max(alpha, 0.003)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 3000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
            f_lo = max(c5_best - 0.02, 0.15)
    c5c = compute_c5(h_sg)
    if c5c < c5_best:
        c5_best = c5c
        h_best = h_sg.copy()
    return h_best, c5_best

def targeted_perturb(h, n, rng, sigma):
    """Perturb near active shifts more aggressively."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    active = corr > 0.95 * np.max(corr)
    center = n - 1
    h_p = h + rng.normal(0, sigma, n)
    # Extra perturbation near active regions
    for k in range(len(active)):
        if not active[k]:
            continue
        s = k - center
        if 0 <= s < n:
            start = max(0, s - 3)
            end = min(n, s + n - 2)
            h_p[start:end] += rng.normal(0, sigma * 2, end - start)
        elif s < 0:
            start = max(0, -s - 3)
            end = min(n, n + s + 2)
            h_p[start:end] += rng.normal(0, sigma * 2, end - start)
    return project_feasible(np.clip(h_p, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    for T in [64, 16, 4, 1]:
        if time.time() - t0 > budget_s - 350:
            break
        h_new = do_slsqp(h_best, n, T, 5000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov subgradient (long run)
    rem = budget_s - (time.time() - t0)
    h_sg, c5_sg = nesterov_sub(h_best, n, t0, budget_s - 200, 500000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP fine polish
    for T in [1, 2, 4, 8, 2]:
        if time.time() - t0 > budget_s - 100:
            break
        h_new = do_slsqp(h_best, n, T, 8000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Targeted perturbation restarts
    for pi in range(60):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.004 + 0.012 * (pi % 6)
        h_p = targeted_perturb(h_best, n, rng, sigma)
        h_new = do_slsqp(h_p, n, 4, 3000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try n*2 resolution
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        for T in [16, 4, 1, 1]:
            if time.time() - t0 > budget_s - 30:
                break
            h_hi = do_slsqp(h_hi, n2, T, 4000, 1e-17)
            h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n2

    # Phase 6: Final Nesterov + SLSQP
    if time.time() - t0 < budget_s - 15:
        h_sg2, c5_sg2 = nesterov_sub(h_best, n, t0, budget_s - 8, 100000)
        if c5_sg2 < c5_best:
            c5_best = c5_sg2
            h_best = h_sg2
        h_new = do_slsqp(h_best, n, 1, 5000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
