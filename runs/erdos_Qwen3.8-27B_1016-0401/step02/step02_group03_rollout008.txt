        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i] += delta
            h[j] -= delta
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Smooth max SLSQP with decreasing temperature
    for T in [256, 128, 64, 32, 16, 8, 4, 2, 1]:
        if time.time() > deadline - 350:
            break
        h_new = slsqp_smooth(h_best, n, T, 4000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Targeted single-shift SLSQP on worst shifts
    for round in range(3):
        if time.time() > deadline - 300:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        for oi in top_idx:
            shift = int(oi) - (n - 1)
            h_new = slsqp_single_shift(h_best, n, shift, 2000)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 3: Nesterov subgradient with stall detection
    h_nest, c5_nest = nesterov_targeted(h_best, n, t0, deadline - 250, 500000)
    if c5_nest < c5_best:
        c5_best = c5_nest
        h_best = h_nest
    else:
        h_best = project_feasible(h_best, n)

    # Phase 4: SLSQP fine polish
    for T in [1, 2, 4, 8, 1]:
        if time.time() > deadline - 120:
            break
        h_new = slsqp_smooth(h_best, n, T, 5000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Pair-swap local search
    h_best, c5_swap = pair_swap_local(h_best, n, 8000, t0, deadline - 80)
    if c5_swap < c5_best:
        c5_best = c5_swap
        h_best = h_best.copy()

    # Phase 6: Perturbation + SLSQP restarts (diverse)
    n_trials = 30
    for pi in range(n_trials):
        if time.time() > deadline - 20:
            break
        mode = pi % 4
        if mode == 0:
            sigma = 0.005 + 0.02 * (pi // 4)
            h_p = h_best + rng.normal(0, sigma, n)
        elif mode == 1:
            # Shift perturbation
            sh = rng.integers(-3, 4)
            h_p = np.roll(h_best, sh)
            h_p = np.clip(h_p, 0.0, 1.0)
        elif mode == 2:
            # Local flip
            k = rng.integers(0, n)
            w = rng.integers(5, 20)
            h_p = h_best.copy()
            sl = slice(max(0, k - w), min(n, k + w))
            h_p[sl] = 1.0 - h_p[sl]
        else:
            # Random restart with structure
            h_p = np.zeros(n)
            h_p[rng.choice(n, n // 2, replace=False)] = 1.0
        h_p = project_feasible(h_p, n)
        T_use = 4 if mode != 3 else 16
        h_new = slsqp_smooth(h_p, n, T_use, 2500, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 7: Try n*2 resolution if time permits
    n2 = n * 2
    if time.time() < deadline - 200:
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n2
        for T in [16, 4, 1]:
            if time.time() > deadline - 60:
                break
            h_new = slsqp_smooth(h_best, n, T, 3000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 8: Final pair-swap + SLSQP polish
    if time.time() < deadline - 10:
        h_best, c5_fin = pair_swap_local(h_best, n, 3000, t0, deadline - 5)
        if c5_fin < c5_best:
            c5_best = c5_fin
        h_new = slsqp_smooth(h_best, n, 1, 3000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
