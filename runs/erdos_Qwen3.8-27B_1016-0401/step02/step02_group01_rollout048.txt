        h2 = np.interp(x2, x1, h_best)
        h2 = project_constraint(h2, target2)
        schedule2 = [(30, 50), (100, 55), (300, 60), (600, 60), (1200, 65)]
        for T, K in schedule2:
            if time.time() > budget - 30:
                break
            h2, c5_2 = pg_opt(h2, target2, T, t0, budget - 20, K, 0.035, 10000)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2.copy()

    # Phase 4: Downsample best back to 560 if it went to 840, or stay
    # Actually keep whatever resolution gave the best C5

    # Phase 5: Final high-T polish at best resolution
    nf = len(h_best)
    tf = nf / 2.0
    for T in [800, 1500, 3000]:
        if time.time() > budget - 8:
            break
        h_best, c5 = pg_opt(h_best, tf, T, t0, budget - 3, 60, 0.02, 5000)
        if c5 < c5_best:
            c5_best = c5

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = project_constraint(h_best, len(h_best) / 2.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
