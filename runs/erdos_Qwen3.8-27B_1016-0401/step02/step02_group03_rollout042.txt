    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def subgrad_topk(h, k=5):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -k)[-k:]
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in top_idx:
        shift = oi - center
        if shift >= 0:
            grad[:n-shift] += (1.0 - h[shift:])
            grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    return grad * dx

def coord_descent_step(h, n):
    """One full pass of coordinate descent minimizing max corr."""
    dx = 2.0 / n
    center = n - 1
    corr = compute_corr(h)
    for i in range(n):
        t0 = h[i]
        if t0 < 1e-12 or t0 > 1.0 - 1e-12:
            pass
        # Compute affected correlations as functions of t = h[i]
        # First-factor: corr[center+k] for k=0..n-1-i, slope = (1-h[i+k])
        # Second-factor: corr[center-k] for k=0..i, slope = -h[i-k]
        # Base values (with t0): we already have corr
        # New corr = corr + (t - t0) * slope
        k_range_f = np.arange(0, n - i)  # first factor shifts
        k_range_s = np.arange(0, i + 1)  # second factor shifts
        slopes = np.zeros(2 * n - 1)
        slopes[center + k_range_f] += (1.0 - h[i + k_range_f])
        slopes[center - k_range_s] -= h[i - k_range_s]
        base = corr.copy()
        base += (t0 - t0) * slopes  # t0 cancels, base is current corr
        
        def obj_1d(t):
            new_corr = base + (t - t0) * slopes
            return np.max(new_corr) * dx
        
        res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                              options={'maxiter': 80, 'xatol': 1e-10})
        h[i] = res.x
        corr = base + (res.x - t0) * slopes
    return h, corr

def nesterov_loop(h_init, n, t0, remain_s, max_iters=200000):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.3
    for it in range(max_iters):
        if time.time() - t0 > remain_s:
            break
        beta = (float(it) / (it + 2.0)) if it > 2 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = compute_corr(h_y)
        # Use top-3 for sharper gradient
        k = 3 if it < 30000 else 5
        idx = int(np.argmax(corr))
        top_set = np.argpartition(corr, -k)[-k:]
        center = n - 1
        grad = np.zeros(n)
        for oi in top_set:
            shift = oi - center
            if shift >= 0:
                grad[:n-shift] += (1.0 - h_y[shift:])
                grad[shift:] -= h_y[:n-shift]
            else:
                s = -shift
                grad[:n-s] += (1.0 - h_y[s:])
                grad[s:] -= h_y[:n-s]
        grad *= dx
        alpha = alpha0 / (1.0 + 0.005 * it)
        h_new = h_y - alpha * grad
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 5000 == 0:
            c5_now = compute_c5(h_sg)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_sg.copy()
    c5_final = compute_c5(h_sg)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h_sg.copy()
    return h_best, c5_best

def slsqp_smooth(h, n, T, maxiter=2000):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -500, 0)
        return (mc + np.log(np.exp(sh).sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -500, 0)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-12:
                continue
            shift = oi - center
            if shift >= 0:
                grad[:n-shift] += w * (1.0 - x[shift:])
                grad[shift:] -= w * x[:n-shift]
            else:
                s = -shift
                grad[:n-s] += w * (1.0 - x[s:])
                grad[s:] -= w * x[:n-s]
        return grad * dx
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coordinate descent sweeps (key new method)
    for sweep in range(30):
        if time.time() - t0 > budget_s - 400:
            break
        h_cd, _ = coord_descent_step(h_best, n)
        h_cd = project_feasible(h_cd, n)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 2: Nesterov subgradient (top-k)
    if time.time() - t0 < budget_s - 250:
        h_ng, c5_ng = nesterov_loop(h_best, n, t0, budget_s - 300)
        if c5_ng < c5_best:
            c5_best = c5_ng
            h_best = h_ng

    # Phase 3: More coordinate descent
    for sweep in range(40):
        if time.time() - t0 > budget_s - 150:
            break
        h_cd, _ = coord_descent_step(h_best, n)
        h_cd = project_feasible(h_cd, n)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 4: SLSQP smooth polish
    for T in [2.0, 1.0, 0.5, 1.0, 2.0]:
        if time.time() - t0 > budget_s - 60:
            break
        h_new = slsqp_smooth(h_best, n, T, 2000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Tiny perturbations + coord descent
    for pi in range(60):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.003 + 0.004 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_cd, _ = coord_descent_step(h_p, n)
        h_cd = project_feasible(h_cd, n)
        c5 = compute_c5(h_cd)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_cd

    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s - 10:
        h_new = slsqp_smooth(h_best, n, 1.0, 3000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
