import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def topK_grad(h, K):
    """Gradient using only top-K correlations"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    # Find top K lags
    top_idx = np.argsort(corr)[::-1][:K]
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] -= h[o:]
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    return grad * dx / K

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgsb_with_smoothing(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def momentum_topK(h0, target, K, steps, lr_init, decay):
    """Momentum-based top-K subgradient descent"""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.9
    lr = lr_init
    for it in range(steps):
        g = topK_grad(h, K)
        v = mu * v + (1 - mu) * g
        h_new = project_box_sum(h - lr * v, target)
        if np.linalg.norm(h_new - h) < 1e-14:
            break
        h = h_new
        lr *= decay
    return h

    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    return np.correlate(h, 1 - h, mode="full")

    n = len(h)
    corr = sm_corr(h)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Very low T L-BFGS-B finishing
    for T in [3, 2, 1, 0.5]:
        if time.time() - t0 > budget_s - 170:
            break
        h_new = lbfgsb_opt(h_best, n, target, T, 1.0, 4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Upsample to 2n and re-optimize
    if time.time() - t0 < budget_s - 220:
        n2 = 2 * n
        t2 = n2 / 2.0
        xo = np.arange(n) * (2.0 / n)
        xn = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_try = np.interp(xn, xo, h_best)
        h_try = project_box_sum(h_try, t2)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best + 0.005:
            for T in [50, 20, 10, 5, 3, 2, 1]:
                if time.time() - t0 > budget_s - 100:
                    break
                h_t = lbfgsb_opt(h_try, n2, t2, T, 5.0, 2000)
                c5_t = compute_c5(h_t)
                if c5_t < c5_try:
                    c5_try = c5_t
                    h_try = h_t
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try
                n = n2
                target = t2

    # Phase 5: Diverse random restarts
    cnt = 0
    while time.time() - t0 < budget_s - 55:
        cnt += 1
        sc = max(0.015, 0.07 / (1.0 + 0.02 * cnt))
        hp = make_perturbation(h_best, rng, sc, cnt)
        hp = project_box_sum(hp, target)
        T = 10 + int(rng.integers(0, 50))
        hn = lbfgsb_opt(hp, n, target, T, 5.0, 800)
        hn = project_box_sum(hn, target)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    # Phase 6: Final ultra-low-T polish
    for T in [2, 1, 0.5]:
        if time.time() - t0 > budget_s - 3:
            break
        hn = lbfgsb_opt(h_best, n, target, T, 1.0, 3000)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
