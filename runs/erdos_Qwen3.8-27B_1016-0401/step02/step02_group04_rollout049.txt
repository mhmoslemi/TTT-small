def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def topk_grad(h, T, k_top):
    """Gradient of soft-top-k objective. Focus on top k lags."""
    n = len(h)
    corr = get_corr(h)

    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n

    """One step: compute gradient from top-k lags only, descend."""
    n = len(h)
    corr = get_corr(h)
    # Get top-k lag indices
    idx = np.argsort(corr)[-top_k:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / top_k

    dx = 2.0 / n

        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def topk_grad(h, top_k=5):
    n = len(h)
    corr = get_corr(h)
    idx = np.argsort(corr)[-top_k:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / top_k
    c = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        wi = w[c + o]
        if wi < 1e-15:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    tk = 1.0
    step = step_init
    for it in range(max_iter):
        y = h + ((tk - 1.0) / (tk + 2.0)) * (h - zh)
        g = softmax_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        d = np.linalg.norm(zh - h)
        if d < 1e-14:
            break
        if it > 200 and d < 1e-12:
            step *= 0.95
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        h = zh
        tk = tk_new
    return h

def greedy_descent(h0, target, max_iter, top_k, lr0):
    n = len(h0)
    h = h0.copy()
    h_by = h0.copy()
    c5_by = compute_c5(h0)
    lr = lr0
    for it in range(max_iter):
        g = topk_grad(h, top_k)
        h_new = project_box_sum(h - lr * g, target)
        c5 = compute_c5(h_new)
        if c5 < c5_by:
            h = h_new
            h_by = h_new.copy()
            c5_by = c5
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return h_by, c5_by

def lbfgsb_refine(h0, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = get_corr(x)
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / n)
    def grad(x):
        return softmax_grad(x, T)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def discrete_search(h, target, rng, n_moves, c5_cur):
    n = len(h)
    best = h.copy()
    bc5 = c5_cur
    for _ in range(n_moves):
        i = rng.integers(0, n)
        trial = best.copy()
        trial[i] = 1.0 - trial[i]
        trial = project_box_sum(trial, target)
        c = compute_c5(trial)
        if c < bc5:
            best = trial
            bc5 = c
        if i < n - 1 and rng.random() < 0.4:
            t2 = best.copy()
            t2[i:i+2] = 1.0 - t2[i:i+2]
            t2 = project_box_sum(t2, target)
            c2 = compute_c5(t2)
            if c2 < bc5:
                best = t2
                bc5 = c2
    return best, bc5

def round_to_discrete(h, target):
    n = len(h)
    n_ones = int(round(target))
    if n_ones >= n:
        n_ones = n - 1
    idx = np.argsort(-h)[:n_ones]
    h_d = np.zeros(n)
    h_d[idx] = 1.0
    return h_d

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: FISTA temperature annealing
    for T in [30, 60, 100, 180, 300, 500, 800, 1200]:
        if elapsed() > budget_s - 350:
            break
        step = min(0.5, 6.0 / (T + 3.0))
        h_new = fista(h_best, target, T, 3500, step)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if elapsed() < budget_s - 330:
            h_l = lbfgsb_refine(h_best, n, target, min(T, 400), 1200)
            c5l = compute_c5(h_l)
            if c5l < c5_best:
                c5_best = c5l
                h_best = h_l

    # Phase 2: Greedy top-k descent (focus on worst lags)
    if elapsed() < budget_s - 280:
        for tk in [3, 5, 8]:
            if elapsed() > budget_s - 280:
                break
            for lr0 in [0.05, 0.02]:
                h_g, c5_g = greedy_descent(h_best, target, 800, tk, lr0)
                if c5_g < c5_best:
                    c5_best = c5_g
                    h_best = h_g

    # Phase 3: Discrete rounding + local search
    if elapsed() < budget_s - 220:
        h_d = round_to_discrete(h_best, target)
        c5_d = compute_c5(h_d)
        if c5_d < c5_best:
            c5_best = c5_d
            h_best = h_d
        h_d, c5_d = discrete_search(h_d, target, rng, 4000, c5_d)
        if c5_d < c5_best:
            c5_best = c5_d
            h_best = h_d

    # Phase 4: Discrete search from continuous solution
    if elapsed() < budget_s - 180:
        h_best, c5_best = discrete_search(h_best, target, rng, 3000, c5_best)

    # Phase 5: Multi-strategy perturbation + re-optimize
    attempts = 0
    while elapsed() < budget_s - 160:
        attempts += 1
        strat = attempts % 4
        if strat == 0:
            sc = max(0.015, 0.08 / (1.0 + 0.03 * attempts))
            h_p = h_best + rng.normal(0, sc, n)
        elif strat == 1:
            h_p = h_best.copy()
            bl = int(rng.integers(4, 15))
            bs = int(rng.integers(0, n - bl))
            h_p[bs:bs + bl] = 1.0 - h_p[bs:bs + bl]
        elif strat == 2:
            h_p = np.roll(h_best, int(rng.integers(1, 4)))
        else:
            h_p = round_to_discrete(h_best + rng.normal(0, 0.05, n), target)
        h_p = project_box_sum(h_p, target)
        T = 80 + int(rng.integers(0, 500))
        h_new = fista(h_p, target, T, 2000, min(0.35, 4.0 / (T + 3)))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if elapsed() < budget_s - 160:
            h_g, c5_g = greedy_descent(h_new, target, 400, 5, 0.03)
            if c5_g < c5_best:
                c5_best = c5_g
                h_best = h_g
            h_d, c5_d = discrete_search(h_best, target, rng, 800, c5_best)
            if c5_d < c5_best:
                c5_best = c5_d
                h_best = h_d

    # Phase 6: Resolution changes
    for n_try in [140, 560]:
        if elapsed() > budget_s - 50:
            break
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_t = np.interp(x_new, x_old, h_best)
        h_t = project_box_sum(h_t, t_try)
        bc5_t = compute_c5(h_t)
        for T in [80, 250, 600]:
            if elapsed() > budget_s - 30:
                break
            h_t = fista(h_t, t_try, T, 2000, min(0.3, 4.0 / (T + 3)))
            ct = compute_c5(h_t)
            if ct < bc5_t:
                bc5_t = ct
        if bc5_t < c5_best:
            c5_best = bc5_t
            h_best = h_t
            n = n_try
            target = t_try

    # Phase 7: Final polish
    if elapsed() < budget_s - 8:
        h_new = fista(h_best, target, 1500, 3000, 0.08)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_g, c5_g = greedy_descent(h_best, target, 500, 3, 0.02)
        if c5_g < c5_best:
            c5_best = c5_g
            h_best = h_g
        if elapsed() < budget_s - 4:
            h_best, c5_best = discrete_search(h_best, target, rng, 2000, c5_best)

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
