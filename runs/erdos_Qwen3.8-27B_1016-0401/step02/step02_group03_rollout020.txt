    lo, hi = y.min() - 1.0, y.max() + 1.0

    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr, idx

def shift_grad(h, shift, n):
    """Gradient of corr[shift] w.r.t. h"""
    grad = np.zeros(n)
    if shift >= 0:
        a = np.arange(n - shift)
        grad[a] += (1.0 - h[a + shift])
        b = np.arange(shift, n)
        grad[b] -= h[b - shift]
    else:
        s = -shift
        a = np.arange(s, n)
        grad[a] += (1.0 - h[a - s])
        b = np.arange(0, n - s)
        grad[b] -= h[b + s]
    return grad

    h_prev = h_init.copy()

        h_y = np.clip(h_y, 0.0, 1.0)

            if trial_val <= cur_val + 1e-10 or step < 1e-6:

            step *= 0.5

        h_prev = h_cur
        h_cur = h_new

            if c < best_c5:

                best_h = h_cur.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _smax_val(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    return (m + np.log(e.sum()) / T) * dx

def _smax_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    ct = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-11)[0]
    for oi in active:
        wv = w[oi]
        sh = oi - ct
        if sh >= 0:
            a = np.arange(n - sh)
            grad[a] += wv * (1.0 - h[a + sh])
            if sh > 0:
                b = np.arange(sh, n)
                grad[b] -= wv * h[b - sh]
        else:
            s2 = -sh
            a = np.arange(s2, n)
            grad[a] += wv * (1.0 - h[a - s2])
            b = np.arange(0, n - s2)
            grad[b] -= wv * h[b + s2]
    return grad * dx

def _lbfgs(h0, n, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_val, h0, method='L-BFGS-B', jac=_smax_grad,
                       args=(T, n), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxls': 50})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

def _nesterov(h_init, n, t_start, budget_s):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    it = 0
    while time.time() - t_start < budget_s:
        it += 1
        beta = (it - 1.0) / (it + 1.5)
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        sh = idx - (n - 1)
        grad = np.zeros(n)
        if sh >= 0:
            a = np.arange(n - sh)
            grad[a] += (1.0 - h_y[a + sh])
            if sh > 0:
                b = np.arange(sh, n)
                grad[b] -= h_y[b - sh]
        else:
            s2 = -sh
            a = np.arange(s2, n)
            grad[a] += (1.0 - h_y[a - s2])
            b = np.arange(0, n - s2)
            grad[b] -= h_y[b + s2]
        grad *= dx
        step = 0.25 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - step * grad, n)
        h_prev = h_cur
        h_cur = h_new
        if it % 15000 == 0:
            c = compute_c5(h_cur)
            if c < best_c5:
                best_c5 = c
                best_h = h_cur.copy()
    c = compute_c5(h_cur)
    if c < best_c5:
        best_c5 = c
        best_h = h_cur.copy()
    return best_h, best_c5

def _topk_smax_grad(h, T, n, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    idx = np.argpartition(-corr, K)[:K]
    c_k = corr[idx]
    e = np.exp(T * (c_k - c_k.max()))
    w = e / e.sum()
    ct = n - 1
    grad = np.zeros(n)
    for ii in range(K):
        oi = idx[ii]
        wv = w[ii]
        sh = oi - ct
        if sh >= 0:
            a = np.arange(n - sh)
            grad[a] += wv * (1.0 - h[a + sh])
            if sh > 0:
                b = np.arange(sh, n)
                grad[b] -= wv * h[b - sh]
        else:
            s2 = -sh
            a = np.arange(s2, n)
            grad[a] += wv * (1.0 - h[a - s2])
            b = np.arange(0, n - s2)
            grad[b] -= wv * h[b + s2]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine
    for T in [20, 10, 5, 3, 2, 1, 0.5, 0.3]:
        if time.time() - t0 > budget_s - 350:
            break
        h_new = _lbfgs(h_best, n, T, 4000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov subgradient fine-tuning
    h_ng, c5_ng = _nesterov(h_best, n, t0, budget_s - 250)
    if c5_ng < c5_best:
        c5_best = c5_ng
        h_best = h_ng

    # Phase 3: L-BFGS-B polish (low temperature, many iters)
    for T in [0.5, 0.3, 0.2]:
        if time.time() - t0 > budget_s - 80:
            break
        h_new = _lbfgs(h_best, n, T, 6000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Targeted perturbation restarts (small, diverse)
    for pi in range(80):
        if time.time() - t0 > budget_s - 15:
            break
        sigma = 0.002 + 0.006 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = _lbfgs(h_p, n, 1.5, 2500)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            # After finding improvement, Nesterov fine-tune
            h_ng2, c5_ng2 = _nesterov(h_best, n, t0, min(60, budget_s - time.time() - 20))
            if c5_ng2 < c5_best:
                c5_best = c5_ng2
                h_best = h_ng2

    # Phase 5: Final Nesterov + SLSQP-like local search
    if time.time() - t0 < budget_s - 30:
        h_ng3, c5_ng3 = _nesterov(h_best, n, t0, min(80, budget_s - time.time() - 10))
        if c5_ng3 < c5_best:
            c5_best = c5_ng3
            h_best = h_ng3
        h_new = _lbfgs(h_best, n, 0.2, 4000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
