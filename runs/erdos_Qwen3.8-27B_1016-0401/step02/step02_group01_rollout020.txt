    idx = np.argsort(corr)[-m:]

def _scl_fun(x):
    return x.sum() - _G_TARGET

def _scl_jac(x):
    return np.ones(_G_N)

def _fobj(x):
    return sm_values(x, _G_T)[0]

def _fgrad(x):
    _, sw = sm_values(x, _G_T)
    return sm_grad_from_sw(x, sw)

def scl_solve(h0, target, T, maxiter, t0, budget):
    global _G_T, _G_TARGET, _G_N
    _G_TARGET = target
    _G_N = len(h0)
    _G_T = T
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _scl_fun, 'jac': _scl_jac}]
    try:
        res = minimize(_fobj, h0, method='SLSQP', jac=_fgrad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

            elif 0 < k < n:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] -= h[:n-k]
            elif -n < k < 0:
                p = -k
                grad[p:] += (1.0 - h[:n-p])
                grad[:n-p] -= h[p:]
        grad *= cx
        if momentum > 0:
            grad = (1.0 - momentum) * grad + momentum * prev_grad
        prev_grad = grad
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        c5_cur = compute_c5(h)
        accepted = False
        for _ in range(30):
            h_new = project_constraint(h - lr * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                accepted = True
                break
            lr *= 0.4
        if not accepted:
            stuck += 1
            prev_grad *= 0.5
            if stuck > 60:
                break
        else:
            stuck = 0
            lr = min(lr * 1.05, lr0)
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def scl_solve(h0, target, T, maxiter, t0, budget):
    global _G_T, _G_TARGET, _G_N
    _G_TARGET = target
    _G_N = len(h0)
    _G_T = T
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _scl_fun, 'jac': _scl_jac}]
    try:
        res = minimize(_fobj, h0, method='SLSQP', jac=_fgrad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n0 = len(h_best)
    target0 = n0 / 2.0
    h_best = project_constraint(h_best, target0)
    c5_best = compute_c5(h_best)

    # Phase 1: Quick SLSQP at moderate T
    if time.time() - t0 < budget - 650:
        h, c5 = scl_solve(h_best, target0, 80, 150, t0, budget - 640)
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 2: PG with momentum, anneal T downward
    for T in [60, 30, 15, 8, 5, 3]:
        if time.time() - t0 > budget - 320:
            break
        h_best, c5 = pg_solve(h_best, target0, T, 5000, 0.025, t0, budget - 300, momentum=0.5)
        if c5 < c5_best:
            c5_best, h_best = c5, h_best.copy()

    # Phase 3: Direct max-subgradient
    if time.time() - t0 < budget - 250:
        h_tmp, c5 = max_subgrad_solve(h_best, target0, t0, budget - 230,
                                       n_iter=6000, lr0=0.01, n_worst=2, momentum=0.4)
        if c5 < c5_best:
            c5_best, h_best = c5, h_tmp.copy()

    # Phase 4: Very low T PG refinement
    if time.time() - t0 < budget - 150:
        h_best, c5 = pg_solve(h_best, target0, 3, 8000, 0.012, t0, budget - 130, momentum=0.6)
        if c5 < c5_best:
            c5_best, h_best = c5, h_best.copy()

    # Phase 5: Diverse perturbations + reoptimize
    for r in range(30):
        if time.time() - t0 > budget - 70:
            break
        h_p = h_best.copy()
        m = r % 6
        if m == 0:
            idx = rng.integers(0, n0)
            sz = rng.integers(4, 18)
            for i in range(sz):
                h_p[(idx + i) % n0] = 1.0 - h_p[(idx + i) % n0]
        elif m == 1:
            idx = rng.integers(0, max(n0 - 40, 10))
            sz = rng.integers(8, min(40, n0 // 5))
            h_p[idx:idx+sz] = h_p[idx:idx+sz][::-1]
        elif m == 2:
            h_p = np.clip(h_best + rng.normal(0, 0.02, n0), 0, 1)
        elif m == 3:
            h_p = np.roll(h_best, rng.integers(1, 7))
        elif m == 4:
            i1, i2 = sorted(rng.integers(0, max(n0 - 15, 10), size=2))
            sz = min(rng.integers(6, 15), n0 - i2)
            tmp = h_p[i1:i1+sz].copy()
            h_p[i1:i1+sz] = h_p[i2:i2+sz]
            h_p[i2:i2+sz] = tmp
        else:
            thr = rng.uniform(0.35, 0.65)
            h_p = (h_best > thr).astype(float)

        h_p = project_constraint(h_p, target0)
        T_p = 5 + (r % 10) * 2
        h_p, c5_p = pg_solve(h_p, target0, T_p, 800, 0.02, t0, budget - 40, momentum=0.4)
        if c5_p < c5_best:
            c5_best, h_best = c5_p, h_p.copy()

    # Phase 6: Final SLSQP polish
    if time.time() - t0 < budget - 25:
        h_best, c5 = scl_solve(h_best, target0, 25, 200, t0, budget - 10)
        if c5 < c5_best:
            c5_best = c5

    # Phase 7: Ultra-fine PG at T=2
    if time.time() - t0 < budget - 15:
        h_best, c5 = pg_solve(h_best, target0, 2, 3000, 0.008, t0, budget - 3, momentum=0.6)
        if c5 < c5_best:
            c5_best, h_best = c5, h_best.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
