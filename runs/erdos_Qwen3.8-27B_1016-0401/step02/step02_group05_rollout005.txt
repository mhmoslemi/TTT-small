                   jac=soft_max_grad, args=(T,),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if res.fun < 50:
        return fix_sum(res.x.copy(), sum(h))
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target,)}]

    # Phase 1: SLSQP soft-max from initial (moderate temp)
    for T in [15, 8, 4, 2, 1.0]:
        if time.time() - t0 > budget_s * 0.25:
            break
        h_try = slsqp_run(h_best, T, bounds, cons, 3000)
        if h_try is not None:
            c = compute_c5(h_try)
            if c < c5_best - 1e-13:
                c5_best = c
                h_best = h_try

    # Phase 2: Subgradient with Nesterov momentum
    h_sg, c5_sg = subgrad_nesterov(h_best, target, 8000, t0, budget_s * 0.45)
    if c5_sg < c5_best - 1e-14:
        c5_best = c5_sg
        h_best = h_sg

    # Phase 3: 2-opt local search
    h_lo, c5_lo = two_opt(h_best, target, 5000, t0, budget_s * 0.62, rng)
    if c5_lo < c5_best - 1e-14:
        c5_best = c5_lo
        h_best = h_lo

    # Phase 4: SLSQP from perturbed starts
    for restart in range(8):
        if time.time() - t0 > budget_s * 0.82:
            break
        r = restart
        if r % 3 == 0:
            h_init = h_best + rng.normal(0, 0.05, n)
        elif r % 3 == 1:
            h_init = h_best.copy()
            st = int(rng.integers(0, max(1, n - 40)))
            ln = int(rng.integers(5, 35))
            h_init[st:st + ln] = 1.0 - h_init[st:st + ln]
        else:
            h_init = h_best + rng.normal(0, 0.02, n)
        h_init = fix_sum(h_init, target)
        T = 3
        h_try = slsqp_run(h_init, T, bounds, cons, 2000)
        if h_try is not None:
            c = compute_c5(h_try)
            if c < c5_best - 1e-13:
                c5_best = c
                h_best = h_try
        h_lo2, c5_lo2 = two_opt(h_best, target, 1500, t0, budget_s * 0.85, rng)
        if c5_lo2 < c5_best - 1e-14:
            c5_best = c5_lo2
            h_best = h_lo2

    # Phase 5: Fine SLSQP at very low temp
    for T in [0.8, 0.5, 0.3]:
        if time.time() - t0 > budget_s - 60:
            break
        h_try = slsqp_run(h_best, T, bounds, cons, 4000)
        if h_try is not None:
            c = compute_c5(h_try)
            if c < c5_best - 1e-14:
                c5_best = c
                h_best = h_try

    # Phase 6: Final 2-opt
    h_fin, c5_fin = two_opt(h_best, target, 3000, t0, budget_s - 2, rng)
    if c5_fin < c5_best - 1e-14:
        c5_best = c5_fin
        h_best = h_fin

    # Final verification
    h_best = fix_sum(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
