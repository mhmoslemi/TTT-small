    for T in [50, 200, 600, 1500, 5000, 15000, 50000]:
        if elapsed() > budget_s - 450:
            break
        h_new = lbfgs_sm(h_best, n, target, T, 5.0, 4000, 1e-14)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Hinge-loss targeted optimization
    for frac in [0.9995, 0.999, 0.998, 0.996, 0.993, 0.988]:
        if elapsed() > budget_s - 400:
            break
        T_target = c5_best * frac
        h_new = lbfgs_hinge(h_best, n, target, T_target, 5.0, 3000, 1e-13)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Alternate soft-max and hinge (fine polishing)
    for it in range(15):
        if elapsed() > budget_s - 250:
            break
        T = 10000 + it * 5000
        h_new = lbfgs_sm(h_best, n, target, T, 5.0, 3000, 1e-15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        T_target = c5_best * (0.9995 - 0.0001 * it)
        h_new = lbfgs_hinge(h_best, n, target, T_target, 8.0, 3000, 1e-15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Random perturbations + optimization (escape local minima)
    best_streak = 0
    while elapsed() < budget_s - 120:
        best_streak += 1
        scale = max(0.005, 0.04 / (1.0 + 0.02 * best_streak))
        mode = best_streak % 4
        if mode == 0:
            h_pert = h_best + rng.normal(0, scale, n)
        elif mode == 1:
            h_pert = h_best.copy()
            s = rng.integers(0, n - 8)
            h_pert[s:s+8] += rng.uniform(-0.2, 0.2)
        elif mode == 2:
            h_pert = h_best.copy()
            swap_idx = rng.permutation(n)[:4]
            for i in range(2):
                idx1, idx2 = swap_idx[2*i], swap_idx[2*i+1]
                h_pert[idx1], h_pert[idx2] = h_best[idx2], h_best[idx1]
        else:
            # Complementary flip on a segment
            h_pert = h_best.copy()
            s = rng.integers(0, n - 20)
            e = s + rng.integers(5, 20)
            h_pert[s:e] = 1.0 - h_best[s:e]
        h_pert = project_box_sum(h_pert, target)
        T = 5000 + int(rng.integers(0, 30000))
        h_new = lbfgs_sm(h_pert, n, target, T, 8.0, 1500, 1e-13)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            best_streak = 0

    # Phase 5: Try different resolutions
    for n_try in [560, 140, 420, 70]:
        if elapsed() > budget_s - 60:
            break
        t_try = n_try / 2.0
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        for T in [3000, 15000, 50000]:
            if elapsed() > budget_s - 25:
                break
            h_t = lbfgs_sm(h_try, n_try, t_try, T, 5.0, 2500, 1e-14)
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        # Hinge polish at this resolution
        if elapsed() < budget_s - 10 and c5_try < c5_best + 0.001:
            T_target = c5_try * 0.999
            h_t = lbfgs_hinge(h_try, n_try, t_try, T_target, 5.0, 2000, 1e-14)
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n = n_try
            target = t_try

    # Phase 6: Final ultra-high-T polish
    if elapsed() < budget_s - 5:
        h_new = lbfgs_sm(h_best, n, target, 100000, 10.0, 3000, 1e-16)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        T_target = c5_best * 0.999
        h_new = lbfgs_hinge(h_best, n, target, T_target, 10.0, 2000, 1e-16)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Verify and return
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_box_sum(h_best, target)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
