import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(-(1.0 - h))
            for i in order:
                if diff <= 0: break
                s = min(diff, 1.0 - h[i])
                h[i] += s; diff -= s
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= 0: break
                s = min(-diff, h[i])
                h[i] -= s; diff += s
    return h

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    if T < 1e-6:
        return mc * dx
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    if T < 1e-6:

        o = im - (n - 1)

def local_search(h, target, n_iter, t0, deadline, step_sizes=None):
    """Targeted local search on worst lag contributors."""
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    corr = np.correlate(h_curr, 1 - h_curr, mode="full")
    best = float(np.max(corr) * dx)
    if step_sizes is None:
        step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005]
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        if p == 0:
            # Self-correlation: try adjusting individual values
            for delta in step_sizes:
                vals = h_curr * (1 - h_curr)
                idx = np.argmax(vals)
                h_try = h_curr.copy()
                if h_curr[idx] > 0.5:
                    h_try[idx] -= delta
                else:

                    corr = np.correlate(h_curr, 1 - h_curr, mode="full")
                    stall = 0

                if h_curr[idx] > 0.5:
                    h_try2[idx] += delta
                else:

                h_try2 = np.clip(h_try2, 0, 1)
                h_try2 = fix_constraint(h_try2, target)

                    corr = np.correlate(h_curr, 1 - h_curr, mode="full")
                    stall = 0
                    break
            else:
                stall += 1
        else:

            if s > 0:
                contrib = h_curr[p:] * (1 - h_curr[:n-p])
            else:
                contrib = h_curr[:n-p] * (1 - h_curr[p:])
            # Try top-k contributors

                    h_try[j1] = np.clip(h_try[j1] - delta, 0, 1)

                        stall = 0

                        break
                    # Try reverse
                    h_try = h_curr.copy()
                    h_try[j1] = np.clip(h_try[j1] + delta, 0, 1)
