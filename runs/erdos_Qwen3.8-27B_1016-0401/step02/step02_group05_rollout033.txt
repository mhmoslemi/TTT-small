            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] += -wi * h[s:]
        else:
            p = -s
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    return grad * dx, mc

def project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            frac = 1.0 - h
            mask = frac > 1e-14
            if not mask.any():
                break
            tf = frac[mask].sum()
            if tf < 1e-14:
                break
            h[mask] += (diff / tf) * frac[mask]
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            tm = h[mask].sum()
            if tm < 1e-14:
                break
            h[mask] -= ((-diff) / tm) * h[mask]
            h = np.clip(h, 0.0, 1.0)
    return h

def block_local_search(h, target, n_iter, t0, budget, delta_start=0.06, rng_seed=99):
    n = len(h)
    c5, corr = compute_c5(h)
    h_curr = h.copy()
    delta = delta_start
    stall = 0
    rng = np.random.default_rng(rng_seed)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 150:
            delta *= 0.7
            if delta < 0.002:
                delta = delta_start * 0.4
            stall = 0
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        mt = rng.integers(0, 5)
        h_try = h_curr.copy()
        if mt == 0 and p > 0:
            if s > 0:
                contrib = h_curr[p:] * (1.0 - h_curr[:n - p])
                idx = int(np.argmax(contrib))
                h_try[idx + p] -= delta
                h_try[idx] += delta
            else:
                contrib = h_curr[:n - p] * (1.0 - h_curr[p:])
                idx = int(np.argmax(contrib))
                h_try[idx] -= delta
                h_try[idx + p] += delta
        elif mt == 1:
            blen = int(rng.integers(2, max(3, n // 25)))
            st1 = int(rng.integers(0, n - blen))
            st2 = int(rng.integers(0, n - blen))
            if abs(st1 - st2) < blen * 2:
                continue
            h_try[st1:st1 + blen] -= delta
            h_try[st2:st2 + blen] += delta
        elif mt == 2:
            cp = int(rng.integers(n // 5, 4 * n // 5))
            width = int(rng.integers(3, max(4, n // 12)))
            amp = delta * float(rng.choice([1, -1]))
            for j in range(n):
                d = j - cp
                if abs(d) < width:
                    h_try[j] += amp * np.cos(np.pi * d / (2.0 * width))
        elif mt == 3:
            sr = int(rng.integers(-n // 3, n // 3 + 1))
            if sr == 0:
                idx = int(rng.integers(n))
                h_try[idx] += delta * float(rng.choice([1, -1]))
            else:
                pr = abs(sr)
                idx = int(rng.integers(max(1, n - pr)))
                if sr > 0:
                    h_try[idx + pr] -= delta
                    h_try[idx] += delta
                else:
                    h_try[idx] -= delta
                    h_try[idx + pr] += delta
        else:
            idx1 = int(rng.integers(n))
            idx2 = int(rng.integers(n))
            if idx1 != idx2:
                move = delta * float(rng.choice([1, -1]))
                h_try[idx1] -= move
                h_try[idx2] += move
        h_try = project(h_try, target)
        c5_try, corr_try = compute_c5(h_try)
        if c5_try < c5 - 1e-14:
            c5 = c5_try
            h_curr = h_try
            corr = corr_try
            stall = 0
        else:
            stall += 1
    return h_curr, c5

def slsqp_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def sum_fun(x, target):
    return x.sum() - target

def sum_jac(x, target):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project(h_best, target)
    c5_best, _ = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(len(x))}]

    # Phase 1: Subgradient descent with momentum
    mom = 0.85
    v = np.zeros(n)
    for T in [30, 15, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5]:
        if time.time() - t0 > budget_s - 350:
            break
        for it in range(400):
            if time.time() - t0 > budget_s - 300:
                break
            g, _ = subgrad_max(h_best, T)
            eta = 0.02 / max(1.0, np.sqrt(it + 1))
            v = mom * v - eta * g
            h_new = project(h_best + v, target)
            c5_new, _ = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
                v = 0.3 * v

    # Phase 2: Intensive block local search
    if time.time() - t0 < budget_s - 250:
        h_ls, c5_ls = block_local_search(h_best, target, 6000, t0, budget_s - 250, 0.06, 42)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls

    # Phase 3: SLSQP at multiple temperatures
    if time.time() - t0 < budget_s - 150:
        for T in [8, 4, 2, 1.2]:
            if time.time() - t0 > budget_s - 120:
                break
            try:
                res = minimize(slsqp_obj, h_best, method='SLSQP', args=(T, n),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-16})
                if res.fun < 100:
                    h_new = project(res.x, target)
                    c5_new, _ = compute_c5(h_new)
                    if c5_new < c5_best:
                        c5_best = c5_new
                        h_best = h_new
            except Exception:
                pass

    # Phase 4: Fine local search with small delta
    if time.time() - t0 < budget_s - 50:
        h_ls2, c5_ls2 = block_local_search(h_best, target, 5000, t0, budget_s - 50, 0.02, 77)
        if c5_ls2 < c5_best:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 5: Random perturbation + re-optimize (escape local optima)
    if time.time() - t0 < budget_s - 30:
        for _ in range(3):
            if time.time() - t0 > budget_s - 25:
                break
            h_pert = h_best + rng.normal(0, 0.03, n)
            h_pert = project(h_pert, target)
            c5_pert, _ = compute_c5(h_pert)
            h_ls3, c5_ls3 = block_local_search(h_pert, target, 1500, t0, budget_s - 25, 0.03, int(rng.integers(1000, 9999)))
            if c5_ls3 < c5_best:
                c5_best = c5_ls3
                h_best = h_ls3

    # Phase 6: Ultra-fine polish
    if time.time() - t0 < budget_s - 5:
        h_ls4, c5_ls4 = block_local_search(h_best, target, 2000, t0, budget_s - 5, 0.008, 2024)
        if c5_ls4 < c5_best:
            c5_best = c5_ls4
            h_best = h_ls4

    c5_final, _ = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
