import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_obj(h, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / len(h))

def sm_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    sw = e / np.sum(e)
    c = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[c + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * dx

def proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(280)
        h[::2] = 1.0

    n = len(h)
    tgt = n / 2.0
    h = proj(h, tgt)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    bnd = [(0.0, 1.0)] * n
    ones = np.ones(n)

    def _do_opt(h0, T, pen, iters):
        def f(x):
            return sm_obj(x, T) + pen * (x.sum() - tgt) ** 2
        def g(x):
            return sm_grad(x, T) + 2.0 * pen * (x.sum() - tgt) * ones
        try:
            r = minimize(f, h0.copy(), method='L-BFGS-B', jac=g, bounds=bnd,
                         options={'maxiter': iters, 'gtol': 1e-15, 'factr': 1e12})
            return proj(r.x, tgt)
        except Exception:
            return proj(h0.copy(), tgt)

    # Phase 1: Progressive temperature (coarse to fine)
    for T in [30, 80, 200, 500, 1200, 3000]:
        if time.time() - t0 > budget_s - 300:
            break
        hn = _do_opt(h, T, 10.0, 4000)
        c = compute_c5(hn)
        if c < best_c5:
            best_c5, best_h = c, hn
            h = hn

    # Phase 2: Very high T for focused max optimization
    for T in [3000, 6000, 12000]:
        if time.time() - t0 > budget_s - 250:
            break
        hn = _do_opt(h, T, 50.0, 5000)
        c = compute_c5(hn)
        if c < best_c5:

            h = hn

        h_shift = np.roll(h, rng.integers(1, max(2, n // 10)))
        h_shift = proj(h_shift, tgt)
        hn = _do_opt(h_shift, 2000, 50.0, 3000)
        c = compute_c5(hn)
        if c < best_c5:

    return (best_h, c5_final, n)
