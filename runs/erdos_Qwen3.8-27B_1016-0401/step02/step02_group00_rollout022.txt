            hi = mid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode="full")

    mc = corr.max()
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def smooth_max_grad(x, T, dx):
    n = len(x)
    corr = corr_full(x)
    mc = corr.max()
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    center = n - 1

        if w < 1e-12:

            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]

    n = len(h)

    y = h.copy()
    z = h.copy()
    t_k = 1.0
    L = L_init

    best_h = h.copy()

        # Gradient step

        z = z_new

            if v < best_val:

                best_h = y.copy()

                    h_alt = h_new
                    stall = 0
                else:
                    stall += 1
            except Exception:
                stall += 1
            if stall == 10:
                break
        try:
            x_new = _slsqp_sm(h_alt, 400, bounds_alt, 1500, 1e-14)
            h_new = project_box_sum(np.clip(x_new, 0, 1), target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt - 1e-10:
                c5_alt = c5_new
                h_alt = h_new
        except Exception:
            pass
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt.copy()
            n = n_alt

    # Phase 4: Final polish on best solution (remaining budget minus 15s)
    p4_end = t0 + budget_s - 15.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f
    _g_n = n_f
    _g_dx = dx_f
    _g_target = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    for _ in range(20):
        if time.time() - t0 > p4_end:
            break
        corr_f = _correlation(h_best)
        top_idx = np.argsort(corr_f)[-4:]
        for lag in top_idx:
            if time.time() - t0 > p4_end:
                break
            try:
                x_new = _slsqp_lag(h_best, int(lag), bounds_f, 1500, 1e-16)
                h_new = project_box_sum(np.clip(x_new, 0, 1), target_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-12:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
