def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
    else:
        order = np.argsort(h)
    for i in order:
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            s = min(diff, 1.0 - h[i])
        else:
            s = max(diff, -h[i])
        h[i] += s
        diff -= s
    return h

def grad_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    mc = corr.max()
    sh = T * (corr - mc)
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o]
        if wi < 1e-14:
            continue

            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] -= wi * h[o:]
        elif o < 0:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad * dx
