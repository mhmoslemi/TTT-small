import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    if mc < -1e10:
        return 0.0
    shifted = T * (corr - mc)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = mc + log_sum / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-14:

        if o == 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = float(np.max(corr))
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = float(np.max(corr))
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def projected_grad_desc(h, target, T, n_iter, t0, budget, lr0):
    n = len(h)
    h_c = h.copy()
    best_c5 = compute_c5(h_c)
    best_h = h_c.copy()
    lr = lr0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        g = soft_max_jac(h_c, T)
        h_new = h_c - lr * g
        h_new = fix_constraint(h_new, target)
        h_c = h_new
        c5 = compute_c5(h_c)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_c.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.6
                stall = 0
        if lr < 1e-10:
            break
    return best_h, best_c5

def local_search(h, target, n_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    corr = np.correlate(h_c, 1 - h_c, mode="full")
    best_c5 = float(np.max(corr) * dx)
    base_deltas = [0.25, 0.15, 0.1, 0.06, 0.04, 0.025, 0.015, 0.01]
    di = 0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 400:
            di = min(di + 1, len(base_deltas) - 1)
            stall = 0
            if di >= len(base_deltas) - 1:
                break
        delta = base_deltas[di]
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        improved = False
        if p == 0:
            vals = h_c * (1 - h_c)
            idx = int(np.argmax(vals))
            for d in [delta, 0.5 * delta, 2.0 * delta]:
                h_t = h_c.copy()
                if h_c[idx] > 0.5:
                    h_t[idx] = max(0.0, h_t[idx] - d)
                else:
                    h_t[idx] = min(1.0, h_t[idx] + d)
                h_t = fix_constraint(h_t, target)
                ct = np.correlate(h_t, 1 - h_t, mode="full")
                c5t = float(np.max(ct) * dx)
                if c5t < best_c5 - 1e-15:
                    h_c = h_t
                    corr = ct
                    best_c5 = c5t
                    improved = True
                    break
        else:
            if s > 0:
                contrib = h_c[p:] * (1 - h_c[:n - p])
                idx = int(np.argmax(contrib))
                j1, j2 = idx + p, idx
            else:
                contrib = h_c[:n - p] * (1 - h_c[p:])
                idx = int(np.argmax(contrib))
                j1, j2 = idx, idx + p
            for d in [delta, 0.5 * delta, 2.0 * delta]:
                h_t = h_c.copy()
                h_t[j1] -= d
                h_t[j2] += d
                h_t = np.clip(h_t, 0, 1)
                h_t = fix_constraint(h_t, target)
                ct = np.correlate(h_t, 1 - h_t, mode="full")
                c5t = float(np.max(ct) * dx)
                if c5t < best_c5 - 1e-15:
                    h_c = h_t
                    corr = ct
                    best_c5 = c5t
                    improved = True
                    break
            if not improved:
                for d in [delta, 0.5 * delta]:
                    h_t = h_c.copy()
                    h_t[j1] += d
                    h_t[j2] -= d
                    h_t = np.clip(h_t, 0, 1)
                    h_t = fix_constraint(h_t, target)
                    ct = np.correlate(h_t, 1 - h_t, mode="full")
                    c5t = float(np.max(ct) * dx)
                    if c5t < best_c5 - 1e-15:
                        h_c = h_t
                        corr = ct
                        best_c5 = c5t
                        improved = True
                        break
        if improved:
            stall = 0
            di = 0
        else:
            stall += 1
    return h_c, best_c5
