```python
    hi = y.max() + 2.0

        else:

            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")

                else:

        # Re-project to maintain sum constraint
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _shift_grad(h, n, shift):
    dx = 2.0 / n
    j = np.arange(n)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif shift > 0:
        m = n - shift
        grad[:m] += (1.0 - h[shift:m+shift]) * dx
        grad[shift:] -= h[:m] * dx
    else:
        s = -shift
        m = n - s
        grad[s:] += (1.0 - h[:m]) * dx
        grad[:m] -= h[s:] * dx
    return grad

# Global state for SLSQP callbacks
_G = {}

def _obj(x):
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    T = _G['T']
    sh = np.clip(T * (corr - mc), -50.0, 0.0)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _jac(x):
    n = _G['n']
    T = _G['T']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = np.clip(T * (corr - mc), -50.0, 0.0)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        s = oi - center
        if s == 0:
            grad += w * (1.0 - 2.0 * x)
        elif s > 0:
            m = n - s
            grad[:m] += w * (1.0 - x[s:s+m])
            grad[s:] -= w * x[:m]
        else:
            ss = -s
            m = n - ss
            grad[ss:] += w * (1.0 - x[:m])
            grad[:m] -= w * x[ss:]
    return grad * dx

def _cons_fun(x):
    return x.sum() - _G['target']

def _cons_jac(x):
    return np.ones(_G['n'])

def do_slsqp(h, n, T, maxiter=3000, ftol=1e-16):
    _G['n'] = n
    _G['target'] = n / 2.0
    _G['T'] = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coord_descent(h, n, sweeps=5):

            old = h[i]
            h[i] = 0.0

            h[i] = 1.0

                    b = t1
                else:
                    a = t2
            h[i] = (a + b) / 2.0
        h = project_feasible(h, n)
    return h

def nesterov_subgrad(h_init, n, t0, budget_s, max_iters=400000):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.3
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))

        g = _shift_grad(h_y, n, shift)

        else:

            diff = target - h_new.sum()

        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
```

The issue with my coordinate descent: I'm computing two full correlations per coordinate. For n=280, that's 2 × 280 × O(n²) = 2 × 280 × 78K ≈ 44M operations per sweep. With numpy, this should take about 0.5-1 second per sweep. 10 sweeps = 5-10 seconds. That's fine.

But wait, I can optimize: instead of computing full correlations, I can compute just the slope vector analytically. Let me do that.

For coordinate i, the slope for each shift m (index in the correlate output):
- m corresponds to shift s = m - (n-1)
- h[i] appears in correlate[m] as:
  - h[k] with k=i: valid if 0 ≤ i-s < n → s ≤ i < n+s
    Contribution to derivative: (1-h[i-s])
  - (1-h[k-s]) with k-s=i, k=i+s: valid if 0 ≤ i+s < n → -s ≤ i < n-s
    Contribution to derivative: -h[i+s]

So slope[m] = [s ≤ i < n+s] * (1-h[i-s]) + [-s ≤ i < n-s] * (-h[i+s])

Where the conditions mean the indices are valid.

Let me vectorize this:
```python
def compute_slopes(h, n, i):
    m = np.arange(2*n - 1)
    s = m - (n - 1)  # shifts from -(n-1) to n-1
    # Condition 1: s <= i < n+s, i.e., 0 <= i-s < n
    c1 = (i - s >= 0) & (i - s < n)  # i-s in [0, n-1]
    # Condition 2: -s <= i < n-s, i.e., 0 <= i+s < n
    c2 = (i + s >= 0) & (i + s < n)  # i+s in [0, n-1]
    term1 = np.where(c1, 1.0 - h[i - s[c1]], 0.0)  # careful with indexing
    term2 = np.where(c2, -h[i + s[c2]], 0.0)
    return term1 + term2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _shift_grad(h, n, shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif shift > 0:
        m = n - shift
        grad[:m] += (1.0 - h[shift:shift + m]) * dx
        grad[shift:] -= h[:m] * dx
    else:
        s = -shift
        m = n - s
        grad[s:] += (1.0 - h[:m]) * dx
        grad[:m] -= h[s:] * dx
    return grad

def _obj(x):
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    T = _G['T']
    sh = np.clip(T * (corr - mc), -60.0, 0.0)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _jac(x):
    n = _G['n']
    T = _G['T']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = np.clip(T * (corr - mc), -60.0, 0.0)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        s = oi - center
        if s == 0:
            grad += w * (1.0 - 2.0 * x)
        elif s > 0:
            m = n - s
            grad[:m] += w * (1.0 - x[s:s + m])
            grad[s:] -= w * x[:m]
        else:
            ss = -s
            m = n - ss
            grad[ss:] += w * (1.0 - x[:m])
            grad[:m] -= w * x[ss:]
    return grad * dx

def _cons_fun(x):
    return x.sum() - _G['target']

def _cons_jac(x):
    return np.ones(_G['n'])

def do_slsqp(h, n, T, maxiter=3000, ftol=1e-16):
    _G['n'] = n
    _G['target'] = n / 2.0
    _G['T'] = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coord_descent(h, n, sweeps=5, seed=0):
    rng = np.random.default_rng(seed)
    h = h.copy()
    for sweep in range(sweeps):
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            slopes = c1 - c0
            h[i] = old
            a, b = 0.0, 1.0
            for _ in range(35):
                t1 = a + (b - a) / 3.0
                t2 = b - (b - a) / 3.0
                if np.max(c0 + t1 * slopes) < np.max(c0 + t2 * slopes):
                    b = t1
                else:
                    a = t2
            h[i] = (a + b) / 2.0
        h = project_feasible(h, n)
    return h

def nesterov_subgrad(h_init, n, t0, budget_s, max_iters=400000):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.35
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = _shift_grad(h_y, n, shift)
        alpha = alpha0 / (1.0 + 0.004 * it) ** 0.6
        h_new = h_y - alpha * g
        if (it + 1) % 5 == 0:
            h_new = project_feasible(h_new, n)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - h_new.sum()
            if abs(diff) > 1e-10:
                h_new = np.clip(h_new + diff / n, 0.0, 1.0)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 15000 == 0:
            c5a = compute_c5(project_feasible(h_avg, n))
            if c5a < c5_best:
                c5_best = c5a
                h_best = project_feasible(h_avg, n).copy()
    return h_best, c5_best
```
