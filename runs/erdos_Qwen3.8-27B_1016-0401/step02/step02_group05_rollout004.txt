import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx), corr

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            if not mask.any():
                break
            h[mask] += room[mask] * (diff / room[mask].sum())
        else:
            avail = h.copy()
            mask = avail > 1e-14
            if not mask.any():
                break
            h[mask] -= avail[mask] * (-diff / avail[mask].sum())
        h = np.clip(h, 0, 1)
    if abs(target - h.sum()) > 1e-12:
        diff = target - h.sum()
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(-h)
        for i in order:
            if diff > 0:
                s = min(diff, 1.0 - h[i]); h[i] += s; diff -= s
            else:
                s = min(-diff, h[i]); h[i] -= s; diff += s
            if abs(diff) < 1e-14:
                break
    return h

def smax_og(x, T, n, dx):
    c = np.correlate(x, 1 - x, mode="full")
    mx = c.max()
    sh = T * (c - mx)
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mx + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    ctr = n - 1
    for o in range(-(n - 1), n):
        w = sw[ctr + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] -= w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    return val, grad * dx

def slsqp_phase(h, target, bounds, t0, deadline, temps, maxiter=3000):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    c5_c, _ = compute_c5(h_c)
    for T in temps:
        if time.time() > deadline:
            break
        def obj(x, _T=T):
            v, _ = smax_og(x, _T, n, dx)
            return v
        def jac(x, _T=T):
            _, g = smax_og(x, _T, n, dx)
            return g
        def cf(x):
            return x.sum() - target
        def cj():
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(obj, h_c, method='SLSQP', jac=jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            if res.fun < 5.0:
                h_new = fix_constraint(res.x.copy(), target)
                c5_new, _ = compute_c5(h_new)
                if c5_new < c5_c - 1e-13:
                    c5_c = c5_new
                    h_c = h_new
        except Exception:
            pass
    return h_c, c5_c
