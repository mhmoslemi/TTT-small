    """Project v onto {0 <= h <= 1, sum(h) = target}"""
    n = len(v)
    # Bisection on lambda
    lo, hi = v.min() - target, v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

Wait, I realize the local search might be too slow if I'm doing np.correlate for every trial. Let me think about whether there's a faster way.

For a single pair move (j1 -= d, j2 += d), the change in correlation at lag s is:
- For lag s: the terms involving h[j1] or h[j2] change
- Specifically, at lag s (where s = k - (n-1)):
  - If j1 >= s (for s>0): the term h[j1] * (1-h[j1-s]) changes from h[j1]*(1-h[j1-s]) to (h[j1]-d)*(1-h[j1-s])
    → change = -d*(1-h[j1-s])
  - If j1 <= n-1-s (for s>0): the term h[j1+s] * (1-h[j1]) changes
    → change = -(-d)*h[j1+s] = d*h[j1+s]... wait

This is getting complicated. For n=280, np.correlate should still be fast enough. Let me just use it directly and keep the number of evaluations per iteration low.

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            s = min(diff, 1.0 - h[idx])
            h[idx] += s
        else:
            idx = int(np.argmax(h))
            s = min(-diff, h[idx])
            h[idx] -= s
    return h

def project_feasible(v, target):
    n = len(v)
    lo, hi = v.min() - target - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] -= w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] -= w * x[:n - p]
    return grad * dx

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def run_slsqp(h, T, target, bounds, maxiter=2000):
    cons = [{'type': 'eq', 'fun': partial(sum_eq_fun, target=target),
             'jac': partial(sum_eq_jac, target=target)}]
    try:
        res = minimize(soft_max_obj, h, method='SLSQP',
                       jac=soft_max_jac, args=(T,),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 50:
            return fix_constraint(res.x.copy(), target)
    except Exception:
        pass
    return h

def local_search(h, target, n_iter, t0, time_limit, delta_init=0.02):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    corr = np.correlate(h_curr, 1 - h_curr, mode="full")
    best_c5 = float(np.max(corr) * dx)
    delta = delta_init
    stall = 0

    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        if stall > 400:
            delta *= 0.5
            stall = 0
            if delta < 0.001:
                break
        if stall > 100:
            delta = min(delta * 1.3, 0.1)

        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        p = abs(s)
        improved = False

        if p == 0:
            vals = h_curr * (1 - h_curr)
            idx = int(np.argmax(vals))
            for d in [delta, delta * 0.5]:
                ht = h_curr.copy()
                if ht[idx] > 0.5:
                    ht[idx] -= d
                else:
                    ht[idx] += d
                np.clip(ht, 0, 1, out=ht)
                ht = fix_constraint(ht, target)
                c = float(np.max(np.correlate(ht, 1 - ht, "full")) * dx)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    h_curr = ht
                    corr = np.correlate(ht, 1 - ht, "full")
                    improved = True
                    break
        else:
            if s > 0:
                contrib = h_curr[p:] * (1 - h_curr[:n - p])
            else:
                contrib = h_curr[:n - p] * (1 - h_curr[p:])
            top_k = min(4, len(contrib))
            top_idx = np.argsort(-contrib)[:top_k]
            for idx_i in top_idx:
                if improved:
                    break
                idx = int(idx_i)
                if s > 0:
                    j1, j2 = idx + p, idx
                else:
                    j1, j2 = idx, idx + p
                for d in [delta, delta * 0.5, delta * 2.0, delta * 0.1]:
                    ht = h_curr.copy()
                    ht[j1] -= d
                    ht[j2] += d
                    if ht[j1] < -1e-10 or ht[j2] > 1 + 1e-10:
                        continue
                    np.clip(ht, 0, 1, out=ht)
                    ht = fix_constraint(ht, target)
                    c = float(np.max(np.correlate(ht, 1 - ht, "full")) * dx)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                        h_curr = ht
                        corr = np.correlate(ht, 1 - ht, "full")
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1

    return h_curr, best_c5

def projected_gradient_loop(h, target, t0, time_limit):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    step = 0.02
    T = 20.0

    for it in range(5000):
        if time.time() - t0 > time_limit:
            break
        grad = soft_max_jac(h_curr, T)
        v = h_curr - step * grad
        h_new = project_feasible(v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
            step = min(step * 1.1, 0.15)
            T = max(T * 0.97, 1.0)
        else:
            step *= 0.7
            if step < 0.001:
                break
        if it % 20 == 19:
            h_ls, c5_ls = local_search(h_curr, target, n_iter=100,
                                        t0=t0, time_limit=time_limit, delta_init=0.015)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h_curr = h_ls

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP warm-up (60s)
    t_phase1 = 60
    for T in [80, 30, 10, 5, 2]:
        if time.time() - t0 > t0 + t_phase1:
            break
        h_new = run_slsqp(h_best, T, target, bounds, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new

    # Phase 2: Projected gradient + local search (350s)
    t_phase2_end = t0 + t_phase1 + 350
    h_pg, c5_pg = projected_gradient_loop(h_best, target, t0, t_phase2_end)
    if c5_pg < c5_best:
        c5_best = c5_pg
        h_best = h_pg

    # Phase 3: Intensive local search (250s)
    t_phase3_end = t_phase2_end + 250
    h_ls, c5_ls = local_search(h_best, target, n_iter=50000,
                                t0=t0, time_limit=t_phase3_end, delta_init=0.01)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 4: SLSQP polish (100s)
    t_phase4_end = t_phase3_end + 100
    for T in [3, 1.5, 0.8, 0.4]:
        if time.time() - t0 > t_phase4_end:
            break
        h_new = run_slsqp(h_best, T, target, bounds, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new

    # Phase 5: Very fine local search (150s)

    h_ls2, c5_ls2 = local_search(h_best, target, n_iter=50000,

    if c5_ls2 < c5_best:
        c5_best = c5_ls2
        h_best = h_ls2

    # Final check
    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_constraint(h_best, target)
    return (h_best, c5_final, n)
