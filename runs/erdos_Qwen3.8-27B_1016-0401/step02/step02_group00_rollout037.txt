import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            for i in order:
                a = min(diff, 1.0 - h[i])
                if a <= 0: continue
                h[i] += a; diff -= a
                if abs(diff) < 1e-13: break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                s = min(-diff, h[i])
                if s <= 0: continue
                h[i] -= s; diff += s
                if abs(diff) < 1e-13: break
    return h

    c5_best = compute_c5(h_best)
    
    def elapsed():

    # SCA optimizer
    def sca(h_init, n_pts, tgt, deadline, max_it=1000):
        h = h_init.copy()
        best = compute_c5(h)
        bh = h.copy()
        for it in range(max_it):
            if time.time() > deadline:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            ctr = n_pts - 1
            c_vec = np.zeros(n_pts + 1); c_vec[-1] = 1.0
            A = np.zeros((2*n_pts-1, n_pts+1))

    h_best = fix_constraint(h_best, len(h_best)/2.0)
    return (h_best, c5_final, len(h_best))

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            for i in order:
                a = min(diff, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                diff -= a
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                s = min(-diff, h[i])
                if s <= 0:
                    continue
                h[i] -= s
                diff += s
                if abs(diff) < 1e-14:
                    break
    return h

def sca_optimize(h_init, n_pts, tgt, deadline, rng, max_it=2000):
    h = h_init.copy()
    best = compute_c5(h)
    bh = h.copy()
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        ctr = n_pts - 1
        c_vec = np.zeros(n_pts + 1)
        c_vec[-1] = 1.0
        n_lags = 2 * n_pts - 1
        A = np.zeros((n_lags, n_pts + 1))
        b = np.zeros(n_lags)
        for ki in range(n_lags):

            if o == 0:
                g = 1.0 - 2.0 * h
            elif o > 0:

                g[:n_pts - o] += (1.0 - h[o:])
                g[o:] -= h[:n_pts - o]

                g[p:] += (1.0 - h[:n_pts - p])
                g[:n_pts - p] -= h[p:]
            A[ki, :n_pts] = g
            A[ki, -1] = -1.0
            b[ki] = float(np.dot(g, h)) - corr[ki]
        Ae = np.zeros((1, n_pts + 1))

        be = np.array([tgt])
