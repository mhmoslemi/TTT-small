import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 0:
                    continue
                h[i] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def corr_lag(h, o):
    n = len(h)
    if o == 0:
        return float(np.sum(h * (1.0 - h)))
    if o > 0:
        return float(np.sum(h[o:] * (1.0 - h[:n - o])))
    p = -o
    return float(np.sum(h[:n - p] * (1.0 - h[p:])))

def corr_grad_lag(h, o):
    n = len(h)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n - o]
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - h[p:]
        g[p:] += -h[:n - p]
    return g

def smax_obj(x, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -500, 500)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * 2.0 / len(x)

def smax_grad(x, T):
    n = len(x)
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -500, 500)
    e = np.exp(sh)
    sw = e / np.sum(e)
    g = np.zeros(n)
    for oi in range(2 * n - 1):
        if sw[oi] < 1e-10:
            continue
        g += sw[oi] * corr_grad_lag(x, oi - center)
    return g * 2.0 / n

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def lag_obj(x, o):
    return corr_lag(x, o) * 2.0 / len(x)

def lag_grad(x, o):
    return corr_grad_lag(x, o) * 2.0 / len(x)

def sm_obj_fn(x, T):
    return smax_obj(x, T)

def sm_grad_fn(x, T):
    return smax_grad(x, T)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - proj)[::-1] if diff > 0 else np.argsort(proj)[::-1]
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - proj[i])
                if a <= 0:
                    continue
                proj[i] += a
                diff -= a
            else:
                a = min(-diff, proj[i])
                if a <= 0:
                    continue
                proj[i] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return proj

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    c5_best = compute_c5(h_best)
    h_best = fix_constraint(h_best, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    # Phase 1: FISTA projected gradient with adaptive T (35% budget)
    p1_end = t0 + budget_s * 0.35
    h_x = h_best.copy()
    h_y = h_best.copy()
    k_f = 1.0
    T = 60.0
    no_imp = 0
    best_T = T
    for it in range(60000):
        if time.time() - t0 > p1_end:
            break
        g = smax_grad(h_y, T)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k_f * k_f)) / 2.0
        alpha = 1.0 / (T * 1.2)
        h_new = project_box_sum(h_y - alpha * g, target)
        h_y = h_new + ((k_f - 1.0) / t_k) * (h_new - h_x)
        h_y = np.clip(h_y, 0.0, 1.0)
        h_x = h_new
        k_f = t_k
        T = min(T * 1.0008, 4000.0)

        if it % 400 == 0:
            c5_cur = compute_c5(h_x)
            if c5_cur < c5_best - 1e-10:
                c5_best = c5_cur
                h_best = h_x.copy()
                no_imp = 0
                best_T = T
            else:
                no_imp += 1
            if no_imp > 10:
                h_x = h_best + rng.normal(0, 0.015, n)
                h_x = fix_constraint(h_x, target)
                h_y = h_x.copy()
                k_f = 1.0
                T = max(60.0, best_T * 0.5)
                no_imp = 0

    # Phase 2: Active-set SLSQP cycling top lags (30% budget)
    p2_end = t0 + budget_s * 0.65
    h_act = h_best.copy()
    for outer in range(60):
        if time.time() - t0 > p2_end:
            break
        corr = np.correlate(h_act, 1.0 - h_act, mode="full")
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for lag in top_idx:
            if time.time() - t0 > p2_end:
                break
            o = lag - (n - 1)
            try:
                res = minimize(lag_obj, h_act, args=(o,), method='SLSQP',
                              jac=lag_grad, bounds=bounds, constraints=cons,
                              options={'maxiter': 1000, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
                    h_act = h_new
                    improved = True
            except Exception:
                pass
        if not improved and outer > 10:
            break

    # Phase 3: Multi-T soft-max SLSQP (25% budget)
    p3_end = t0 + budget_s * 0.90
    for T_val in [200, 400, 800, 1600, 3200]:
        if time.time() - t0 > p3_end:
            break
        try:
            res = minimize(sm_obj_fn, h_best, args=(T_val,), method='SLSQP',
                          jac=sm_grad_fn, bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-16})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Final active-set polish (10% budget)
    p4_end = t0 + budget_s - 5.0
    for _ in range(20):
        if time.time() - t0 > p4_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        o_f = int(np.argmax(corr_f)) - (n - 1)
        try:
            res = minimize(lag_obj, h_best, args=(o_f,), method='SLSQP',
                          jac=lag_grad, bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Try a slightly different n if time permits
    p5_end = t0 + budget_s - 3.0
    if time.time() - t0 < p5_end:
        n2 = 336
        target2 = n2 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h_best)
        h2 = fix_constraint(h2, target2)
        c5_2 = compute_c5(h2)
        _TARGET = target2
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        for _ in range(5):
            if time.time() - t0 > p5_end:
                break
            corr_2 = np.correlate(h2, 1.0 - h2, mode="full")
            o2 = int(np.argmax(corr_2)) - (n2 - 1)
            try:
                res = minimize(lag_obj, h2, args=(o2,), method='SLSQP',
                              jac=lag_grad, bounds=bounds2, constraints=cons2,
                              options={'maxiter': 800, 'ftol': 1e-15})
                h2 = fix_constraint(np.clip(res.x, 0, 1), target2)
            except Exception:
                pass
        c5_2 = compute_c5(h2)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2

    _TARGET = len(h_best) / 2.0
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
