def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] -= h[k:]
        if k > 0:
            g[k:] += (1.0 - h[:n - k])
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def subgrad_step(h, target, lr, top_k=8, t_scale=40.0):

        if w[j] < 1e-10:
