import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def softmax_val(corr, T, dx):
    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def softmax_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    g = np.zeros(n)
    center = n - 1
    active = np.where(w > 1e-12)[0]
    for j in active:
        k = int(j) - center
        if k == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w[j] * (1.0 - h[:n - k])
            g[:n - k] -= w[j] * h[k:]
        else:
            p = -k
            g[:n - p] += w[j] * (1.0 - h[p:])
            g[p:] -= w[j] * h[:n - p]
    return g * dx

def fista(h, n, target, T, iters):

    h_pm1 = h.copy()

    dx = 2.0 / n

        g = softmax_grad(corr, h_c, T, n)

        t = t_new

def subgrad_topk(h, n, target, iters, K=6):

    for it in range(iters):
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        top = np.argsort(corr)[-K:]
        dx = 2.0 / n
        center = n - 1
        g = np.zeros(n)
        for j in top:
            k = int(j) - center
            if k == 0:
                g += (1.0 - 2.0 * h_c)
            elif k > 0:
                g[k:] += (1.0 - h_c[:n - k])
                g[:n - k] -= h_c[k:]
            else:
                p = -k
                g[:n - p] += (1.0 - h_c[p:])
                g[p:] -= h_c[:n - p]
        g *= dx / K
        lr = 0.03 / (1.0 + 0.005 * it)
        h_new = project_sum(h_c - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_c:
            h_c, c5_c = h_new, c5_new
    return h_c, c5_c

def binary_swap_search(h, n, target, n_rounds, rng):
    c5_c = compute_c5(h)
    for rd in range(n_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_try = min(800, len(ones) * len(zeros))
        for _ in range(n_try):
            i = int(rng.integers(len(ones)))
            j = int(rng.integers(len(zeros)))
            h_try = h.copy()
            h_try[ones[i]] = 0.0
            h_try[zeros[j]] = 1.0
            c5_try = compute_c5(h_try)
            if c5_try < c5_c - 1e-12:
                h, c5_c = h_try, c5_try
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
        if not improved and rd > 3:
            break
    return h, c5_c

def slsqp_polish(h, n, target, T=2.0, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        return softmax_val(np.correlate(x, 1.0 - x, mode="full"), T, dx)
    def jac(x):
        return softmax_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_sum(res.x, target)
    except Exception:
        return h
