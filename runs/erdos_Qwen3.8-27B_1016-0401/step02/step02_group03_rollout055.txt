```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def corr_and_grad(h, n):
    """Compute correlations and their gradients w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # actual integral values

def shift_grad(h, n, shift):
    """Gradient of corr[shift] w.r.t. h, times dx."""
    dx = 2.0 / n
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad * dx
```

```python
def nesterov_subgrad(h_init, n, t0, time_budget, max_iters=500000, K=10):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best_local = compute_c5(h_init)
    h_best_local = h_init.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        # Top-K shifts
        top_idx = np.argsort(corr)[-K:]
        # Weights: softmax over top K
        top_vals = corr[top_idx]
        top_vals_shifted = top_vals - top_vals.max()
        top_w = np.exp(top_vals_shifted) / np.exp(top_vals_shifted).sum()
        
        g = np.zeros(n)
        center = n - 1
        for (t_w, t_i) in zip(top_w, top_idx):
            s = t_i - center
            if s == 0:
                g += t_w * (1.0 - 2.0 * h_y)
            elif s > 0:
                if s < n:
                    g[s:] += t_w * (1.0 - h_y[:n-s])
                if n-s > 0:
                    g[:n-s] -= t_w * h_y[s:]
            else:
                sp = -s
                if n-sp > 0:
                    g[:n-sp] += t_w * (1.0 - h_y[sp:])
                if sp < n:
                    g[sp:] -= t_w * h_y[:n-sp]
        g *= dx
        
        # Adaptive step size
        alpha = 0.5 / (1.0 + 0.01 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        
        if (it + 1) % 20000 == 0:
            c5a = compute_c5(h_avg)
            if c5a < c5_best_local:
                c5_best_local = c5a
                h_best_local = h_avg.copy()
            c5c = compute_c5(h_sg)
            if c5c < c5_best_local:
                c5_best_local = c5c
                h_best_local = h_sg.copy()
    
    return h_best_local, c5_best_local
```

```python
    h_best = h_init.copy()

            break

        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)

        tw /= tw.sum()

        h_sg = h_new

            if c5_tmp < c5_best:
```

```python
def project_feasible(y, n):
```
