import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def coord_descent_sweep(h, c, n, dx):
    """One sweep of coordinate descent. h and c are modified in place.
    c is the current correlation array (length 2n-1).
    """
    for i in range(n):
        old = h[i]
        # Compute coeff: c_new[s] = c[s] + (v-old)*coeff[s]
        # coeff[s] = (f_s - g_s) * dx
        # where f_s = 1-h[i+s] if 0<=i+s<n, g_s = h[i-s] if 0<=i-s<n
        shift = np.arange(-(n-1), n)  # actual shifts, length 2n-1
        
        # Vectorized computation of coeff
        # f: for each shift s, is i+s in [0,n)? If so, 1-h[i+s], else 0
        f_idx = i + shift  # array of length 2n-1
        f_valid = (f_idx >= 0) & (f_idx < n)
        f = np.where(f_valid, 1.0 - h[np.where(f_valid, f_idx, 0)], 0.0)
        
        g_idx = i - shift
        g_valid = (g_idx >= 0) & (g_idx < n)
        g = np.where(g_valid, h[np.where(g_valid, g_idx, 0)], 0.0)
        
        coeff = (f - g) * dx
        
        # Now minimize max_s (c + (v-old)*coeff) over v in [0,1]
        # = max_s (coeff*s * v + (c[s] - old*coeff[s]))

        if old < 0.99:
            test_vs = np.concatenate([test_vs, np.array([old+0.25*(1-old), old+0.5*(1-old), 0.75+0.25*old])])
        test_vs = np.clip(test_vs, 0.0, 1.0)
        
        best_v = old
        best_obj = np.max(c)
        for v in test_vs:
            obj = np.max(c + (v - old) * coeff)
            if obj < best_obj:
                best_obj = obj
                best_v = v
        
        # Ternary search refinement
        if 0.01 < best_v < 0.99:
            lo_b, hi_b = max(0.0, best_v - 0.3), min(1.0, best_v + 0.3)
            for _ in range(25):
