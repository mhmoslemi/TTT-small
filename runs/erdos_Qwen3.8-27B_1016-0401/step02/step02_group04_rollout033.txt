  - Similarly for h[i]

Actually, updating all deltas after a swap is O(n) (each delta[p] at specific o values changes). But in practice, it's easier to just recompute everything O(n²) each round. For n=280, O(n²) = 78K operations, which is nothing.

Let me just recompute corr and deltas from scratch each round. It's O(n²) and fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -50, 50)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -50, 50)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def _opt_obj(x, T, pen, target):
    return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2

def _opt_grad(x, T, pen, target):
    n = len(x)
    return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * np.ones(n)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def run_lbfgsb(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_opt_obj, h0.copy(), method='L-BFGS-B', jac=_opt_grad,
                       args=(T, pen, target), bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-13, 'maxls': 60})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def make_deltas(h):
    n = len(h)
    onen = n - 1
    deltas = np.zeros((n, 2 * n - 1))
    for p in range(n):
        for o in range(-(n - 1), n):
            val = 0.0
            ip = p + o
            if 0 <= ip < n:
                val += 1.0 - h[ip]
            ip2 = p - o
            if 0 <= ip2 < n:
                val -= h[ip2]
            deltas[p, onen + o] = val
    return deltas

def binary_local_search(h, target, max_rounds=8):
    n = len(h)
    h = h.copy()
    onen = n - 1
    corr = np.correlate(h, 1 - h, mode="full")
    best_max = np.max(corr)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    for rnd in range(max_rounds):
        deltas = make_deltas(h)
        improved = False
        best_gain = 1e-12
        best_pair = None
        for j in ones:
            base = corr - deltas[j]
            if not zeros:
                break
            new_maxes = np.max(base[None, :] + deltas[zeros], axis=1)
            idx = np.argmin(new_maxes)
            i = zeros[idx]
            gain = best_max - new_maxes[idx]
            if gain > best_gain:
                best_gain = gain
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[j] = 0.0
            h[i] = 1.0
            corr = np.correlate(h, 1 - h, mode="full")
            best_max = np.max(corr)
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            improved = True
        if not improved:
            break
    return h, best_max * (2.0 / n)

def subgrad_polish(h, target, n_steps=500):
    n = len(h)
    h = h.copy()
    lr = 0.3 / n
    for s in range(n_steps):
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr) - (n - 1)
        grad = np.zeros(n)
        o = k_max
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] -= h[o:]
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
        h = project_box_sum(h - lr * grad, target)
        lr *= 0.997
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Progressive soft-max L-BFGS-B
    for T in [80, 160, 320, 640, 1200, 2500, 5000]:
        if elapsed() > budget_s - 350:
            break
        h_new = run_lbfgsb(h_best, n, target, T, 5.0, 4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Subgradient polish
    if elapsed() < budget_s - 250:
        h_sub = subgrad_polish(h_best, target, 600)
        c5 = compute_c5(h_sub)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_sub

    # Phase 3: Round to binary + local search
    if elapsed() < budget_s - 200:
        h_bin = np.where(h_best > 0.5, 1.0, 0.0)
        h_bin = project_box_sum(h_bin, target)
        # Ensure exactly target ones
        if h_bin.sum() != target:
            order = np.argsort(np.abs(h_bin - 0.5))[::-1]
            for idx in order:
                if h_bin.sum() > target and h_bin[idx] == 1.0:
                    h_bin[idx] = 0.0
                elif h_bin.sum() < target and h_bin[idx] == 0.0:
                    h_bin[idx] = 1.0
                if h_bin.sum() == target:
                    break
        h_bin, c5_bin = binary_local_search(h_bin, target, max_rounds=12)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin.copy()

    # Phase 4: Perturbation + re-optimize (escape local minima)
    n_att = 0
    while elapsed() < budget_s - 180:
        n_att += 1
        scale = max(0.03, 0.1 / (1.0 + 0.06 * n_att))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T = 150 + 100 * (n_att % 4)
        h_new = run_lbfgsb(h_pert, n, target, T, 5.0, 2500)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            # Also try binary local search from this
            h_b2 = np.where(h_new > 0.5, 1.0, 0.0)
            h_b2 = project_box_sum(h_b2, target)
            if h_b2.sum() != target:
                order2 = np.argsort(np.abs(h_b2 - 0.5))[::-1]
                for idx in order2:
                    if h_b2.sum() > target and h_b2[idx] == 1.0:
                        h_b2[idx] = 0.0
                    elif h_b2.sum() < target and h_b2[idx] == 0.0:
                        h_b2[idx] = 1.0
                    if h_b2.sum() == target:
                        break
            h_b2, c5_b2 = binary_local_search(h_b2, target, max_rounds=8)
            if c5_b2 < c5_best:
                c5_best = c5_b2
                h_best = h_b2.copy()

    # Phase 5: Final binary local search from best
    if elapsed() < budget_s - 30:
        h_bf = np.where(h_best > 0.5, 1.0, 0.0)
        h_bf = project_box_sum(h_bf, target)
        if h_bf.sum() != target:
            order3 = np.argsort(np.abs(h_bf - 0.5))[::-1]
            for idx in order3:
                if h_bf.sum() > target and h_bf[idx] == 1.0:
                    h_bf[idx] = 0.0
                elif h_bf.sum() < target and h_bf[idx] == 0.0:
                    h_bf[idx] = 1.0
                if h_bf.sum() == target:
                    break
        h_bf, c5_bf = binary_local_search(h_bf, target, max_rounds=15)
        if c5_bf < c5_best:
            c5_best = c5_bf
            h_best = h_bf.copy()

    # Phase 6: Final soft-max polish (very high T)
    if elapsed() < budget_s - 10:
        h_pol = run_lbfgsb(h_best, n, target, 8000, 1.0, 2000)
        c5 = compute_c5(h_pol)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_pol

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
