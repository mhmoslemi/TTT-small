import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target} via bisection."""
    n = len(h)
    if n == 0:
        return h
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return np.clip(h, 0, 1)
    lo, hi = h.min() - 0.01, h.max() + 0.01
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_clipped = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_clipped > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj_grad(h, T):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc) - T * corr.max()  # numerical stability
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se  # softmax weights
    
    # Gradient
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= (2.0 / n)
    return obj, grad

def fista_solve(h0, target, T_schedule, n_iter, t0, budget, lr_override=None):
    """FISTA with backtracking line search for smooth max objective."""
    n = len(h0)
    x = h0.copy()
    x_prev = h0.copy()
    t_mom = 1.0
    L = lr_override if lr_override else 1.0 / max(np.linalg.norm(h0 - x_prev, ord=np.inf) * T_schedule[0] * 2 + 1, 0.1)
    best_c5 = compute_c5(x)
    best_x = x.copy()
    
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        # Determine T for this iteration
        frac = it / max(n_iter - 1, 1)
        T = T_schedule[0] * (T_schedule[-1] / T_schedule[0]) ** frac
        
        # FISTA momentum
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = x + (t_mom - 1.0) / t_new * (x - x_prev)
            t_mom = t_new
        else:
            y = x.copy()
        
        obj_y, grad_y = sm_obj_grad(y, T)
        gn = np.linalg.norm(grad_y)
        if gn < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / L
        for _bs in range(30):
            x_new = project_constraint(y - step * grad_y, target)
            obj_new, _ = sm_obj_grad(x_new, T)
            if obj_new <= obj_y + 0.5 * step * gn * gn - 1e-15:
                break
            step *= 0.5
            L *= 2.0
        else:
            break

        x = x_new
        L = max(L * 0.95, 0.01)  # slowly decrease L (allow larger steps)

            c5 = compute_c5(x)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_x = x.copy()
    
    c5 = compute_c5(x)
    if c5 < best_c5:
        best_c5 = c5
        best_x = x.copy()
    return best_x, best_c5

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_constraint(h, target):
    lo, hi = float(h.min()) - 0.01, float(h.max()) + 0.01
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = float(e.sum())
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= (2.0 / n)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_constraint(h, target):
    lo, hi = float(h.min()) - 0.01, float(h.max()) + 0.01
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = float(e.sum())
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-14:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= (2.0 / n)
    return obj, grad

def fista_solve(h0, target, T_lo, T_hi, n_iter, t0, deadline):
    n = len(h0)
    x = h0.copy()
    x_prev = h0.copy()
    mu = 1.0
    L = 4.0 * T_hi / n
    best_c5 = compute_c5(x)
    best_x = x.copy()
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = min(it / max(n_iter - 1, 1), 1.0)
        T = T_hi * (T_lo / T_hi) ** frac
        if it > 0:
            mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) * 0.5
            y = x + (mu - 1.0) / mu_new * (x - x_prev)
            mu = mu_new
        else:
            y = x.copy()
        obj_y, gy = sm_obj_grad(y, T)
        gn2 = float(np.dot(gy, gy))
        if gn2 < 1e-28:
            break
        step = 1.0 / L
        for _ in range(25):
            x_new = project_constraint(y - step * gy, target)
            obj_new, _ = sm_obj_grad(x_new, T)
            if obj_new <= obj_y + 0.5 * step * gn2 - 1e-15:
                break
            step *= 0.5
            L *= 2.0
        else:
            break
        if obj_new > obj_y + 0.05:
            x_prev = x.copy()
            mu = 1.0
            x = x_new
            continue
        x_prev = x
        x = x_new
        L = max(L * 0.85, 1e-10)
        c5_check = compute_c5(x)
        if c5_check < best_c5 - 1e-10:
            best_c5 = c5_check
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800:
            x_prev = x.copy()
            mu = 1.0
            stall = 0
    c5 = compute_c5(x)
    if c5 < best_c5:
        best_c5, best_x = c5, x.copy()
    return best_x, best_c5

def coord_pair(h, target, n_trials, t0, deadline, seed):
    n = len(h)
    rng_l = np.random.default_rng(seed)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dvals = np.array([0.02, 0.04, 0.07, 0.1, 0.15])
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = int(rng_l.integers(0, n))
        j = int(rng_l.integers(0, n))
        if i == j:
            continue
        d = dvals[int(rng_l.integers(0, len(dvals)))]
        for sgn in (1, -1):
            ni = h[i] + sgn * d
            nj = h[j] - sgn * d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
                break
            h[i], h[j] = best_h[i], best_h[j]
            if sgn == 1:
                break
    return best_h, best_c5

_SCL_TARGET = 0.0
_SQP_T = 100.0

def _scl_fun(x):
    return x.sum() - _SCL_TARGET

def _scl_jac(x):
    return np.ones(len(x))

def _sqp_fobj(x):
    return sm_obj_grad(x, _SQP_T)[0]

def _sqp_fgrad(x):
    return sm_obj_grad(x, _SQP_T)[1]

    # Phase 3: Diverse restarts + short FISTA
    d = t_start + rem * 0.70
    for r in range(10):
        if time.time() > d:
            break
        h_p = h_best.copy()
        pt = r % 5
        if pt == 0:
            hf = np.fft.rfft(h_p)

        elif pt == 2:

    return (h_best, c5_final, n)
