def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_eval(corr, T, n):
    mc = np.max(corr)
    raw = T * (corr - mc)
    sh = raw - np.max(raw)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    raw = T * (corr - mc)
    sh = raw - np.max(raw)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

        c = compute_c5(h)
        if c < c5b:
            c5b = c; hb = h.copy()

    # Phase 4: Pair swap local search
    if el() < budget_s - 100:
        h, cl = pair_swap(hb.copy(), n, target, 800, rng)
        c = compute_c5(h)
        if c < c5b:
            c5b = c; hb = h.copy()
        # Second round of pair swap with smaller steps
        if el() < budget_s - 60:
            h2, cl2 = pair_swap(hb.copy(), n, target, 500, rng)
            c2 = compute_c5(h2)
            if c2 < c5b:
                c5b = c2; hb = h2.copy()

    # Phase 5: Final low-T SLSQP
    if el() < budget_s - 40:
        for T in [1.0, 0.6, 0.4]:
            if el() > budget_s - 20:
                break
            h = slsqp(hb, n, target, T, 2000)
            c = compute_c5(h)
            if c < c5b:
                c5b = c; hb = h.copy()

    # Phase 6: Restart with perturbation from best
    if el() < budget_s - 60:
        for sc in [0.05, 0.02]:
            if el() > budget_s - 30:
                break
            h_p = project_feasible(hb + rng.normal(0, sc, n), target)
            h_p = slsqp(h_p, n, target, 3, 800)
            c = compute_c5(h_p)
            if c < c5b:
                c5b = c; hb = h_p.copy()
            h_p2, _ = subgrad(h_p, n, target, 200, 2, 0.01)
            c2 = compute_c5(h_p2)
            if c2 < c5b:
                c5b = c2; hb = h_p2.copy()

    # Phase 7: Lower resolution attempt
    if el() < budget_s - 25 and n > 160:
        for nl in [140, 200]:
            if el() > budget_s - 12:
                break
            tl = nl / 2.0
            xo = np.arange(n) * (2.0 / n)
            xn = np.linspace(0, 2.0, nl, endpoint=False)
            hl = np.interp(xn, xo, hb)
            hl = project_feasible(hl, tl)
            for T in [20, 5, 2, 1]:
                if el() > budget_s - 8:
                    break
                hl = slsqp(hl, nl, tl, T, 400)
            cl = compute_c5(hl)
            if cl < c5b:
                c5b = cl; hb = hl.copy(); n = nl

    # Phase 8: Ultra-fine subgradient
    if el() < budget_s - 10:
        h, _ = subgrad(hb, n, n / 2.0, 300, 2, 0.005)
        c = compute_c5(h)
        if c < c5b:
            c5b = c; hb = h.copy()

    c5_final = min(compute_c5(hb), c5b)
    return (hb, c5_final, len(hb))
