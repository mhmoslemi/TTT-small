            g[:n-p] += -wi * h[p:]

            hn = fix_h(np.clip(r.x, 0, 1), tgt)
            cn = compute_c5(hn)
            if cn < best - 1e-16:
                best = cn
                hb = hn
        except Exception:
            pass

    # Phase 5: Random restarts if time remains
    if el() < t4e - 15:
        for _r in range(8):
            if el() > t4e:
                break
            hq = fix_h(hb + rng.normal(0, 0.005, n), tgt)
            cq = compute_c5(hq)
            start_h = hq
            start_c = cq
            c_q = np.correlate(start_h, 1.0 - start_h, mode="full")
            w_q = np.argmax(c_q)
            try:
                r = minimize(_lag_obj, start_h, args=(w_q,), method='SLSQP',
                             jac=lambda x, l=w_q: _lag_grad(x, l),
                             bounds=bounds, constraints=_cons(),
                             options={'maxiter': 2000, 'ftol': 1e-18})
                hn = fix_h(np.clip(r.x, 0, 1), tgt)
                cn = compute_c5(hn)
                if cn < start_c:
                    start_c = cn
                    start_h = hn
            except Exception:
                pass
            if start_c < best:
                best = start_c
                hb = start_h

    return (hb, min(best, compute_c5(hb)), len(hb))
