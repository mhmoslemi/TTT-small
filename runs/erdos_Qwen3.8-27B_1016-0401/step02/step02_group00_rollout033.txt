                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-15:

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += (1.0 - x[:n-o])
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - x[p:])
        g[p:] += -x[:n-p]
    return g * (2.0 / n)

def sm_obj_pen(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    obj = mc + np.log(np.sum(np.exp(shifted))) / T
    pen = mu * (x.sum() - _EM) ** 2
    return obj + pen

def sm_grad_pen(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        w = sw[idx]
        if w < 1e-12:
            continue
        o = idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    grad *= dx
    grad += 2.0 * mu * (x.sum() - _EM)
    return grad

def eq_fun(x):
    return x.sum() - _EM

def eq_jac(x):
    return np.ones(len(x))

import numpy as np
from scipy.optimize import minimize
import time

_EM = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                diff -= a
            else:
                s = min(-diff, h[i])
                if s <= 0:
                    continue
                h[i] -= s
                diff += s
            if abs(diff) < 1e-15:
                break
    return h

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * x)
    elif o > 0:
        # CORRECTED: grad[:n-o] += (1-x[o:]); grad[o:] += -x[:n-o]
        g[:n-o] += (1.0 - x[o:])
        g[o:] += -x[:n-o]
    else:
        p = -o
        # CORRECTED: grad[:n-p] += -x[p:]; grad[p:] += (1-x[:n-p])
        g[:n-p] += -x[p:]
        g[p:] += (1.0 - x[:n-p])
    return g * (2.0 / n)

def sm_obj_pen(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    obj = mc + np.log(np.sum(np.exp(shifted))) / T
    return obj + mu * (x.sum() - _EM) ** 2

def sm_grad_pen(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = sw[idx]
        if w < 1e-12:
            continue
        o = idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[:n-o] += w * (1.0 - x[o:])
            grad[o:] += -w * x[:n-o]
        else:
            p = -o
            grad[:n-p] += -w * x[p:]
            grad[p:] += w * (1.0 - x[:n-p])
    grad *= dx
    grad += 2.0 * mu * (x.sum() - _EM)
    return grad

def eq_fun(x):
    return x.sum() - _EM

def eq_jac(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    global _EM
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _EM = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    # Phase 1: SLSQP active-set cycling with CORRECTED gradients (30%)
    p1_end = t0 + budget_s * 0.30
    while time.time() < p1_end:
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        center = n - 1
        top_k = min(12, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for lag in top_idx:
            if time.time() > p1_end:
                break
            try:
                res = minimize(lag_obj, h_best, args=(lag,), method='SLSQP',
                              jac=lag_grad, bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-18})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
                    improved = True
            except Exception:
                pass
        if not improved:
            break

    # Phase 2: L-BFGS-B smoothed max with penalty (20%)
    p2_end = t0 + budget_s * 0.50
    for T in [150, 300, 600, 1200]:
        if time.time() > p2_end:
            break
        for mu in [500.0, 5e4]:
            if time.time() > p2_end:
                break
            try:
                res = minimize(sm_obj_pen, h_best, args=(T, mu),
                              method='L-BFGS-B', jac=sm_grad_pen,
                              bounds=bounds,
                              options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-12})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 3: Multi-lag subgradient with iteration averaging (20%)
    p3_end = t0 + budget_s * 0.70
    h_sg = h_best.copy()
    h_avg = h_best.copy()
    center = n - 1
    for it in range(30000):
        if time.time() > p3_end:
            break
        corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        # Use top-3 lags weighted
        top3 = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for lag in top3:
            o = lag - center
            if o == 0:
                g += (1.0 - 2.0 * h_sg)
            elif o > 0:
                g[:n-o] += (1.0 - h_sg[o:])
                g[o:] += -h_sg[:n-o]
            else:
                p = -o
                g[:n-p] += -h_sg[p:]
                g[p:] += (1.0 - h_sg[:n-p])
        g *= 2.0 / n / 3.0
        step = 0.8 / np.sqrt(float(it + 1))
        h_new_sg = np.clip(h_sg - step * g, 0, 1)
        h_new_sg = fix_constraint(h_new_sg, target)
        h_avg = ((it + 1) * h_avg + h_new_sg) / (it + 2)
        h_sg = h_new_sg
        if it % 3000 == 0:
            c5_avg = compute_c5(fix_constraint(h_avg.copy(), target))
            if c5_avg < c5_best:
                c5_best = c5_avg
                h_best = fix_constraint(h_avg.copy(), target)
                h_sg = h_best.copy()
                h_avg = h_best.copy()

    # Phase 4: Finer discretization n=560 (15%)
    p4_end = t0 + budget_s * 0.85
    n_alt = 560
    target_alt = n_alt / 2.0
    _EM = target_alt
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
    h_alt = np.interp(x_new, x_old, h_best)
    h_alt = fix_constraint(h_alt, target_alt)
    c5_alt = compute_c5(h_alt)
    bounds_alt = [(0.0, 1.0)] * n_alt
    cons_alt = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    while time.time() < p4_end:
        corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
        worst_a = int(np.argmax(corr_a))
        try:
            res = minimize(lag_obj, h_alt, args=(worst_a,), method='SLSQP',
                          jac=lag_grad, bounds=bounds_alt, constraints=cons_alt,
                          options={'maxiter': 2000, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt - 1e-15:
                c5_alt = c5_new
                h_alt = h_new
            else:
                break
        except Exception:
            break
    if compute_c5(h_alt) < c5_best:
        c5_best = compute_c5(h_alt)
        h_best = h_alt
        n = n_alt
        target = target_alt
        _EM = target
        bounds = [(0.0, 1.0)] * n

    # Phase 5: Final aggressive polish (15%)
    p5_end = t0 + budget_s - 10.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    _EM = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    cons_f = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    stall = 0
    while time.time() < p5_end:
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            res = minimize(lag_obj, h_best, args=(worst_f,), method='SLSQP',
                          jac=lag_grad, bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 5000, 'ftol': 1e-20})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
                stall = 0
            else:
                stall += 1
                if stall > 8:
                    # Small perturbation to escape
                    pert = rng.normal(0, 0.01, n_f)
                    h_pert = fix_constraint(np.clip(h_best + pert, 0, 1), target_f)
                    h_best = h_pert
                    stall = 0
        except Exception:
            break

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
