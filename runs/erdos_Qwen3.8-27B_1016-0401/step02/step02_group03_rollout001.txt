def _jac_ls(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        if sw[oi] >= 1e-11:
            grad += sw[oi] * shift_grad(x, n, oi - center)
    return grad * dx

def _obj_wavg(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    T = _G_T
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    return float(sw @ corr) * (2.0 / n)

def _jac_wavg(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    T = _G_T
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        if sw[oi] >= 1e-11:
            grad += sw[oi] * shift_grad(x, n, oi - center)
    return grad * dx

def _cfun(x):
    return x.sum() - _G_TARGET

def _cjac(x):
    return np.ones(_G_N)

def do_opt(h, n, T, maxiter, ftol, use_wavg=False):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    obj = _obj_wavg if use_wavg else _obj_ls
    jac = _jac_wavg if use_wavg else _jac_ls
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def accel_grad(h_init, n, t0, tlimit, lr0=0.0015, decay=0.99992):
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    lr = lr0
    for it in range(150000):
        if time.time() - t0 > tlimit:
            break
        beta = min(0.9, float(it) / (it + 10.0))
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = get_corr(h_y)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(h_y, n, shift) * dx
        h_new = h_y - lr * g
        if (it + 1) % 5 == 0:
            h_new = project_feasible(h_new, n)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - h_new.sum()
            if abs(diff) > 1e-12:
                h_new = np.clip(h_new + diff / n, 0.0, 1.0)
        h_prev = h_cur
        h_cur = h_new
        lr *= decay
        if (it + 1) % 20000 == 0:
            c5c = compute_c5(project_feasible(h_cur, n))
            if c5c < c5_best:
                c5_best = c5c
                h_best = project_feasible(h_cur, n).copy()
    return h_best, c5_best

def coord_descent(h, n, t0, tlimit, max_iters=50000):
    """Iteratively adjust single coordinates to reduce max correlation."""
    dx = 2.0 / n
    target = n / 2.0
    h_c = h.copy()
    c5_best = compute_c5(h_c)
    h_best = h_c.copy()
    step = 0.01
    for it in range(max_iters):
        if time.time() - t0 > tlimit:
            break
        i = it % n
        corr = get_corr(h_c)
        mc = np.max(corr)
        # Try increasing and decreasing h[i]
        best_delta = 0.0
        for delta in [step, step/2, -step, -step/2, 2*step, -2*step]:
            hv = h_c.copy()
            hv[i] = np.clip(hv[i] + delta, 0.0, 1.0)
            # Fix sum
            ds = target - hv.sum()
            if abs(ds) > 1e-12:
                hv = np.clip(hv + ds / n, 0.0, 1.0)
            c = float(np.max(get_corr(hv))) * dx
            if c < mc - 1e-10:
                best_delta = delta
                mc = c
        if best_delta != 0.0:
            h_c[i] = np.clip(h_c[i] + best_delta, 0.0, 1.0)
            ds = target - h_c.sum()
            if abs(ds) > 1e-12:
                h_c = np.clip(h_c + ds / n, 0.0, 1.0)
            c5c = compute_c5(h_c)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_c.copy()
        if (it + 1) % n == 0:
            step = max(0.001, step * 0.85)
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Multi-temperature SLSQP (log-sum-exp)
    temps = [256, 128, 64, 32, 16, 8, 4, 2, 1, 0.5, 0.25]
    for T in temps:
        if time.time() - t0 > budget_s - 450:
            break
        h_new = do_opt(h_best, n, T, 4000, 1e-18, use_wavg=False)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Weighted-average objective (different landscape)
    for T in [16, 4, 1, 0.5]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = do_opt(h_best, n, T, 3000, 1e-18, use_wavg=True)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Accelerated gradient on max correlation
    h_gd, c5_gd = accel_grad(h_best, n, t0, budget_s - 300, 0.002, 0.99993)
    if c5_gd < c5_best:
        c5_best = c5_gd
        h_best = h_gd

    # Phase 4: SLSQP polish
    for T in [0.5, 1, 2]:
        if time.time() - t0 > budget_s - 120:
            break
        h_new = do_opt(h_best, n, T, 5000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Coordinate descent for fine tuning
    h_cd, c5_cd = coord_descent(h_best, n, t0, budget_s - 100, 40000)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = h_cd

    # Phase 6: Perturbation restarts
    for pi in range(50):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.002 + 0.003 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_opt(h_p, n, 2, 2500, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 7: Try n=320 if time
    if time.time() - t0 < budget_s - 100:
        n2 = 320
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n2
        for T in [4, 1, 0.5]:
            if time.time() - t0 > budget_s - 20:
                break
            h_new = do_opt(h_best, n, T, 3500, 1e-18)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
