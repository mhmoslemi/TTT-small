    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    h_bin[order[:k]] = 1.0
    if k < n:
        h_bin[order[k]] = target - k
    return h_bin

    zeros = list(np.where(h_bin < 0.5)[0])

    reps = n // period
    h[:reps * period] = np.tile(base, reps)
    if n % period > 0:

    return h

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def subgrad_polish(h, n, target, n_steps, K=5, t_end=None):
    c5_local = compute_c5(h)
    for it in range(n_steps):
        if t_end is not None and time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-K:]
        center = n - 1
        g = np.zeros(n)
        for idx in top_idx:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g = g * (2.0 / n) / K
        lr = 0.08 / (1.0 + it * 0.15)
        h_cand = project_feasible(h - lr * g, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_local:
            c5_local = c5_cand
            h = h_cand
        else:

            h_cand = project_feasible(h - lr * g, target)

            if c5_cand < c5_local:

                h = h_cand

def slsqp_refine(h, n, target, T, maxiter, t_end):
    dx = 2.0 / n
    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = sh - np.max(sh)
        e = np.exp(sh)
        w = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        nz = np.where(w > 1e-12)[0]
        for idx in nz:
            o = int(idx) - center
            wi = w[idx]
            if o == 0:
                grad += wi * (1.0 - 2.0 * x)
            elif o > 0:
                grad[o:] += wi * (1.0 - x[:n - o])
                grad[:n - o] -= wi * x[o:]
            else:
                p = -o
                grad[:n - p] += wi * (1.0 - x[p:])
                grad[p:] -= wi * x[:n - p]
        return grad * dx
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h
