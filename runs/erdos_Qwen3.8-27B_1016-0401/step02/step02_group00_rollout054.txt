import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                if diff <= 0: break
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= 0: break
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def compute_deltas(h):
    """Compute delta_remove[i] for h[i]=1 and delta_add[j] for h[j]=0"""
    n = len(h)
    corr = get_corr(h)
    center = n - 1
    one_pos = np.where(h > 0.5)[0]
    zero_pos = np.where(h < 0.5)[0]
    
    # For a binary sequence, delta_remove[i] when h[i]: 1->0:
    # At lag k: 
    #   if k != 0: -(1-h[i-k])*I[0<=i-k<n] + h[i+k]*I[0<=i+k<n]
    #   if k == 0: 0
    # (using the formula derived above, with h being the OLD values)
    
    # Actually, let me just compute numerically for correctness
    delta_remove = {}
    delta_add = {}
    
    for i in one_pos:
        h[i] = 0.0
        new_corr = get_corr(h)
        delta_remove[i] = new_corr - corr
        h[i] = 1.0
    
    for j in zero_pos:
        h[j] = 1.0
        new_corr = get_corr(h)
        delta_add[j] = new_corr - corr
        h[j] = 0.0
    
    return corr, delta_remove, delta_add, one_pos, zero_pos

def binary_local_search(h, max_iters=100, time_limit=None, t_start=None):
    """Hill climbing on 0/1 sequence using single swaps"""
    n = len(h)
    h = np.where(h >= 0.5, 1.0, 0.0).copy()
    target = n / 2.0
    h = fix_sum(h, target)  # ensure exactly n/2 ones
    
    c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time_limit and (t_start + time_limit) < time.time():
            break
        corr, delta_remove, delta_add, one_pos, zero_pos = compute_deltas(h)
        old_max = np.max(corr)
        
        best_swap = None
        best_new_max = old_max + 1e-10  # must improve
        
        for i in one_pos:
            for j in zero_pos:
                new_corr = corr + delta_remove[i] + delta_add[j]
                nm = np.max(new_corr)
                if nm < best_new_max:
                    best_new_max = nm
                    best_swap = (i, j)
        
        if best_swap is None:
            break  # local minimum
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
    
    return h, c5

            if min_val < best_val - 1e-15:
                best_val = min_val
                best_i = one_pos[idx_i]
                best_j = zero_pos[np.argmin(new_maxes)]
        if best_i < 0:
            stuck += 1
            if stuck > 8:
                break
            if rng is not None:
                i = one_pos[rng.integers(len(one_pos))]
                j = zero_pos[rng.integers(len(zero_pos))]
                h[i] = 0.0
                h[j] = 1.0
                corr = get_corr(h)
                c5 = compute_c5(h)
            continue
        h[best_i] = 0.0
        h[best_j] = 1.0
        corr = get_corr(h)
        c5 = compute_c5(h)
        stuck = 0
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    global _g_n, _g_target, _g_T
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    _g_n = n
    _g_target = n / 2.0
    h_best = fix_sum(h_best, _g_target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP soft-max warm-up (15% budget)
    p1_end = t0 + budget_s * 0.15
    for T in [100, 300, 800]:
        if time.time() > p1_end:
            break
        _g_T = float(T)
        cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
        try:
            res = minimize(sm_obj, h_best, method='SLSQP', jac=sm_grad,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = fix_sum(np.clip(res.x, 0, 1), _g_target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 2: Binary local search on main n (45% budget)
    p2_end = t0 + budget_s * 0.60
    h_bin, c5_bin = binary_local_search(h_best, max_iters=500, t_deadline=p2_end, rng=rng)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin

    # Multi-start: try a few random binary configurations (10% budget)
    p3_end = t0 + budget_s * 0.70
    for trial in range(3):
        if time.time() > p3_end:
            break
        h_rand = np.zeros(n)
        positions = rng.permutation(n)[:n//2]
        h_rand[positions] = 1.0
        h_r, c5_r = binary_local_search(h_rand, max_iters=100, t_deadline=p3_end, rng=rng)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r

    # Phase 3: Continuous SLSQP polish (15% budget)
    p4_end = t0 + budget_s * 0.85
    n = len(h_best)
    _g_n = n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for T in [2000, 5000, 10000]:
        if time.time() > p4_end:
            break
        _g_T = float(T)
        cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
        try:
            res = minimize(sm_obj, h_best, method='SLSQP', jac=sm_grad,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = fix_sum(np.clip(res.x, 0, 1), _g_target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Try alternate n values (15% budget)
    p5_end = t0 + budget_s - 10.0
    n_main = len(h_best)
    for n_alt in [276, 288, 300, 336, 252]:
        if time.time() > p5_end:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(n_main) * (2.0 / n_main)
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_sum(h_alt, target_alt)
        h_alt_b, c5_alt = binary_local_search(h_alt, max_iters=80, t_deadline=p5_end, rng=rng)
        _g_n = n_alt
        _g_target = target_alt
        bounds_alt = [(0.0, 1.0)] * n_alt
        _g_T = 3000.0
        cons_alt = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
        try:
            res = minimize(sm_obj, h_alt_b, method='SLSQP', jac=sm_grad,
                         bounds=bounds_alt, constraints=cons_alt,
                         options={'maxiter': 1000, 'ftol': 1e-16})
            h_new = fix_sum(np.clip(res.x, 0, 1), target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt_b = h_new
        except Exception:
            pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt_b
            n_main = n_alt
            _g_n = n_main
            _g_target = n_main / 2.0

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
