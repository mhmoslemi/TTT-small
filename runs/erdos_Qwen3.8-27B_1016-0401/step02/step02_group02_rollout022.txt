        if _elapsed() > budget_s - 150:
            break
        scale = 0.02 + 0.04 * (it % 5) / 5.0
        h_try, c5_try = _block_noise(h_best, n, target, rng, scale, c5_best)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()
            # Quick SLSQP polish
            if _elapsed() < budget_s - 160:
                h_new = _slsqp_optimize(h_best, n, target, 3.0, 800)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new.copy()

    # Phase 4: Pair swap local search
    if _elapsed() < budget_s - 60:
        h_best, c5_best = _pair_swap_search(h_best, n, target, rng, 300, c5_best)

    # Phase 5: Final SLSQP at very low T
    if _elapsed() < budget_s - 30:
        for T in [0.5, 0.3, 0.2]:
            if _elapsed() > budget_s - 15:
                break
            h_new = _slsqp_optimize(h_best, n, target, T, 8000)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()

    # Phase 6: One more subgradient pass from best
    if _elapsed() < budget_s - 10:
        h = h_best.copy()
        for it in range(200):
            lr = 0.05 / (1.0 + 0.01 * it)
            h_new, c5_new = _subgrad_step(h, n, target, 2, lr)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
            h = h_new

    h_best = project_feasible(h_best, target)
    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)

    return (h_best, c5_best, n)
