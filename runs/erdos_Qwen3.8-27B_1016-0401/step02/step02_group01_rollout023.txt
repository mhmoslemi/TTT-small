def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def active_max_values(h, T, topm):
    """Softmax over top m shifts only."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    # Find top m values
    idx = np.argpartition(corr, -topm)[-topm:]
    mc = corr[idx]
    # Softmax over these top m
    sh = T * (mc - mc.max())
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc.max() + np.log(se) / T) * (2.0 / n)
    sw = e / se  # weights over top m
    return obj, sw, idx

def active_max_grad(h, sw, idx, topm):
    """Gradient using only top m shifts."""
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in idx:
        w = sw[j - idx[0]]  # weight for this shift
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            p = -k
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    grad *= (2.0 / n)
    # Normalize by sum of weights (should be 1)
    return grad

        grad[:n-p] += (1.0 - h[p:]) * dx

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    m = np.max(corr)
    return float(m * dx)

def compute_c5_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return corr * dx, int(np.argmax(corr))

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj_grad(h, T, topm):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    # Get top m indices
    if topm >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -topm)[-topm:]
    vals = corr[idx]
    vmax = vals.max()
    ex = np.exp(T * (vals - vmax))
    se = ex.sum()
    obj = (vmax + np.log(se)/T) * (2.0/n)
    w = ex / se
    # Gradient
    grad = np.zeros(n)
    c = n - 1
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-15:
            continue
        j = idx[ii]
        k = j - c
        if k == 0:
            grad += wi * (1.0 - 2.0 * h) * (2.0/n)
        elif k > 0:

            p = -k
            grad[:n-p] += wi * (1.0 - h[p:]) * (2.0/n)

def softmax_topm(h, T, topm):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = len(corr)
    if topm >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(corr, -topm)[-topm:]
    vals = corr[idx]
    vmax = vals.max()
    ex = np.exp(T * (vals - vmax))
    se = ex.sum()
    obj = (vmax + np.log(se) / T) * (2.0 / n)
    w = ex / se
    return obj, w, idx

def softmax_topm_grad(h, w, idx):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    dx2 = 2.0 / n
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-15:
            continue
        k = idx[ii] - c
        if k == 0:
            grad += wi * (1.0 - 2.0 * h) * dx2
        elif k > 0:
            grad[k:] += wi * (1.0 - h[:n - k]) * dx2
            grad[:n - k] -= wi * h[k:] * dx2
        else:
            p = -k
            grad[:n - p] += wi * (1.0 - h[p:]) * dx2
            grad[p:] -= wi * h[:n - p] * dx2
    return grad

def pg_optimize(h0, target, T, topm, n_iter, lr0, t0, budget):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        obj, w, idx = softmax_topm(h, T, topm)
        grad = softmax_topm_grad(h, w, idx)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = h
        for _ in range(50):
            h_new = project_constraint(h - lr * grad, target)
            obj_new, _, _ = softmax_topm(h_new, T, topm)
            if obj_new < obj - 1e-16:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        lr = min(lr * 1.06, lr0)
        if it % 80 == 79:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_refine(h0, target, t0, budget, n_trials, rng):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    deltas = np.array([-0.04, -0.025, -0.015, -0.01, 0.01, 0.015, 0.025, 0.04])
    nd = len(deltas)
    for _ in range(n_trials):
        if time.time() - t0 > budget:
            break
        i = rng.integers(0, n)
        d = deltas[rng.integers(0, nd)]
        old = h[i]
        nv = np.clip(old + d, 0.0, 1.0)
        if nv == old:
            continue
        h[i] = nv
        hp = project_constraint(h, target)
        c5 = compute_c5(hp)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            h = hp
        else:
            h[i] = old
    return h, best_c5

def pair_refine(h0, target, t0, budget, n_trials, rng):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    deltas = np.array([-0.02, -0.01, 0.01, 0.02])
    nd = len(deltas)
    for _ in range(n_trials):
        if time.time() - t0 > budget:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        d = deltas[rng.integers(0, nd)]
        oi, oj = h[i], h[j]
        ni = np.clip(oi + d, 0.0, 1.0)
        nj = np.clip(oj - d, 0.0, 1.0)
        if ni == oi and nj == oj:
            continue
        h[i], h[j] = ni, nj
        hp = project_constraint(h, target)
        c5 = compute_c5(hp)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            h = hp
        else:
            h[i], h[j] = oi, oj
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 560
        h_best = np.zeros(n0)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Active-set PG at n=560, sweeping T and topm
    for topm in [3, 5, 8]:
        for T in [60, 150, 400, 1000, 2000, 5000]:
            if time.time() - t0 > budget - 400:
                break
            ht, ct = pg_optimize(h_best, target, T, topm, 1200, 0.025, t0, budget - 350)
            if ct < c5_best:
                c5_best = ct
                h_best = ht

    # Phase 2: Coordinate and pair refinement at n=560
    if time.time() - t0 < budget - 150:
        ht, ct = coord_refine(h_best, target, t0, budget - 130, 8000, rng)
        if ct < c5_best:
            c5_best = ct
            h_best = ht
        ht, ct = pair_refine(h_best, target, t0, budget - 100, 4000, rng)
        if ct < c5_best:
            c5_best = ct
            h_best = ht

    # Phase 3: Upsample to n*2 and optimize
    if time.time() - t0 < budget - 250:
        n2 = n * 2
        target2 = n2 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x2, x_old, h_best)
        h2 = project_constraint(h2, target2)
        for topm in [3, 5]:
            for T in [100, 300, 800, 2000, 5000]:
                if time.time() - t0 > budget - 80:
                    break
                ht, ct = pg_optimize(h2, target2, T, topm, 800, 0.015, t0, budget - 70)
                if ct < c5_best:
                    c5_best = ct
                    h_best = ht
                    n = len(h_best)
                    target = n / 2.0
        if time.time() - t0 < budget - 50:
            ht, ct = coord_refine(h_best, target, t0, budget - 30, 4000, rng)
            if ct < c5_best:
                c5_best = ct
                h_best = ht

    # Phase 4: Restart from perturbed best, re-optimize
    for r in range(5):
        if time.time() - t0 > budget - 120:
            break
        h_r = h_best.copy()
        if r == 0:
            h_r += rng.normal(0, 0.02, len(h_r))
        elif r == 1:
            idx = rng.integers(0, len(h_r), 40)
            h_r[idx] = 1.0 - h_r[idx]
        elif r == 2:
            sh = rng.integers(2, 10)
            h_r = np.roll(h_r, sh)
        elif r == 3:
            mask = rng.random(len(h_r)) < 0.1
            h_r[mask] = 1.0 - h_r[mask]
        else:
            h_r = (h_r > rng.uniform(0.3, 0.7)).astype(float)
        h_r = project_constraint(h_r, target)
        for T in [500, 2000]:
            if time.time() - t0 > budget - 30:
                break
            ht, ct = pg_optimize(h_r, target, T, 5, 800, 0.02, t0, budget - 20)
            if ct < c5_best:
                c5_best = ct
                h_best = ht

    # Phase 5: Final ultra-high-T refinement
    if time.time() - t0 < budget - 10:
        ht, ct = pg_optimize(h_best, target, 10000, 3, 2000, 0.005, t0, budget - 3)
        if ct < c5_best:
            c5_best = ct
            h_best = ht
        ht, ct = coord_refine(h_best, target, t0, budget - 1, 2000, rng)
        if ct < c5_best:
            c5_best = ct
            h_best = ht

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
