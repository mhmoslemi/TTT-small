    s = T * (c - mc)

            p = -o

def lbfgsb_opt(h0, target, T, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return smax_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return smax_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-16, 'maxls': 50})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def slsqp_opt(h0, target, T, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smax_obj(x, T)
    def grad(x):
        return smax_grad(x, T)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = res.x.copy()
        if abs(h_out.sum() - target) > 0.5:
            return h0.copy()
        return project_box_sum(h_out, target)
    except Exception:
        return h0.copy()

def make_bump_start(n, centers, widths):
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False)
    for c, w in zip(centers, widths):
        h += np.exp(-((x - c) ** 2) / (2 * w ** 2))
    h = project_box_sum(np.clip(h, 0, 1), n / 2.0)
    return h

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Aggressive L-BFGS-B with increasing temperature
    for T in [300, 1000, 3000, 8000, 20000, 50000]:
        if elapsed() > budget_s - 120:
            break
        h_new = lbfgsb_opt(h_best, target, T, 200.0, 2500)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: SLSQP exact-constraint polish
    if elapsed() < budget_s - 80:
        for T in [2000, 10000]:
            if elapsed() > budget_s - 80:
                break
            h_new = slsqp_opt(h_best, target, T, 400)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 3: Structured perturbations + re-optimize
    attempts = 0
    stuck_count = 0
    while elapsed() < budget_s - 100:
        attempts += 1
        strat = attempts % 5
        if strat == 0:
            h_pert = h_best + rng.normal(0, 0.04, n)
        elif strat == 1:
            shift = int(rng.integers(1, 4))
            h_pert = np.roll(h_best, shift).copy()
        elif strat == 2:
            h_pert = h_best.copy()
            start = int(rng.integers(0, max(1, n - 30)))
            length = int(rng.integers(8, 30))
            h_pert[start:start + length] = 1.0 - h_pert[start:start + length]
        elif strat == 3:
            freq = rng.uniform(0.5, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n)
            h_pert = h_best + 0.04 * np.sin(2 * np.pi * freq * x / n + phase)
        else:
            # Targeted: perturb where overlap is high
            c = corr_dx(h_best)
            top_region = np.argsort(c)[::-1][:20]
            offsets = top_region - (n - 1)
            h_pert = h_best.copy()
            for o in offsets[:10]:
                o_int = int(o)
                if o_int > 0 and o_int < n:
                    h_pert[o_int:] += rng.normal(0, 0.03, n - o_int)
                elif o_int < 0 and -o_int < n:
                    h_pert[:-o_int] += rng.normal(0, 0.03, -o_int)
        
        h_pert = project_box_sum(np.clip(h_pert, 0, 1), target)
        T = [500, 2000, 8000, 20000, 50000][int(rng.integers(0, 5))]
        maxit = 2000 if elapsed() < budget_s - 150 else 800
        h_new = lbfgsb_opt(h_pert, target, T, 100.0, maxit)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            stuck_count = 0
        else:
            stuck_count += 1

    # Phase 4: Try nearby resolutions
    for dn in [16, 32, -16, 48, 64, -32, 80]:
        if elapsed() > budget_s - 40:
            break
        n_try = n + dn
        if n_try < 40 or n_try % 2 != 0:
            continue
        t_try = n_try / 2.0
        h_try = interp_to_n(h_best, n_try)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        for T in [1000, 5000, 20000]:
            if elapsed() > budget_s - 15:
                break
            h_t = lbfgsb_opt(h_try, t_try, T, 200.0, 1500)
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n = n_try
            target = t_try

    # Phase 5: Alternative starting points if still above target
    if c5_best > 0.380860 and elapsed() < budget_s - 60:
        starts = [
            ([0.5, 1.5], [0.25, 0.25]),
            ([0.3, 1.0, 1.7], [0.18, 0.18, 0.18]),
            ([0.4, 1.2], [0.22, 0.22]),
        ]
        for centers, widths in starts:
            if elapsed() > budget_s - 20:
                break
            h_alt = make_bump_start(n, centers, widths)
            for T in [1000, 10000]:
                if elapsed() > budget_s - 10:
                    break
                h_new = lbfgsb_opt(h_alt, target, T, 200.0, 3000)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Final ultra-polish
    if elapsed() < budget_s - 3:
        h_new = lbfgsb_opt(h_best, target, 30000, 500.0, 5000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        # SLSQP final
        if elapsed() < budget_s - 1:
            h_new = slsqp_opt(h_best, target, 50000, 200)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    c5_best = min(c5_best, compute_c5(h_best))
    return (h_best, c5_best, len(h_best))
