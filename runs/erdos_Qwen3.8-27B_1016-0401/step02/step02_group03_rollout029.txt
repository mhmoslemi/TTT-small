# Global state for SLSQP callbacks
_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    T = _G['T']
    sh = T * (corr - mc)
    e = np.exp(sh)
    es = e.sum()
    return (mc + np.log(es) / T) * (2.0 / _G['n'])

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    T = _G['T']
    n = _G['n']
    sh = T * (corr - mc)
    e = np.exp(sh)
    es = e.sum()
    sw = e / es
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad_raw(x, n, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G['target']

def _g_cons_jac(x):
    return np.ones(_G['n'])

def do_slsqp(h, n, T, maxiter, ftol):
    _G['n'] = n
    _G['target'] = n / 2.0
    _G['T'] = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0

        center = n - 1

        top_k.sort()
        g = np.zeros(n)

        g *= dx / K

        h_new = project_feasible(h_new, n)

        h_sg = h_new

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _full_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    T = _G['T']
    n = _G['n']
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _full_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    T = _G['T']
    n = _G['n']
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad_raw(x, n, oi - center)
    return grad * dx

    vals = corr[_G['act_idx']]
    mc = vals.max()

    n = _G['n']
    sh = T * (vals - mc)
    e = np.exp(sh)

def _active_jac(x):

    vals = corr[_G['act_idx']]
    mc = vals.max()

    n = _G['n']

    return x.sum() - _G['target']

def do_slsqp(h, n, T, maxiter, ftol, active_idx=None):
    _G['n'] = n
    _G['target'] = n / 2.0
    _G['T'] = float(T)
    if active_idx is not None:
        _G['act_idx'] = np.array(active_idx, dtype=int)
        obj_f, jac_f = _active_obj, _active_jac
    else:
        obj_f, jac_f = _full_obj, _full_jac
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(obj_f, h, method='SLSQP', jac=jac_f,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

            break

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")

        g = np.zeros(n)
