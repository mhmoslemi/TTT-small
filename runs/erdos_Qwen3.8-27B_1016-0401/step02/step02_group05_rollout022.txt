import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def sum_fun(x, target):
    return x.sum() - target

def sum_jac(x, target):
    return np.ones(len(x))

def local_search(h, target, n_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    best = float(np.max(corr) * dx)
    hc = h.copy()
    stall = 0
    delta = 0.12
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 180:
            delta *= 0.5
            if delta < 0.006:
                break
            stall = 0
        ks = np.argmax(corr)
        s = ks - (n - 1)
        p = abs(s)
        improved = False
        if p == 0:
            vals = hc * (1.0 - hc)
            idx = np.argmax(vals)
            for d in [delta, delta * 2, 0.3, 0.5]:
                ht = hc.copy()
                ht[idx] += d if hc[idx] < 0.5 else -d
                ht = project_constraint(ht, target)
                c = np.correlate(ht, 1 - ht, mode="full")
                c5t = float(np.max(c) * dx)
                if c5t < best - 1e-14:
                    best, hc, corr, stall, improved = c5t, ht, c, 0, True
                    break
        else:
            if s > 0:
                contrib = hc[p:] * (1.0 - hc[:n - p])
            else:
                contrib = hc[:n - p] * (1.0 - hc[p:])
            top_k = min(6, len(contrib))
            top_idx = np.argsort(-contrib)[:top_k]
            for ti in top_idx:
                j1, j2 = (ti + p, ti) if s > 0 else (ti, ti + p)
                if hc[j1] < delta * 0.4 or hc[j2] > 1 - delta * 0.4:
                    continue
                for d in [delta, delta * 0.5, 0.2, 0.35]:
                    mv = min(d, hc[j1], 1.0 - hc[j2])
                    if mv < 0.004:
                        continue
                    ht = hc.copy()
                    ht[j1] -= mv
                    ht[j2] += mv
                    ht = project_constraint(ht, target)
                    c = np.correlate(ht, 1 - ht, mode="full")
                    c5t = float(np.max(c) * dx)
                    if c5t < best - 1e-14:
                        best, hc, corr, stall, improved = c5t, ht, c, 0, True
                        break
                if improved:
                    break
        if not improved:
            stall += 1
    return hc, best

    try:
        res = minimize(soft_max_obj, h, method='SLSQP', jac=soft_max_jac,
                       args=(T,), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 50:
            return project_constraint(res.x, target)
    except Exception:
        pass
    return h

    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(sum_fun, target=target),
             'jac': partial(sum_jac, target=target)}]

    # Phase 1: Aggressive local search
    h_ls, c5_ls = local_search(h_best, target, 8000, t0, budget_s * 0.2)
    if c5_ls < c5_best:
        c5_best, h_best = c5_ls, h_ls

    # Phase 2: SLSQP annealing
    for T in [150, 80, 40, 20, 10, 5, 2.5, 1]:
        if elapsed() > budget_s - 380:
            break
        h_new = slsqp_step(h_best, target, T, bounds, cons, maxiter=2500)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-13:
            c5_best, h_best = c5, h_new
        h_ls, c5_ls = local_search(h_best, target, 2000, t0, budget_s * 0.4)
        if c5_ls < c5_best:
            c5_best, h_best = c5_ls, h_ls

    # Phase 3: Random restarts with diverse perturbations
    while elapsed() < budget_s - 180:
        r = rng.integers(0, 6)
        if r == 0:
            h_init = h_best + rng.normal(0, 0.07, n)
        elif r == 1:
            h_init = h_best + rng.normal(0, 0.03, n)
        elif r == 2:
            h_init = h_best.copy()
            st = int(rng.integers(0, max(1, n - 40)))
            ln = int(rng.integers(4, min(40, n - st)))
            h_init[st:st + ln] = 1.0 - h_init[st:st + ln]
        elif r == 3:
            h_init = h_best.copy()
            for _ in range(int(rng.integers(2, 7))):
                st = int(rng.integers(0, max(1, n - 12)))
                ln = int(rng.integers(2, min(10, n - st)))
                h_init[st:st + ln] = 1.0 - h_init[st:st + ln]
        elif r == 4:
            # Swap two blocks
            h_init = h_best.copy()
            b1 = int(rng.integers(0, n // 2))
            b2 = int(rng.integers(n // 2, n - 10))
            ln = int(rng.integers(3, min(15, n - max(b1, b2))))
            h_init[b1:b1+ln], h_init[b2:b2+ln] = h_init[b2:b2+ln].copy(), h_init[b1:b1+ln].copy()
        else:
            h_init = (rng.random(n) >= 0.5).astype(float)

        h_init = project_constraint(h_init, target)
        h_new = slsqp_step(h_init, target, 20, bounds, cons, maxiter=800)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-13:
            c5_best, h_best = c5, h_new
        h_ls, c5_ls = local_search(h_best, target, 800, t0, budget_s - 130)
        if c5_ls < c5_best:
            c5_best, h_best = c5_ls, h_ls

    # Phase 4: Fine SLSQP + final local search
    for T in [4, 1.5, 0.7]:
        if elapsed() > budget_s - 35:
            break
        h_new = slsqp_step(h_best, target, T, bounds, cons, maxiter=4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-13:
            c5_best, h_best = c5, h_new

        c5_best, h_best = c5_ls, h_ls

    return (h_best, c5_final, len(h_best))
