from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_obj(h, T, topk):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
    else:
        vals = corr
    m = vals.max()
    s = T * (vals - m)
    s = np.clip(s, -50, 50)
    return (m + np.log(np.exp(s).sum()) / T) * (2.0 / n)

def sm_grad(h, T, topk):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    if topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        wfull = np.zeros(len(corr))
        vals = corr[idx]
        m = vals.max()
        s = np.clip(T * (vals - m), -50, 50)
        e = np.exp(s)
        wfull[idx] = e / e.sum()
    else:
        m = corr.max()
        s = np.clip(T * (corr - m), -50, 50)
        e = np.exp(s)
        wfull = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(1 - n, n):
        w = wfull[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n-o])
            g[:n-o] -= w * h[o:]
        else:
            p = -o
            g[p:] += w * (1.0 - h[:n-p])
            g[:n-p] -= w * h[p:]
    return g * dx

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def lbfgs(h0, target, T, topk, pen, maxiter):
    n = len(h0)
    ones = np.ones(n)
    def obj(x):
        return sm_obj(x, T, topk) + pen * (x.sum() - target) ** 2
    def grad(x):
        return sm_grad(x, T, topk) + 2 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                      bounds=[(0,1)]*n, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_box_sum(res.x, target)
    except:
        return h0.copy()
