def get_corr(h):

def sm_obj(corr, T, k_top=0):
    if k_top > 0 and k_top < len(corr):
        idx = np.argpartition(corr, len(corr) - k_top)[len(corr) - k_top:]
        vals = corr[idx]
        vmax = vals.max()
        sh = T * (vals - vmax)
        sh -= sh.max()
        e = np.exp(sh)
        se = e.sum()
        obj = vmax + np.log(se) / T
        w = np.zeros(len(corr))
        w[idx] = e / se
    else:
        mc = corr.max()
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        se = e.sum()
        obj = mc + np.log(se) / T
        w = e / se
    return obj, w

def grad_w(h, w):
    n = len(h)
    c = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        m = j - c
        if m == 0:
            g += wj * (1.0 - 2.0 * h)
        elif m > 0:
            g[m:] += wj * (1.0 - h[:n-m])
            g[:n-m] -= wj * h[m:]
        else:
            p = -m
            g[:n-p] += wj * (1.0 - h[p:])
            g[p:] -= wj * h[:n-p]
