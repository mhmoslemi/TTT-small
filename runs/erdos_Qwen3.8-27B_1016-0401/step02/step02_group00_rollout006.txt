    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]

            if abs(diff) < 1e-13:
                break

                add = min(diff, 1.0 - h[i])

    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sub = T * (corr[active_idx] - mc)
    val = mc + np.log(np.sum(np.exp(sub))) / T
    return val * (2.0 / n) + lam * (x.sum() - target) ** 2

            p = -o
            grad[:n - p] += w * (1.0 - x[p:])

def compute_c5(h):

    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            n_free = mask.sum()
            if n_free == 0:
                break

            if uniform > 1e-15:
                h[mask] += uniform
                diff -= uniform
        else:
            mask = h > 0.0
            n_free = mask.sum()
            if n_free == 0:
                break
            uniform = min(-diff / n_free, h[mask]).min() if n_free > 0 else 0
            if uniform > 1e-15:

            else:

            h[i] += add
            diff -= add
