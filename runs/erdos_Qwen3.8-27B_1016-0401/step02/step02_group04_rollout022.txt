def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_box_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sm_obj_and_grad(h, T, target, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    for i in range(len(sw)):
        w = sw[i]
        if w < 1e-14:
            continue
        k = i - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= dx
    s = h.sum() - target
    val += pen * s * s
    grad += 2.0 * pen * s
    return val, grad

                       options={'maxiter': maxiter, 'gtol': 1e-16, 'ftol': 1e-18})

        return h0.copy()

            grad[:n - p] += 1.0 - h[p:]

        c5_best = c5_final

    return (h_best, c5_best, len(h_best))
