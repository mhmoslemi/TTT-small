import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_single_shift(h, n, m):
    """Gradient of corr_m w.r.t. h."""
    j = np.arange(n)
    grad = np.zeros(n)
    # (1-h[j-m]) for 0 <= j-m < n
    valid1 = (j - m >= 0) & (j - m < n)
    grad[valid1] += (1.0 - h[j[valid1] - m])
    # -h[j+m] for 0 <= j+m < n
    valid2 = (j + m >= 0) & (j + m < n)
    grad[valid2] -= h[j[valid2] + m]
    return grad

def topm_subgrad(h, n, M):
    """Average subgradient over top-M shifts."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    idxs = np.argsort(corr)[-M:]
    grad = np.zeros(n)
    for idx in idxs:
        m = int(idx) - (n - 1)
        grad += grad_single_shift(h, n, m)
    return grad / M * dx

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-14:
            continue
        m = oi - center
        grad += w * grad_single_shift(x, _G_N, m)
    return grad * dx

def _g_cons(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_single_shift(h, n, m):
    j = np.arange(n)
    grad = np.zeros(n)
    v1 = (j - m >= 0) & (j - m < n)
    grad[v1] += (1.0 - h[j[v1] - m])
    v2 = (j + m >= 0) & (j + m < n)
    grad[v2] -= h[j[v2] + m]
    return grad

def topm_subgrad(h, n, M):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    idxs = np.argsort(corr)[-M:]
    grad = np.zeros(n)
    for idx in idxs:
        m = int(idx) - (n - 1)
        grad += grad_single_shift(h, n, m)
    return grad / M * dx

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-14:
            continue
        m = oi - center
        grad += w * grad_single_shift(x, _G_N, m)
    return grad * dx

    return x.sum() - _G_TARGET

def _g_cjac(x):
    return np.ones(_G_N)

    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons, 'jac': _g_cjac}]
    try:
        res = minimize(_g_obj, h.copy(), method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
