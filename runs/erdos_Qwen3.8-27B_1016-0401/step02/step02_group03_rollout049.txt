    j = np.arange(n)

            m1 = j < n - lag
            m2 = j >= lag
            if m1.any():
                grad[m1] += (1.0 - h[j[m1] + lag])
            if m2.any():
                grad[m2] -= h[j[m2] - lag]
        else:
            s = -lag
            m1 = j + s < n
            m2 = j >= s
            if m1.any():
                grad[m1] += (1.0 - h[j[m1] + s])
            if m2.any():

                break

            f_try = compute_c5(h_try)
        
        # Update Lipschitz
        if ls > 0:
            L *= 2.0
        elif ls < 3 and it > 10:
            L = max(L * 0.8, 0.1)
        
        h_prev = h_m
        h_m = h_try
        
        if it % 2000 == 0:
            c5_m = compute_c5(h_m)
            if c5_m < c5_best:
                c5_best = c5_m
                h_best = h_m.copy()

    if c5_m < c5_best:
        c5_best = c5_m
        h_best = h_m.copy()
    return h_best, c5_best

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shift = _G_T * (corr - mc)
    shift = shift - shift.max()
    e = np.exp(shift)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    j = np.arange(_G_N)
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):

        if w < 1e-14:
            continue
        lag = oi - center

            m2 = j >= lag
