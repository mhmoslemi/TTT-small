            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_values(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    return obj, sw

def sm_grad_from_sw(h, sw):
    n = len(h)
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * (2.0 / n)

def pg_nesterov(h0, target, T, n_iter, lr0, deadline, mu=0.88, decay=0.9995):
    h = h0.copy()
    v = np.zeros_like(h)
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    obj_prev = sm_values(h, T)[0]
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        h_y = h + mu * v
        obj, sw = sm_values(h_y, T)
        grad = sm_grad_from_sw(h_y, sw)
        h_new = project_constraint(h_y - lr * grad, target)
        v = 0.5 * (h_new - h)
        h = h_new
        lr *= decay
        if it % 200 == 199:
            obj_now = sm_values(h, T)[0]
            if obj_now > obj_prev - 1e-12:
                stall += 1
                if stall > 3:
                    break
            else:
                stall = 0
            obj_prev = obj_now
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _scl_fun(x):
    return x.sum() - _G_TARGET

def _scl_jac(x):
    return np.ones(_G_N)

def _fobj(x):
    return sm_values(x, _G_T)[0]

def _fgrad(x):
    _, sw = sm_values(x, _G_T)
    return sm_grad_from_sw(x, sw)

def scl_solve(h0, target, T, maxiter, deadline):
    global _G_T, _G_TARGET, _G_N
    _G_T = T
    _G_TARGET = target
    _G_N = len(h0)
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _scl_fun, 'jac': _scl_jac}]
    try:
        res = minimize(_fobj, h0, method='SLSQP', jac=_fgrad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def binary_round(h, target):
    n = len(h)
    above = h > 0.5
    m = above.sum()
    h_b = np.where(h > 0.5, 1.0, 0.0)
    if m > target:
        order = np.argsort(h[above])
        flip = int(m - target)
        idxs = np.where(above)[0]
        h_b[idxs[order[:flip]]] = 0.0
    elif m < target:
        order = np.argsort(h[~above])
        flip = int(target - m)
        idxs = np.where(~above)[0]
        h_b[idxs[order[:flip]]] = 1.0
    return h_b

def local_flip_search(h0, target, deadline, max_iters=2000):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(50):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if h[i] > 0.5 and h[j] < 0.5:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(560)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    h = h_best.copy()

    # Phase 1: Nesterov PG multi-T
    T_list = [30, 80, 150, 300, 600, 1200, 2500]
    for T in T_list:
        if time.time() > deadline - 250:
            break
        dl = min(t0 + 120, deadline - 200)
        h, c5 = pg_nesterov(h, target, T, 4000, 0.035, dl, mu=0.88)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 2: SLSQP polish
    for T in [300, 800]:
        if time.time() > deadline - 150:
            break
        h, c5 = scl_solve(h, target, T, 400, deadline - 80)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 3: Binary rounding + local flip search
    if time.time() < deadline - 100:
        h_bin = binary_round(h_best, target)
        c5_bin = compute_c5(h_bin)
        h_flipped, c5_flip = local_flip_search(h_bin, target, deadline - 60, max_iters=3000)
        if c5_flip < c5_best:
            c5_best = c5_flip
            h_best = h_flipped.copy()

    # Phase 4: Perturbation + re-optimize from both continuous and binary
    for r in range(25):
        if time.time() > deadline - 40:
            break
        base = h_best.copy()
        if r % 3 == 0:
            idx = rng.integers(0, n)
            sz = rng.integers(4, 25)
            for i in range(sz):
                base[(idx + i) % n] = 1.0 - base[(idx + i) % n]
            base = project_constraint(base, target)
        elif r % 3 == 1:
            base = np.clip(base + rng.normal(0, 0.04, n), 0, 1)
            base = project_constraint(base, target)
        else:
            base = binary_round(base + rng.normal(0, 0.1, n), target)
        T_p = 300 + r * 40
        h_p, c5_p = pg_nesterov(base, target, T_p, 2000, 0.025, deadline - 15, mu=0.87)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()

    # Phase 5: Very high T Nesterov (near-exact max)
    for T in [1500, 3000, 6000]:
        if time.time() > deadline - 20:
            break
        h, c5 = pg_nesterov(h_best, target, T, 6000, 0.012, deadline - 8, mu=0.92, decay=0.9998)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 6: Final SLSQP
    if time.time() < deadline - 10:
        h, c5 = scl_solve(h_best, target, 1000, 250, deadline - 2)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
