```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)
```

```python
    for k in range(2*n-1):
        s = k - center
        # (1-h[i+s]) for valid i
        i_plus = np.arange(max(0, -s), min(n, n-s))
        G[k, i_plus] += (1 - h[i_plus + s])
        # -h[i-s] for valid i
        i_minus = np.arange(max(0, s), min(n, n+s))
        G[k, i_minus] -= h[i_minus - s]
    return G
```
