    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Multi-T refinement on initial
    h_best, c5_best = multi_t_refine(h_best, target, t0, budget - 300)

    # Phase 2: Coordinate polish
    h_best, c5 = coord_polish(h_best, target, t0, budget - 200, max_rounds=30, rng=rng)
    if c5 < c5_best:
        c5_best = c5
        h_best = h_best.copy()

    # Phase 3: Diverse Fourier starts
    for trial in range(6):
        if time.time() - t0 > budget - 120:
            break
        h_f = fourier_init(n, target, rng)
        h_f, c5_f = multi_t_refine(h_f, target, t0, budget - 60)
        h_f, c5_f = coord_polish(h_f, target, t0, budget - 30, max_rounds=10, rng=rng)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f.copy()

    # Phase 4: Nudge + re-optimize from best
    for nd in range(8):
        if time.time() - t0 > budget - 60:
            break
        h_p = h_best.copy()
        npass = int(rng.integers(3, 12))
        for _ in range(npass):
            h_p[int(rng.integers(0, n))] = rng.uniform(0.0, 1.0)
        h_p = project_constraint(h_p, target)
        h_p, c5_p = multi_t_refine(h_p, target, t0, budget - 30)
        h_p, c5_p = coord_polish(h_p, target, t0, budget - 15, max_rounds=8, rng=rng)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()

    # Phase 5: Try lower resolution n2=280 for potentially simpler optimum
    n2 = 280
    target2 = n2 / 2.0
    if time.time() - t0 < budget - 80:
        xc = np.arange(n) * (2.0 / n)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x2, xc, h_best)
        h2 = project_constraint(h2, target2)
        h2, c5_2 = multi_t_refine(h2, target2, t0, budget - 30)
        h2, c5_2 = coord_polish(h2, target2, t0, budget - 15, max_rounds=20, rng=rng)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2
            n = n2
            target = target2

    # Phase 6: Final micro-refinement
    nf = len(h_best)
    tf = nf / 2.0
    if time.time() - t0 < budget - 3:
        h_best, c5 = pg_solve(h_best, tf, 2.0, 1500, 0.015, t0, budget - 2)
        if c5 < c5_best:
            c5_best = c5
        h_best, c5 = coord_polish(h_best, tf, t0, budget - 1, max_rounds=20, rng=rng)
        if c5 < c5_best:
            c5_best = c5

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
