    return np.correlate(x, 1.0 - x, mode="full")[lag] * (2.0 / n)

def lag_grad(x, lag, n):
    o = lag - (n - 1)
    return corr_grad(x, o) * (2.0 / n)

def sum_eq(x, target):
    return x.sum() - target

def sum_jac(x, target):
    return np.ones(len(x))
