from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def logsumexp_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def logsumexp_grad(x, T):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o] * dx
        if w < 1e-14:

        else:
            p = -o

    else:
        p = -o

            free = (v < 1.0 - 1e-12)
            steps = (1.0 - v[free])
            total_cap = steps.sum()
            if total_cap > 0:

    n = len(x0)

    def grad(x):
        return corr_grad_single(x, lag_idx)
    def eq_fun(x):
        return x.sum() - target

        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})

    except:

def fix_constraint(h, target):

                diff -= add
