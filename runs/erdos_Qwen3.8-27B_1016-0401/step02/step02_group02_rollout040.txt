    for _ in range(n_steps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_local
        best_pair = None
        n_try = min(30, max(5, len(ones) * len(zeros) // 500))
        for _ in range(n_try):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h_c = h.copy()
            h_c[i] = 0.0
            h_c[j] = 1.0
            h_c = project_feasible(h_c, target)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            h = project_feasible(h, target)
            c5_local = best_c5
            ones = np.where(h >= 0.99)[0]
            zeros = np.where(h <= 0.01)[0]
        else:
            break
    return h, c5_local

def round_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(280) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP continuation (moderate T)
    if elapsed() < budget_s - 350:
        for T, mit in [(60, 200), (30, 300), (15, 400), (8, 500), (4, 600)]:
            if elapsed() > budget_s - 250:
                break
            h = slsqp_phase(h, n, target, bounds, T, mit)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 2: Momentum subgradient (K=1 first for focus)
    h = h_best.copy()
    if elapsed() < budget_s - 180:
        for K_val, steps, lr, mu in [(1, 400, 0.1, 0.75), (2, 300, 0.07, 0.7),
                                      (1, 300, 0.05, 0.8), (3, 200, 0.06, 0.65)]:
            if elapsed() > budget_s - 80:
                break
            h, c5_local = mom_subgrad(h, n, target, steps, K_val, lr, 0.03, mu)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 3: SLSQP at very low T (near-exact max)
    if elapsed() < budget_s - 100:
        for T in [1.5, 0.8, 0.4]:
            if elapsed() > budget_s - 40:
                break
            h = slsqp_phase(h_best, n, target, bounds, T, 2000)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 4: Coordinate descent polish
    if elapsed() < budget_s - 50:
        for nsw, dets in [(3, [-0.15, -0.05, 0.05, 0.15]), (2, [-0.08, -0.03, 0.03, 0.08]),
                          (2, [-0.03, -0.01, 0.01, 0.03])]:
            if elapsed() > budget_s - 20:
                break
            h_best, c5_cd = coord_polish(h_best, n, target, nsw, dets)
            if c5_cd < c5_best:
                c5_best = c5_cd

    # Phase 5: Try binary + local search, keep best of continuous vs binary
    h_bin = round_binary(h_best, target)
    c5_bin = compute_c5(h_bin)
    if elapsed() < budget_s - 30 and c5_bin < c5_best + 0.001:
        h_bin, c5_bin = binary_local_search(h_bin, target, 800, rng)
    c5_bin_final = compute_c5(h_bin)
    if c5_bin_final < c5_best:
        c5_best = c5_bin_final
        h_best = h_bin.copy()

    # Phase 6: Explore different resolutions
    if elapsed() < budget_s - 80:
        for n_try in [n // 2, int(n * 1.5)]:
            if elapsed() > budget_s - 30:
                break
            if n_try < 50:
                continue
            target_t = n_try / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, h_best)
            h_t = project_feasible(h_t, target_t)
            c5_t = compute_c5(h_t)
            bounds_t = [(0.0, 1.0)] * n_try
            for T in [20, 8, 3]:
                if elapsed() > budget_s - 20:
                    break
                h_t = slsqp_phase(h_t, n_try, target_t, bounds_t, T, 500)
                c5_t = compute_c5(h_t)
            if c5_t < c5_best:
                c5_best = c5_t
                h_best = h_t.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
