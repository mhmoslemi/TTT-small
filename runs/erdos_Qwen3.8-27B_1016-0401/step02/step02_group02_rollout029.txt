  Contribution: h[i-lag] * (1-h[i]) → if we increase h[i] by a, change is -h[i-lag]*a = -a*h[i-lag]

So the "influence vector" of position i is:
inf_i[lag] = (1-h[i+lag]) - h[i-lag]
(valid when both indices are in range, 0 otherwise)

In terms of the full correlation index k = lag + (n-1):
inf_i[k] = (1-h[i+k-(n-1)]) - h[i-(k-(n-1))] for valid indices

Now, if I transfer a from position j to position i (h[i] += a, h[j] -= a):
Δcorr[k] = a * (inf_i[k] - inf_j[k])

```python
        lag_lo2 = i - (n - 1)
        lag_hi2 = i
        for lag in range(lag_lo2, lag_hi2 + 1):
            inf[i, lag + n - 1] -= h[i - lag]
    return inf
```

Let me vectorize the influence computation:

```python
def compute_influence_matrix(h):
    n = len(h)
    N = 2 * n - 1
    # For position i, the influence at correlation index k (corresponding to lag = k-n+1):
    # First factor: (1-h[i+lag]) valid when 0<=i+lag<n → lag in [-i, n-1-i]
    # Second factor: -h[i-lag] valid when 0<=i-lag<n → lag in [i-n+1, i]
    
    # Build using broadcasting
    i_idx = np.arange(n)[:, None]  # (n, 1)
    lag_idx = np.arange(N)[None, :]  # (1, N), lag = lag_idx - n + 1
    lags = lag_idx - (n - 1)  # (1, N), actual lag values
    
    # First factor: i + lag, valid when 0 <= i+lag < n
    ial = i_idx + lags  # (n, N)
    valid1 = (ial >= 0) & (ial < n)
    # h[i+lag] where valid
    clamped_ial = np.clip(ial, 0, n-1)
    h_ial = h[clamped_ial]  # (n, N)
    
    # Second factor: i - lag, valid when 0 <= i-lag < n
    ial2 = i_idx - lags  # (n, N)
    valid2 = (ial2 >= 0) & (ial2 < n)
    clamped_ial2 = np.clip(ial2, 0, n-1)
    h_ial2 = h[clamped_ial2]  # (n, N)
    
    inf = valid1 * (1.0 - h_ial) - valid2 * h_ial2
    return inf
```

This is O(n²) but vectorized in numpy. For n=280, it's 280×559 ≈ 157K elements, very fast.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    h = x
    dx = 2.0 / n
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def cfun_eq(x, target):
    return x.sum() - target

def cjac_eq(x, target):
    return np.ones(len(x))

def slsqp_phase(h, n, target, bounds, T, maxiter):
    cons = [{'type': 'eq', 'fun': cfun_eq, 'jac': cjac_eq, 'args': (target,)}]
    try:
        res = minimize(lse_obj, h, method='SLSQP', jac=lse_grad,
                       args=(T, n), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def compute_influence_matrix(h):
    n = len(h)
    N = 2 * n - 1
    i_idx = np.arange(n)[:, None]
    lags = np.arange(N)[None, :] - (n - 1)
    ial = i_idx + lags
    valid1 = (ial >= 0) & (ial < n)
    h_ial = h[np.clip(ial, 0, n - 1)]
    ial2 = i_idx - lags
    valid2 = (ial2 >= 0) & (ial2 < n)
    h_ial2 = h[np.clip(ial2, 0, n - 1)]
    return valid1 * (1.0 - h_ial) - valid2 * h_ial2

def pairwise_polish(h, n, target, n_rounds, n_pairs, rng, t0, deadline):
    dx = 2.0 / n
    c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    inf = compute_influence_matrix(h)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved_any = False
        for _ in range(n_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            val = h[i] - h[j]
            if abs(val) < 0.015:
                continue
            best_gain = 0.0
            best_na = 0.0
            for sign in (1, -1):
                for frac in (0.5, 0.3, 0.7, 0.2):
                    na = sign * val * frac
                    hi_n = h[i] + na
                    hj_n = h[j] - na
                    if hi_n < -1e-10 or hi_n > 1 + 1e-10:
                        continue
                    if hj_n < -1e-10 or hj_n > 1 + 1e-10:
                        continue
                    new_max = float(np.max(corr + na * (inf[i] - inf[j]))) * dx
                    gain = c5 - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_na = na
            if best_gain > 1e-13:
                na = best_na
                h[i] += na
                h[j] -= na
                corr = np.correlate(h, 1.0 - h, mode="full")
                inf = compute_influence_matrix(h)
                c5 = float(np.max(corr) * dx)
                improved_any = True
        if not improved_any:
            break
    return h, c5

def binary_swap_polish(h, n, target, n_rounds, rng, t0, deadline):
    dx = 2.0 / n
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:int(target)]] = 1.0
    c5 = compute_c5(h_bin)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        improved = False
        n_tries = min(1500, len(ones) * len(zeros))
        corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
        for _ in range(n_tries):
            i = int(ones[rng.integers(0, len(ones))])
            j = int(zeros[rng.integers(0, len(zeros))])
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            corr[i] = np.correlate(h_bin, 1.0 - h_bin, mode="full")
            new_c5 = float(np.max(corr) * dx)
            if new_c5 < c5 - 1e-13:
                c5 = new_c5
                improved = True
                ones = np.where(h_bin == 1.0)[0]
                zeros = np.where(h_bin == 0.0)[0]
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
        if not improved:
            break
    return h_bin, c5

def subgrad_polish(h, n, target, n_steps, K, t0, deadline):
    dx = 2.0 / n
    c5 = compute_c5(h)
    center = n - 1
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for idx in top_idx:
            o = int(idx) - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        gx = g * dx / K
        for lr in (0.02, 0.01, 0.005, 0.002):
            h_cand = project_feasible(h - lr * gx, target)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5 - 1e-13:
                h = h_cand
                c5 = c5_cand
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.zeros(280)
        h[:140] = 1.0

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP
    for T, mit in [(50, 200), (15, 300), (5, 400)]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h = slsqp_phase(h, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 2: Binary structure optimization
    dl2 = t0 + budget_s * 0.32
    h_bin, c5_bin = binary_swap_polish(h_best, n, target, 300, rng, t0, dl2)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    # Relax from binary with SLSQP
    h = h_best.copy() + rng.normal(0, 0.08, n)
    h = project_feasible(h, target)
    for T, mit in [(30, 300), (8, 400)]:
        if time.time() - t0 > budget_s * 0.42:
            break
        h = slsqp_phase(h, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 3: Intensive pairwise polish
    dl3 = t0 + budget_s * 0.72
    h = h_best.copy()
    h, c5_pw = pairwise_polish(h, n, target, 500, 2500, rng, t0, dl3)
    if c5_pw < c5_best:
        c5_best = c5_pw
        h_best = h.copy()

    # Phase 4: SLSQP polish
    for T, mit in [(3, 400), (2, 500)]:
        if time.time() - t0 > budget_s * 0.8:
            break
        h = slsqp_phase(h_best, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 5: More pairwise
    dl5 = t0 + budget_s * 0.9
    h = h_best.copy()
    h, c5_pw2 = pairwise_polish(h, n, target, 400, 2000, rng, t0, dl5)
    if c5_pw2 < c5_best:
        c5_best = c5_pw2
        h_best = h.copy()

    # Phase 6: Subgradient polish
    dl6 = t0 + budget_s - 10
    for K_val in (5, 3, 2):
        if time.time() - t0 > dl6:
            break
        h = h_best.copy()
        h, c5_sg = subgrad_polish(h, n, target, 300, K_val, t0, dl6)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h.copy()
        # Quick pairwise after subgrad
        h = h_best.copy()
        h, c5_pw3 = pairwise_polish(h, n, target, 100, 1500, rng, t0, dl6)
        if c5_pw3 < c5_best:
            c5_best = c5_pw3
            h_best = h.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
```
