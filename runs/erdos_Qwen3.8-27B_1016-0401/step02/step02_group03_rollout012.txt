import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def full_soft_grad(h, n, T):
    """Compute soft-max gradient over all shifts."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad_raw(h, n, oi - center)
    return grad * dx

def soft_obj(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    dx = 2.0 / n
    if T < 1e-6:
        return mc * dx
    return (mc + np.log(np.exp(T * (corr - mc)).sum()) / T) * dx

def soft_obj_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    dx = 2.0 / n
    if T < 1e-6:
        idx = int(np.argmax(corr))
        return shift_grad_raw(h, n, idx - (n-1)) * dx
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad_raw(h, n, oi - center)
    return grad * dx

def slsqp_optimize(h, n, T, maxiter, ftol):
    def obj(x):
        return soft_obj(x, n, T)
    def grad(x):
        return soft_obj_grad(x, n, T)
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def adam_optimize(h_init, n, t0, time_budget, lr=0.05, T_init=8.0, T_end=0.5):
    """Adam optimizer with soft-max gradient."""
    dx = 2.0 / n
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    n_iter = 0
    total_iters = 200000
    T = T_init
    T_factor = (T_end / T_init) ** (1.0 / max(total_iters, 1))
    
    while time.time() - t0 < time_budget:
        n_iter += 1
        T = max(T_end, T_init * (T_factor ** n_iter))
        g = full_soft_grad(h, n, T)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** n_iter)
        v_hat = v / (1 - beta2 ** n_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_feasible(h_new, n)
        h = h_new
        if n_iter % 5000 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def soft_obj_grad(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    dx = 2.0 / n
    if T < 1e-8:
        idx = int(np.argmax(corr))
        return shift_grad_raw(x, n, idx - (n - 1)) * dx
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    thresh = 1e-8
    for oi in range(len(sw)):
        w = sw[oi]
        if w < thresh:
            continue
        grad += w * shift_grad_raw(x, n, oi - center)
    return grad * dx

def soft_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    dx = 2.0 / n
    if T < 1e-8:
        return mc * dx
    return (mc + np.log(np.exp(T * (corr - mc)).sum()) / T) * dx

def do_slsqp(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(soft_obj, h, method='SLSQP', jac=lambda x: soft_obj_grad(x, n, T),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def adam_soft(h_init, n, t0, wall_time, lr=0.04, T_init=16.0, T_end=0.3):
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    it = 0
    while time.time() - t0 < wall_time:
        it += 1
        frac = min(1.0, it / 50000.0)
        T = T_init * (T_end / T_init) ** frac
        lr_t = lr * (1.0 / (1.0 + 0.002 * it))
        g = soft_obj_grad(h, n, T)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        m_hat = m_arr / (1 - b1 ** it)
        v_hat = v_arr / (1 - b2 ** it)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        if it % 8000 == 0:
            c5c = compute_c5(h)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine SLSQP
    temps1 = [256.0, 128.0, 64.0, 32.0, 16.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25]
    for T in temps1:

        miters = 3000 if T >= 4 else 4000

        if c5 < c5_best:
            c5_best = c5

    # Phase 2: Adam soft-max optimization
    elapsed = time.time() - t0

            if time.time() - t0 > budget_s - 50:

            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        # Quick Adam at higher res

            h_a2, c5_a2 = adam_soft(h_best, n, t0, time.time() - t0 + 50, lr=0.03)

    return (h_best, c5_final, n)
