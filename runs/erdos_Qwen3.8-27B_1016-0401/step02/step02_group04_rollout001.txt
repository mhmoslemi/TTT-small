from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_binary(h, target):
    n = len(h)
    t = int(round(target))
    idx = np.argsort(-h)[:t]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

_pT = 0
_pk = 0
_pn = 0
_ptarget = 0
_pen = 0

def _obj(x):
    corr = get_corr(x)
    top = np.sort(corr)[:-_pk] if _pk < len(corr) else corr
    top = np.sort(corr)[-max(_pk,1):]
    mc = top[-1]
    sh = _pT * (top - mc)
    sh = np.clip(sh, -500, 500)
    val = mc + np.log(np.sum(np.exp(sh))) / _pT
    return val * (2.0 / _pn) + _pen * (x.sum() - _ptarget) ** 2

def _grad(x):
    corr = get_corr(x)
    k = max(_pk, 1)
    top_idx = np.argsort(corr)[-k:]
    mc = corr[top_idx[-1]]
    sh = _pT * (corr[top_idx] - mc)
    sh = np.clip(sh, -500, 500)
    e = np.exp(sh)
    w = e / np.sum(e)
    center = _pn - 1
    g = np.zeros(_pn)
    for ii in range(k):
        if w[ii] < 1e-8:
            continue
        o = int(top_idx[ii]) - center
        wi = w[ii]
        if o == 0:
            g += wi * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += wi * (1.0 - x[:_pn - o])
            g[:_pn - o] -= wi * x[o:]
        else:
            p = -o
            g[:_pn - p] += wi * (1.0 - x[p:])
            g[p:] -= wi * x[:_pn - p]
    return g * (2.0 / _pn) + 2.0 * _pen * (x.sum() - _ptarget) * np.ones(_pn)

def optimize_lbfgs(x0, T, k, n, target, pen, maxiter):
    global _pT, _pk, _pn, _ptarget, _pen
    _pT, _pk, _pn, _ptarget, _pen = T, k, n, target, pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, x0.copy(), method='L-BFGS-B', jac=_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_box_sum(res.x, target)
    except Exception:
        return x0.copy()

def binary_sanneal(b, target, n_steps, rng, T_start=5.0, T_end=0.5):
    n = len(b)
    b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    corr = get_corr(b)
    best_c = float(corr.max())
    cur_c = best_c
    for idx in range(n_steps):
        T_sa = T_start * (T_end / T_start) ** (idx / max(n_steps - 1, 1))
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        b[i], b[j] = 0.0, 1.0
        new_corr = get_corr(b)
        new_c = float(new_corr.max())
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T_sa, 1e-6)):
            cur_c = new_c
            corr = new_corr
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if new_c < best_c:
                best_c = new_c
        else:
            b[i], b[j] = 1.0, 0.0
    return b, best_c
