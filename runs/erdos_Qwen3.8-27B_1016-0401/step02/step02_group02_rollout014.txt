def lse_value(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        s = int(idx) - center
        wi = w[idx]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:n] += wi * (1.0 - h[0:n - s])
            grad[0:n - s] -= wi * h[s:n]
        else:
            p = -s
            grad[0:n - p] += wi * (1.0 - h[p:n])
            grad[p:n] -= wi * h[0:n - p]
    return grad * dx

def fista_opt(h_init, target, n, T, maxiter=400):
    x = h_init.copy()
    y = h_init.copy()
    x_old = h_init.copy()
    t = 1.0
    L = 6.0 * (T + 1.0) / n
    if L < 0.05:
        L = 0.05
    for it in range(maxiter):
        yc = np.clip(y, 0.0, 1.0)
        corr = np.correlate(yc, 1.0 - yc, mode="full")
        g = lse_grad(corr, yc, T, n)
        z = yc - g / L
        x_new = project_feasible(z, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + (t / t_new - 1.0) * (x_new - x_old)
        x_old = x
        x = x_new
        t = t_new
    return x

def proj_grad_opt(h_init, target, n, T, maxiter=600):
    x = h_init.copy()
    step = 2.0 / (4.0 * T + 4.0)
    momentum = 0.7
    v = np.zeros(n)
    for it in range(maxiter):
        corr = np.correlate(x, 1.0 - x, mode="full")
        g = lse_grad(corr, x, T, n)
        v = momentum * v - step * g
        z = x + v
        x_new = project_feasible(z, target)
        f_new = lse_value(np.correlate(x_new, 1.0 - x_new, mode="full"), T, n)
        f_old = lse_value(np.correlate(x, 1.0 - x, mode="full"), T, n)
        if f_new > f_old + 0.001:
            step *= 0.5
            v = -0.5 * g
            z = x + v
            x_new = project_feasible(z, target)
        elif step < 2.0 / (4.0 * T + 4.0):
            step = min(step * 1.05, 2.0 / (4.0 * T + 4.0))
        x = x_new
    return x

def gradient_swap(h, n, target, max_steps=400):
    c5 = compute_c5(h)
    for step in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = lse_grad(corr, h, 1.0, n)
        g_src = np.argsort(g)[::-1]
        g_tgt = np.argsort(g)
        src = -1
        tgt = -1
        for idx in g_src[:40]:
            if h[idx] > 0.01:
                src = int(idx)
                break
        for idx in g_tgt[:40]:
            if h[idx] < 0.99:
                tgt = int(idx)
                break
        if src < 0 or tgt < 0 or src == tgt:
            break
        lo, hi = 0.0, min(h[src], 1.0 - h[tgt])
        best_eps = 0.0
        best_c5 = c5
        for _ in range(10):
            mid = (lo + hi) / 2
            if mid < 1e-5:
                break
            h_cand = h.copy()
            h_cand[src] -= mid
            h_cand[tgt] += mid
            c5c = compute_c5(h_cand)
            if c5c < best_c5:
                best_c5 = c5c
                best_eps = mid
                lo = mid
            else:
                hi = mid
        if best_eps < 1e-5:
            eps = 0.003 * min(h[src], 1.0 - h[tgt])
            if eps < 1e-5:
                break
            h[src] -= eps
            h[tgt] += eps
            h = project_feasible(h, target)
            c5 = compute_c5(h)
        else:
            h[src] -= best_eps
            h[tgt] += best_eps
            c5 = best_c5
    return h, c5

def topk_swap(h, n, target, K, max_steps=300):
    c5 = compute_c5(h)
    center = n - 1
    dx = 2.0 / n
    for step in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_lags = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for idx in top_lags:
            s = int(idx) - center
            if s == 0:
                g += (1.0 - 2.0 * h)
            elif s > 0:
                g[s:n] += (1.0 - h[0:n - s])
                g[0:n - s] -= h[s:n]
            else:
                p = -s
                g[0:n - p] += (1.0 - h[p:n])
                g[p:n] -= h[0:n - p]
        g *= dx / K
        g_src = np.argsort(g)[::-1]
        g_tgt = np.argsort(g)
        tried = 0
        improved = False
        for si in range(min(8, len(g_src))):
            src = int(g_src[si])
            if h[src] < 0.02:
                continue
            for ti in range(min(8, len(g_tgt))):
                tgt = int(g_tgt[ti])
                if h[tgt] > 0.98 or tgt == src:
                    continue
                tried += 1
                eps_max = min(h[src], 1.0 - h[tgt])
                best_eps = 0.0
                best_c5 = c5
                for f in [0.1, 0.25, 0.5, 0.75, 0.95]:
                    eps = f * eps_max
                    h_c = h.copy()
                    h_c[src] -= eps
                    h_c[tgt] += eps
                    c5c = compute_c5(h_c)
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_eps = eps
                if best_eps > 0.005 and best_c5 < c5 - 1e-9:
                    h[src] -= best_eps
                    h[tgt] += best_eps
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    c5 = best_c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(280)
        h0[::2] = 1.0

    n = len(h0)
    target = n / 2.0
    h0 = project_feasible(h0, target)
    c5_best = compute_c5(h0)
    h_best = h0.copy()

    # Phase 1: FISTA with decreasing temperature
    for T in [200, 100, 50, 25, 15, 10, 7, 5, 3, 2, 1.5, 1.0]:
        if elapsed() > budget_s - 350:
            break
        h = fista_opt(h_best, target, n, T, maxiter=300)
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-9:
            c5_best = c5
            h_best = h.copy()

    # Phase 2: Projected gradient with momentum
    for T in [5, 3, 2, 1.5, 1.0]:
        if elapsed() > budget_s - 250:
            break
        h = proj_grad_opt(h_best, target, n, T, maxiter=400)
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-9:
            c5_best = c5
            h_best = h.copy()

    # Phase 3: Gradient-guided swap search
    h = h_best.copy()
    h, c5 = gradient_swap(h, n, target, max_steps=400)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()

    # Phase 4: Top-K focused swap
    for K in [6, 4, 3, 2]:
        if elapsed() > budget_s - 150:
            break
        h = h_best.copy()
        h, c5 = topk_swap(h, n, target, K, max_steps=250)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 5: Very low T FISTA from best
    for T in [0.8, 0.5, 0.3]:
        if elapsed() > budget_s - 100:
            break
        h = fista_opt(h_best, target, n, T, maxiter=500)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 6: Final swap polish
    h = h_best.copy()
    h, c5 = gradient_swap(h, n, target, max_steps=300)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    h, c5 = topk_swap(h, n, target, 1, max_steps=200)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()

    # Phase 7: Multi-start perturbations
    if elapsed() < budget_s - 200:
        for restart in range(8):
            if elapsed() > budget_s - 60:
                break
            scale = 0.03 + 0.04 * (restart // 3)
            noise = rng.normal(0, scale, n)
            h_r = project_feasible(h_best + noise, target)
            h_r = fista_opt(h_r, target, n, 3, maxiter=200)
            c5_r = compute_c5(h_r)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r.copy()
            h_r, c5_r = gradient_swap(h_r, n, target, max_steps=150)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_r.copy()

    # Phase 8: Try different resolution
    if elapsed() < budget_s - 100 and n > 140:
        n_lo = n // 2
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        for T in [30, 10, 5, 2, 1]:
            if elapsed() > budget_s - 30:
                break
            h_lo = fista_opt(h_lo, target_lo, n_lo, T, maxiter=200)
            c5_lo = compute_c5(h_lo)
            if c5_lo < c5_best:
                c5_best = c5_lo
                h_best = h_lo.copy()
        h_lo, c5_lo = gradient_swap(h_lo, n_lo, target_lo, max_steps=200)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()

    # Final verification
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
