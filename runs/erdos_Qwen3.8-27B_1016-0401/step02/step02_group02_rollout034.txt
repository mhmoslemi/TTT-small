import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_and_grad(h, T, n):
    """Compute log-sum-exp objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    Tc = T * corr
    Tc_max = np.max(Tc)
    exp_shifted = np.exp(Tc - Tc_max)
    log_sum = Tc_max + np.log(np.sum(exp_shifted))
    obj = log_sum / T * dx
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    grad *= dx
    return obj, grad

def slsqp_one(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = lse_obj_and_grad(x, T, n)
        return v
    
    def jac(x):
        _, g = lse_obj_and_grad(x, T, n)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h

def periodic_pattern_search(n, p):
    """Search for best periodic pattern of period p."""
    if n % p != 0:
        return None, None
    n_tiles = n // p
    if p % 2 != 0:

        # With n_tiles tiles, total ones = n/2
        # Each tile has floor(p/2) or ceil(p/2) ones
        # This is complicated, skip for now

    total_ones = n // 2
    if n_tiles * half_p != total_ones:
        # Can't form valid periodic pattern
        return None, None
    
    best = None

import numpy as np
import time
import itertools
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    Tc = T * corr
    mx = np.max(Tc)
    es = np.exp(Tc - mx)
    s = np.sum(es)
    obj = (mx + np.log(s)) / T * dx
    w = es / s
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    grad *= dx
    return obj, grad

def slsqp_phase(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj_grad(x, T, n)[0]
    def jac(x):
        return lse_obj_grad(x, T, n)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def projected_grad(h, n, target, T, n_iter, lr_init=0.1):
    """Projected gradient descent with backtracking."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):

        lr = lr_init / (1.0 + 0.05 * it)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    if k_full < n:

    return h_bin
