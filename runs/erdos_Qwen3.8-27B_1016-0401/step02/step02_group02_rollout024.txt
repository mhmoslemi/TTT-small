    """Project x onto {y: 0<=y<=1, sum(y)=target} using bisection on the threshold."""
    lo, hi = -2.0, 2.0

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

    except:

        if c5_cand < best_c5:

            best = h_cand.copy()

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad(corr, h, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-12:
            continue
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * (2.0 / n)

def slsqp_polish(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        mc = np.max(np.correlate(x, 1.0 - x, mode="full"))
        sh = T * (np.correlate(x, 1.0 - x, mode="full") - mc)
        sh -= np.max(sh)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)

        return h

    best_c5 = compute_c5(best)
    for it in range(n_steps):
        if time.time() - t0 > budget - 10 or time.time() > time_limit + t0:
            break
        # Get top correlations
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_k = np.argsort(corr)[-3:]
        center = n - 1
        # For each top lag, find positions that contribute most
        for idx in top_k:
            o = int(idx) - center
            # Try to reduce this correlation by perturbing contributing positions
            if o == 0:
                contrib = best * (1.0 - best)
            elif o > 0:
                contrib = np.zeros(n)
                contrib[o:] = best[o:] * (1.0 - best[:n - o])
                contrib[:n - o] += best[o:] * (1.0 - best[:n - o])
            else:
                p = -o
                contrib = np.zeros(n)
                contrib[:n - p] = best[p:] * (1.0 - best[:n - p])
                contrib[p:] += best[:n - p] * (1.0 - best[p:])
