import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def exact_subgrad(h, top_k, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    w = 1.0 / len(idx)
    center = n - 1
    grad = np.zeros(n)
    for p in idx:
        s = p - center
        ws = w * dx
        if s >= 0:
            if n - s > 0:
                grad[:n - s] += ws * (1.0 - h[s:])
                grad[s:] -= ws * h[:n - s]
        else:
            p2 = -s
            if n - p2 > 0:
                grad[p2:] += ws * (1.0 - h[:n - p2])
                grad[:n - p2] -= ws * h[p2:]
    return grad

def soft_subgrad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    logits = T * (corr - mc)
    logits -= logits.max()
    e = np.exp(logits)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        wi = w[center + o] * dx
        if wi < 1e-13:
            continue
        if o >= 0:
            if n - o > 0:
                grad[:n - o] += wi * (1.0 - h[o:])
                grad[o:] -= wi * h[:n - o]
        else:
            p2 = -o
            if n - p2 > 0:
                grad[p2:] += wi * (1.0 - h[:n - p2])
                grad[:n - p2] -= wi * h[p2:]
    return grad

def adam_opt(h0, target, top_k, lr, iters, b1=0.9, b2=0.999, eps=1e-8):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for i in range(iters):
        g = exact_subgrad(h, top_k, dx)
        t = i + 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        h = project_box_sum(h - lr * mhat / (np.sqrt(vhat) + eps), target)
        if i % 100 == 99:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def lbfgs_soft(h0, n, target, T, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        mc = corr.max()
        s = T * (corr - mc)
        s -= s.max()
        return (mc + np.log(np.sum(np.exp(s))) / T) * dx
    def jcn(x):
        return soft_subgrad(x, T, dx)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj_pen(x):
        return obj(x) + 20.0 * (x.sum() - target) ** 2
    def jcn_pen(x):
        return jcn(x) + 40.0 * (x.sum() - target) * ones
    try:
        res = minimize(obj_pen, h0.copy(), method='L-BFGS-B', jac=jcn_pen,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-13})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Adam with exact top-k subgradient, multiple configs
    for top_k in [3, 5, 8]:
        if elapsed() > budget_s * 0.30:
            break
        for lr in [0.008, 0.004, 0.002]:
            if elapsed() > budget_s * 0.30:
                break
            h_new, c5_new = adam_opt(h_best, target, top_k, lr, 4000)
            h_new = project_box_sum(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 2: L-BFGS-B with soft-max at high temperatures
    for T in [500, 1000, 3000]:
        if elapsed() > budget_s * 0.55:
            break
        h_new = lbfgs_soft(h_best, n, target, T, 5000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Perturbation + Adam restart (escape local minima)
    npert = 0
    while elapsed() < budget_s * 0.80:
        npert += 1
        scale = max(0.02, 0.12 / (1.0 + 0.08 * npert))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        tk = int(rng.integers(3, 12))
        lr = float(rng.uniform(0.002, 0.012))
        h_new, c5_new = adam_opt(h_pert, target, tk, lr, 2500)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Resolution sweep
    for n_try in [140, 200, 280, 350, 420]:
        if elapsed() > budget_s - 80:
            break
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        for tk in [3, 5]:
            if elapsed() > budget_s - 30:
                break
            h_t, c5_t = adam_opt(h_try, t_try, tk, 0.005, 3000)
            h_t = project_box_sum(h_t, t_try)
            c5_t = compute_c5(h_t)
            if c5_t < c5_try:
                c5_try = c5_t
                h_try = h_t
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n = n_try
            target = t_try

    # Phase 5: Final polish with very low learning rate Adam
    if elapsed() < budget_s - 10:
        h_new, c5_new = adam_opt(h_best, target, 3, 0.001, 5000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final L-BFGS-B polish
    if elapsed() < budget_s - 3:
        h_new = lbfgs_soft(h_best, n, target, 2000, 3000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
