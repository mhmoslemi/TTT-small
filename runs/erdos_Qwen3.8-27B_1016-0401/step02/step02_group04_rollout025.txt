def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(h, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    n = len(h)
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, T, pen):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se
    obj = (mc + np.log(se) / T) * (2.0 / n)
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= dx
    if pen > 0:
        obj += pen * (h.sum() - n / 2.0) ** 2
        grad += 2.0 * pen * (h.sum() - n / 2.0)
    return obj, grad

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return soft_max_obj_grad(x, T, pen)[0]
    def grad(x):
        return soft_max_obj_grad(x, T, pen)[1]
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return project_box_sum(h0.copy(), target)

def topm_subgrad_step(h, m, step):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argpartition(corr, -m)[-m:]
    center = n - 1
    dx = 2.0 / n
    w = 1.0 / m
    grad = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    h_new = h - step * grad * dx
    return project_box_sum(h_new, n / 2.0)

def coord_descent(h, target, n_sweeps):
    n = len(h)
    h = h.copy()
    o_range = np.arange(-(n - 1), n)
    for _sweep in range(n_sweeps):
        for i in range(n):
            old = h[i]
            # Slope of each corr component w.r.t. h[i]
            s = np.zeros(2 * n - 1)
            vp = (i + o_range >= 0) & (i + o_range < n)
            vm = (i - o_range >= 0) & (i - o_range < n)
            s[vp] += 1.0 - h[i + o_range[vp]]
            s[vm] -= h[i - o_range[vm]]
            corr = np.correlate(h, 1 - h, mode="full")
            lo_d, hi_d = -old, 1.0 - old

                    hi_d = m2
                else:
                    lo_d = m1
            h[i] = old + (lo_d + hi_d) / 2.0
        h = project_box_sum(h, target)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, T, pen):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se
    obj = (mc + np.log(se) / T) * (2.0 / n)
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad *= dx
    if pen > 0:
        obj += pen * (h.sum() - n / 2.0) ** 2
        grad += 2.0 * pen * (h.sum() - n / 2.0)
    return obj, grad

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    def _obj(x):
        return soft_max_obj_grad(x, T, pen)[0]
    def _grad(x):
        return soft_max_obj_grad(x, T, pen)[1]
    try:
        res = minimize(_obj, h0.copy(), method='L-BFGS-B', jac=_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15})
        return project_box_sum(res.x, target)
    except Exception:
        return project_box_sum(h0.copy(), target)

def topm_subgrad_step(h, m, step):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argpartition(corr, -m)[-m:]
    center = n - 1
    dx = 2.0 / n
    w = 1.0 / m
    grad = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    h_new = h - step * grad * dx
    return project_box_sum(h_new, n / 2.0)

def coord_descent(h, target, n_sweeps):
    n = len(h)
    h = h.copy()
    o_range = np.arange(-(n - 1), n)
    for _sweep in range(n_sweeps):
        for i in range(n):
            old = h[i]
            s = np.zeros(2 * n - 1)
            vp = (i + o_range >= 0) & (i + o_range < n)
            vm = (i - o_range >= 0) & (i - o_range < n)
            s[vp] += 1.0 - h[i + o_range[vp]]
            s[vm] -= h[i - o_range[vm]]
            corr = np.correlate(h, 1 - h, mode="full")
            lo_d, hi_d = -old, 1.0 - old
            for _ in range(28):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                if np.max(corr + m1 * s) < np.max(corr + m2 * s):
                    hi_d = m2
                else:
                    lo_d = m1
            h[i] = old + (lo_d + hi_d) / 2.0
        h = project_box_sum(h, target)
    return h

    # Phase 3: Coordinate descent (exact 1D per coordinate)
