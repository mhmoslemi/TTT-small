                if c5_f < c5_best:
                    c5_best = c5_f
                    h_best = h_f
                break

    # Phase 4: Structured perturbations + re-optimize
    n_blocks = 14
    bsize = n // n_blocks
    for trial in range(50):
        if time.time() - _timer[0] > budget_s - 80:
            break
        h_pert = h_best.copy()
        k = int(rng.integers(1, 4))
        for _ in range(k):
            b = int(rng.integers(0, n_blocks))
            s = b * bsize
            e = min(s + bsize, n)
            h_pert[s:e] += rng.normal(0, 0.08, e - s)
        h_pert = project_box_sum(h_pert, target)
        T = 5 + int(rng.integers(0, 15))
        h_new = run_lbfgs(h_pert, n, target, T, 100.0, 5000, 1e-14)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Binary snap + re-optimize
    if time.time() - _timer[0] < budget_s - 50:
        h_snap = h_best.copy()
        h_snap[h_snap < 0.1] = 0.0
        h_snap[h_snap > 0.9] = 1.0
        h_snap = project_box_sum(h_snap, target)
        for T in [15, 6, 3]:
            if time.time() - _timer[0] > budget_s - 20:
                break
            h_new = run_lbfgs(h_snap, n, target, T, 100.0, 6000, 1e-15)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 6: Try coarser n=140
    if time.time() - _timer[0] < budget_s - 20:
        nc = 140
        tc = nc / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, nc, endpoint=False)
        h_c = project_box_sum(np.interp(x_new, x_old, h_best), tc)
        for T in [20, 6, 3]:
            if time.time() - _timer[0] > budget_s - 5:
                break
            h_new = run_lbfgs(h_c, nc, tc, T, 100.0, 6000, 1e-15)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                n = nc
                target = tc

    # Final polish
    if time.time() - _timer[0] < budget_s - 2:
        h_new = run_lbfgs(h_best, n, target, 2, 200.0, 5000, 1e-15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_box_sum(h_best, len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    return (h_best, min(c5_final, c5_best), len(h_best))
