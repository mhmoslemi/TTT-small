def fix_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    # Adjust with weighted steps
    for iterations in range(1000):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            # Add where possible
            idx = np.argsort(1.0 - h)[-min(max(int(abs(diff) * 5), 3), len(h)):]  # Sample most significant points
        else:
            idx = np.argsort(h)[:min(max(int(abs(diff) * 5), 3), len(h))]
        # Adjust one by one
        if not idx.size:
            break
        i = idx[0]
        if diff > 0 and 1.0 - h[i] >= abs(diff) or diff < 0 and h[i] >= abs(diff):
            h[i] = max(min(h[i], 1), 0)
            diff = 0
            break
        else:
            h[i] += np.clip(sgn(diff) * min(1, abs(diff) / 2), -h[i], 1-h[i])
            diff -= sgn(diff) * min(1, abs(diff))
    return h
