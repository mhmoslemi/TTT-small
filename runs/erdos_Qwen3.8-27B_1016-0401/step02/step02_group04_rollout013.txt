def compute_c5(h):

    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:

        if o == 0:

        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx

def lbfgs_opt(h0, target, T, pen, maxiter, gtol):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

        return h0.copy()
