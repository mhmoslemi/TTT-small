    h_init = h_init.copy()
    corr = np.correlate(h_init, 1.0 - h_init, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    shifts = (top_idx - (n - 1)).astype(int)
    tau0 = float(np.max(corr))
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    cons = []

    for si, s in enumerate(shifts):
        def crow_fun(x, s=s):
            h = x[:n]
            return x[-1] - corr_k_vec(h, s)

            h = x[:n]
            g = np.zeros(n+1)
            g[:n] = -corr_k_grad_vec(h, s)
            g[-1] = 1.0
            return g
        cons.append({'type': 'ineq', 'fun': crow_fun, 'jac': crow_jac})

        return x[:n].sum() - n/2.0

        g[:n] = 1.0
        return g

        return 0.0
    if s >= 0:
        return float(np.dot(h[:n-t], 1.0 - h[t:]))

    t = abs(s)
    g = np.zeros(n)

        return g
    if s >= 0:
        g[:n-t] += 1.0 - h[t:]

    return g

def lse_slsqp(h, n, T, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        return (mc + np.log(np.exp(T*(c - mc)).sum()) / T) * dx
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        w = np.exp(T * (c - mc))
        w /= w.sum()
        g = np.zeros(n)
        center = n - 1
        for i in range(len(w)):
            if w[i] < 1e-14:
                continue
            g += w[i] * corr_shift_grad(x, i - center)
        return g * dx
    def cons_eq(x):
        return x.sum() - target
    def cons_eq_j(x):
        return np.ones(n)
    res = minimize(obj, h, method='SLSQP', jac=jac,
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':cons_eq,'jac':cons_eq_j}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

    J = np.zeros((k, _AS_N + 1))
    for i in range(k):
        J[i, :_AS_N] = -corr_shift_grad(h, _AS_SHIFTS[i])
        J[i, -1] = 1.0
    return J

def _as_eq_fun(x):
    return x[:_AS_N].sum() - _AS_TARGET

def _as_eq_jac(x):
    g = np.zeros(_AS_N + 1)
    g[:_AS_N] = 1.0
    return g

def active_set_slsqp(h, n_active, maxiter, deadline):
    global _AS_N, _AS_SHIFTS, _AS_TARGET
    n = len(h)
    _AS_N = n
    _AS_TARGET = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    for outer in range(40):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        _AS_SHIFTS = (top_idx - (n - 1)).astype(int)
        tau0 = float(corr[top_idx[0]]) + 1e-8
        x0 = np.concatenate([h_cur, np.array([tau0])])
        bounds = [(0.0, 1.0)] * (n + 1)
        cons = [{'type': 'ineq', 'fun': _as_ineq_fun, 'jac': _as_ineq_jac},
                {'type': 'eq', 'fun': _as_eq_fun, 'jac': _as_eq_jac}]
        try:
            res = minimize(_as_obj, x0, method='SLSQP', jac=_as_obj_g,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            if res.x is not None and res.x[-1] < tau0:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new
                    h_cur = h_new
                elif c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

_LSE_N = 0
_LSE_T = 1.0
_LSE_DX = 1.0

def _lse_obj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    sh = _LSE_T * (c - mc)
    sh -= sh.max()
    return (mc + np.log(np.exp(sh).sum()) / _LSE_T) * _LSE_DX

def _lse_jac(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    sh = _LSE_T * (c - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    n = _LSE_N
    center = n - 1
    g = np.zeros(n)
    for i in range(len(w)):
        if w[i] < 1e-13:
            continue
        g += w[i] * corr_shift_grad(x, i - center)
    return g * _LSE_DX

def _lse_eq(x):
    return x.sum() - _LSE_N / 2.0

def _lse_eq_j(x):
    return np.ones(_LSE_N)

def lse_slsqp(h, n, T, maxiter):
    global _LSE_N, _LSE_T, _LSE_DX
    _LSE_N = n
    _LSE_T = float(T)
    _LSE_DX = 2.0 / n
    try:
        res = minimize(_lse_obj, h, method='SLSQP', jac=_lse_jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _lse_eq, 'jac': _lse_eq_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h.copy()

def multi_sg(h_init, n, t0, deadline, max_iters, k):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    T_sg = 40.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[::-1][:k]
        w = np.exp(T_sg * (corr[top_idx] - corr[top_idx[0]]))
        w /= w.sum()
        g = np.zeros(n)
        center = n - 1
        for i in range(k):
            if w[i] > 1e-12:
                g += w[i] * corr_shift_grad(h_y, top_idx[i] - center)
        alpha = 0.25 / (1.0 + 0.004 * it)
        h_new = h_y - alpha * g * dx
        if (it + 1) % 5 == 0:
            h_new = project_feasible(h_new, n)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - h_new.sum()
            if abs(diff) > 1e-10:
                h_new = np.clip(h_new + diff / n, 0.0, 1.0)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 40000 == 0:
            for hh in [project_feasible(h_avg, n), project_feasible(h_sg, n)]:
                c = compute_c5(hh)
                if c < c5_best:
                    c5_best = c
                    h_best = hh.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: LSE SLSQP
    for T in [64, 16, 4]:
        if time.time() > budget_s - 500:
            break
        hx = lse_slsqp(h_best, n, T, 2500)
        hx = project_feasible(hx, n)
        c = compute_c5(hx)
        if c < c5_best:
            c5_best = c
            h_best = hx

    # Phase 2: Active-set SLSQP
    dl = min(t0 + 300, budget_s - 350)
    h_a, c5_a = active_set_slsqp(h_best, 5, 4000, dl)
    if c5_a < c5_best:
        c5_best = c5_a
        h_best = h_a

    # Phase 3: Multi-shift subgradient
    h_sg, c5_sg = multi_sg(h_best, n, t0, budget_s - 250, 120000, k=6)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg

    # Phase 4: Active-set polish (more active shifts)
    dl4 = min(t0 + 200, budget_s - 120)
    h_a2, c5_a2 = active_set_slsqp(h_best, 7, 6000, dl4)
    if c5_a2 < c5_best:
        c5_best = c5_a2
        h_best = h_a2

    # Phase 5: Perturbation + active-set restarts
    for pi in range(40):
        if time.time() > budget_s - 15:
            break
        sigma = 0.004 + 0.008 * (pi % 5)
        h_p = project_feasible(h_best + rng.normal(0, sigma, n), n)
        h_p2, c_p2 = active_set_slsqp(h_p, 5, 2000, time.time() + 40)
        if c_p2 < c5_best:
            c5_best = c_p2
            h_best = h_p2

    # Phase 6: Final LSE SLSQP polish
    if time.time() < budget_s - 60:
        for T in [8, 4]:
            if time.time() > budget_s - 30:
                break
            hx = lse_slsqp(h_best, n, T, 3000)
            hx = project_feasible(hx, n)
            c = compute_c5(hx)
            if c < c5_best:
                c5_best = c
                h_best = hx

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
