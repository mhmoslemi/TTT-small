import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        m2 = j >= shift
        grad[m1] += (1.0 - h[j[m1] + shift])
        grad[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j < (n - s)
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _smooth_obj(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _smooth_grad(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-11:
            continue
        grad += w * _shift_grad(x, n, oi - center)
    return grad * dx

def do_slsqp(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        for oi in range(len(sw)):
            if sw[oi] < 1e-11:
                continue
            g += sw[oi] * _shift_grad(x, n, oi - center)
        return g * dx
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def adam_optimize(h_init, n, t0, budget_s, T_sched, max_iters=500000):

    m = np.zeros(n)

    h_a = h_init.copy()
    m_pow = 1.0
    v_pow = 1.0
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    no_improve = 0
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        T = T_sched(it)
        corr = np.correlate(h_a, 1.0 - h_a, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()

        grad = np.zeros(n)
        for oi in range(len(sw)):
            if sw[oi] < 1e-11:
                continue
            grad += sw[oi] * _shift_grad(h_a, n, oi - center)
        grad *= dx
        m_pow *= beta1
        v_pow *= beta2
        bc1 = 1.0 - m_pow
        bc2 = 1.0 - v_pow
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / bc1
        v_hat = v / bc2

def t_sched_factory(T_start, T_end, total_steps):

        frac = min(1.0, it / total_steps)
        return T_start * (T_end / T_start) ** frac
