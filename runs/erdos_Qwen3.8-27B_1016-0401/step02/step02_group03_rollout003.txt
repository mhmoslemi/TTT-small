def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def multi_shift_nesterov(h_init, n, t0, time_budget, K=5, max_iters=300000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = float(it) / (it + 3.0) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = corr[top_idx].max()
        w = np.exp(5.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            shift = ik - center
            g += wk * shift_grad_raw(h_y, n, shift)
        g *= dx
        alpha = 0.10 / (1.0 + 0.004 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 15000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best:
                c5_best = cv
                h_best = h_sg.copy()
    cv = compute_c5(project_feasible(h_sg, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_sg, n).copy()
    return h_best, c5_best

def local_polish(h, n, rng, steps=5000):
    """Coordinate-level local search: nudge pairs to reduce max correlation."""
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        ak = int(np.argmax(corr))
        shift = ak - center
        g = shift_grad_raw(h_cur, n, shift) * dx
        # Pick coordinate with largest positive gradient to decrease or negative to increase
        pos_mask = g > 0.01
        neg_mask = g < -0.01
        if pos_mask.any() and neg_mask.any():
            ip = rng.choice(np.where(pos_mask)[0])
            inn = rng.choice(np.where(neg_mask)[0])
            delta = min(0.05, h_cur[ip], 1.0 - h_cur[inn])
            h_try = h_cur.copy()
            h_try[ip] -= delta
            h_try[inn] += delta
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-8:
                h_cur = h_try
                c5_cur = c5_try
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    temps = [128, 64, 32, 16, 8, 4, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25]
    for T in temps:
        if time.time() - t0 > budget_s - 550:
            break
        h_new = do_slsqp(h_best, n, T, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Multi-shift Nesterov subgradient
    h_sg, c5_sg = multi_shift_nesterov(h_best, n, t0, budget_s - 450, K=5, max_iters=250000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP fine polish
    for T in [1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = do_slsqp(h_best, n, T, 12000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Local coordinate polish
    if time.time() - t0 < budget_s - 80:
        h_lp, c5_lp = local_polish(h_best, n, rng, steps=6000)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp
        h_new = do_slsqp(h_best, n, 0.5, 10000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Perturbation + SLSQP restarts
    for pi in range(20):
        if time.time() - t0 > budget_s - 30:
            break
        sigma = 0.003 + 0.007 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 2, 5000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final polish
    if time.time() - t0 < budget_s - 10:
        h_new = do_slsqp(h_best, n, 0.3, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
