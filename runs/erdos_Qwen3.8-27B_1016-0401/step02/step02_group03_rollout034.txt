        h = project_feasible(h, n)
        corr = compute_corr(h)
    
    return h, corr.max()

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:

            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad
