import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    n = _G_N
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = oi - center
        if shift >= 0:
            j = np.arange(n)
            m1 = j >= shift
            m2 = (j + shift) < n
            grad[m1] += (1.0 - x[j[m1] - shift])
            grad[m2] -= x[j[m2] + shift]
        else:
            s = -shift
            j = np.arange(n)
            m1 = (j + s) < n
            m2 = j >= s
            grad[m1] += (1.0 - x[j[m1] + s])
            grad[m2] -= x[j[m2] - s]
    return grad * dx

def _cons_fun(x):
    return x.sum() - _G_TARGET

def _cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter=3000, ftol=1e-16):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coord_descend(h, n, max_sweeps=200, time_limit_s=600):
    """Coordinate descent exploiting linearity in each variable."""
    h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit_s:
            break
        improved = False
        for i in range(n):
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            l = np.arange(-(n-1), n)
            
            valid_plus = (l >= -i) & (l <= n-1-i)
            valid_minus = (l >= i-(n-1)) & (l <= i)
            
            ip = np.clip(i + l, 0, n-1)
            im = np.clip(i - l, 0, n-1)
            hp = np.where(valid_plus, 1.0 - h[ip], 0.0)
            hm = np.where(valid_minus, h[im], 0.0)
            a = hp - hm
            b = corr - h[i] * a
            
            # Minimize max(a*t + b) over t in [0,1]
            # Quick grid + refinement
            ts = np.linspace(0, 1, 51)
            vals = np.max(a * ts[:, None] + b[None, :], axis=1)
            idx = np.argmin(vals)
            lo = max(0.0, ts[max(0, idx-1)] - 0.01)
            hi = min(1.0, ts[min(len(ts)-1, idx+1)] + 0.01)
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            f1 = np.max(a * t1 + b)
            f2 = np.max(a * t2 + b)
            for _ in range(25):
                if f1 < f2:
                    hi = t2; t2 = t1; f2 = f1
                    t1 = hi - gr * (hi - lo)
                    f1 = np.max(a * t1 + b)
                else:
                    lo = t1; t1 = t2; f1 = f2
                    t2 = lo + gr * (hi - lo)
                    f2 = np.max(a * t2 + b)
            t_opt = (lo + hi) / 2
            
            if abs(t_opt - h[i]) > 1e-12:
                h[i] = t_opt
                improved = True
        
        h = project_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
        if not improved and sweep > 3:
            break
    
    return h, best_c5

        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            # Quick coord descent after improvement

                if c5_cd < c5_best:
                    c5_best = c5_cd
                    h_best = h_cd

    for n_try in [360, 420]:
        if elapsed() > budget_s - 80:
            break
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n_try)
        c5_hi = compute_c5(h_hi)
        h_new = do_slsqp(h_hi, n_try, 4, 1500, 1e-16)
        h_new = project_feasible(h_new, n_try)
        c5_new = compute_c5(h_new)
        h_cd, c5_cd = coord_descend(h_new, n_try, max_sweeps=100, time_limit_s=30)
        c5_cd = min(c5_cd, compute_c5(h_cd))
        best_local = min(c5_new, c5_cd)
        if best_local < c5_best:

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    n = _G_N
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = oi - center
        if shift >= 0:
            m1 = j >= shift
            m2 = (j + shift) < n
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s = -shift
            m1 = (j + s) < n
            m2 = j >= s
            grad[m1] += w * (1.0 - x[j[m1] + s])
            grad[m2] -= w * x[j[m2] - s]
    return grad * dx

def _cf(x):
    return x.sum() - _G_TARGET

def _cj(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter=3000, ftol=1e-16):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def _solve_1d_convex_pl(a, b):
    """Minimize max_k(a_k*t + b_k) over t in [0,1]."""
    # Sample grid
    n_grid = 51
    ts = np.linspace(0.0, 1.0, n_grid)
    vals = np.max(a * ts[:, None] + b[None, :], axis=1)
    idx = int(np.argmin(vals))
    half = 1.0 / (n_grid - 1)
    lo = max(0.0, (idx - 1) * half)
    hi = min(1.0, (idx + 2) * half)
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)
    f1 = float(np.max(a * t1 + b))
    f2 = float(np.max(a * t2 + b))
    for _ in range(30):
        if f1 < f2:
            hi, t2, f2 = t2, t1, f1
            t1 = hi - gr * (hi - lo)
            f1 = float(np.max(a * t1 + b))
        else:
            lo, t1, f1 = t1, t2, f2
            t2 = lo + gr * (hi - lo)
            f2 = float(np.max(a * t2 + b))
    return (lo + hi) * 0.5

def coord_descend(h, n, max_sweeps=500, time_limit_s=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    omh = (1.0 - h).copy()
    lags = np.arange(-(n - 1), n)
    n_corr = len(corr)
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit_s:
            break
        improved = False
        order = range(n) if sweep % 2 == 0 else range(n - 1, -1, -1)
        for i in order:
            vp = (lags >= -i) & (lags <= n - 1 - i)
            vm = (lags >= i - (n - 1)) & (lags <= i)
            ip = np.clip(i + lags, 0, n - 1)
            im = np.clip(i - lags, 0, n - 1)
            hp = np.where(vp, omh[ip], 0.0)
            hm = np.where(vm, h[im], 0.0)
            a = hp - hm
            b = corr - h[i] * a
            t_opt = _solve_1d_convex_pl(a, b)
            t_opt = max(0.0, min(1.0, t_opt))
            if abs(t_opt - h[i]) > 1e-12:
                delta = t_opt - h[i]
                corr += delta * a
                h[i] = t_opt
                omh[i] = 1.0 - t_opt
                improved = True
        h = project_feasible(h, n)
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full').copy()
        c5_new = float(np.max(corr) * dx)
        if c5_new < best_c5:
            best_c5 = c5_new
        if not improved and sweep > 5:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    def elapsed():
        return time.time() - t0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: SLSQP coarse-to-fine
    for T in [64, 16, 4, 1]:
        if elapsed() > budget_s - 550:
            break

        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 2: Coordinate descent (main new component)
    if elapsed() < budget_s - 150:

        c5_cd = min(c5_cd, compute_c5(h_cd))
        if c5_cd < c5_best:

    # Phase 3: SLSQP fine polish

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
