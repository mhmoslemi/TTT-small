import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * (2.0 / n)
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad

def fista(h0, target, T, n_iter, lr0, deadline):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = sm_obj_grad(y, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        lr_t = lr
        for _ in range(60):
            h_new = project_constraint(y - lr_t * grad, target)
            obj_new, _ = sm_obj_grad(h_new, T)
            if obj_new <= obj - 1e-16:
                break
            lr_t *= 0.5
        else:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = project_constraint(y, target)
        h = h_new
        t_k = t_new
        lr = min(lr_t * 1.12, lr0)
        if it % 80 == 79:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def local_search(h0, target, n_passes, deadline, seed=77):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(seed)
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(800):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or abs(h[i] - h[j]) < 1e-6:
                continue
            h[i], h[j] = h[j], h[i]
            h = project_constraint(h, target)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
                h = project_constraint(h, target)
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
