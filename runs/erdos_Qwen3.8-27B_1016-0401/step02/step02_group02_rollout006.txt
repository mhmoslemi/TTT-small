import time
from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_smooth(corr, T, n):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    w = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * (2.0 / n)
    return val, w

def subgrad_from_weights(w, h, n):
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

def momentum_opt(h, n, target, T, n_iter, lr0=0.08, beta=0.85):
    x = h.copy().astype(float)
    v = np.zeros(n)
    best = x.copy()
    best_c5 = compute_c5(x)
    for it in range(n_iter):
        corr = np.correlate(x, 1.0 - x, mode="full")
        _, w = lse_smooth(corr, T, n)
        g = subgrad_from_weights(w, x, n)
        lr = lr0 / (1.0 + 0.003 * it)
        v = beta * v + (1.0 - beta) * g
        x_new = project_feasible(x - lr * v, target)
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_c5 = c5
            best = x_new.copy()
        x = x_new
    return best, best_c5

def slsqp_polish(h, n, target, T, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return lse_smooth(corr, T, n)[0]
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        _, w = lse_smooth(corr, T, n)
        return subgrad_from_weights(w, x, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)

        return h

def binary_round(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    frac = target - k_full
    if 0 < frac < 1 and k_full < n:

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_smooth(corr, T, n):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    w = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * (2.0 / n)
    return val, w

def subgrad_weights(w, h, n):
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            g += wi * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += wi * (1.0 - h[:n - o])
            g[:n - o] -= wi * h[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

def momentum_subgrad(h, n, target, T, n_iter, lr0=0.1, beta=0.9):
    x = h.copy().astype(float)
    v = np.zeros(n)

        corr = np.correlate(x, 1.0 - x, mode="full")

            best = x_new.copy()
        x = x_new

def slsqp_polish(h, n, target, T, maxit=300):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_smooth(np.correlate(x, 1.0 - x, mode="full"), T, n)[0]
    def jac(x):
        _, w = lse_smooth(np.correlate(x, 1.0 - x, mode="full"), T, n)
        return subgrad_weights(w, x, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h

def binary_round(h, target):
    n = len(h)
    h_b = np.zeros(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    h_b[order[:k]] = 1.0
    frac = target - k
    if 0 < frac < 1 and k < n:
        h_b[order[k]] = frac
    return h_b

def binary_sa(h, n, target, n_iter=20000, T0=1.0, T1=0.01, rng=None):
    """Simulated annealing on binary structure."""
    if rng is None:
        rng = np.random.default_rng(7)
    best = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    for it in range(n_iter):

        i = ones[rng.integers(len(ones))]

                best = h.copy()
        else:

    return best, best_c5

        if elapsed() < budget_s - 40:
            for T in [15, 8, 4, 2, 1.2]:
                if elapsed() > budget_s - 30:
                    break
                h2, c5_2 = momentum_subgrad(h2, n2, target2, T, 250, 0.09, 0.89)
                if c5_2 < c5_best:
                    c5_best = c5_2
                    h_best = h2.copy()
            h2b = binary_round(h_best, target2)
            c5_2 = compute_c5(h2b)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h2b.copy()

    # Final check
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
