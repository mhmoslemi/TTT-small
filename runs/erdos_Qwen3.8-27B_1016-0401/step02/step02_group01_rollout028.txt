def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_values(h, K, T):

    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se

For lag L (positive), c[n-1+L] = sum_{j=0}^{n-1-L} h[j+L]*(1-h[j])

∂c[n-1+L]/∂h[i]:
- If i-L is valid (0 ≤ i-L ≤ n-1-L, i.e., L ≤ i ≤ n-1): contributes (1-h[i-L])
- If i is valid as j (0 ≤ i ≤ n-1-L): contributes -h[i+L]

So:
g[i] = (1-h[i-L]) * [L ≤ i ≤ n-1] - h[i+L] * [0 ≤ i ≤ n-1-L]

For negative lag -L (L > 0):

∂/∂h[i]:
- h[i] as h[j]: contributes (1-h[i+L]) * [0 ≤ i ≤ n-1-L]
- h[i] as h[j+L] where j+L=i, j=i-L: contributes -h[i-L] * [L ≤ i ≤ n-1]

g[i] = (1-h[i+L]) * [0 ≤ i ≤ n-1-L] - h[i-L] * [L ≤ i ≤ n-1]

    if L > 0:

        #       - h[i+L] for i in [0, n-1-L]  (h[i] as h[j] in (1-h[j]))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_and_sw(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    sh = (corr - mc) * T
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = mc + np.log(se) / T
    sw = e / se
    return obj, sw, corr

def grad_from_sw(h, sw):
    n = len(h)
    c = n - 1
    g = np.zeros(n)
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-12:
            continue
        k = j - c
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] -= w * h[k:]
        else:
            p = -k
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g

def pg_momentum(h, target, T, n_iter, lr0, beta, t0, deadline):

    dx = 2.0 / n
    v = np.zeros(n)
    h_cur = h.copy()

    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = grad_from_sw(h_cur, sw) * dx
        # Nesterov-like: evaluate gradient at lookahead
        # For simplicity, just use momentum
        v = beta * v - lr0 * g
        h_try = project_constraint(h_cur + v, target)
        obj_try, sw_try, _ = corr_and_sw(h_try, T)
        if obj_try < obj - 1e-16:
            h_cur = h_try
            obj = obj_try
            sw = sw_try
        else:
            v = 0.5 * v
            h_try = project_constraint(h_cur + v, target)
            obj_try, sw_try, _ = corr_and_sw(h_try, T)
            h_cur = h_try
            obj = obj_try
            sw = sw_try
        if it % 300 == 299:

            if c5 < best_c5:

                best_h = h_cur.copy()
    c5 = compute_c5(h_cur)

    return best_h, best_c5

def exact_subgrad_polish(h, target, n_iter, t0, deadline):

    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        lags = np.arange(-(n-1), n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        L_star = lags[np.argmax(corr)]

            g[L_star:] += (1.0 - h[:n - L_star])
            g[:n - L_star] -= h[L_star:]
        elif L_star < 0:

            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_sw(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(corr.max())
    sh = (corr - mc) * T
    sh -= sh.max()
    e = np.exp(sh)
    se = float(e.sum())
    obj = mc + np.log(se) / T
    sw = e / se
    return obj, sw, corr

def grad_from_sw(h, sw):
    n = len(h)
    c = n - 1
    g = np.zeros(n)
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-13:
            continue
        k = j - c
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] -= w * h[k:]
        else:
            p = -k
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g

def pg_momentum(h, target, T, n_iter, lr0, beta, deadline):
    n = len(h)
    dx = 2.0 / n
    v = np.zeros(n)
    h_cur = h.copy()
    obj, sw, _ = corr_sw(h_cur, T)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = grad_from_sw(h_cur, sw) * dx
        v = beta * v - lr0 * g
        h_try = project_constraint(h_cur + v, target)
        obj_try, sw_try, _ = corr_sw(h_try, T)
        if obj_try < obj - 1e-16:
            h_cur = h_try
            obj = obj_try
            sw = sw_try
        else:
            v *= 0.25
            h_try = project_constraint(h_cur + v, target)
            obj_try, sw_try, _ = corr_sw(h_try, T)
            h_cur = h_try
            obj = obj_try
            sw = sw_try
        if it % 200 == 199:
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    c5 = compute_c5(h_cur)
    if c5 < best_c5:
        best_c5, best_h = c5, h_cur.copy()
    return best_h, best_c5

def exact_polish(h, target, n_iter, deadline):
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        L_star = int(lags[idx_max])
        mc = float(corr.max())
        g = np.zeros(n)
        if L_star > 0:
            g[L_star:] += (1.0 - h[:n - L_star])
            g[:n - L_star] -= h[L_star:]
        elif L_star < 0:
            p = -L_star
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
        else:
            g[:] = 1.0 - 2.0 * h
        improved = False
        for lr in [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_try = project_constraint(h - lr * g, target)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            if float(corr_try.max()) < mc - 1e-12:
                h = h_try
                improved = True
                break
        if not improved:
            # Try top-2 lags combined
            idx2 = np.argsort(corr)[-2:][0]
            L2 = int(lags[idx2])
            g2 = np.zeros(n)
            if L2 > 0:
                g2[L2:] += (1.0 - h[:n - L2])
                g2[:n - L2] -= h[L2:]
            elif L2 < 0:
                p2 = -L2
                g2[:n - p2] += (1.0 - h[p2:])
                g2[p2:] -= h[:n - p2]
            else:
                g2[:] = 1.0 - 2.0 * h
            g_comb = 0.7 * g + 0.3 * g2
            for lr in [0.01, 0.005, 0.002, 0.001]:
                h_try = project_constraint(h - lr * g_comb, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
                if float(corr_try.max()) < mc - 1e-12:
                    h = h_try
                    break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):

    ]
    for T, iters, lr, beta in schedules:

        if remaining < 60:
            break

    # Phase 2: Exact subgradient polishing

    # Phase 3: Mid-T momentum re-polish (recover from subgradient roughness)

        if c5 < best_c5:

        # Another exact polish

        elapsed = time.time() - t0
        if budget - elapsed < 50:

        h_p = best_h.copy()
        mode = r % 5

            h_p[idx:idx + sz] = 1.0 - h_p[idx:idx + sz]
        elif mode == 1:
            h_p = np.clip(h_p + rng.normal(0, 0.025, n), 0, 1)
        elif mode == 2:
            sh = int(rng.integers(1, 12))
            h_p = np.roll(h_p, sh)
        elif mode == 3:
            h_p = (h_p > rng.uniform(0.3, 0.7)).astype(float)
        else:
            i1 = int(rng.integers(0, n // 3))
            i2 = int(rng.integers(2 * n // 3, n - 12))
            sz = int(rng.integers(3, 12))
            tmp = h_p[i1:i1 + sz].copy()
            h_p[i1:i1 + sz] = h_p[i2:i2 + sz]
            h_p[i2:i2 + sz] = tmp
        h_p = project_constraint(h_p, target)
        T_p = 800 + r * 50
        dl_p = min(t0 + budget - 30, t0 + elapsed + 120)
        h_p, c5_p = pg_momentum(h_p, target, T_p, 1200, 0.005, 0.5, dl_p)
        # Quick exact polish on perturbed

            dl_ep = t0 + budget - 25
            h_p, c5_p2 = exact_polish(h_p, target, 800, dl_ep)
            if c5_p2 < c5_p:
                c5_p, h_p = c5_p2, h_p
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 5: Higher resolution if time allows
    elapsed = time.time() - t0
    if budget - elapsed > 120:
        n2 = 840
        target2 = n2 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x2, x_old, best_h)
        h2 = project_constraint(h2, target2)
        for T2, it2, lr2, b2 in [(300, 1500, 0.008, 0.5), (1500, 2000, 0.003, 0.6)]:
            if time.time() - t0 > budget - 40:
                break
            dl2 = min(t0 + budget - 20, t0 + (time.time() - t0) + 100)
            h2, c5_2 = pg_momentum(h2, target2, T2, it2, lr2, b2, dl2)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()
        if time.time() - t0 < budget - 25:

    best_h = project_constraint(best_h, len(best_h) / 2.0)
    return (best_h, c5_final, len(best_h))
