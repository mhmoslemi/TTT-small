        if time.time() > p3_end:
            break
        target_alt = n_alt / 2.0
        x_alt = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_alt, x_best, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for outer in range(10):
            if time.time() > p3_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = int(np.argmax(corr_a))
            try:
                res = minimize(lag_obj, h_alt, args=(worst_a, target_alt),
                              method='L-BFGS-B', jac=lag_grad, bounds=bounds_alt,
                              options={'maxiter': 1500, 'ftol': 1e-17, 'gtol': 1e-11})
                h_new = fix_constraint(res.x, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        if time.time() < p3_end:
            try:
                res = minimize(sm_obj, h_alt, args=(500, target_alt),
                              method='L-BFGS-B', jac=sm_grad, bounds=bounds_alt,
                              options={'maxiter': 3000, 'ftol': 1e-17, 'gtol': 1e-11})
                h_new = fix_constraint(res.x, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = target_alt
            bounds = bounds_alt
            x_best = np.arange(n) * (2.0 / n)

    # Phase 4: SLSQP final polish for precision (25%)
    p4_end = t0 + budget_s - 5.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    bounds_f = [(0.0, 1.0)] * n_f
    for it in range(12):
        if time.time() > p4_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_k2 = min(3, 2 * n_f - 1)
        top_f = np.argsort(corr_f)[-top_k2:]
        for lag_f in top_f:
            if time.time() > p4_end:
                break
            lag_f = int(lag_f)
            try:
                res = minimize(lag_obj, h_best, args=(lag_f, target_f),
                              method='SLSQP', jac=lag_grad, bounds=bounds_f,
                              options={'maxiter': 500, 'ftol': 1e-16})
                h_new = fix_constraint(res.x, target_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 5: One more L-BFGS-B pass on the very best
    if time.time() < p4_end:
        corr_end = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_end = int(np.argmax(corr_end))
        try:
            res = minimize(lag_obj, h_best, args=(worst_end, target_f),
                          method='L-BFGS-B', jac=lag_grad, bounds=bounds_f,
                          options={'maxiter': 5000, 'ftol': 1e-19, 'gtol': 1e-14})
            h_new = fix_constraint(res.x, target_f)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
