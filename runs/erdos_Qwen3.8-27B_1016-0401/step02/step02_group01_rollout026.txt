    # corr[m] = sum_n h[n] * (1-h[n + m - (n-1)])

    # For m = n-1+k: corr = sum_i h[i+k]*(1-h[i]) (lag k)

        #     contribution: -h[j+lag]
        for j in range(lag, n):

    h_best = h_curr.copy()

            break

        # Find top K lags
        top_idx = np.argpartition(overlaps, -K)[-K:]
        top_vals = overlaps[top_idx]

The gradient of F w.r.t. h is:

where sw_j = exp(T*(corr_j - max)) / sum_k exp(T*(corr_k - max))

And d(corr_j)/dh is what the original code computes (without the dx factor).

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

def project_constraint(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    z = T * (corr - mc)
    z -= z.max()
    e = np.exp(z)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        w = sw[j]
        if w < 1e-15:
            continue
        k = j - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            p = -k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad * dx

def pg_solve(h0, target, T, n_iter, lr0, t0, budget):

    lr = lr0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        obj, grad = sm_obj_grad(h, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = h
        for _ in range(50):
            h_new = project_constraint(h - lr * grad, target)
            obj_new, _ = sm_obj_grad(h_new, T)
            if obj_new <= obj + 1e-16:

            lr *= 0.5
        h = h_new
        lr = min(lr * 1.1, lr0)
        if it % 30 == 29:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, h_best = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, h_best = c5, h.copy()
    return h_best, best_c5

def active_refine(h, target, n_iter, t0, budget):
    """Refine using direct subgradient on top-K lags."""
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    best_c5 = compute_c5(h_c)
    h_best = h_c.copy()
    K = max(3, n // 50)  # number of active lags

        if time.time() - t0 > budget:
            break
        corr = np.correlate(h_c, 1 - h_c, mode="full")
        overlaps = corr * dx
        # Top K lags
        top_k = np.argpartition(overlaps, -K)[-K:]

        z = 200.0 * (top_v - mv)
        z -= z.max()

        grad = np.zeros(n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

def project_constraint(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_obj_grad(h, T, topK=None):
    """Soft-max objective and gradient. If topK set, only use top-K lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    if topK is not None and topK < len(corr):
        idx = np.argpartition(corr, -topK)[-topK:]
        c_use = corr[idx]
    else:
        idx = np.arange(len(corr))
        c_use = corr
    mc = c_use.max()
    z = T * (c_use - mc)
    z -= z.max()
    e = np.exp(z)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    grad = np.zeros(n)
    c = n - 1
    for a in range(len(idx)):
        w = sw[a]
        if w < 1e-15:
            continue
        k = idx[a] - c
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            p = -k
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return obj, grad * dx

def pg_solve(h0, target, T, n_iter, lr0, t0, budget, topK=None):
    h = h0.copy()
    lr = lr0
    best_c5 = compute_c5(h)

        best_c5, h_best = c5, h.copy()

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)


def project_constraint(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def _lift(h, k, n):
    g = np.zeros(n)
    sh = k - (n - 1)
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        g[sh:] += 1.0 - h[:n - sh]
        g[:n - sh] -= h[sh:]
    else:
        p = -sh
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]
    return g


def _sof(h, T, topK):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    if topK is not None and topK < len(corr):
        idx = np.argpartition(corr, -topK)[-topK:]
    else:
        idx = np.arange(len(corr))
    cu = corr[idx]
    mc = float(cu.max())
    z = T * (cu - mc)
    z -= z.max()
    e = np.exp(z)
    se = float(e.sum())
    obj = (mc + np.log(se) / T) * dx
    sw = e / se
    grad = np.zeros(n)
    for a in range(len(idx)):
        w = sw[a]
        if w < 1e-15:
            continue
        grad += w * _lift(h, idx[a], n)
    return obj, grad * dx

        hp = np.clip(hp + rng.normal(0, 0.04, n), 0, 1)
    elif typ == 2:
        hp = (hp > rng.uniform(0.3, 0.7)).astype(float)
    elif typ == 3:
        hp = np.roll(hp, int(rng.integers(1, 10)))
    elif typ == 4:
        m = int(rng.integers(4, max(5, n // 3)))
        i0 = int(rng.integers(0, max(1, n - m)))
        hp[i0:i0 + m] = 1.0 - hp[i0:i0 + m]
    else:
        step = int(rng.integers(2, 5))
        hp[np.arange(0, n, step)] = 1.0
    return hp


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(560)
        h[::2] = 1.0
    n = len(h)
    tgt = n / 2.0
    h = project_constraint(h, tgt)
    hb = h.copy()
    c5b = compute_c5(hb)

    # Phase 1: Full soft-max projected gradient (annealed T)
    for T in [25, 60, 150, 400]:
        rem = dl - time.time()
        if rem < 50:
            break
        h, c5 = _pg(h, tgt, T, max(300, int(rem * 0.35)), 0.04, t0, t0 + rem * 0.85, None)
        if c5 < c5b:
            c5b, hb = c5, h.copy()

    # Phase 2: Top-K focused PG (sharper, faster)
    for T in [300, 800, 2000]:
        rem = dl - time.time()
        if rem < 40:
            break
        h, c5 = _pg(h, tgt, T, max(300, int(rem * 0.4)), 0.03, t0, t0 + rem * 0.85, 20)
        if c5 < c5b:
            c5b, hb = c5, h.copy()

    # Phase 3: Active-set subgradient polish
    rem = dl - time.time()
    if rem > 30:
        h, c5 = _act(hb, tgt, max(200, int(rem * 0.4)), t0, t0 + rem * 0.85)
        if c5 < c5b:
            c5b, hb = c5, h.copy()

    # Phase 4: Perturbation restarts
    for r in range(25):
        rem = dl - time.time()
        if rem < 45:
            break
        hp = _pert(hb, n, rng, r % 6)
        hp = project_constraint(hp, tgt)
        hp, cp = _pg(hp, tgt, 200 + r * 20, max(150, int(rem * 0.25)), 0.03, t0, t0 + rem * 0.6, 15)
        if cp < c5b:
            c5b, hb = cp, hp.copy()
            hp, cp = _act(hp, tgt, 80, t0, t0 + (dl - time.time()) * 0.3)
            if cp < c5b:
                c5b, hb = cp, hp.copy()

    # Phase 5: Final micro-polish at very high T
    rem = dl - time.time()
    if rem > 15:
        hf, cf = _pg(hb, tgt, 5000, max(100, int(rem * 0.4)), 0.007, t0, t0 + rem * 0.7, 10)
        if cf < c5b:
            c5b, hb = cf, hf.copy()
        hf, cf = _act(hb, tgt, max(80, int(rem * 0.35)), t0, t0 + rem * 0.95)
        if cf < c5b:
            c5b, hb = cf, hf.copy()

    # Phase 6: Also check n=280 discretization
    if n > 280 and dl - time.time() > 8:
        xo = np.arange(n) * 2.0 / n
        xn = np.linspace(0, 2, 280, endpoint=False)
        h280 = project_constraint(np.interp(xn, xo, hb), 140.0)
        c280 = compute_c5(h280)
        if c280 < c5b:
            c5b, hb = c280, h280

    c_final = compute_c5(hb)
    return (hb, min(c_final, c5b), len(hb))
