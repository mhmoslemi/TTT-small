import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_and_grad(corr, h, T, n):
    # Returns (objective, gradient) for log-sum-exp smoothed max
    dx = 2.0 / n
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    lse = mc + np.log(np.sum(e)) / T
    obj = lse * dx
    # Gradient
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def slsqp_phase(h, n, target, T, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        o, _ = lse_obj_and_grad(corr, x, T, n)
        return o
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        _, g = lse_obj_and_grad(corr, x, T, n)
        return g
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except:
        return h

def binary_local_search(h_bin, n, target, max_iters, rng):
    # Swap-based local search on binary vector
    c5_cur = compute_c5(h_bin)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new <= c5_cur:
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            c5_cur = c5_new
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, c5_cur

def round_to_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    rem = target - k_full
    if rem > 0 and k_full < n:
        h_bin[order[k_full]] = rem
    return h_bin

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_eval(corr, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    s = T * (corr - mc) - T * mc
    s -= np.max(s)
    e = np.exp(s)
    return (mc + np.log(e.sum()) / T) * dx

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= np.max(s)
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for idx in np.where(w > 1e-7)[0]:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_round(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_eval(np.correlate(x, 1.0-x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0-x, mode="full"), x, T, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, target)
    except:
        return h

    order = np.argsort(h0)[::-1]
    h[order[:int(target)]] = 1.0
    rem = target - int(target)

        else:

        t = max(t * 0.999, t_min)
