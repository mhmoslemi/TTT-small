```python
                s = min(-d, h[i])
                h[i] -= s
```

- Damage metric: number of other lags k' where h[i]=1 contributes to corr[k'] but h[i+k*]=1 would also contribute

```python
            g[:n-o] = -x[o:]
        else:
            p = -o
            g[:n-p] = 1.0 - x[p:]
            g[p:] = -x[:n-p]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h)[::-1] if d > 0 else np.argsort(h)[::-1]
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                s = min(-d, h[i])
                h[i] -= s
                d += s
            if abs(d) < 1e-13:
                break
    return h

def swap_search(h, target, rng, t_end):
    n = len(h)
    dx = 2.0 / n
    ones = list(np.where(h >= 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    no_imp = 0
    it = 0
    while time.time() < t_end and no_imp < 400:
        batch = 500
        best_c = c5
        best_mv = None
        used = set()
        for _ in range(batch):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            k = (min(i, j), max(i, j))
            if k in used:
                continue
            used.add(k)
            vi, vj = h[i], h[j]
            h[i], h[j] = vj, vi
            c = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
            if c < best_c - 1e-15:
                best_c = c
                best_mv = (i, j)
            h[i], h[j] = vi, vj
        if best_mv is not None:
            i, j = best_mv
            vi, vj = h[i], h[j]
            h[i], h[j] = vj, vi
            if vi >= 0.5:
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                zeros.remove(i)
                zeros.append(j)
                ones.remove(j)
                ones.append(i)
            no_imp = 0
        else:
            no_imp += 1
        c5 = best_c
        it += 1
        if it % 100 == 0:
            ones = list(np.where(h >= 0.5)[0])
            zeros = list(np.where(h <= 0.5)[0])
    return h, c5

def topk_obj(x, K):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    return float(np.sort(corr)[-K:].sum())

def topk_grad(x, K):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    dx = 2.0 / n
    center = n - 1
    top_k = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for k_idx in top_k:
        o = k_idx - center
        g = np.zeros(n)
        if o == 0:
            g[:] = 1.0 - 2.0 * x
        elif o > 0:
            g[o:] = 1.0 - x[:n - o]
            g[:n - o] = -x[o:]
        else:
            p = -o
            g[:n - p] = 1.0 - x[p:]
            g[p:] = -x[:n - p]
        grad += g
    return grad * dx

def eq_fun(x, target):
    return x.sum() - target

def eq_jac(x, target):
    return np.ones(len(x))

def slsqp_optimize(h, target, t_end, K=5, maxiter=3000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(topk_obj, h, args=(K,), method='SLSQP',
                       jac=lambda x: topk_grad(x, K),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'catol': 1e-16, 'disp': False})
        if res.success or compute_c5(fix_sum(np.clip(res.x, 0, 1), target)) < compute_c5(h):
            return fix_sum(np.clip(res.x, 0, 1), target)
    except Exception:
        pass
    return h

def target_swap_search(h, target, rng, t_end):
    """Bias swaps toward reducing the worst lag."""
    n = len(h)
    dx = 2.0 / n
    ones = list(np.where(h >= 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
    no_imp = 0
    while time.time() < t_end and no_imp < 300:
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(corr)
        o = int(k_star - (n - 1))
        batch = 600
        best_c = c5
        best_mv = None
        used = set()
        # Generate biased candidates
        for _ in range(batch):
            if o > 0:
                # Try to reduce: i with h[i]=1, h[i+o]=0 -> swap i and i+o
                candidates_i = [i for i in range(n - o) if h[i] >= 0.5]
                if candidates_i:
                    i = candidates_i[rng.integers(len(candidates_i))]
                    j = i + o
                    if h[j] <= 0.5:
                        pass
                    else:
                        i = ones[rng.integers(len(ones))]
                        j = zeros[rng.integers(len(zeros))]
                elif rng.random() < 0.5:
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                else:
                    continue
            elif o < 0:
                p = -o
                candidates_i = [i for i in range(p, n) if h[i] <= 0.5 and h[i-p] >= 0.5]
                if candidates_i:
                    j = candidates_i[rng.integers(len(candidates_i))]
                    i = j - p
                elif rng.random() < 0.5:
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                else:
                    continue
            else:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            k = (min(i, j), max(i, j))
            if k in used:
                continue
            used.add(k)
            vi, vj = h[i], h[j]
            h[i], h[j] = vj, vi
            c = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
            if c < best_c - 1e-15:
                best_c = c
                best_mv = (i, j)
            h[i], h[j] = vi, vj
        if best_mv is not None:
            i, j = best_mv
            vi, vj = h[i], h[j]
            h[i], h[j] = vj, vi
            if vi >= 0.5:
                if i in ones:
                    ones.remove(i)
                ones.append(j)
                if j in zeros:
                    zeros.remove(j)
                zeros.append(i)
            else:
                if i in zeros:
                    zeros.remove(i)
                zeros.append(j)
                if j in ones:
                    ones.remove(j)
                ones.append(i)
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                ones = list(np.where(h >= 0.5)[0])
                zeros = list(np.where(h <= 0.5)[0])
        c5 = best_c
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 280
        h = np.zeros(n_init)
        h[::2] = 1.0
    n = len(h)
    target = n / 2.0
    h = fix_sum(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: Targeted swap search (35%)
    t1 = t0 + budget_s * 0.35
    h1, c5_1 = target_swap_search(h, target, rng, t1)
    if c5_1 < c5_best:
        c5_best, h_best = c5_1, h1
    else:
        h_best = fix_sum(h1, target)
        c5_best = min(c5_best, compute_c5(h_best))

    # Phase 2: Standard swap search (15%)
    t2 = t0 + budget_s * 0.50
    h2, c5_2 = swap_search(h_best.copy(), target, rng, t2)
    if c5_2 < c5_best:
        c5_best, h_best = c5_2, h2

    # Phase 3: SLSQP continuous polish (20%)
    t3 = t0 + budget_s * 0.70
    h3 = h_best.copy()
    for K in [3, 5, 8]:
        if time.time() > t3:
            break
        h3 = slsqp_optimize(h3, target, t3, K=K, maxiter=2000)
        c5_3 = compute_c5(h3)
        if c5_3 < c5_best:
            c5_best, h_best = c5_3, h3

    # Phase 4: Multi-n exploration (20%)
    t4 = t0 + budget_s * 0.90
    for n_alt in [252, 300, 336, 364, 280]:
        if time.time() > t4:
            break
        if n_alt == n:
            continue
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.arange(n_alt) * (2.0 / n_alt)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_sum(h_alt, n_alt / 2.0)
        # Brief swap search
        t_sub = min(t4, time.time() + budget_s * 0.04)
        h_alt, c5_alt = target_swap_search(h_alt, n_alt / 2.0, rng, t_sub)
        # SLSQP
        h_alt = slsqp_optimize(h_alt, n_alt / 2.0, min(t4, time.time() + 30), K=5, maxiter=1500)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best, h_best = c5_alt, h_alt

    # Phase 5: Final polish (10%)
    t5 = t0 + budget_s - 15.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    # Quick swap pass
    h_f, c5_f = target_swap_search(h_best.copy(), target_f, rng, min(t5, t0 + budget_s * 0.95))
    if c5_f < c5_best:
        c5_best, h_best = c5_f, h_f
    # SLSQP final
    h_f2 = slsqp_optimize(h_best.copy(), target_f, t5, K=3, maxiter=2000)
    c5_f2 = compute_c5(h_f2)
    if c5_f2 < c5_best:
        c5_best, h_best = c5_f2, h_f2

    # Ensure constraint and bounds
    h_best = fix_sum(h_best, len(h_best) / 2.0)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
```
