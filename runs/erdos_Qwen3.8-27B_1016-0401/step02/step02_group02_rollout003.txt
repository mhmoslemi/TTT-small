def lse_obj_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

        _, g = lse_obj_grad(corr, h, T, n)
        h = project_feasible(h - lr * g, target)
    return h

    for _ in range(steps):

                p = -o
                g[:n-p] += (1.0 - h[p:])
                g[p:] -= h[:n-p]

            c = compute_c5(hc)
