For the worst lag k, the overlap is sum_i h[i]*(1-h[i+k]). To reduce this, we want to:

def local_search(h, target, n_iters):

        # Overlap = sum_i h[i]*(1-h[i+o])

        # to a low h[j] (where h[j+o] is high) or to h[i+o]

        best_improve = 1e-15

        for i in range(0, n, 3):  # subsample for speed

                h_test = h.copy()
                # Move delta from i to i+1 (or some other position)
                j = (i + 1) % n
                h_test[i] += delta
                h_test[j] -= delta

                c5_test = compute_c5(h_test)

_TARGET = 0.0
_LAG_IDX = 0
_TEMP = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def corr_vec(x):
    return np.correlate(x, 1.0 - x, mode="full")

def sm_obj_impl(x, T):
    n = len(x)
    corr = corr_vec(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def sm_grad_impl(x, T):
    n = len(x)
    corr = corr_vec(x)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def lag_obj_impl(x, lag_idx):
    n = len(x)
    return corr_vec(x)[lag_idx] * (2.0 / n)

def lag_grad_impl(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:

    elif o > 0:

        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += (1.0 - x[p:])
        g[p:] += -x[:n - p]
    return g * dx

    return x.sum() - _TARGET

def sm_grad_wrapper(x):

def lag_obj_wrapper(x):

def lag_grad_wrapper(x):
