def project_constraint(h, target):

def _fobj(x, T, target, n):

def scl_solve(h0, target, T, maxiter, t0, budget):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: _scl_jac(x, target)}]

                       jac=lambda x: _fgrad(x, T, target, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 10:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:

    return h0, compute_c5(h0)
