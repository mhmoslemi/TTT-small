import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        grad[shift:] += (1.0 - h[:n - shift])
        grad[:n - shift] -= h[shift:]
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    return grad

def slim_obj_jac(x, T, mu, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    esum = e.sum()
    obj = (mc + np.log(esum) / T) * dx + mu * (x.sum() - n / 2.0) ** 2
    sw = e / esum
    center = n - 1
    grad = np.zeros(n)
    active = np.where(sw > 1e-10)[0]
    for oi in active:
        grad += sw[oi] * shift_grad_raw(x, n, oi - center)
    grad *= dx
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return obj, grad

def do_lbfgsb(h, n, T, mu, maxiter, time_limit):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    res = minimize(slim_obj_jac, h, args=(T, mu, n), method='L-BFGS-B',
                   jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxfun': maxiter * 2})
    return res.x

def coord_descent(h, n, sweeps=3):
    for sw in range(sweeps):
        for i in range(n):
            h0 = h.copy()
            save = h0[i]
            h0[i] = 0.0
            c0 = np.correlate(h0, 1.0 - h0, mode="full")
            h0[i] = 1.0
            c1 = np.correlate(h0, 1.0 - h0, mode="full")
            d = c1 - c0
            # minimize max(c0 + t*d) for t in [0,1]
            def obj1d(t):
                return np.max(c0 + t * d)
            try:
                res = minimize_scalar(obj1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
                h[i] = res.x
            except:
                pass
        h = project_feasible(h, n)
    return h

def do_slsqp(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        active = np.where(sw > 1e-10)[0]
        for oi in active:
            g += sw[oi] * shift_grad_raw(x, n, oi - center)
        return g * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        grad[shift:] += (1.0 - h[:n - shift])
        grad[:n - shift] -= h[shift:]
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    return grad

def _sm_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    es = e.sum()
    sw = e / es
    center = n - 1
    g = np.zeros(n)
    active = np.where(sw > 1e-10)[0]
    for oi in active:
        g += sw[oi] * shift_grad_raw(x, n, oi - center)
    return g * dx

def do_lbfgsb(h, n, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * (2.0 / n)
    try:
        res = minimize(f, h, method='L-BFGS-B', jac=_sm_grad,
                       args=(n, T), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-12})
        return res.x
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        active = np.where(sw > 1e-10)[0]
        for oi in active:
            g += sw[oi] * shift_grad_raw(x, n, oi - center)
        return g * dx
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coord_descent(h, n, sweeps=3):
    for _ in range(sweeps):
        for i in range(n):
            save = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = save
            d = c1 - c0
            def obj1d(t):
                return np.max(c0 + t * d)
            try:
                r = minimize_scalar(obj1d, bounds=(0.0, 1.0),
                                    method='bounded', options={'xatol': 1e-12})
                h[i] = max(0.0, min(1.0, r.x))
            except Exception:
                pass
        h = project_feasible(h, n)
    return h

def polyak_subgrad(h, n, t0, budget_s, max_iters=200000):
    target = n / 2.0
    dx = 2.0 / n
    h_s = h.copy()
    h_p = h.copy()
    h_best_l = h.copy()
    c5_best_l = compute_c5(h)
    alpha0 = 0.3
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_s + beta * (h_s - h_p), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = int(np.argmax(corr))
        g = shift_grad_raw(h_y, n, idx - (n - 1)) * dx
        alpha = alpha0 / (1.0 + 0.01 * it)
        h_new = h_y - alpha * g
        if (it + 1) % 5 == 0:
            h_new = project_feasible(h_new, n)
        else:
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new += (target - h_new.sum()) / n
            h_new = np.clip(h_new, 0.0, 1.0)
        h_p = h_s
        h_s = h_new
        if (it + 1) % 20000 == 0:
            c = compute_c5(project_feasible(h_s, n))
            if c < c5_best_l:
                c5_best_l = c
                h_best_l = project_feasible(h_s, n).copy()
    return h_best_l, c5_best_l

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:

    return (h_best, c5_final, n)
