        h_best, c5_new = fine_search(h_best, target_best, t0, lim, 2000)
        if c5_new < c5_best:
            c5_best = c5_new

    # Final fine pass
    if time.time() - t0 < budget_s - 5:
        h_best, c5_new = fine_search(h_best, target_best, t0, t0 + budget_s - 2, 2000)
        if c5_new < c5_best:
            c5_best = c5_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
