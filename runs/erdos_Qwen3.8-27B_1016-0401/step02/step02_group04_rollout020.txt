import numpy as np
from scipy.optimize import minimize
import time

_SM = {'T': 100.0, 'top_k': 15}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_grad(h, T, top_k=15):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    k = min(top_k, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    ct = corr[idx]
    sh = T * (ct - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    sw = e / e.sum()
    c0 = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(idx)):
        o = int(idx[i]) - c0
        w = sw[i]
        if w < 1e-14:
            continue
        if o == 0:
            g = g + w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] = g[o:] + w * (1.0 - h[:n - o])
            g[:n - o] = g[:n - o] - w * h[o:]
        else:
            p = -o
            g[:n - p] = g[:n - p] + w * (1.0 - h[p:])
            g[p:] = g[p:] - w * h[:n - p]
    return g * dx

def sm_obj(h, T, top_k=15):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def _obj(x):
    return sm_obj(x, _SM['T'], _SM['top_k'])

def _grad(x):
    return sm_grad(x, _SM['T'], _SM['top_k'])

def proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def do_lbfgs(h0, target, T, mi, tk):
    _SM['T'] = float(T)
    _SM['top_k'] = int(tk)
    n = len(h0)
    try:
        res = minimize(_obj, h0.copy(), jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': mi, 'gtol': 1e-13, 'ftol': 1e-15})
        return proj(res.x, target)
    except Exception:
        return proj(h0, target)

def do_fista(h0, target, T, mi, step, tk):
    h = h0.astype(float).copy()
    zh = h.copy()
    t_k = 1.0
    for _ in range(mi):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = sm_grad(y, T, tk)
        zh = proj(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-14:
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
    return h

def targeted(h, target, nsteps):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for _ in range(nsteps):
        corr = np.correlate(h, 1 - h, mode="full")
        kmax = int(np.argmax(corr))
        o = kmax - (n - 1)
        g = np.zeros(n)
        if o == 0:
            g = 1.0 - 2.0 * h
        elif o > 0:
            g[o:] = 1.0 - h[:n - o]
            g[:n - o] = -h[o:]
        else:
            p = -o
            g[:n - p] = 1.0 - h[p:]
            g[p:] = -h[:n - p]
        g = g * dx
        improved = False
        for s in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = proj(h - s * g, target)
            c5n = compute_c5(h_new)
            if c5n < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def block_perturb(h, rng, n_blocks, block_len, target):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - block_len))
        length = int(rng.integers(block_len // 2, block_len + 1))
        delta = rng.normal(0, 0.1, length)
        h_new[start:start+length] += delta
    return proj(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(140)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = proj(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS with cooling
    for T in [80.0, 200.0, 500.0, 1500.0, 4000.0]:
        if time.time() - t0 > budget_s - 300:
            break
        h_new = do_lbfgs(h_best, target, T, 3000, 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA at several temperatures
    for T, st in [(40.0, 0.3), (150.0, 0.18), (500.0, 0.10)]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = do_fista(h_best, target, T, 4000, st, 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Targeted worst-lag descent
    if time.time() - t0 < budget_s - 120:
        h_new, c5n = targeted(h_best, target, 300)
        if c5n < c5_best:
            c5_best = c5n
            h_best = h_new

    # Phase 4: Perturbation + re-optimize (main search)
    np_ = 0
    while time.time() - t0 < budget_s - 120:
        np_ += 1
        sc = 0.02 + 0.10 / (1.0 + 0.12 * np_)
        hp = h_best + rng.normal(0, sc, n)
        hp = proj(hp, target)
        if np_ % 3 == 0:
            hp = block_perturb(h_best, rng, 2, max(3, n // 30), target)
        T = 40.0 + rng.integers(0, 600).astype(float)
        tk = 10 + rng.integers(0, 10)
        h_new = do_lbfgs(hp, target, T, 1500, tk)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if time.time() - t0 < budget_s - 130:
            h_t, c5t = targeted(h_new, target, 50)
            if c5t < c5_best:
                c5_best = c5t
                h_best = h_t

    # Phase 5: Multi-resolution search
    cur_n = n
    for n_try in [140, 70, 280, 560, 980]:
        if time.time() - t0 > budget_s - 50:
            break
        tt = n_try / 2.0
        xo = np.arange(cur_n) * (2.0 / cur_n)
        xn = np.linspace(0.0, 2.0, n_try, endpoint=False)
        ht = np.interp(xn, xo, h_best)
        ht = proj(ht, tt)
        c5t = compute_c5(ht)
        for T in [100.0, 500.0, 2000.0]:
            if time.time() - t0 > budget_s - 20:
                break
            ht2 = do_lbfgs(ht, tt, T, 2000, 15)
            c52 = compute_c5(ht2)
            if c52 < c5t:
                c5t = c52
                ht = ht2
        ht, c5f = targeted(ht, tt, 100)
        if c5f < c5t:
            c5t = c5f
            ht = ht
        if c5t < c5_best:
            c5_best = c5t
            h_best = ht
            n = n_try
            target = tt
            cur_n = n_try

    # Phase 6: Final polish
    if time.time() - t0 < budget_s - 10:
        h_new = do_lbfgs(h_best, target, 25.0, 4000, 25)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    if time.time() - t0 < budget_s - 5:
        h_new, c5n = targeted(h_best, target, 500)
        if c5n < c5_best:
            c5_best = c5n
            h_best = h_new
    if time.time() - t0 < budget_s - 2:
        h_new = do_fista(h_best, target, 100.0, 2000, 0.15, 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = compute_c5(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, len(h_best))
