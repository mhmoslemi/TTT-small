def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(100):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.clip(h - theta, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()

    sh -= sh.max()

    se = e.sum()

    w = e / se

            p = -k

            grad[p:] -= wk * h[:n-p]
    return obj, grad * dx

def project_and_line(h, grad, target, lr):
    h_new = project_constraint(h - lr * grad, target)
    return h_new

    h = h0.copy()
    v = np.zeros_like(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    w = e / se
    grad = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        wk = w[j]
        if wk < 1e-16:
            continue
        k = j - c
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            p = -k
            grad[:n - p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n - p]
    return obj, grad * dx

_G_T = 1.0
_G_TARGET = 0.0
_G_N = 0

def _sf(x):
    return lse_obj_grad(x, _G_T)[0]

def _sg(x):
    return lse_obj_grad(x, _G_T)[1]

def _se(x):
    return x.sum() - _G_TARGET

def _sej(x):
    return np.ones(_G_N)

def slsqp(h0, target, T, maxiter, t0, tlim):
    global _G_T, _G_TARGET, _G_N
    _G_TARGET = target
    _G_N = len(h0)
    _G_T = T
    try:
        res = minimize(_sf, h0, method='SLSQP', jac=_sg,
                       bounds=[(0.0, 1.0)] * _G_N,
                       constraints=[{'type': 'eq', 'fun': _se, 'jac': _sej}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 10:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def pg(h0, target, T, nit, lr0, t0, tlim):
    h = h0.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(nit):
        if time.time() - t0 > tlim:
            break
        obj, grad = lse_obj_grad(h, T)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        h_new = h
        obj_new = obj
        for bt in range(50):
            h_try = project_constraint(h - lr * grad, target)
            obj_try, _ = lse_obj_grad(h_try, T)
            if obj_try < obj - 1e-14:
                h_new = h_try
                obj_new = obj_try
                break
            lr *= 0.5
        else:
            break
        h = h_new
        lr = min(lr * 1.12, lr0 * 3.0)
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-11:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
