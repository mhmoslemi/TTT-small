            return x.sum() - target
        def cjac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
        try:
            res = minimize(obj_level, h, method='SLSQP', jac=jac_level,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) <= compute_c5(h) + 1e-9:
                h = h_new
        except Exception:
            pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(280) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP continuation (aggressive)
    schedule = [(200, 150), (80, 250), (30, 400), (10, 600), (5, 800),
                (3, 1000), (2, 1200), (1.2, 1500), (0.8, 2000), (0.5, 2500)]
    for T, mit in schedule:
        if elapsed() > budget_s - 350:
            break
        h = slsqp_step(h, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 2: Subgradient with varying K
    for K_val in [10, 6, 4, 3, 2]:
        if elapsed() > budget_s - 250:
            break
        h = subgrad_descent(h_best, n, target, 400, K_val, 0.1, 0.03)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 3: Lag leveling via SLSQP
    if elapsed() < budget_s - 200:
        h = lag_leveling(h_best, n, target, bounds, 3, 500, n_rounds=3)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 4: More SLSQP at low T from best
    for T in [0.6, 0.4, 0.3]:
        if elapsed() > budget_s - 150:
            break
        h = slsqp_step(h_best, n, target, bounds, T, 3000)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 5: Directed random search (many small steps)
    if elapsed() < budget_s - 60:
        n_tries = int(min((budget_s - 60 - elapsed()) * 80, 8000))
        h_dir, c5_dir = directed_search(h_best, n, target, rng, n_tries, scale0=0.02)
        if c5_dir < c5_best:
            c5_best = c5_dir
            h_best = h_dir.copy()

    # Phase 6: Final micro-polish
    if elapsed() < budget_s - 40:
        for _ in range(3):
            if elapsed() > budget_s - 30:
                break
            h = slsqp_step(h_best, n, target, bounds, 0.3, 2000)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            h = subgrad_descent(h_best, n, target, 300, 4, 0.015, 0.08)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 7: Try other resolutions
    if elapsed() < budget_s - 100:
        for n_try in [200, 240, 320, 400]:
            if elapsed() > budget_s - 50:
                break
            target_try = n_try / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, h_best)
            h_try = project_feasible(h_try, target_try)
            bounds_try = [(0.0, 1.0)] * n_try
            for T in [30, 10, 3, 1]:
                if elapsed() > budget_s - 40:
                    break
                h_try = slsqp_step(h_try, n_try, target_try, bounds_try, T, 600)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try.copy()
                n = n_try
                target = target_try

    # Final micro-search with whatever time remains
    if elapsed() < budget_s - 5:
        n_tries = int((budget_s - 5 - elapsed()) * 60)
        if n_tries > 100:
            h_f, c5_f = directed_search(h_best, len(h_best), len(h_best)/2.0, rng, n_tries, scale0=0.005)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
