import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(-(1.0 - h))
            for i in idx:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            idx = np.argsort(h)
            for i in idx:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def sum_eq(x, t):
    return x.sum() - t

def sum_eq_jac(x):
    return np.ones(len(x))

    delta = 0.12

        if time.time() - t0 > budget:
            break

            delta *= 0.72
            sa_temp *= 0.75
            stall = 0

        s = k_star - (n - 1)
        p = abs(s)

        if p == 0:
            vals = hc * (1 - hc)
            idx = int(np.argmax(vals))

                ht[idx] -= delta
            else:
                ht[idx] += delta
        elif mt == 0:
            if s > 0:
                contrib = hc[p:] * (1 - hc[:n - p])
            else:
                contrib = hc[:n - p] * (1 - hc[p:])

            ws = w.sum()

            else:

            d = delta if hc[j1] >= 0.5 else -delta

            ht[j2] += d
