    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return grad * dx

def fix_sum(h, target):
    """Fix sum to target while keeping h in [0,1]"""
    h = np.clip(h, 0.0, 1.0).copy()

            if h.sum() < 1e-15:

            h -= h * (-diff / h.sum())

    diff = target - h.sum()

        # Find a point to adjust
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:

        # Change in corr_{s}: -(grad[a] - grad[b]) * delta

                    for d in [0.05, 0.1, 0.15, 0.2, 0.3]:

                        h_try[a] -= d
                        h_try[b] += d
                        if h_try[a] < -0.01 or h_try[b] > 1.01:

                        c5_try = compute_c5(h_try)

            h_curr[b] += d

            best_c5 = compute_c5(h_curr)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:

                # ...
                break

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    ctr = n - 1
    for o in range(-(n - 1), n):
        w = sw[ctr + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return grad * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += room * (diff / room.sum())
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h -= h * (-diff / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] += min(diff, 1.0 - h[idx])
