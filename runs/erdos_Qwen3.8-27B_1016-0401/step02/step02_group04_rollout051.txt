import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    return dx * (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T)

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n - o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n - o]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def tk_grad(h, bk, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    center = n - 1
    idx = np.argsort(corr)[-bk:]
    cmax = corr.max()
    wts = np.exp(beta * (corr[idx] - cmax))
    wts /= wts.sum()
    grad = np.zeros(n)
    for i, m in enumerate(idx):
        w = wts[i]
        lag = m - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            p = -lag
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def lbfgsb_opt(h0, target, T, pen, maxiter):
    n = len(h0)
    ones = np.ones(n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return sm_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return sm_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-15, 'maxls': 80})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def backtrack_line_search(h0, target, bk, beta, steps):
    h = h0.copy().astype(float)
    c5_cur = compute_c5(h)
    for _ in range(steps):
        g = tk_grad(h, bk, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn * gn
        accepted = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_box_sum(h - alpha * g, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-12:
                h = h_trial
                c5_cur = c5_trial
                accepted = True
                break
        if not accepted:
            break
    return h, c5_cur

def nesterov_bgd(h0, target, bk, beta, steps, step_init):
    h = h0.copy().astype(float)
    y = h.copy()
    t_k = 1.0
    step = step_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(steps):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - y)
        g = tk_grad(y, bk, beta)
        y = project_box_sum(y - step * g, target)
        if np.linalg.norm(y - h) < 1e-14:
            h = y
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = y
        t_k = t_new
    c5_end = compute_c5(h)
    if c5_end < best_c5:
        best_h = h.copy()
        best_c5 = c5_end
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def el():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with high-T soft-max (close to exact max)
    for T in [300, 800, 2000, 5000]:
        if el() > budget_s - 400:
            break
        h_new = lbfgsb_opt(h_best, target, T, 10.0, 4000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Backtracking line search with exact subgradient
    if el() < budget_s - 300:
        h_new, c5_new = backtrack_line_search(h_best, target, 3, 40.0, 200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
    if el() < budget_s - 250:
        h_new, c5_new = backtrack_line_search(h_best, target, 2, 50.0, 200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
    if el() < budget_s - 200:
        h_new, c5_new = backtrack_line_search(h_best, target, 4, 30.0, 200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 3: Nesterov accelerated BGD with exact subgradient
    if el() < budget_s - 200:
        for bk in [2, 3]:
            h_new, c5_new = nesterov_bgd(h_best, target, bk, 40.0, 3000, 0.003)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new

    # Phase 4: Perturbation + re-optimization (structured escapes)
    niter = 0
    while el() < budget_s - 150:
        niter += 1
        scale = max(0.012, 0.08 * np.exp(-0.025 * niter))
        h_pert = h_best + rng.normal(0, scale, n)
        # Structured: add low-frequency sinusoid
        nfreq = rng.integers(1, 4)
        for _ in range(nfreq):
            f = rng.integers(1, max(2, n // 8))
            a = 0.008 * rng.uniform(0.5, 1.5)
            phi = rng.uniform(0, 2 * np.pi)
            h_pert += a * np.sin(2 * np.pi * f * np.arange(n) / n + phi)
        h_pert = project_box_sum(h_pert, target)
        T = 300 + rng.integers(0, 2000)
        h_new = lbfgsb_opt(h_pert, target, T, 10.0, 2000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            # Quick polish after improvement
            if el() < budget_s - 120:
                h_pol, c5_pol = backtrack_line_search(h_new, target, 3, 40.0, 100)
                if c5_pol < c5_best:
                    c5_best = c5_pol
                    h_best = h_pol
            niter = max(0, niter - 3)

    # Phase 5: Try lower resolution n=140
    if el() < budget_s - 100:
        n2 = 140
        tgt2 = n2 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h_best)
        h2 = project_box_sum(h2, tgt2)
        c5_2 = compute_c5(h2)
        c5b2 = c5_2
        hb2 = h2.copy()
        for T in [300, 1000, 3000]:
            if el() > budget_s - 40:
                break
            h_new2 = lbfgsb_opt(hb2, tgt2, T, 10.0, 3000)
            h_new2 = project_box_sum(h_new2, tgt2)
            c5n2 = compute_c5(h_new2)
            if c5n2 < c5b2:
                c5b2 = c5n2
                hb2 = h_new2
        if c5b2 < c5_best:
            c5_best = c5b2
            h_best = hb2
            n = n2
            target = tgt2

    # Phase 6: Final intensive polish
    if el() < budget_s - 15:
        h_new = lbfgsb_opt(h_best, target, 200, 5.0, 10000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new, c5_new = backtrack_line_search(h_best, target, 2, 50.0, 300)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
        h_new, c5_new = backtrack_line_search(h_best, target, 3, 30.0, 200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
