    for it in range(500000):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 0.5 / (1.0 + 0.002 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 8000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a

def _perturbed(h_best, n, rng, pi):
    mode = pi % 4
    if mode == 0:
        sig = 0.002 + 0.003 * (pi // 4) % 5
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h_best.copy()
        lb = rng.integers(0, n - 20)
        ln = rng.integers(5, 25)
        h_p[lb:lb+ln] = 1.0 - h_p[lb:lb+ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h_best.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        h_p[[i, j]] = h_p[[j, i]]
        return project_feasible(h_p, n)
    else:
        sig = 0.005 + 0.002 * (pi // 4) % 3
        h_p = h_best + rng.normal(0, sig, n)
        h_p[rng.integers(0, n)] = 1.0
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Active-set SLSQP coarse to fine
    t1 = t0 + budget_s * 0.30
    for T, tk, mi in [(64, 30, 2000), (32, 25, 2500), (16, 20, 3000),
                        (8, 15, 3500), (4, 12, 4000), (2, 10, 5000), (1, 8, 6000)]:
        if time.time() > t1:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Nesterov multi-shift
    t2 = t0 + budget_s * 0.55
    hs, c5s = _nesterov(h_best, n, t2, rng, top_k=10)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Very fine SLSQP
    t3 = t0 + budget_s * 0.80
    for T, tk, mi in [(4, 12, 4000), (2, 10, 5000), (1, 8, 6000),
                        (0.5, 6, 8000), (0.3, 5, 10000)]:
        if time.time() > t3:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 4: Structured perturbation + SLSQP
    for pi in range(60):
        if time.time() > budget_s - 20:
            break
        hp = _perturbed(h_best, n, rng, pi)
        hn = _slsqp(hp, n, 2, 10, 2000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Final ultra-fine polish
    if time.time() < budget_s - 5:
        hn = _slsqp(h_best, n, 0.5, 5, 5000, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
