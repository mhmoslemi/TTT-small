def lse_obj_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    obj = (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    e = np.exp(sh)
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def init_thue_morse(n, rng):
    seq = np.zeros(n)
    for i in range(n):
        if bin(i).count('1') % 2 == 0:
            seq[i] = 1.0
    ones = int(seq.sum())
    target = n // 2
    if ones > target:
        idx = np.where(seq == 1)[0]
        to_remove = rng.choice(idx, size=ones - target, replace=False)
        seq[to_remove] = 0
    elif ones < target:
        idx = np.where(seq == 0)[0]
        to_add = rng.choice(idx, size=target - ones, replace=False)
        seq[to_add] = 1
    return seq

def init_random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    seq = np.zeros(n)
    seq[idx] = 1.0
    return seq

def init_block(n, rng):
    seq = np.zeros(n)
    n_blocks = int(np.sqrt(n))
    per_block = n // n_blocks
    for b in range(n_blocks):
        w = rng.integers(max(2, per_block // 3), per_block + 1)
        start = b * per_block
        seq[start:start + min(w, n - start)] = 1.0
    return to_binary(seq, n)

def binary_local_search(h, n, n_iter, rng, deadline):
    h = h.copy()
    best_val = compute_c5(h)
    h_best = h.copy()
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top5 = np.argsort(corr)[-5:]
        improved = False
        n_tries = 60
        for t in range(n_tries):
            if rng.random() < 0.75:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                nv = compute_c5(h)
                if nv < best_val - 1e-15:
                    best_val = nv
                    h_best = h.copy()
                    improved = True
                    stall = 0
                else:
                    h[i], h[j] = h[j], h[i]
            else:
                idx = rng.choice(n, size=4, replace=False)
                i1, j1, i2, j2 = int(idx[0]), int(idx[1]), int(idx[2]), int(idx[3])
                if h[i1] == h[j1] or h[i2] == h[j2]:
                    continue
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]
                nv = compute_c5(h)
                if nv < best_val - 1e-15:
                    best_val = nv
                    h_best = h.copy()
                    improved = True
                    stall = 0
                else:
                    h[i1], h[j1] = h[j1], h[i1]
                    h[i2], h[j2] = h[j2], h[i2]
        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(4):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and h[i] != h[j]:
                        h[i], h[j] = h[j], h[i]
                nv = compute_c5(h)
                if nv > best_val + 0.01:
                    h = h_best.copy()
    return h_best, best_val

def lse_obj_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    obj = (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    e = np.exp(sh)
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def proj_grad_polish(h, n, target, T, n_iter, deadline):
    h = h.copy()
    best_val = compute_c5(h)
    h_best = h.copy()
    lr = 0.1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = lse_obj_grad(h, T, n)
        h_new = project_feasible(h - lr * grad, target)
        val = compute_c5(h_new)
        if val < best_val - 1e-14:
            best_val = val
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.4)
        else:
            lr *= 0.6
            h_new = project_feasible(h - lr * grad, target)
            v2 = compute_c5(h_new)
            if v2 < best_val:
                best_val = v2
                h_best = h_new.copy()
            h = h_new
    return h_best, best_val

def slsqp_step(h, n, target, T, maxiter):
    from scipy.optimize import minimize
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = sh - np.max(sh)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = init_random_binary(200, rng)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    n_list = [128, 160, 200, 256, 280, 320, 384]
    deadline_bi = t0 + budget_s * 0.55

    for n in n_list:
        if time.time() - t0 > deadline_bi:
            break
        target = n // 2
        for init_id in range(3):
            if time.time() - t0 > deadline_bi:
                break
            if n == n_init and init_id == 0:
                h = to_binary(h_init, n)
            elif init_id == 0:
                h = init_thue_morse(n, rng)
            elif init_id == 1:
                h = init_random_binary(n, rng)
            else:
                h = init_block(n, rng)
            h2, c5 = binary_local_search(h, n, 2000, rng, deadline_bi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    dl_cont = t0 + budget_s * 0.80
    n_b = len(best_h)
    tgt_b = n_b / 2.0
    for T in [80, 40, 20, 10, 6, 4, 3, 2]:
        if time.time() - t0 > dl_cont:
            break
        h_p = best_h.copy()
        h_p, c5_p = proj_grad_polish(h_p, n_b, tgt_b, T, 1500, dl_cont)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    if time.time() - t0 < t0 + budget_s - 120:
        n_b = len(best_h)
        h_bin = to_binary(best_h, n_b)
        h_bin, c5_bin = binary_local_search(h_bin, n_b, 3000, rng, t0 + budget_s - 60)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    if time.time() - t0 < t0 + budget_s - 40:
        n_b = len(best_h)
        h_s = best_h.copy()
        h_s = slsqp_step(h_s, n_b, n_b / 2.0, 3, 800)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n_final)
