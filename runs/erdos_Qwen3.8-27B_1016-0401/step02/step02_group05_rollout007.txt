def sa_local_search(h_init, target, n_iter, t0, budget, T_start, seed):
    n = len(h_init)
    h_curr = h_init.copy()
    c5_curr = compute_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c5_curr
    T = T_start
    rng = np.random.default_rng(seed)
    cool_rate = max(0.998, 1.0 - 2.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        mt = rng.integers(0, 5)
        h_try = h_curr.copy()
        if mt == 0:
            i = rng.integers(n)
            d = rng.uniform(-0.12, 0.12)
            h_try[i] += d
            j = rng.integers(n)
            if j != i:
                h_try[j] -= d
        elif mt == 1:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            d = rng.uniform(-0.25, 0.25)
            h_try[i] -= d
            h_try[j] += d
        elif mt == 2:
            length = int(rng.integers(2, 7))
            start = int(rng.integers(0, max(1, n - length)))
            h_try[start:start + length] = 1.0 - h_try[start:start + length]
        elif mt == 3:
            i1 = rng.integers(n)
            j1 = rng.integers(n)
            i2 = rng.integers(n)
            j2 = rng.integers(n)
            if len(set([i1, j1, i2, j2])) < 4:
                continue
            d1 = rng.uniform(-0.1, 0.1)
            d2 = rng.uniform(-0.1, 0.1)
            h_try[i1] -= d1
            h_try[j1] += d1
            h_try[i2] -= d2
            h_try[j2] += d2
        else:
            # Binarize a random element + compensate
            i = rng.integers(n)
            target_val = 0.0 if h_curr[i] > 0.5 else 1.0
            delta = target_val - h_curr[i]
            h_try[i] = target_val
            j = rng.integers(n)
            if j != i:
                h_try[j] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = fix_constraint(h_try, target)
        c5_try = compute_c5(h_try)
        dc = c5_try - c5_curr
        if dc < 0.0 or (T > 1e-7 and rng.random() < np.exp(-dc / T)):
            h_curr = h_try
            c5_curr = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        T = max(T * cool_rate, 0.005)
    return best_h, best_c5

def grep_search(h_init, target, n_iter, t0, budget, seed):
    n = len(h_init)
    h_curr = h_init.copy()
    c5_curr = compute_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c5_curr
    rng = np.random.default_rng(seed)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Exact 1D: vary h[i] in [lo, hi], h[j] compensates
        fixed_sum = target - h_curr[i] - h_curr[j]
        lo = max(0.0, fixed_sum - 1.0)
        hi = min(1.0, fixed_sum)
        if hi - lo < 1e-8:
            continue
        # Sample 5 candidates in [lo, hi]
        candidates = np.linspace(lo + 0.01 * (hi - lo), hi - 0.01 * (hi - lo), 5)
        best_local_c = c5_curr
        best_local_h = None
        for cv in candidates:
            h_try = h_curr.copy()
            h_try[i] = cv
            h_try[j] = fixed_sum - cv
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = fix_constraint(h_try, target)
            c = compute_c5(h_try)
            if c < best_local_c - 1e-14:
                best_local_c = c
                best_local_h = h_try.copy()
        if best_local_h is not None:
            h_curr = best_local_h
            c5_curr = best_local_c
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(sum_eq_fun, target=target),
             'jac': sum_eq_jac}]

    # Phase 1: SLSQP soft-max annealing
    temps = [300, 150, 80, 50, 30, 20, 12, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5]
    for T in temps:
        if time.time() - t0 > budget_s - 420:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-16})
            if res.fun < 100:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 2: Simulated annealing (multiple rounds, decreasing T)
    for rnd in range(6):
        if time.time() - t0 > budget_s - 250:
            break
        T_s = 0.04 * (0.65 ** rnd)
        h_sa, c5_sa = sa_local_search(h_best, target, 10000, t0,
                                       budget_s - 230, T_s, 1000 + rnd * 7919)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa

    # Phase 3: Greedy pair search (exact 1D per pair)
    for rnd in range(4):
        if time.time() - t0 > budget_s - 120:
            break
        h_gp, c5_gp = grep_search(h_best, target, 20000, t0,
                                   budget_s - 110, 5000 + rnd * 3301)
        if c5_gp < c5_best:
            c5_best = c5_gp
            h_best = h_gp

    # Phase 4: More SLSQP at very low temperature for final polish
    for T in [1.0, 0.5, 0.3, 0.15, 0.1, 0.05, 0.03]:
        if time.time() - t0 > budget_s - 40:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 6000, 'ftol': 1e-18})
            if res.fun < 100:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 5: Final SA + greedy interleaved
    if time.time() - t0 < budget_s - 15:
        h_sa, c5_sa = sa_local_search(h_best, target, 6000, t0,
                                       budget_s - 8, 0.008, 99991)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa
    if time.time() - t0 < budget_s - 3:
        h_gp, c5_gp = grep_search(h_best, target, 8000, t0,
                                   budget_s - 1, 77777)
        if c5_gp < c5_best:
            c5_best = c5_gp
            h_best = h_gp

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
