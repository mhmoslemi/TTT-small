import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0: continue
                h[i] += add; diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0: continue
                h[i] -= sub; diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def lag_obj(x, lag, n):
    return np.correlate(x, 1.0 - x, mode="full")[lag] * (2.0 / n)

def lag_grad(x, lag, n):
    o = lag - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] += -x[:n-p]
    return g * (2.0 / n)

def sm_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def sm_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    g = np.zeros(n)
    center = n - 1
    for oi in range(2 * n - 1):
        w = sw[oi]
        if w < 1e-12: continue
        o = oi - center
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n-o])
            g[:n-o] += -w * x[o:]
        else:
            p = -o
            g[:n-p] += w * (1.0 - x[p:])
            g[p:] += -w * x[:n-p]
    return g * (2.0 / n)

    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h0, target, n, T, max_iter, t_end, t0):
    h_f = h0.copy()
    h_m = h0.copy()
    t_k = 1.0
    L = 50.0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = sm_grad(h_f, T, n)
        step = 1.0 / L
        h_try = project_box_sum(h_f - step * g, target)
        f_try = sm_obj(h_try, T, n)
        f_curr = sm_obj(h_f, T, n)
        if f_try > f_curr + 1e-13:
            L *= 2.0
            step = 1.0 / L
            h_try = project_box_sum(h_f - step * g, target)
        else:
            L = max(10.0, L * 0.7)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_m = h_try + ((t_k - 1.0) / t_new) * (h_try - h_f)
        h_m = np.clip(h_m, 0.0, 1.0)
        h_f = h_try
        t_k = t_new
    return fix_constraint(h_f, target)

def slsqp_lag(h, lag, n, target, maxiter, t_end, t0):
    global _TARGET
    _TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(lag_obj, h, args=(lag, n), method='SLSQP',
                       jac=lag_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = fix_constraint(np.clip(res.x, 0, 1), target)
        return h_new
    except Exception:
        return h

def slsqp_smooth(h, T, n, target, maxiter, t_end, t0):

    _TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

                       jac=sm_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = fix_constraint(np.clip(res.x, 0, 1), target)

    except Exception:
        return h
