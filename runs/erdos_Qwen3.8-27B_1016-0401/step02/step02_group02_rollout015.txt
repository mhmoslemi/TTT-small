    lo, hi = -1.0, 1.0
    for _ in range(50):

        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:

def smooth_max(corr, T):
    mc = np.max(corr)
    sh = T * (corr - mc) - np.max(T * (corr - mc))
    return (mc + np.log(np.sum(np.exp(sh))) / T)

def corr_gradient_single(h, idx):
    """Gradient of corr[idx] w.r.t. h"""
    n = len(h)
    center = n - 1
    o = idx - center
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] = 1.0 - h[:n-o]
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] = 1.0 - h[p:]
        g[p:] -= h[:n-p]
    return g

    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return mc + np.log(np.sum(np.exp(sh))) / T

    n = len(h)
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-12:
            continue
        g = corr_gradient_single(h, idx)
        grad += w[idx] * g
    return grad

def lse_val(corr, T):
    m = np.max(corr)
    s = np.exp(T * (corr - m) - np.max(T * (corr - m)))
    return m + np.log(s.sum()) / T

def lse_grad_full(corr, h, T):
    n = len(h)
    m = np.max(corr)
    s = T * (corr - m)
    s -= np.max(s)
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:

        else:
            p = -o

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(x - t, 0.0, 1.0)

def lse_val(corr, T):
    m = np.max(corr)
    s = T * (corr - m)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / T

def lse_grad(corr, h, T):
    n = len(h)
    m = np.max(corr)
    s = T * (corr - m)
    s -= np.max(s)
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad

def subgrad_exact(corr, h, K):

        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T) * dx
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    try:

                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def equalize_top(corr, h, n, target):
