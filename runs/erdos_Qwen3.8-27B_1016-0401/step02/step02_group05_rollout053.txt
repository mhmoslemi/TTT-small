    # Project x onto {y: 0<=y_i<=1, sum(y)=target}
    # Using bisection on lambda
    lo, hi = -max(x) - 1, -min(x) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix any remaining error
    diff = target - y.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-saturated
        free = (y > 0) & (y < 1)
        if free.sum() > 0:
            y[free] += diff / free.sum()
            y = np.clip(y, 0, 1)
    return y

    fine = [0.01, 0.005, 0.002, 0.001]
    # Phase 1: SLSQP warm-up
    for T in [8, 4, 2]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = slsqp_step(h_best, target, T, 3000)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-12:
            c5_best = c5
            h_best = h_new
    # Phase 2: Subgradient descent
    h_sg, c5_sg = subgrad_phase(h_best, target, 3000, t0, budget_s - 350)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    # Phase 3: Local search coarse
    h_ls, c5_ls = local_search(h_best, target, 3000, t0, budget_s - 280, coarse, fine)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls
    # Phase 4: SLSQP fine-tune
    for T in [1.5, 1.0]:
        if time.time() - t0 > budget_s - 180:
            break
        h_new = slsqp_step(h_best, target, T, 5000)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-12:
            c5_best = c5
            h_best = h_new
    # Phase 5: Local search fine
    fine2 = [0.005, 0.003, 0.002, 0.001, 0.0005]
    h_ls2, c5_ls2 = local_search(h_best, target, 15000, t0, budget_s - 60, fine2, fine2, top_lags=4, top_pairs=4)
    if c5_ls2 < c5_best:
        c5_best = c5_ls2
        h_best = h_ls2
    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s - 30:
        h_new = slsqp_step(h_best, target, 0.8, 4000)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-12:
            c5_best = c5
            h_best = h_new
    # Phase 7: Final polish local search
    ultra_fine = [0.002, 0.001, 0.0005, 0.0003]
    h_ls3, c5_ls3 = local_search(h_best, target, 10000, t0, budget_s - 3, ultra_fine, ultra_fine, top_lags=2, top_pairs=2)
    if c5_ls3 < c5_best:
        c5_best = c5_ls3
        h_best = h_ls3
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
