        if mx_f < c5_best:
            c5_best = mx_f
            h_best = h_f.copy()

    if elapsed() < p4_end:
        _TS = tgt_f
        C_fin = np.correlate(h_best, 1.0 - h_best, mode='full')
        for li in np.argsort(C_fin)[-3:]:
            if elapsed() > p4_end:
                break
            _CUR_LAG = int(li)
            try:
                res = minimize(lag_obj, h_best, args=(int(li),), method='SLSQP',
                               jac=_jacf, bounds=bounds_f, constraints=cons_f,
                               options={'maxiter': 3000, 'ftol': 1e-17})
                h_new = fix_sum(np.clip(res.x, 0, 1), tgt_f)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
