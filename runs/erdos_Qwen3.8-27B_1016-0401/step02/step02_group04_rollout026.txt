**Key insight**: The correlation c_k = sum_i h[i](1-h[i+k]) can be written as:
c_k = sum_i h[i] - sum_i h[i]h[i+k] = n/2 - sum_i h[i]h[i+k] (since sum h[i] = n/2)

So minimizing max_k c_k is equivalent to maximizing min_k sum_i h[i]h[i+k].

The minimum of sum_i h[i]h[i+k] over k should be as large as possible. This is like maximizing the minimum autocorrelation of h.

For the lag-0 value: sum_i h[i]² = sum_i h[i]² (which we want to maximize for binary-like h)
For lag-k: sum_i h[i]h[i+k] (which we want to be as large as possible for all k)

But we can't make ALL lags large simultaneously (that would require h to be constant, which gives h=1/2 everywhere, which makes c_0 = 1/2).

So the tradeoff is:
- Make h close to binary → c_0 small, but h is "spiky" → some lags might have low autocorrelation
- Make h smooth → all lags have comparable autocorrelation, but c_0 is larger

The optimal balances these. The previous solution at 0.380908 is close to optimal, so we're in a delicate balance.

    dx = 2.0 / n

    return np.correlate(h, 1 - h, mode="full")

def softmax_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    T = max(T, 1.0)
    shifted = np.clip(T * (corr - mc), -50, 50)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def softmax_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    T = max(T, 1.0)
    shifted = np.clip(T * (corr - mc), -50, 50)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_box_sum(h, target):

            lo = mid
        else:
            hi = mid

    try:

                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-13})

        h_pert = project_box_sum(h_pert, target)
        T = 60 + rng.integers(0, 400)
        if rng.random() < 0.4:
            h_new = lbfgs_run(h_pert, target, T, 8.0, 2500)
        else:
            step = min(0.25, 2.0 / (T + 1.0))
            h_new = fista_run(h_pert, target, T, 3000, step)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Multi-resolution
    for n_try in [140, 210, 350]:
        if elapsed() > budget_s * 0.88:
            break
        t_try = n_try / 2.0
        h_try = interpolate_h(h_best, n_try)
        h_try = project_box_sum(h_try, t_try)
        best_tol = h_try
        best_c5_tol = compute_c5(h_try)
        for T in [40, 150, 400]:
            if elapsed() > budget_s * 0.92:
                break
            h_t = fista_run(best_tol, t_try, T, 3000, min(0.3, 2.0 / (T + 1)))
            h_t = project_box_sum(h_t, t_try)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5_tol:
                best_c5_tol = c5_t
                best_tol = h_t
        if best_c5_tol < c5_best:
            c5_best = best_c5_tol
            h_best = best_tol
            n = n_try
            target = t_try

    # Phase 5: Final aggressive polish
    if elapsed() < budget_s - 30:
        for T in [20, 50, 100, 300]:
            if elapsed() > budget_s - 10:
                break
            h_new = lbfgs_run(h_best, target, T, 1.0, 5000)
            h_new = project_box_sum(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Final targeted refinement
    if elapsed() < budget_s - 5:
        for _ in range(5):
            h_targ = targeted_perturb(h_best, scale=0.01, ntop=1)
            h_targ = project_box_sum(h_targ, target)
            h_targ = fista_run(h_targ, target, 50, 2000, 0.2)
            h_targ = project_box_sum(h_targ, target)
            c5_t = compute_c5(h_targ)
            if c5_t < c5_best:
                c5_best = c5_t
                h_best = h_targ

    h_best = project_box_sum(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
