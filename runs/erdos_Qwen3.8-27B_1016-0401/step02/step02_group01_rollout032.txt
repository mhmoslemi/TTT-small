def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo+hi)/2
        if np.sum(np.clip(h-mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def corr_and_topk(h, topk=10):
    corr = np.correlate(h, 1-h, mode="full")
    idx = np.argpartition(corr, -topk)[-topk:]
    return corr, idx

def sm_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    mc = corr.max()
    s = T*(corr - mc)
    s -= s.max()
    e = np.exp(s)
    return (mc + np.log(e.sum())/T) * (2.0/n)

def sm_grad(h, T):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode="full")
    mc = corr.max()
    s = T*(corr - mc)
    s -= s.max()
    e = np.exp(s)
    w = e/e.sum()
    g = np.zeros(n)
    c = n-1
    for j in range(2*n-1):
        wj = w[j]
        if wj < 1e-16: continue
        k = j - c
        if k == 0:
            g += wj*(1-2*h)
        elif k > 0:
            g[:n-k] += wj*(1-h[k:])
            g[k:] -= wj*h[:n-k]
        else:
            p = -k
            g[:n-p] += wj*(1-h[p:])
            g[p:] -= wj*h[:n-p]
    return g*dx

def slsqp_run(h0, target, T, maxiter, t0, budget):
    n = len(h0)
    bounds = [(0,1)]*n
    def f(x):
        return sm_obj(x, T)
    def jf(x):
        return sm_grad(x, T)
    cons = [{'type':'eq', 'fun': lambda x: x.sum()-target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h0, method='SLSQP', jac=jf, bounds=bounds,
                      constraints=cons, options={'maxiter':maxiter, 'ftol':1e-14, 'maxls':50})
        h = project_sum(res.x, target)
        return h, compute_c5(h)
    except:
        return h0, compute_c5(h0)

    beta = 0.9
    lr = 0.05
    best = h.copy()

        v = beta*v + (1-beta)*grad
        h_new = project_sum(h - lr*v, target)
        # Adaptive step

        obj_old = sm_obj(h, T)
        if obj_new > obj_old + 1e-12:
            lr *= 0.5
            h_new = project_sum(h - lr*v, target)
            if sm_obj(h_new, T) > obj_old + 1e-12:

        else:
            lr = min(lr*1.05, 0.1)
            h = h_new
        if it % 200 == 199:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best = c5, h
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_G_T = 1.0
_G_TGT = 0.0
_G_N = 0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _sm_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = corr.max()
    s = _G_T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    return (mc + np.log(e.sum()) / _G_T) * dx

def _sm_grad(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = corr.max()
    s = _G_T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    c = n - 1
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-16:
            continue
        k = j - c
        if k == 0:
            g += wj * (1.0 - 2.0 * x)
        elif k > 0:
            g[:n-k] += wj * (1.0 - x[k:])
            g[k:] -= wj * x[:n-k]
        else:
            p = -k
            g[:n-p] += wj * (1.0 - x[p:])
            g[p:] -= wj * x[:n-p]
    return g * dx

def _cons_fun(x):
    return x.sum() - _G_TGT

def _cons_jac(x):
    return np.ones(_G_N)

def slsqp_run(h0, target, T, maxiter, t0, budget):
    global _G_T, _G_TGT, _G_N
    _G_TGT = target
    _G_N = len(h0)
    _G_T = T
    bounds = [(0.0, 1.0)] * len(h0)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_sm_obj, h0, method='SLSQP', jac=_sm_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 60})
        h = project_sum(res.x, target)
        return h, compute_c5(h)
    except Exception:
        return h0, compute_c5(h0)

    h = h0.copy()

    lr = 0.08
    best_h = h.copy()

    obj_cache = _sm_obj(h)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        grad = _sm_grad(h)
        v = beta * v + (1.0 - beta) * grad
        h_new = project_sum(h - lr * v, target)
        obj_new = _sm_obj(h_new)
        if obj_new > obj_cache + 1e-14:
            lr *= 0.4
            h_new = project_sum(h - lr * v, target)
            obj_new = _sm_obj(h_new)
            if obj_new > obj_cache + 1e-14:

def perturb(h, rng, r):
    n = len(h)
    hp = h.copy()
    if r % 5 == 0:
        idx = rng.integers(0, n)
        sz = rng.integers(3, 25)
        for i in range(sz):
            hp[(idx + i) % n] = 1.0 - hp[(idx + i) % n]
    elif r % 5 == 1:
        hp = np.clip(hp + 0.04 * rng.standard_normal(n), 0, 1)
    elif r % 5 == 2:
        thr = 0.3 + 0.4 * rng.random()
        hp = (hp > thr).astype(float)
    elif r % 5 == 3:
        shift = rng.integers(1, 20)
        hp = np.roll(hp, int(shift))
    else:
        freq = rng.integers(3, 30)
        amp = 0.1 + 0.2 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        hp = np.clip(hp + amp * np.sin(2*np.pi*freq*t/n + phase), 0, 1)
    return hp

def upscale(h, n_new):

    if n_new == n_old:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
