    pert_limit = time.time() + 100
    if pert_limit < t0 + budget_s - 80:
        for pi in range(70):
            if time.time() - t0 > budget_s - 80:
                break
            sigma = 0.004 + 0.018 * (pi % 6)
            h_p = h_best + rng.normal(0, sigma, n)
            h_p = project_feasible(h_p, n)
            h_new = do_slsqp(h_p, n, 4, 2500, 1e-14)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
            if pi % 5 == 0:
                s = int(rng.integers(1, max(2, n // 7)))
                h_s = np.roll(h_best, s)
                h_s = project_feasible(h_s, n)
                h_new = do_slsqp(h_s, n, 4, 2000, 1e-14)
                h_new = project_feasible(h_new, n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Phase 6: Local sparse perturbations
    h_best, c5_best = local_perturb_search(
        h_best, n, rng, 40, t0 + budget_s - 60)

    # Phase 7: Final micro-polish with very tight tolerance
    for T in [2, 1, 1, 1]:
        if time.time() - t0 > budget_s - 8:
            break
        h_new = do_slsqp(h_best, n, T, 12000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
