def project(h, target):
    """Project onto {0<=x<=1, sum(x)=target} via bisection."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):

        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any tiny residual
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(result < 1.0)[0]
            for i in idx[:len(idx)]:
                s = min(diff, 1.0 - result[i])
                if s > 0:
                    result[i] += s
                    diff -= s
                    if abs(diff) < 1e-14:
                        break
        else:
            idx = np.where(result > 0.0)[0]
            for i in idx[:len(idx)]:
                s = min(-diff, result[i])
                if s > 0:
                    result[i] -= s
                    diff += s
                    if abs(diff) < 1e-14:
                        break
    return result

def softmax_obj(x, T):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T)

def softmax_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted - np.max(shifted))  # numerical stability
    sw = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad

def proj_subgrad(h, target, T, n_iter, alpha, t0, time_limit):
    """Projected subgradient descent on softmax surrogate."""
    h_curr = h.copy()
    best_obj = softmax_obj(h_curr, T)
    dx = 2.0 / len(h_curr)
    best_h = h_curr.copy()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        g = softmax_grad(h_curr, T)
        a = alpha / (1.0 + 0.01 * it)
        h_new = project(h_curr - a * g, target)
        obj = softmax_obj(h_new, T)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_obj

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = target - r.sum()
    if d > 1e-14:
        for i in np.where(r < 1.0)[0]:
            s = min(d, 1.0 - r[i]); r[i] += s; d -= s
            if d < 1e-14: break
    elif d < -1e-14:
        for i in np.where(r > 0.0)[0]:
            s = min(-d, r[i]); r[i] -= s; d += s
            if d > -1e-14: break
    return r

def softmax_val(x, T):
    c = np.correlate(x, 1-x, mode="full")
    mc = c.max()
    return mc + np.log(np.exp(T*(c-mc)).sum())/T

def softmax_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1-x, mode="full")
    mc = c.max()
    sh = T*(c-mc)
    e = np.exp(sh-sh.max())
    w = e/e.sum()
    g = np.zeros(n)
    ctr = n-1
    for o in range(-(n-1), n):
        wi = w[ctr+o]
        if wi < 1e-13: continue
        if o == 0: g += wi*(1-2*x)
        elif o > 0:
            g[o:] += wi*(1-x[:n-o])
            g[:n-o] -= wi*x[o:]
        else:
            p = -o
            g[:n-p] += wi*(1-x[p:])
            g[p:] -= wi*x[:n-p]
    return g

def proj_subgrad(h, target, T, nsteps, alpha, t0, tmax):
    h = h.copy()
    best, best_h = softmax_val(h, T), h.copy()
    for it in range(nsteps):
        if time.time() - t0 > tmax: break
        g = softmax_grad(h, T)
        a = alpha / (1 + 0.005*it)
        h = project(h - a*g, target)
        v = softmax_val(h, T)
        if v < best: best, best_h = v, h.copy()
    return best_h, best
