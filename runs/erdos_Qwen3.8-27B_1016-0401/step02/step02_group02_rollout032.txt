from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode="full")
    return float(np.max(corr)*dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo+hi)/2
        if np.clip(x-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x-(lo+hi)/2, 0, 1)

def lse_obj_and_grad(corr, h, T, n):
    dx = 2.0/n
    mc = np.max(corr)
    sh = T*(corr-mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e/e.sum()
    obj = (mc + np.log(e.sum())/T)*dx
    center = n-1
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-13:
            continue
        o = idx-center
        wi = w[idx]
        if o==0:

        else:

            grad[:n-p] += wi*(1-h[p:])
            grad[p:] -= wi*h[:n-p]

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-13:
            continue
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad * dx

def slsqp_opt(h, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        return lse_obj_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)[0]
    def jacf(x):
        return lse_obj_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)[1]
    try:
        res = minimize(objf, h, method='SLSQP', jac=jacf, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

    best_c5 = compute_c5(h)
    best_h = h.copy()
    nm = np.linalg.norm(g)
    if nm < 1e-12:
        return h, best_c5

    for _ in range(30):
        if time_fn() > budget:

        h_try = project_feasible(h - mid * g, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:

            best_h = h_try.copy()
        # coarse bisection: check midpoint improvement
        if c5_try < best_c5 + 1e-4:
            lo = mid
        else:
            hi = mid
    return best_h, best_c5

def swap_search(h, n, target, max_iters, rng, time_fn, budget):
    best_c5 = compute_c5(h)
    center = n - 1
    stall = 0
    for it in range(max_iters):
        if time_fn() > budget:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(corr)[-4:]
        g = np.zeros(n)
        for idx in top_k:
            o = idx - center
            if o == 0:
                g += (1.0 - 2.0 * h)
            elif o > 0:
                g[o:] += (1.0 - h[:n - o])
                g[:n - o] -= h[o:]
            else:
                p = -o
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
        g /= (np.max(np.abs(g)) + 1e-12)
        improved = False
        candidates = []
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        candidates.append((i_max, i_min))
