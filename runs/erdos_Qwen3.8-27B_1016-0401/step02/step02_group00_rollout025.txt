    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def corr_and_grad(h):
    """Return (correlations, index of max)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr, np.argmax(corr)

def lag_gradient(h, lag_idx):
    """Gradient of correlation at a specific lag."""
    n = len(h)
    center = n - 1
    o = lag_idx - center
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] -= h[:n-p]
    return g

def smax_value_grad(h, T):
    """Soft-max value and gradient of the correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    val = mc + np.log(e.sum()) / T
    center = n - 1
    g = np.zeros(n)
    for idx in range(2*n-1):
        w = sw[idx]
        if w < 1e-13:
            continue
        o = idx - center
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n-o])
            g[:n-o] -= w * h[o:]
        else:
            p = -o
            g[:n-p] += w * (1.0 - h[p:])
            g[p:] -= h[:n-p]
    return val, g
