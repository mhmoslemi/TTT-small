def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

    h_bin[order[:k_full]] = 1.0
    if k_full < n:

    return h_bin

            c5_cur = c5_new

            zeros[zeros == j] = i

            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            ones = np.unique(ones)
            zeros = np.unique(zeros)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def slsqp_step(h, n, target, bounds, T, maxiter):
    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cfun, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h

def round_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    h_bin[order[:k]] = 1.0
    if k < n:
        h_bin[order[k]] = target - k
    return h_bin

def binary_local_search(h, t_end, rng, max_rounds=100):
    """Greedy + SA on binary space."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        ones = np.where(h == 1.0)[0].copy()
        zeros = np.where(h == 0.0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            for j in zeros:
                if time.time() > t_end:
                    return best_h, best_c5
                h[i], h[j] = 0.0, 1.0
                c5_new = compute_c5(h)
                if c5_new < c5 - 1e-15:
                    c5 = c5_new
                    improved = True
                    zeros = zeros[zeros != j]
                    zeros = np.append(zeros, i)
                    ones = ones[ones != i]
                    ones = np.append(ones, j)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best_c5

def sa_search(h, t_end, rng, n_iters=50000):
    """Simulated annealing on binary space."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    T0, T1 = 0.01, 0.0001
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        o = int(idx) - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target

    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

Let me also add: **coordinate descent**. For each coordinate i:
- Fix all other coordinates
- Optimize h[i] to minimize C5
This is a 1D problem (h[i] ∈ [0,1]) and can be solved analytically or with a few evaluations.

The C5 as a function of h[i] (with all else fixed):
C5(t) = dx * max_k corr_k(t)
where corr_k(t) is the k-th correlation value as a function of h[i]=t.

Each corr_k is linear in t (since it's a sum of products, and only terms involving h[i] change). So C5(t) is the max of a set of linear functions, i.e., a convex piecewise-linear function of t. We can minimize it by finding where the "kink" nearest to the unconstrained minimum is.

This is promising! Coordinate descent should converge and handle the fractional values naturally.

For each coordinate i, the correlation at offset o (in the full correlate output, index k = n-1+o) changes as:
- If i appears in the sum for offset o, corr_k changes linearly with t = h[i]
- The slope is ±1 (or 0 if i doesn't appear)

So for coordinate i:
corr_k(t) = a_k * t + b_k for some constants a_k, b_k
where a_k ∈ {-1, 0, 1} (since h[i] appears at most once in each correlation)

Wait, actually h[i] appears in multiple terms of a single correlation. Let me think again.

c_o = sum_{valid k} h[k+o]*(1-h[k])

h[i] appears in this sum when:
- i = k+o (i.e., k = i-o): term is h[i]*(1-h[i-o])
- i = k (i.e., k = i): term is h[i+o]*(1-h[i])

So h[i] contributes to c_o through:
- h[i]*(1-h[i-o]) if i-o is valid
- h[i+o]*(1-h[i]) if i+o is valid

The derivative of c_o with respect to h[i]:
dc_o/dh[i] = (1-h[i-o]) * I(i-o valid) - h[i+o] * I(i+o valid)

This is in {-1, -1, 0, 0, 1} depending on the values.

= (1-h[i-o]) (when only first appears)
= -h[i+o] (when only second appears)
= 0 (when neither appears)

OK so each c_o is affine in h[i] with slope s_o = dc_o/dh[i]. The slope is in [-1, 1].

C5(t) = dx * max_o (c_o(t)) = dx * max_o (c_o(0) + s_o * t)

This is the max of affine functions, hence convex. The minimum is where the subgradient contains 0, or at a boundary (t=0 or t=1).

We can find the minimum by evaluating C5 at a few points and bisecting, or by finding the pair of offsets that "cross" at the optimum.

For each coordinate i (in some order):
