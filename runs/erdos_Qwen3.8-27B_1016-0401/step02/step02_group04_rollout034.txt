def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def topk_obj(h, k):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argsort(corr)[-k:]
    return np.mean(corr[idx]) * (2.0 / n)

def topk_grad(h, k):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argsort(corr)[-k:]
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for oi in idx:
        o = oi - center
        if o == 0:
            grad += (1.0 - 2.0 * h) / k
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o]) / k
            grad[:n - o] -= h[o:] / k
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:]) / k
            grad[p:] -= h[:n - p] / k
    return grad * dx

def penalty_obj(h, C_target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    thresh = C_target / dx
    viol = np.maximum(0.0, corr - thresh)
    return 0.5 * np.sum(viol * viol) * dx

def penalty_grad(h, C_target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    thresh = C_target / dx
    active = corr > thresh
    center = n - 1
    grad = np.zeros(n)
    for oi in np.where(active)[0]:
        o = oi - center
        v = 2.0 * (corr[oi] - thresh)
        if o == 0:
            grad += v * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += v * (1.0 - h[:n - o])
            grad[:n - o] -= v * h[o:]
        else:
            p = -o
            grad[:n - p] += v * (1.0 - h[p:])
            grad[p:] -= v * h[:n - p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgs_soft(h0, n, target, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return soft_max_obj(x, T)
    def grad(x):
        return soft_max_grad(x, T)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-16})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def lbfgs_topk(h0, n, target, k, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return topk_obj(x, k)
    def grad(x):
        return topk_grad(x, k)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-16})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def lbfgs_penalty(h0, n, target, C_target, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return penalty_obj(x, C_target)
    def grad(x):
        return penalty_grad(x, C_target)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-16})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def fista_optimize(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    step = step_init
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = soft_max_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        diff = np.linalg.norm(zh - h)
        if diff < 1e-14 * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def downsample_upsample(h, n_fine, n_coarse):
    x_f = np.arange(n_fine) * (2.0 / n_fine)
    x_c = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    h_c = np.interp(x_c, x_f, h)
    h_c = project_box_sum(h_c, n_coarse / 2.0)
    h_f = np.interp(x_f, x_c, h_c)
    return project_box_sum(h_f, n_fine / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Multi-resolution cycle (downsample -> optimize -> upsample)
    for ds in [4, 2]:
        if elapsed() > budget_s * 0.25:
            break
        n_s = max(30, n // ds)
        x_f = np.arange(n) * (2.0 / n)
        x_s = np.linspace(0.0, 2.0, n_s, endpoint=False)
        h_s = np.interp(x_s, x_f, h_best)
        h_s = project_box_sum(h_s, n_s / 2.0)
        for T in [40, 150, 500]:
            if elapsed() > budget_s * 0.25:
                break
            h_s = lbfgs_soft(h_s, n_s, n_s / 2.0, T, 4000)
        h_f = np.interp(x_f, x_s, h_s)
        h_f = project_box_sum(h_f, target)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 2: Progressive T-cooling at full resolution
    for T in [40, 120, 350, 900, 2000, 5000]:
        if elapsed() > budget_s * 0.50:
            break
        h_new = lbfgs_soft(h_best, n, target, T, 5000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Top-k targeted refinement
    for k in [3, 5, 8, 15]:
        if elapsed() > budget_s * 0.65:
            break
        for rep in range(2):
            h_new = lbfgs_topk(h_best, n, target, k, 4000)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Penalty-based bisection on C_5
    if elapsed() < budget_s * 0.80:
        C_hi = c5_best
        C_lo = 0.3790
        for _ in range(12):
            if elapsed() > budget_s * 0.80:

            C_mid = (C_lo + C_hi) / 2.0
            h_new = lbfgs_penalty(h_best, n, target, C_mid, 2000)

                if c5_check < c5_best:
                    c5_best = c5_check
                    h_best = h_new
            else:
                C_lo = C_mid

    # Phase 5: Random restarts with multi-objective local search
    n_tries = 0
    while elapsed() < budget_s * 0.92:

        scale = 0.06 * (0.85 ** (n_tries * 0.3))
        h_p = h_best + rng.normal(0, scale, n)
        h_p = project_box_sum(h_p, target)
        T = int(50 + rng.integers(0, 400))
        h_new = lbfgs_soft(h_p, n, target, T, 2000)
        h_new = lbfgs_topk(h_new, n, target, 3, 1500)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: FISTA polish with very high T
    if elapsed() < budget_s * 0.96:

            h_best = h_new

            h_best = h_new

    h_best = project_box_sum(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
