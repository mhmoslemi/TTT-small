from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def make_beatty(n, alpha):
    x = (np.arange(n, dtype=float) * alpha) % 1.0
    idx = np.argsort(x)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    h[0] = 0
    for i in range(1, n):
        h[i] = 1.0 - h[i >> 1] if (i & 1) == 0 else h[i >> 1]
    # Actually, standard Thue-Morse: t(i) = popcount(i) mod 2
    # Let me use the recursive definition: t(2i) = t(i), t(2i+1) = 1-t(i)
    # Or just: t(i) = number of 1-bits in i, mod 2
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c += v & 1
            v >>= 1
        h[i] = c & 1
    return h

def sa_search(h_bin, n, t_end, rng):

    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    zeros = list(np.where(h <= 0.5)[0])

    no_imp = 0
    while time.time() < t_end:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0; h[j] = 1.0
        mc = np.max(np.correlate(h, 1.0 - h, mode="full"))
        delta = mc - curr_mc
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            curr_mc = mc
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            c5 = mc * dx
            if c5 < best_c5:
                best_c5 = c5; best_h = h.copy(); no_imp = 0
            else:

        else:

            no_imp += 1
        T = max(T * cool, T_min)

_Tpar = 1.0

def _sobj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    s = np.exp(_Tpar * (c - mc)).sum()
    return (mc + np.log(s) / _Tpar) * (2.0 / _N)

def _sjac(x):
    n = _N
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    w = np.exp(_Tpar * (c - mc))
    w /= w.sum()
    dx = 2.0 / n
    ctr = n - 1
    g = np.zeros(n)
    for oi in range(2 * n - 1):
        wi = w[oi]
        if wi < 1e-14:

        if k >= 0:

            g[k:] -= wi * x[:n - k]
        else:
            m = -k
            g[m:] += wi * (1.0 - x[:n - m])
            g[:n - m] -= wi * x[m:]

def _cfun(x):
    return x.sum() - _N / 2.0

def _cjac(x):
    return np.ones(_N)

def slsqp_solve(h, n, T, maxit, ftol):

    _N, _Tpar = n, float(T)
    try:

                     bounds=[(0, 1)] * n,

                     options={'maxiter': maxit, 'ftol': ftol})
        return r.x
    except Exception:
        return h.copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def make_beatty(n, alpha):
    x = (np.arange(n, dtype=float) * alpha) % 1.0
    order = np.argsort(x)
    h = np.zeros(n)
    h[order[:n // 2]] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v, c = i, 0
        while v:
            c += v & 1
            v >>= 1
        h[i] = c & 1
    return h

_Tpar = 1.0

def _sobj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    s = np.exp(_Tpar * (c - mc)).sum()
    return (mc + np.log(s) / _Tpar) * (2.0 / _N)

def _sjac(x):
    n = _N
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    w = np.exp(_Tpar * (c - mc))
    w /= w.sum()
    dx = 2.0 / n
    ctr = n - 1
    g = np.zeros(n)
    for oi in range(2 * n - 1):
        wi = w[oi]
        if wi < 1e-14:
            continue
        k = oi - ctr
        if k >= 0:
            g[:n - k] += wi * (1.0 - x[k:])
            g[k:] -= wi * x[:n - k]
        else:
            m = -k
            g[m:] += wi * (1.0 - x[:n - m])
            g[:n - m] -= wi * x[m:]
    return g * dx

def _cfun(x):
    return x.sum() - _N / 2.0

def _cjac(x):
    return np.ones(_N)
