def grad_corr_at_shift(h, s):
    """Gradient of correlation at shift s w.r.t. all h[j]"""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        # corr = sum_{i=0}^{n-1-s} h[i](1-h[i+s])
        # d/dh[j]: (1-h[j+s]) if j < n-s, minus h[j-s] if j >= s
        valid_i = np.arange(0, n - s) if s < n else np.array([])
        if len(valid_i) > 0:
            grad[:n-s] += (1 - h[s:])  # (1-h[i+s]) for i=0..n-1-s
        if s > 0:
            grad[s:] += -h[:n-s]  # -h[i] for i=0..n-1-s, at position i+s
    else:
        p = -s

        grad[:n-p] += -h[p:]  # -h[i] for i=p..n-1, at position i-p
    return grad

def fix_constraint(h, target):

def project_onto_feasible(h, target):

    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) < 1e-14:

            grad[:n-p] += -h[p:]
    return grad

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            order = np.argsort(-(1.0 - h))
            for i in order:
                if diff <= 0: break
                s = min(diff, 1.0 - h[i])
                h[i] += s; diff -= s
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= 0: break
                s = min(-diff, h[i])
                h[i] -= s; diff += s
    return h

    h = np.clip(h, 0.0, 1.0)
    return fix_constraint(h, target)

        for j, idx in enumerate(top_idx):

    if time.time() - t0 < budget_s * 0.6:
        for T in [5, 2.5, 1.2]:
            if time.time() - t0 > budget_s * 0.6: break
            try:
                res = minimize(smooth_max_obj, h_best, method='SLSQP',
                              jac=smooth_max_jac, args=(T,),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2500, 'ftol': 1e-15})
                if res.fun < 50:
                    h_new = fix_constraint(res.x.copy(), target)
                    c5 = compute_c5(h_new)
                    if c5 < c5_best - 1e-12: c5_best = c5; h_best = h_new
            except: pass

    # Phase 4: annealed pair search
    if time.time() - t0 < budget_s * 0.8:
        h_a, c5_a = anneal_pair_search(h_best, target, 20000, t0, budget_s * 0.8)
        if c5_a < c5_best: c5_best = c5_a; h_best = h_a

    # Phase 5: SLSQP low temp
    if time.time() - t0 < budget_s * 0.9:
        try:
            res = minimize(smooth_max_obj, h_best, method='SLSQP',
                          jac=smooth_max_jac, args=(0.8,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12: c5_best = c5; h_best = h_new
        except: pass

    # Phase 6: binary rounding check
    h_b, c5_b = try_binary_round(h_best, target)
    if c5_b < c5_best:
        h_from_b = h_b.copy()
        c5_best = c5_b; h_best = h_b
        h_a2, c5_a2 = anneal_pair_search(h_best, target, 5000, t0, budget_s - 10, 0.001)
        if c5_a2 < c5_best: c5_best = c5_a2; h_best = h_a2

    # Phase 7: final polish
    if time.time() - t0 < budget_s - 5:
        h_f, c5_f = multi_pair_search(h_best, target, 4000, t0, budget_s - 5)
        if c5_f < c5_best: c5_best = c5_f; h_best = h_f
        h_a3, c5_a3 = anneal_pair_search(h_best, target, 8000, t0, budget_s - 2, 0.0005)
        if c5_a3 < c5_best: c5_best = c5_a3; h_best = h_a3

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
