import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0: continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0: continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += (1.0 - x[:n-o])
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - x[p:])
        g[p:] += -x[:n-p]
    return g * dx

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def make_eq(n):
    def eq_fun(x):
        return x.sum() - n / 2.0
    def eq_jac(x):
        return np.ones(len(x))
    return {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

def active_set_optimize(h, n, bounds, max_iter_outer, ftol, time_limit, t_start):
    """Cycle through active lags, optimizing each one."""
    target = n / 2.0
    c5 = compute_c5(h)
    cons = [make_eq(n)]
    for outer in range(max_iter_outer):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top lags
        top_k = min(8, 2*n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for lag in sorted(top_idx, key=lambda l: -corr[l]):
            if time.time() - t_start > time_limit:
                break
            try:
                res = minimize(lag_obj, h, args=(lag,), method='SLSQP',
                             jac=lambda x, l=lag: lag_grad(x, l),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': ftol})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-15:
                    c5 = c5_new
                    h = h_new
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5

def soft_max_optimize(h, n, bounds, T_list, time_limit, t_start):
    """Soft-max continuation."""
    target = n / 2.0
    c5 = compute_c5(h)
    cons = [make_eq(n)]
    for T in T_list:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(sm_obj, h, args=(T,), method='SLSQP',
                          jac=lambda x, t=T: sm_grad(x, t),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                h = h_new
        except Exception:
            pass
    return h, c5

def nesterov_polish(h, n, time_limit, t_start):
    """Subgradient Nesterov method for min-max."""
    target = n / 2.0
    c5 = compute_c5(h)
    h_sg = h.copy()
    h_mom = h.copy()
    L = 30.0
    center = n - 1
    for it in range(20000):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        worst = np.argmax(corr)
        o = worst - center
        g = np.zeros(n)
        if o == 0:
            g[:] = (1.0 - 2.0 * h_sg)
        elif o > 0:
            g[o:] += (1.0 - h_sg[:n-o])
            g[:n-o] += -h_sg[o:]
        else:
            p = -o
            g[:n-p] += (1.0 - h_sg[p:])
            g[p:] += -h_sg[:n-p]
        g *= 2.0 / n
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        h_new_sg = h_mom - (L / t_k) * g
        h_new_sg = fix_constraint(h_new_sg, target)
        h_mom = h_new_sg + ((L - 1.0) / t_k) * (h_new_sg - h_sg)
        h_mom = np.clip(h_mom, 0.0, 1.0)
        h_sg = h_new_sg
        L += 1.0
        if it % 200 == 0:
            c5_sg = compute_c5(h_sg)
            if c5_sg < c5:
                c5 = c5_sg
                # Keep h as is for c5 tracking, but don't update h globally here
    return h_sg, c5

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n

    def f(x):
        return x.sum() - n / 2.0

        return np.ones(len(x))
    return {'type': 'eq', 'fun': f, 'jac': j}

def do_active_set(h, n, bounds, max_outer, ftol, t_limit, t_ref):
    target = n / 2.0
    cons = [eq_constraint(n)]
    c5 = compute_c5(h)
    for outer in range(max_outer):
        if time.time() - t_ref > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = min(10, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]

                               jac=lambda x, l=lag: lag_grad(x, l),

def do_nesterov(h, n, n_iter, t_limit, t_ref):
    target = n / 2.0
    c5 = compute_c5(h)
    h_sg = h.copy()
    h_mom = h.copy()

    center = n - 1

        L += 1.0
        if it % 300 == 0:
