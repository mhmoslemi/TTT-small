import time

_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def slsqp_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def slsqp_cf(x):
    return x.sum() - _G_TARGET

def slsqp_cj(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cf, 'jac': slsqp_cj}]
    try:
        res = minimize(slsqp_obj, h, method='SLSQP', jac=slsqp_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def multi_nesterov(h_init, n, t0, budget_s, max_iters=300000):
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    # Adaptive: use top-m shifts with softmax
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        # Adaptive temperature: start warm, cool down
        T_adapt = max(0.05, 2.0 / (1.0 + 0.01 * it))
        sh = (corr - mc) / T_adapt
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-10:

            g += w * shift_grad_raw(h_y, n, oi - center)

            if c5c < c5_best:

    return h_best, c5_best

def binary_local_search(h_cont, n, t0, budget_s, max_rounds=5000):
    """Round to binary, then do local swap search."""
    h_bin = (h_cont >= 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    # Ensure exactly n/2 ones
    ones = np.sum(h_bin > 0.5)

        idx = np.where(h_bin > 0.5)[0]
        order = np.argsort(-h_cont[idx])  # remove ones with lowest original value
        for i in order[:int(ones - n/2)]:
            h_bin[idx[i]] = 0.0
    elif ones < n / 2:
        idx = np.where(h_bin < 0.5)[0]
        order = np.argsort(-h_cont[idx])  # add zeros with highest original value
        for i in order[:int(n/2 - ones)]:

    h_best = h_bin.copy()
    
    idx1 = np.where(h_bin > 0.5)[0]
    idx0 = np.where(h_bin < 0.5)[0]

                h_bin = h_tmp
                c5_cur = c5_tmp
                improved = True
        if improved and c5_cur < compute_c5(h_best):
            h_best = h_bin.copy()
        if not improved and rd > 10:
            break
    return h_best, c5_cur

def periodic_search(n, t0, budget_s):

    periods = [7, 10, 14, 17, 20, 28, 35, 40, 56, 70]
    best_h = None

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0
_AS_N = 0
_AS_TARGET = 0.0
_AS_SHIFTS = []
_AS_NSHIFT = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_corr_single(h, n, shift):
    if shift >= 0:
        return float(np.sum(h[shift:] * (1.0 - h[:n - shift])))
    else:
        s = -shift
        return float(np.sum(h[:n - s] * (1.0 - h[s:])))

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def lse_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def lse_cf(x):
    return x.sum() - _G_TARGET

def lse_cj(x):
    return np.ones(_G_N)

    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lse_cf, 'jac': lse_cj}]
    try:
        res = minimize(lse_obj, h, method='SLSQP', jac=lse_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:

def multi_nesterov(h_init, n, t0, budget_s, max_iters=250000):
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        T_adapt = max(0.08, 1.5 / (1.0 + 0.008 * it))
        sh = (corr - mc) / T_adapt
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-10:
                continue
            g += w * shift_grad_raw(h_y, n, oi - center)
        g *= dx
        alpha = 0.25 / (1.0 + 0.004 * it)
        h_new = h_y - alpha * g
        if (it + 1) % 5 == 0:
            h_new = project_feasible(h_new, n)
        h_prev = h_cur
        h_cur = h_new
        if (it + 1) % 10000 == 0:
            c5c = compute_c5(h_cur)

    c5c = compute_c5(h_cur)
    if c5c < c5_best:
        c5_best = c5c
        h_best = h_cur.copy()
    return h_best, c5_best

def binary_local_search(h_cont, n, t0, budget_s, max_rounds=4000):
    h_bin = project_feasible(h_cont.copy(), n)
    h_bin = np.where(h_bin >= 0.5, 1.0, 0.0)
    ones = int(np.sum(h_bin > 0.5))
    target = n // 2
    if ones > target:
        idx = np.where(h_bin > 0.5)[0]
        order = np.argsort(-h_cont[idx])
        for k in order[:ones - target]:
            h_bin[idx[k]] = 0.0
    elif ones < target:
        idx = np.where(h_bin < 0.5)[0]
        order = np.argsort(-h_cont[idx])
        for k in order[:target - ones]:
            h_bin[idx[k]] = 1.0
    c5_cur = compute_c5(h_bin)
    h_best = h_bin.copy()
    rng_local = np.random.default_rng(7)
    for rd in range(max_rounds):
        if time.time() - t0 > budget_s:
            break
        idx1 = np.where(h_bin > 0.5)[0]
        idx0 = np.where(h_bin < 0.5)[0]
        n_try = min(300, max(50, len(idx1) * len(idx0) // 10))
        improved = False
        for _ in range(n_try):
            i = idx1[rng_local.integers(len(idx1))]
            j = idx0[rng_local.integers(len(idx0))]
            h_tmp = h_bin.copy()
            h_tmp[i], h_tmp[j] = h_tmp[j], h_tmp[i]
            c5_tmp = compute_c5(h_tmp)

                c5_cur = c5_tmp
                improved = True
        if c5_cur < compute_c5(h_best) - 1e-9:
