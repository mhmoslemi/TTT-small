        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try different resolution
    for n_try in [200, 320]:
        if time.time() - t0 > budget_s - 60:
            break
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_t = np.interp(x_new, x_old, h_best)
        h_t = project_feasible(h_t, n_try)
        for (P, K) in [(30, 10), (20, 6)]:
            if time.time() - t0 > budget_s - 30:
                break
            h_new = slsqp_solve(h_t, n_try, P, K, 2500, 1e-14, 0.0)
            h_new = project_feasible(h_new, n_try)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                n = n_try

    # Final: tiny polish
    if time.time() - t0 < budget_s - 5:
        h_new = slsqp_solve(h_best, n, 50, 4, 3000, 1e-15, 0.0)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
