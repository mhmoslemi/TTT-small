    if diff > 0:

def subgradient_step(h, target, alpha):
    """One step of projected subgradient descent on max corr."""
    n = len(h)
    corr = get_corr(h)
    k_star = np.argmax(corr)
    s = k_star - (n - 1)
    p = abs(s)
    grad = np.zeros(n)
    if p == 0:
        grad = 1.0 - 2.0 * h
    elif s > 0:
        grad[:n-p] += -h[p:]
        grad[p:] += (1.0 - h[:n-p])
    else:
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]
    dx = 2.0 / n
    # Projected gradient step
    h_new = h - alpha * dx * grad
    # Project onto simplex-like constraint: [0,1]^n intersect {sum = target}
    h_new = np.clip(h_new, 0, 1)
    h_new = fix_sum(h_new, target)
    return h_new, corr

def softmax_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def softmax_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        p = abs(o)
        if p == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[p:] += w * (1.0 - x[:n-p])
            grad[:n-p] += -w * x[p:]
        else:
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
                diff -= add
            if abs(diff) < 1e-14:
                break
    else:
        order = np.argsort(-h)
        for idx in order:
            rem = min(-diff, h[idx])
            if rem > 0:
                h[idx] -= rem
                diff += rem
            if abs(diff) < 1e-14:
                break
    return h

def smax_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def smax_jac(x, T):

    grad = np.zeros(n)
    center = n - 1

        w = sw[center + o]
        if w < 1e-12:
            continue
        p = abs(o)

            grad += w * (1.0 - 2.0 * x)

        else:

            grad[p:n] += -w * x[:n-p]
    return grad * dx
