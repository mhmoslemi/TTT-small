import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:

            p = -o

            grad[p:] -= w * h[:n - p]

        else:

    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

    h = h0.copy()
    if step is None:
        step = 0.3 * 2.0 / n

        idx = np.argsort(corr)[::-1][:top_k]

        if np.max(np.abs(h_new - h)) < 1e-14:
            break
        # Slowly decay step
        step *= 0.999
        h = h_new
    return h

def fista_optimize(h0, target, T, max_iter, step_init):

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

    return h

            h_pert[idx] += rng.normal(0, 0.3, len(idx))
        elif n_perturb % 3 == 1:
            # Block shift perturbation
            block = n // 10
            shift = rng.integers(1, block)
            h_pert = np.roll(h_best, shift)
            h_pert = np.clip(h_pert + rng.normal(0, 0.05, n), 0, 1)
        else:
            h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T = 20 + rng.integers(0, 60)
        h_new = lbfgsb_optimize(h_pert, n, target, T, 5.0, 2000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            best_pert_scale = scale * 2.0
            best_pert_scale = min(best_pert_scale, 0.08)

    # Phase 5: Fine-tune with very low T (T=3-10) for final precision
    for T in [5, 8, 12, 3]:
        if elapsed() > budget_s - 30:
            break
        h_new = lbfgsb_optimize(h_best, n, target, T, 1.0, 4000, gtol=1e-16)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final cyclic + L-BFGS polish
    if elapsed() < budget_s - 10:
        h_cyc = cyclic_lag_minimize(h_best, target, top_k=3, max_rounds=300)
        c5_cyc = compute_c5(h_cyc)
        if c5_cyc < c5_best:
            c5_best = c5_cyc
            h_best = h_cyc
        h_new = lbfgsb_optimize(h_best, n, target, 8, 1.0, 3000, gtol=1e-16)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = project_box_sum(h_best, target)
    return (h_best, c5_final, len(h_best))
