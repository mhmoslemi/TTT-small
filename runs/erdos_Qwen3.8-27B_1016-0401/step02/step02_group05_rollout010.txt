import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_simplex(v, target):
    lo = -v.max()
    hi = 1.0 - v.min()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def softmax_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = corr_array(x)
    mc = np.max(corr)
    shifted = (corr - mc) / T
    val = mc + T * np.log(np.mean(np.exp(shifted)))
    return val * dx

def softmax_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = corr_array(x)
    mc = np.max(corr)
    shifted = (corr - mc) / T
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def subgrad_step(h, corr, n_active, alpha):
    n = len(h)
    mc = np.max(corr)
    # Active set: lags within small tolerance of max
    active = np.where(corr >= mc - 1e-10)[0]
    # Take top n_active
    if len(active) > n_active:
        top_k = np.argsort(corr[active])[::-1][:n_active]
        active = active[top_k]
    # Average gradient
    center = n - 1
    grad = np.zeros(n)
    for idx in active:
        o = idx - center
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n - o])
            grad[:n - o] += -h[o:]
        else:
            p = -o
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] += -h[:n - p]
    grad /= len(active)
    dx = 2.0 / n
    grad *= dx
    return h - alpha * grad

def local_search_block(h, target, n_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    corr = corr_array(h)
    best_c5 = float(np.max(corr) * dx)
    h_curr = h.copy()
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        if stall > 200:
            break
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        # Try multiple move types
        best_move_c5 = best_c5
        best_move_h = None
        deltas = [0.05, 0.1, 0.15, 0.2, 0.3]
        
        if p == 0:
            for d in deltas:
                idx = np.argmax(h_curr * (1.0 - h_curr))
                h_try = h_curr.copy()
                if h_curr[idx] > 0.5:
                    h_try[idx] -= d
                else:
                    h_try[idx] += d
                h_try = project_simplex(np.clip(h_try, 0, 1), target)
                c5_try = float(np.max(corr_array(h_try)) * dx)
                if c5_try < best_move_c5:
                    best_move_c5 = c5_try
                    best_move_h = h_try
        else:
            if s > 0:
                contrib = h_curr[p:] * (1.0 - h_curr[:n - p])
            else:
                contrib = h_curr[:n - p] * (1.0 - h_curr[p:])
            # Try top few contributing pairs
            top_idx = np.argsort(contrib)[::-1][:5]
            for ti in top_idx:
                if s > 0:

                else:

                    h_try[j1] -= d
                    h_try[j2] += d

                stall += 1
        else:
            stall += 1
        if it % 200 == 0:
            Tf *= 0.85
    return h_best_ls, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[0::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_simplex(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    def eqf(x):
        return x.sum() - target
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]

    # Phase 1: SLSQP annealing
    temps = [200, 100, 50, 25, 12, 6, 3, 2, 1.5, 1.0, 0.8]
    for T in temps:
        if elapsed() > budget_s * 0.30:
            break
        try:
            res = minimize(sm_obj, h_best, method='SLSQP',
                          jac=sm_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-15})
            if res.fun < 200:
                h_new = project_simplex(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 2: Precision local search
    h_ls, c5_ls = preclocal(h_best, target, 8000, t0, budget_s * 0.55, T_sa=0.003)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 3: Restart loop
    while elapsed() < budget_s - 280:
        r = rng.random()
        if r < 0.25:
            h_init = h_best + rng.normal(0, 0.04, n)
        elif r < 0.45:
            h_init = h_best.copy()
            st = int(rng.integers(0, max(1, n - 50)))
            ln = int(rng.integers(8, 50))
            h_init[st:st + ln] = 1.0 - h_init[st:st + ln]
        elif r < 0.65:
            h_init = h_best.copy()
            for _ in range(int(rng.integers(2, 5))):
                st = int(rng.integers(0, max(1, n - 20)))
                ln = int(rng.integers(4, 15))
                h_init[st:st + ln] = 1.0 - h_init[st:st + ln]
        elif r < 0.8:
            h_init = h_best + rng.normal(0, 0.07, n)
        elif r < 0.92:
            kern = np.exp(-0.5 * (np.arange(-8, 9) / 4.0) ** 2)
            h_init = np.convolve(h_best, kern / kern.sum(), mode='same')
        else:
            h_init = (rng.random(n) >= 0.5).astype(float)
        h_init = project_simplex(np.clip(h_init, 0, 1), target)
        try:
            res = minimize(sm_obj, h_init, method='SLSQP',
                          jac=sm_jac, args=(20.0,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-13})
            if res.fun < 200:
                h_new = project_simplex(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-13:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass
        h_ls2, c5_ls2 = preclocal(h_best, target, 1000, t0, budget_s - 230, T_sa=0.002)
        if c5_ls2 < c5_best:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 4: Final SLSQP
    for T in [3, 1.5, 1.0, 0.7, 0.5]:
        if elapsed() > budget_s - 45:
            break
        try:
            res = minimize(sm_obj, h_best, method='SLSQP',
                          jac=sm_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            if res.fun < 200:
                h_new = project_simplex(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-14:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 5: Final precise local search
    h_ls3, c5_ls3 = preclocal(h_best, target, 3000, t0, budget_s - 2, T_sa=0.001)
    if c5_ls3 < c5_best:
        c5_best = c5_ls3
        h_best = h_ls3

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
