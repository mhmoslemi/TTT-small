def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid

def smooth_topk(h, n, T, K):
    """Return (value, gradient) of smooth top-K max of overlaps."""
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    if K >= len(corr):
        K = len(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    tv = corr[top_idx]
    sh = T * (tv - tv.max())
    e = np.exp(sh)
    sw = e / e.sum()
    val = float(np.dot(sw, tv)) * dx
    grad = np.zeros(n)
