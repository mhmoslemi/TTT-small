import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        a = np.arange(shift, n)
        if len(a):
            grad[a] += (1.0 - h[a - shift])
        b = np.arange(n - shift)
        if len(b):
            grad[b] -= h[b + shift]
    else:
        s = -shift
        a = np.arange(n - s)
        if len(a):
            grad[a] += (1.0 - h[a + s])
        b = np.arange(s, n)
        if len(b):
            grad[b] -= h[b - s]
    return grad

def smooth_vg(h, n, K, T):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    if K and K < len(corr):
        top = np.argpartition(corr, -K)[-K:]
        tv = corr[top]
        sh = T * (tv - tv.max())
        e = np.exp(sh)
        sw = e / e.sum()
        val = np.dot(sw, tv) * dx
        grad = np.zeros(n)
        for i in range(K):
            if sw[i] > 1e-11:
                grad += sw[i] * shift_grad(h, n, top[i] - center)
        return val, grad * dx
    sh = T * (corr - corr.max())
    e = np.exp(sh)
    sw = e / e.sum()
    val = (corr.max() + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    for i in range(len(corr)):
        if sw[i] > 1e-11:
            grad += sw[i] * shift_grad(h, n, i - center)
    return val, grad * dx

def nesterov_loop(h0, n, t_end, T0=50.0, T1=0.8, K0=2, K1=12, max_it=300000):
    h_sg = h0.copy()
    h_pv = h0.copy()
    best_c = compute_c5(h0)
    best_h = h0.copy()
    rng = np.random.default_rng(12345)
    for it in range(max_it):
        if time.time() > t_end:
            break
        frac = min(1.0, (it + 1) / float(max_it))
        T = T0 * (T1 / T0) ** (frac ** 0.45)
        K = max(K0, int(K0 + (K1 - K0) * frac ** 2.0))
        beta = it / (it + 1.5)
        y = h_sg + beta * (h_sg - h_pv)
        y = np.clip(y, 0.0, 1.0)
        _, g = smooth_vg(y, n, K, T)
        alpha = 0.48 / (1.0 + 0.0008 * it)
        new = y - alpha * g
        if (it + 1) % 15000 == 0:
            new += rng.normal(0, 0.003, n)
        h_pv = h_sg
        h_sg = project_feasible(new, n)
        if (it + 1) % 8000 == 0:
            c = compute_c5(h_sg)
            if c < best_c:
                best_c = c
                best_h = h_sg.copy()
    c = compute_c5(h_sg)
    if c < best_c:
        best_c = c
        best_h = h_sg.copy()
    return best_h, best_c

def slsqp_polish(h0, n, K, T=1.2, maxit=5000):
    target = n / 2.0
    def obj(x):
        return smooth_vg(x, n, K, T)[0]
    def jac(x):
        return smooth_vg(x, n, K, T)[1]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac,
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxit, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h0.copy()

def binary_flip_search(h0, n, t_end):
    """Local search: try flipping individual values toward 0 or 1."""
    h = h0.copy()
    c = compute_c5(h)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        order = np.random.default_rng(7).permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            for target_val in [0.0, 1.0]:
                if abs(h[i] - target_val) < 0.01:
                    continue
                h[i] = target_val
                h = project_feasible(h, n)
                c2 = compute_c5(h)
                if c2 < c - 1e-10:
                    c = c2
                    improved = True
                    break
                else:
                    h[i] = h0[i]
                    h = project_feasible(h, n)
            h0 = h.copy()
    return h, c

def equalize_top(h0, n, K_top=3, t_end=time.time()+60):
    """Penalize variance of top-K overlaps to push them toward equality."""
    target = n / 2.0
    center = n - 1
    def top_overlap(x, k=K_top):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top = np.sort(corr)[::-1][:k]
        return top
    def obj(x):
        top = top_overlap(x)
        return np.max(top) + 0.1 * (np.var(top) ** 0.5)
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argsort(corr)[::-1][:K_top]
        tv = corr[top_idx]
        t = tv.max()
        var = np.var(tv)
        g = np.zeros(n)
        for j in range(K_top):
            dmax = 1.0 if tv[j] >= t - 1e-9 else 0.0
            dvar = (tv[j] - tv.mean()) / (K_top * var ** 0.5) if var > 1e-20 else 0.0
            w = dmax + 0.1 * dvar
            if w > 0:
                g += w * shift_grad(x, n, top_idx[j] - center)
        return g * (2.0 / n)
    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=jac,
                       bounds=[(0, 1)] * n, maxiter=2000,
                       options={'ftol': 1e-16})
        return res.x
    except Exception:
        return h0.copy()

def transfer_opt(h_cur, n_cur, n_targ, t_end):
    x_old = np.arange(n_cur) * (2.0 / n_cur)
    x_new = np.linspace(0, 2.0, n_targ, endpoint=False)
    h = np.interp(x_new, x_old, h_cur)
    h = project_feasible(h, n_targ)
    h, c = nesterov_loop(h, n_targ, min(t_end - 20, time.time() + 80),
                         T0=40, T1=1.0, K0=3, K1=10, max_it=100000)
    h = project_feasible(h, n_targ)
    if time.time() < t_end - 15:
        h_new = slsqp_polish(h, n_targ, 4, 1.5, 2000)
        h_new = project_feasible(h_new, n_targ)
        c2 = compute_c5(h_new)
        if c2 < c:
            h, c = h_new, c2
    return h, c
