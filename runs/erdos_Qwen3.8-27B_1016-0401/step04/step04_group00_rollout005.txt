def topk_smooth_obj_grad(h, n, T, K):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_vec(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            grad[:n - shift] += (1.0 - h[shift:])
        if shift > 0:
            grad[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] += (1.0 - h[:n - s])
        if s > 0:
            grad[:n - s] -= h[s:]
    return grad

def topk_obj_grad(h, n, T, K):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    nseg = len(corr)
    if K >= nseg:
        K = nseg
    top_idx = np.argpartition(corr, -K)[-K:]
    tv = corr[top_idx]
    tv_max = tv.max()
    sh = T * (tv - tv_max)
    e = np.exp(sh)
    sw = e / e.sum()
    val = np.dot(sw, tv) * dx
    grad = np.zeros(n)
    for i in range(K):
        if sw[i] > 1e-10:
            grad += sw[i] * shift_grad_vec(h, n, top_idx[i] - center)
    grad *= dx
    return val, grad

def nesterov_loop(h_init, n, t_start, t_end, K_s, K_e, T_s, T_e, max_iters, shake_period=20000):
    h_pos = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    rng = np.random.default_rng(7)
    ntot = max(1, max_iters)
    for it in range(ntot):
        if time.time() - t_start >= t_end:
            break
        frac = min((it + 1) / ntot, 1.0)
        T = T_s * (T_e / T_s) ** (frac ** 0.45)
        K = max(1, int(K_s + (K_e - K_s) * frac))
        mom = 0.92 - 0.2 * frac
        h_y = h_pos + mom * (h_pos - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        _, g = topk_obj_grad(h_y, n, T, K)
        alpha = 0.5 / (1.0 + 0.002 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_pos
        h_pos = h_new
        if shake_period > 0 and (it + 1) % shake_period == 0:
            h_pos = h_pos + 0.005 * rng.standard_normal(n)
            h_pos = project_feasible(h_pos, n)
        if (it + 1) % 4000 == 0:
            c = compute_c5(h_pos)
            if c < c5_best:
                c5_best = c
                h_best = h_pos.copy()
    c = compute_c5(h_pos)
    if c < c5_best:
        c5_best = c
        h_best = h_pos.copy()
    return h_best, c5_best

def slsqp_run(h_init, n, K, maxiter=2500, T_fix=3.0):
    target = n / 2.0
    try:
        res = minimize(_slsqp_obj, h_init, args=(n, T_fix, K), method='SLSQP',
                       jac=_slsqp_jac, bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h_init.copy()

def _slsqp_obj(x, n, T, K):
    v, _ = topk_obj_grad(x, n, T, K)
    return v

def _slsqp_jac(x, n, T, K):
    _, g = topk_obj_grad(x, n, T, K)
    return g

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def golden_bin_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n // 2):
        h[int((i * phi) % n)] = 1.0
    return project_feasible(h, n)

def three_peak_init(n):
    n_ones = n // 2
    h = np.zeros(n)
    g = n_ones // 3
    extra = n_ones - 3 * g
    blocks = []
    for k in range(3):
        sz = g + (1 if k < extra else 0)
        start = int(k * n / 3 + 0.5)
        h[start:start + sz] = 1.0
    return project_feasible(h, n)
