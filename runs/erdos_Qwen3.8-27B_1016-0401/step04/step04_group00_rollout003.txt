def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def get_topk_corr(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    idx = np.argpartition(corr, -K)[-K:]
    idx = idx[np.argsort(corr[idx])[::-1]]
    return corr, idx, dx

def grad_corr_shift(h, n, shift):
    """Gradient of ∫h(x)(1-h(x+shift))dx w.r.t. h"""
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def power_mean_grad(h, n, K, p):
    """Gradient of (sum of top-K corr^p)^(1/p) * dx"""
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argpartition(corr, -K)[-K:]
    vals = corr[idx]
    # (sum v^p)^(1/p) gradient: v^(p-1) / (sum v^p)^(1/p) * (1/p) ... 
    # Actually d/dv (sum v^p)^(1/p) = v^(p-1) * (sum v^p)^(1/p - 1)

    for i, ix in enumerate(idx):

    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    if s >= 0:
        m1 = j >= s
        m2 = (j + s) < n
        if m1.any():
            g[m1] += (1.0 - h[j[m1] - s])
        if m2.any():
            g[m2] -= h[j[m2] + s]
    else:
        ss = -s
        m1 = (j + ss) < n
        m2 = j >= ss
        if m1.any():
            g[m1] += (1.0 - h[j[m1] + ss])
        if m2.any():
            g[m2] -= h[j[m2] - ss]
    return g

def pm_val_grad(h, n, K, p):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argpartition(corr, -K)[-K:]
    vals = corr[idx]
    sp = np.sum(vals ** p)
    val = sp ** (1.0 / p) * dx
    pref = sp ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for i in range(K):
        w = vals[i] ** (p - 1) * pref
        if w > 1e-15:
            g += w * grad_shift(h, n, idx[i] - center)
    return val, g * dx

def nesterov_pm(h0, n, t0, t_end, K, p_seq, max_it):
    hs = h0.copy()
    hp = h0.copy()
    cb = compute_c5(h0)
    hb = h0.copy()
    np_ = len(p_seq)
    for it in range(max_it):
        if time.time() - t0 >= t_end:
            break
        fr = min((it + 1) / max(1, max_it), 1.0)
        pi = min(int(fr * np_), np_ - 1)
        p = p_seq[pi]
        b = float(it) / (it + 1.5) if it > 0 else 0.0
        hy = np.clip(hs + b * (hs - hp), 0.0, 1.0)
        _, g = pm_val_grad(hy, n, K, p)
        gn = np.linalg.norm(g)
        alpha = 1.2 / (1.0 + 0.003 * it) if gn > 1e-8 else 0.5
        hn = project_feasible(hy - alpha * g, n)
        hp = hs
        hs = hn
        if (it + 1) % 4000 == 0:
            c = compute_c5(hs)
            if c < cb:
                cb = c
                hb = hs.copy()
    c = compute_c5(hs)
    if c < cb:
        cb = c
        hb = hs.copy()
    return hb, cb

def slsqp_topk(h0, n, K, mx=3500):
    dx = 2.0 / n
    tgt = n / 2.0
    center = n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, -K)[-K:]
        return np.sum(corr[idx]) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, -K)[-K:]
        g = np.zeros(n)
        for ix in idx:
            g += grad_shift(x, n, ix - center)
        return g * dx

    try:
        r = minimize(obj, h0, method='SLSQP', jac=jac,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': mx, 'ftol': 1e-18})
        return r.x
    except Exception:
        return h0.copy()

def lbfgs_topk(h0, n, K, mx=4000):
    dx = 2.0 / n
    tgt = n / 2.0
    center = n - 1

    def obj_p(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, -K)[-K:]
        v = x.sum() - tgt
        return np.sum(corr[idx]) * dx + 1e4 * v * v

    def jac_p(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, -K)[-K:]
        g = np.zeros(n)
        for ix in idx:
            g += grad_shift(x, n, ix - center)
        v = x.sum() - tgt
        return g * dx + 2e4 * v * np.ones(n)

    try:
        r = minimize(obj_p, h0, method='L-BFGS-B', jac=jac_p,
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mx, 'gtol': 1e-16})
        return r.x
    except Exception:
        return h0.copy()

def golden_init(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n // 2):
        h[int((i * phi) % n)] = 1.0
    return project_feasible(h, n)

def perturb_search(hb, n, t0, t_end, seed):
    rng = np.random.default_rng(seed)
    cb = compute_c5(hb)
    for tr in range(4):
        if time.time() - t0 >= t_end:
            break
        hp = hb.copy()
        ns = n // 10
        ones = np.where(hp > 0.3)[0]
        zeros = np.where(hp < 0.7)[0]
        if len(ones) < 10 or len(zeros) < 10:
            continue
        so = rng.choice(ones, size=min(ns, len(ones)), replace=False)
        sz = rng.choice(zeros, size=min(ns, len(zeros)), replace=False)
        hp[so] = 0.0
        hp[sz] = 1.0
        hp = project_feasible(hp, n)
        hn, cn = nesterov_pm(hp, n, t0, min(t_end - 20, time.time() + 100), 5, [6, 3, 2], 80000)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < cb:
            cb = c
            hb = hn
    return hb, cb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        hb = np.array(initial_h_values, dtype=float)
    else:
        hb = golden_init(280)
    n = len(hb)
    hb = project_feasible(hb, n)
    cb = compute_c5(hb)

    # Phase 1: Nesterov power-mean (K=5, p: 10->2)
    t1 = t0 + (budget_s - 120) * 0.30
    hb, cb = nesterov_pm(hb, n, t0, t1, 5, [10, 7, 5, 4, 3, 2], 180000)
    hb = project_feasible(hb, n)

    # Phase 2: SLSQP K=5
    t2 = t0 + (budget_s - 120) * 0.48
    if time.time() - t0 < t2:
        hn = slsqp_topk(hb, n, 5, 4000)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < cb:
            cb, hb = c, hn

    # Phase 3: L-BFGS-B K=5
    t3 = t0 + (budget_s - 120) * 0.62
    if time.time() - t0 < t3:
        hn = lbfgs_topk(hb, n, 5, 5000)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < cb:
            cb, hb = c, hn

    # Phase 4: SLSQP with smaller K (focus on worst 3-4)
    t4 = t0 + (budget_s - 120) * 0.78
    for K in [3, 4, 6]:
        if time.time() - t0 >= t4:
            break
        hn = slsqp_topk(hb, n, K, 3500)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < cb:
            cb, hb = c, hn

    # Phase 5: Perturbation search
    t5 = t0 + (budget_s - 120) * 0.90
    if time.time() - t0 < t5:
        hb, cb = perturb_search(hb, n, t0, t5, seed + 7)

    # Phase 6: Final tight SLSQP K=3
    if time.time() - t0 < budget_s - 30:
        hn = slsqp_topk(hb, n, 3, 4000)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < cb:
            cb, hb = c, hn

    # Phase 7: Mini Nesterov burst
    if time.time() - t0 < budget_s - 10:
        hn, cn = nesterov_pm(hb, n, t0, budget_s - 2, 4, [4, 2], 80000)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < cb:
            cb, hb = c, hn

    hb = project_feasible(hb, n)
    cfin = compute_c5(hb)
    return (hb, min(cfin, cb), n)
