    return val, grad * dx

def nesterov_phase(h_init, n, t0, t_end, max_iters=1000000):
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    T_s, T_e = 120.0, 0.5
    K_s, K_e = 3, 10
    for it in range(max_iters):
        if time.time() - t0 >= t_end:
            break
        frac = (it + 1) / max(1, max_iters)
        T = T_s * (T_e / T_s) ** (frac ** 0.35)
        K = max(K_s, int(K_s + (K_e - K_s) * frac ** 2.0))
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        _, g = topk_vg(h_y, n, T, K)
        alpha = 0.45 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 4000 == 0:
            c = compute_c5(h_sg)
            if c < c5_best:
                c5_best = c
                h_best = h_sg.copy()
    c = compute_c5(h_sg)
    if c < c5_best:
        c5_best = c
        h_best = h_sg.copy()
    return h_best, c5_best

def slsqp_refine(h_init, n, K, t_end, maxiter=5000):
    target = n / 2.0

    def obj(x):
        v, _ = topk_vg(x, n, 3.0, K)
        return v

    def jac(x):
        _, g = topk_vg(x, n, 3.0, K)
        return g

    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() < t_end:
            return res.x
    except Exception:
        pass
    return h_init.copy()

def binary_cd(h, n, t0, t_end, max_rounds=3000):
    target = n / 2.0
    h = project_feasible(h, n)
    c5_best = compute_c5(h)
    center = n - 1
    rng = np.random.default_rng(12345)
    for rnd in range(max_rounds):
        if time.time() - t0 >= t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        K = 6
        top_idx = np.argpartition(corr, -K)[-K:]
        top_corr = corr[top_idx]
        top_max = top_corr.max()
        best_gain = 1e-12
        best_pair = None
        ones = np.where(h > 0.99)[0]
        zeros = np.where(h < 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            frac = np.abs(h - 0.5).argmin()
            if h[frac] > 0.5:
                ones = np.array([frac])
            else:
                zeros = np.array([frac])
        for ii in range(min(len(ones), 60)):
            i = ones[ii]
            for jj in range(min(len(zeros), 60)):
                j = zeros[jj]
                if i == j:
                    continue
                dc_i = np.zeros(K)
                dc_j = np.zeros(K)
                for m in range(K):
                    k = int(top_idx[m]) - center
                    if 0 <= i - k < n:
                        dc_i[m] += -1.0 * (1.0 - h[i - k])
                    if 0 <= i + k < n:
                        dc_i[m] -= -1.0 * h[i + k]
                    if 0 <= j - k < n:
                        dc_j[m] += 1.0 * (1.0 - h[j - k])
                    if 0 <= j + k < n:
                        dc_j[m] -= 1.0 * h[j + k]
                new_top = top_corr + dc_i + dc_j
                new_max = new_top.max()
                gain = top_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        if best_pair is None:
            idx = rng.integers(n)
            if h[idx] > 0.5:
                continue
            h[best_pair[0]] = 0.0 if best_pair else h[best_pair[0]]
            h[best_pair[1]] = 1.0 if best_pair else h[best_pair[1]]
        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
        c = compute_c5(h)
        if c < c5_best:
            c5_best = c
    return h, c5_best

def lbfgs_polish(h_init, n, T, K, maxiter=3000):
    target = n / 2.0

    def obj_pen(x):
        v, _ = topk_vg(x, n, T, K)
        viol = x.sum() - target
        return v + 1e4 * viol * viol

    def jac_pen(x):
        _, g = topk_vg(x, n, T, K)
        viol = x.sum() - target
        return g + 2e4 * viol * np.ones(n)

    try:
        res = minimize(obj_pen, h_init, method='L-BFGS-B', jac=jac_pen,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-16})
        return res.x
    except Exception:
        return h_init.copy()

def golden_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n // 2):
        idx = int((i * phi) % n)
        h[idx] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = golden_init(280)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Main Nesterov with slow cooling (55% time)
    t1 = t0 + (budget_s - 60) * 0.55
    h_best, c5_best = nesterov_phase(h_best, n, t0, t1, max_iters=900000)
    h_best = project_feasible(h_best, n)

    # Phase 2: SLSQP with small K (20% time, split among K values)
    t2 = t0 + (budget_s - 60) * 0.75
    for K in [3, 4, 5]:
        if time.time() - t0 >= t2:
            break
        rem = max(1, t2 - time.time())
        h_new = slsqp_refine(h_best, n, K, t2, maxiter=int(200 * rem) + 2000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 3: Binary coordinate descent (10% time)
    t3 = t0 + (budget_s - 60) * 0.85
    if time.time() - t0 < t3:
        h_b, c_b = binary_cd(h_best, n, t0, t3, max_rounds=2000)
        if c_b < c5_best:
            c5_best = c_b
            h_best = h_b

    # Phase 4: Final SLSQP polish at K=3,4 (8% time)
    t4 = t0 + (budget_s - 60) * 0.93
    for K in [3, 4]:
        if time.time() - t0 >= t4:
            break
        h_new = slsqp_refine(h_best, n, K, t4, maxiter=4000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 5: L-BFGS polish + final Nesterov burst
    if time.time() - t0 < budget_s - 15:
        h_new = lbfgs_polish(h_best, n, 1.5, 3, 2000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new
        if time.time() - t0 < budget_s - 5:
            h_new, c_new = nesterov_phase(h_best, n, t0, budget_s - 3, max_iters=100000)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new)
            if c < c5_best:
                c5_best = c
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
