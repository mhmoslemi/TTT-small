        h_try = h0 + rng.normal(0, eps, n)
        h_try = np.clip(h_try, 0, 1)
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < c5_best:
            c5_best = c_try
            h_best = h_try.copy()
        h_new, c_new = nesterov_run(h_try, n, t0, min(te - 1, time.time() + slab * 0.8),
                                     max_iters=int(60000 * slab / 100))
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        h_best = np.zeros(n)
        for i in range(n // 2):
            idx = int((i * phi) % n)
            h_best[idx] = 1.0
        h_best = project_feasible(h_best, n)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Nesterov from initial (25% budget)
    t_p1 = t0 + (budget_s - 30) * 0.22
    h_best, c5_best = nesterov_run(h_best, n, t0, t_p1, max_iters=200000)
    h_best = project_feasible(h_best, n)

    # Phase 2: Multi-restart with perturbations (35% budget)
    t_p2 = t0 + (budget_s - 30) * 0.57
    h_best, c5_best = multi_restart(h_best, n, t0, t_p2, seed=seed + 7, n_starts=5)
    h_best = project_feasible(h_best, n)

    # Phase 3: L-BFGS refinement at multiple temperatures (15%)
    t_p3 = t0 + (budget_s - 30) * 0.72
    for T in [8.0, 4.0, 2.0]:
        if time.time() - t0 >= t_p3:
            break
        h_new = lbfgs_refine(h_best, n, T, 5, maxiter=1500)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 4: SLSQP polish with tight K (15%)
    t_p4 = t0 + (budget_s - 30) * 0.87
    for K in [3, 4, 5, 7]:
        if time.time() - t0 >= t_p4:
            break
        h_new = slsqp_polish(h_best, n, K, 3000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 5: Coordinate swap local search
    if time.time() - t0 < t_p4 + 20:
        h_new, c_new = coord_swap(h_best, n, n_rounds=40, seed=seed)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new

    # Phase 6: Final Nesterov burst + SLSQP
    if time.time() - t0 < budget_s - 25:
        h_new, c_new = nesterov_run(h_best, n, t0, budget_s - 18, max_iters=40000)
        h_new = project_feasible(h_new, n)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new

    if time.time() - t0 < budget_s - 8:
        h_new = slsqp_polish(h_best, n, 3, 1500)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
