import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_feas(h, n):
    target = n / 2.0
    lo, hi = -5.0, 6.0
    for _ in range(75):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - mid, 0.0, 1.0)

def shift_g(h, sh, n):
    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        m1 = j >= sh
        if m1.any():
            g[m1] += 1.0 - h[j[m1] - sh]
        m2 = (j + sh) < n
        if m2.any():
            g[m2] -= h[j[m2] + sh]
    else:
        s = -sh
        m1 = (j + s) < n
        if m1.any():
            g[m1] += 1.0 - h[j[m1] + s]
        m2 = j >= s
        if m2.any():
            g[m2] -= h[j[m2] - s]
    return g

def obj_grad(h, T, K, n):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argpartition(corr, -K)[-K:]
    vals = corr[top]
    mx = vals.max()
    e = np.exp(T * (vals - mx))
    w = e / e.sum()
    val = (mx + np.log(e.sum()) / T) * dx
    g = np.zeros(n)
    for i in range(K):
        if w[i] > 1e-14:
            g += w[i] * shift_g(h, int(top[i]) - center, n)
    return val, g * dx

def adam_run(h0, n, t0, t_end, T0=10.0, T1=1.8, K0=2, K1=7, lr=0.02):
    h = h0.copy().astype(float)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    tc = 0
    b1, b2, eps = 0.82, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    while time.time() - t0 < t_end:
        frac = min(tc / 80000.0, 1.0)
        T = T0 * (T1 / T0) ** (frac ** 0.7)
        K = max(2, int(K0 + (K1 - K0) * frac * frac))
        _, g = obj_grad(h, T, K, n)
        tc += 1
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        mhat = m_v / (1 - b1 ** tc)
        vhat = v_v / (1 - b2 ** tc)
        step = lr / (1.0 + 0.003 * tc * 0.01)
        h = h - step * mhat / (np.sqrt(vhat) + eps)
        h = proj_feas(h, n)
        if tc % 8000 == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def slsqp_run(h0, n, T, K, maxiter=3000):
    target = n / 2.0
    def fob(x):
        v, _ = obj_grad(x, T, K, n)
        return v
    def fjac(x):
        _, g = obj_grad(x, T, K, n)
        return g
    try:
        res = minimize(fob, h0, method='SLSQP', jac=fjac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h0.copy()

def fw_step(h, n, t_coeff):
    g = np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    km = int(np.argmax(corr))
    g = shift_g(h, km - center, n)
    v = np.zeros(n)
    v[np.argsort(g)[:n // 2]] = 1.0
    return (1.0 - t_coeff) * h + t_coeff * v

def multi_perturb(h0, n, t0, t_end, seed, n_perturb=3):
    rng = np.random.default_rng(seed)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    time_per = (t_end - t0) / (n_perturb + 1)
    for p in range(n_perturb):
        t_start = t0 + p * time_per
        t_stop = t_start + time_per * 0.6
        if time.time() - t0 >= t_end - 30:
            break
        noise_scale = 0.03 + 0.02 * p
        h_p = h0 + noise_scale * rng.standard_normal(n)
        h_p = proj_feas(h_p, n)
        h_p, c_p = adam_run(h_p, n, t0, t_stop, T0=8.0, T1=2.0, K0=2, K1=5, lr=0.018)
        if c_p < best_c5:
            best_c5 = c_p
            best_h = h_p.copy()
            h0 = h_p
    return best_h, best_c5

def transfer_res(h_cur, n_cur, n_targ, t0, t_end):
    x_old = np.arange(n_cur) * (2.0 / n_cur)
    x_new = np.linspace(0, 2.0, n_targ, endpoint=False)
    h = np.interp(x_new, x_old, h_cur)
    h = proj_feas(h, n_targ)
    c5 = compute_c5(h)
    if time.time() - t0 < t_end - 40:
        h, c5 = adam_run(h, n_targ, t0, min(t_end - 20, time.time() + 80),
                         T0=8.0, T1=2.0, K0=2, K1=5, lr=0.015)
    if time.time() - t0 < t_end - 10:
        h_new = slsqp_run(h, n_targ, 3.0, 4, 2000)
        h_new = proj_feas(h_new, n_targ)
        cn = compute_c5(h_new)
        if cn < c5:
            h, c5 = h_new, cn
    return h, c5, n_targ

def golden_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n // 2):
        idx = int((i * phi) % n)
        h[idx] = 1.0
    return proj_feas(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = golden_init(280)
    n = len(h_best)
    h_best = proj_feas(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Main ADAM optimization
    t_p1_end = t0 + (budget_s - 100) * 0.45
    h_best, c5_best = adam_run(h_best, n, t0, t_p1_end, T0=12.0, T1=2.0, K0=2, K1=6, lr=0.022)
    h_best = proj_feas(h_best, n)
    c5_best = min(compute_c5(h_best), c5_best)

    # Phase 2: SLSQP refinements at different smoothing levels
    t_p2_end = t0 + (budget_s - 100) * 0.62
    for T in [5.0, 3.0, 2.0]:
        if time.time() - t0 >= t_p2_end:
            break
        h_new = slsqp_run(h_best, n, T, 4, 3000)
        h_new = proj_feas(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 3: FW diversification + re-optimize
    t_p3_end = t0 + (budget_s - 100) * 0.75
    if time.time() - t0 < t_p3_end:
        h_fw = h_best.copy()
        for it in range(200):
            tc = 2.0 / (it + 2)
            h_fw = fw_step(h_fw, n, tc)
            h_fw = proj_feas(h_fw, n)
            if (it + 1) % 50 == 0:
                c = compute_c5(h_fw)
                if c < c5_best:
                    c5_best = c
                    h_best = h_fw.copy()
        h_ad, c_ad = adam_run(h_best, n, t0, t_p3_end, T0=6.0, T1=2.0, K0=2, K1=4, lr=0.015)
        if c_ad < c5_best:
            c5_best = c_ad
            h_best = h_ad

    # Phase 4: Perturbation-based multi-start
    t_p4_end = t0 + (budget_s - 60) * 0.88
    if time.time() - t0 < t_p4_end - 50:
        h_best, c5_best = multi_perturb(h_best, n, t0, t_p4_end, seed, n_perturb=3)
        h_best = proj_feas(h_best, n)
        c5_best = min(compute_c5(h_best), c5_best)

    # Phase 5: SLSQP final polish
    t_p5_end = t0 + (budget_s - 40) * 0.95
    for K in [3, 4]:
        if time.time() - t0 >= t_p5_end:
            break
        h_new = slsqp_run(h_best, n, 2.0, K, 2500)
        h_new = proj_feas(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 6: Try different resolution
    t_p6_end = budget_s - 30
    if time.time() - t0 < t_p6_end - 200:
        for n2 in [n - 40, n + 40]:
            if n2 < 100 or n2 > 500 or time.time() - t0 >= t_p6_end - 100:
                continue
            h2, c52, n2 = transfer_res(h_best, n, n2, t0, t_p6_end - 30)
            if c52 < c5_best:
                c5_best = c52
                h_best = h2
                n = n2

        h_new, c_new = adam_run(h_best, n, t0, budget_s - 5, T0=6.0, T1=1.5, K0=2, K1=3, lr=0.01)
        h_new = proj_feas(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            h_best, c5_best = h_new, c

    h_best = proj_feas(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
