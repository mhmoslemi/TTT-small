    # Ensure f(lam_lo) > 0 and f(lam_hi) < 0
    while f(lam_lo) <= 0:
        lam_lo -= 1
    while f(lam_hi) >= 0:
        lam_hi += 1
    lam = brentq(f, lam_lo, lam_hi, xtol=1e-14)
    return np.clip(h - lam, 0, 1)
