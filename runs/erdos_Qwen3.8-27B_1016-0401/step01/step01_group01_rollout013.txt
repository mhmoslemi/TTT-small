```python
import numpy as np
from scipy.optimize import minimize
import time

_CUR_DX = 1.0
_CUR_T_SMOOTH = 0.1
_CUR_N = 70

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _smooth_obj(x):
    dx = _CUR_DX
    t = _CUR_T_SMOOTH
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(c)
    return dx * (c_max + t * np.log(np.sum(np.exp((c - c_max) / t))))

def _smooth_jac(x):
    n = _CUR_N
    dx = _CUR_DX
    t = _CUR_T_SMOOTH
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(c)
    w = np.exp((c - c_max) / t)
    w /= np.sum(w)
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += w[m] * (1.0 - x[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += -w[m] * x[ks[v2] + s]
    return dx * grad

def _cons_fun(x):
    return float(np.sum(x) - _CUR_N / 2.0)

def _cons_jac(x):
    return np.ones(_CUR_N)

def _normalize(h, T):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 1e-10:
        h = h * T / s
    return np.clip(h, 0, 1)

def _optimize_annealed(h_init, n, t_schedule, max_iter_per_phase, t_start=0.0):
    global _CUR_DX, _CUR_N, _CUR_T_SMOOTH
    _CUR_DX = 2.0 / n
    _CUR_N = n
    T = n / 2.0
    dx = 2.0 / n
    h = _normalize(h_init, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for t in t_schedule:
        if t_start > 0 and time.time() - t_start > 1050:
            break
        _CUR_T_SMOOTH = t
        try:
            r = minimize(_smooth_obj, h, method='SLSQP', jac=_smooth_jac,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': max_iter_per_phase, 'ftol': 1e-12})
            h_new = _normalize(r.x, T)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def _lbfgs_polish(h_init, n, t, max_iter):
    global _CUR_DX, _CUR_N, _CUR_T_SMOOTH
    _CUR_DX = 2.0 / n
    _CUR_N = n
    _CUR_T_SMOOTH = t
    T = n / 2.0
    dx = 2.0 / n
    h = _normalize(h_init, T)
    bounds = [(0.0, 1.0)] * n
    
    def _pen_obj(x):
        return _smooth_obj(x) + 1e6 * _cons_fun(x) ** 2
    
    def _pen_jac(x):
        return _smooth_jac(x) + 2e6 * _cons_fun(x) * np.ones(n)
    
    try:
        r = minimize(_pen_obj, h, method='L-BFGS-B', jac=_pen_jac,
                    bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-10})
        h_new = _normalize(r.x, T)
        c5_new = compute_c5(h_new, dx)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h, dx)

def _structured_perturbations(h, rng):
    n = len(h)
    T = n / 2.0
    candidates = []
    candidates.append(1.0 - h)
    candidates.append(h[::-1])
    for k in [1, 2, 3, 5, 7, 10]:
        if k < n:
            candidates.append(np.roll(h, k))
    for _ in range(4):
        blk_size = rng.randint(5, n // 2)
        start = rng.randint(0, n - blk_size)
        h_p = h.copy()
        shift = rng.randint(1, n - blk_size)
        h_p[start:start+blk_size] = 0
        end = (start + shift) % n
        h_p[end:end+blk_size] = np.clip(h_p[end:end+blk_size] * 0.3, 0, 1)
        candidates.append(_normalize(h_p, T))
    for _ in range(3):
        noise = rng.normal(0, 0.15, n)
        candidates.append(_normalize(h + noise, T))
    return candidates

def _subgradient_polish(h, n, num_steps, lr_start):
    global _CUR_DX, _CUR_N
    _CUR_DX = 2.0 / n
    _CUR_N = n
    T = n / 2.0
    dx = 2.0 / n
    h = _normalize(h, T)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for step in range(num_steps):
        lr = lr_start * (0.95 ** step)
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(c))
        s = idx - (n - 1)
        grad = np.zeros(n)
        ks = np.arange(n)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += -h[ks[v2] + s]
        grad *= dx
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        for _ in range(50):
            s_sum = h_new.sum()
            if abs(s_sum - T) < 1e-12:
                break
            if s_sum > 0:
                h_new = np.clip(h_new * (T / s_sum), 0, 1)
            else:
                h_new = np.full(n, T / n)
                break
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
```
