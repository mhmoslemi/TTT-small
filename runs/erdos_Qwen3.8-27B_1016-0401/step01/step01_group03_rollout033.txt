    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]

            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

            h = best_h + 0.1 * rng.standard_normal(n)
            h = project_feasible(h, ts)
    
    return best_h, best_c5

def simulated_annealing(h, dx, t_limit, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    target_ones = n // 2
    b = round_to_binary(h, target_ones).copy()
    best_b = b.copy()
    best_c5 = compute_c5(b, dx)
    cur_c5 = best_c5
    
    T_init = 0.5
    T = T_init
    cooling = 0.9999
    iteration = 0
    best_iter = 0
    
    while time.time() < t_limit:
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose move
        r = rng.random()
        if r < 0.12 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            do_move = True
            moves = [(i1, 0), (i2, 0), (j1, 1), (j2, 1)]
        elif r < 0.35:
            i = ones[rng.randint(len(ones))]
            d = 1 + rng.randint(0, max(1, n // 5))
            j = i + d if rng.random() < 0.5 else i - d
            if 0 <= j < n and b[j] == 0:
                do_move = True
                moves = [(i, 0), (j, 1)]
            else:
                j = zeros[rng.randint(len(zeros))]
                do_move = True
                moves = [(i, 0), (j, 1)]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            do_move = True
            moves = [(i, 0), (j, 1)]
        
        for idx, val in moves:
            b[idx] = val
        new_c5 = compute_c5(b, dx)
        delta = new_c5 - cur_c5
        
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-max(delta, 0) / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_b = b.copy()
                best_iter = iteration
                T = min(T * 1.01, T_init)
        else:
            for idx, val in moves:
                b[idx] = 1 - val
        
        iteration += 1
        T *= cooling
        
        if iteration - best_iter > 30000:
            if T > 0.1:
                T = 0.3
            elif iteration - best_iter > 80000:
                b = best_b.copy()
                cur_c5 = best_c5
                best_iter = iteration
                T = T_init
    
    return best_b, best_c5
