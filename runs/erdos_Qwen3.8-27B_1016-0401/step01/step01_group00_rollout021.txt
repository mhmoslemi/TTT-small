import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_extreme(h):
    """Return (c5_value, argmax_index_in_full_corr)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx

def grad_at_shift(h, k_idx):
    """Compute gradient of f_k where k_idx is the index in full correlation output."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    k = k_idx - center  # actual shift
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif k > 0:
        # h[i]*(1-h[i+k]) for i = 0..n-1-k
        grad[:n-k] += (1.0 - h[k:]) * dx
        grad[k:] -= h[:n-k] * dx
    else:
        p = -k
        grad[p:] += (1.0 - h[:n-p]) * dx
        grad[:n-p] -= h[p:] * dx
    return grad

def project(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target}"""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            # Need to add diff total
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-15:
                    break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff >= -1e-15:
                    break
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr * dx, idx

def grad_at_shift(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    k = k_idx - center
    grad = np.zeros(n)
    if k == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif k > 0:
        grad[:n-k] += (1.0 - h[k:]) * dx
        grad[k:] -= h[:n-k] * dx
    else:
        p = -k
        grad[p:] += (1.0 - h[:n-p]) * dx
        grad[:n-p] -= h[p:] * dx
    return grad

def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            idx = idx[np.argsort(h[idx])]
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            idx = idx[np.argsort(-h[idx])]
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break
    return h

def soft_max_gradient(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shift_min = shifted.min()
    e = np.exp(shifted - shift_min)
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-9:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n-o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n-o]
        else:
            p = -o
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):

    n = len(h_best)

    c5_best = compute_c5(h_best)

    # Phase 1: Projected subgradient on exact max (40% budget)
    p1_end = budget_s * 0.40
    step = 0.8
    h_cur = h_best.copy()
    no_improve = 0
    for it in range(5000000):
        if elapsed() > p1_end:
            break
        _, k_idx = compute_corr_max(h_cur)
        grad = grad_at_shift(h_cur, k_idx)

        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:

        else:
