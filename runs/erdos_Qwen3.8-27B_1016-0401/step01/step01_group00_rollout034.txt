def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def soft_max_val(corr, T):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T

def project_onto_simplex(h, target):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target}"""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            # Need to add: add to smallest values first
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-15:
                    break
        else:
            # Need to subtract: subtract from largest values first
            order = np.argsort(h)[::-1]
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff > -1e-15:
                    break
    return h

def lse_gradient(h, T, n):
    """Gradient of soft_max_val(compute_corr(h), T) * (2/n) with respect to h"""
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-8:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx

    delta = np.zeros(2*n-1)
    center = n-1

    # As h-part: -(1-h[i-k]) for k where 0<=i-k<n → k in [i-(n-1), i]
    #   Index in delta: center + k, so indices [center+i-(n-1), center+i] = [i, center+i]

    k_hi = min(n-1, i)

        delta[idx] -= (1 - h[i - np.arange(k_lo, k_hi+1)])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def lse_obj(corr, T):
    mc = np.max(corr)
    s = T * (corr - mc)
    return mc + np.log(np.sum(np.exp(s))) / T

def lse_grad(h, T):
    n = len(h)
    corr = compute_corr(h)
    mc = np.max(corr)
    s = T * (corr - mc)
    e = np.exp(s)
    w = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o]
        if wi < 1e-9:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] -= wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx

def project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff > -1e-14:
                    break
    return h

def projected_gradient_descent(h_init, target, T, steps=500, lr=None):
    n = len(h_init)
    h = h_init.copy()
    dx = 2.0 / n
    if lr is None:
        lr = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(steps):
        grad = lse_grad(h, T)
        h_new = h - lr * grad
        h_new = project(h_new, target)
        # Line search (simple)
        obj_new = lse_obj(compute_corr(h_new), T)
        obj_old = lse_obj(compute_corr(h), T)
        if obj_new > obj_old:
            # Halve step
            for _ in range(10):
                lr /= 2
                h_new = project(h - lr * grad, target)
                obj_new = lse_obj(compute_corr(h_new), T)
                if obj_new <= obj_old:
                    break
        h = h_new
        lr_prev = lr
        lr = min(lr * 1.05, 0.5)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_local_search(h, max_passes=200):
    n = len(h)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix to have exactly n/2 ones
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        # Remove excess ones (those with lowest value in original h)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[order[:num_ones-target]]] = 0
    elif num_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:target-num_ones]]] = 1
    
    corr = compute_corr(h_bin)
    current_max = float(np.max(corr))
    
    for _ in range(max_passes):
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        best_gain = 0
        best_i, best_j = -1, -1
        best_new_max = current_max
        
        for i in ones:
            for j in zeros:
                # Compute delta corr
                # A_i[k] = -(1-h[i-k])*[0<=i-k<n] + h[i+k]*[0<=i+k<n]
                # B_j[k] = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]
                delta = np.zeros(2*n-1)
                
                # A_i part
                k_lo = max(0, i-(n-1))
                k_hi = min(n-1, i)
                if k_lo <= k_hi:
                    ks = np.arange(k_lo, k_hi+1)
                    delta[n-1+ks] -= (1 - h_bin[i-ks])
                k_lo2 = max(-(n-1), -i)
                k_hi2 = min(n-1, n-1-i)
                if k_lo2 <= k_hi2:
                    ks2 = np.arange(k_lo2, k_hi2+1)
                    delta[n-1+ks2] += h_bin[i+ks2]
                
                # B_j part
                k_lo = max(0, j-(n-1))

                if k_lo <= k_hi:
                    ks = np.arange(k_lo, k_hi+1)
                    delta[n-1+ks] += (1 - h_bin[j-ks])
                k_lo2 = max(-(n-1), -j)
                k_hi2 = min(n-1, n-1-j)
                if k_lo2 <= k_hi2:
                    ks2 = np.arange(k_lo2, k_hi2+1)
                    delta[n-1+ks2] -= h_bin[j+ks2]
                
                new_max = np.max(corr + delta)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_gain = current_max - new_max
                    best_i, best_j = i, j
                    if best_gain > 0.5:  # Early termination for large gains
                        break
            if best_gain > 0.5:
                break
        
        if best_i < 0:
            break
        
        h_bin[best_i] = 0
        h_bin[best_j] = 1
        corr = compute_corr(h_bin)
        current_max = float(np.max(corr))
    
    return h_bin, current_max * (2.0/n)

    return grad * dx

def _obj(x):
    T, tgt, pen = _params['T'], _params['target'], _params['pen']
    return lse_val(compute_corr(x), T) + pen * (x.sum() - tgt)**2

def _grad(x):
    T, tgt, pen = _params['T'], _params['target'], _params['pen']
    return lse_grad(x, T) + 2.0*pen*(x.sum()-tgt)*np.ones(len(x))

def project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if diff < 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                s_ = min(-diff, h[idx])
                h[idx] -= s_
                diff += s_
                if diff > -1e-14:
                    break
    return h

def _run_lbfgsb(h0, n, target, T, pen, maxiter, gtol):
    _params['T'] = T
    _params['target'] = target
    _params['pen'] = pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0.copy(), method='L-BFGS-B', jac=_grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': maxiter*5})
        return project(res.x, target)
    except Exception:
        return None

def _binary_swap_eval(h, corr, i, j, n):
    delta = np.zeros(2*n - 1)
    c = n - 1
    kl = max(0, i - n + 1); kh = min(n-1, i)
    if kl <= kh:
        ks = np.arange(kl, kh+1)
        delta[c+ks] -= (1.0 - h[i-ks])
    kl2 = max(-(n-1), -i); kh2 = min(n-1, n-1-i)
    if kl2 <= kh2:
        ks2 = np.arange(kl2, kh2+1)
        delta[c+ks2] += h[i+ks2]
    kl = max(0, j - n + 1); kh = min(n-1, j)
    if kl <= kh:
        ks = np.arange(kl, kh+1)
        delta[c+ks] += (1.0 - h[j-ks])
    kl2 = max(-(n-1), -j); kh2 = min(n-1, n-1-j)
    if kl2 <= kh2:
        ks2 = np.arange(kl2, kh2+1)
        delta[c+ks2] -= h[j+ks2]
    return float(np.max(corr + delta))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project(h_best, target)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Multi-temperature L-BFGS-B annealing
    for T in [3, 8, 20, 50, 120, 300]:
        if elapsed() > budget_s - 400:
            break
        for pen in [100, 1000]:
            if elapsed() > budget_s - 400:
                break
            h_new = _run_lbfgsb(h_best, n, target, T, pen, 600, 1e-11)
            if h_new is not None:
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Phase 2: Binary local search with random restarts
    if elapsed() < budget_s - 250:
        h_bin = (h_best > 0.5).astype(float)
        cnt = int(h_bin.sum())
        tgt_int = int(target)
        if cnt > tgt_int:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h_best[ones])
            h_bin[ones[order[:cnt - tgt_int]]] = 0
        elif cnt < tgt_int:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h_best[zeros])
            h_bin[zeros[order[:tgt_int - cnt]]] = 1

        corr = compute_corr(h_bin)
        cur_max = float(np.max(corr))
        n_passes = 60
        for _ in range(n_passes):
            if elapsed() > budget_s - 250:
                break
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            n_try = min(2500, len(ones) * len(zeros))
            ii = rng.integers(0, len(ones), size=n_try)
            jj = rng.integers(0, len(zeros), size=n_try)
            best_gain = 0.0
            best_i = best_j = -1
            for t in range(n_try):
                i = int(ones[ii[t]])
                j = int(zeros[jj[t]])
                nm = _binary_swap_eval(h_bin, corr, i, j, n)
                gain = cur_max - nm
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
            if best_i < 0:
                break
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            corr = compute_corr(h_bin)
            cur_max = float(np.max(corr))
        c5_bin = cur_max * (2.0 / n)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin.copy()

    # Phase 3: Continuous refinement
    if elapsed() < budget_s - 150:
        for T in [5, 15, 40, 100]:
            if elapsed() > budget_s - 150:
                break
            for pen in [500, 5000]:
                if elapsed() > budget_s - 150:
                    break
                x0 = h_best + rng.normal(0, 0.02, n)
                h_new = _run_lbfgsb(x0, n, target, T, pen, 800, 1e-12)
                if h_new is not None:
                    c5 = compute_c5(h_new)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = h_new

    # Phase 4: Higher resolution
    if elapsed() < budget_s - 80:
        for n_hi in [n + 70, n + 140]:
            if n_hi > 700 or elapsed() > budget_s - 40:
                continue
            tgt_hi = n_hi / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project(h_hi, tgt_hi)
            for T in [30, 80]:
                if elapsed() > budget_s - 20:
                    break
                h_new = _run_lbfgsb(h_hi, n_hi, tgt_hi, T, 2000, 400, 1e-10)
                if h_new is not None:
                    c5 = compute_c5(h_new)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = h_new
                        n = n_hi
                        h_hi = h_new.copy()
            n = len(h_best)
            target = n / 2.0

    # Phase 5: Final fine-tuning
    if elapsed() < budget_s - 10:
        h_best = project(h_best, len(h_best) / 2.0)
        h_new = _run_lbfgsb(h_best, len(h_best), len(h_best)/2.0, 20, 10000, 500, 1e-13)
        if h_new is not None:
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project(h_best, len(h_best) / 2.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
