from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > ts:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = ts - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            m = proj < 1.0 - 1e-10
        else:
            m = proj > 1e-10
        if m.any():
            proj[m] += diff / m.sum()
    return np.clip(proj, 0.0, 1.0)

def soft_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    mx = np.max(c)
    ec = np.exp(beta * (c - mx))
    sp = np.sum(ec)
    obj = mx + np.log(sp) / beta
    p = ec / sp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        m = k - n + 1
        if m >= 0:
            grad[m:] += pk * dx * (1.0 - h[:n - m])
            grad[:n - m] -= pk * dx * h[m:]
        else:
            pp = -m
            grad[:n - pp] += pk * dx * (1.0 - h[pp:])
            grad[pp:] -= pk * dx * h[:n - pp]
    return obj, grad
