        for it in range(phases):
            if time.time() > t_end:
                break
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            oi = ones[np.random.randint(len(ones))]
            zi = zeros[np.random.randint(len(zeros))]
            b[oi] = 0.0
            b[zi] = 1.0
            c5_new = compute_c5(b, dx)
            delta = c5_new - c5
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                c5 = c5_new
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_b = b.copy()
            else:
                b[oi] = 1.0
                b[zi] = 0.0
    return best_b, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0

    best_h = project_feasible(h_init.copy(), ts_init)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    # Phase 1: Smooth optimization from initial (30% time)
    t1 = start_time + budget_s * 0.30
    h_s, c5_s = optimize_smooth(best_h, dx_init, ts_init, t1)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 2: Binary optimization (35% time)
    t2 = start_time + budget_s * 0.65
    b_init = threshold_to_binary(best_h, int(ts_init))
    b_g, c5_g = greedy_binary(b_init, dx_init, max_passes=800)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = b_g.astype(float).copy()

    # SA from multiple random starts
    t_sa_end = t2
    n_starts = 0
    while time.time() < t_sa_end and n_starts < 6:
        n_starts += 1
        if time.time() < t_sa_end - 60:
            b_rand = np.zeros(n_init)
            idx = np.random.choice(n_init, size=int(ts_init), replace=False)
            b_rand[idx] = 1.0
            b_g2, c5_g2 = greedy_binary(b_rand, dx_init, max_passes=300)
        else:
            b_g2, c5_g2 = b_g, c5_g
        t_sub = min(t_sa_end, time.time() + 40)
        b_sa, c5_sa = sa_binary(b_g2, dx_init, t_sub, T0=0.06)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = b_sa.astype(float).copy()

    # Phase 3: Smooth optimization from best binary and random starts (25% time)
    t3 = start_time + budget_s * 0.92
    for attempt in range(8):
        if time.time() > t3:
            break
        if attempt == 0:
            h_start = project_feasible(best_h, ts_init)
        elif attempt == 1:
            b_thresh = threshold_to_binary(best_h, int(ts_init))
            h_start = project_feasible(b_thresh + 0.3 * np.random.randn(n_init) * 0.05, ts_init)
        else:
            h_start = project_feasible(np.random.uniform(0, 1, n_init), ts_init)
        c5_0 = compute_c5(h_start, dx_init)
        t_sub = min(t3, time.time() + max(20, (t3 - time.time()) * 0.35))
        h_opt, c5_opt = optimize_smooth(h_start, dx_init, ts_init, t_sub)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Final polish - smooth from current best
    t4 = start_time + budget_s * 0.97
    h_final, c5_final = optimize_smooth(best_h, dx_init, ts_init, t4)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    best_h = project_feasible(np.clip(best_h, 0, 1), ts_init)
    best_c5 = compute_c5(best_h, dx_init)

    return best_h, best_c5, best_n


def threshold_to_binary(h, ts):
    n = len(h)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:int(ts)]] = 1.0
    return b
