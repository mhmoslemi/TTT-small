import numpy as np

def fast_corr_full(a, b):
    n = len(a)
    N = 2 * n - 1
    return np.fft.irfft(np.fft.rfft(a, N) * np.conj(np.fft.rfft(b, N)), N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_corr_full(h, 1.0 - h)
    return float(np.max(corr) * dx)

def obj_grad(h, T):
    n = len(h)
    corr = fast_corr_full(h, 1.0 - h)
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    se = e.sum()
    sw = e / se
    obj = (mc + np.log(se) / T) * (2.0 / n)
    dx = 2.0 / n
    grad = np.zeros(n)
    c = n - 1
    for o in range(-(n-1), n):
        w = sw[c + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n-o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n-o]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return obj, grad * dx

def project(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo+hi)*0.5, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        mask = (h > 1e-6) & (h < 1 - 1e-6)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += d / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            i = np.argmin(h) if d > 0 else np.argmax(h)
            h[i] = np.clip(h[i] + d, 0.0, 1.0)
    return h

def adam_opt(h0, n_steps, T0, T1, lr0, target):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c = compute_c5(h)
    best_h = h.copy()
    b1, b2 = 0.9, 0.999
    for s in range(n_steps):
        frac = s / max(n_steps - 1, 1)
        T = T0 * (T1 / T0) ** frac
        _, g = obj_grad(h, T)
        sr = s + 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1**sr)
        vh = v / (1 - b2**sr)
        lr = lr0 * (T1 / T0) ** (0.5 * frac)
        h -= lr * mh / (np.sqrt(vh) + 1e-8)
        h = project(h, target)
        if s % 40 == 0 or s == n_steps - 1:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

            grad[:n - o] += w * (1.0 - h[o:])
            grad[o:] -= w * h[:n - o]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return obj, grad * dx

def project(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += d / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = np.argmin(h) if d > 0 else np.argmax(h)
            h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))
    return h

def adam_opt(h0, n_steps, T0, T1, lr0, target):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c = compute_c5(h)
    best_h = h.copy()
    b1, b2 = 0.9, 0.999
    for s in range(n_steps):
        frac = s / max(n_steps - 1, 1)
        T = T0 * (T1 / T0) ** frac
        _, g = obj_grad(h, T)
        sr = s + 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mh = m / (1.0 - b1 ** sr)
        vh = v / (1.0 - b2 ** sr)
        lr = lr0 * (1.0 + 0.5 * frac)
        h = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h = project(h, target)
        if s % 50 == 0 or s == n_steps - 1:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def coord_refine(h, target, n_iter=500):
    n = len(h)
    h = h.copy()
    c_best = compute_c5(h)
    rng_local = np.random.default_rng(12345)
    for _ in range(n_iter):
        idx = rng_local.integers(0, n)
        delta = rng_local.uniform(0.01, 0.05) * (1 if rng_local.random() < 0.5 else -1)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        h_new = project(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < c_best - 1e-7:
            c_best = c_new
            h = h_new
    return h, c_best

def make_patterns(h_best, n, target, rng):
    pats = []
    for bp in [3, 4, 5, 6, 7, 9, 12]:
        half = bp // 2
        p = np.tile(np.concatenate([np.ones(half), np.zeros(bp - half)]), n // bp + 1)[:n].astype(float)
        pats.append(project(p, target))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [2, 3, 5, 7]:
        s = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        pats.append(project(s, target))
    for _ in range(3):
        r = rng.random(n)
        pats.append(project((r > rng.uniform(0.35, 0.65)).astype(float), target))
    for sc in [0.05, 0.15, 0.3]:
        pats.append(project(np.clip(h_best + rng.normal(0, sc, n), 0, 1), target))
    return pats

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project(h_best, target)
    c5_best = compute_c5(h_best)

    def remain():
        return max(0.0, budget_s - (time.time() - t0))

    # Calibrate speed
    t_cal = time.time()
    for _ in range(50):
        _, _ = obj_grad(h_best, 50)
    time_per_step = max((time.time() - t_cal) / 50.0, 0.0001)
    steps_per_sec = 1.0 / time_per_step

    # Phase 1: Refine parent (25% budget)
    if remain() > 80:
        bu = min(remain() * 0.25, 250)
        ns = int(min(bu * steps_per_sec * 0.7, 8000))
        ns = max(ns, 2000)
        h1, c1 = adam_opt(h_best, ns, 100, 3, 0.04, target)
        if c1 < c5_best:
            c5_best, h_best = c1, h1
        h_best, c5_best = coord_refine(h_best, target, 300)

    # Phase 2: Diverse restarts (45% budget)
    if remain() > 80:
        patterns = make_patterns(h_best, n, target, rng)
        per_pat_time = remain() * 0.45 / len(patterns)
        per_pat_steps = int(min(per_pat_time * steps_per_sec * 0.6, 4000))
        per_pat_steps = max(per_pat_steps, 800)
        for hp in patterns:
            if remain() < 40:
                break
            h2, c2 = adam_opt(hp, per_pat_steps, 60, 3, 0.035, target)
            h2, c2 = coord_refine(h2, target, 150)
            if c2 < c5_best:
                c5_best, h_best = c2, h2

    # Phase 3: Higher resolution refinement (30% budget)
    for n_hi in [420, 560]:
        if remain() < 50:
            break
        target_hi = n_hi / 2.0
        n_old = len(h_best)
        x_old = np.arange(n_old) * (2.0 / n_old)
        x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project(h_hi, target_hi)
        # Recalibrate for new n
        t_cal2 = time.time()
        for _ in range(30):
            _, _ = obj_grad(h_hi, 40)
        tps2 = max((time.time() - t_cal2) / 30.0, 0.0002)
        spp2 = 1.0 / tps2
        ns_hi = int(min(remain() * 0.15 * spp2 * 0.6, 5000))
        ns_hi = max(ns_hi, 1500)
        h_hi, c_hi = adam_opt(h_hi, ns_hi, 50, 2, 0.03, target_hi)
        h_hi, c_hi = coord_refine(h_hi, target_hi, 100)
        if c_hi < c5_best:
            c5_best, h_best = c_hi, h_hi

    c5_final = compute_c5(h_best)
    return (h_best, min(c5_final, c5_best), len(h_best))
