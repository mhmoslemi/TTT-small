    """Minimize f_k(h) = sum_i h[i+k]*(1-h[i]) via projected gradient."""
    h = h0.copy()
    dx_step = 1.0 / (n)  # Lipschitz constant related
    
    for it in range(n_iter):
        # Compute gradient of f_k
        grad = np.zeros(n)
        # f_k = sum_{i=0}^{n-1-k} h[i+k] - sum_{i=0}^{n-1-k} h[i+k]*h[i]
        # d/dh[j]: 
        # First term: 1 if j in [k, n-1]
        # Second term: -h[j-k] (if j >= k) - h[j+k] (if j+k < n)
        grad[k:n] += 1.0  # from linear term
        if k > 0:
            grad[k:] -= h[:-k]  # -h[j-k] for j >= k
        grad[:n-k] -= h[k:]  # -h[j+k] for j < n-k

            np.clip(h_new, 0, 1, out=h_new)

from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sopt(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

_GDX = 0.0
_GSC = 50.0

        return _proj(r.x, n)

    h = h0.copy()
    best = h0.copy()
    best_c = _c5(best)
