        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Main simulated annealing
    sa_end = deadline - 120
    if time.time() < sa_end:
        h_sa, c5_sa = simulated_annealing(best_h, n, target, rng, sa_end)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa

    # Phase 3: SLSQP refinement chain
    for alpha in [400, 800, 1600, 3200]:
        if time.time() > deadline - 40:
            break
        h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=120)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Multi-start exploration
    if time.time() < deadline - 40:
        h_ms, c5_ms = multi_start_slsqp(best_h, n, target, rng, deadline - 20)
        if c5_ms < best_c5:
            best_h = h_ms
            best_c5 = c5_ms

    # Phase 5: Final short SA polish
    if time.time() < deadline - 8:
        h_pol, c5_pol = simulated_annealing(best_h, n, target, rng, deadline - 3)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol

    # Phase 6: Final SLSQP
    if time.time() < deadline - 2:
        h_final = slsqp_optimize(best_h, n, target, 5000, maxiter=100)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Ensure feasibility
    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
