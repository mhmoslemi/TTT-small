def compute_c5_and_argmax(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    am = int(np.argmax(corr))
    return float(corr[am] * dx), am

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):

    x_src = np.linspace(0, 2, n_src)

    h_new = np.clip(h_new, 0, 1)

def grad_single_k(h, k, n):
    """Gradient of corr[k] with respect to h, where corr = np.correlate(h, 1-h, 'full')"""
    g = np.zeros(n)
    partner = k - np.arange(n)
    valid = (partner >= 0) & (partner < n)
    g[valid] = 1.0 - 2.0 * h[partner[valid]]
    return g

    partner = k - np.arange(n)
    valid = (partner >= 0) & (partner < n)
    return np.sum(h[np.arange(n)[valid]] * (1.0 - h[partner[valid]]))

    k = _g_k

    n = _g_n

    return float(np.max(corr) * dx)

        if diff > 0:

        else:
            mask = h > 1e-15

import time

_gk = 0
_gn = 0
_gt = 0.0

def _obj(x):
    p = _gk - np.arange(_gn)
    v = (p >= 0) & (p < _gn)
    return float(np.sum(x[np.arange(_gn)[v]] * (1.0 - x[p[v]])))

def _grad(x):
    p = _gk - np.arange(_gn)
    v = (p >= 0) & (p < _gn)
    g = np.zeros(_gn)
    g[v] = 1.0 - 2.0 * x[p[v]]
    return g

    return float(np.sum(x) - _gt)

def c5_of(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), int(np.argmax(c))

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(80):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 - 1e-15 if d > 0 else h > 1e-15
        if not np.any(m):
            break
        h[m] += d / np.count_nonzero(m)
        h = np.clip(h, 0.0, 1.0)
    return h

    n1 = len(h)
    x1 = np.linspace(0, 2, n1)
    x2 = np.linspace(0, 2, n2)
    h2 = np.clip(np.interp(x2, x1, h), 0, 1)
    return fix(h2, n2, n2 / 2.0)
