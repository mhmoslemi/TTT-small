import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, S):
    """Project h to satisfy 0<=h<=1 and sum(h)=S"""
    h = np.clip(h, 0.0, 1.0)
    # Adjust to satisfy sum constraint
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference to positions not at bounds
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if free.any():
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
        # Final adjustment
        h += (S - h.sum()) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_grad_descent(h, n, S, n_iters, t_start, t_end, T_init, T_final):
    """Projected gradient descent with soft-max objective"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        # Adaptive temperature
        frac = it / max(n_iters, 1)
        T = T_init * (T_final / T_init) ** frac
        
        corr = compute_corr(h)
        mx = corr.max()

        shifted = T * (corr - mx)
        p = np.exp(shifted)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(compute_corr(h).max() * dx)

def project_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = (h < 1.0 - 1e-9).astype(float)
    else:
        mask = (h > 1e-9).astype(float)
    if mask.sum() > 1e-12:
        h += diff * mask / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def compute_softgrad(h, n, T):
    corr = compute_corr(h)
    mx = corr.max()
    shifted = np.clip(T * (corr - mx), -400, 400)
    p = np.exp(shifted)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= ps
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        jo = k - (n - 1)
        if jo >= 0:
            if jo < n:
                grad[jo:] += pk * (1.0 - h[:n - jo])
                grad[:n - jo] -= pk * h[jo:]
        else:
            jj = -jo
            if jj < n:
                grad[:n - jj] += pk * (1.0 - h[jj:])
                grad[jj:] -= pk * h[:n - jj]
    return h, grad, mx

def soft_grad_descent(h, n, S, n_iters, t_deadline, T_start, T_end, eta_scale=1.0):
    h = project_h(h.astype(float).copy(), S)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if time.time() > t_deadline:
            break
        frac = it / max(n_iters - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        _, grad, _ = compute_softgrad(h, n, T)
        grad -= grad.mean()
        eta = eta_scale / (T * 0.2 + 1.0)
        eta = min(eta, 2.0 / (n + 1))
        h_new = project_h(h - eta * grad, S)
        if not np.all(np.isfinite(h_new)):
            break
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def flip_delta(h, pos, to_one, n):
    delta = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        jo = k - (n - 1)
        sign = 1.0 if to_one else -1.0
        if 0 <= pos - jo < n:
            delta[k] += sign * (1.0 - h[pos - jo])
        if 0 <= pos + jo < n:
            delta[k] -= sign * h[pos + jo]
    return delta

def binary_hill_climb(h, n, S, t_deadline, max_swaps=500):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    no_improve = 0
    n_swaps = 0
    while time.time() < t_deadline and no_improve < 8 and n_swaps < max_swaps:
        corr = compute_corr(h)
        M = float(corr.max())
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        # Precompute deltas
        d10 = {}
        d01 = {}
        for i in one_idx:
            d10[i] = flip_delta(h, i, False, n)
        for j in zero_idx:
            d01[j] = flip_delta(h, j, True, n)
        improved = False
        best_new_max = M
        best_pair = None
        for i in one_idx:
            for j in zero_idx:
                nc = corr + d10[i] + d01[j]
                nm = float(nc.max())
                if nm < best_new_max - 1e-12:
                    best_new_max = nm
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            n_swaps += 1
            no_improve = 0
            new_c5 = best_new_max * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            no_improve += 1
    return best_h, best_c5

def coordinate_refine(h, n, S, t_deadline, n_sweeps=50):
    h = project_h(h.astype(float).copy(), S)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        if time.time() > t_deadline:
            break
        improved_any = False
        for m in range(n):
            if time.time() > t_deadline:
                break
            old = h[m]
            # Compute beta[k] = dc[k]/dh[m] for all k
            beta = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                jo = k - (n - 1)
                bk = 0.0
                if 0 <= m - jo < n:
                    bk += (1.0 - h[m - jo])
                if 0 <= m + jo < n:
                    bk -= h[m + jo]
                beta[k] = bk
            # alpha[k] = corr[k] - old * beta[k]
            corr_cur = compute_corr(h)
            alpha = corr_cur - old * beta
            # Minimize max_k (alpha[k] + beta[k]*t) over t in [0,1]
            # Candidates: t=0, t=1, and intersections of top lines
            vals_0 = alpha.copy()
            vals_1 = alpha + beta
            best_t = 0.0
            best_val = max(vals_0.max(), vals_1.max())
            # Check intersections of lines that are near current max
            top_k = np.argsort(-np.abs(beta))[:20]
            for ii in range(len(top_k)):
                for jj in range(ii + 1, len(top_k)):
                    i1, i2 = top_k[ii], top_k[jj]
                    if abs(beta[i1] - beta[i2]) < 1e-12:
                        continue
                    t_star = (alpha[i2] - alpha[i1]) / (beta[i1] - beta[i2])
                    if 0.01 < t_star < 0.99:
                        v = vals_0.max()
                        cand = alpha + t_star * beta
                        cm = cand.max()
                        if cm < best_val - 1e-10:
                            best_val = cm
                            best_t = t_star
            # Check boundaries more carefully
            v0 = alpha.max()
            v1 = (alpha + beta).max()
            if v0 < best_val - 1e-10:
                best_val = v0
                best_t = 0.0
            if v1 < best_val - 1e-10:
                best_val = v1
                best_t = 1.0
            if abs(best_t - old) > 1e-8:
                new_h = h.copy()
                new_h[m] = best_t
                new_h = project_h(new_h, S)
                c5_new = compute_c5(new_h)
                if c5_new < best_c5 - 1e-12:
                    h = new_h
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved_any = True
        if not improved_any:
            break
    return best_h, best_c5
