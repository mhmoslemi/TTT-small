def _smooth_grad(h, dx, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(sc * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k_idx in range(2*n - 1):
        if w[k_idx] < 1e-14:
            continue
        k = k_idx - (n - 1)  # actual lag
        wk = w[k_idx]
        if k >= 0:
            # (1-h[j+k]) contributes to grad[j] for j in [0, n-1-k]
            grad[:n-k] += wk * one_minus_h[k:]
            # -h[j-k] contributes to grad[j] for j in [k, n-1]
            grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            # (1-h[j-kk]) contributes to grad[j] for j in [kk, n-1]
            grad[kk:] += wk * one_minus_h[:n-kk]
            # -h[j+kk] contributes to grad[j] for j in [0, n-1-kk]
            grad[:n-kk] -= wk * h[kk:]
    return dx * grad

    # Simpler: (m + log(sum(exp(sc*(corr-m))))/sc) * dx

def _smooth_grad(h, dx, sc):

    corr = np.correlate(h, 1.0 - h, mode="full")

    log_w = sc * (corr - m)

                grad[k:] -= wk * h[:n-k]

            np.clip(h, 0.0, 1.0, out=h)

    if abs(r) > 1e-12:

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_obj(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = np.logaddexp.reduce(sc * (corr - m))
    return (m + s / sc) * dx

def _smooth_grad(h, dx, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    lw = sc * (corr - m)
    lw -= np.logaddexp.reduce(lw)
    w = np.exp(lw)
    grad = np.zeros(n)
    omh = 1.0 - h
    for ki in range(2 * n - 1):
        if w[ki] < 1e-13:
            continue
        k = ki - (n - 1)
        wk = w[ki] * dx
        if k >= 0:
            if k < n:
                grad[:n-k] += wk * omh[k:]
                grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * omh[:n-kk]
            grad[:n-kk] -= wk * h[kk:]

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] -= d / free.sum()
        else:

        np.clip(h, 0.0, 1.0, out=h)

    if abs(r) > 1e-12:

        if msk.any():

def _local_search(h, tmax, mode='grad'):

import numpy as np

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_obj(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = np.logaddexp.reduce(sc * (corr - m))
    return (m + s / sc) * dx

def _smooth_grad(h, dx, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    lw = sc * (corr - m)
    lw -= np.logaddexp.reduce(lw)
    w = np.exp(lw)
    grad = np.zeros(n)
    omh = 1.0 - h
    for ki in range(2 * n - 1):
        if w[ki] < 1e-13:
            continue
        k = ki - (n - 1)
        wk = w[ki] * dx
        if k >= 0:
            if k < n:
                grad[:n-k] += wk * omh[k:]
                grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * omh[:n-kk]
            grad[:n-kk] -= wk * h[kk:]
    return grad

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] -= d / free.sum()
        else:
            h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

        frac = min(1.0, elapsed / max(1e-6, tmax))
        sc = sc0 * (sc1 / sc0) ** frac
        obj_cur = _smooth_obj(h, dx, sc)
        g = _smooth_grad(h, dx, sc)
        gnorm = np.linalg.norm(g)

        step *= 0.98
        accepted = False

            if obj_new <= obj_cur + 1e-15:
                h = h_new
                step = min(step * 1.05, 50.0 / n)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            flat_cnt += 1
            if flat_cnt > 200:
                break
        else:
            flat_cnt = 0
        c = _c5(h)
        if c < bestc:

    return best, bestc
