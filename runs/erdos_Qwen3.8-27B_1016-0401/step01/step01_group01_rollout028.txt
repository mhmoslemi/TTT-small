    dx, tau = _P['dx'], _P['tau']
    c = np.correlate(x, 1.0 - x, mode='full')

    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    w = np.exp((c - cm) / tau)

    grad = np.zeros(n)

        wm = w[s + n - 1]
        if wm < 1e-10:
            continue
        if s >= 0:

            grad[i] += wm * (1.0 - x[i + s])

            grad[i] += wm * (1.0 - x[i - sp])
            grad[i - sp] += wm * (-x[i])
