import time

def compute_corr(h):

def make_feasible(h, n, target):

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_of_lag(h, m):
    """Subgradient of correlation at lag magnitude m (0 to n-1)."""
    n = len(h)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        for i in range(m, n):
            grad[i] += (1.0 - h[i - m])
        for i in range(0, n - m):
            grad[i] -= h[i + m]
    return grad

def subgradient_descent(h, n, target, deadline, top_k=3):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    n_corr = 2 * n - 1
    
    lr = 0.3
    momentum = np.zeros(n)
    beta = 0.9
    
    it = 0
    while time.time() < deadline:
        it += 1
        corr = compute_corr(h_curr)
        c5_val = corr.max() * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_curr.copy()
        
        # Find top_k worst lags
        top_indices = np.argsort(corr)[-top_k:]
        # Average subgradient
        grad = np.zeros(n)
        for ki in top_indices:
            m = abs(int(ki) - (n - 1))
            grad += subgrad_of_lag(h_curr, m)
        grad /= top_k
        
        # Momentum
        momentum = beta * momentum - lr * grad
        h_new = h_curr + momentum
        h_new = make_feasible(h_new, n, target)
        
        # Adaptive lr
        if it % 200 == 0:
            lr *= 0.98
            lr = max(lr, 0.01)
        
        h_curr = h_new
    
    return best_h, best_c5

def slsqp_optimize(h, n, target, alpha, maxiter=200, deadline=None):

    dx = 2.0 / n_pts
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    
    def sum_cons(x):
        return np.sum(x) - target
    
    try:
        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n_pts,
            constraints=[{'type': 'eq', 'fun': sum_cons}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return make_feasible(res.x, n, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_subgrad_descent(h, n, target, deadline, beta0=10.0, beta_end=300.0, total_iters=80000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    n_corr = 2 * n - 1
    lr = 0.12
    momentum = np.zeros(n)
    beta_mom = 0.85
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        frac = min(it / total_iters, 1.0)
        beta = beta0 + (beta_end - beta0) * (frac ** 0.6)
        
        corr = compute_corr(h_curr)
        c5_val = corr.max() * (2.0 / n)
        if c5_val < best_c5 - 1e-12:

            best_h = h_curr.copy()
        
        vals = np.clip(beta * (corr - corr.max()), -50.0, 0.0)
        exps = np.exp(vals)
        wsum = np.sum(exps)
        weights = exps / wsum
        
        wb = np.zeros(n)
        center = n - 1
        for ki in range(n_corr):
            wb[abs(ki - center)] += weights[ki]
        
        grad = np.zeros(n)

def slsqp_optimize(h, n, target, alpha, maxiter=150):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def cons(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_subgrad_descent(h, n, target, deadline, beta0=10.0, beta_end=400.0, total_iters=60000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    n_corr = 2 * n - 1
    center = n - 1
    lr = 0.12
    momentum = np.zeros(n)
    beta_mom = 0.85
    for it in range(total_iters):
        if time.time() > deadline:
            break
        frac = min(it / total_iters, 1.0)
        beta = beta0 + (beta_end - beta0) * (frac ** 0.6)
        corr = compute_corr(h_curr)
        c5_val = corr.max() * (2.0 / n)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_curr.copy()
        vals = np.clip(beta * (corr - corr.max()), -50.0, 0.0)
        exps = np.exp(vals)
        weights = exps / np.sum(exps)
        wb = np.zeros(n)
        for ki in range(n_corr):
            wb[abs(ki - center)] += weights[ki]
        grad = np.zeros(n)
        if wb[0] > 1e-12:
            grad += wb[0] * (1.0 - 2.0 * h_curr)
        for m in range(1, n):
            w = wb[m]
            if w < 1e-12:
                continue
            grad[m:] += w * (1.0 - h_curr[:-m])
            grad[:-m] -= w * h_curr[m:]
        momentum = beta_mom * momentum + lr * grad
        h_new = h_curr - momentum
        h_new = make_feasible(h_new, n, target)
        if it > 0 and it % 2000 == 0:
            lr *= 0.99
        h_curr = h_new
    return best_h, best_c5

def slsqp_optimize(h, n, target, alpha, maxiter=150):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def cons(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def targeted_local_search(h, n, target, rng, deadline, step=0.02):
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    it = 0
    while time.time() < deadline:
        it += 1
        corr = compute_corr(h_curr)
        c5_val = corr.max() * (2.0 / n)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
        k_star = int(np.argmax(corr))
        m = abs(k_star - (n - 1))
        h_try = h_curr.copy()
        if m > 0:
            valid = np.arange(n - m)
            contrib = h_curr[valid] * (1.0 - h_curr[valid + m])
            tsum = contrib.sum()
            if tsum > 1e-10:
                i = rng.choice(n - m, p=contrib / tsum)
            else:
                i = rng.randint(n - m)
            j = i + m
            d = step * rng.uniform(0.5, 1.0)
            if h_curr[i] > 0.4:
                h_try[i] -= d
                h_try[j] += d
            else:
                h_try[i] += d
                h_try[j] -= d
        else:
            i = rng.randint(n)
            d = step * rng.uniform(0.5, 1.0)
            if h_try[i] < 0.5:
                h_try[i] += d
            else:
                h_try[i] -= d
        if rng.random() < 0.4:
            a, b = rng.choice(n, 2, replace=False)
            h_try[a] += rng.uniform(-step, step)
            h_try[b] -= rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_curr = h_try
            best_c5 = c5_try
        elif it % 20000 == 10000:
            idx = rng.choice(n, size=2, replace=False)
            h_curr[idx] += rng.uniform(-0.04, 0.04, 2)
            h_curr = make_feasible(h_curr, n, target)
            best_c5 = compute_c5(h_curr)
    return h_curr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, n, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm start
    p1_end = t0 + min(100, budget_s * 0.12)
    for alpha in [30, 100, 400, 1000]:
        if time.time() > p1_end:
            break
        h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=80)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Softmax subgradient descent (first-order, momentum)
    p2_end = t0 + min(350, budget_s * 0.45)
    h_sg, c5_sg = softmax_subgrad_descent(best_h, n, target, p2_end,
                                           beta0=10, beta_end=500, total_iters=60000)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # Phase 3: Targeted local search (worst-lag guided)
    p3_end = t0 + min(650, budget_s * 0.75)
    h_ls, c5_ls = targeted_local_search(best_h, n, target, rng, p3_end, step=0.02)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls, c5_ls

    # Phase 4: SLSQP refinement at high alpha
    if time.time() < dl - 50:
        for alpha in [1000, 2500, 6000]:
            if time.time() > dl - 50:
                break
            h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=80)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Second subgradient pass with sharp beta
    if time.time() < dl - 20:
        h_sg2, c5_sg2 = softmax_subgrad_descent(best_h, n, target, dl - 20,
                                                 beta0=80, beta_end=900, total_iters=30000)
        if c5_sg2 < best_c5:
            best_h, best_c5 = h_sg2, c5_sg2

    # Phase 6: Fine local search polish
    if time.time() < dl - 5:
        h_ls, c5_ls = targeted_local_search(best_h, n, target, rng, dl - 5, step=0.007)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), final_c5, n
