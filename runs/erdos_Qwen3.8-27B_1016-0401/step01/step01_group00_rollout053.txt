import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to [0,1]^n with sum = target via bisection on shift."""
    h = h.copy()
    lo, hi = -h.max(), 1.0 - h.min()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def coordinate_descent(h, n_passes=100, target=None):
    """Exact coordinate descent for the Erdos overlap problem."""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    
    c = np.correlate(h, 1 - h, mode="full")  # current correlation, length 2n-1
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Precompute lag offsets
    lags = np.arange(-(n-1), n)  # length 2n-1
    
    for it in range(n_passes):
        # Randomize order
        order = np.random.permutation(n)
        
        for j in order:
            # Compute a[lag] for this j
            # a[lag] = (1-h[j-lag])*[0<=j-lag<n] - h[j+lag]*[0<=j+lag<n]
            # j-lag ranges over: for lag in [-(n-1), n-1], j-lag in [j-(n-1), j+(n-1)]
            # Valid: 0 <= j-lag < n
            jl = j - lags  # j - lag for each lag
            valid1 = (jl >= 0) & (jl < n)
            jr = j + lags  # j + lag for each lag
            valid2 = (jr >= 0) & (jr < n)
            
            a = np.zeros(2*n-1)
            a[valid1] = 1.0 - h[jl[valid1]]
            a[valid2] -= h[jr[valid2]]
            
            # b[lag] = c[lag] - h[j] * a[lag]
            b = c - h[j] * a
            
            # Golden section search for x in [0,1] minimizing max(a*x + b)
            lo_x, hi_x = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio - 1 ≈ 0.618
            x1 = hi_x - gr * (hi_x - lo_x)
            x2 = lo_x + gr * (hi_x - lo_x)
            f1 = np.max(a * x1 + b)
            f2 = np.max(a * x2 + b)
            
            for _ in range(60):
                if f1 < f2:
                    hi_x = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi_x - gr * (hi_x - lo_x)
                    f1 = np.max(a * x1 + b)
                else:
                    lo_x = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo_x + gr * (hi_x - lo_x)
                    f2 = np.max(a * x2 + b)
            
            x_star = (lo_x + hi_x) / 2.0

            if abs(delta) > 1e-15:

                c += delta * a
        
        # Project to satisfy constraint
        h = project_sum(h, target)
        c = np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -h.max() - 0.1, 1.0 - h.min() + 0.1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def coord_descent(h, n_passes, rng, target):
    n = len(h)
    h = h.astype(float).copy()
    h = project_sum(h, target)
    c = np.correlate(h, 1 - h, mode="full")
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lags = np.arange(-(n-1), n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for it in range(n_passes):
        order = rng.permutation(n)
        for j in order:
            jl = j - lags
            jr = j + lags
            v1 = (jl >= 0) & (jl < n)
            v2 = (jr >= 0) & (jr < n)
            a = np.zeros(2*n-1)
            a[v1] = 1.0 - h[jl[v1]]
            a[v2] -= h[jr[v2]]
            b = c - h[j] * a
            # Golden section
            lo_x, hi_x = 0.0, 1.0
            x1 = hi_x - gr * (hi_x - lo_x)
            x2 = lo_x + gr * (hi_x - lo_x)
            f1 = np.max(a * x1 + b)
            f2 = np.max(a * x2 + b)
            for _ in range(45):
                if f1 < f2:
                    hi_x, x2, f2 = x2, x1, f1
                    x1 = hi_x - gr * (hi_x - lo_x)
                    f1 = np.max(a * x1 + b)
                else:
                    lo_x, x1, f1 = x1, x2, f2
                    x2 = lo_x + gr * (hi_x - lo_x)
                    f2 = np.max(a * x2 + b)
            x_star = (lo_x + hi_x) * 0.5
            delta = x_star - h[j]
            if abs(delta) > 1e-14:
                h[j] = x_star
                c += delta * a
        h = project_sum(h, target)
        c = np.correlate(h, 1 - h, mode="full")
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 140
        h0 = np.zeros(n0)
        h0[::2] = 1.0
    
    n = len(h0)
    target = n / 2.0
    h_best = project_sum(h0, target)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Coordinate descent at current resolution
    remaining = budget_s - elapsed()
    n_passes1 = min(500, int(remaining * 30))  # ~30 passes per second
    h_best, c5_cd = coord_descent(h_best, n_passes1, rng, target)
    c5_best = min(c5_best, c5_cd)
    
    # Phase 2: Try different resolutions
    for n_hi in [2*n, 3*n]:
        if elapsed() > budget_s - 60:
            break
        if n_hi > 1000:
            break
        target_hi = n_hi / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_sum(h_hi, target_hi)
        c5_hi = compute_c5(h_hi)
        
        remaining = budget_s - elapsed() - 30
        n_passes2 = min(300, int(remaining * 20))
        h_hi, c5_hi2 = coord_descent(h_hi, n_passes2, rng, target_hi)
        c5_hi2 = min(c5_hi, c5_hi2)
        
        if c5_hi2 < c5_best:
            c5_best = c5_hi2
            h_best = h_hi
    
    # Phase 3: Random restarts at best resolution
    while elapsed() < budget_s - 30:
        h_init = h_best + rng.normal(0, 0.1, n)
        h_init = project_sum(h_init, target)
        n_passes3 = 50
        h_r, c5_r = coord_descent(h_init, n_passes3, rng, target)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r
    
    # Final
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))

def softmax_refine(h, target, T, t0, time_limit):
    """Polish with soft-max gradient using L-BFGS-B."""
    from scipy.optimize import minimize
    n = len(h)
    dx = 2.0 / n
    ln = 2*n - 1
    
    def obj(x):
        corr = np.correlate(x, 1-x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= sh.max()
        return dx * (mc + np.log(np.sum(np.exp(sh))) / T)

        corr = np.correlate(x, 1-x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= sh.max()

        w = e / np.sum(e)
        wr = w[::-1]
        fft_size = 1
        while fft_size < 3*n - 2:
            fft_size *= 2
        c1 = np.fft.irfft(np.fft.rfft(1-x, fft_size) * np.fft.rfft(w, fft_size), fft_size)[:3*n-2]

        return obj(np.clip(h_full, 0, 1))

        h_full = np.empty(n)
        h_full[0] = target - x.sum()
        h_full[1:] = x
        g = grad(np.clip(h_full, 0, 1))
        return np.ones(n-1) * g[0] + g[1:]
    
    bounds = [(0.0, 1.0)] * (n-1)
    cons = [
        {'type': 'ineq', 'fun': lambda x: x.sum() - (target - 1.0), 'jac': lambda x: np.ones(n-1)},
        {'type': 'ineq', 'fun': lambda x: (target - 0.0) - x.sum(), 'jac': lambda x: -np.ones(n-1)}

Wait, I realize the constraint handling for L-BFGS-B is tricky. The constraint h[0] = target - sum(x) means:
- 0 ≤ target - sum(x) ≤ 1
- 0 ≤ x_i ≤ 1 for all i

The first constraint gives: target - 1 ≤ sum(x) ≤ target
Which is: sum(x) - (target-1) ≥ 0 and target - sum(x) ≥ 0

In scipy's format: fun(x) ≥ 0 for 'ineq' constraints.
