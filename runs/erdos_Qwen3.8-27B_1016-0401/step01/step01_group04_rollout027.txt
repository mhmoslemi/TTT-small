def targeted_fc(h, n, S, t_end, rng):
    """Focused hill-climbing targeting the worst lags."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(c.max())
    best_h = h.copy()
    best_c = c_max
    stuck = 0
    
    while time.time() < t_end:
        # Find hot lags
        hot_threshold = c_max - 0.5
        hot_indices = np.where(c >= hot_threshold)[0]
        
        # For each hot lag, try to find and fix a contributing pair
        found = False
        for hi in hot_indices:
            # hi corresponds to lag hi - (n-1)
            lag = hi - (n - 1)
            k = abs(lag)
            if k == 0:
                continue
            # Components contributing to this c[hi]:
            # pairs (i, i+k) or (i+k, i) where they differ
            # Try random such pairs
            if lag > 0:
                valid_i = range(0, n - k)
            else:
                valid_i = range(-lag, n)
            
            # Find contributing pairs
            if lag > 0:
                contrib = [(i, i+k) for i in range(n-k) if h[i]*(1-h[i+k]) > 0.9]
            else:
                ik = -lag
                contrib = [(i-ik, i) for i in range(ik, n) if h[i-ik]*(1-h[i]) > 0.9]
            
            if not contrib:
                continue
            
            # Try a random contributing pair
            a, b = contrib[rng.randint(len(contrib))]
            if h[a] == h[b]:
                continue
            
            # Swap and check if max decreases or the hot lag specifically decreases
            h[a], h[b] = h[b], h[a]
            c_new = np.correlate(h, 1.0 - h, mode='full')
            
            if c_new.max() < c_max - 1e-12 or (c_new[hi] < c[hi] - 0.5 and c_new.max() <= c_max + 1e-12):
                c = c_new
                c_max = float(c.max())
                stuck = 0
                if c_max < best_c:
                    best_c = c_max
                    best_h = h.copy()
                found = True
                break
            else:
                h[a], h[b] = h[b], h[a]
        
        if not found:
            stuck += 1
            if stuck > 100:
                # Kick
                for _ in range(2):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if h[a] != h[b]:
                        h[a], h[b] = h[b], h[a]
                c = np.correlate(h, 1.0 - h, mode='full')
                c_max = float(c.max())
                stuck = 0
    
    return best_h, best_c

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    S = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        h_b[one_idx[order[:ones - S]]] = 0
    elif ones < S:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        h_b[zero_idx[order[:S - ones]]] = 1
    return h_b

def random_binary(n, S, rng):
    h = np.zeros(n)
    idx = rng.choice(n, S, replace=False)
    h[idx] = 1.0
    return h

def fixed_count(h, S, rng):
    h = h.astype(float).copy()
    S = int(S)
    ones = int(h.sum())
    if ones > S:
        one_idx = np.where(h == 1)[0]
        perm = rng.permutation(len(one_idx))
        h[one_idx[perm[:ones - S]]] = 0
    elif ones < S:
        zero_idx = np.where(h == 0)[0]
        perm = rng.permutation(len(zero_idx))
        h[zero_idx[perm[:S - ones]]] = 1
    return h
