    T1 = 0.01
    total_iters = 20000000  # cap
    cool_rate = (T1/T0)**(1.0/total_iters)
    
    it = 0
    no_improve = 0
    kick_interval = 100000
    
    while time.time() < t_limit:
        T = T0 * (cool_rate ** it)
        T = max(T, T1)
        
        a = zeros[rng.randint(len(zeros))]
        b = ones[rng.randint(len(ones))]

        mc = int(np.max(np.correlate(h, 1-h, 'full')))
        delta = mc - max_corr
        
        # Move a in zeros->ones, b in ones->zeros
        ones.remove(b); ones.append(a)
        zeros.remove(a); zeros.append(b)
        
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            max_corr = mc
            if mc < best_mc:
                best_mc = mc
                best = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            h[b] = 1.0; h[a] = 0.0

            zeros.remove(b); zeros.append(a)

                    ones.remove(kb); ones.append(ka)
                    zeros.remove(ka); zeros.append(kb)
            max_corr = int(np.max(np.correlate(h, 1-h, 'full')))

One more optimization: I can precompute the correlation array and update it incrementally for each swap. For a swap (a: 0→1, b: 1→0):
- corr[m] changes for specific m values
- Specifically, the change involves positions a and b

Let me derive the incremental update:
corr[m] = sum_i h[i] * (1-h[i+m]) for valid i

Change in corr[m] due to h[a]: 0→1:
- If a+m is valid: term h[a]*(1-h[a+m]) changes from 0 to (1-h[a+m]). Δ += (1-h[a+m])
- If a-m is valid (i.e., (a-m)+m = a): term h[a-m]*(1-h[a]) changes from h[a-m]*1 to h[a-m]*0. Δ -= h[a-m]

Change in corr[m] due to h[b]: 1→0:
- If b+m is valid: term h[b]*(1-h[b+m]) changes from (1-h[b+m]) to 0. Δ -= (1-h[b+m])
- If b-m is valid: term h[b-m]*(1-h[b]) changes from h[b-m]*0 to h[b-m]*1. Δ += h[b-m]

So: Δcorr[m] = (1-h[a+m])*I(a+m valid) - h[a-m]*I(a-m valid) - (1-h[b+m])*I(b+m valid) + h[b-m]*I(b-m valid)

This is O(1) per m, and we need to compute it for all m from -(n-1) to n-1, giving O(n) per swap.

But wait, the only m values where Δcorr[m] ≠ 0 are:
m = a-m' for some valid m' → m ranges around a
m = b-m' for some valid m' → m ranges around b

Actually, for a given a:
- (1-h[a+m]) is non-zero only when a+m is in [0, n-1], i.e., m ∈ [-a, n-1-a]
- h[a-m] is non-zero only when a-m is in [0, n-1], i.e., m ∈ [a-(n-1), a]

So the Δcorr is non-zero only for m in [-a, n-1-a] ∪ [a-(n-1), a] ∪ [-(b-(n-1)), n-1-b] ∪ [b-(n-1), b]... this is getting complicated.

def swap_delta(corr, h, a, b, n):
    """Compute change in correlation array when h[a]:0->1, h[b]:1->0."""
    # a changes from 0 to 1, b changes from 1 to 0
    delta = np.zeros(2*n - 1)  # index m+(n-1)
    
    for m in range(-(n-1), n):
        d = 0.0
        # Contribution of h[a] changing 0->1
        if 0 <= a + m < n:
            d += (1.0 - h[a + m])  # was 0, now (1-h[a+m])
        if 0 <= a - m < n:
            d -= h[a - m]  # was h[a-m]*1, now h[a-m]*0
        # Contribution of h[b] changing 1->0
        if 0 <= b + m < n:
            d -= (1.0 - h[b + m])  # was (1-h[b+m]), now 0
        if 0 <= b - m < n:
            d += h[b - m]  # was h[b-m]*0, now h[b-m]*1
        delta[m + n - 1] = d
    
    return delta

            best_h[i] = 0.0
    elif oc < ts:
        idxs = np.where(best_h < 0.5)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            best_h[i] = 1.0
    best_h = (best_h >= 0.5).astype(float)
    dx_f = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
