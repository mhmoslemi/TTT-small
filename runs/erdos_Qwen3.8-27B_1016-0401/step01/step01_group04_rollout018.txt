import numpy as np
from scipy.optimize import minimize
import time

_Tg = 100.0
_Sg = 35.0

def _obj_sm(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = corr.max()
    return float(mx + np.log(np.sum(np.exp(_Tg * (corr - mx)))) / _Tg)

def _con_eq(x):
    return x.sum() - _Sg

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def normalize_h(h, S):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * (S / s)
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, S):
    target = int(round(S))
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def slsqp(h, n, S, T, maxiter=150):
    global _Tg, _Sg
    _Tg = float(T)
    _Sg = float(S)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_sm, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': _con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()

def ls_1swap(h, n, S, deadline, max_sweeps=200):
    h_ls = h.astype(float).copy()
    best_h = h_ls.copy()
    best_c5 = compute_c5(h_ls)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        oi = np.where(h_ls == 1)[0]
        zi = np.where(h_ls == 0)[0]
        for i in oi:
            if time.time() > deadline:
                break
            for j in zi:
                if i == j:
                    continue
                h_ls[i] = 0.0
                h_ls[j] = 1.0
                c = compute_c5(h_ls)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = h_ls.copy()
                    improved = True
                else:
                    h_ls[i] = 1.0
                    h_ls[j] = 0.0
        if not improved:
            break
    return best_h, best_c5

def ls_2swap(h, n, S, rng, deadline, max_sweeps=10, n_trials=800):
    h_ls = h.astype(float).copy()
    best_h = h_ls.copy()
    best_c5 = compute_c5(h_ls)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        oi = np.where(h_ls == 1)[0]
        zi = np.where(h_ls == 0)[0]
        for _t in range(n_trials):
            if time.time() > deadline:
                break
            i1, i2 = rng.choice(oi, 2, replace=False)
            j1, j2 = rng.choice(zi, 2, replace=False)
            h_ls[i1] = 0; h_ls[i2] = 0
            h_ls[j1] = 1; h_ls[j2] = 1
            c = compute_c5(h_ls)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = h_ls.copy()
                improved = True
                oi = np.where(h_ls == 1)[0]
                zi = np.where(h_ls == 0)[0]
            else:
                h_ls[i1] = 1; h_ls[i2] = 1
                h_ls[j1] = 0; h_ls[j2] = 0
        if not improved:
            break
    return best_h, best_c5

def sa(h, n, S, rng, n_iter, deadline, best_c5_in):
    h_sa = h.astype(float).copy()
    c5_sa = compute_c5(h_sa)
    best_h = h_sa.copy()

        j = zi[rng.randint(len(zi))]

                best_c5_loc = c5_sa
                best_h = h_sa.copy()
        else:
            h_sa[i] = 1; h_sa[j] = 0
    return best_h, best_c5_loc

def gen_start(n, S, rng, v):
    S_int = int(S)
    h = np.zeros(n)
    if v == 0:
        pos = np.round(np.linspace(0, n - 1, S_int)).astype(int)
        pos = np.unique(pos)
        if len(pos) < S_int:
            rem = np.setdiff1d(np.arange(n), pos)
            pos = np.sort(np.concatenate([pos, rng.choice(rem, S_int - len(pos), replace=False)]))
        h[pos] = 1
    elif v == 1:
        phi = (1 + 5**0.5) / 2
        pos = np.unique(((np.arange(S_int) * phi * n).astype(int) % n))
        if len(pos) < S_int:
            rem = np.setdiff1d(np.arange(n), pos)
            pos = np.sort(np.concatenate([pos, rng.choice(rem, S_int - len(pos), replace=False)]))
        h[pos] = 1
    elif v == 2:
        idx = rng.choice(n, S_int, replace=False)
        h[idx] = 1
    elif v == 3:
        for k in range(S_int):
            c = int(k * n / S_int + rng.uniform(-1.0, 1.0))
            h[c % n] = 1
        if h.sum() != S_int:
            h = np.zeros(n)
            idx = rng.choice(n, S_int, replace=False)
            h[idx] = 1
    else:
        idx = rng.choice(n, S_int, replace=False)
        h[idx] = 1
        h = np.roll(h, rng.randint(1, n))
    return h
