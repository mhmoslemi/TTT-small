                if zi == oi:
                    continue
                best[zi], best[oi] = 1.0, 0.0
                c = compute_c5(best, dx)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    improved = True
                    break
                best[zi], best[oi] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def to_binary(h, ts, rng):
    n = len(h)
    b = np.where(h > 0.5, 1.0, 0.0).copy()
    oc = int(np.sum(b))
    if oc > ts:
        idxs = np.where(b == 1)[0]
        rng.shuffle(idxs)
        b[idxs[:oc - ts]] = 0.0
    elif oc < ts:
        idxs = np.where(b == 0)[0]
        rng.shuffle(idxs)
        b[idxs[:ts - oc]] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    dx0 = 2.0 / n0
    ts0 = n0 / 2.0
    best_h = project_feasible(h_init, ts0)
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0

    t1 = t0 + budget_s * 0.38
    h1, c1 = optimize_smooth(best_h, dx0, ts0, t1, rng)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    t2 = t0 + budget_s * 0.62
    h2, c2 = pgd_phase(best_h, dx0, ts0, t2, rng)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    t3 = t0 + budget_s * 0.80
    h3, c3 = binary_polish(best_h, dx0, ts0, t3, rng)
    if c3 < best_c5:
        best_c5, best_h = c3, h3
        best_n = n0

    # Try multiple resolutions and restarts
    t4 = t0 + budget_s * 0.96
    n_list = [n0, 48, 92, 96, 184, 194]
    n_list = sorted(set(x for x in n_list if x % 2 == 0 and x >= 10), key=lambda x: abs(x - n0))
    for n_try in n_list[:5]:
        if time.time() > t4 or n_try == best_n:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=True)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=True)
        h_interp = np.interp(x_new, x_old, best_h)
        for rs in range(3):
            if time.time() > t4:
                break
            if rs == 0:
                h_try = to_binary(h_interp, ts_t, rng)
            elif rs == 1:
                h_try = project_feasible(h_interp + 0.3 * rng.randn(n_try), ts_t)
            else:
                h_try = project_feasible(rng.uniform(0, 1, n_try), ts_t)
            rem = max(5.0, t4 - time.time())
            t_sub = time.time() + rem * 0.25
            hr, cr = optimize_smooth(h_try, dx_t, ts_t, t_sub, rng)
            if cr < best_c5:
                best_c5, best_h, best_n = cr, hr, n_try
            t_sub2 = time.time() + max(2.0, (t4 - time.time()) * 0.15)
            hr2, cr2 = pgd_phase(hr, dx_t, ts_t, t_sub2, rng)
            if cr2 < best_c5:
                best_c5, best_h, best_n = cr2, hr2, n_try

    # Final subgradient polish
    dx_f = 2.0 / best_n
    ts_f = best_n / 2.0
    t_end = t0 + budget_s * 0.99
    hf, cf = subgrad_cycle(best_h, dx_f, ts_f, t_end, rng)
    if cf < best_c5:
        best_c5, best_h = cf, hf

    dx_f = 2.0 / best_n
    best_h = project_feasible(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
