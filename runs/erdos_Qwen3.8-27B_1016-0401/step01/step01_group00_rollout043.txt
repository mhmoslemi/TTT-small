        h[i] = 1.0 - (bin(i).count('1') % 2)
    return h

def rotational_h(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = thue_morse_h(280)
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Anneal from parent with L-BFGS-B
    T_schedule = [200, 100, 60, 40, 25, 15, 10, 7, 5, 3, 2]
    for T in T_schedule:
        if elapsed() > budget_s - 120:
            break
        try:
            res = minimize(sm_obj, h_best, args=(T, 1e4), method='L-BFGS-B',
                          jac=sm_grad, bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-12,
                                   'maxls': 50})
            h_new = fix_constraint(res.x.copy(), target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    # Phase 2: Alternative initializations
    alts = [
        fix_constraint(thue_morse_h(n), target),
        fix_constraint(rotational_h(n, (1 + np.sqrt(5)) / 2.0), target),
        fix_constraint(rotational_h(n, 1.0 / np.sqrt(2.0)), target),
        fix_constraint((rng.random(n) >= 0.5).astype(float), target),
    ]
    for h_alt in alts:
        if elapsed() > budget_s - 200:
            break
        for T in [80, 40, 20, 10]:
            if elapsed() > budget_s - 180:
                break
            try:
                res = minimize(sm_obj, h_alt, args=(T, 1e4), method='L-BFGS-B',
                              jac=sm_grad, bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-11})
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                h_alt = h_new
            except Exception:
                pass

    # Phase 3: Many short restarts from best
    n_restarts = 0
    while elapsed() < budget_s - 200:
        n_restarts += 1
        r = n_restarts
        strat = r % 8
        if strat == 0:
            h_init = h_best + rng.normal(0, 0.03, n)
        elif strat == 1:
            h_init = np.roll(h_best, int(rng.integers(-3, 4)))
        elif strat == 2:
            h_init = h_best.copy()
            flips = rng.choice(n, size=int(rng.integers(2, 7)), replace=False)
            h_init[flips] = 1.0 - h_init[flips]
        elif strat == 3:
            h_init = 0.7 * h_best + 0.3 * h_best[::-1]
        elif strat == 4:
            h_init = h_best + rng.normal(0, 0.1, n)
        elif strat == 5:
            h_init = h_best.copy()
            for _ in range(3):
                lo = int(rng.integers(0, n - 20))
                hi = lo + int(rng.integers(5, 25))
                h_init[lo:hi] = 1.0 - h_init[lo:hi]
        elif strat == 6:
            kernel = np.ones(5) / 5.0
            h_init = np.convolve(h_best, kernel, mode='same')
            h_init += rng.normal(0, 0.08, n)
        else:
            h_init = h_best.copy()
            seg = int(rng.integers(10, 40))
            pos = int(rng.integers(0, n - seg))
            h_init[pos:pos+seg] = 1.0 - h_init[pos:pos+seg]

        h_init = fix_constraint(h_init, target)
        T = float(rng.choice([20, 30, 50, 80, 120]))
        try:
            res = minimize(sm_obj, h_init, args=(T, 1e4), method='L-BFGS-B',
                          jac=sm_grad, bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
            h_new = fix_constraint(res.x.copy(), target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Multi-resolution exploration
    for n_hi in [200, 350, 420]:
        if elapsed() > budget_s - 80:
            break
        if n_hi == n:
            continue
        target_hi = n_hi / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, int(n_hi), endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = fix_constraint(h_hi, int(n_hi) / 2.0)
        nh = len(h_hi)
        bnd_hi = [(0.0, 1.0)] * nh
        for T in [60, 30, 15, 8]:
            if elapsed() > budget_s - 40:
                break
            try:
                res = minimize(sm_obj, h_hi, args=(T, 1e4), method='L-BFGS-B',
                              jac=sm_grad, bounds=bnd_hi,
                              options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-11})
                h_new = fix_constraint(res.x.copy(), nh / 2.0)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    h_hi = h_new.copy()
                    n = nh
                    target = nh / 2.0
            except Exception:
                pass

    # Phase 5: Final fine-tuning
    n = len(h_best)
    target = n / 2.0
    bnd = [(0.0, 1.0)] * n
    for T in [5, 3, 2, 1.5]:
        if elapsed() > budget_s - 8:
            break
        try:
            res = minimize(sm_obj, h_best, args=(T, 1e5), method='L-BFGS-B',
                          jac=sm_grad, bounds=bnd,
                          options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-14,
                                   'maxls': 80})
            h_new = fix_constraint(res.x.copy(), target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    # Final micro-perturbations
    if elapsed() < budget_s - 5:
        for _ in range(20):
            if elapsed() > budget_s - 3:
                break
            h_try = h_best + rng.normal(0, 0.01, n)
            h_try = fix_constraint(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
