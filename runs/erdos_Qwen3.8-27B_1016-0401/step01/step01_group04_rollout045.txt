        if s > 0:

    try:
        r = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return np.clip(r.x, 0, 1)
    except:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.93
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(70, 0.5)
    n = len(h0)
    S = n / 2.0
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Continuous SLSQP (moderate)
    for T in [5, 30, 150, 600, 2500]:
        if time.time() > t_end:
            break
        h_new = slsqp_smooth(best_h, n, S, T, maxiter=400)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c
    
    # Phase 2: Binary iterated local search (heavy)
    h_bin = to_binary(best_h, S)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin
    
    # IL: many perturbations + hill climb
    t_il_end = t0 + budget_s * 0.72
    current_bin = h_bin.copy()
    current_c5 = c5_bin
    for it in range(200):
        if time.time() > t_il_end:
            break
        # Random perturbation
        k = rng.randint(1, 6)
        h_p = current_bin.copy()
        for _ in range(k):
            oi = np.where(h_p == 1)[0]
            zi = np.where(h_p == 0)[0]
            if len(oi) == 0 or len(zi) == 0:
                break
            h_p[rng.choice(oi)], h_p[rng.choice(zi)] = 0, 1
        
        h_hc, c5_hc = hill_climb_binary(h_p, min(t_il_end, time.time() + 8),
                                          max_sweeps=40, rng=rng)
        if c5_hc < current_c5:
            current_c5 = c5_hc
            current_bin = h_hc.copy()
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc.copy()
    
    # Also try from scratch with random initializations
    for ri in range(5):
        if time.time() > t_il_end:
            break
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, size=int(S), replace=False)] = 1.0
        h_hc, c5_hc = hill_climb_binary(h_rand, min(t_il_end, time.time() + 5),
                                          max_sweeps=30, rng=rng)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc.copy()
            current_bin = h_hc.copy()
            current_c5 = c5_hc
    
    # Phase 3: SA from best
    t_sa_end = t0 + budget_s * 0.82
    h_sa = best_h.copy()
    c5_sa = compute_c5(h_sa)
    best_sa_c5 = c5_sa
    best_sa_h = h_sa.copy()
    temp = 0.004
    one_idx_all = np.where(h_sa == 1)[0]
    zero_idx_all = np.where(h_sa == 0)[0]
    
    for it in range(80000):
        if time.time() > t_sa_end:
            break
        if it % 5000 == 0 and it > 0:
            temp *= 0.97
        i = int(rng.choice(one_idx_all))
        j = int(rng.choice(zero_idx_all))
        if h_sa[i] != 1 or h_sa[j] != 0:
            continue
        h_sa[i], h_sa[j] = 0, 1
        c5_new = compute_c5(h_sa)
        d = c5_new - c5_sa
        if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-12)):
            c5_sa = c5_new
            if c5_new < best_sa_c5:
                best_sa_c5 = c5_new
                best_sa_h = h_sa.copy()
        else:
            h_sa[i], h_sa[j] = 1, 0
    
    if best_sa_c5 < best_c5:
        best_c5 = best_sa_c5
        best_h = best_sa_h
    
    # Phase 4: Final continuous refinement
    if time.time() < t_end:
        for T in [200, 1000, 5000]:
            if time.time() > t_end:
                break
            h_c = best_h.astype(float) + rng.normal(0, 0.03, n)
            h_c = np.clip(h_c, 0, 1)
            s = h_c.sum()
            if s > 0:
                h_c = h_c * S / s
            h_c = np.clip(h_c, 0, 1)
            h_new = slsqp_smooth(h_c, n, S, T, maxiter=200)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5 = c
                best_h = h_new
    
    # Final binary check
    h_final_bin = to_binary(best_h, S)
    c5_fb = compute_c5(h_final_bin)
