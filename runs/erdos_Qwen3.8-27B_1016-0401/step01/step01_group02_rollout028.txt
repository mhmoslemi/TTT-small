```python
import numpy as np
from scipy.optimizeimport minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        if msk.any() and abs(r) / msk.sum() < 1.0:
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _smooth_obj(h, dx, s):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    v = s * (corr - m)
    v = np.clip(v, -500, 500)
    return (m + np.log(np.exp(v).sum()) / s) * dx

def _smooth_grad(h, dx, s):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    v = s * (corr - m)
    v = np.clip(v, -500, 500)
    e = np.exp(v)
    w = e / e.sum()  # softmax weights
    # grad[j] = dx * sum_k w[k] * (dcorr[k]/dh[j])
    # corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])
    # dcorr[k]/dh[j] = (1-h[j+k-(n-1)]) - h[j+(k-1)]  (with bounds)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        off1 = k - (n - 1)
        off2 = k - 1
        # (1-h[j+off1]) for valid j
        j1 = np.arange(n)
        i1 = j1 + off1
        v1 = (i1 >= 0) & (i1 < n)
        if v1.any():
            grad[v1] += wk * (1.0 - h[i1[v1]])
        # -h[j+off2] for valid j
        i2 = j1 + off2
        v2 = (i2 >= 0) & (i2 < n)
        if v2.any():
            grad[v2] -= wk * h[i2[v2]]
    return grad * dx

def _lbf(h0, n, dx, scales, maxiter, tmax):
    t0 = time.time()
    best = _proj(h0, n)
    bestc = _c5(best)
    for s in scales:
        elapsed = time.time() - t0
        if elapsed > tmax * 0.8:
            break
        h_cur = best.copy()
        try:
            res = minimize(
                lambda x: _smooth_obj(x, dx, s),
                h_cur,
                jac=lambda x: _smooth_grad(x, dx, s),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            h_new = _proj(res.x, n)
            c_new = _c5(h_new)
            if c_new < bestc:
                best, bestc = h_new, c_new
        except Exception:
            pass
    return best, bestc
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        if msk.any():
            step = r / msk.sum()
            if abs(step) < 1.0:
                h[msk] += step
                np.clip(h, 0.0, 1.0, out=h)
    return h

def _smooth_val(corr, s):
    m = corr.max()
    v = np.clip(s * (corr - m), -500, 500)
    return m + np.log(np.exp(v).sum()) / s

def _obj(h, dx, s, mu, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    sv = _smooth_val(corr, s) * dx
    pen = mu * (h.sum() - 0.5 * n) ** 2
    return sv + pen

def _grad(h, dx, s, mu, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    v = np.clip(s * (corr - m), -500, 500)
    e = np.exp(v)
    w = e / e.sum()
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        i1 = j + k - (n - 1)
        v1 = (i1 >= 0) & (i1 < n)
        if v1.any():
            grad[v1] += wk * (1.0 - h[i1[v1]])
        i2 = j + (n - 1 - k)
        v2 = (i2 >= 0) & (i2 < n)
        if v2.any():
            grad[v2] -= wk * h[i2[v2]]
    grad *= dx
    grad += 2.0 * mu * (h.sum() - 0.5 * n) * np.ones(n)
    return grad

def _lbf(h0, n, dx, tmax, mu=1e4):
    t0 = time.time()
    best = _proj(h0, n)
    bestc = _c5(best)
    scales = [20, 50, 100, 200, 400]
    for s in scales:
        if time.time() - t0 > tmax * 0.85:
            break
        hcur = best.copy()
        try:
            res = minimize(
                lambda x: _obj(x, dx, s, mu, n),
                hcur,
                jac=lambda x: _grad(x, dx, s, mu, n),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 10000}
            )
            hnew = _proj(res.x, n)
            cn = _c5(hnew)
            if cn < bestc:
                best, bestc = hnew, cn
        except Exception:
            pass
    return best, bestc

def _peel(h0, n, dx, tmax, nit=50000):
    t0 = time.time()
    h = _proj(h0, n)
    c = _c5(h)
    best = h.copy()
    bestc = c
    step = 0.02
    for i in range(nit):
        if time.time() - t0 > tmax:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        kstar = np.argmax(corr)
        # gradient of corr[kstar] w.r.t. h
        n_ = n
        j = np.arange(n_)
        i1 = j + kstar - (n_ - 1)
        v1 = (i1 >= 0) & (i1 < n_)
        i2 = j + (n_ - 1 - kstar)
        v2 = (i2 >= 0) & (i2 < n_)
        g = np.zeros(n_)
        g[v1] += (1.0 - h[i1[v1]])
        g[v2] -= h[i2[v2]]
        # step in negative gradient
        h2 = h - step * g
        h2 = _proj(h2, n_)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2.copy(), c2
        if c2 < c:
            h, c = h2, c2
        else:
            step = max(step * 0.9, 0.001)
        if i % 5000 == 0:
            step = min(step * 1.1, 0.05)
    return best, bestc

def _rsearch(h0, n, dx, tmax, nit=200000):
    t0 = time.time()
    h = _proj(h0, n)
    c = _c5(h)
    best = h.copy()
    bestc = c
    for i in range(nit):
        if time.time() - t0 > tmax:
            break
        h2 = h.copy()
        # Random pair swap
        idx = np.random.choice(n, 2, replace=False)
        dv = np.random.uniform(0.01, 0.15) * (1 if np.random.random() < 0.5 else -1)
        h2[idx[0]] = np.clip(h2[idx[0]] + dv, 0, 1)
        h2[idx[1]] = np.clip(h2[idx[1]] - dv, 0, 1)
        # Or block perturbation (30% chance)
        if np.random.random() < 0.3:
            blen = np.random.randint(3, 15)
            bstart = np.random.randint(0, n - blen)
            delta = np.random.uniform(-0.1, 0.1, blen)
            delta -= delta.mean()  # maintain sum
            h2[bstart:bstart+blen] += delta
            np.clip(h2, 0, 1, out=h2)
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2.copy(), c2
        if c2 < c or np.random.random() < 0.1:
            h, c = h2, c2
        else:
            h = best.copy()
            c = bestc
    return best, bestc

def _up(h, factor):
    n = len(h)
    n2 = n * factor
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, n2)
    return np.interp(x1, x0, h)
```
