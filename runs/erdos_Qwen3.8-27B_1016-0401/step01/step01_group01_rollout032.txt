from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_grad(h, dx, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_args = (c - c_max) / t
    w = np.exp(exp_args)
    w_sum = w.sum()
    w /= w_sum
    obj = dx * (c_max + t * np.log(w_sum))
    
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        s = m - (n - 1)
        idx1 = np.clip(ks - s, 0, n - 1)
        valid1 = (ks - s >= 0) & (ks - s < n)
        idx2 = np.clip(ks + s, 0, n - 1)
        valid2 = (ks + s >= 0) & (ks + s < n)
        grad[valid1] += wm * (1.0 - h[idx1[valid1]])
        grad[valid2] += wm * (-h[idx2[valid2]])
    return obj, dx * grad

def project_simplex(h, T):
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def pg_solve(h0, dx, T, temps, n_iters, lr_init, mu_init):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for ti, t in enumerate(temps):
        lr = lr_init / (1.0 + 0.1 * ti)
        mu = mu_init * (0.99 ** ti)
        for it in range(n_iters):
            _, g = smooth_obj_grad(h, dx, t)
            h_trial = project_simplex(h - lr * (g + mu * v), T)
            if h_trial is None:
                h_trial = project_simplex(h - lr * g, T)
            v = h_trial - h
            h = h_trial
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def cons_fun(x, T):
    return float(np.sum(x) - T)

def cons_jac(x, n):
    return np.ones(n)

def slsqp_sharp(h0, dx, T, max_iter):
    n = len(h0)
    def obj(x):
        return compute_c5(x, dx)
    def jac(x):

        idx = int(np.argmax(c))

        ks = np.arange(n)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += (1.0 - x[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += (-x[ks[v2] + s])
        return dx * grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: cons_fun(x, T), 'jac': lambda x: cons_jac(x, n)}]
    try:

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_grad(h, dx, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_args = (c - c_max) / t
    w = np.exp(exp_args)
    w_sum = w.sum()
    w /= w_sum
    obj = dx * (c_max + t * np.log(w_sum))
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        v2 = (ks + s >= 0) & (ks + s < n)
        if np.any(v1):
            grad[v1] += wm * (1.0 - h[ks[v1] - s])
        if np.any(v2):
            grad[v2] += wm * (-h[ks[v2] + s])
    return obj, dx * grad

def project_simplex(h, T):
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (T / s)
    else:
        h = np.full_like(h, T / len(h))
    return np.clip(h, 0.0, 1.0)

    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()

            if c5 < best_c5:

                best_h = h.copy()

from scipy.optimize import minimize

_G_T = 35.0
_G_N = 70
_G_DX = 2.0 / 70
_G_MOOC = 0.01

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_grad(h, dx, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_args = (c - c_max) / t
    w = np.exp(exp_args)
    w_sum = w.sum()
    w /= w_sum
    obj = dx * (c_max + t * np.log(w_sum))
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        v2 = (ks + s >= 0) & (ks + s < n)
        if np.any(v1):
            grad[v1] += wm * (1.0 - h[ks[v1] - s])
        if np.any(v2):
            grad[v2] += wm * (-h[ks[v2] + s])
    return obj, dx * grad

def project_simplex(h, T):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def normalize(h, T):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (T / s)
    else:
        h = np.full_like(h, T / len(h))
    return np.clip(h, 0.0, 1.0)

            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        v = np.zeros(len(h0))
    return best_h, best_c5

    n = len(h)

                trial[j] += d
