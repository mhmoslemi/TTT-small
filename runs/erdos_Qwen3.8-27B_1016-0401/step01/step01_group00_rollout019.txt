    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            frac = h < 1.0 - 1e-15
        else:
            frac = h > 1e-15
        cnt = np.sum(frac)
        if cnt == 0:
            break
        h[frac] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def focused_obj(h, n, idx, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    tv = corr[idx]
    mc = np.max(tv)
    if T < 1e-8:
        return mc
    sh = T * (tv - mc)
    return mc + np.log(np.sum(np.exp(sh))) / T

def focused_grad(h, n, idx, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    tv = corr[idx]
    mc = np.max(tv)
    if T < 1e-8:
        w = np.zeros(len(idx))
        mask = tv >= mc - 1e-12
        w[mask] = 1.0 / np.sum(mask)
    else:
        sh = T * (tv - mc)
        e = np.exp(sh)
        w = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-14:
            continue
        o = idx[ii] - center
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] += -wi * h[o:]
        else:
            p = -o
            grad[p:] += wi * (1.0 - h[:n-p])
            grad[:n-p] += -wi * h[p:]
    return grad * dx

def subgradient_step(h, n, alpha, top_k):
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / n
    mc = np.max(corr)
    active = np.where(corr >= mc - 1e-10)[0]
    if len(active) > top_k:
        active = np.argsort(corr)[-top_k:]
    center = n - 1
    grad = np.zeros(n)
    for o_idx in active:
        o = o_idx - center
        if o == 0:
            grad += (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += (1.0 - h[:n-o])
            grad[:n-o] += -h[o:]
        else:
            p = -o
            grad[p:] += (1.0 - h[:n-p])
            grad[:n-p] += -h[p:]
    grad /= len(active)
    grad *= dx
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_constraint(h_new, n / 2.0)

def do_slsqp(h, n, idx, T, target, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return focused_obj(x, n, idx, T)
    def jac(x):
        return focused_grad(x, n, idx, T)
    res = minimize(obj, h, method='SLSQP', jac=jac,
                  bounds=bounds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.fun < 200:
        return fix_constraint(res.x.copy(), target)
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Subgradient exploration (fast, many steps)
    alpha0 = 0.02
    total_steps = 80000
    for s in range(total_steps):
        if elapsed() > budget_s - 300:
            break
        alpha = alpha0 / (1 + s * 0.0001)
        top_k = 5 + (s // 20000) * 3
        h_curr = subgradient_step(h_best, n, alpha, min(top_k, 2*n-1))
        c5 = compute_c5(h_curr)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_curr

    # Phase 2: Focused SLSQP - progressively widen lag set
    for rnd in range(35):
        if elapsed() > budget_s - 250:
            break
        corr = get_corr(h_best)
        M = 2 + min(rnd, 28)
        top_M = np.argsort(corr)[-M:]
        T = max(5.0, 80.0 - rnd * 2.0)
        h_new = do_slsqp(h_best, n, top_M, T, target, 1200)
        if h_new is not None:
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 3: Perturbed restarts with focused optimization
    for attempt in range(40):
        if elapsed() > budget_s - 180:
            break
        scale = 0.05 + 0.25 * (attempt % 6) / 6.0
        h_try = h_best + rng.normal(0, scale, n)
        h_try = fix_constraint(h_try, target)
        corr = get_corr(h_try)
        M = 4 + (attempt % 8)
        top_M = np.argsort(corr)[-M:]
        T = 10 + 30 * (attempt % 5)
        h_new = do_slsqp(h_try, n, top_M, T, target, 600)
        if h_new is not None:
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Full soft-max polishing
    all_idx = np.arange(2 * n - 1)
    for T in [5, 10, 25, 60, 150, 300]:
        if elapsed() > budget_s - 100:
            break
        h_new = do_slsqp(h_best, n, all_idx, T, target, 2000)
        if h_new is not None:
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 5: Try n=70 binary and upsample
    if elapsed() < budget_s - 80:
        n70 = 70
        t70 = n70 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x70 = np.arange(n70) * (2.0 / n70)
        h70 = np.interp(x70, x_old, h_best)
        h70_bin = (h70 >= 0.5).astype(float)
        h70_bin = fix_constraint(h70_bin, t70)
        c5_70 = compute_c5(h70_bin)
        # Local search on binary
        for it in range(800):
            if elapsed() > budget_s - 60:
                break
            improved = False
            for j in range(n70):
                if h70_bin[j] > 0.5:
                    continue
                h70_bin[j] = 1.0
                if abs(h70_bin.sum() - t70) > 1e-8:
                    h70_bin[j] = 0.0
                    continue
                c5_try = compute_c5(h70_bin)
                if c5_try < c5_70 - 1e-9:
                    c5_70 = c5_try
                    improved = True
                    break
                h70_bin[j] = 0.0
            if not improved:
                break
        h_up = np.interp(x_old, x70, h70_bin)
        h_up = fix_constraint(h_up, target)
        c5 = compute_c5(h_up)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_up
            h_new = do_slsqp(h_best, n, all_idx, 20, target, 1500)
            if h_new is not None:
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Phase 6: Final focused refinement
    if elapsed() < budget_s - 10:
        corr = get_corr(h_best)
        top_5 = np.argsort(corr)[-5:]
        for T in [3, 8, 15]:
            if elapsed() > budget_s - 5:
                break
            h_new = do_slsqp(h_best, n, top_5, T, target, 2000)
            if h_new is not None:
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
