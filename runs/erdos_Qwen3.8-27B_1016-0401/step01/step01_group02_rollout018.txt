import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - target
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = target - h.sum()
    if abs(r) > 1e-12:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _subgrad_step(h, n, dx, alpha):
    """One subgradient step"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    # Subgradient: dC_k/dh[j] = dx * [(1-h[j+lag]) - h[j-lag]] with bounds
    g = np.zeros(n)
    if lag >= 0:
        # h[j]*(1-h[j+lag]): d/dh[j] = (1-h[j+lag]) for j+lag < n
        g[:n-lag] += (1.0 - h[lag:n])  # (1-h[j+lag]) for j = 0..n-1-lag
        # h[j-lag]*(1-h[j]): d/dh[j] = -h[j-lag] for j >= lag
        g[lag:] -= h[:n-lag]  # -h[j-lag] for j = lag..n-1
    else:
        lag2 = -lag
        g[lag2:] += (1.0 - h[:n-lag2])  # (1-h[j+lag]) where j+lag = j-lag2
        g[:n-lag2] -= h[lag2:]  # -h[j-lag] where j-lag = j+lag2
    return g * dx

def _nesterov_subgrad(h0, n, tmax, lr_init=0.01):
    """Nesterov accelerated subgradient with projection"""
    t0 = time.time()
    h = h0.copy()
    y = h0.copy()
    best = h0.copy()
    bestc = _c5(best)
    dx = 2.0 / n
    m = 0.0
    m_best = max(1, int(tmax / 0.01))
    
    it = 0
    while time.time() - t0 < tmax:
        m += 1
        t = m / m_best
        alpha = lr_init / (1.0 + m * 0.001)
        
        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * m * m)) / (2.0 * m + 1.0)
        
        g = _subgrad_step(y, n, dx, 1.0)
        y = h - alpha * g

        h = _proj(h_new, n)
        
        c = _c5(h)
        if c < bestc:

    return best, bestc

    m = corr.max()
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * (2.0 / n)

def _slsqp(h0, n, sc, maxit, ftol=1e-14):
    dx = 2.0 / n
    def _obj(x):
        return _smooth_obj(x, n, sc)
    def _con(x):
        return x.sum() - 0.5 * n
    def _jac(x):
        return np.ones(n)
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con, 'jac': _jac}],
                       options={'maxiter': maxit, 'ftol': ftol})
        return _proj(res.x, n)
    except Exception:
        return _proj(h0, n)

def _sa_ls(h0, n, tmax, rng, k_anneal=3):
    t0 = time.time()
    target = 0.5 * n
    h = h0.copy()
    curc = _c5(h)
    best = h0.copy()
    bestc = curc
    t_total = max(1.0, tmax)
    it = 0
    while time.time() - t0 < tmax:
        it += 1
        progress = min(1.0, 0.1 + 0.9 * it / (t_total * 800))
        temp = 0.06 * (1.0 - progress) ** k_anneal + 0.001
        r = rng.random()
        if r < 0.45:
            corr = _corr(h)
            k_star = np.argmax(corr)
            lag = k_star - (n - 1)
            if lag == 0:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(max(1, n - 2))) % n
            elif lag > 0 and lag < n:
                valid = np.arange(n - lag)
                contrib = h[valid] * (1.0 - h[valid + lag])
                tot = contrib.sum()
                if tot > 1e-10:
                    p = contrib / tot
                    i = rng.choice(valid, p=p)
                else:
                    i = rng.randint(n - lag)
                j = i + lag
            elif lag < 0 and lag > -n:
                lag2 = -lag
                valid = np.arange(n - lag2)
                contrib = h[valid] * (1.0 - h[valid + lag2])
                tot = contrib.sum()
                if tot > 1e-10:
                    p = contrib / tot
                    i = rng.choice(valid, p=p)
                else:
                    i = rng.randint(n - lag2)
                j = i + lag2
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(max(1, n - 2))) % n
            amt = rng.uniform(0.08, 0.45)
            h2 = h.copy()
            h2[i] = max(0.0, h2[i] - amt)
            h2[j] = min(1.0, h2[j] + amt)
            ds = h2.sum() - target
            if abs(ds) > 1e-12:
                h2 -= ds / n
                np.clip(h2, 0.0, 1.0, out=h2)
        elif r < 0.75:
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            amt = rng.uniform(0.05, 0.35)
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + sgn * amt, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - sgn * amt, 0.0, 1.0)
            ds = h2.sum() - target
            if abs(ds) > 1e-12:
                h2 -= ds / n
                np.clip(h2, 0.0, 1.0, out=h2)
        else:
            blen = int(rng.randint(2, min(12, n // 4)))
            start = int(rng.randint(0, n - blen))
            shift = rng.uniform(-0.12, 0.12)
            h2 = h.copy()
            h2[start:start + blen] += shift
            np.clip(h2, 0.0, 1.0, out=h2)
            ds = h2.sum() - target
            if abs(ds) > 1e-12:
                h2 -= ds / n
                np.clip(h2, 0.0, 1.0, out=h2)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2.copy(), c2
            h, curc = h2, c2
        elif c2 - curc < temp * rng.standard_normal():
            h, curc = h2, c2
    return best, bestc

def _kronecker_init(n, seed=0):
    rng_l = np.random.RandomState(seed)
    alpha = rng_l.uniform(0.3, 0.7)
    h = ((np.arange(n, dtype=float) * alpha) % 1.0) < 0.5
    h = h.astype(float)
    ones = h.sum()
    if ones != n // 2:
        idx = np.where(h == ((ones > n // 2) * 1.0))[0]
        flip = len(idx) - abs(n // 2 - ones)
        if flip > 0:
            h[idx[:flip]] = 1.0 - h[idx[:flip]]
    return _proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = _kronecker_init(180, seed)

    n = len(h0)
    bh = h0.copy()
    bc = _c5(bh)

    # Try a few SLSQP scales at initial n
    for sc in [40.0, 80.0]:
        rem = budget_s - (time.time() - ts)
        if rem < 30:
            break
        h1 = _slsqp(bh, n, sc, 2500)
        c1 = _c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Local search phase 1
    rem = budget_s - (time.time() - ts)
    if rem > 30:
        hl, cl = _sa_ls(bh, n, min(50.0, rem * 0.25), rng, k_anneal=3)
        if cl < bc:
            bh, bc = hl, cl

    # Upscale to 2n
    n2 = 2 * n
    rem = budget_s - (time.time() - ts)
    if n2 <= 720 and rem > 80:
        # Upscale with interpolation-like approach
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h_interp = np.interp(x_new, x_old, bh, period=2.0)
        h2 = _proj(h_interp + rng.uniform(-0.04, 0.04, n2), n2)
        # SLSQP
        for sc in [60.0, 120.0]:
            rem = budget_s - (time.time() - ts)
            if rem < 40:
                break
            hh = _slsqp(h2, n2, sc, 2000)
            cc = _c5(hh)
            if cc < _c5(h2):
                h2 = hh
        c2 = _c5(h2)
        rem = budget_s - (time.time() - ts)
        if rem > 50:
            hl, cl = _sa_ls(h2, n2, min(70.0, rem * 0.35), rng, k_anneal=4)
            if cl < c2:
                h2, c2 = hl, cl
        if c2 < bc:
            bh, bc, n = h2, c2, n2

    # Upscale to 4n (from original)
    n3 = 2 * n
    rem = budget_s - (time.time() - ts)
    if n3 <= 720 and rem > 100:
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n3, endpoint=False) + 1.0 / n3
        h_interp = np.interp(x_new, x_old, bh, period=2.0)
        h3 = _proj(h_interp + rng.uniform(-0.03, 0.03, n3), n3)
        for sc in [100.0, 200.0]:
            rem = budget_s - (time.time() - ts)
            if rem < 40:
                break
            hh = _slsqp(h3, n3, sc, 2000)
            cc = _c5(hh)
            if cc < _c5(h3):
                h3 = hh
        c3 = _c5(h3)
        rem = budget_s - (time.time() - ts)
        if rem > 40:
            hl, cl = _sa_ls(h3, n3, min(60.0, rem * 0.35), rng, k_anneal=5)
            if cl < c3:
                h3, c3 = hl, cl
        if c3 < bc:
            bh, bc, n = h3, c3, n3

    # Final long local search
    rem = budget_s - (time.time() - ts)
    if rem > 15:
        hl, cl = _sa_ls(bh, n, min(rem * 0.8, 250.0), rng, k_anneal=3)
        if cl < bc:
            bh, bc = hl, cl

    # Tiny final polish
    rem = budget_s - (time.time() - ts)
    if rem > 10:
        hl, cl = _sa_ls(bh, n, min(rem * 0.5, 60.0), rng, k_anneal=5)
        if cl < bc:
            bh, bc = hl, cl

    fc = _c5(bh)
    return (bh, fc, len(bh))
