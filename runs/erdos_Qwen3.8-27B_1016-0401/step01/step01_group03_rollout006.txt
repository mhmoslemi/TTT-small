def compute_c5(h, dx):

    return np.max(corr) * dx

def to_binary(h, target_ones):
    """Convert to binary with exactly target_ones ones."""
    n = len(h)
    b = (h > 0.5).astype(float)
    oc = int(b.sum())
    if oc > target_ones:
        idxs = np.where(b == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - target_ones]:
            b[i] = 0.0
    elif oc < target_ones:

        np.random.shuffle(idxs)
        for i in idxs[:target_ones - oc]:

    return b

def fast_local_search(h, dx, t_limit, max_rounds=200):
    """Greedy binary local search: try all swaps, accept best improvement."""
    n = len(h)
    target = n // 2
    h = to_binary(h, target).copy()
    c5 = compute_c5(h, dx)
    
    for rnd in range(max_rounds):

            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_a, best_b = -1, -1
        for a in ones:

        if not improved:

def project_feasible_cont(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

        h_r = project_binary(h_r, ts_use)
        h_rr, c5_r = iterated_local_search(h_r, dx_use, min(time.time() + 15, t3), seed=seed + 1000 + rs, k_perturb=4)
        update(h_rr, c5_r, n_use)

    # Phase 4: Continuous refinement (10% time)
    t4 = start_time + budget_s * 0.97
    n_use = best_n
    dx_use = 2.0 / n_use
    ts_use = n_use / 2.0
    h_cont = project_feasible_cont(best_h, ts_use)
    h_cont, c5_cont = continuous_refine(h_cont, dx_use, ts_use, t4)
    if c5_cont < best_c5 - 1e-12:
        best_h, best_c5, best_n = h_cont, c5_cont, n_use

    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    ts_f = best_n / 2.0
    best_h = project_feasible_cont(best_h, ts_f)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, best_n
