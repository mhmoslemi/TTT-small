```python
def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def swap_delta(h, i, j, corr):
    """Compute new corr after swapping h[i] and h[j] (h[i]=1, h[j]=0)"""
    n = len(h)
    # After swap: h[i]=0, h[j]=1
    # c(k) = sum valid l: h[l] * (1-h[l+k])
    # Changes at l=i (h goes 1->0) and l=j (h goes 0->1)
    # Also changes where l+k = i or l+k = j
    # i.e., l = i-k or l = j-k
    
    new_corr = corr.copy()
    for k in range(-(n-1), n):
        if k == 0:
            continue
        # Affected indices: l where l in {i, j} or l+k in {i, j}
        affected = set()
        affected.add(i)
        affected.add(j)
        li = i - k  # where l+k = i
        lj = j - k  # where l+k = j
        if 0 <= li < n:
            affected.add(li)
        if 0 <= lj < n:
            affected.add(lj)
        
        old_sum = 0
        new_sum = 0
        for l in affected:
            if not (0 <= l < n and 0 <= l+k < n):
                continue
            old_h_l = h[l]
            old_h_lk = h[l+k]
            new_h_l = h[l]
            new_h_lk = h[l+k]
            if l == i:
                new_h_l = 0
            elif l == j:
                new_h_l = 1
            if l+k == i:
                new_h_lk = 0
            elif l+k == j:
                new_h_lk = 1
            old_sum += old_h_l * (1 - old_h_lk)
            new_sum += new_h_l * (1 - new_h_lk)
        
        # corr index: k + (n-1)
        idx = k + (n-1)
        new_corr[idx] += new_sum - old_sum
    return new_corr
```
