import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _normalize(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h = h * target_sum / s
    else:
        h = np.full_like(h, target_sum / len(h))
    return np.clip(h, 0.0, 1.0)

def _f_1d(x, a_m, b_m, quad_idx):
    vals = a_m * x + b_m
    if quad_idx >= 0:
        vals[quad_idx] -= x * x
    return float(np.max(vals))

def _cd_pass(h, n):
    n_lags = 2 * n - 1
    quad_idx = n - 1
    c_full = np.correlate(h, 1.0 - h, mode='full')
    s_vals = np.arange(-(n - 1), n)
    for j in range(n):
        h_j = h[j]
        a_m = np.zeros(n_lags)
        b_m = c_full.copy()
        for m in range(n_lags):
            s = s_vals[m]
            js = j + s
            jm = j - s
            if 0 <= js < n:
                t1 = h_j * (1.0 - h[js])
                b_m[m] -= t1
                if s == 0:
                    a_m[m] += 1.0
                else:
                    a_m[m] += (1.0 - h[js])
            if s != 0 and 0 <= jm < n:
                t2 = h[jm] * (1.0 - h_j)
                b_m[m] -= t2
                a_m[m] += (-h[jm])
        a_lo, a_hi = 0.0, 1.0
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        invgr = 1.0 / gr
        x1 = a_lo + (1.0 - invgr) * (a_hi - a_lo)
        x2 = a_lo + invgr * (a_hi - a_lo)
        f1 = _f_1d(x1, a_m, b_m, quad_idx)
        f2 = _f_1d(x2, a_m, b_m, quad_idx)
        for _ in range(50):
            if f1 < f2:
                a_hi = x2
                x2 = x1
                f2 = f1
                x1 = a_lo + (1.0 - invgr) * (a_hi - a_lo)
                f1 = _f_1d(x1, a_m, b_m, quad_idx)

                a_lo = x1
                x1 = x2
                f1 = f2
                x2 = a_lo + invgr * (a_hi - a_lo)
                f2 = _f_1d(x2, a_m, b_m, quad_idx)
        h[j] = max(0.0, min(1.0, (a_lo + a_hi) / 2.0))
    return h

def coord_descent(h, n, max_passes=500, tol=1e-13):
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    prev = compute_c5(h, dx)
    for _ in range(max_passes):
        h = _cd_pass(h, n)
        h = _normalize(h, target)
        cur = compute_c5(h, dx)
        if prev - cur < tol:
            break
        prev = cur
    return h

_SMOOTH_T = 0.01
_CUR_DX = 0.02
_CONST_T = 35.0
_CONST_N = 70

def _smooth_obj(x):
    c = np.correlate(x, 1.0 - x, mode='full') * _CUR_DX
    t = _SMOOTH_T
    m = np.max(c)
    return m + t * np.log(np.sum(np.exp((c - m) / t)))

def _smooth_grad(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * _CUR_DX
    t = _SMOOTH_T
    m = np.max(c)
    w = np.exp((c - m) / t)
    w /= w.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        s = m_idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - x[j + s])
            if 0 <= j - s < n:
                g += (-x[j - s])
            grad[j] += wm * g
    return _CUR_DX * grad

def _cons_eq(x):
    return float(np.sum(x) - _CONST_T)

def _cons_eq_jac(x):
    return np.ones(_CONST_N)

def _slsqp(h_init, n, t, max_iter):
    global _SMOOTH_T, _CUR_DX, _CONST_T, _CONST_N
    _SMOOTH_T = t
    _CUR_DX = 2.0 / n
    _CONST_T = n / 2.0
    _CONST_N = n
    h_try = _normalize(h_init, n / 2.0).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        r = minimize(_smooth_obj, h_try, method='SLSQP', jac=_smooth_grad,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': 1e-14})
        return _normalize(r.x, n / 2.0)
    except Exception:
        return h_try
