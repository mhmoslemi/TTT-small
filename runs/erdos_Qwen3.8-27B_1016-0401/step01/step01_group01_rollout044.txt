    """Project x onto {0 ≤ x ≤ 1, sum(x) = T}."""
    n = len(x)
    # If already feasible, return
    x = np.clip(x, 0, 1)
    s = x.sum()
    if abs(s - T) < 1e-12:
        return x
    # Bisection on lambda: x_i = clip(x_i + lambda, 0, 1)
    # Want sum(clip(x_i + lambda, 0, 1)) = T
    lo, hi = -x.max() - 1, -x.min() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        x_try = np.clip(x + mid, 0, 1)
        if x_try.sum() > T:
            lo = mid
        else:
            hi = mid

    # Fine-tune to get exact sum
    d = T - x_proj.sum()
    if abs(d) > 1e-14:

import numpy as np
from scipy.optimize import minimize

_CUR_DX = 1.0
_CONST_T = 35.0
_CONST_N = 70
_TEMP = 0.02

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _obj_smooth(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(c)
    t = _TEMP
    return _CUR_DX * (c_max + t * np.log(np.sum(np.exp((c - c_max) / t))))

def _jac_smooth(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(c)
    t = _TEMP
    w = np.exp((c - c_max) / t)
    w /= np.sum(w)
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v1] += w[m] * (1.0 - x[ks[v1] - s])
        grad[v2] += w[m] * (-x[ks[v2] + s])
    return _CUR_DX * grad

def _obj_sharp(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(c) * _CUR_DX)

def _jac_sharp(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    grad = np.zeros(n)
    ks = np.arange(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v1] += (1.0 - x[ks[v1] - s])
    grad[v2] += (-x[ks[v2] + s])
    return _CUR_DX * grad

def _cons_fun(x):
    return float(np.sum(x) - _CONST_T)

def _cons_jac(x):
    return np.ones(_CONST_N)

def _normalize(h, T):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 1e-10:
        h = h * T / s
    else:
        h = np.full_like(h, T / len(h))
    h = np.clip(h, 0, 1)
    d = T - h.sum()
    if abs(d) > 1e-12:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += d / free.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0, 1)
    return h

def _project_parent(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_parent)
    return _normalize(h_new, n_new / 2.0)

def _complement_flip(h):
    return 1.0 - h[::-1].copy()

def _perturb_structured(h, rng, n_modes=6, scale=0.35):
    n = len(h)
    x = np.linspace(0, 2, n, endpoint=False)
    h_new = h.copy()
    for k in range(1, n_modes + 1):
        a1 = rng.uniform(-scale, scale) / k
        a2 = rng.uniform(-scale, scale) / k
        h_new += a1 * np.sin(np.pi * k * x)
        h_new += a2 * np.cos(np.pi * k * x)
    return _normalize(h_new, n / 2.0)

def _slsqp_run(h_init, n, max_iter, ftol, temp):
    global _CUR_DX, _CONST_T, _CONST_N, _TEMP
    _CUR_DX = 2.0 / n
    _CONST_T = n / 2.0
    _CONST_N = n
    _TEMP = temp
    T = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_try = _normalize(h_init.copy(), T)
    if temp > 0:
        f, g = _obj_smooth, _jac_smooth
    else:
        f, g = _obj_sharp, _jac_sharp
    try:
        r = minimize(f, h_try, method='SLSQP', jac=g,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        h_out = _normalize(r.x, T)
        return h_out, compute_c5(h_out, _CUR_DX)
    except Exception:
        return h_try, compute_c5(h_try, _CUR_DX)

def _multi_phase(h_init, n, max_iter_per_phase):
    phases = [
        (0.10, max_iter_per_phase, 1e-12),
        (0.03, max_iter_per_phase, 1e-13),
        (0.005, max_iter_per_phase, 1e-14),
        (0.0, max_iter_per_phase, 1e-14),
    ]
    h_cur = _normalize(h_init, n / 2.0)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur, 2.0 / n)
    for temp, iters, ftol in phases:
        h_cur, c5 = _slsqp_run(h_cur, n, iters, ftol, temp)
        if c5 < best_c5:
            best_c5, best_h = c5, h_cur.copy()
    return best_h, best_c5
