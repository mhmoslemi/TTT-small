    sum_exp = np.sum(exp_c)
    return dx * (max_c + np.log(sum_exp) / beta)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    """Project h to [0,1]^n with sum = target using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    # For numerical stability, only use top values
    exp_c = np.exp(np.clip(beta * shifted, -50, 0))
    sum_exp = np.sum(exp_c)
    val = max_c + np.log(sum_exp) / beta
    return dx * val

def smooth_gradient(h, dx, beta):

    corr = np.correlate(h, 1.0 - h, mode='full')

    grad = np.zeros(n)

        if wi < 1e-12:

            idx1 = np.arange(0, n - m)
            grad[idx1] += wi * (1.0 - h[idx1 + m])
            idx2 = np.arange(m, n)
            grad[idx2] -= wi * h[idx2 - m]

            idx2 = np.arange(0, n - p)
            grad[idx2] -= wi * h[idx2 + p]

        return np.sum(x) - target_sum
