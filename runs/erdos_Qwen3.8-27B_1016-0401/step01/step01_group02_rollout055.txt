import time

    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        if msk.sum() > 0:
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sm_idf(h, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    return (m + np.log(e.sum()) / sc) * (2.0 / n)

def _sm_grad(h, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    w = e / e.sum()
    A = 1.0 - h
    B = h
    jl = np.arange(n)[:, None] - np.arange(-(n-1), n)[None, :]
    jj = np.arange(n)[:, None] + np.arange(-(n-1), n)[None, :]
    A_m = np.where((jl >= 0) & (jl < n), A[jl], 0.0)
    B_m = np.where((jj >= 0) & (jj < n), B[jj], 0.0)
    return np.dot(A_m - B_m, w) * (2.0 / n)

    def obj(x):
        return _sm_idf(x, sc)
    def jacob(x):
        return _sm_grad(x, sc)
    def ceq(x):
        return x.sum() - 0.5 * n
    def ceqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': ceqj}]
    bds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=jacob, method='SLSQP',
                  bounds=bds, constraints=cons,
                  options={'maxiter': maxiter, 'ftol': 1e-15})
    return _proj(res.x, n)

    h = h0.copy()
    best = h.copy()
    bestc = _c5(best)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        h = np.clip(h, 0.0, 1.0)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        cnt = msk.sum()
        if cnt > 0:
            h[msk] += r / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sm_val(h, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    return (m + np.log(e.sum()) / sc) * (2.0 / n)

def _sm_grad(h, sc):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    w = e / e.sum()
    j = np.arange(n)
    l = np.arange(-(n - 1), n)
    jl = j[:, None] - l[None, :]
    jj = j[:, None] + l[None, :]
    Am = np.where((jl >= 0) & (jl < n), 1.0 - h[jl], 0.0)
    Bm = np.where((jj >= 0) & (jj < n), h[jj], 0.0)
    return (Am - Bm).dot(w) * (2.0 / n)

def _slsqp_step(h, n, sc, mi=300):
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]

                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _proj(r.x, n)
    except Exception:
        return h
