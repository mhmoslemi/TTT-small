import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_DX = 0.0
_G_SCALE = 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sopt(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip(sc * (corr - m), -400.0, 400.0)
    return (m + np.log(np.exp(shifted).sum()) / sc) * dx

def _f_slq(x):
    return _sopt(np.clip(x, 0.0, 1.0), _G_DX, _G_SCALE)

def _ceq(x):
    return x.sum() - 0.5 * _G_N

def _ceqj(x):
    return np.ones(len(x))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.any():
            h[free] -= d / free.sum()
        else:
            h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _slsqp(h0, n, dx, sc, mi, ftol):
    global _G_N, _G_DX, _G_SCALE
    _G_N, _G_DX, _G_SCALE = n, dx, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_f_slq, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': mi, 'ftol': ftol, 'maxls': 40})
        return _proj(r.x, n)
