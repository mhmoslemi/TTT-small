_G_T = 0.1
_G_MU = 1.0
_G_TARGET = 35.0

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_grad(h, dx, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    w = np.exp((c - c_max) / t)
    w_sum = np.sum(w)
    w = w / w_sum
    obj = dx * (c_max + t * np.log(w_sum))
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += wm * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += wm * (-h[ks[v2] + s])
    return obj, dx * grad
