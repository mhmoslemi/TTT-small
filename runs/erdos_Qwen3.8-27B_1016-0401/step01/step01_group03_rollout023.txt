    topk_sum = np.sum(sorted_c[:k])

import numpy as np
from scipy.optimize import minimize_scalar
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)


def eval_pair_t(t, h, i, j, dx):
    h2 = h.copy()
    s = h[i] + h[j]
    h2[i] = s - t
    h2[j] = t
    h2 = np.clip(h2, 0.0, 1.0)
    return compute_c5(h2, dx)


def pair_optimize(h, dx, t_limit, max_passes=30):
    n = len(h)
    cur_c5 = compute_c5(h, dx)
    for _ in range(max_passes):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > t_limit:
                return h, cur_c5
            for j in range(i + 1, n):
                if time.time() > t_limit:
                    return h, cur_c5
                s = h[i] + h[j]
                lo_t = max(0.0, s - 1.0)
                hi_t = min(1.0, s)
                if hi_t - lo_t < 1e-10:
                    continue
                try:
                    res = minimize_scalar(eval_pair_t, args=(h, i, j, dx),
                                          bounds=(lo_t, hi_t), method='bounded',
                                          options={'xatol': 1e-10, 'maxiter': 25})
                except Exception:
                    continue
                if res.fun < cur_c5 - 1e-13:
                    h[i] = s - res.x
                    h[j] = res.x
                    h = np.clip(h, 0.0, 1.0)
                    cur_c5 = compute_c5(h, dx)
                    improved = True
        if not improved:
            break
    return h, cur_c5


def hill_climb_bin(h, dx, t_limit):
    n = len(h)
    cur_c5 = compute_c5(h, dx)
    improved = True
    while improved and time.time() < t_limit:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            if time.time() > t_limit:
                return h, cur_c5
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h, dx)
                if new_c5 < cur_c5 - 1e-14:
                    cur_c5 = new_c5
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
    return h, cur_c5


def sa_binary(n, ones_cnt, dx, t_limit, n_restarts=3, seed=0):

    return h_b
