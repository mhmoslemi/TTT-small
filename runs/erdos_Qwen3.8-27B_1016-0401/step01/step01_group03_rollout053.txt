        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary(h, target):
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    if s > target:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for i in ones[:s - target]:
            h[i] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target - s]:
            h[i] = 1.0
    return h

def binary_local_search(h, dx, t_limit, max_passes=500):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(max_passes):
        if time.time() > t_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-15
        best_oi = -1
        best_zi = -1
        best_c5_trial = best_c5
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                c5_new = compute_c5(h, dx)
                delta = best_c5 - c5_new
                if delta > best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
                    best_c5_trial = c5_new
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
        if best_oi >= 0:
            h[best_oi] = 0.0
            h[best_zi] = 1.0
            best_c5 = best_c5_trial
        else:
            break
    return h, best_c5

def binary_sa(h_init, dx, t_limit, n_iter=50000):
    n = len(h_init)
    target = n // 2
    h = make_binary(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    T = 0.03
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        T = max(1e-7, T * 0.9997)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        oi = ones[np.random.randint(len(ones))]
        zi = zeros[np.random.randint(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        c5_new = compute_c5(h, dx)
        delta = c5_new - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best_c5

def p_norm_obj_grad(h, dx, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    abs_c = np.abs(corr)
    sp = np.sum(abs_c ** p)
    if sp < 1e-30:
        return 0.0, np.zeros(n)
    obj = sp ** (1.0 / p) * dx
    if p == 2:
        w = corr / np.sqrt(sp)
    else:
        w = sp ** (1.0 / p - 1.0) * corr * abs_c ** (p - 2)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-20:
            continue
        m = k - (n - 1)
        if m >= 0:
            idx1 = np.arange(0, n - m)
            grad[idx1] += wk * (1.0 - h[idx1 + m])
            idx2 = np.arange(m, n)
            grad[idx2] -= wk * h[idx2 - m]
        else:
            ps = -m
            idx1 = np.arange(ps, n)
            grad[idx1] += wk * (1.0 - h[idx1 - ps])
            idx2 = np.arange(0, n - ps)
            grad[idx2] -= wk * h[idx2 + ps]
    return obj, grad * dx

def continuous_optimize(h0, dx, target, t_limit, p_list=None):
    if p_list is None:
        p_list = [8, 32, 128, 512, 2048]
    h = project_unit_sum(h0, target).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for p in p_list:
        if time.time() > t_limit:
            break
        lr = 2.0
        for it in range(100000):
            if time.time() > t_limit:
                break
            obj, grad = p_norm_obj_grad(h, dx, p)
            h_new = project_unit_sum(h - lr * grad, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                lr = min(lr * 1.08, 80.0)
            else:
                lr *= 0.92
            h = h_new
            if lr < 1e-9:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    target_init = n_init / 2.0
    best_h = project_unit_sum(h_init, target_init)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    # Phase 1: Continuous optimization at initial n
    t1 = start_time + budget_s * 0.22
    h1, c5_1 = continuous_optimize(best_h, dx_init, target_init, t1, p_list=[8, 32, 128, 512])
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Binary local search + SA at multiple n values
    t2 = start_time + budget_s * 0.72
    n_list = [46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 92, 96]
    x_ref = np.linspace(0.0, 2.0, best_n)
    for n_try in n_list:
        if time.time() > t2:
            break
        dx_t = 2.0 / n_try
        target_t = n_try // 2
        remaining = t2 - time.time()
        if remaining < 3:
            break
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_ref, best_h)
        h_bin = make_binary(h_interp, target_t)
        # Local search
        t_ls = min(t2, time.time() + max(3, remaining * 0.35))
        h_ls, c5_ls = binary_local_search(h_bin, dx_t, t_ls, max_passes=400)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n_try
            x_ref = np.linspace(0.0, 2.0, best_n)
        # SA (brief)
        if time.time() < t2 - 8:
            t_sa = min(t2, time.time() + min(12, remaining * 0.15))
            h_sa, c5_sa = binary_sa(h_ls, dx_t, t_sa, n_iter=15000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_try
                x_ref = np.linspace(0.0, 2.0, best_n)
            # Quick LS after SA
            if time.time() < t2 - 3:
                h_ls2, c5_ls2 = binary_local_search(h_sa, dx_t, min(t2, time.time() + 4), max_passes=100)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()
                    best_n = n_try
                    x_ref = np.linspace(0.0, 2.0, best_n)

    # Phase 3: Multi-start random binary at best_n and a couple others
    t3 = start_time + budget_s * 0.85
    n_cands = [best_n]
    if best_n != 48:
        n_cands.append(48)
    if best_n != 52:
        n_cands.append(52)
    for n_try in n_cands:
        if time.time() > t3:
            break
        dx_t = 2.0 / n_try
        target_t = n_try // 2
        for restart in range(4):
            if time.time() > t3:
                break
            h_rand = np.random.uniform(0.0, 1.0, n_try)
            h_rand_b = make_binary(h_rand, target_t)
            t_sub = min(t3, time.time() + max(2, (t3 - time.time()) * 0.3))
            h_r, c5_r = binary_local_search(h_rand_b, dx_t, t_sub, max_passes=200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 4: Continuous refinement of best binary
    t4 = start_time + budget_s * 0.95
    dx_best = 2.0 / best_n
    h_cont, c5_cont = continuous_optimize(best_h, dx_best, best_n / 2.0, t4, p_list=[32, 128, 512, 2048])
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont

    # Phase 5: Final binary local search
    t5 = start_time + budget_s * 0.98
    h_bin_f = make_binary(best_h, best_n // 2)
    h_f, c5_f = binary_local_search(h_bin_f, 2.0 / best_n, t5, max_passes=80)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_unit_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
