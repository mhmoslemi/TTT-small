import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -80, 80)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    n = len(h_start)
    def obj(x):
        return soft_max_c5(make_feasible(x, target), alpha)
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h_start, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, target)
    except Exception:
        return h_start

    c5_cur = compute_c5(h_cur)

        g[:n-j] -= w_neg * h[j:n]         # -h[i+j] for i=0..n-1-j
    
    # For j=0: corr[n-1] = sum h[i](1-h[i]), d/dh[i] = 1-2h[i]
    g += weights[n-1] * (1 - 2*h)
    
    return dx * g

def slsqp_step(h_start, target, alpha, maxiter):
    n = len(h_start)
    def obj(x):
        return soft_max_c5(make_feasible(x, target), alpha)
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h_start, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, target)
    except Exception:
        return h_start

def projected_grad_descent(h, target, alpha, lr_start, n_steps, deadline):
    """Gradient descent with projection onto {sum=target, 0<=h<=1}."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr_start
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        if it % 1000 == 0 and it > 0:
            lr *= 0.95
        
        # Compute gradient
        g = soft_max_grad_c5(h_cur, alpha)
        
        # Gradient step
        h_try = h_cur - lr * g
        
        # Project onto feasible set
        h_try = np.clip(h_try, 0, 1)
        # Adjust sum
        h_try = make_feasible(h_try, target)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h_cur = h_try
    
    return best_h, best_c5

def coord_sa(h, target, rng, n_steps, T0, deadline):
    """Coordinate descent with simulated annealing."""
    n = len(h)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    step = 0.03
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(n_steps, 1)
        T = T0 * 1e-8 ** frac  # geometric cooling
        
        if it % 3000 == 0 and it > 0:
            step *= 0.85
        
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        delta = rng.uniform(-step, step)
        h_try = h_cur.copy()
        h_try[i] = np.clip(h_try[i] + delta, 0, 1)
        h_try[j] = np.clip(h_try[j] - delta, 0, 1)
        h_try = make_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        
        if c5_try <= c5_cur:
            h_cur = h_try
            c5_cur = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        elif T > 1e-15:
            if rng.uniform() < np.exp(-(c5_try - c5_cur) / T):
                h_cur = h_try
                c5_cur = c5_try
    
    return best_h, best_c5
