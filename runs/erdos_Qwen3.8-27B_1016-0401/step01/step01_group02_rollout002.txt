                h2[i] = np.clip(h2[i] + delta, 0, 1)
                d = h2.sum() - tgt
                if abs(d) > 1e-10:
                    msk = (h2 < 1 - 1e-8) if d < 0 else (h2 > 1e-8)
                    if msk.sum() == 0:
                        continue
                    h2[msk] += d / msk.sum()
                    np.clip(h2, 0.0, 1.0, out=h2)
                c = _c5(h2)
                if c < bestc - 1e-12:
                    best, bestc = h2, c
                    improved = True
                    break
        if not improved:
            break
    return best, bestc

def _upscale(h, n_new, n_old):
    x_old = np.linspace(0.0, 2.0, n_old + 1)
    x_new = np.linspace(0.0, 2.0, n_new + 1)
    h_pad = np.concatenate([h, [h[-1]]])
    h_new = np.interp(x_new[:-1], x_old, h_pad)
    return _proj(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5
    n = len(h)
    h = _proj(h, n)
    bh, bc = h.copy(), _c5(h)

    def rem():
        return budget_s - (time.time() - ts)

    # Phase 1: SLSQP at base resolution
    for sc in [40, 80, 150, 300]:
        if rem() < 40:
            break
        h1 = _slsqp_opt(bh, n, sc, 6000)
        c1 = _c5(h1)
        if c1 < bc - 1e-12:
            bh, bc = h1, c1

    # Phase 2: Local search
    if rem() > 50:
        bh, bc = _local_search(bh, n, min(rem() * 0.12, 70.0), rng)

    # Phase 3: Upscale x2 and optimize
    n2 = 2 * n
    if n2 <= 720 and rem() > 150:
        h2 = _upscale(bh, n2, n)
        for sc in [60, 120, 250]:
            if rem() < 60:
                break
            h2s = _slsqp_opt(h2, n2, sc, 6000)
            if _c5(h2s) < _c5(h2):
                h2 = h2s
        if _c5(h2) < bc:
            bh, bc, n = h2, _c5(h2), n2
        if rem() > 50:
            bh, bc = _local_search(bh, n, min(rem() * 0.18, 130.0), rng)

    # Phase 4: Upscale x4 total
    n3 = 2 * n
    if n3 <= 720 and rem() > 180:
        h3 = _upscale(bh, n3, n)
        for sc in [100, 200, 400]:
            if rem() < 60:
                break
            h3s = _slsqp_opt(h3, n3, sc, 6000)
            if _c5(h3s) < _c5(h3):
                h3 = h3s
        if _c5(h3) < bc:
            bh, bc, n = h3, _c5(h3), n3
        if rem() > 40:
            bh, bc = _local_search(bh, n, min(rem() * 0.2, 160.0), rng)

    # Phase 5: Coordinate descent
    if rem() > 30:
        bh, bc = _coord_descent(bh, n, min(rem() * 0.25, 100.0), rng)

    # Phase 6: Final local search + SLSQP polish
    if rem() > 30:
        bh, bc = _local_search(bh, n, min(rem() * 0.3, 80.0), rng)
    if rem() > 25:
        h_f = _slsqp_opt(bh, n, 500, 3000)
        if _c5(h_f) < bc:
            bh, bc = h_f, _c5(h_f)

    bh = _proj(bh, n)
    fc = _c5(bh)
    return (bh, fc, n)
