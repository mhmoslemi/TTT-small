import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def coord_descent(h, dx, t_limit, patience=200, use_random_order=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    stall = 0
    sweep = 0
    while time.time() < t_limit and stall < patience:
        if use_random_order and sweep % 5 == 0:
            order = np.random.permutation(n)
        else:
            order = range(n)
        for j in order:
            old_val = h[j]
            best_j = old_val
            best_obj = np.inf
            for x in np.linspace(0.0, 1.0, 41):
                h[j] = x
                c = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c < best_obj:
                    best_obj = c
                    best_j = x
            lo_f = max(0.0, best_j - 0.06)
            hi_f = min(1.0, best_j + 0.06)
            for x in np.linspace(lo_f, hi_f, 41):
                h[j] = x
                c = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c < best_obj:
                    best_obj = c
                    best_j = x
            h[j] = best_j
        h = project_feasible(h, target)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        sweep += 1
    return best_h, best_c5

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return dx * (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)


def _cons_fun(x, ts):
    return np.sum(x) - ts


def _cons_jac(x):
    return np.ones(len(x))


def slsqp_refine(h, dx, t_limit):
    n = len(h)
    ts = n / 2.0
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for beta in [300.0, 1500.0, 6000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_obj, best_h, args=(dx, beta),
                           method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 1500, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5
