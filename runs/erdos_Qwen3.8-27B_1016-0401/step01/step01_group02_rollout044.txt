import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def _sopt(h, dx, p):
    """p-norm approximation of max"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use the top values primarily
    top = np.sort(corr)[-min(10, len(corr)):]
    return (np.sum(top**p) / len(top))**(1.0/p) * dx

def _lsq_obj(x, dx, p):
    h = np.clip(x, 0, 1)
    return _sopt(h, dx, p)

def _mc(x):
    return x.sum() - _G_N * 0.5

def _mc_jac(x):
    return np.ones(len(x))

_G_N = 0

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0, 1)
    for _ in range(200):
        h -= (h.sum() - tgt) / n
        if h.min() >= 0 and h.max() <= 1:
            break
        np.clip(h, 0, 1, out=h)
    # Final fix
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.argsort(-h)[:int(abs(r))+1]
            for i in idx:
                add = min(1-h[i], r)
                h[i] += add
                r -= add
                if r < 1e-14: break
        else:
            idx = np.argsort(h)[:int(abs(r))+1]
            for i in idx:
                sub = min(h[i], -r)
                h[i] -= sub
                r += sub
                if r > -1e-14: break
    return h

    global _G_N
    _G_N = n
    bds = [(0,1)]*n
    cons = [{'type':'eq', 'fun':_mc, 'jac':_mc_jac}]
    def f(x):
        return _lsq_obj(x, dx, p)
    try:
        res = minimize(f, h0, method='SLSQP', bounds=bds, constraints=cons,
                       options={'maxiter':maxiter, 'ftol':1e-12})
        return _proj(res.x, n)
    except:
        return _proj(h0, n)

    best = h.copy()
    bestc = _c5(best)
    cur = h.copy()
    curc = bestc
    tgt = 0.5 * n

    for it in range(niter):

            break
        move_type = rng.random()
        h2 = cur.copy()
        if move_type < 0.5:

            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:

                h2[i] += d
                h2[j] -= d
        elif move_type < 0.8:

            i = rng.integers(n - blen)
            block = h2[i:i+blen].copy()
            shift = rng.uniform(-0.15, 0.15)
            h2[i:i+blen] = np.clip(block + shift, 0, 1)
            # Compensate elsewhere
            j_start = (i + blen + 5) % n
            j_end = min(j_start + blen, n)
            if j_end == j_start:
