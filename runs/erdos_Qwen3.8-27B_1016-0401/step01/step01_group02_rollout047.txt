import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sopt(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    # Use log-sum-exp for numerical stability
    s = sc * (corr - m)
    max_s = s.max()
    return (m + max_s / sc + np.log(np.exp(s - max_s).sum()) / sc) * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

# SLSQP wrapper
_G_N = 0
_G_DX = 0.0
_G_SC = 0.0

def _f_slsqp(x):
    return _sopt(np.clip(x, 0.0, 1.0), _G_DX, _G_SC)

def _ceq(x):
    return x.sum() - 0.5 * _G_N

def _ceqj(x):
    return np.ones(len(x))

def _slsqp(h0, n, dx, sc, maxiter):
    global _G_N, _G_DX, _G_SC
    _G_N, _G_DX, _G_SC = n, dx, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_f_slsqp, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

            mk = (cw < 1.0) if d2 < 0 else (cw > 0.0)
            if mk.any():
                cw[mk] -= d2 / mk.sum()
                np.clip(cw, 0.0, 1.0, out=cw)
        ct = _c5(cw)
        if ct < bestc:
            best, bestc = cw.copy(), ct
        elif rng.random() < 0.1:
            pass
        else:
            cw = best.copy()
        temp = max(0.004, temp * 0.9997)
    return best, bestc

def _two_pt_search(h, n, t_start, t_end, rng):
    cw = h.copy()
    best = h.copy()
    bestc = _c5(best)
    while time.time() < t_end:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        sgn = 1.0 if rng.random() < 0.5 else -1.0
        dv = sgn * rng.uniform(0.02, 0.25)
        h2 = cw.copy()
        h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
        h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        ds = h2.sum() - 0.5 * n
        if abs(ds) > 1e-12:
            h2 -= ds / n
            np.clip(h2, 0.0, 1.0, out=h2)
        c5t = _c5(h2)
        if c5t < bestc:
            best, bestc = h2, c5t
            cw = h2
        elif rng.random() < 0.12:
            cw = h2
        else:
            cw = best.copy()
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        bh = np.array(initial_h_values, dtype=float)
    else:
        bh = np.ones(96) * 0.5

    n = len(bh)
    bc = _c5(bh)

    # Phase 1: Coordinate descent at initial resolution
    rem = budget_s - (time.time() - t0)
    t_end = t0 + min(150, rem * 0.18)
    bh, bc = _coord_down(bh, n, t0, t_end, rng)

    # Phase 2: SLSQP with multiple smoothing scales
    rem = budget_s - (time.time() - t0)
    for sc in [25.0, 50.0, 100.0]:
        if rem < 50:
            break
        h2 = _slsqp(bh, n, sc, 2500)
        c2 = _c5(h2)
        if c2 < bc:
            bh, bc = h2, c2
        rem = budget_s - (time.time() - t0)

    # Phase 3: Block + two-point local search
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        t_e3 = t0 + min(rem * 0.22, 200)
        bh, bc = _blk_search(bh, n, t0, t_e3, rng)
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        t_e3b = t0 + min(rem * 0.15, 100)
        bh, bc = _two_pt_search(bh, n, t0, t_e3b, rng)

    # Phase 4: Upscale to 2n
    rem = budget_s - (time.time() - t0)
    n2 = 2 * n
    if rem > 180 and n2 <= 720:
        bh2 = _proj(np.repeat(bh, 2), n2)
        bc2 = _c5(bh2)
        for sc in [50.0, 100.0, 200.0]:
            rem = budget_s - (time.time() - t0)
            if rem < 60:
                break
            hh = _slsqp(bh2, n2, sc, 2000)
            cc = _c5(hh)
            if cc < bc2:
                bh2, bc2 = hh, cc
        rem = budget_s - (time.time() - t0)
        if rem > 80:
            t_e4 = t0 + min(rem * 0.28, 200)
            bh2, bc2 = _coord_down(bh2, n2, t0, t_e4, rng)
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            t_e4b = t0 + min(rem * 0.2, 130)
            bh2, bc2 = _blk_search(bh2, n2, t0, t_e4b, rng)
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            t_e4c = t0 + min(rem * 0.15, 80)
            bh2, bc2 = _two_pt_search(bh2, n2, t0, t_e4c, rng)
        if bc2 < bc:
            bh, bc, n = bh2, bc2, n2

    # Phase 5: Final polish - coordinate descent + SLSQP
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        t_e5 = t0 + min(rem * 0.35, 120)
        bh, bc = _coord_down(bh, n, t0, t_e5, rng)
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        hh = _slsqp(bh, n, 80.0, 1500)
        cc = _c5(hh)
        if cc < bc:
            bh, bc = hh, cc

    # Phase 6: Final fine local search
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        t_e6 = t0 + min(rem * 0.5, 80)
        bh, bc = _two_pt_search(bh, n, t0, t_e6, rng)

    fc = _c5(bh)
    return (bh, fc, len(bh))
