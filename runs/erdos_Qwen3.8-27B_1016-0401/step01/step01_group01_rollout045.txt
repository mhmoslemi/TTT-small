import numpy as np
from scipy.optimize import minimize

_G_DX = 1.0
_G_T = 35.0
_G_N = 70
_G_PEN = 1000.0
_G_TMP = 0.1

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _sharp_obj(h):
    return compute_c5(h, _G_DX)

def _sharp_grad(h):
    n = _G_N
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    ks = np.arange(n)
    grad = np.zeros(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    grad[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])
    return _G_DX * grad

def _sm_obj(h):
    return _sm_obj_grad(h)[0]

def _sm_obj_grad(h):
    n = _G_N
    dx = _G_DX
    t = _G_TMP
    T = _G_T
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    obj = dx * (cmax + t * np.log(np.sum(np.exp((c - cmax) / t))))
    exponents = (c - cmax) / t
    exponents -= np.max(exponents)
    w = np.exp(exponents)
    w /= np.sum(w)
    active = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in active:
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += w[m] * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += w[m] * (-h[ks[v2] + s])
    grad = dx * grad
    pen = h.sum() - T
    obj += _G_PEN * pen * pen
    grad += 2.0 * _G_PEN * pen
    return obj, grad

def _cons_eq(h):
    return h.sum() - _G_T

def _cons_eq_jac(h):
    return np.ones(_G_N)

def _normalize(h, T):
    h = np.clip(h, 0, 1).astype(float)
    s = h.sum()
    if s > 1e-12:
        h = h * (T / s)
    else:
        h = np.full(h.shape, T / len(h))
    h = np.clip(h, 0, 1)
    d = T - h.sum()
    h += d / len(h)
    return np.clip(h, 0, 1)

def _interp_to_n(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_parent, period=2.0)
    return _normalize(h_new, n_new / 2.0)

def _optimize_from(h_init, n, max_iter_smooth=300, max_iter_slsqp=1000):
    global _G_DX, _G_T, _G_N, _G_PEN, _G_TMP
    _G_DX = 2.0 / n
    _G_T = n / 2.0
    _G_N = n
    T = n / 2.0
    dx = 2.0 / n
    h = _normalize(h_init, T).copy()
    bounds = [(0.0, 1.0)] * n
    temps = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    import time
    global _run_t0, _run_budget
    for t_i, t_val in enumerate(temps):
        if time.time() - _run_t0 > _run_budget - 40:
            break
        _G_TMP = t_val
        _G_PEN = 500.0 + 500.0 * t_i
        try:
            r = minimize(_sm_obj, h, jac=_sm_obj_grad, method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': max_iter_smooth, 'ftol': 1e-16,
                                 'gtol': 1e-12, 'maxls': 50})
            if r.x is not None:
                h_new = _normalize(r.x, T)
                if compute_c5(h_new, dx) <= compute_c5(h, dx) + 1e-9:
                    h = h_new
        except Exception:
            pass
    # SLSQP polish on sharp max
    if time.time() - _run_t0 > _run_budget - 20:
        return _normalize(h, T), compute_c5(h, dx)
    _G_PEN = 0.0
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:

    # Second SLSQP from smooth result

                    bounds=bounds, constraints=cons,

        h_new = _normalize(r.x, T)
        if compute_c5(h_new, dx) < compute_c5(h, dx):
            h = h_new
    except Exception:
        pass
    h = _normalize(h, T)
    return h, compute_c5(h, dx)
