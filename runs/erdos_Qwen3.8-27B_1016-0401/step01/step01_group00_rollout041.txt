import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def get_subgrad(h, top_k=3, beta=20.0):
    """Weighted subgradient from top-k shifts"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    
    # Get top-k
    top_idx = np.argsort(corr)[::-1][:top_k]
    max_c = corr[top_idx[0]]
    weights = np.exp(beta * (corr[top_idx] - max_c))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - center
        g = np.zeros(n)
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            g[:n-k] += 1.0
            g[:n-k] -= h[k:]
            g[k:] -= h[:n-k]
        else:
            p = -k
            g[p:] += 1.0 - h[:n-p]
            g[:n-p] -= h[p:]
        grad += w * g
    
    return grad * dx, corr * dx

def project_onto_feasible(h, target):
    """Project h onto {x : 0<=x_i<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Water-filling
    if diff > 0:
        # Need to add mass; add to elements with most room
        idx = np.argsort(1.0 - h)[::-1]  # most room first
        for i in idx:
            if diff <= 0:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        # Need to remove mass; remove from elements with most mass
        idx = np.argsort(h)[::-1]  # most mass first
        for i in idx:
            if diff >= 0:
                break
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
    return h

def projected_grad_descent(h, target, budget_s, t0, top_k=5, beta=30.0, 
                            momentum=0.9, n_iter_max=100000):
    """Projective gradient descent with momentum"""
    h_cur = h.copy()
    v = np.zeros_like(h)
    lr = 1.0 / (top_k * np.max(np.abs(get_subgrad(h_cur, top_k, beta)[0])))
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for it in range(n_iter_max):
        if time.time() - t0 > budget_s - 30:
            break
        
        grad, corr_vals = get_subgrad(h_cur, top_k, beta)
        # Backtracking line search
        alpha = lr
        for _ in range(30):
            h_new = project_onto_feasible(h_cur - alpha * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 or alpha < 1e-8:
                break
            alpha *= 0.5
        
        v = momentum * v - alpha * grad
        h_new = project_onto_feasible(h_cur + v, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            # Slowly increase top_k and decrease beta
        else:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    top_idx = np.argsort(corr)[::-1][:top_k]

            grad[k:] += w * (-h[:n-k])
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] += w * (-h[p:])

            diff -= add

            h[i] -= rem
            diff += rem

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_topk(h, top_k=5, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    top_idx = np.argsort(corr)[::-1][:top_k]
    max_c = corr[top_idx[0]]
    weights = np.exp(beta * (corr[top_idx] - max_c))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad * dx

def gradient_softmax(h, T=10.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    for o_idx in range(len(corr)):
        w = sw[o_idx]
        if w < 1e-12:
            continue
        k = o_idx - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad * dx, (mc + np.log(np.sum(e)) / T) * dx

            h[i] += add
            diff -= add
        else:

            h[i] -= rem
            diff += rem
    return h

def pgd_solve(h, target, deadline, top_k=5, beta=40.0, lr_init=1.0, max_iter=80000):
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr_init
    v = np.zeros_like(h_cur)
    momentum = 0.85
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = gradient_topk(h_cur, top_k, beta)
        h_try = project_feasible(h_cur - lr * grad, target)
        f_try = compute_c5(h_try)
        if f_try < best_c5:
            best_c5 = f_try
            best_h = h_try.copy()
            v = momentum * v - lr * grad
            h_cur = project_feasible(h_cur + v, target)
            lr = min(lr * 1.05, 3.0)
            stuck = 0

                stuck = 0
        if it % 2000 == 0 and it > 0:

    return best_h, best_c5

def softmax_solve(h, target, deadline, T_init=20.0, max_iter=300):
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)

        if time.time() > deadline:
            break

        def obj(x):
            _, val = gradient_softmax(x, T)
            return val + 50.0 * (x.sum() - target) ** 2
        def jac(x):
            g, _ = gradient_softmax(x, T)
            g += 100.0 * (x.sum() - target) * np.ones(len(x))
            return g
        try:

                          options={'maxiter': 50, 'ftol': 1e-14, 'gtol': 1e-10})

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
                h_cur = h_new

            pass
