        temps_short = [0.1, 0.05, 0.02, 0.01]
        h_r, c5_r = _anneal_optimize(h_r, n_init, temps_short, 150, 1e-12)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r
    
    # Phase 4: Try other grid sizes
    for n_alt in [40, 50, 100, 140]:
        if time.time() - t0 > budget_s - 60:
            break
        # Interpolate best to new size
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_alt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = _normalize(h_interp, n_alt / 2.0)
        h_interp, c5_i = _anneal_optimize(h_interp, n_alt, [0.05, 0.02, 0.01], 250, 1e-12)
        if c5_i < best_c5:
            best_c5, best_h = c5_i, h_interp
        # Random at this size
        if time.time() - t0 > budget_s - 30:
            break
        h_r = rng.random(n_alt)
        h_r, c5_r = _anneal_optimize(h_r, n_alt, [0.1, 0.05, 0.02], 150, 1e-12)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r
    
    # Phase 5: Perturbation restarts from best
    n_b = len(best_h)
    for sigma in [0.05, 0.02, 0.01]:
        if time.time() - t0 > budget_s - 20:
            break
        for i in range(8):
            if time.time() - t0 > budget_s - 15:
                break
            h_p = best_h + sigma * rng.randn(n_b)
            h_p = _fix_sum(h_p, n_b / 2.0)
            h_p, c5_p = _anneal_optimize(h_p, n_b, [0.02, 0.01, 0.005], 150, 1e-12)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_p
    
    # Phase 6: Final sharp polish
    if time.time() - t0 < budget_s - 5:
        _setup_globals(n_b)
        T = n_b / 2.0
        h_final = best_h.copy()
        for _ in range(3):
            if time.time() - t0 > budget_s - 3:
                break
            try:
                r = minimize(_sharp_obj, h_final, method='SLSQP', jac=_sharp_jac,
                             bounds=_BOUNDS, constraints=_CONS,
                             options={'maxiter': 2000, 'ftol': 1e-15})
                h_final = r.x
            except Exception:
                break
        h_final = _fix_sum(h_final, T)
        c5_final = compute_c5(h_final, 2.0 / n_b)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final
    
    # Final cleanup
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    best_h = _fix_sum(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    
    return best_h, best_c5, n_f
