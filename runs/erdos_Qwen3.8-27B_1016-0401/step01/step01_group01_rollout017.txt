    # Phase 4: Random restarts
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    T_best = n_best / 2.0
    t_short = [0.05, 0.02, 0.01, 0.005, 0.001]
    for i in range(30):
        if time.time() - t0 > budget_s - 60:
            break
        style = i % 4
        if style == 0:
            h_r = rng.random(n_best)
        elif style == 1:
            h_r = (rng.random(n_best) > 0.5).astype(float)
        elif style == 2:
            h_r = np.sin(np.arange(n_best) * rng.uniform(0.1, 0.5) + rng.uniform(0, 3)) * 0.4 + 0.5
        else:
            h_r = np.tile(np.linspace(0.2, 0.8, 7), n_best // 7 + 1)[:n_best] + rng.random(n_best) * 0.1
        h_r = project_feasible(h_r, T_best)
        h_try, c5_try = anneal_optimize(h_r, n_best, t_short, 500, dx_best)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 5: Perturbation around best
    if time.time() - t0 < budget_s - 30:
        for i in range(15):
            if time.time() - t0 > budget_s - 20:
                break
            scale = 0.05 + 0.05 * (i / 15.0)
            h_pert = best_h + rng.randn(n_best) * scale
            h_pert = project_feasible(h_pert, T_best)
            h_try, c5_try = anneal_optimize(h_pert, n_best, [0.01, 0.005, 0.001, 0.0005], 600, dx_best)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

    # Phase 6: Final ultra-fine polish
    if time.time() - t0 < budget_s - 10:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        T_f = n_f / 2.0
        _NS['dx'] = dx_f
        _NS['T'] = T_f
        _NS['N'] = n_f
        bounds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        h_f = project_feasible(best_h, T_f)
        _NS['t'] = 0.0001
        try:
            res = minimize(_obj_smooth, h_f, method='SLSQP', jac=_grad_smooth,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            h_f = project_feasible(res.x, T_f)
            c5 = compute_c5(h_f, dx_f)
            if c5 < best_c5:
                best_c5, best_h = c5, h_f
        except Exception:
            pass
        try:
            res = minimize(_obj_sharp, best_h, method='SLSQP', jac=_grad_sharp,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-16})
            h_f = project_feasible(res.x, T_f)
            c5 = compute_c5(h_f, dx_f)
            if c5 < best_c5:
                best_c5, best_h = c5, h_f
        except Exception:
            pass

    n_f = len(best_h)
    dx_f = 2.0 / n_f
    best_h = project_feasible(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, n_f
