def compute_c5(h, dx):

def project_binary(h, target):
    """Project to binary with exactly target ones."""
    n = len(h)
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    if s == target:
        return h
    if s > target:
        # Need to remove s - target ones
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:s - target]:
            h[i] = 0
    else:
        # Need to add target - s ones
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:target - s]:
            h[i] = 1
    return h

def make_evenly_spaced(n, k):
    """Place k ones evenly in n positions."""
    h = np.zeros(n)
    positions = np.unique((np.arange(k) * n / k).astype(int))
    h[positions] = 1
    return h

def make_quadratic_res(n, k):
    """Place ones at quadratic residue positions mod p."""
    p = n - 1  # or nearest prime
    # Find nearby prime
    while True:
        if p > 2 and all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)):
            break
        p += 1
    h = np.zeros(n)
    count = 0
    for x in range(1, p):
        if pow(x, 2, p) == 1 or pow(x, (p-1)//2, p) == 1:
            h[x % n] = 1
            count += 1
        if count >= k:
            break
    return project_binary(h, k)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.minimum(beta * (corr - max_c), 500.0)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        m = k - (n - 1)
        if m >= 0:
            a, b = 0, n - m
            grad[a:b] += wk * (1.0 - h[a + m:b + m])
            a2, b2 = m, n
            grad[a2:b2] -= wk * h[a2 - m:b2 - m]
        else:
            p = -m
            a, b = p, n
            grad[a:b] += wk * (1.0 - h[a - p:b - p])
            a2, b2 = 0, n - p
            grad[a2:b2] -= wk * h[a2 + p:b2 + p]
    return obj, grad

def projected_grad_opt(h0, dx, target_sum, t_end, beta_start=500.0, n_steps=20000):
    n = len(h0)
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    beta = beta_start
    lr = 0.5
    for it in range(n_steps):
        if time.time() > t_end:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_try = project_feasible(h - lr * grad, target_sum)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            lr = min(lr * 1.05, 10.0)
        else:
            h = project_feasible(h - lr * 0.5 * grad, target_sum)
            lr *= 0.9
        if it % 5000 == 0 and it > 0:
            beta = min(beta * 2.0, 20000.0)
    return best_h, best_c5

def binary_local_search(h, dx, t_end, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h, dx)
    for _ in range(200):
        if time.time() > t_end:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_gain = 1e-15
        best_pair = None
        for zi in zeros:
            if time.time() > t_end:
                return h, c5
            for oi in ones:
                h[zi] = 1.0
                h[oi] = 0.0
                nc5 = compute_c5(h, dx)
                h[zi] = 0.0
                h[oi] = 1.0
                if c5 - nc5 > best_gain:
                    best_gain = c5 - nc5
                    best_pair = (zi, oi)
        if best_pair is None:
            break
        h[best_pair[0]] = 1.0
        h[best_pair[1]] = 0.0
        c5 -= best_gain
    return h, c5

def binary_sa(h0, dx, t_end, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    T0 = 0.03
    n_total = 300000
    for step in range(n_total):
        if time.time() > t_end:
            break
        T = T0 * max(1.0 - step / n_total, 0.01) ** 1.5
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        zi = zeros[rng.integers(len(zeros))]
        oi = ones[rng.integers(len(ones))]
        h[zi] = 1.0
        h[oi] = 0.0
        nc5 = compute_c5(h, dx)
        d = nc5 - c5_cur
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            c5_cur = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[zi] = 0.0
            h[oi] = 1.0
    return best_h, best_c5
