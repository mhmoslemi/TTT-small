import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_DX = 0.0
_G_SC = 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sopt(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx

def _f(x):
    return _sopt(np.clip(x, 0.0, 1.0), _G_DX, _G_SC)

def _ceq(x):
    return x.sum() - 0.5 * _G_N

def _ceqj(x):
    return np.ones(len(x))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

    global _G_N, _G_DX, _G_SC
    _G_N, _G_DX, _G_SC = n, dx, sc

                     constraints=cs, options={'maxiter': mi, 'ftol': 1e-14})
        return _proj(r.x, n)
