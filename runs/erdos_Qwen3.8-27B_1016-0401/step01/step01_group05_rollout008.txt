```python
    m = np.max(corr)

        h[mask] += diff / np.count_nonzero(mask)
```

```python
        js2 = max(0, lag); je2 = min(n, n + lag)
        if js2 < je2:
            grad[js2:je2] -= h[js2-lag:je2-lag]
    return grad * dx / top_k
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.clip(alpha * (corr - m), -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    lags = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = int(lags[k])
        js = max(0, -lag); je = min(n, n - lag)
        if js < je:
            grad[js:je] += wk * (1.0 - h[js+lag:je+lag])
        js2 = max(0, lag); je2 = min(n, n + lag)
        if js2 < je2:
            grad[js2:je2] -= wk * h[js2-lag:je2-lag]
    return grad * dx

def grad_topk(h, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    thresh = np.max(corr) - 0.05 * (np.max(corr) - np.min(corr))
    idx = np.where(corr >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(corr)])
    lags = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for k in idx:
        lag = int(lags[k])
        js = max(0, -lag); je = min(n, n - lag)
        if js < je:
            grad[js:je] += (1.0 - h[js+lag:je+lag])
        js2 = max(0, lag); je2 = min(n, n + lag)
        if js2 < je2:
            grad[js2:je2] -= h[js2-lag:je2-lag]
    return grad * dx / len(idx)

def project(h, target):
    h = h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_optimize(h, n, target, alpha, lr, max_iter, deadline):
    h_curr = h.copy()
    h_prev = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        beta = 0.9 if it > 10 else 0.0
        y = h_curr + beta * (h_curr - h_prev)
        g = grad_smooth_max(np.clip(y, 0, 1), alpha)
        h_new = project(y - lr * g, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_prev = h_curr
        h_curr = h_new
    return best_h, best_c5

def topk_optimize(h, n, target, lr, max_iter, deadline):
    h_curr = h.copy()
    h_prev = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        beta = 0.85 if it > 20 else 0.0
        y = h_curr + beta * (h_curr - h_prev)
        g = grad_topk(np.clip(y, 0, 1), top_k=3)
        h_new = project(y - lr * g, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_prev = h_curr
        h_curr = h_new
    return best_h, best_c5

def coord_descend(h, n, target, max_iters, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    rng = np.random.RandomState(777)
    step = np.zeros(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        s = h_curr[i] + h_curr[j]
        t_lo = max(0.0, s - 1.0)
        t_hi = min(1.0, s)
        if t_lo >= t_hi:
            continue
        ts = np.linspace(t_lo, t_hi, 16)
        best_t = ts[0]
        best_v = 1e9
        for t in ts:
            h_try = h_curr.copy()
            h_try[i] = t
            h_try[j] = s - t
            v = compute_c5(h_try)
            if v < best_v:
                best_v = v
                best_t = t
        h_curr[i] = best_t
        h_curr[j] = s - best_t
        if best_v < best_c5:
            best_c5 = best_v
    return h_curr, best_c5

def interpolate_to(h_src, n_tgt):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    return np.interp(x_tgt, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_src = np.array(initial_h_values, dtype=float)
    else:
        h_src = np.full(70, 0.5)

    best_h = h_src.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_src)

    resolutions = [len(h_src), 140, 210]
    for n_target in resolutions:
        if time.time() > deadline - 80:
            break
        if n_target == len(h_src):
            h = h_src.copy()
        else:
            h = interpolate_to(h_src, n_target)
        n = len(h)
        target = n / 2.0
        h = project(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

        # Phase A: smooth-max Nesterov (coarse to fine alpha)
        for alpha, lr, n_iter in [(30, 2.0/(30), 300), (100, 2.0/(100), 400),
                                   (300, 1.5/(300), 500), (800, 1.0/(800), 600)]:
            if time.time() > deadline - 40:
                break
            h, c5 = nesterov_optimize(h, n, target, alpha, lr, n_iter, deadline - 40)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), n

        # Phase B: top-k subgradient refinement
        for lr_val in [0.02, 0.01, 0.005, 0.002]:
            if time.time() > deadline - 30:
                break
            h, c5 = topk_optimize(h, n, target, lr_val, 400, deadline - 30)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), n

        # Phase C: coordinate descent
        if time.time() < deadline - 15:
            h, c5 = coord_descend(h, n, target, 8000, deadline - 15)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), n

    # Final polish on global best
    if time.time() < deadline - 5:
        h_fin = project(best_h, best_n / 2.0)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin
        # One more coord descent on the global best
        if time.time() < deadline - 3:
            h_fin, c5_fin = coord_descend(best_h, best_n, best_n/2, 3000, deadline - 3)
            if c5_fin < best_c5:

    best_h = project(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), final_c5, best_n
```
