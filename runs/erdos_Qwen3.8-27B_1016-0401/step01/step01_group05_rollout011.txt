    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(np.clip(x, 0, 1), 1.0 - np.clip(x, 0, 1), mode='full')
    m = np.max(corr)
    vals = alpha * (corr - m)
    vals = np.clip(vals, -50, 50)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

        return np.ones(n)
