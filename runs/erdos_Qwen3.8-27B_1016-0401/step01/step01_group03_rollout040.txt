    for _ in range(100000):
        if time.time() > t_limit:
            break
        _, g = smooth_loss_grad(h, dx, beta)
        h_new = project_feasible(h - lr * g, ts)
        c5_n = compute_c5(h_new, dx)
        if c5_n < best_c5 - 1e-15:
            best_c5 = c5_n
            best_h = h_new.copy()
            lr = min(lr * 1.02, 5.0)
        else:
            lr *= 0.95
        h = h_new
        if lr < 1e-7:
            break
    return best_h, best_c5

def sa_binary(h0, dx, t_limit):
    n = len(h0)
    n_ones = n // 2
    h = to_binary(h0, n_ones).copy()
    mc = np.max(get_corr(h))
    best_mc = mc
    best_h = h.copy()
    T = 1.0
    while time.time() < t_limit:
        T = max(T * 0.99997, 0.003)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if np.random.random() < 0.25:
            a1, a2 = np.random.choice(ones, 2, replace=False)
            b1, b2 = np.random.choice(zeros, 2, replace=False)
            h[a1] = h[a2] = 0; h[b1] = h[b2] = 1
            nm = np.max(get_corr(h))
            if nm < mc or np.random.random() < np.exp(-(nm - mc) / T):
                mc = nm
                if mc < best_mc:
                    best_mc = mc; best_h = h.copy()
            else:
                h[a1] = h[a2] = 1; h[b1] = h[b2] = 0
        else:
            a = np.random.choice(ones)
            b = np.random.choice(zeros)
            h[a] = 0; h[b] = 1
            nm = np.max(get_corr(h))
            if nm < mc or np.random.random() < np.exp(-(nm - mc) / T):
                mc = nm
                if mc < best_mc:
                    best_mc = mc; best_h = h.copy()
            else:
                h[a] = 1; h[b] = 0
    return best_h, best_mc * dx
