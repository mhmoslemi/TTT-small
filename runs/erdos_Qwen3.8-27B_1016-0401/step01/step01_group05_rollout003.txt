import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_active_shift(h, shift, n):
    """Gradient of correlation at a given shift index in full correlate output."""
    d = shift - (n - 1)  # actual lag
    j = np.arange(n)
    grad = np.zeros(n)
    m1 = (j + d >= 0) & (j + d < n)
    grad[m1] += (1.0 - h[j[m1] + d])
    m2 = (j - d >= 0) & (j - d < n)
    grad[m2] += -h[j[m2] - d]
    return grad

def slsqp_optimize(h, n, target, alpha, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def cons(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':cons}],
                       options={'maxiter':maxiter, 'ftol':1e-12})
        return make_feasible(res.x, n, target)
    except:
        return h.copy()

def gradient_annealing(h, n, target, rng, deadline, T0=0.005, cool=0.9997):
    """Gradient-based simulated annealing."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = T0
    step_size = 0.02
    stagnant = 0
    
    while time.time() < deadline:
        # Compute all correlations
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        active = np.argmax(corr)
        cur_c5_actual = corr[active] * (2.0/n)
        
        # Gradient of active constraint
        g = grad_active_shift(cur_h, active, n)
        
        # Project gradient (remove mean component for sum constraint)
        g = g - np.mean(g)  # project onto sum=0 subspace
        
        # Gradient descent step
        scale = step_size / (np.linalg.norm(g) + 1e-10)
        new_h = cur_h - scale * g
        new_h = make_feasible(new_h, n, target)
        
        # Also try a random perturbation sometimes
        if rng.random() < 0.15:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.normal(0, 0.03)
            new_h[i] += delta
            new_h[j] -= delta
            new_h = make_feasible(new_h, n, target)
        
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        
        # Acceptance criterion
        if new_c5 < cur_c5 or rng.random() < np.exp(-delta_c5 / T):
            cur_h = new_h
            cur_c5 = new_c5
            # Reduce step size slightly on acceptance
            step_size *= 0.999
        else:

            step_size = min(step_size * 1.05, 0.15)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Reheat if stagnant
        if stagnant > 500:

            stagnant = 0
        
        T *= cool
        if T < T0 * 0.01:

    return best_h, best_c5

   This is a weighted average of the gradients of all constraints, weighted by how close they are to the max.

3. For very large α, this approaches the gradient of the single active constraint. For moderate α, it's a "descent direction" that pushes down all near-max constraints simultaneously.

Let me implement this more carefully:

```python
def softmax_gradient(h, n, alpha):
    """Compute gradient of soft-max of all correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - m), -50, 0))
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual gradients
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        if weights[k] < 1e-10:
            continue
        d = k - (n - 1)
        j = np.arange(n)
        gk = np.zeros(n)
        m1 = (j + d >= 0) & (j + d < n)
        gk[m1] += (1.0 - h[j[m1] + d])
        m2 = (j - d >= 0) & (j - d < n)
        gk[m2] += -h[j[m2] - d]
        grad += weights[k] * gk
    
    # Project onto sum=0 subspace
    grad -= np.mean(grad)
    return grad
```

This is O(n * n_full) = O(n²) per gradient evaluation, which for n=70 is about 70*139 ≈ 9730 operations. Very fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_grad(h, n, alpha, top_k=30):
    """Analytic gradient of soft-max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    n_full = 2 * n - 1
    if n_full > top_k:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(n_full)
    top_corr = corr[top_idx]
    w = np.exp(np.clip(float(alpha) * (top_corr - m), -60, 0))
    w /= np.sum(w)
    j = np.arange(n)
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-12:
            continue
        d = int(idx) - (n - 1)
        m1 = (j + d >= 0) & (j + d < n)
        m2 = (j - d >= 0) & (j - d < n)
        grad[m1] += wi * (1.0 - h[j[m1] + d])
        grad[m2] += wi * (-h[j[m2] - d])
    grad -= np.mean(grad)
    return grad, corr, m

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    vals = np.clip(float(alpha) * (corr - m), -60, 0)
    return (m + np.sum(np.exp(vals)) / float(alpha)) * dx

def slsqp_run(h, n, target, alpha, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        vals = np.clip(float(alpha) * (corr - m), -60, 0)
        return (m + np.sum(np.exp(vals)) / float(alpha)) * dx
    def cons(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def gradient_anneal(h, n, target, rng, deadline, T0=0.003, cool=0.9995, max_random_frac=0.2):
    """Soft-max gradient descent with simulated annealing."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = T0
    lr = 0.04
    stagnant = 0
    it = 0
    while time.time() < deadline:
        it += 1
        # Adaptive alpha: start high, decrease
        alpha = 200.0 * max(0.1, 1.0 - it / 50000.0)
        g, corr, mval = softmax_grad(cur_h, n, alpha)
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-14:
            g = g / gnorm
        # Backtracking line search
        for _ in range(6):
            new_h = make_feasible(cur_h - lr * gnom, n, target) if False else make_feasible(cur_h - lr * g, n, target)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 or rng.random() < np.exp(-max(0, new_c5 - cur_c5) / max(T, 1e-8)):
                break
            lr *= 0.6
        # Sometimes add random perturbation
        if rng.random() < max_random_frac:
            ii = rng.choice(n, 2, replace=False)
            delta = rng.normal(0, 0.02)
            new_h[ii[0]] += delta
            new_h[ii[1]] -= delta
            new_h = make_feasible(new_h, n, target)

            else:

                new_c5 = compute_c5(new_h)
        cur_h = new_h.copy()
        cur_c5 = new_c5
        if new_c5 < best_c5:

            stagnant = 0
        else:

        lr = min(lr * 0.9997, 0.12)
