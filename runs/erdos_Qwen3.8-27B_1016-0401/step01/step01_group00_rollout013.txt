import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def sm_obj(h, T, lam, target):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sm = mc + np.log(np.sum(e)) / T
    return sm * (2.0 / n) + lam * (h.sum() - target) ** 2

def sm_grad(h, T, lam, target):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - target)
    return grad

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(d / cnt, 1.0 - h[mask].min() if mask.any() else 1.0)
            # Just add d/cnt to those that can absorb
            h[mask] = np.minimum(1.0, h[mask] + d / cnt)
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + d / cnt)
    return h

The soft-max objective: f(h) = (2/n) * [mc + (1/T) * log(sum(exp(T*(corr - mc))))]

where sw[o] = exp(T*(corr[o]-mc)) / sum(exp(T*(corr[mc]-mc)))

Note: since mc = max(corr), the max of T*(corr-mc) is 0, so exp(0) = 1 for the max element. The other elements have negative exponents. This is numerically stable.

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
