import time

_NPT = 0
_DX = 0.0
_SC = 50.0

def _c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * (2.0 / len(h))

    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:

        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)

    return h

    return _soft_obj(np.clip(x, 0.0, 1.0), _SC)

def _ceq(x):
    return x.sum() - 0.5 * _NPT

def _ceqj(x):
    return np.ones(len(x))

def _slsqp(h0, n, sc, maxiter):
    global _NPT, _DX, _SC
    _NPT, _DX, _SC = n, 2.0 / n, sc
    bds = [(0.0, 1.0)] * n

    v = sc * (corr - m)
    v[v > 500] = 500
    w = np.exp(v)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - center
        if lag >= 0:
            if lag < n:
                grad[lag:] += wk * (1.0 - h[:n - lag])
            if n - 1 - lag >= 0:
                grad[:n - lag] -= wk * h[lag:]
        else:
            l = -lag
            if n - 1 - l >= 0:
                grad[:n - l] += wk * (1.0 - h[l:])
            if l < n:
                grad[l:] -= wk * h[:n - l]
    return grad * dx

def _proj_grad(h, n, tb, sc=80.0):
    t0 = time.time()
    alpha = 0.01
    best = h.copy()
    bestc = _c5(best)
    for it in range(50000):
        if time.time() - t0 > tb:
            break
        g = _grad_soft(h, n, sc)
        h2 = h - alpha * g
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2.copy(), c2
            alpha = min(alpha * 1.1, 0.1)
        else:
            alpha *= 0.7
            h = best.copy()
            if alpha < 1e-6:
                break
        h = h2
    return best, bestc

def _local_search(h0, n, tb, nit_max=500000):
    t0 = time.time()
    tgt = 0.5 * n
    cw = h0.copy()
    c5w = _c5(cw)
    best = h0.copy()
    bestc = c5w
    temp = 0.015
    stall = 0
    for it in range(nit_max):
        if time.time() - t0 > tb:
            break
        if it % 1000 == 500:
            temp *= 0.997
            if temp < 0.001:
                temp = 0.001
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        sgn = 1.0 if np.random.random() < 0.5 else -1.0
        dv = sgn * np.random.uniform(0.01, 0.25)
        h2 = cw.copy()
        ni = np.clip(h2[i] + dv, 0.0, 1.0)
        actual = ni - h2[i]
        nj = np.clip(h2[j] - actual, 0.0, 1.0)
        if abs(nj - h2[j]) > abs(dv) * 1.05:
            continue
        h2[i] = ni
        h2[j] = nj
        c5t = _c5(h2)
        if c5t < bestc - 1e-12:
            best, bestc = h2.copy(), c5t
            cw, c5w = h2, c5t
            stall = 0
        elif c5t < c5w + temp:
            prob = min(1.0, np.exp(-(c5t - c5w) / max(temp, 1e-10)))
            if np.random.random() < prob:
                cw, c5w = h2, c5t
                stall += 1
            else:
                stall += 1
        else:
            stall += 1
            if stall > 2000:
                cw = best.copy()
                c5w = bestc
                stall = 0
    return best, bestc

def _block_search(h0, n, tb, nit_max=100000):
    t0 = time.time()
    tgt = 0.5 * n
    best = h0.copy()
    bestc = _c5(best)
    for it in range(nit_max):
        if time.time() - t0 > tb:
            break
        blen = np.random.randint(2, max(3, n // 8))
        start = np.random.randint(0, n - blen)
        h2 = h0.copy()
        seg = h2[start:start + blen].copy()
        shift = np.random.randint(-3, 4)
        new_seg = np.roll(seg, shift)
        h2[start:start + blen] = new_seg
        ds = h2.sum() - tgt
        if abs(ds) > 1e-12:
            h2 -= ds / n
            np.clip(h2, 0.0, 1.0, out=h2)
        c5t = _c5(h2)
        if c5t < bestc:
            best, bestc = h2.copy(), c5t
            h0 = h2
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 180
        x = np.linspace(0, 2, n_init + 1)[:-1]
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x + 0.5)
        h = np.clip(h, 0.0, 1.0)
        h = _proj(h, n_init)

    n = len(h)
    h = _proj(h, n)
    best_h = h.copy()
    best_c = _c5(best_h)

    # Phase 1: SLSQP with multiple scales
    for sc in [30, 60, 100, 150, 200]:
        rem = budget_s - (time.time() - ts)
        if rem < 40:
            break
        mi = min(8000, max(2000, int(rem * 8)))
        h_s = _slsqp(h, n, sc, mi)
        c_s = _c5(h_s)
        if c_s < best_c:
            best_h, best_c = h_s, c_s
            h = h_s

    # Phase 2: Projected gradient refinement
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        hg, cg = _proj_grad(h, n, min(rem * 0.15, 50.0), sc=100)
        if cg < best_c:
            best_h, best_c = hg, cg
            h = hg

    # Phase 3: Local search
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        hl, cl = _local_search(best_h, n, min(rem * 0.25, 150.0), 800000)
        if cl < best_c:
            best_h, best_c = hl, cl

    # Phase 4: Block search
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        hb, cb = _block_search(best_h, n, min(rem * 0.15, 60.0), 80000)
        if cb < best_c:
            best_h, best_c = hb, cb

    # Phase 5: SLSQP again after local search
    rem = budget_s - (time.time() - ts)
    if rem > 50:
        for sc in [100, 200]:
            h_s = _slsqp(best_h, n, sc, 5000)
            c_s = _c5(h_s)
            if c_s < best_c:
                best_h, best_c = h_s, c_s

    # Phase 6: Upscale to 2n
    n2 = 2 * n
    rem = budget_s - (time.time() - ts)
    if n2 <= 360 and rem > 150:
        h2 = _proj(np.repeat(best_h, 2), n2)
        for sc in [80, 150, 250]:
            rem = budget_s - (time.time() - ts)
            if rem < 50:
                break
            h2 = _slsqp(h2, n2, sc, min(6000, int(rem * 5)))
        c2 = _c5(h2)
        if c2 < best_c:
            best_h, best_c = h2, c2
            n = n2
        rem = budget_s - (time.time() - ts)
        if n == n2 and rem > 80:
            hl, cl = _local_search(best_h, n, min(rem * 0.35, 180.0), 600000)
            if cl < best_c:
                best_h, best_c = hl, cl
            rem = budget_s - (time.time() - ts)
            if rem > 40:
                h_s = _slsqp(best_h, n, 200, 4000)
                c_s = _c5(h_s)
                if c_s < best_c:
                    best_h, best_c = h_s, c_s

    # Phase 7: Final local search polish
    rem = budget_s - (time.time() - ts)
    if rem > 25:
        hl, cl = _local_search(best_h, len(best_h), min(rem * 0.4, 100.0), 400000)
        if cl < best_c:
            best_h, best_c = hl, cl

    # Phase 8: Final SLSQP polish
    rem = budget_s - (time.time() - ts)
    if rem > 20:
        nf = len(best_h)
        h_s = _slsqp(best_h, nf, 250, 3000)
        c_s = _c5(h_s)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    best_h = _proj(best_h, len(best_h))
    final_c = _c5(best_h)
    return (best_h, final_c, len(best_h))
