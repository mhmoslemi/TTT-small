import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    logsum = np.logaddexp.reduce(shifted)
    return (m + logsum / alpha) * dx

def slsqp_opt(h, target, alpha, maxiter=300):
    n = len(h)
    def obj(x):
        return smooth_max_obj(x, alpha)
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, target)
    except Exception:
        return h.copy()

def sa_refine(h0, target, rng, n_steps, T0, T1, s0, s1, deadline):
    n = len(h0)
    cur = h0.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    stale = 0
    for i in range(n_steps):
        if time.time() > deadline:
            break
        p = i / max(n_steps, 1)
        T = T0 * (T1 / T0) ** p
        step = s0 * (s1 / s0) ** p
        # 2-coord swap perturbation
        idx = rng.choice(n, size=2, replace=False)
        d = rng.uniform(-step, step)
        cand = cur.copy()
        cand[idx[0]] += d
        cand[idx[1]] -= d
        cand = make_feasible(cand, target)
        nc = compute_c5(cand)
        if nc <= cur_c5:
            cur = cand
            cur_c5 = nc
            stale = 0
            if nc < best_c5:
                best = cand.copy()
                best_c5 = nc
        else:
            if rng.random() < np.exp(-(nc - cur_c5) / max(T, 1e-12)):
                cur = cand
                cur_c5 = nc
            stale += 1
        # Multi-coordinate jump every 5000 steps
        if i % 5000 == 4999:
            k = rng.choice([2, 3, 4, 5, 6])
            idx2 = rng.choice(n, size=k, replace=False)
            deltas = rng.uniform(-step, step, k)
            deltas -= np.sum(deltas) / k
            cand2 = cur.copy()
            cand2[idx2] += deltas
            cand2 = make_feasible(cand2, target)
            nc2 = compute_c5(cand2)
            if nc2 < best_c5 * 1.01:
                cur = cand2
                cur_c5 = nc2
        # Restart if stuck
        if stale > 4000:
            cur = best.copy()
            cur_c5 = best_c5
            stale = 0
    return best, best_c5
