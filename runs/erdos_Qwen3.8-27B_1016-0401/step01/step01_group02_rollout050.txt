import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _softmax_val(corr, sc):
    m = corr.max()
    return m + np.log(np.exp(sc * (corr - m)).sum()) / sc

def _softobj(x, dx, sc, pen):
    h = np.clip(x, 0, 1)
    corr = _corrs(h)
    val = _softmax_val(corr, sc) * dx

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _softmax_approx(corr, sc):
    m = corr.max()
    return m + np.log(np.exp(sc * (corr - m)).sum()) / sc

def _softobj_pen(x, dx, sc, pen, n):
    h = np.clip(x, 0.0, 1.0)
    corr = _corrs(h)
    val = _softmax_approx(corr, sc) * dx
    s = h.sum() - 0.5 * n
    val += pen * s * s
    return val

def _softobj_pen_grad(x, dx, sc, pen, n):
    h = np.clip(x, 0.0, 1.0)
    corr = _corrs(h)
    m = corr.max()
    w = np.exp(sc * (corr - m))
    w /= w.sum()
    grad_h = np.zeros(n)
    for i in range(2 * n - 1):
        lag = i - (n - 1)
        wi = w[i]
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            grad_h[lo1:hi1] += wi * (1.0 - h[lo1 - lag: hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            grad_h[lo2:hi2] += -wi * h[lo2 + lag: hi2 + lag]
    g = grad_h * dx
    s = h.sum() - 0.5 * n
    g += 2.0 * pen * s
    return g

def _project_sum(h, tgt):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > tgt:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        msk = (h > 1e-12) & (h < 1.0 - 1e-12)
        if msk.any():
            h[msk] += d / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _lbfgs_opt(h0, n, dx, sc, pen, maxiter):
    bds = [(0.0, 1.0)] * n
    def f(x):
        return _softobj_pen(x, dx, sc, pen, n)
    def g(x):
        return _softobj_pen_grad(x, dx, sc, pen, n)
    try:
        r = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return _project_sum(r.x, 0.5 * n)
    except Exception:
        return _project_sum(h0, 0.5 * n)

def _coord_descent(h, n, dx, sc, nsweeps, tmax=None, t0=None):
    tgt = 0.5 * n
    pts = np.linspace(0.0, 1.0, 21)
    for sw in range(nsweeps):
        if tmax is not None and t0 is not None and time.time() - t0 > tmax:
            break
        for r in range(n):
            old = h[r]
            best_v = old
            best_f = _softmax_approx(_corrs(h), sc)
            for pv in pts:
                if pv == old:
                    continue
                h[r] = pv
                f = _softmax_approx(_corrs(h), sc)
                if f < best_f:
                    best_f = f
                    best_v = pv
            h[r] = best_v
    h = _project_sum(h, tgt)
    return h

def _local_search(h, n, tmax, nit, t0):
    tgt = 0.5 * n
    best = h.copy()
    bc = _c5(best)
    cw = h.copy()
    c5w = bc
    rng = np.random.default_rng(123)
    for it in range(nit):
        if time.time() - t0 > tmax:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        sgn = 1.0 if rng.random() < 0.5 else -1.0
        dv = sgn * rng.uniform(0.005, 0.15)
        h2 = cw.copy()
        h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
        h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        h2 = _project_sum(h2, tgt)
        c5t = _c5(h2)
        if c5t < bc:
            best, bc = h2, c5t
            cw, c5w = h2, c5t
        elif rng.random() < 0.1 and c5t < c5w + 0.01:
            cw, c5w = h2, c5t
        else:
            cw, c5w = best.copy(), bc
    return best, bc

def _try_binarize(h, n):
    """Threshold to 0/1 with exactly n/2 ones"""
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):

    t_start = time.time()

    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _c5(best_h)

    # Phase 1: L-BFGS at various soft-max scales
    for sc in [30.0, 60.0, 120.0, 200.0, 400.0]:
        rem = budget_s - (time.time() - t_start)
        if rem < 15:

        h1 = _lbfgs_opt(best_h, n, dx, sc, 500.0, 2000)
        c1 = _c5(h1)
        if c1 < best_c:

    # Phase 2: Coordinate descent
    rem = budget_s - (time.time() - t_start)
    if rem > 30:
        budget_cd = min(rem * 0.2, 60.0)
        h2 = _coord_descent(best_h.copy(), n, dx, 400.0, 200, tmax=budget_cd, t0=time.time())
        c2 = _c5(h2)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Try binarization
    hb = _try_binarize(best_h, n)
    cb = _c5(hb)
    if cb < best_c:
        best_h, best_c = hb, cb
        # Re-optimize from binarized
        for sc in [100.0, 300.0]:
            rem = budget_s - (time.time() - t_start)
            if rem < 15:
                break
            h1 = _lbfgs_opt(best_h, n, dx, sc, 500.0, 1500)
            c1 = _c5(h1)
            if c1 < best_c:
                best_h, best_c = h1, c1
        h2 = _coord_descent(best_h.copy(), n, dx, 400.0, 100)
        c2 = _c5(h2)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 4: Local search
    rem = budget_s - (time.time() - t_start)
    if rem > 40:
        t0 = time.time()
        h3, c3 = _local_search(best_h, n, min(rem * 0.25, 80.0), 500000, t0)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 5: Upscale to 2n if time permits
    rem = budget_s - (time.time() - t_start)
    n2 = 2 * n
    if n2 <= 720 and rem > 120:
        h_up = _project_sum(np.repeat(best_h, 2).copy(), 0.5 * n2)
        for sc in [80.0, 200.0]:
            rem = budget_s - (time.time() - t_start)
            if rem < 60:
                break
            h_u1 = _lbfgs_opt(h_up, n2, 2.0/n2, sc, 500.0, 1500)
            c_u1 = _c5(h_u1)
            if c_u1 < _c5(h_up):
                h_up = h_u1
        t0 = time.time()

                              tmax=min(rem*0.15, 40.0), t0=t0)
        c_u2 = _c5(h_u2)
        if c_u2 < best_c:
            best_h, best_c = h_u2, c_u2
            n = n2

    # Phase 6: Final local search
    rem = budget_s - (time.time() - t_start)
    if rem > 30:
        t0 = time.time()
        h4, c4 = _local_search(best_h, len(best_h), min(rem * 0.7, 200.0), 800000, t0)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Final polish coordinate descent
    rem = budget_s - (time.time() - t_start)
    if rem > 10:
        nf = len(best_h)
        h5 = _coord_descent(best_h.copy(), nf, 2.0/nf, 500.0, 50)

        if c5 < best_c:
            best_h, best_c = h5, c5

    final_c = _c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return (best_h, best_c, len(best_h))
