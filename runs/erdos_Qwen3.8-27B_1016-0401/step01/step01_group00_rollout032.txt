def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, "full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, "full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, "full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    return grad * dx

    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                s = min(diff, 1.0 - h[i])
                h[i] += s
            else:
                s = min(-diff, h[i])
                h[i] -= s
            diff = target - h.sum()
            if abs(diff) < 1e-14:

        d_scale = t_max * (t_min / t_max) ** frac
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = float(rng.uniform(0, d_scale))
        if rng.random() > 0.5:
            d = -d
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d
        h[j] -= d
