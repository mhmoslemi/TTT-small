def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def pair_coeffs(h, j, k, n):
    """Compute C and D arrays for minimizing max(C*t+D) over pair (j,k) with h[j]=t, h[k]=s-t."""
    corr = compute_corr(h)
    m_arr = np.arange(2*n-1)
    off = m_arr - (n-1)
    lo_i = np.maximum(0, -off)
    hi_i = np.minimum(n, n - off)
    
    # Coefficient of h[j] in corr[m]:
    # Term 1: h[j]*(1-h[j+off]) if j in [lo_i,hi_i) and 0<=j+off<n
    idx1 = j + off

    idx2 = j - off
    v2 = (idx2 >= lo_i) & (idx2 < hi_i) & (idx2 >= 0) & (idx2 < n)
    
    A1 = np.where(v1, 1.0 - np.take(h, np.clip(idx1, 0, n-1), mode='clip'), 0.0)
    A1 -= np.where(v2, np.take(h, np.clip(idx2, 0, n-1), mode='clip'), 0.0)
    
    # Coefficient of h[k] in corr[m]:
    idx1k = k + off
    v1k = (k >= lo_i) & (k < hi_i) & (idx1k >= 0) & (idx1k < n)
    idx2k = k - off
    v2k = (idx2k >= lo_i) & (idx2k < hi_i) & (idx2k >= 0) & (idx2k < n)
    
    A2 = np.where(v1k, 1.0 - np.take(h, np.clip(idx1k, 0, n-1), mode='clip'), 0.0)
    A2 -= np.where(v2k, np.take(h, np.clip(idx2k, 0, n-1), mode='clip'), 0.0)
    
    s = h[j] + h[k]
    C = A1 - A2
    D = A2 * s + (corr - A1 * h[j] - A2 * h[k])

    t_hi = min(1.0, s)

        else:

        h[j] = m2; h[k] = s - m2
        f2 = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_opt = np.clip((lo + hi) / 2.0, t_lo, t_hi)
    h[j] = t_opt
    h[k] = s - t_opt
    return t_opt, s - t_opt

def sweep_sequential(h, n):
    changed = 0.0
    for j in range(n - 1):
        old = h[j]
        pair_opt(h, j, j + 1, n)
        changed += abs(h[j] - old)
    return changed

def sweep_cross(h, n):
    changed = 0.0
    half = n // 2
    for j in range(n):
        k = (j + half) % n
        if k != j:
            old = h[j]
            pair_opt(h, j, k, n)
            changed += abs(h[j] - old)
    return changed

def sweep_random(h, n, rng, count):
    changed = 0.0
    for _ in range(count):
        j = rng.randint(n)
        k = rng.randint(n)
        if j != k:
            old = h[j]
            pair_opt(h, j, k, n)
            changed += abs(h[j] - old)
    return changed

_slsqp_target = 0.0

def _slsqp_obj(x):
    return compute_c5(x)

def _slsqp_constr(x):
    return np.sum(x) - _slsqp_target

def slsqp_refine(h, n, target, maxiter=200):
    global _slsqp_target
    _slsqp_target = target
    try:
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _slsqp_constr}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, n, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Structured pair descent (alternate patterns)
    h_curr = best_h.copy()
    for sweep in range(800):
        if time.time() > deadline - 300:
            break
        if sweep % 3 == 0:
            ch = sweep_sequential(h_curr, n)
        elif sweep % 3 == 1:
            ch = sweep_cross(h_curr, n)
        else:
            ch = sweep_random(h_curr, n, rng, n)
        c5 = compute_c5(h_curr)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_curr.copy()
        if ch < 1e-13:
            break

    # Phase 2: Random pair descent with kicks
    h_curr = best_h.copy()
    for sweep in range(1000):
        if time.time() > deadline - 150:
            break
        ch = sweep_random(h_curr, n, rng, n * 2)
        c5 = compute_c5(h_curr)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_curr.copy()
        if sweep % 200 == 199 and ch < 1e-13:
            kick_idx = rng.choice(n, size=5, replace=False)
            h_curr[kick_idx] += rng.uniform(-0.02, 0.02, size=5)
            h_curr = make_feasible(h_curr, n, target)

    # Phase 3: SLSQP refinement
    if time.time() < deadline - 80:
        for _ in range(2):
            if time.time() > deadline - 80:
                break
            h_new = slsqp_refine(best_h, n, target, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Fine pair descent after SLSQP
    if time.time() < deadline - 30:
        h_curr = best_h.copy()
        for sweep in range(600):
            if time.time() > deadline - 30:
                break
            if sweep % 2 == 0:
                ch = sweep_sequential(h_curr, n, )
            else:
                ch = sweep_cross(h_curr, n)
            c5 = compute_c5(h_curr)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_curr.copy()
            if ch < 1e-14:
                break

    # Phase 5: Multi-restart
    for trial in range(4):
        if time.time() > deadline - 20:
            break
        h_r = best_h + rng.uniform(-0.03, 0.03, n)
        h_r = make_feasible(h_r, n, target)
        c5_r = compute_c5(h_r)
        for sweep in range(200):
            if time.time() > deadline - 20:
                break
            ch = sweep_sequential(h_r, n)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            if ch < 1e-14:
                break
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final validation
    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
