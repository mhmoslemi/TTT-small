```python
def smart_hill_climb(h, t_deadline=None, max_sweeps=10000):
    n = len(h)
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)  # 2n-1
    lags = np.arange(n_corr) - (n - 1)  # lags from -(n-1) to n-1
    c5_raw = corr.max()
    
    for sweep in range(max_sweeps):
        if t_deadline and time.time() > t_deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        
        # For each (i, j) pair, compute delta for all lags
        # delta[i,j,m] = -(1-h[i-lag]) + h[i+lag] + (1-h[j-lag]) - h[j+lag]
        # where h[x] = 0 if x not in [0,n)
        
        # Vectorize: i_idx shape (35,), j_idx shape (35,), lags shape (139,)
        i_arr = one_idx[:, None, None]  # (35, 1, 1)
        j_arr = zero_idx[None, :, None]  # (1, 35, 1)
        lag_arr = lags[None, None, :]  # (1, 1, 139)
        
        # Positions
        i_minus = i_arr - lag_arr  # (35, 35, 139)
        i_plus = i_arr + lag_arr
        j_minus = j_arr - lag_arr
        j_plus = j_arr + lag_arr
        
        # Clamp to [0, n)
        def safe_get(pos):
            # pos is an array, return h[pos] where valid, 0 elsewhere
            mask = (pos >= 0) & (pos < n)
            return np.where(mask, np.take(h, np.clip(pos, 0, n-1), mode='clip'), 0.0)
        
        delta = (-1 + safe_get(i_minus)) + safe_get(i_plus) + (1 - safe_get(j_minus)) - safe_get(j_plus)
        
        new_corr_max = np.max(corr[None, None, :] + delta, axis=2)  # (35, 35)
        
        min_val, min_pos = np.unravel_index(np.argmin(new_corr_max), new_corr_max.shape)
        
        if new_corr_max[min_val, min_pos] < c5_raw - 1e-12:
            i_swap = one_idx[min_val]
            j_swap = zero_idx[min_pos]
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_raw = corr.max()
        else:
            break
    
    return h, c5_raw * (2.0 / n)
```
