    return (m + logsum / sc) * (2.0 / n)

def _smooth_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sc = _G_SC
    vals = np.minimum(sc * (corr - m), 60.0)
    w = np.exp(vals)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        js1 = max(0, -lag)
        je1 = min(n - 1, n - 1 - lag)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 + lag:je1 + 1 + lag])
        js2 = max(0, lag)
        je2 = min(n - 1, n - 1 + lag)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - lag:je2 + 1 - lag]
    return grad * (2.0 / n)

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _obj_sl(h):
    return _smooth_obj(h)

def _grad_sl(h):
    return _smooth_grad(h)

def _ceq(h):
    return h.sum() - 0.5 * _G_N

def _ceqj(h):
    return np.ones(_G_N)

def _slsqp_grad(h0, n, sc, maxiter):
    global _G_N, _G_DX, _G_SC
    _G_N, _G_DX, _G_SC = n, 2.0 / n, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_obj_sl, h0.copy(), jac=_grad_sl, method='SLSQP',
                     bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _grad_local_search(h0, n, tmax, nit, seed=0):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    tgt = 0.5 * n
    cur = h0.copy()
    curc = _c5(cur)
    best = h0.copy()
    bestc = curc
    accepted = 0
    for it in range(nit):
        if time.time() - t0 > tmax:
            break
        if it % 500 == 0:
            g = _smooth_grad(cur)
            g -= g.mean()
            gsign = np.sign(g).astype(float)
        elif it % 50 == 0:
            gsign = np.zeros(n)
        if rng.random() < 0.6 and it % 50 != 0:
            # gradient-guided pair move
            idx = np.argsort(np.abs(gsign))
            n_trials = 3
            for _ in range(n_trials):
                i = idx[rng.randint(len(idx))]
                j = rng.randint(n)
                if i == j:
                    continue
                sgn = -gsign[i] if gsign[i] != 0 else (1.0 if rng.random() < 0.5 else -1.0)
                dv = sgn * rng.uniform(0.02, 0.15)
                h2 = cur.copy()
                h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
                h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
                ds = h2.sum() - tgt
                if abs(ds) > 1e-12:
                    h2 -= ds / n
                    np.clip(h2, 0.0, 1.0, out=h2)
                c5t = _c5(h2)
                if c5t < bestc:
                    best, bestc = h2, c5t
                    cur, curc = h2, c5t
                    accepted += 1
                    break
                elif rng.random() < 0.1 and c5t < curc + 0.015:
                    cur, curc = h2, c5t
                    break
            else:
                cur = best.copy()
                curc = bestc
        else:
            # random block move
            blen = rng.randint(2, 8)
            i = rng.randint(n - blen)
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            dv = sgn * rng.uniform(0.02, 0.12)
            h2 = cur.copy()
            j = rng.randint(n - blen)
            h2[i:i + blen] = np.clip(h2[i:i + blen] + dv, 0.0, 1.0)
            h2[j:j + blen] = np.clip(h2[j:j + blen] - dv, 0.0, 1.0)
            ds = h2.sum() - tgt
            if abs(ds) > 1e-12:
                h2 -= ds / n
                np.clip(h2, 0.0, 1.0, out=h2)
            c5t = _c5(h2)
            if c5t < bestc:
                best, bestc = h2, c5t
                cur, curc = h2, c5t
                accepted += 1
            elif rng.random() < 0.12 and c5t < curc + 0.02:
                cur, curc = h2, c5t
            else:
                if rng.random() < 0.3:
                    cur = best.copy()
                    curc = bestc
    return best, bestc

def _upscale(h0, factor):
    n0 = len(h0)
    n1 = n0 * factor
    x0 = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
    x1 = np.linspace(0, 2, n1, endpoint=False) + 1.0 / n1
    h1 = np.interp(x1, x0, h0, period=2.0)
    return _proj(h1, n1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5

    n0 = len(h)
    bh = h.copy()
    bc = _c5(bh)
    bn = n0

    # Phase 1: SLSQP with gradient at initial resolution
    for sc in [30.0, 50.0, 80.0]:
        rem = budget_s - (time.time() - ts)
        if rem < 30:
            break
        h1 = _slsqp_grad(bh, bn, sc, 3000)
        c1 = _c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2: Gradient-guided local search at n0
    rem = budget_s - (time.time() - ts)
    if rem > 80:
        hl, cl = _grad_local_search(bh, bn, min(rem * 0.12, 40.0), 400000, seed=seed)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 3: Upscale to 2*n0 and optimize
    n1 = 2 * bn
    rem = budget_s - (time.time() - ts)
    if n1 <= 720 and rem > 200:
        h2 = _upscale(bh, 2)
        for sc in [50.0, 80.0, 120.0]:
            rem = budget_s - (time.time() - ts)
            if rem < 40:
                break
            hh = _slsqp_grad(h2, n1, sc, 3000)
            cc = _c5(hh)
            if cc < _c5(h2):
                h2 = hh
        c2 = _c5(h2)
        if c2 < bc:
            bh, bc, bn = h2, c2, n1

        rem = budget_s - (time.time() - ts)
        if rem > 100:
            hl, cl = _grad_local_search(bh, bn, min(rem * 0.2, 150.0), 700000, seed=seed+1)
            if cl < bc:
                bh, bc = hl, cl

    # Phase 4: Try different structure as starting point
    rem = budget_s - (time.time() - ts)
    if rem > 150:
        # Cosine-based starting point
        x = np.linspace(0, 2, n0, endpoint=False)
        h_cos = 0.5 + 0.45 * np.cos(np.pi * x)
        h_cos = _proj(h_cos, n0)
        c_cos = _c5(h_cos)
        if c_cos < bc + 0.05:
            hc = h_cos
            for sc in [40.0, 70.0]:
                rem = budget_s - (time.time() - ts)
                if rem < 40:
                    break
                hh = _slsqp_grad(hc, n0, sc, 2000)
                cc = _c5(hh)
                if cc < _c5(hc):
                    hc = hh
            c_x = _c5(hc)
            rem = budget_s - (time.time() - ts)
            if rem > 40:
                hl, cl = _grad_local_search(hc, n0, min(rem * 0.15, 50.0), 300000, seed=seed+2)
                if cl < c_x:
                    hc, c_x = hl, cl
            if c_x < bc:
                bh, bc, bn = hc, c_x, n0

    # Phase 5: If we're at n0 still, try upscaling once more
    rem = budget_s - (time.time() - ts)
    if bn == n0 and rem > 180:
        n_up = 2 * bn
        if n_up <= 720:
            h_up = _upscale(bh, 2)
            for sc in [60.0, 100.0]:
                rem = budget_s - (time.time() - ts)
                if rem < 40:
                    break
                hh = _slsqp_grad(h_up, n_up, sc, 3000)
                cc = _c5(hh)
                if cc < _c5(h_up):
                    h_up = hh
            c_up = _c5(h_up)
            if c_up < bc:
                bh, bc, bn = h_up, c_up, n_up
            elif c_up < bc + 0.005:
                # still worth local searching
                rem = budget_s - (time.time() - ts)
                if rem > 60:
                    hl, cl = _grad_local_search(h_up, n_up, min(rem * 0.3, 100.0), 500000, seed=seed+3)
                    if cl < bc:
                        bh, bc, bn = hl, cl, n_up

    # Phase 6: Final local search on best
    rem = budget_s - (time.time() - ts)
    if rem > 30:
        t_final = min(rem * 0.7, 120.0)
        hl, cl = _grad_local_search(bh, bn, t_final, 500000, seed=seed + 10)
        if cl < bc:
            bh, bc = hl, cl

    # Final polish with smaller steps
    rem = budget_s - (time.time() - ts)
    if rem > 15:
        t_pol = min(rem * 0.5, 40.0)
        hl, cl = _grad_local_search(bh, bn, t_pol, 200000, seed=seed + 20)
        if cl < bc:
            bh, bc = hl, cl

    fc = _c5(bh)
    return (bh, fc, len(bh))
