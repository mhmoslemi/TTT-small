import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(corr, T):
    mx = np.max(corr)
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def obj_and_grad(h, T, dx):
    n = len(h)
    a = 1.0 - h
    corr = np.correlate(h, a, mode='full')
    mx = corr.max()
    exp_t = np.exp(T * (corr - mx))
    weights = exp_t / exp_t.sum()
    f_val = dx * (mx + np.log(exp_t.sum()) / T)
    
    # Gradient using convolution
    w = weights  # length 2n-1
    a_conv = np.convolve(a, w, 'full')  # length 3n-1
    w_flip = w[::-1]
    h_conv = np.convolve(h, w_flip, 'full')  # length 3n-1
    off = n - 1
    grad = dx * (a_conv[off:off+n] - h_conv[off:off+n])
    
    return f_val, grad

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

    return float(mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T)

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return dx * _smax(corr, _G['T'])

    a = 1.0 - x
    corr = np.correlate(x, a, mode='full')

        if abs(diff) < 1e-12:

    h = h.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() > t_end:

        improved = False
        for i in range(n):

import numpy as np
from scipy.optimize import minimize
import time

_G = {'T': 100.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _smax(corr, T):
    mx = np.max(corr)
    return float(mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T)

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return dx * _smax(corr, _G['T'])

def _grad(x):
    n = len(x)
    dx = 2.0 / n
    a = 1.0 - x
    corr = np.correlate(x, a, mode='full')
    mx = np.max(corr)
    w = np.exp(_G['T'] * (corr - mx))
    w /= w.sum()
    A = np.convolve(a, w, 'full')
    B = np.convolve(x, w[::-1], 'full')
    off = n - 1
    return dx * (A[off:off + n] - B[off:off + n])

def _con_fun(x):
    return x.sum() - x.shape[0] / 2.0

def _con_jac(x):
    return np.ones(x.shape[0])

def project(h, n):
    S = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = S - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_opt(h, n, T, maxiter):
    _G['T'] = T
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h.copy(), jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(res.x, n)
    except Exception:
        return h.copy()

    S = n / 2.0
    for sweep in range(max_sweeps):

        improved = False

                src = i if h[i] > h[j] else j
                dst = j if h[i] > h[j] else i

                    if c5_new < c5 - 1e-15:

                        improved = True
                        break
                    else:
                        h[src] += delta
                        h[dst] -= delta
                        h = project(h, n)
        if not improved:
            break
    return h, c5

                    return h, c5
                i, j = idx[ii], idx[jj]
                if i == j or abs(h[i] - h[j]) < 1e-8:
                    continue
                if h[i] < h[j]:
                    i, j = j, i
                delta = 0.4 * (h[i] - h[j])
                h[i] -= delta
                h[j] += delta
                h = project(h, n)
                c5_new = compute_c5(h)
                if c5_new < c5 - 1e-15:
                    c5 = c5_new
                    improved = True
                else:
                    h2 = h.copy()
                    h2[i] += delta
                    h2[j] -= delta
                    h = project(h2, n)
        if not improved:
            break
    return h, c5

def make_init(n, mode, rng):
    x = np.arange(n)
    if mode == 0:
        h = 0.5 + 0.25 * np.cos(np.pi * x / n * 2) + 0.08 * np.cos(4 * np.pi * x / n)
    elif mode == 1:
        h = 0.5 + 0.15 * np.cos(2 * np.pi * x / n * 3) + 0.1 * np.sin(2 * np.pi * x / n * 5)
    elif mode == 2:
        h = 0.5 + rng.normal(0, 0.12, n)
    elif mode == 3:
        h = 0.5 + 0.2 * np.cos(np.pi * x * x / (n * n))
    elif mode == 4:
        h = 0.5 + 0.3 * np.sign(np.cos(2 * np.pi * x / n))
    else:
        h = np.full(n, 0.5)
    return project(np.clip(h, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_start = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_start = np.full(70, 0.5)

    n0 = len(h_start)
    best_h = project(h_start, n0)
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([n0, 56, 70, 84, 100, 120, 140, 200]))
    T_sched = [5, 20, 80, 300, 1200, 5000, 15000]

    for n in n_list:
        if time.time() - t0 > budget_s * 0.82:
            break
        remaining_n = len(n_list) - n_list.index(n)
        t_phase_end = t0 + (budget_s * 0.84 - (time.time() - t0)) / remaining_n
        t_phase_end = max(t_phase_end, time.time() + 30)

        for mode in [0, 1, 2]:
            if time.time() > t_phase_end:
                break
            if n == n0 and mode == 0:
                h_curr = best_h.copy()
            else:
                h_curr = make_init(n, mode, rng)

            for T in T_sched:
                if time.time() > t_phase_end:
                    break
                h_curr = slsqp_opt(h_curr, n, T, maxiter=300)
                c5 = compute_c5(h_curr)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_curr.copy()

            if time.time() < t_phase_end:
                T_gd = T_sched[min(len(T_sched) - 1, 3)]
                h_gd, c5_gd = grad_descent(h_curr, n, T_gd, t_phase_end, n_iter=5000)
                if c5_gd < best_c5:
                    best_c5 = c5_gd
                    best_h = h_gd.copy()
                h_ls, c5_ls = local_search(h_gd, n, t_phase_end, max_sweeps=30)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Final refinement on best solution
    bn = len(best_h)
    t_final = t0 + budget_s * 0.95
    for T in [50, 200, 800]:
        if time.time() > t_final:
            break
        best_h = slsqp_opt(best_h, bn, T, maxiter=400)
        c5 = compute_c5(best_h)
        if c5 > best_c5:
            best_c5 = c5

    h_gd, c5_gd = grad_descent(best_h, bn, 200, t_final, n_iter=10000)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd

    h_ls, c5_ls = local_search(best_h, bn, t0 + budget_s * 0.97, max_sweeps=80)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
