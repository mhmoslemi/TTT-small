import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = alpha * (corr - m)
    vals = np.clip(vals, -700, 700)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

def slsqp_obj(x, target, alpha):
    return soft_max_c5(x, alpha)

def slsqp_constraint(x, target):
    return np.sum(x) - target

def slsqp_optimize(h, target, alpha, maxiter=200, ftol=1e-12):
    n = len(h)
    try:
        res = minimize(
            lambda x: slsqp_obj(x, target, alpha),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: slsqp_constraint(x, target)}],

        )
        return make_feasible(res.x, n, target)

def simulated_annealing(h, n, target, rng, deadline, t0=5.0, cooling=0.9999, slsqp_every=200):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_h = h.copy()
    curr_c5 = best_c5
    
    step_size = 0.05
    steps_since_improve = 0
    step_count = 0
    
    while time.time() < deadline:
        step_count += 1
        T = t0 * (cooling ** step_count)
        
        # Perturb 2 random coordinates
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            j = (j + 1) % n
        
        delta = rng.uniform(-step_size, step_size)
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = make_feasible(new_h, n, target)
        
        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            curr_h = new_h
            curr_c5 = new_c5
            steps_since_improve = 0

        h_sa, c5_sa = simulated_annealing(
            best_h, n, target, rng, deadline - 80,
            t_init=5.0, cool_rate=0.99996, slsqp_interval=350
        )
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 3: High-alpha SLSQP polish
    if time.time() < deadline - 25:
        for alpha in [3000, 8000, 15000]:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_optimize(best_h, target, alpha, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Try larger n via interpolation
    for n_try in [100, 140, 200]:
        if time.time() > deadline - 35:
            break
        h_interp = np.interp(
            np.linspace(0, 2, n_try),
            np.linspace(0, 2, len(best_h)),
            best_h
        )
        tgt = n_try / 2.0
        h_interp = make_feasible(h_interp, n_try, tgt)
        # SLSQP on the finer grid
        h_fin = slsqp_optimize(h_interp, tgt, 500, maxiter=300)
        if time.time() < deadline - 30:
            h_fin = slsqp_optimize(h_fin, tgt, 2000, maxiter=200)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 5: Final short SA on best solution
    if time.time() < deadline - 15:
        h_fin, c5_fin = simulated_annealing(
            best_h, len(best_h), len(best_h)/2.0, rng, deadline - 10,
            t_init=1.0, cool_rate=0.9999, slsqp_interval=600
        )
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final SLSQP if time allows
    if time.time() < deadline - 5:
        n_f = len(best_h)
        t_f = n_f / 2.0
        h_f = slsqp_optimize(best_h, t_f, 20000, maxiter=200)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
            n_f = len(best_h)

    # Validation
    nf = len(best_h)
    tf = nf / 2.0
    best_h = make_feasible(best_h, nf, tf)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, nf
