    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_obj(h, n, sc):

    corr = np.correlate(h, 1.0 - h, mode="full")

    mcorr = corr.max()

            if 0 <= l2 < n:

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_obj(h, n, sc):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mcorr = corr.max()
    return (mcorr + np.log(np.exp(sc * (corr - mcorr)).sum()) / sc) * dx

def _smooth_grad(h, n, sc):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mcorr = corr.max()
    exps = np.exp(sc * (corr - mcorr))
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            l1 = j + k - n + 1
            if 0 <= l1 < n:
                s += w[k] * (1.0 - h[l1])
            l2 = j - k + n - 1
            if 0 <= l2 < n:
                s -= w[k] * h[l2]
        grad[j] = dx * s
    return grad

def _project(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

            np.clip(h, 0.0, 1.0, out=h)

            if c4 < c2:
                h2, c2 = h4, c4
        if c2 < bc:
            bh, bc, n = h2, c2, n2
    # Phase 5: Final local search
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        t6 = min(rem * 0.5, 200.0)
        h5, c5 = _local_search(bh, n, t6)
        if c5 < bc:
            bh, bc = h5, c5
    bh = _project(bh, len(bh))
    fc = _c5(bh)
    return (bh, fc, len(bh))
