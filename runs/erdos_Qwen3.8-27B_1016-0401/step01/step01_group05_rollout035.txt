```python
def project_to_constraint(v, n, target):
    """Project v onto {0 ≤ x ≤ 1, sum(x) = target}"""
    # Bisection on μ: sum(clip(v - μ, 0, 1)) = target
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2
    return np.clip(v - mu, 0, 1)
```

```python
def subgradient(h, n):
    """Compute subgradient of f(h) = max_m C(m) * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)  # index in [0, 2n-2]
    m_star = k_star - (n - 1)  # actual lag
    
    g = np.zeros(n)
    for i in range(n):
        # (1-h[i+m]) if i+m is in [0,n)
        if 0 <= i + m_star < n:
            g[i] += (1 - h[i + m_star])
        # -h[i-m] if i-m is in [0,n)
        if 0 <= i - m_star < n:
            g[i] -= h[i - m_star]
    return g * dx
```

```python
def smooth_gradient(h, n, alpha=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for m_idx in range(len(corr)):
        m = m_idx - (n - 1)
        w = weights[m_idx]
        if w < 1e-8:
            continue
        for i in range(n):
            if 0 <= i + m < n:
                g[i] += w * (1 - h[i + m])
            if 0 <= i - m < n:
                g[i] -= w * h[i - m]
    return g * dx
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_h(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} via bisection on offset."""
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
```
