import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary_from_continuous(h, S):
    n = len(h)
    h_b = (h >= 0.5).astype(float)
    target = int(round(S))
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def compute_A(h, n):
    """Compute A[p, k_idx] = h[p-lag]*I(0<=p-lag<n) - (1-h[p+lag])*I(0<=p+lag<n)"""
    lags = np.arange(0, 2*n-1) - (n-1)
    A = np.zeros((n, 2*n-1))
    for p in range(n):
        valid_l = (p - lags >= 0) & (p - lags < n)
        A[p, valid_l] += h[p - lags[valid_l]]
        valid_r = (p + lags >= 0) & (p + lags < n)
        A[p, valid_r] -= (1 - h[p + lags[valid_r]])
    return A

def local_search_binary(h, n, max_iters=500, t_deadline=None, kick_size=0):

    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    stagnation = 0
    
    for it in range(max_iters):
        if t_deadline and time.time() > t_deadline:
            break
        
        ones_idx = np.where(h_cur == 1)[0]
        zeros_idx = np.where(h_cur == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        A = compute_A(h_cur, n)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

        new_corr = corr[None, None, :] + A[ones_idx][:, None, :] - A[zeros_idx][None, :, :]
        new_C5 = new_corr.max(axis=2) * dx
        
        best_val = new_C5.min()
        if best_val < c5_cur - 1e-14:
            bi = np.unravel_index(new_C5.argmin(), new_C5.shape)
            i = ones_idx[bi[0]]
            j = zeros_idx[bi[1]]
            h_cur[i] = 0
            h_cur[j] = 1

            stagnation = 0
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        else:
            stagnation += 1
            if kick_size > 0 and stagnation >= kick_size:

                for _ in range(kick_size):
                    oi = np.where(h_cur == 1)[0]

                    h_cur[a], h_cur[b] = h_cur[b], h_cur[a]
                c5_cur = compute_c5(h_cur)
                stagnation = 0
            else:
                break
    
    return best_h, best_c5
