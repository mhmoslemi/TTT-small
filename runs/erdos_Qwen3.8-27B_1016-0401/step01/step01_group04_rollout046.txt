def best_swap(h, c, n):
    """Find the swap (i,j) with i in ones, j in zeros that minimizes max(c + delta_c)."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_val = np.inf
    best_pair = None
    nm1 = n - 1
    
    for i in ones:
        for j in zeros:
            # Compute delta_c
            # Term 1: -(1-h[i+k-(n-1)]) for valid k
            # i+k-(n-1) in [0,n-1] means k in [n-1-i, n-1-i+n-1] = [n-1-i, 2n-2-i]
            # and k in [0, 2n-2]
            
            # Let's just compute directly
            delta = np.zeros(2*n-1)
            # Term 1: -(1-h[i+k-(n-1)])
            idx1 = i + np.arange(2*n-1) - nm1
            mask1 = (idx1 >= 0) & (idx1 < n)
            delta[mask1] -= (1 - h[idx1[mask1]])
            # Term 2: +h[i-k+(n-1)]
            idx2 = i - np.arange(2*n-1) + nm1
            mask2 = (idx2 >= 0) & (idx2 < n)
            delta[mask2] += h[idx2[mask2]]
            # Term 3: +(1-h[j+k-(n-1)])
            idx3 = j + np.arange(2*n-1) - nm1
            mask3 = (idx3 >= 0) & (idx3 < n)
            delta[mask3] += (1 - h[idx3[mask3]])
            # Term 4: -h[j-k+(n-1)]
            idx4 = j - np.arange(2*n-1) + nm1
            mask4 = (idx4 >= 0) & (idx4 < n)
            delta[mask4] -= h[idx4[mask4]]
            
            new_max = np.max(c + delta)
            if new_max < best_val:
                best_val = new_max
                best_pair = (i, j)
    
    return best_pair, best_val

def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def to_binary(h, n):
    S = n // 2
    h_b = (h >= 0.5).astype(float)
    cnt = int(h_b.sum())
    if cnt > S:
        idx = np.where(h_b == 1)[0]
        h_b[idx[np.argsort(h[idx])[:cnt - S]]] = 0
    elif cnt < S:
        idx = np.where(h_b == 0)[0]
        h_b[idx[np.argsort(-h[idx])[:S - cnt]]] = 1
    return h_b

def local_search(h, n, deadline=None, max_rounds=500):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(float)

        best_nm = cmax

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
