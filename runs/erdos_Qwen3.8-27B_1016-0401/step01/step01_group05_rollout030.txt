import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 - 1e-10 if diff > 0 else h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(alpha * (corr - m), -700, 0)
    return (m + np.log(np.sum(np.exp(sh))) / alpha) * dx

def subgrad_descent(h, n, target, max_iters, deadline, top_k=3):
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    step = 0.03
    stagnation = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_idx:
            k = m - (n - 1)
            for j in range(n):
                val = 0.0
                if 0 <= j - k < n:
                    val += 1.0 - h_curr[j - k]
                if 0 <= j + k < n:
                    val -= h_curr[j + k]
                g[j] += val
        g /= top_k
        h_try = make_feasible(h_curr - step * g, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_curr = h_try
            stagnation = 0
            step = min(step * 1.01, 0.06)
        else:
            h_curr = h_try
            stagnation += 1
            if stagnation > 50:
                step *= 0.7
                stagnation = 0
        if it % 200 == 199:
            step *= 0.92
    return best_h, best_c5

def fine_coord_descent(h, n, target, rng, max_iters, deadline):
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    deltas = [0.005, 0.01, 0.02, 0.03, 0.05, -0.005, -0.01, -0.02, -0.03, -0.05]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        improved = False
        for d in deltas:
            h_try = h_curr.copy()
            h_try[i] = np.clip(h_try[i] + d, 0, 1)
            h_try = make_feasible(h_try, n, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_curr = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved and it % 3 == 0:
            j = rng.randint(n - 1)
            if j >= i:
                j += 1
            for d in [0.01, 0.02, 0.04, -0.01, -0.02, -0.04]:
                h_try = h_curr.copy()
                h_try[i] = np.clip(h_try[i] + d, 0, 1)
                h_try[j] = np.clip(h_try[j] - d, 0, 1)
                h_try = make_feasible(h_try, n, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    h_curr = h_try
                    best_c5 = c5_try
                    break
    return h_curr, best_c5

def lfbgbs_opt(h, n, target, alpha, maxiter, deadline):
    def obj(x):
        return logsumexp_obj(np.clip(x, 0, 1), alpha)
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxfun': 5000})
        h_new = make_feasible(res.x, n, target)
        return h_new
    except Exception:
        return h.copy()

def targeted_coord(h, n, target, max_iters, deadline):
    """Coordinate descent targeting the worst shift specifically."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    for it in range(max_iters):

        k = m_worst - (n - 1)
        # Find indices contributing most to this shift
        contrib = np.zeros(n)

        # Try to reduce worst contributors
        order = np.argsort(contrib)[::-1]
        improved = False
        for idx in order[:5]:

                    if c5_try < best_c5:
                        h_curr = h_try

                    h_try = h_curr.copy()
                    h_try[idx] = np.clip(h_try[idx] + d, 0, 1)

                    if c5_try < best_c5:
                        h_curr = h_try
