        if time.time() - t0 > budget_s - 200:

        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac,
                 'args': (target, n)}]
        try:
            res = minimize(obj_fn, h_best, method='SLSQP', jac=jac_fn,
                          args=(T,), bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-15})
            if res.fun < 1e2:
                h_new = fix_constraint(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: Boundary-focused optimization
    # Identify bins that are "boundary" (not clearly 0 or 1)
    boundary_mask = (h_best > 0.01) & (h_best < 0.99)
    n_boundary = boundary_mask.sum()
    if n_boundary > 5 and n_boundary < n - 5 and time.time() - t0 < budget_s - 150:
        boundary_idx = np.where(boundary_mask)[0]
        fixed_vals = h_best.copy()
        fixed_sum = h_best[~boundary_mask].sum()
        b_target = target - fixed_sum

        def b_obj(x, T):
            h_full = fixed_vals.copy()
            h_full[boundary_idx] = x
            return soft_max_obj(h_full, T)

        def b_jac(x, T):
            h_full = fixed_vals.copy()
            h_full[boundary_idx] = x
            g_full = soft_max_grad(h_full, T)
            return g_full[boundary_idx]

        for T in [10, 5, 3, 2, 1]:
            if time.time() - t0 > budget_s - 100:
                break
            try:
                b_bounds = [(0.0, 1.0)] * n_boundary
                res = minimize(penalty_obj, h_best[boundary_idx], method='L-BFGS-B',
                              jac=penalty_jac, args=(T, b_target, 100.0),
                              bounds=b_bounds,
                              options={'maxiter': 2000, 'gtol': 1e-10})
                h_full = fixed_vals.copy()
                h_full[boundary_idx] = np.clip(res.x, 0, 1)
                h_full = fix_constraint(h_full, target)
                c5 = compute_c5(h_full)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_full
                    boundary_mask = (h_best > 0.01) & (h_best < 0.99)
                    boundary_idx = np.where(boundary_mask)[0]
                    fixed_vals = h_best.copy()
                    fixed_sum = h_best[~boundary_mask].sum()
                    b_target = target - fixed_sum
                    n_boundary = boundary_mask.sum()
                    if n_boundary < 5:
                        break
            except Exception:
                pass

    # Phase 4: Projected gradient with momentum for final polish
    h_best, c5_best = subgr_search(h_best, c5_best, n, target, budget_s - 30, t0)

    # Phase 5: Resolution upgrade if time remains
    if time.time() - t0 < budget_s - 60:
        n_hi = n + 70
        target_hi = n_hi / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = fix_constraint(h_hi, target_hi)
        bounds_hi = [(0.0, 1.0)] * n_hi
        for T in [50, 20, 10, 5, 3]:
            if time.time() - t0 > budget_s - 20:
                break
            try:
                res = minimize(penalty_obj, h_hi, method='L-BFGS-B',
                              jac=penalty_jac, args=(T, target_hi, 200.0),
                              bounds=bounds_hi,
                              options={'maxiter': 1500, 'gtol': 1e-10})
                h_new = fix_constraint(res.x, target_hi)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    h_hi = h_new.copy()
            except Exception:
                pass

    # Final SLSQP polish
    n_final = len(h_best)
    target_final = n_final / 2.0
    h_best = fix_constraint(h_best, target_final)
    bounds_f = [(0.0, 1.0)] * n_final
    cons_f = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac,
               'args': (target_final, n_final)}]
    if time.time() - t0 < budget_s - 5:
        try:
            res = minimize(obj_fn, h_best, method='SLSQP', jac=jac_fn,
                          args=(1.0,), bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            if res.fun < 1e2:
                h_new = fix_constraint(res.x, target_final)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
