import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    # Log-sum-exp for numerical stability
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

    """Convert to binary with exactly S_int ones."""
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S_int:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - S_int):
            h_b[one_idx[order[i]]] = 0
    elif ones < S_int:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(S_int - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

    dx = 2.0 / n

    c5 = corr.max() * dx
    best_h = h_hc.copy()
    best_c5 = c5
    
    for sweep in range(max_sweeps):

            break
        improved = False
        one_idx = list(np.where(h_hc == 1)[0])
        zero_idx = list(np.where(h_hc == 0)[0])
        
        for i in one_idx:
            if time.time() > t_deadline:
                break
            for j in zero_idx:
                # Compute delta for each lag
                # Swap: h[i]: 1->0, h[j]: 0->1
                # delta_k for lag k in range [-(n-1), n-1], indexed as k+(n-1) in full array
                new_corr = corr.copy()

                    d = 0.0
                    # Term 1: i' = i, need 0 <= i+k < n
                    if 0 <= i + k < n:
                        d -= (1.0 - h_hc[i + k])
                    # Term 2: i' = i-k, need 0 <= i-k < n
                    if 0 <= i - k < n:

                    # Term 3: i' = j, need 0 <= j+k < n
                    if 0 <= j + k < n:

                    zero_idx.append(i)
                    one_idx.remove(i)
                # else: keep as is (didn't swap)

    """Compute delta for all lags when swapping h[i]:1->0 and h[j]:0->1."""
    delta = np.zeros(2*n-1)
    # For each lag k (absolute), the index in full corr is k + n - 1
    k = np.arange(-(n-1), n)  # all lags

    delta[valid1] -= (1.0 - h[i + k[valid1]])

    valid2 = (i - k >= 0) & (i - k < n)
    delta[valid2] += h[i - k[valid2]]

    valid3 = (j + k >= 0) & (j + k < n)
    delta[valid3] += (1.0 - h[j + k[valid3]])
    
    # Term 4: -h[j-k] where 0<=j-k<n
    valid4 = (j - k >= 0) & (j - k < n)
    delta[valid4] -= h[j - k[valid4]]
    
    return delta

    return float(corr.max() * dx)

def project_to_binary(h, S_int):
    h_b = np.zeros(len(h))
    idx = np.argsort(-h)
    h_b[idx[:S_int]] = 1.0
    return h_b

def smooth_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = corr.max()
    shifted = T * (corr - mx)
    shifted = np.clip(shifted, -500, 500)
    return (mx + np.log(np.sum(np.exp(shifted))) / T) * dx

def slsqp_step(h, n, S, T, maxiter=200):
    bounds = [(0.0, 1.0)] * n

        return smooth_max_obj(x, T)

                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-12})

            x = x / x.sum() * S
            x = np.clip(x, 0, 1)
            return x

        pass

                    best_c5 = float(new_max * dx)
                    best_h = h_wc.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def double_swap_hill_climb(h, n, t_deadline, max_passes=500, rng=None):
    """Try swapping two 1s with two 0s (4-position move)."""
    if rng is None:
        rng = np.random.RandomState(1)
    dx = 2.0 / n
    h_wc = h.copy().astype(float)
    best_c5 = compute_c5(h_wc)
    best_h = h_wc.copy()
    corr = np.correlate(h_wc, 1.0 - h_wc, mode='full').astype(float)

    for p in range(max_passes):
        if time.time() > t_deadline:
            break
        improved = False
        ones = np.where(h_wc == 1)[0]
        zeros = np.where(h_wc == 0)[0]
        n_try = min(500, len(ones) * (len(ones) - 1) // 2 * 2)
        for _ in range(n_try):
            if time.time() > t_deadline:
                return best_h, best_c5
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                continue
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2 or j1 == i1 or j2 == i1 or j1 == i2 or j2 == i2:
                continue
            h_wc[i1] = 0.0
            h_wc[i2] = 0.0
            h_wc[j1] = 1.0
            h_wc[j2] = 1.0
            new_corr = np.correlate(h_wc, 1.0 - h_wc, mode='full').astype(float)
            new_c5 = new_corr.max() * dx
            if new_c5 < best_c5 - 1e-15:
                best_c5 = float(new_c5)
                best_h = h_wc.copy()
                corr = new_corr
                improved = True
            else:
                h_wc[i1] = 1.0
                h_wc[i2] = 1.0
                h_wc[j1] = 0.0
                h_wc[j2] = 0.0
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_parent = None

    best_h = None
    best_c5 = 1.0
    best_n = 70

    if h_parent is None:
        h_parent = make_random_binary(70, rng)

    n_parent = len(h_parent)
    S_parent = n_parent / 2.0

    n_candidates = [n_parent]
    for dn in [-4, -2, 2, 4, -6, 6]:
        nc = n_parent + dn
        if 40 <= nc <= 120 and nc % 2 == 0 and nc not in n_candidates:
            n_candidates.append(nc)
    n_candidates.sort(reverse=True)  # parent first

    # Time allocation
    per_n_time = budget_s * 0.08
    max_n_attempts = len(n_candidates)

    for n in n_candidates:
        if time.time() - t0 > budget_s * 0.92:
            break
        n_start = time.time()
        n_budget = min(per_n_time, budget_s * 0.92 - (n_start - t0))
        S = n / 2.0
        S_int = n // 2

        # Build starting points
        starts = []
        if n_parent == n:
            starts.append(h_parent.copy())
            starts.append(project_to_binary(h_parent, S_int))
        elif h_parent is not None:
            x_old = np.linspace(0, 2, n_parent, endpoint=False) + 1.0 / n_parent
            x_new = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            interp = np.interp(x_new, x_old, h_parent)
            starts.append(np.clip(interp, 0, 1))

        starts.append(make_random_binary(n, rng))
        starts.append(make_random_binary(n, rng))

        for start_h in starts:
            if time.time() - t0 > budget_s * 0.92:
                break

            # Phase A: Continuous SLSQP
            h_cont = np.clip(start_h, 0.0, 1.0)
            if h_cont.sum() > 1e-10:
                h_cont = h_cont / h_cont.sum() * S
            h_cont = np.clip(h_cont, 0.0, 1.0)

            t_cont_end = time.time() + min(n_budget * 0.25, 60)
            for T in [5, 20, 80, 300, 1500, 8000]:
                if time.time() > t_cont_end:
                    break
                h_cont = slsqp_step(h_cont, n, S, T, maxiter=120)

            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
                best_n = n

            # Phase B: Binary hill climb from continuous
            h_bin = project_to_binary(h_cont, S_int)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n

            # Also try binary from raw start
            h_bin2 = project_to_binary(np.clip(start_h, 0, 1), S_int)
            c5_bin2 = compute_c5(h_bin2)
            if c5_bin2 < c5_bin:
                h_bin = h_bin2
                c5_bin = c5_bin2
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n

            # Phase C: Intensive binary hill climbing
            t_hc_end = time.time() + min(n_budget * 0.5, 180)
            h_bc, c5_bc = binary_hill_climb(h_bin, n, t_hc_end,
                                             max_passes=8000, perturb_every=60, rng=rng)
            if c5_bc < best_c5:
                best_c5 = c5_bc
                best_h = h_bc.copy()
                best_n = n

            # Phase D: Double-swap refinement
            if time.time() < t0 + budget_s * 0.90:
                t_ds_end = time.time() + min(30, (t0 + budget_s * 0.90) - time.time())
                h_ds, c5_ds = double_swap_hill_climb(h_bc, n, t_ds_end,
                                                      max_passes=800, rng=rng)
                if c5_ds < best_c5:
                    best_c5 = c5_ds
                    best_h = h_ds.copy()
                    best_n = n
                    # Re-run single-swap HC from double-swap result
                    t_hc2_end = time.time() + 20
                    h_bc2, c5_bc2 = binary_hill_climb(h_ds, n, t_hc2_end,
                                                       max_passes=3000, perturb_every=100, rng=rng)
                    if c5_bc2 < best_c5:
                        best_c5 = c5_bc2
                        best_h = h_bc2.copy()
                        best_n = n

            # Phase E: Continuous from best binary
            if time.time() < t0 + budget_s * 0.90:
                h_c2 = best_h.astype(float).copy() if len(best_h) == n else h_bc.astype(float)
                h_c2 += rng.normal(0, 0.08, len(h_c2))
                h_c2 = np.clip(h_c2, 0, 1)
                if h_c2.sum() > 1e-10:
                    h_c2 = h_c2 / h_c2.sum() * S
                h_c2 = np.clip(h_c2, 0, 1)
                for T in [30, 200, 1000]:
                    if time.time() > t0 + budget_s * 0.91:
                        break
                    h_c2 = slsqp_step(h_c2, n, S, T, maxiter=100)
                c5_c2 = compute_c5(h_c2)
                if c5_c2 < best_c5:
                    best_c5 = c5_c2
                    best_h = h_c2.copy()
                    best_n = n

    if best_h is None:
        best_h = make_random_binary(70, rng)
        best_c5 = compute_c5(best_h)
        best_n = 70

    # Final binary check
    if len(best_h) == best_n and np.all(np.abs(best_h * (1 - best_h)) < 1e-6):
        pass  # Already binary
    else:
        h_fb = project_to_binary(best_h, best_n // 2)
        c5_fb = compute_c5(h_fb)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
