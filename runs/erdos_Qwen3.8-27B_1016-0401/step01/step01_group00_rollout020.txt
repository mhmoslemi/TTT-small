```python
    c[m] = coeff
    
    # Negative lags pn=1..n-1
    pn = np.arange(1, n)
    m2 = np.arange(n-2, -1, -1)
    coeff2 = np.zeros(n-1)
    # First part: (1-h[p-pn]) for pn <= p
    mask3 = pn <= p
    if mask3.any():
        coeff2[mask3] += (1.0 - h[p-1:p-1-mask3.sum()+1][::-1])
        # h[p-pn] for pn=1,2,...,p: h[p-1], h[p-2], ..., h[0]
    # Second part: -h[p+pn] for pn <= n-1-p
    mask4 = pn <= (n-1-p)
    if mask4.any():
        coeff2[mask4] -= h[p+1:p+1+mask4.sum()]
        # h[p+pn] for pn=1,2,...,n-1-p: h[p+1], h[p+2], ..., h[n-1]
    
    c[m2] = coeff2
    return c
```

Now, for the coordinate update, given coeffs (shape 2n-1) and old corr (shape 2n-1):
- For t in [0,1]: new_vals[m] = corr[m] + (t - h[p]) * coeffs[m] for m ≠ n-1
- For m = n-1 (o=0): new_val = corr[n-1] - h[p]*(1-h[p]) + t*(1-t)

So: f(t) = max( max_{m≠n-1}(corr[m] + (t-h[p])*coeffs[m]), corr[n-1] - h[p]*(1-h[p]) + t*(1-t) )

Let me vectorize the evaluation:

```python
def evaluate_coord_update(corr, coeffs, h_p, ts):
    """Evaluate f(t) for array of t values. Returns array of f values."""
    n = len(corr) // 2 + 1  # original n
    # Non-zero lags
    mask = np.ones(len(corr), dtype=bool)
    mask[n-1] = False  # exclude o=0
    # new_vals shape: (len(ts), 2n-1)
    new_vals = corr[None, :] + (ts[:, None] - h_p) * coeffs[None, :]
    max_nz = np.max(new_vals[:, mask], axis=1)
    # o=0
    val_0 = corr[n-1] - h_p * (1 - h_p) + ts * (1 - ts)
    return np.maximum(max_nz, val_0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    # Water-filling projection
    h = np.array(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        tau = (lo + hi) / 2
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    h_new = np.clip(h - tau, 0.0, 1.0)
    # Fix residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-h_new)
        else:
            order = np.argsort(h_new)
        for i in order:
            if diff > 0:
                adj = min(diff, 1.0 - h_new[i])
            else:
                adj = min(-diff, h_new[i])
            h_new[i] += np.sign(diff) * adj
            diff -= np.sign(diff) * adj
            if abs(diff) < 1e-12:
                break
    return h_new

        h = h.copy()
        n_pts = len(h)
        tgt = n_pts / 2.0
        ts = np.linspace(0.0, 1.0, 41)

            if check_time is not None and check_time() > budget_s - 60:
                break
            # Shuffle order
            order = rng_cd.permutation(n_pts)
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode="full")
            # For each coordinate
            for p in order:
                # Compute coefficients
                c = np.zeros(2 * n_pts - 1)
                c[n_pts - 1] = -999
                # Positive lags
                for o in range(1, n_pts):
                    m = n_pts - 1 + o
                    coeff = 0.0
                    if p + o < n_pts:
                        coeff += (1.0 - h[p + o])
                    if p - o >= 0:

                    c[m] = coeff
                # Negative lags
                for pn in range(1, n_pts):
                    m = n_pts - 1 - pn
                    coeff = 0.0
                    if p - pn >= 0:
                        coeff += (1.0 - h[p - pn])
                    if p + pn < n_pts:
                        coeff -= h[p + pn]
                    c[m] = coeff
                
                # Evaluate at grid points
                # Non-zero lags
                mask = np.ones(2 * n_pts - 1, dtype=bool)
                mask[n_pts - 1] = False
                new_vals = corr[None, :] + (ts[:, None] - h[p]) * c[None, :]
                max_nz = np.max(new_vals[:, mask], axis=1)
                val_0 = corr[n_pts - 1] - h[p] * (1 - h[p]) + ts * (1 - ts)
```
