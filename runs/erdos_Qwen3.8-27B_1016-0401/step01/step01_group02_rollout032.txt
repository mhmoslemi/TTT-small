import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0-h, mode="full").max() * (2.0/n))

def _proj_sum(h, n):
    """Project h to {0<=h_i<=1, sum(h)=n/2}"""
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    # Bisection for lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def _grad_sm(h, sc):
    """Gradient of smooth max of correlate(h, 1-h, 'full') w.r.t. h"""
    n = len(h)
    c = np.correlate(h, 1.0-h, mode="full")
    m = c.max()
    w = np.exp(sc*(c-m))
    ws = w.sum()
    w /= ws
    g = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2*n-1):
            mk = k - (n-1)
            if mk >= 0:
                ds = 1.0 if i >= mk else 0.0
            else:
                ds = 1.0 if i <= k else 0.0
            dr = 0.0
            im = i - mk
            ip = i + mk
            if 0 <= im < n:
                dr += h[im]
            if 0 <= ip < n:
                dr += h[ip]
            s += w[k] * (ds - dr)
        g[i] = s
    return g

def _pgd(h0, n, sc, tmax, lr_init=0.001, mom=0.85):
    """Projected gradient descent with momentum"""
    t0 = time.time()
    h = h0.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c = _c5(h)
    lr = lr_init
    step = 0
    bad = 0
    while time.time() - t0 < tmax:
        g = _grad_sm(h, sc)
        v = mom * v - lr * g
        h_new = _proj_sum(h + v, n)
        c_new = _c5(h_new)
        if c_new < best_c:
            best = h_new.copy()
            best_c = c_new
            h = h_new
            bad = 0
        else:
            bad += 1
            if bad > 50:
                lr *= 0.9
                v *= 0.5
                bad = 0
            h = h  # stay
        step += 1
        if step % 200 == 0:
            lr *= 0.995
    return best, best_c

def _local_search(h0, n, tmax, n_iter=500000):
    """Random local search with sum-preserving moves"""
    t0 = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = _c5(h)
    target = 0.5 * n
    for _ in range(n_iter):
        if time.time() - t0 > tmax:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Random amount to move
        dv = np.random.uniform(-0.5, 0.5)
        h2 = h.copy()
        h2[i] += dv
        h2[j] -= dv
        h2[i] = min(max(h2[i], 0.0), 1.0)
        h2[j] = min(max(h2[j], 0.0), 1.0)
        # Fix sum
        ds = h2.sum() - target
        if abs(ds) > 1e-10:
            h2 -= ds / n
            h2 = np.clip(h2, 0.0, 1.0)
        c2 = _c5(h2)
        if c2 < best_c - 1e-15:
            best = h2
            best_c = c2
            h = h2
        elif c2 < best_c + 0.01 and np.random.random() < 0.1:
            h = h2  # explore
    return best, best_c

def _slsqp_opt(h0, n, sc, tmax):
    """SLSQP with exact gradient and smooth max"""
    global _G_PARAMS
    _G_PARAMS = (n, sc)
    
    def obj(x):
        h = np.clip(x, 0, 1)
        c = np.correlate(h, 1-h, mode="full")
        m = c.max()
        return (m + np.log(np.exp(sc*(c-m)).sum())/sc) * (2.0/n)

        g = _grad_sm(h, sc) * (2.0/n)
        # Zero out where clipped
        g[x <= 0] = 0
        g[x >= 1] = 0
        return g
    
    def con_fun(x):
        return x.sum() - 0.5*n
    
    def con_jac(x):
        return np.ones(n)
    
    t0 = time.time()

    r = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                 bounds=bds, constraints=cons,
                 options={'maxiter': 5000, 'ftol': 1e-16})
    return _proj_sum(r.x, n)
