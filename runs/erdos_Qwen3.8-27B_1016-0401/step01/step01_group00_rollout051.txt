def soft_max_obj(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    return (mc + np.log(np.sum(e)) / T) * (2.0 / n)

def soft_max_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:

        if o == 0:
            grad += w * (1.0 - 2.0 * h)

            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:

            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    return grad * dx

                h[i] -= s
