def subgradient_step(h, n, alpha):
    dx = 2.0 / n
    # Compute all correlations
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    # corr has length 2n-1, centered at n-1
    k_max = np.argmax(corr)
    shift = k_max - (n - 1)  # actual shift value
    
    # Compute gradient of corr[k_max] with respect to h
    # corr[k_max] = sum_i h[i] * (1-h[i+shift]) for valid i
    # (depending on np.correlate's convention)
    
    # Actually, np.correlate(a, v, mode="full")[n-1+k] = sum_i a[i+k]*v[i]
    # So corr[n-1+shift] = sum_i h[i+shift] * (1-h[i])
    
    grad = np.zeros(n)
    if shift >= 0:
        # h[i+shift] * (1-h[i]) for i in [0, n-1-shift]
        # d/dh[j]: (1-h[j-shift]) if j-shift >= 0 [from h[j]=h[i+shift], i=j-shift]
        #          -h[j+shift] if j+shift < n [from (1-h[j]), i=j]

        # h[i+shift] term: for each i, contributes to h[i+shift]
        for j in range(n):
            g = 0.0
            # From h[i+shift] when i+shift = j, i.e., i = j-shift
            if j - shift >= 0 and j - shift < n - shift:
                g += (1.0 - h[j - shift])
            # From (1-h[i]) when i = j, i.e., h[j+shift]
            if j + shift < n:
                g -= h[j + shift]
            grad[j] = g * dx

            g = 0.0

            # From (1-h[i]) when i = j, i.e., h[j-s]
            if j - s >= 0:
                g -= h[j - s]
            grad[j] = g * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search on lambda
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.clip(y - mid, 0.0, 1.0)
        s = val.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def subgradient_grad(h, n, shift):
    """Compute gradient of C(shift) = sum_i h[i+shift]*(1-h[i]) * dx with respect to h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        # C = sum_{i=0}^{n-1-shift} h[i+shift]*(1-h[i])
        # dC/dh[j] = (1-h[j-shift])*1[j>=shift and j<=n-1] - h[j+shift]*1[j+shift<n]
        mask1 = np.arange(n) >= shift
        mask2 = np.arange(n) < (n - shift)
        grad[mask1] += (1.0 - h[np.arange(n)[mask1] - shift]) 
        grad[mask2] -= h[np.arange(n)[mask2] + shift]
    else:
        s = -shift
        # C = sum_{i=s}^{n-1} h[i-s]*(1-h[i])
        # dC/dh[j] = (1-h[j+s])*1[j+s<n] - h[j-s]*1[j>=s]
        mask1 = np.arange(n) < (n - s)
        mask2 = np.arange(n) >= s
        grad[mask1] += (1.0 - h[np.arange(n)[mask1] + s])
        grad[mask2] -= h[np.arange(n)[mask2] - s]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    
    rng = np.random.default_rng(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Projected subgradient method
    # Step size schedule
    max_iters = 500000
    for it in range(max_iters):
        if elapsed() > budget_s - 300:
            break
        if it % 5000 == 0 and elapsed() - t0 > 10:
            pass  # timing check
        
        dx = 2.0 / n
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        
        # Compute subgradient
        grad = subgradient_grad(h_best, n, shift)
        
        # Step size: decreasing
        alpha = 0.1 / (1.0 + it * 0.0001)
        
        h_new = h_best - alpha * grad
        h_new = project_feasible(h_new, n)
        
        # Check if improved (every 1000 iters)
        if it % 5000 == 0:
            c5_cur = compute_c5(h_new)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_new.copy()
        else:
            h_best = h_new
    
    # Phase 2: SLSQP polish
    # ... (similar to before but with better initialization)
    
    c5_final = min(compute_c5(h_best), c5_best)

        for T in [16, 4, 1]:
            if time.time() - t0 > budget_s - 60:
                break
            h_new = do_slsqp(h_best, n, T, 3000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
