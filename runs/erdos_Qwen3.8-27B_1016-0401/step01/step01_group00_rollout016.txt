    return g

def coordinate_polish(h, n, n_sweeps=200):
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c5_init = compute_c5(h)
    for sweep in range(n_sweeps):
        perm = np.random.permutation(n)
        improved = False
        for j in perm:
            others_sum = h.sum() - h[j]
            def f_1d(v):
                h[j] = np.clip(v, 0, 1)
                corr = np.correlate(h, 1 - h, mode="full")
                return np.max(corr) * dx
            res = minimize_scalar(f_1d, bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-10})
            h[j] = np.clip(res.x, 0, 1)
            improved = True
        h = project_simplex(h, target)
        if not improved:
            break
    return h

def make_bounds(n):
    return [(0.0, 1.0)] * n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_simplex(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Subgradient descent at parent resolution
    h_curr = h_best.copy()
    for it in range(3000):
        if elapsed() > budget_s - 300:
            break
        step = 0.3 / np.sqrt(it + 5)
        g, _ = subgrad_active(h_curr, n, n_active=3)
        h_curr = project_simplex(h_curr - step * g, target)
        if it % 100 == 0:
            c5_c = compute_c5(h_curr)
            if c5_c < c5_best:
                c5_best = c5_c
                h_best = h_curr.copy()

    # Phase 2: L-BFGS-B with power-sum at multiple temps
    dx = 2.0 / n
    bounds = make_bounds(n)
    for p in [40, 80, 150]:
        if elapsed() > budget_s - 200:
            break
        for pen in [10, 50, 200]:
            if elapsed() > budget_s - 150:
                break
            h_try = h_best + rng.normal(0, 0.01, n)
            h_try = project_simplex(h_try, target)
            try:
                res = minimize(ps_obj, h_try, jac=lambda x: ps_grad(x, n, p, pen, target, dx),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12,
                                        'maxls': 50})
                h_try = project_simplex(res.x, target)
                c5_t = compute_c5(h_try)
                if c5_t < c5_best:
                    c5_best = c5_t
                    h_best = h_try.copy()
            except Exception:
                pass

    # Phase 3: Work at lower resolution for global search
    for n_low in [70, 100, 140]:
        if elapsed() > budget_s - 150:
            break
        if n_low >= n:
            continue
        x_old = np.arange(n) * dx
        x_new = np.linspace(0, 2.0, n_low, endpoint=False)
        h_low = np.interp(x_new, x_old, h_best)
        h_low = project_simplex(h_low, n_low / 2.0)
        tgt_low = n_low / 2.0
        dx_low = 2.0 / n_low
        bounds_low = make_bounds(n_low)

        for attempt in range(6):
            if elapsed() > budget_s - 100:
                break
            if attempt == 0:
                h_init = h_low.copy()
            elif attempt == 1:
                h_init = (rng.random(n_low) >= 0.5).astype(float)
                h_init = project_simplex(h_init, tgt_low)
            elif attempt == 2:
                h_init = np.zeros(n_low)
                h_init[::2] = 1.0
                h_init = np.roll(h_init, int(rng.integers(-20, 21)))
                h_init = project_simplex(h_init, tgt_low)
            else:
                h_init = h_low + rng.normal(0, 0.15, n_low)
                h_init = project_simplex(h_init, tgt_low)

            # Subgradient at low res
            h_sg = h_init.copy()
            for it in range(2000):
                if elapsed() > budget_s - 80:
                    break
                step = 0.3 / np.sqrt(it + 5)
                g, _ = subgrad_active(h_sg, n_low, n_active=3)
                h_sg = project_simplex(h_sg - step * g, tgt_low)

            # L-BFGS-B polish
            for pp in [50, 120]:
                if elapsed() > budget_s - 70:
                    break
                try:
                    res = minimize(ps_obj, h_sg,
                                   jac=lambda x: ps_grad(x, n_low, pp, 100, tgt_low, dx_low),
                                   method='L-BFGS-B', bounds=bounds_low,
                                   options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12})
                    h_sg = project_simplex(res.x, tgt_low)
                except Exception:
                    pass

            # Upsample and evaluate
            h_up = np.interp(x_old, x_new, h_sg)
            h_up = project_simplex(h_up, target)
            c5_u = compute_c5(h_up)
            if c5_u < c5_best:
                c5_best = c5_u
                h_best = h_up.copy()

    # Phase 4: Coordinate descent polish at full resolution
    if elapsed() < budget_s - 60:
        h_cd = h_best.copy()
        h_cd = coordinate_polish(h_cd, n, n_sweeps=300)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 5: Final L-BFGS-B refinement
    if elapsed() < budget_s - 30:
        for pp in [200, 400]:
            if elapsed() > budget_s - 10:
                break
            try:
                h_try = h_best + rng.normal(0, 0.005, n)
                h_try = project_simplex(h_try, target)
                res = minimize(ps_obj, h_try,
                               jac=lambda x: ps_grad(x, n, pp, 500, target, dx),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                                        'maxls': 80})
                h_try = project_simplex(res.x, target)
                c5_t = compute_c5(h_try)
                if c5_t < c5_best:
                    c5_best = c5_t
                    h_best = h_try.copy()
            except Exception:
                pass

    # Phase 6: Short subgradient polish
    if elapsed() < budget_s - 5:
        h_sg2 = h_best.copy()
        for it in range(1000):
            if elapsed() > budget_s - 2:
                break
            step = 0.1 / np.sqrt(it + 50)
            g, _ = subgrad_active(h_sg2, n, n_active=2)
            h_sg2 = project_simplex(h_sg2 - step * g, target)
            if it % 50 == 0:
                c5_s = compute_c5(h_sg2)
                if c5_s < c5_best:
                    c5_best = c5_s
                    h_best = h_sg2.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = project_simplex(h_best, target)
    return (h_best, c5_final, len(h_best))
