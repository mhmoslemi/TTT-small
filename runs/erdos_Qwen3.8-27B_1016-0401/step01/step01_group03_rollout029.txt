import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary(h, ts):
    n = len(h)
    b = (h > 0.5).astype(float)
    oc = int(np.sum(b))
    if oc > ts:
        idxs = np.where(b == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            b[i] = 0.0
    elif oc < ts:
        idxs = np.where(b == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            b[i] = 1.0
    return b

def softmax_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        p = max(0, -m)
        q = min(n, n - m)
        for i in range(p, q):
            grad[i] += wk * (1.0 - h[i + m])
            grad[i + m] += wk * (-h[i])
    return obj, grad

            lr *= 0.96
        h = h_new
