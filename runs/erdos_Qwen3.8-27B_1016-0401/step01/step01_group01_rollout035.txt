    return dx * (c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau))))

    h = np.clip(h, 0, 1)

        if abs(s - target) < 1e-12:

        else:

            if mask.sum() == 0:

            h[mask] += diff / mask.sum()

def _coord_descent_fast(h, dx, n_sweeps=30):
    n = len(h)
    h = h.copy()

    for sweep in range(n_sweeps):
        for i in range(n):
            # Compute D_k (coefficient of h[i] in corr(k))
            D = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                s = k_idx - (n - 1)
                d = 0.0
                if 0 <= i + s < n:
                    d += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    d += (-h[i - s])
                D[k_idx] = d
            
            # C_k = c[k] - h[i] * D[k]
            C = c - h[i] * D
            
            # Minimize max_k (C_k + D_k * x) over x in [0,1]
            # Ternary search
            def f_1d(x):
                return np.max(C + D * x)
            
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if f_1d(m1) < f_1d(m2):
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo + hi) / 2.0
            
            # Update h and c incrementally
            delta = x_opt - h[i]
            if abs(delta) > 1e-15:
                h[i] = x_opt
                c += delta * D
        
        h = _project(h, target)
        # Recompute c to avoid drift
        c = np.correlate(h, 1.0 - h, mode='full')
    
    return h

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _obj_tau(h, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    if tau < 1e-12:
        return cm * dx
    return dx * (cm + tau * np.log(np.sum(np.exp((c - cm) / tau))))

def _jac_tau(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    ks = np.arange(n)
    if tau < 1e-12:
        idx = int(np.argmax(c))
        s = idx - (n - 1)
        g = np.zeros(n)
        v1 = (ks - s >= 0) & (ks - s < n)
        g[v1] += (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        g[v2] += (-h[ks[v2] + s])
        return dx * g
    w = np.exp((c - cm) / tau)
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        s = m - (n - 1)
        g = np.zeros(n)
        v1 = (ks - s >= 0) & (ks - s < n)
        g[v1] += (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        g[v2] += (-h[ks[v2] + s])
        grad += w[m] * g
    return dx * grad

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

        if _eval_1d(C, D, m1) < _eval_1d(C, D, m2):
            hi = m2
        else:

    n = len(h)
    h = h.copy()
    target = n / 2.0
    for sweep in range(n_sweeps):
        c = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            D = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                s = k_idx - (n - 1)
                d = 0.0
                if 0 <= i + s < n:
                    d += (1.0 - h[i + s])
                if 0 <= i - s < n:

                D[k_idx] = d

            delta = x_opt - h[i]
            if abs(delta) > 1e-15:

                c += delta * D
        h = _project(h, target)

def _slsqp(h_init, n, tau, max_iter):

    target = n / 2.0
