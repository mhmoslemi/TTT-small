import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr.max()) * dx)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def to_binary(h, S):
    n = len(h)
    target = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def steepest_descend_binary(h0, max_sweeps=1000, t_deadline=None):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        best_delta = 0
        best_a, best_b = -1, -1
        for a in one_idx:
            for b in zero_idx:
                h[a], h[b] = h[b], h[a]
                c = compute_c5(h)
                delta = best_c5 - c
                if delta > best_delta:
                    best_delta = delta
                    best_a, best_b = a, b
                h[a], h[b] = h[b], h[a]
        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]
            best_c5 = compute_c5(h)
            best_h = h.copy()
        else:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, S):
    target = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def steepest_descend_binary(h0, max_sweeps=2000, t_deadline=None, patience=5):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    no_improve = 0
    for sweep in range(max_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        best_delta = 1e-15
        best_a, best_b = -1, -1
        for a in one_idx:
            for b in zero_idx:
                h[a], h[b] = h[b], h[a]
                c = compute_c5(h)
                delta = best_c5 - c
                if delta > best_delta:
                    best_delta = delta
                    best_a, best_b = a, b
                h[a], h[b] = h[b], h[a]
        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]
            best_c5 = compute_c5(h)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
    return best_h, best_c5

def lbbf_optimize(h0, n, S, T, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    def obj(x):
        return smooth_max_obj(x, T)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', bounds=bounds,
                       constraints=constraint, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return normalize_h(res.x, S)
    except Exception:
        return h0.copy()

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def perturbed_binary(h, n, rng, k=3):
    h2 = h.copy()
    one_idx = np.where(h2 == 1)[0]
    zero_idx = np.where(h2 == 0)[0]
    for _ in range(k):
        if len(one_idx) > 0 and len(zero_idx) > 0:
            a = one_idx[rng.randint(len(one_idx))]
            b = zero_idx[rng.randint(len(zero_idx))]
            h2[a], h2[b] = h2[b], h2[a]
            one_idx = np.where(h2 == 1)[0]
            zero_idx = np.where(h2 == 0)[0]
    return h2
