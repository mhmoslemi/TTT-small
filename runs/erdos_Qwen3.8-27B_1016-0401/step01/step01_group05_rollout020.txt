        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)

def lse_obj(h, alpha, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700, 700)
    lse = (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    s = np.sum(h) - target
    lse += penalty * s * s
    return lse

def slsqp_opt(h, n, target, alpha, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return lse_obj(x, alpha, target, penalty)
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, target)
    except:
        return h.copy()

    n = len(h)

        if 0 <= jk < n:
            g -= h[jk]
        grad[j] = g
    h_new = h - lr * grad * dx
