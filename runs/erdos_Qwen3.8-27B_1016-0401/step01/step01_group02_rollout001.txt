        c1 = _c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 7: Another directed search pass (different seed)
    if rem() > 60:
        t1 = min(rem() * 0.2, 130.0)
        hl, cl = _directed_search(bh, bn, t1, seed + 999)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 8: Coordinate descent polish
    if rem() > 20:
        t1 = min(rem() * 0.3, 80.0)
        hl, cl = _coord_descent(bh, bn, t1, seed + 1500)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 9: Final tiny SLSQP
    if rem() > 25:
        h1 = _soft_slsqp(bh, bn, 150.0, 1000)
        c1 = _c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Final directed search polish
    if rem() > 15:
        t1 = min(rem() * 0.5, 60.0)
        hl, cl = _directed_search(bh, bn, t1, seed + 2500)
        if cl < bc:
            bh, bc = hl, cl

    fc = _c5(bh)
    return (bh, fc, len(bh))
