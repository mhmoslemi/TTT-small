    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-15})
        return normalize_h(res.x, S)
    except Exception:
        return h0.copy()

def hill_climb_binary(h0, n, S, max_sweeps=300, t_deadline=None):
    h_hc = h0.copy()
    c5 = compute_c5(h_hc)
    best_h = h_hc.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        improved = False
        oi = np.where(h_hc == 1)[0]
        zi = np.where(h_hc == 0)[0]
        for i in oi:
            if t_deadline is not None and time.time() > t_deadline:
                break
            for j in zi:
                h_hc[i], h_hc[j] = h_hc[j], h_hc[i]
                c5_new = compute_c5(h_hc)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_hc.copy()
                    improved = True
                else:
                    h_hc[i], h_hc[j] = h_hc[j], h_hc[i]
        if not improved:
            break
    return best_h, best_c5

def sa_binary(h0, n, S, rng, t_iter, t_deadline, best_c5_in):
    h_sa = h0.copy()
    c5_sa = compute_c5(h_sa)
    best_local = h_sa.copy()
    bc5 = min(c5_sa, best_c5_in)
    temp = 0.0025
    no_imp = 0
    it = 0
    while time.time() < t_deadline and it < t_iter:
        it += 1
        if no_imp > 25000:
            break
        if it % 2000 == 0 and it > 0:
            temp *= 0.985
        i = rng.randint(n)
        j = rng.randint(n)
        if h_sa[i] == h_sa[j]:
            continue
        h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        c5_new = compute_c5(h_sa)
        d = c5_new - c5_sa
        if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-12)):
            c5_sa = c5_new
            if c5_new < bc5:
                bc5 = c5_new
                best_local = h_sa.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            no_imp += 1
    return best_local, bc5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(70, 0.5)

    n = len(h0)
    S = n / 2.0
    h0 = normalize_h(h0, S)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuous SLSQP - smooth max with annealing
    T_list = [5, 15, 50, 200, 800, 3000]
    for T in T_list:
        if elapsed() > budget_s * 0.18:
            break
        h_new = slsqp_run(best_h, n, S, lambda x: slsqp_obj_smoothmax(x, T), 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: p-norm and top-k objectives
    for p in [8, 20, 50]:
        if elapsed() > budget_s * 0.30:
            break
        h_new = slsqp_run(best_h, n, S, lambda x: slsqp_obj_pnorm(x, p), 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    for k in [2, 3, 5]:
        if elapsed() > budget_s * 0.38:
            break
        h_new = slsqp_run(best_h, n, S, lambda x: slsqp_obj_topk(x, k), 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Multi-start: perturb and re-optimize
    for _ in range(3):
        if elapsed() > budget_s * 0.45:
            break
        h_pert = best_h + rng.normal(0, 0.05, n)
        h_pert = normalize_h(h_pert, S)
        for T in [30, 200]:
            h_new = slsqp_run(h_pert, n, S, lambda x: slsqp_obj_smoothmax(x, T), 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                h_pert = h_new

    # Phase 3: Binary optimization
    h_bin = to_binary(best_h, S)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    # Hill climb from binary
    h_hc, c5_hc = hill_climb_binary(h_bin, n, S, 200, t0 + budget_s * 0.55)
    if c5_hc < best_c5:
        best_h = h_hc
        best_c5 = c5_hc

    # SA with restarts
    t_sa_end = t0 + budget_s * 0.72
    h_sa, c5_sa = sa_binary(best_h, n, S, rng, 60000, t_sa_end, best_c5)
    if c5_sa < best_c5:
        best_h = h_sa
        best_c5 = c5_sa

    # Second SA from perturbed best
    if elapsed() < budget_s * 0.72:
        h_pert2 = best_h.copy()
        oi = np.where(h_pert2 == 1)[0]
        zi = np.where(h_pert2 == 0)[0]
        for _ in range(5):
            i, j = rng.choice(oi, 2, replace=False), rng.choice(zi, 2, replace=False)
            h_pert2[i], h_pert2[j] = h_pert2[j], h_pert2[i]
        h_sa2, c5_sa2 = sa_binary(h_pert2, n, S, rng, 40000, t_sa_end, best_c5)
        if c5_sa2 < best_c5:
            best_h = h_sa2
            best_c5 = c5_sa2

    # More hill climbing
    h_hc2, c5_hc2 = hill_climb_binary(best_h, n, S, 300, t0 + budget_s * 0.82)
    if c5_hc2 < best_c5:
        best_h = h_hc2
        best_c5 = c5_hc2

    # Phase 4: Continuous refinement from best binary
    if elapsed() < budget_s * 0.85:
        h_cont = best_h.astype(float) + rng.normal(0, 0.02, n)
        h_cont = normalize_h(h_cont, S)
        for T in [50, 300, 1500]:
            if elapsed() > budget_s * 0.90:
                break
            h_new = slsqp_run(h_cont, n, S, lambda x: slsqp_obj_smoothmax(x, T), 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_cont = h_new

    # Top-k refinement
    if elapsed() < budget_s * 0.92:
        for k in [3, 5]:
            h_new = slsqp_run(best_h, n, S, lambda x: slsqp_obj_topk(x, k), 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Final binary check
    if elapsed() < budget_s * 0.94:
        h_fb = to_binary(best_h, S)
        c5_fb = compute_c5(h_fb)
        if c5_fb < best_c5:
            best_h = h_fb
            best_c5 = c5_fb

    # Final hill climb
    h_fin, c5_fin = hill_climb_binary(best_h, n, S, 200, t0 + budget_s * 0.97)
    if c5_fin < best_c5:
        best_h = h_fin
        best_c5 = c5_fin

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
