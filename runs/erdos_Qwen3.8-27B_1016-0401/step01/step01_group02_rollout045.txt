import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth(h, sc):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    return (m + np.log(np.exp(sc * (c - m)).sum()) / sc) * dx

def _smooth_g(h, sc):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    e = np.exp(sc * (c - m))
    e /= e.sum()
    active = np.where(e > 1e-10)[0]
    g = np.zeros(n)
    for k in active:
        wk = e[k]
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            s1 = lo1 + k - n + 1
            g[lo1:hi1 + 1] += wk * (1.0 - h[s1:s1 + hi1 - lo1 + 1])
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k)
        if lo2 <= hi2:
            s2 = lo2 - k + n - 1
            g[lo2:hi2 + 1] -= wk * h[s2:s2 + hi2 - lo2 + 1]
    return g * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0 - 1e-12) if r > 0 else (h > 1e-12)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _slsqp(h0, sc, mxit, ftl):
    n = len(h0)
    def _f(x):
        return _smooth(np.clip(x, 0, 1), sc)
    def _df(x):
        return _smooth_g(np.clip(x, 0, 1), sc)
    def _c(x):
        return x.sum() - 0.5 * n
    def _dc(x):
        return np.ones(n)
    try:
        res = minimize(_f, h0, jac=_df, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _c, 'jac': _dc}],
                       options={'maxiter': mxit, 'ftol': ftl, 'iprint': 0})
        hx = _proj(res.x, n)
        return hx if _c5(hx) <= _c5(h0) + 1e-15 else _proj(h0, n)
    except Exception:
        return _proj(h0, n)

def _pgd(h0, n, steps, lr, sc):
    h = h0.copy()
    for _ in range(steps):
        g = _smooth_g(h, sc)
        h -= lr * g
        h = _proj(h, n)
    return h

def _lsearch(h0, n, tmax, niter, rng):
    tgt = 0.5 * n
    best = h0.copy()
    bestc = _c5(best)
    cur = h0.copy()
    curc = bestc
    for _ in range(niter):
        if time.time() > tmax:
            break
        r = rng.random()
        if r < 0.4:
            # Gradient-informed pair
            g = _smooth_g(cur, 80.0)
            top = np.argsort(np.abs(g))[-20:]
            i = top[rng.randint(len(top))]
            j = rng.randint(n)
            if i == j:
                j = (i + 1) % n
            dv = rng.uniform(0.005, 0.12) * (-np.sign(g[i]))
        elif r < 0.6:
            # Block shift
            blen = rng.randint(3, min(9, n // 4))
            st = rng.randint(0, n - blen)
            idx = np.arange(st, st + blen)
            j = rng.randint(n)
            while j in idx:
                j = rng.randint(n)
            dv = rng.uniform(-0.08, 0.08)
        else:
            # Random pair
            i = rng.randint(n)
            j = rng.randint(n)
            while j == i:
                j = rng.randint(n)
            dv = rng.uniform(0.005, 0.15) * (1 if rng.random() < 0.5 else -1)
            blen = 1
            st = i
            idx = np.array([i])

        h2 = cur.copy()
        if r >= 0.6:
            h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - (h2[i] - cur[i]), 0.0, 1.0)
        else:
            h2[idx] = np.clip(h2[idx] + dv, 0.0, 1.0)
            adj = (h2[idx].sum() - cur[idx].sum())
            h2[j] = np.clip(h2[j] - adj, 0.0, 1.0)
        # Fix sum
        h2 -= (h2.sum() - tgt) / n
        np.clip(h2, 0.0, 1.0, out=h2)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2, c2
            cur, curc = h2, c2
        elif rng.random() < 0.12 and c2 < curc + 0.015:
            cur, curc = h2, c2
        else:
            cur = best.copy()
            curc = bestc
    return best, bestc

def _upsample(h, factor):
    flat = np.repeat(h, factor)
    return flat
