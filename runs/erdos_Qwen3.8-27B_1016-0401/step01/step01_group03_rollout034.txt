def compute_c5(h, dx):

    return np.max(corr) * dx

def project_sum_bounds(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1} using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(80):

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    mc = np.max(corr)
    sh = corr - mc
    ec = np.exp(np.clip(beta * sh, -100, 100))
    se = np.sum(ec)
    obj = dx * (mc + np.log(se) / beta)

                grad[j] += wi * (1.0 - h[j + p])
            for j in range(p, n):
                grad[j] -= wi * h[j - p]

    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return dx * (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta)

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ec = np.exp(np.clip(beta * (corr - mc), -700, 700))
    w = dx * ec / np.sum(ec)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        p = abs(idx - (n - 1))
        wk = w[idx]
        if wk < 1e-14:
            continue
        if p == 0:
            continue
        grad[p:] += wk * (1.0 - h[:n - p])
        grad[:n - p] -= wk * h[p:]
    return grad

def smooth_opt(h0, dx, target, t_limit, beta_start=200.0, max_iter=80000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    n = len(h)
    beta = beta_start
    lr = 2.0 / (beta * dx * n)
    stall = 0
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        g = lse_grad(h, dx, beta)
        h_new = project_sum(h - lr * g, target)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.04, 20.0)
            stall = 0
        else:
            lr *= 0.92
            stall += 1
        if it % 300 == 0 and beta < 500000:
            beta *= 2.0
            lr = 2.0 / (beta * dx * n)
        if stall > 800 and lr < 1e-8:
            break
    return best_h, best_c5

def make_binary_pattern(n, block_ones, block_zeros):
    period = block_ones + block_zeros
    reps = n // period
    rem = n % period
    h = np.tile(np.concatenate([np.ones(block_ones), np.zeros(block_zeros)]), reps)
    if rem > 0:
        h = np.append(h, np.concatenate([np.ones(block_ones), np.zeros(block_zeros)])[:rem])
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:n_ones - target]:
            h[i] = 0.0
    elif n_ones < target:
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:target - n_ones]:
            h[i] = 1.0
    return h

def sa_binary(h_bin, dx, t_limit, T0=0.05, cool=0.99997):
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    while time.time() < t_limit:
        if T < 1e-9:
            break
        oi = ones[np.random.randint(len(ones))]
        zi = zeros[np.random.randint(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h, dx)
        d = new_c5 - curr_c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-12)):
            curr_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
        T *= cool
    return best_h, best_c5
