```python
import numpy as np
from scipy.optimize import minimize
import time

_dx = 1.0
_T = 35.0
_tau = 0.1

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def grad_sharp(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    grad = np.zeros(n)
    ks = np.arange(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    grad[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])
    return dx * grad

def obj_soft(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    return (cm + _tau * np.log(np.sum(np.exp((c - cm) / _tau)))) * _dx

def grad_soft(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    w = np.exp((c - cm) / _tau)
    w /= w.sum()
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += wm * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += wm * (-h[ks[v2] + s])
    return _dx * grad

def obj_sharp(h):
    return compute_c5(h, _dx)

def cons_fun(x):
    return float(x.sum() - _T)

def cons_jac(x):
    return np.ones(len(x))

def normalize(h, T):
    h = np.clip(h, 0, 1).copy()
    s = h.sum()
    if s > 1e-10:
        h *= T / s
    h = np.clip(h, 0, 1)
    d = T - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for j in idx:
                a = min(d, room[j])
                h[j] += a
                d -= a
                if d < 1e-12:
                    break
        else:
            d = -d
            idx = np.argsort(h)
            for j in idx:
                a = min(d, h[j])
                h[j] -= a
                d -= a
                if d < 1e-12:
                    break
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h)

def do_optimize(h_init, n, t0, budget_s):
    global _dx, _T, _tau
    _dx = 2.0 / n
    _T = n / 2.0
    T = n / 2.0
    h = normalize(h_init, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    best_local = h.copy()
    best_c5 = compute_c5(h, _dx)
    
    # Soft-max annealing
    for tau in [0.5, 0.2, 0.08, 0.03, 0.01]:
        if time.time() - t0 > budget_s - 30:
            break
        _tau = tau
        try:
            r = minimize(obj_soft, best_local, method='SLSQP', jac=grad_soft,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 1000, 'ftol': 1e-13})
            c5 = compute_c5(r.x, _dx)
            if c5 < best_c5:
                best_c5 = c5
                best_local = r.x.copy()
        except Exception:
            pass
    
    # Sharp max polish
    if time.time() - t0 < budget_s - 15:
        try:
            r2 = minimize(obj_sharp, best_local, method='SLSQP',
                         jac=lambda x: grad_sharp(x, _dx),
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 3000, 'ftol': 1e-15})
            c5 = compute_c5(r2.x, _dx)
            if c5 < best_c5:
                best_c5 = c5
                best_local = r2.x
        except Exception:
            pass
    
    return normalize(best_local, T), best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_dx = 1.0
_T = 35.0
_tau = 0.1

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def grad_sharp(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    grad = np.zeros(n)
    ks = np.arange(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    grad[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])
    return dx * grad

def grad_sharp_g(h):
    return grad_sharp(h, _dx)

def obj_sharp(h):
    return compute_c5(h, _dx)

def obj_soft(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    return (cm + _tau * np.log(np.sum(np.exp((c - cm) / _tau)))) * _dx

def grad_soft(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    w = np.exp((c - cm) / _tau)
    w /= w.sum()
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += wm * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += wm * (-h[ks[v2] + s])
    return _dx * grad

def cons_fun(x):
    return float(x.sum() - _T)

def cons_jac(x):
    return np.ones(len(x))

def normalize(h, T):
    h = np.clip(h, 0, 1).copy()
    s = h.sum()
    if s > 1e-10:
        h *= T / s
    h = np.clip(h, 0, 1)
    d = T - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for j in idx:
                a = min(d, room[j])
                h[j] += a
                d -= a
                if d < 1e-12:
                    break
        else:
            d = -d
            idx = np.argsort(h)
            for j in idx:
                a = min(d, h[j])
                h[j] -= a
                d -= a
                if d < 1e-12:
                    break
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h)

def do_optimize(h_init, n, t0, budget_s, max_iters_soft=800, max_iters_sharp=2000):
    global _dx, _T, _tau
    _dx = 2.0 / n
    _T = n / 2.0
    T = n / 2.0
    h = normalize(h_init, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    best_local = h.copy()
    best_c5 = compute_c5(h, _dx)

    for tau in [1.0, 0.4, 0.15, 0.05, 0.02]:
        if time.time() - t0 > budget_s - 30:
            break
        _tau = tau
        try:
            r = minimize(obj_soft, best_local, method='SLSQP', jac=grad_soft,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': max_iters_soft, 'ftol': 1e-13})
            c5 = compute_c5(r.x, _dx)
            if c5 < best_c5:
                best_c5 = c5
                best_local = r.x.copy()
        except Exception:
            pass

    if time.time() - t0 < budget_s - 15:
        try:
            r2 = minimize(obj_sharp, best_local, method='SLSQP',
                         jac=grad_sharp_g, bounds=bounds, constraints=cons,
                         options={'maxiter': max_iters_sharp, 'ftol': 1e-15})
            c5 = compute_c5(r2.x, _dx)
            if c5 < best_c5:
                best_c5 = c5
                best_local = r2.x.copy()
        except Exception:
            pass

    return normalize(best_local, T), best_c5

def local_search(h, dx, T, t0, budget_s):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    for i in range(n):
        if time.time() - t0 > budget_s - 10:
            break
        for val in [0.0, 1.0]:
            h_try = best.copy()
            h_try[i] = val
            h_try = normalize(h_try, T)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5 - 1e-12:
                best = h_try.copy()
                best_c5 = c5_try
    return best, best_c5
```
