    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5_grad(h, dx):
    """Compute C5 and its subgradient (for the active shift)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.argmax(corr)
    # Subgradient of max is the gradient at the active point
    n = len(h)
    grad = np.zeros(n)

            grad[i] += (1.0 - h[i + shift])  # d/dh[i] of h[i]*(1-h[i+shift])
            grad[i + shift] -= h[i]  # d/dh[i+shift] of h[i]*(1-h[i+shift])
