    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0

        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:

        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.clip(h - theta, 0, 1)

def smooth_max_obj(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    val = dx * (c_max + t * np.log(np.sum(np.exp((c - c_max) / t))))
    penalty = mu * (np.sum(h) - n / 2.0) ** 2

def smooth_max_grad(h, t, mu):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)

    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])

**Idea: Nesterov-accelerated proximal gradient for the smooth max**

The smooth max function f(h) = t * log(Σ exp(c_k(h)/t)) + c_max is smooth with Lipschitz constant related to 1/t. The proximal gradient method with projection onto the constraint set would be:

This is similar to my subgradient approach but with a smooth gradient. With Nesterov acceleration, it converges at rate O(1/k²) for smooth convex functions (though our function isn't convex).

_GMU = 50.0
_GN = 70

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    t = min(max(float(target_sum), 0.0), float(n))
    if t < 1e-15:
        return np.zeros(n)
    if t > float(n) - 1e-15:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sobj(x):
    h = x; t = _GT; mu = _GMU; n = _GN; dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    return dx * (cm + t * np.log(np.sum(np.exp((c - cm) / t)))) + mu * (h.sum() - n * 0.5) ** 2

def _sjac(x):
    h = x; t = _GT; mu = _GMU; n = _GN; dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)

    g = np.zeros(n); ks = np.arange(n)

        if wt < 1e-13:

        g[mk1] += wt * (1.0 - h[i1[mk1]])
        i2 = ks + s; mk2 = (i2 >= 0) & (i2 < n)
        g[mk2] += wt * (-h[i2[mk2]])

    except Exception:
