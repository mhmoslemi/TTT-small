import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(z, T):
    n = len(z)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj(x):
    dx, t = _G['dx'], _G['t']
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    return dx * (cm + t * np.log(np.sum(np.exp((c - cm) / t))))

def _smooth_jac(x):
    dx, t = _G['dx'], _G['t']
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    w = np.exp((c - cm) / t)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-13:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[:n - s] += w[m] * (1.0 - x[s:])
            grad[s:] += w[m] * (-x[:n - s])
        else:
            s2 = -s
            grad[s2:] += w[m] * (1.0 - x[:n - s2])
            grad[:n - s2] += w[m] * (-x[s2:])
    return dx * grad

def _exact_obj(x):
    dx = _G['dx']
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = int(np.argmax(c))
    return float(c[idx] * dx)

def _exact_jac(x):
    dx = _G['dx']
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - x[s:])
        grad[s:] += (-x[:n - s])
    else:
        s2 = -s
        grad[s2:] += (1.0 - x[:n - s2])
        grad[:n - s2] += (-x[s2:])
    return dx * grad

def _cons_fun(x):
    return float(x.sum() - _G['T'])

def _cons_jac(x):
    return np.ones(len(x))

def _run_slsqp(h, n, t, max_iter, ftol):
    _G['dx'] = 2.0 / n
    _G['t'] = t
    _G['T'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_c = project_box_sum(h.copy(), n / 2.0)
    best_h, best_c5 = h_c.copy(), compute_c5(h_c, 2.0 / n)
    try:
        r = minimize(_smooth_obj, h_c, jac=_smooth_jac, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        c5 = compute_c5(r.x, 2.0 / n)
        if c5 < best_c5:
            best_c5, best_h = c5, r.x.copy()
    except Exception:
        pass
    try:
        r2 = minimize(_exact_obj, best_h, jac=_exact_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': ftol})
        c5 = compute_c5(r2.x, 2.0 / n)
        if c5 < best_c5:
            best_c5, best_h = c5, r2.x.copy()
    except Exception:
        pass
    return project_box_sum(best_h, n / 2.0), best_c5

def _coord_descent(h, n, num_sweeps, deltas):
    dx = 2.0 / n
    T = n / 2.0
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for _ in range(num_sweeps):
        improved = False
        for i in range(n):
            for d in deltas:
                h_t = h.copy()
                h_t[i] = np.clip(h[i] + d, 0.0, 1.0)
                h_t = project_box_sum(h_t, T)
                c5 = compute_c5(h_t, dx)
                if c5 < best_c5 - 1e-15:
                    h = h_t
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def _interpolate(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

    if time.time() - t0 < budget_s - 15:

        T_f = n_f / 2.0
