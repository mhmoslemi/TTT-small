def resample_n(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    h_new = np.zeros(n_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    if T < 500:
        return mx + float(np.log(np.sum(np.exp(T * (corr - mx))))) / T
    return mx

def normalize_h(h, S):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(50):
        d = S - h.sum()
        if abs(d) < 1e-12: break
        m = (h > 1e-8) & (h < 1 - 1e-8)
        if m.any(): h[m] += d / m.sum()
        else: h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, S):
    t = int(round(S))
    b = (h >= 0.5).astype(float)
    c = int(b.sum())
    if c > t:
        ix = np.where(b == 1)[0]; o = np.argsort(h[ix])
        for k in range(c - t): b[ix[o[k]]] = 0
    elif c < t:
        ix = np.where(b == 0)[0]; o = np.argsort(-h[ix])
        for k in range(t - c): b[ix[o[k]]] = 1
    return b

def ls_best(h, n, S, rounds, dl=None):
    h = to_binary(h, S) if h.sum() != int(round(S)) else h.astype(float).copy()
    t = int(round(S))
    c1 = int(np.sum(h == 1))
    if c1 > t:
        ix = np.where(h==1)[0]
        for k in range(c1-t): h[ix[k]] = 0
    elif c1 < t:
        ix = np.where(h==0)[0]
        for k in range(t-c1): h[ix[k]] = 1
    cv = compute_c5(h); bh = h.copy(); bc = cv
    for _ in range(rounds):
        if dl and time.time() > dl: break
        oi = np.where(h==1)[0]; zi = np.where(h==0)[0]
        if not len(oi) or not len(zi): break
        bm, bi, bj = cv, -1, -1
        for i in oi:
            for j in zi:
                h[i]=0; h[j]=1
                c = compute_c5(h)
                h[i]=1; h[j]=0
                if c < bm - 1e-15: bm=c; bi,bj=i,j
        if bi >= 0:
            h[bi]=0; h[bj]=1; cv=bm
            if cv < bc: bc=cv; bh=h.copy()
        else: break
    return bh, bc

def sa_iter(h, n, S, rng, dl, nks=3):
    h = h.astype(float).copy(); t = int(round(S))
    c1 = int(np.sum(h==1))
    if c1 != t:
        if c1 > t:
            ix = np.where(h==1)[0]
            for k in range(c1-t): h[ix[k]]=0
        else:
            ix = np.where(h==0)[0]
            for k in range(t-c1): h[ix[k]]=1
    bh = h.copy(); bc = compute_c5(h)
    for _ in range(nks):
        if time.time() > dl: break
