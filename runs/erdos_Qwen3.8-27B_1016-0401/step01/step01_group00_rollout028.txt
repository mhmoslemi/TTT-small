    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0).copy()
    # Binary search for threshold
    # We want sum(clip(h - tau, 0, 1)) = target
    # f(tau) = sum(clip(h - tau, 0, 1)) is decreasing in tau
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(h_proj)[::-1] if diff > 0 else np.argsort(h_proj)
        for i in idx:
            if diff > 0:
                s = min(diff, 1.0 - h_proj[i])
                h_proj[i] += s
            else:
                s = min(-diff, h_proj[i])
                h_proj[i] -= s
            diff = target - h_proj.sum()
            if abs(diff) < 1e-14:

    return h_proj

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constrained(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(h_p)[::-1] if diff > 0 else np.argsort(h_p)
        for i in idx:
            a = min(abs(diff), 1.0 - h_p[i]) if diff > 0 else min(abs(diff), h_p[i])
            h_p[i] += a if diff > 0 else -a
            diff = target - h_p.sum()
            if abs(diff) < 1e-14:
                break
    return h_p

def thue_morse(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t.astype(float)

def soft_max_and_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    tot = e.sum()
    sw = e / tot
    obj = (mc + np.log(tot) / T) * (2.0 / n)
    ctr = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[ctr + o]
        if w < 1e-8:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return obj, grad * dx

def fix_binary_count(h, target, h_pref, rng):
    h = h.copy()
    d = target - h.sum()
    if d > 0:
        z = np.where(h == 0)[0]
        order = z[np.argsort(-h_pref[z])]
        h[:0]
        for idx in order[:int(d)]:
            h[idx] = 1.0
    elif d < 0:
        o = np.where(h == 1)[0]
        order = o[np.argsort(h_pref[o])]
        for idx in order[:int(-d)]:
            h[idx] = 0.0
    return h

def local_search(h_init, t0, budget_s, reserve, rng):
    n = len(h_init)
    h = h_init.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T_sa = 0.005
    it = 0
    while time.time() - t0 < budget_s - reserve:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T_sa, 1e-8)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        if it % 1000 == 0:
            T_sa = max(T_sa * 0.99, 0.0001)
    return best_h, best_c5
