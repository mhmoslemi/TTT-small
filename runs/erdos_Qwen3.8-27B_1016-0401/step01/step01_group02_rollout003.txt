def _c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _softmax_c5(h, scale, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sm = m + np.log(np.exp(scale * (corr - m)).sum()) / scale
    return sm * dx

def _softmax_grad(h, scale, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(scale * (corr - m))
    w = e / e.sum()
    # w has length 2n-1, w[n-1+k] is weight for lag k
    grad = np.zeros(n)
    for k in range(1, n):
        wk = w[n-1+k] + w[n-1-k]
        if wk < 1e-15:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    grad *= dx
    return grad

def _project(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-12:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.sum() > 0:
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _nesterov(h0, n, tmax, scale_start=50.0, lr_start=0.01, momentum=0.9):

    h = h0.copy()
    y = h0.copy()
    vel = np.zeros(n)
    best = h0.copy()
    best_c = _c5(h)
    lr = lr_start
    mu = momentum
    scale = scale_start
    
    while time.time() - ts < tmax:
        # Nesterov step
        fy = _softmax_c5(y, scale, 2.0/n)
        gy = _softmax_grad(y, scale, 2.0/n)
        
        y = h + mu * (h - y)  # Nesterov interpolation
        y = y - lr * gy  # gradient step
        y = _project(y, n)
        
        h_new = y
        
        # Check if we made progress
        c_new = _softmax_c5(h_new, scale, 2.0/n)
        c_cur = _softmax_c5(h, scale, 2.0/n)

            lr *= 0.5
            y = h + mu * (h - y)
            y = _project(y, n)
            mu *= 0.9
        else:

            if c_new < c_cur:
                lr = min(lr * 1.05, 0.1)
            y = h_new
            h = h_new
        
        c_actual = _c5(h)
        if c_actual < best_c:
            best_c = c_actual
            best = h.copy()
        
        # Adapt scale
        if scale < 500:
            scale *= 1.01
    
    # Final exact evaluation
    best_c = _c5(best)
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_DX = 0.0
_G_SCL = 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sm_obj(h, scale):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(scale * (corr - m)).sum()) / scale) * dx

def _sm_grad(h, scale):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(scale * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(1, n):
        wk = w[n - 1 + k] + w[n - 1 - k]
        if wk > 1e-14:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
    return grad * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.sum() > 0:
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _nesterov(h0, n, tmax, scale0=100.0, lr0=0.03):
    ts = time.time()
    h = _proj(h0.copy(), n)
    y = h.copy()
    beta = 0.85
    lr = lr0
    scale = scale0
    best = h.copy()
    best_c = _c5(h)
    stall = 0
    while time.time() - ts < tmax:
        gy = _sm_grad(y, scale)
        lnrm = np.linalg.norm(gy)
        if lnrm < 1e-14:
            break
        y_new = h + beta * (h - y)
        y_new = _proj(y_new - lr * gy, n)
        f_new = _sm_obj(y_new, scale)
        f_old = _sm_obj(h, scale)
        if f_new <= f_old + 1e-13:
            h = y_new
            beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            beta = min(beta, 2.0)
            if f_new < f_old - 1e-10:
                lr = min(lr * 1.01, 0.4)
                stall = 0
            else:
                stall += 1
        else:
            lr *= 0.6
            beta *= 0.9
            y_new = h.copy()
        if stall > 100:
            scale = min(scale * 1.05, 400.0)
            lr = lr * 0.8
            stall = 0
        c = _c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
            stall = 0
    return best, _c5(best)

def _obj_fn(x):
    return _sm_obj(np.clip(x, 0.0, 1.0), _G_SCL)

def _obj_jac(x):
    return _sm_grad(np.clip(x, 0.0, 1.0), _G_SCL)

def _cons_fun(x):
    return x.sum() - 0.5 * _G_N

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h0, n, scale, iters=3000):
    global _G_N, _G_SCL
    _G_N = n
    _G_SCL = scale
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        r = minimize(_obj_fn, h0, jac=_obj_jac, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': iters, 'ftol': 1e-16})
        if r.fun < _sm_obj(h0, scale):
            return _proj(r.x, n)
    except Exception:
        pass
    return _proj(h0, n)

def _swap_opt(h0, n, tmax):
    ts = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = _c5(h)
    tabu_age = np.full(n, -999)
    tabu_len = 30
    it = 0
    ones = np.where(h > 0.4)[0]
    zeros = np.where(h < 0.6)[0]
    while time.time() - ts < tmax:
        it += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        if it - tabu_age[i] < tabu_len or it - tabu_age[j] < tabu_len:
            continue
        h[i], h[j] = h[j], h[i]
        c = _c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
            tabu_age[i] = tabu_age[j] = it
            ones = np.where(h > 0.4)[0]
            zeros = np.where(h < 0.6)[0]
        else:
            h[j], h[i] = h[i], h[j]
    return best, _c5(best)

def _perturb(h, n, amt):
    ts = np.random.randint(n, size=min(20, n // 5))
    h2 = h.copy()
    for i in ts:
        h2[i] = np.clip(h2[i] + np.random.uniform(-amt, amt), 0.0, 1.0)
    return _proj(h2, n)

def _gold_seq(n, frac):
    """Fill positions by golden-ratio ordering to get ~frac ones."""
    pos = np.array([(i * 0.618033988749895) % 1.0 for i in range(n)])
    order = np.argsort(pos)
    h = np.zeros(n)
    k = int(round(frac * n))
    h[order[:k]] = 1.0
    return _proj(h, n)

def _thue_morse(n, frac):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c ^= 1
            x >>= 1
        h[i] = float(c)
    return _proj(h, n)
