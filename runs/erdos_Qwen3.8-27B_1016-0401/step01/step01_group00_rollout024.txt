    """Gradient of correlation at a specific lag.
    lag is in range 0..2n-2 (index into 'full' output).
    The actual shift is k = lag - (n-1).
    """
    n = len(h)

        for j in range(p, n):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            for i in np.argsort(1.0 - h)[::-1]:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
                if abs(diff) < 1e-13:
                    break
        else:
            for i in np.argsort(h)[::-1]:
                amt = min(-diff, h[i])
                h[i] -= amt
                diff += amt
                if abs(diff) < 1e-13:
                    break
    return h

def sm_obj(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def sm_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    dx = 2.0 / n

            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            for i in np.argsort(1.0 - h)[::-1]:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
                if abs(diff) < 1e-13:
                    break
        else:
            for i in np.argsort(h)[::-1]:
                amt = min(-diff, h[i])
                h[i] -= amt
                diff += amt
                if abs(diff) < 1e-13:
                    break
    return h

def sm_obj(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def sm_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[n - 1 + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def make_cons(n, target):
    def cf(x): return x.sum() - target
    def cj(x): return np.ones(n)
    return [{'type': 'eq', 'fun': cf, 'jac': cj}]

def run_binary_sa(h_init, max_iter, rng, deadline):
    h = h_init.copy()
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    ks = np.arange(2 * n - 1) - (n - 1)
    for step in range(max_iter):
        if step % 50000 == 0 and time.time() > deadline:
            break
        frac = step / max_iter
        T = max(0.05, 2.0 * (1.0 - frac) ** 1.5)
        opa = np.where(h == 1)[0]
        oza = np.where(h == 0)[0]
        if len(opa) == 0 or len(oz) == 0:
            break
        a = opa[rng.integers(len(opa))]
        b = oza[rng.integers(len(oz))]
        if a == b:
            continue
        d = np.zeros(2 * n - 1)
        ix = a + ks
        v = (ix >= 0) & (ix < n)
        d[v] -= 1.0 - h[ix[v]]
        ix = b + ks
        v = (ix >= 0) & (ix < n)
        d[v] += 1.0 - h[ix[v]]
        ix = a - ks
        v = (ix >= 0) & (ix < n)
        d[v] += h[ix[v]]
        ix = b - ks
        v = (ix >= 0) & (ix < n)
        d[v] -= h[ix[v]]
        old_mx = corr.max()
        nc = corr + d
        new_mx = nc.max()
        if new_mx <= old_mx or rng.random() < np.exp(-(new_mx - old_mx) / T):
            h[a] = 0.0
            h[b] = 1.0
            corr = nc
            c5n = float(new_mx * (2.0 / n))
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
    return best_h, best_c5

def slsqp_optimize(h_init, n, target, T, maxiter=3000):

    def jac(x): return sm_grad(x, n, T)

                      bounds=bounds, constraints=cons,

                return h_new, c5

        pass
    return h_init, 999.0
