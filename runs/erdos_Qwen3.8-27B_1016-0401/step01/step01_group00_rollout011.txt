def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = shifted - np.max(shifted)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-11:
            continue

            grad += w * (1.0 - 2.0 * h)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
            if abs(diff) < 1e-14:
                break
    return h

def slsqp_polish(h, target, T_val=40.0, maxiter=400):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        mc = np.max(corr)
        s = np.exp(T_val * (corr - mc))
        return (mc + np.log(np.sum(s)) / T_val) * (2.0 / n)

    def jac(x):
        return softmax_grad(x, T_val)

    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < 100:
            h_new = project_feasible(res.x.copy(), target)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)
