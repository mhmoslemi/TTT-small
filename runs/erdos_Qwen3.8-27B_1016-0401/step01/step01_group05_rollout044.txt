    idx = rng.permutation(n)[:ones]
    h[idx] = 1.0
    h += rng.uniform(-noise, noise, n)
    return make_feasible(h, n, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, n, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient + SLSQP on initial
    sg_h, sg_c5 = subgradient_descent(best_h, n, target, 0.01, 200, deadline - 60)
    if sg_c5 < best_c5:
        best_h, best_c5 = sg_h, sg_c5
    for alpha in [100, 300, 600]:
        if time.time() > deadline - 80:
            break
        h_new = slsqp_step(best_h, n, target, alpha, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Main simulated annealing
    if time.time() < deadline - 120:
        sa_h, sa_c5 = simulated_annealing(best_h, n, target, rng,
                                           max_steps=80000, deadline=deadline - 120, T0=0.012)
        if sa_c5 < best_c5:
            best_h, best_c5 = sa_h, sa_c5

    # Phase 3: Multi-start with binary inits
    for trial in range(5):
        if time.time() > deadline - 100:
            break
        h_alt = random_binary_start(n, target, rng, noise=0.05 + 0.04 * trial)
        h_alt, c5_sg = subgradient_descent(h_alt, n, target, 0.008, 100, deadline - 100)
        for alpha in [100, 400]:
            if time.time() > deadline - 100:
                break
            h_alt = slsqp_step(h_alt, n, target, alpha, maxiter=150)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt, c5_alt

    # Phase 4: Another SA round from best
    if time.time() < deadline - 50:
        sa_h2, sa_c5_2 = simulated_annealing(best_h, n, target, rng,
                                              max_steps=40000, deadline=deadline - 50, T0=0.006)
        if sa_c5_2 < best_c5:
            best_h, best_c5 = sa_h2, sa_c5_2

    # Phase 5: Final SLSQP polish
    for alpha in [500, 1000, 2000, 4000]:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_step(best_h, n, target, alpha, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Final subgradient + local search polish
    if time.time() < deadline - 5:
        sg_h2, sg_c5_2 = subgradient_descent(best_h, n, target, 0.003, 100, deadline - 5)
        if sg_c5_2 < best_c5:
            best_h, best_c5 = sg_h2, sg_c5_2
        # Tiny local search
        h_ls = best_h.copy()
        c5_ls = best_c5
        for it in range(5000):
            if time.time() > deadline - 3:
                break
            i, j = rng.choice(n, size=2, replace=False)
            d = rng.uniform(-0.01, 0.01)
            trial = h_ls.copy()
            trial[i] += d
            trial[j] -= d
            trial = make_feasible(trial, n, target)
            c5_t = compute_c5(trial)
            if c5_t < c5_ls:
                h_ls, c5_ls = trial, c5_t
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
