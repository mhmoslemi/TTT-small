```python
            p = -s
            idx = np.arange(p, n)
            terms = [Y[i, i+s] for i in idx]

                break
            prob.solve(solver=solver, verbose=False)
            if h.value is not None:
                break
        except Exception:
            continue
    
    if h.value is None:
        return None, None
    
    h_opt = np.clip(h.value, 0, 1)
    diff = target - h_opt.sum()
    h_opt = h_opt + diff / n
    h_opt = np.clip(h_opt, 0, 1)
    
    c5 = compute_c5(h_opt)
    return h_opt, c5
```

Let me speed it up. Instead of calling compute_c5 (which does a full correlate), I can compute just the change in the max correlation.

When we swap h[i] (1→0) and h[j] (0→1):
The correlation at shift s changes by:
ΔC(s) = -(1-h[i+s]) + h[i-s] + (1-h[j+s]) - h[j-s]
       = h[i+s] - 1 + h[i-s] + 1 - h[j+s] - h[j-s]
       = (h[i+s] + h[i-s]) - (h[j+s] + h[j-s])

where out-of-bounds indices give 0.

So ΔC(s) = (h[i+s] + h[i-s]) - (h[j+s] + h[j-s]) for each shift s.

This is O(1) per shift, O(n) for all shifts. And then the new max is max(max(C + ΔC), other values).

Actually, since ΔC(s) can be both positive and negative, the new max could be at any shift. So I need to compute max over all s of (C(s) + ΔC(s)).

But ΔC(s) = (h[i+s] + h[i-s]) - (h[j+s] + h[j-s]) depends on s in a structured way. Let me define:
f_a(s) = h[a+s] + h[a-s] (with out-of-bounds = 0)

Then ΔC(s) = f_j(s) - f_i(s)... wait, let me re-derive.

Actually, I defined the swap as h[i]: 1→0 and h[j]: 0→1.

Original C(s) = Σ_k h[k]*(1-h[k+s]) for valid k.

After swap:
- At k=i: contributes 0*(1-h[i+s]) = 0 instead of 1*(1-h[i+s]) = 1-h[i+s]
  Change: -(1-h[i+s]) = h[i+s] - 1
- At k=i-s (if valid): h[i-s]*(1-h[i]) changes from h[i-s]*0 to h[i-s]*1
  Change: +h[i-s]
- At k=j: contributes 1*(1-h[j+s]) instead of 0*(1-h[j+s]) = 0
  Change: +(1-h[j+s])
- At k=j-s (if valid): h[j-s]*(1-h[j]) changes from h[j-s]*1 to h[j-s]*0
  Change: -h[j-s]

So ΔC(s) = (h[i+s] - 1) + h[i-s] + (1-h[j+s]) - h[j-s]
          = h[i+s] + h[i-s] - h[j+s] - h[j-s]

where h[a±s] = 0 if a±s is out of [0, n-1].

This is clean! For a given swap (i,j), ΔC(s) for all s can be computed as:
ΔC = f_i - f_j where f_a(s) = h[a+s] + h[a-s] (with 0 padding)

And f_a(s) for all s can be computed in O(n) (just shift h by a and add to its reverse-shift).

So the total cost of evaluating one swap is O(n) to compute ΔC for all shifts, plus O(n) to find the new max. Total O(n) per swap.

For all 19600 swaps: O(n * n²) = O(n³) ≈ 22M for n=280. At ~1B ops/s in numpy, this is about 0.02s per exhaustive pass. That's fast!

```python
def eval_swap_delta(h, i, j):
    """Compute ΔC(s) for all shifts s when swapping h[i]:1->0 and h[j]:0->1."""
    n = len(h)
    # f(a, s) = h[a+s] if 0<=a+s<n else 0, plus h[a-s] if 0<=a-s<n else 0
    # We need this for s in range(-(n-1), n)
    # Total 2n-1 shifts
    
    # Vectorized: for shift s, h[i+s] is at position i+s if valid
    # Let's construct the full array
    
    # Actually, let's just do it with numpy
    # f_i(s) for s = -(n-1), ..., n-1
    # h[i+s] is valid when 0 <= i+s <= n-1, i.e., -i <= s <= n-1-i
    # h[i-s] is valid when 0 <= i-s <= n-1, i.e., 1-n+i <= s <= i
    
    si = np.zeros(2*n-1)  # indexed by s + (n-1)
    sj = np.zeros(2*n-1)
    
    # h[i+s]: s from -i to n-1-i
    lo = max(-i, -(n-1))
    hi = min(n-1-i, n-1)
    if lo <= hi:
        shifts = np.arange(lo, hi+1)
        si[shifts + (n-1)] += h[i + shifts]
    
    # h[i-s]: s from 1-n+i to i
    lo2 = max(1-n+i, -(n-1))
    hi2 = min(i, n-1)
    if lo2 <= hi2:
        shifts = np.arange(lo2, hi2+1)
        si[shifts + (n-1)] += h[i - shifts]
    
    # Same for j
    lo = max(-j, -(n-1))
    hi = min(n-1-j, n-1)
    if lo <= hi:
        shifts = np.arange(lo, hi+1)
        sj[shifts + (n-1)] += h[j + shifts]
    
    lo2 = max(1-n+j, -(n-1))
    hi2 = min(j, n-1)
    if lo2 <= hi2:
        shifts = np.arange(lo2, hi2+1)
        sj[shifts + (n-1)] += h[j - shifts]
    
    return si - sj  # ΔC(s) for all s
```

```python
    # h[a+s] for s = -(n-1), ..., n-1:
    #   = h[a-(n-1)], h[a-(n-2)], ..., h[a], ..., h[a+n-1]
    #   with out-of-bounds = 0
    
    f = np.zeros((n, 2*n-1))
    for a in range(n):
        # h[a+s] for s = -(n-1) to n-1
        idx_pos = a + np.arange(-(n-1), n)
        valid = (idx_pos >= 0) & (idx_pos < n)
        f[a, valid] += h[idx_pos[valid]]
        # h[a-s] for s = -(n-1) to n-1 → h[a+s'] for s' = n-1 to -(n-1)
        # = reverse of h[a+s]
        idx_neg = a - np.arange(-(n-1), n)
        valid2 = (idx_neg >= 0) & (idx_neg < n)
        f[a, valid2] += h[idx_neg[valid2]]
    return f
```
