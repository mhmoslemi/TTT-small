import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def _cons_fun(x, ts):
    return np.sum(x) - ts

def _cons_jac(x):
    return np.ones(len(x))

def smooth_obj_grad(h, dx, beta):
    # Log-sum-exp approximation to max of correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    
    # Gradient
    n = len(h)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            # corr[n-1+m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            # d/dh[j] = (1-h[j-m])*I(j>=m) - h[j+m]*I(j+m<n)
            for j in range(m, n):
                grad[j] += wi * (1.0 - h[j - m])
            for j in range(0, n - m):
                grad[j] -= wi * h[j + m]
        else:
            p = -m
            for j in range(0, n - p):
                grad[j] += wi * (1.0 - h[j + p])
            for j in range(p, n):
                grad[j] -= wi * h[j - p]
    return obj, grad

from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def smooth_obj_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    n = len(h)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

    return np.sum(x) - ts

def _cons_jac(x):
    return np.ones(len(x))

    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for beta in [200.0, 1000.0, 5000.0, 20000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='L-BFGS-B', bounds=bnds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass

    momentum = 0.9

np.correlate(h, 1-h, 'full')[m] where m goes from 0 to 2n-2.

The relationship: for m = n-1+k (k can be negative), this is:
sum_{i: max(0,k) ≤ i ≤ min(n-1, n-1+k)} h[i] * (1-h[i-k])

For the swap, the change in this sum:
- Position i: was 1, now 0. Contributes -(1-h[i-k]) for valid i, i-k
- Position j: was 0, now 1. Contributes +(1-h[j-k]) for valid j, j-k
- Also, i and j appear in the (1-h[...]) factor for other terms:
  - For any p where p-k = i (i.e., p = i+k): the term h[p]*(1-h[i]) changes: was h[p]*(1-1)=0, now h[p]*(1-0)=h[p]. So gain h[i+k] if valid.
  - For any p where p-k = j (i.e., p = j+k): was h[p]*(1-0)=h[p], now h[p]*(1-1)=0. So loss h[j+k] if valid.

So the total change in corr[n-1+k] for a swap (i:1→0, j:0→1):
delta[k] = -(1-h[i-k]) * I(valid) + (1-h[j-k]) * I(valid) + h[i+k] * I(valid) - h[j+k] * I(valid)

Where "valid" means the indices are in [0, n-1].

This is O(1) per shift! So I can compute all 2n-1 deltas in O(n) instead of O(n^2). Then the new max can be computed if I track which shifts are at the max.

def swap_delta(h, i, j, n):
    """Compute delta of each correlation value when swapping h[i] (1→0) and h[j] (0→1)."""
    # corr[m] for m = 0 to 2n-2, where m = n-1+k, k ∈ [-(n-1), n-1]
    # delta[k] = -(1-h[i-k])*I(0≤i-k<n) + (1-h[j-k])*I(0≤j-k<n) 
    #            + h[i+k]*I(0≤i+k<n) - h[j+k]*I(0≤j+k<n)
    deltas = np.zeros(2*n-1)
    # For each of the 4 terms, only shifts k where the index is valid contribute
    
    # Term 1: -(1-h[i-k]) for k in [i-(n-1), i] i.e., k ∈ [i-n+1, i]
    k1_lo = max(-(n-1), i - (n-1))
    k1_hi = min(n-1, i)
    for k in range(k1_lo, k1_hi + 1):
        idx = i - k
        if 0 <= idx < n:
            deltas[n-1+k] -= (1.0 - h[idx])
    
    # Term 2: +(1-h[j-k]) for k in [j-(n-1), j]
    k2_lo = max(-(n-1), j - (n-1))
    k2_hi = min(n-1, j)
    for k in range(k2_lo, k2_hi + 1):
        idx = j - k
        if 0 <= idx < n:
            deltas[n-1+k] += (1.0 - h[idx])
    
    # Term 3: +h[i+k] for k in [-i, n-1-i]
    k3_lo = max(-(n-1), -i)
    k3_hi = min(n-1, n-1-i)
    for k in range(k3_lo, k3_hi + 1):
        idx = i + k
        if 0 <= idx < n:
            deltas[n-1+k] += h[idx]
    
    # Term 4: -h[j+k] for k in [-j, n-1-j]
    k4_lo = max(-(n-1), -j)
    k4_hi = min(n-1, n-1-j)
    for k in range(k4_lo, k4_hi + 1):
        idx = j + k
        if 0 <= idx < n:
            deltas[n-1+k] -= h[idx]
    
    return deltas

        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h, dx)
        if c5_new <= best_c5:
            best, best_c5, no_imp = h.copy(), c5_new, 0
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
            no_imp += 1
            if no_imp > 80000:
                h = best.copy()
                T = max(T, 3.0)
                no_imp = 0
        else:
            h[i], h[j] = 1.0, 0.0
        T *= cool
    return best, best_c5

def binary_ls(h0, dx, t_limit):
    n = len(h0)
    h = h0.copy()
    best_c5, best_h = compute_c5(h, dx), h.copy()
    while time.time() < t_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > t_limit or improved:
                break
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5 - 1e-15:
                    best_c5, best_h = c5_new, h.copy()
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best_c5

def binary_ls_double(h0, dx, t_limit):
    n = len(h0)
    h = h0.copy()
    best_c5, best_h = compute_c5(h, dx), h.copy()
    rng = np.random.RandomState(12345)
    while time.time() < t_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        trial = 0
        while not improved and trial < 200 and time.time() < t_limit:
            trial += 1
            idx_o = rng.choice(len(ones), 2, replace=False)
            idx_z = rng.choice(len(zeros), 2, replace=False)
            i1, i2 = int(ones[idx_o[0]]), int(ones[idx_o[1]])
            j1, j2 = int(zeros[idx_z[0]]), int(zeros[idx_z[1]])
            h[i1] = h[i2] = 0.0
            h[j1] = h[j2] = 1.0
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5 - 1e-15:
                best_c5, best_h = c5_new, h.copy()
                improved = True
            else:
                h[i1] = h[i2] = 1.0
                h[j1] = h[j2] = 0.0
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    best_h = project_feasible(h_init, n_init / 2.0)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    n_list = [46, 48, 49, 50, 52, 54, 56, 60, 64, 72, 80, 96, 128, 194]
    t_bin_end = start + budget_s * 0.60
    for idx, n in enumerate(n_list):
        if time.time() > t_bin_end:
            break
        rem = t_bin_end - time.time()
        if rem < 15:
            break
        t_sa = min(rem * 0.65, 55)
        t_ls = min(rem * 0.25, 20)
        target = n // 2
        if n == n_init:
            h_bin = (h_init > 0.5).astype(float)
        else:
            h_bin = np.zeros(n)
            h_bin[np.random.choice(n, target, replace=False)] = 1.0
        oc = int(np.round(np.sum(h_bin)))
        if oc != target:
            if oc > target:
                idxs = np.where(h_bin == 1)[0]
                np.random.shuffle(idxs)
                for k in range(oc - target):
                    h_bin[idxs[k]] = 0
            else:
                idxs = np.where(h_bin == 0)[0]
                np.random.shuffle(idxs)
                for k in range(target - oc):
                    h_bin[idxs[k]] = 1
        dx_n = 2.0 / n
        h_sa, c5_sa = binary_sa(h_bin, dx_n, time.time() + t_sa, seed + idx * 7)
        h_ls, c5_ls = binary_ls(h_sa, dx_n, time.time() + t_ls)
        if c5_ls < best_c5:
            best_c5, best_h, best_n = c5_ls, h_ls.copy(), n
        if time.time() < t_bin_end and n <= 80:
            h_ds, c5_ds = binary_ls_double(h_ls, dx_n, min(time.time() + 15, t_bin_end))
            if c5_ds < best_c5:
                best_c5, best_h, best_n = c5_ds, h_ds.copy(), n

    t_cont_end = start + budget_s * 0.88
    if time.time() < t_cont_end:
        h_c, c5_c = optimize_continuous(best_h, 2.0 / best_n, best_n / 2.0, t_cont_end)
        if c5_c < best_c5:
            best_c5, best_h, best_n = c5_c, h_c.copy(), best_n
        for n_try in [n_init, 92, 96, 194]:
            if time.time() > t_cont_end:
                break
            dx_t = 2.0 / n_try
            x_old = np.linspace(0.0, 2.0, len(h_init))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_old, h_init)
            h_t = project_feasible(h_interp, n_try / 2.0)
            h_c2, c5_c2 = optimize_continuous(h_t, dx_t, n_try / 2.0, min(t_cont_end, time.time() + 40))
            if c5_c2 < best_c5:
                best_c5, best_h, best_n = c5_c2, h_c2.copy(), n_try

    t_final = start + budget_s * 0.96
    if time.time() < t_final and best_n % 2 == 0:
        dx_f = 2.0 / best_n
        if np.all((best_h == 0) | (best_h == 1)):
            h_ref, c5_ref = binary_ls(best_h, dx_f, t_final)
            if c5_ref < best_c5:
                best_c5, best_h = c5_ref, h_ref.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, best_n
