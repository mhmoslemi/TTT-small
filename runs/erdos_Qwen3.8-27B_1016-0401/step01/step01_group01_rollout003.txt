import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def grad_sharp(h, dx):
    n = len(h)
    c = get_corr(h)
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] += (-h[s:])
    else:
        ss = -s
        grad[:n-ss] += (1.0 - h[ss:])

    c = get_corr(h)
    cm = np.max(c)
    return dx * (cm + T * np.log(np.sum(np.exp((c - cm) / T))))

    n = len(h)

            h_f = project_feasible(h_f, n_f)
            c_f = compute_c5(h_f, dx_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f.copy()

    # Final verification
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    best_h = project_feasible(best_h, n_f)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, n_f
