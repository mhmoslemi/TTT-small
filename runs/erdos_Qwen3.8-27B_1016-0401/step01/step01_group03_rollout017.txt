Let me also consider: instead of evaluating all 2-swaps, I can use the delta formula to quickly compute the change in max.

For a 2-swap (a1,a2→1, b1,b2→0), the change in corr at shift s:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (m + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        k = i - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            p = -k
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return obj, grad

def _eq_fun(x, ts):
    return np.sum(x) - ts

def _eq_jac(x):
    return np.ones(len(x))

def optimize_continuous(h0, dx, t_limit):
    n = len(h0)
    ts = n / 2.0
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cons = [{'type': 'eq', 'fun': lambda x: _eq_fun(x, ts), 'jac': lambda x: _eq_jac(x)}]
    bnds = [(0.0, 1.0)] * n
    for beta in [500.0, 2000.0, 8000.0]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(smooth_max_obj_grad, h, args=(dx, beta), jac=True,
                          method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def snap_binary(h, n):
    ts = n // 2
    h = np.clip(h, 0.0, 1.0)
    ranks = np.argsort(-h)[:ts]
    out = np.zeros(n)
    out[ranks] = 1.0
    return out

def single_swap_search(h, dx, t_limit, max_passes=200):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for _ in range(max_passes):
        if time.time() > t_limit:
            break
        best_gain = 1e-14
        best_a, best_b = -1, -1
        zlist = sorted(zeros)
        olist = sorted(ones)
        for a in zlist:
            for b in olist:
                h[a] = 1.0
                h[b] = 0.0
                c = compute_c5(h, dx)
                h[a] = 0.0
                h[b] = 1.0
                gain = best_c5 - c
                if gain > best_gain:
                    best_gain = gain
                    best_a, best_b = a, b
                if time.time() > t_limit:
                    break
            if time.time() > t_limit:
                break
        if best_a >= 0:
            h[best_a] = 1.0
            h[best_b] = 0.0
            ones.discard(best_b)
            ones.add(best_a)
            zeros.discard(best_a)
            zeros.add(best_b)
            best_c5 -= best_gain
        else:
            break
    best_c5 = compute_c5(h, dx)
    return h, best_c5

def two_swap_search(h, dx, t_limit, max_passes=30):
    n = len(h)
    if n > 72:
        return h, compute_c5(h, dx)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for _ in range(max_passes):
        if time.time() > t_limit:
            break
        best_gain = 1e-14
        best_move = None
        zlist = sorted(zeros)

                a1, a2 = zlist[i], zlist[j]

                        b1, b2 = olist[p], olist[q]
                        h[a1] = 1.0; h[a2] = 1.0
                        h[b1] = 0.0; h[b2] = 0.0
                        c = compute_c5(h, dx)
                        h[a1] = 0.0; h[a2] = 0.0
                        h[b1] = 1.0; h[b2] = 1.0
                        gain = best_c5 - c
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (a1, a2, b1, b2)
                        if time.time() > t_limit:
                            return h, compute_c5(h, dx)
        if best_move is not None:
            a1, a2, b1, b2 = best_move
            h[a1] = 1.0; h[a2] = 1.0
            h[b1] = 0.0; h[b2] = 0.0
            ones.discard(b1); ones.discard(b2)
            ones.add(a1); ones.add(a2)
            zeros.discard(a1); zeros.discard(a2)
            zeros.add(b1); zeros.add(b2)
            best_c5 -= best_gain
        else:
            break
    best_c5 = compute_c5(h, dx)
    return h, best_c5

def simulated_anneal_binary(h0, dx, t_limit, n_iter=200000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    cur = compute_c5(h, dx)
    cur_temp = 0.01
    cool_rate = 0.9997
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        zlist = list(zeros)
        olist = list(ones)
        a = zlist[np.random.randint(len(zlist))]
        b = olist[np.random.randint(len(olist))]
        h[a] = 1.0
        h[b] = 0.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-max(delta, 1e-12) / cur_temp):
            cur_c5 = new_c5
            ones.discard(b); ones.add(a)
            zeros.discard(a); zeros.add(b)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 0.0
            h[b] = 1.0
        cur_temp *= cool_rate
        if cur_temp < 1e-6:
            cur_temp = 1e-6
    return best_h, best_c5

def periodic_init(n, period_frac=0.382):
    h = np.zeros(n)
    period = max(2, int(n * period_frac))
    for i in range(n):
        if (i % period) < period // 2:
            h[i] = 1.0
    ts = n // 2
    if h.astype(float).sum() != ts:
        h = snap_binary(h, n)
    return h
