from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shift = corr.max()
    return shift + np.log(np.sum(np.exp(T * (corr - shift)))) / T

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def slsqp_refine(h, n, S, T, maxiter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    def obj(x):
        return smooth_max_obj(x, T)
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': ftol})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()

def greedy_binary(h_init, n, S, max_sweeps, t_deadline, best_c5):
    """Best-improvement greedy: try all swaps, pick the best one per sweep."""
    h_g = h_init.copy()
    c5_g = compute_c5(h_g)
    best_h = h_g.copy()
    best_c5 = c5_g if c5_g < best_c5 else best_c5
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h_g == 1)[0]
        zero_idx = np.where(h_g == 0)[0]
        best_delta = 1e-18  # require improvement
        best_pair = None
        for i in one_idx:
            for j in zero_idx:
                h_g[i], h_g[j] = h_g[j], h_g[i]
                c5_new = compute_c5(h_g)
                h_g[i], h_g[j] = h_g[j], h_g[i]
                delta = c5_new - c5_g
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is not None:
            h_g[best_pair[0]], h_g[best_pair[1]] = h_g[best_pair[1]], h_g[best_pair[0]]
            c5_g = c5_g + best_delta
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
        else:
            break
    return best_h, best_c5

def first_improve_binary(h_init, n, S, rng, n_iter, t_deadline, best_c5, window=7):

    h_f = h_init.copy()

        c5_new = compute_c5(h_f)
        if c5_new <= c5_f + 1e-15:

            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
        else:

    return best_h, best_c5

def block_move_binary(h_init, n, S, rng, n_iter, t_deadline, best_c5, block_size=2):
    """Move a block of consecutive 1s to a new position."""
    h_b = h_init.copy()
    c5_b = compute_c5(h_b)
    best_h = h_b.copy()
    best_c5 = c5_b if c5_b < best_c5 else best_c5
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        # Find a block of consecutive 1s
        one_idx = np.where(h_b == 1)[0]
        if len(one_idx) < block_size:
            break
        start = rng.choice(one_idx)
        block = []
        for k in range(block_size):
            pos = (start + k) % n
            if h_b[pos] == 1:
                block.append(pos)
            else:
                break
        if len(block) < 2:
            continue
        # Try shifting the block by ±1 or ±2
        shift = rng.choice([-2, -1, 1, 2])
        new_block = [(p + shift + n) % n for p in block]
        # Check no overlap
        if len(set(new_block)) != len(new_block):
            continue
        if any(h_b[p] == 1 for p in new_block):
            continue
        # Apply
        h_b[block] = 0
        h_b[new_block] = 1
        c5_new = compute_c5(h_b)
        if c5_new <= c5_b + 1e-15:
            c5_b = c5_new
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
        else:
            h_b[new_block] = 0
            h_b[block] = 1
    return best_h, best_c5

            best_h, best_c5 = h_t2, c5_t2

    # Phase 6: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.92:
        h_cont = best_h.astype(float) + rng.normal(0, 0.02, n)
        h_cont = normalize_h(h_cont, S)
        for T in [200, 1000, 5000]:
            if time.time() - t0 > budget_s * 0.95:
                break
            h_new = slsqp_refine(h_cont, n, S, T, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_cont = h_new

    # Final binary check + quick greedy
    if time.time() - t0 < budget_s * 0.96:
        h_fb = to_binary(best_h, S)
        c5_fb = compute_c5(h_fb)
        if c5_fb < best_c5:
            best_h = h_fb
            best_c5 = c5_fb
        h_fg, c5_fg = greedy_binary(best_h, n, S, 8, t0 + budget_s * 0.97, best_c5)
        if c5_fg < best_c5:
            best_h, best_c5 = h_fg, c5_fg

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
