def simulated_annealing(h, n, target, rng, steps=20000, t_start=0.1, t_end=0.001, deadline=None):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(steps):

        T = t_start * (t_end / t_start) ** (step / steps)
        c5_cur = compute_c5(h)
        
        # Random move

        if move_type == 0:
            # Two-coordinate move
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.1, 0.1)
            h_try = h.copy()

def slsqp_polish(h, n, target, top_k=10, alpha=100, maxiter=200):
    """SLSQP with sum of top_k correlations as objective."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-top_k:]
        return np.sum(corr[idx])
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k in idx:
            lag = k - (n - 1)
            gg = np.zeros(n)
            if lag == 0:
                gg[:] = 1.0 - 2.0 * x
            elif lag > 0:
                gg[lag:] += (1.0 - x[:n-lag])
                gg[:n-lag] -= x[lag:]
            else:
                la = -lag
                gg[:n-la] += (1.0 - x[la:])
                gg[la:] -= x[:n-la]
            g += gg
        return g
    
    def constraint(x):
        return np.sum(x) - target
    
    def constraint_grad(x):
        return np.ones(n)
    
    try:
        res = minimize(
            obj, h.copy(), jac=obj_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': constraint_grad}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = project_feasible(res.x, n, target)
        return h_new
    except Exception:
        return h.copy()
