    s = h.sum()

        return h

        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def sgs_smooth(h, dx, beta):
    """Smooth max objective and gradient."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()

    e = np.exp(ea)
    es = e.sum()
    obj = cm + np.log(es) / beta
    w = e / es
    grad = np.zeros(n)

        wm = w[m]
        if wm < 1e-15:
            continue

            grad[s:] += wm * (1.0 - h[:n - s])
            grad[:n - s] += wm * (-h[s:])
        else:
            r = -s
            grad[:n - r] += wm * (1.0 - h[r:])
            grad[r:] += wm * (-h[:n - r])
    return obj, dx * grad

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(c))
    s = m - (n - 1)
    return dx * _grad_at_lag(h, s, n)

def slsqp_run(h0, n, max_iter, ftol, beta):
    dx = 2.0 / n
    S = n / 2.0
    h = project_h(h0.copy(), S)
    bounds = [(0.0, 1.0)] * n

    def cf(x):
        return float(x.sum() - S)
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]

    if beta is not None:
        f = lambda x: smooth_obj(x, dx, beta)
        jf = lambda x: smooth_grad(x, dx, beta)
    else:
        f = lambda x: sharp_obj(x, dx)
        jf = lambda x: sharp_grad(x, dx)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    try:
        r = minimize(f, h, method='SLSQP', jac=jf, bounds=bounds,
                     constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        if r.x is not None:
            c5 = compute_c5(r.x, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = project_h(r.x, S)
    except Exception:
        pass
    return project_h(best_h, S), best_c5

def fista_run(h0, n, max_steps, beta0, beta_final, time_limit):
    dx = 2.0 / n
    S = n / 2.0
    h = project_h(h0.copy(), S)
    y = h.copy()
    h_prev = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t_start = time.time()
    n_decay = max(1, max_steps // 3)
    for k in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        beta = beta0 * (beta_final / beta0) ** (k / max(1, max_steps - 1))
        gy = smooth_grad(y, dx, beta)
        L = dx * 4.0 * n * beta * beta + 1.0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t / t_new - 1.0) * (h - h_prev)
        h = project_h(h - gy / L, S)
        h_prev = h_prev.copy()
        h_prev[:] = h
        h_prev = h.copy()
        t = t_new
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return project_h(best_h, S), best_c5

def pair_cd(h, n, max_rounds, t_grid_size=15):
    dx = 2.0 / n
    S = n / 2.0
    h = project_h(h.copy(), S)
    best_c5 = compute_c5(h, dx)
    ts = np.linspace(-0.04, 0.04, t_grid_size)
    for rnd in range(max_rounds):
        improved = False
        order = np.random.permutation(n * (n - 1) // 2)
        for idx in order:
            i = int(idx // (n - 1 - idx % n))
            j = n - 1 - idx % n
            if i >= j:
                i, j = j, i
            best_t = 0.0
            best_val = best_c5
            for t in ts:
                hi, hj = h[i] + t, h[j] - t
                if hi < -1e-9 or hi > 1 + 1e-9 or hj < -1e-9 or hj > 1 + 1e-9:
                    continue
                h_new = h.copy()
                h_new[i] = hi
                h_new[j] = hj
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_val:
                    best_val = c5_new
                    best_t = t
            if best_t != 0.0 and best_val < best_c5 - 1e-13:
                h[i] += best_t
                h[j] -= best_t
                h = project_h(h, S)
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = np.full(70, 0.5)
    n0 = len(h0)

    best_h = project_h(h0.copy(), n0 / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n0)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: SLSQP at n0 with decreasing beta (homotopy)
    for beta in [200.0, 80.0, 30.0, 12.0]:
        if remaining() < 80:
            break
        h_try, c5_try = slsqp_run(best_h, n0, 4000, 1e-14, beta)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 2: Sharp max SLSQP
    if remaining() > 60:
        h_try, c5_try = slsqp_run(best_h, n0, 6000, 1e-15, None)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 3: Pair coordinate descent
    if remaining() > 40:
        h_try, c5_try = pair_cd(best_h, n0, 15, 11)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 4: FISTA with annealing
    if remaining() > 120:
        h_try, c5_try = fista_run(best_h, n0, 3000, 50.0, 5.0, remaining() - 80)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        # Sharp polish after FISTA
        if remaining() > 60:
            h_try2, c5_try2 = slsqp_run(best_h, n0, 3000, 1e-15, None)
            if c5_try2 < best_c5:
                best_c5, best_h = c5_try2, h_try2

    # Phase 5: Try n=100
    if remaining() > 200:
        n2 = 100
        h_proj = np.interp(np.linspace(0, 2, n2, endpoint=False),
                           np.linspace(0, 2, n0, endpoint=False), best_h)
        h_proj = project_h(h_proj, n2 / 2.0)
        for beta in [60.0, 20.0]:
            if remaining() < 60:
                break
            h_try, c5_try = slsqp_run(h_proj, n2, 3000, 1e-14, beta)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try
        if remaining() > 60:
            h_try, c5_try = slsqp_run(best_h, len(best_h), 4000, 1e-15, None)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try
        if remaining() > 30:
            h_try, c5_try = pair_cd(best_h, len(best_h), 10, 11)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

    # Phase 6: Try n=140 if time allows
    if remaining() > 250:
        n3 = 140
        n_cur = len(best_h)
        h_proj = np.interp(np.linspace(0, 2, n3, endpoint=False),
                           np.linspace(0, 2, n_cur, endpoint=False), best_h)
        h_proj = project_h(h_proj, n3 / 2.0)
        for beta in [40.0, 15.0]:
            if remaining() < 60:
                break
            h_try, c5_try = slsqp_run(h_proj, n3, 2500, 1e-14, beta)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try
        if remaining() > 50:
            h_try, c5_try = slsqp_run(best_h, len(best_h), 3000, 1e-15, None)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

    # Phase 7: Perturbation restarts at best size
    n_best = len(best_h)
    for p in range(15):
        if remaining() < 25:
            break
        scale = 0.005 + 0.003 * p
        h_pert = best_h + rng.normal(0, scale, n_best)
        h_pert = project_h(h_pert, n_best / 2.0)
        h_try, c5_try = slsqp_run(h_pert, n_best, 1500, 1e-14, 25.0)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Final polish
    if remaining() > 20:
        n_f = len(best_h)
        h_try, c5_try = slsqp_run(best_h, n_f, 3000, 1e-15, None)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        h_try, c5_try = pair_cd(best_h, n_f, 8, 9)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Final cleanup
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    best_h = project_h(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, n_f
