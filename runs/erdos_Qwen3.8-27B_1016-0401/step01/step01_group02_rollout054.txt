    n = len(h)
    c = _corr(h)
    old = h[i]
    # d_k = dc_k/dh_i
    k_shift = np.arange(-(n-1), n)  # shifts
    d = np.zeros(2*n-1)
    for k_idx, k in enumerate(k_shift):
        # dc_k/dh_i = (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)
        idx1 = i - k

        if 0 <= idx1 < n:

        if 0 <= idx2 < n:

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _lse_obj(h, s, dx):
    c = _corr(h)
    cmax = c.max()
    e = np.exp(s * (c - cmax))
    return (cmax + np.log(e.sum()) / s) * dx

def _grad_lse(h, s, dx):
    n = len(h)
    c = _corr(h)
    cmax = c.max()
    e = np.exp(s * (c - cmax))
    w = e / e.sum()
    A = np.convolve(w, 1.0 - h, mode='full')
    B = np.correlate(w, h, mode='full')
    return s * dx * (A[n-1:n-1+n] - B[0:n][::-1])

def _coord_opt(h, i, s):
    n = len(h)
    c = _corr(h)
    old = h[i]
    # Compute d_k = dc_k/dh_i
    d = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        val = 0.0
        idx1 = i - shift
        if 0 <= idx1 < n:
            val += (1.0 - h[idx1])

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_lse(h, s, dx):
    n = len(h)
    c = _corr(h)
    cmax = c.max()
    e = np.exp(s * (c - cmax))
    w = e / e.sum()

    B = np.correlate(w, h, mode='full')
