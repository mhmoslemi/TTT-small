- Find j- with most positive s[j-] (decreasing h[j-]... wait, we need to increase it)
- Actually: to reduce corr[k*], we want to decrease h where s>0 and increase h where s<0
- But we need to maintain sum(h) = n/2, so we transfer mass from one position to another
- Best move: transfer from argmax(s) to argmin(s)

This is a "greedy" move that directly targets the top lag.

from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _smooth_max(corr, n, scale):
    m = corr.max()
    val = (m + np.log(np.exp(scale * (corr - m)).sum()) / scale) * (2.0 / n)
    return val

def _slsqp_smooth(h0, n, scale, maxiter):
    dx = 2.0 / n
    g_n = n
    g_dx = dx
    g_scale = scale
    
    def obj(x):
        c = _corr(x)
        m = c.max()
        v = (m + np.log(np.exp(g_scale * (c - m)).sum()) / g_scale) * g_dx
        return v
    
    def obj_grad(x):
        c = _corr(x)
        m = c.max()
        w = np.exp(g_scale * (c - m))
        w /= w.sum()
        shift_base = g_n - 1
        g = np.zeros(g_n)
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-10:
                continue
            shift = k_idx - shift_base
            if shift >= 0:
                g[:g_n-shift] += w[k_idx] * (1.0 - x[shift:])
                g[shift:] -= w[k_idx] * x[:g_n-shift]
            else:
                g[-shift:] += w[k_idx] * (1.0 - x[:g_n+shift])
                g[:g_n+shift] -= w[k_idx] * x[-shift:]
        return g * g_dx
    
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5*n,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _proj(res.x, n)
    except:
        return _proj(h0, n)

    t0 = time.time()
    bh = h.copy()
    bc = _c5(bh)
    cur = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > tmax:
            break
        curr = _corr(cur)
        k_star = np.argmax(curr)
        shift = k_star - (n - 1)
        # Compute sensitivity
        if shift >= 0:
            sens = np.zeros(n)
            sens[:n-shift] += (1.0 - cur[shift:])
            sens[shift:] -= cur[:n-shift]
        else:
            sens = np.zeros(n)
            sens[-shift:] += (1.0 - cur[:n+shift])
            sens[:n+shift] -= cur[-shift:]
        # Best transfer: from argmax(sens) to argmin(sens)
        j_out = np.argmax(sens)
        j_in = np.argmin(sens)
        if j_out == j_in:
            continue
        # Try transferring some mass
        for frac in [0.3, 0.5, 0.7, 1.0]:
            amt = frac * cur[j_out]
            h2 = cur.copy()
            h2[j_out] -= amt
            h2[j_in] += amt
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc:
                bh, bc = h2, c2
                cur = h2
                break
    return bh, bc
