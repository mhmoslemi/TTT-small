import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def best_improvement_local_search(h, n, t_deadline, max_iter=10000):
    """Best-improvement local search for binary h."""
    h = h.copy()
    c5 = compute_c5(h)
    target = n // 2
    
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        
        best_delta = 0
        best_swap = None
        
        for i in one_idx:
            for j in zero_idx:
                h[i], h[j] = 0, 1
                c5_new = compute_c5(h)
                delta = c5 - c5_new
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                h[i], h[j] = 1, 0
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i], h[j] = 0, 1
        c5 -= best_delta
    
    return h, c5

def iterated_local_search(h, n, rng, n_kicks, kick_size, t_deadline):
    """Iterated local search: local search + perturbation."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for kick in range(n_kicks):
        if time.time() > t_deadline:
            break
        
        # Local search
        h, c5 = best_improvement_local_search(h, n, t_deadline, max_iter=5000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Perturbation

        k = rng.randint(2, kick_size + 1)
        for _ in range(k):
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
            i = one_idx[rng.randint(len(one_idx))]
            j = zero_idx[rng.randint(len(zero_idx))]
            h[i], h[j] = 0, 1
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, n):
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def best_improvement_ls(h, n, t_deadline, max_sweeps=80):
    h = h.copy()
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        best_d = 1e-14
        best_p = None
        for i in oi:
            hi_save = h[i]
            h[i] = 0.0
            for j in zi:
                h[j] = 1.0
                c = np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)
                d = c5 - c
                if d > best_d:
                    best_d = d
                    best_p = (i, j)
                h[j] = 0.0
            h[i] = hi_save
        if best_p is None:
            break
        i, j = best_p
        h[i], h[j] = 0.0, 1.0
        c5 -= best_d
    return h, c5

def first_improvement_ls(h, n, t_deadline, max_sweeps=200):
    h = h.copy()
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        improved = False
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        rng_order = np.random.permutation(len(oi))
        rng_order2 = np.random.permutation(len(zi))
        for ai in rng_order:
            for aj in rng_order2:
                i, j = oi[ai], zi[aj]
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h)
                if c < c5 - 1e-14:
                    c5 = c
                    improved = True
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, c5

def make_starts(n, rng, seed_offset=0):
    starts = []
    # Random starts
    for _ in range(3):
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(h)
    # Spaced pattern
    h = np.zeros(n)
    pos = (np.arange(n // 2) * n + n // 4) % n
    h[pos] = 1.0
    starts.append(h)
    # Quadratic pattern
    h = np.zeros(n)
    for k in range(n // 2):
        p = (k * k + seed_offset) % n
        h[p] = 1.0
    if h.sum() != n // 2:
        h = to_binary(h, n)
    starts.append(h)
    return starts
