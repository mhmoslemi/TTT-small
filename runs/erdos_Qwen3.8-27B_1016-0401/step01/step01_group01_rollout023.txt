_NM_DX = 1.0
_NM_T = 35.0

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_at(h, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    return dx * (cmax + temp * np.log(np.sum(np.exp((c - cmax) / temp))))

def smooth_grad_at(h, dx, temp):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    w = np.exp((c - cmax) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        v2 = (ks + s >= 0) & (ks + s < n)
        g = np.zeros(n)
        g[v1] += 1.0 - h[ks[v1] - s]
        g[v2] += -h[ks[v2] + s]
        grad += w[m] * g
    return dx * grad

def _obj(h):
    return smooth_obj_at(h, _G_DX, _G_TEMP)

def _grad(h):
    return smooth_grad_at(h, _G_DX, _G_TEMP)

def _obj_sharp(h):
    return compute_c5(h, _G_DX)

def _grad_sharp(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    ks = np.arange(n)
    grad = np.zeros(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v1] += 1.0 - h[ks[v1] - s]
    grad[v2] += -h[ks[v2] + s]
    return _G_DX * grad

def _cons_fun(h):
    return float(np.sum(h) - _G_T)

def _cons_jac(h):
    return np.ones(_G_N)

def _nm_obj(x):
    x = np.clip(x, 0, 1)
    x = _normalize(x, _NM_T)
    return compute_c5(x, _NM_DX)

def _normalize(h, T):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h = h * (T / s)
    else:
        h = np.full_like(h, T / len(h))
    h = np.clip(h, 0.0, 1.0)
    d = T - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = int(np.argmax(h < 1.0))
        else:
            idx = int(np.argmin(h > 0.0))
        h[idx] = np.clip(h[idx] + d, 0.0, 1.0)
    return h

def _slsqp_pass(h, n, max_iter, ftol, temp):
    global _G_DX, _G_TEMP, _G_N, _G_T
    _G_DX = 2.0 / n
    _G_N = n
    _G_T = n / 2.0
    T = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h = _normalize(h, T)
    best = h.copy()
    best_c5 = compute_c5(h, _G_DX)
    _G_TEMP = temp
    try:
        r = minimize(_obj, h, method='SLSQP', jac=_grad,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        c5 = compute_c5(r.x, _G_DX)
        if c5 < best_c5:
            best_c5, best = c5, r.x.copy()
    except Exception:
        pass
    try:
        r = minimize(_obj_sharp, best, method='SLSQP', jac=_grad_sharp,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        c5 = compute_c5(r.x, _G_DX)
        if c5 < best_c5:
            best_c5, best = c5, r.x.copy()
    except Exception:
        pass
    best = _normalize(best, T)
    best_c5 = compute_c5(best, _G_DX)
    return best, best_c5

def _nm_polish(h, n):
    global _NM_DX, _NM_T
    _NM_DX = 2.0 / n
    _NM_T = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(h, _NM_DX)
    try:
        r = nelder_mead(_nm_obj, h.copy(), maxiter=800, xatol=1e-9, fatol=1e-13)
        rx = _normalize(np.array(r.x), _NM_T)
        c5 = compute_c5(rx, _NM_DX)
        if c5 < best_c5:
            best_c5, best = c5, rx
    except Exception:
        pass
    return best, best_c5

def _anneal_optimize(h, n, time_limit):
    temps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    iters = [600, 800, 1000, 1200, 1500, 2000, 3000]
    for i, (t, mi) in enumerate(zip(temps, iters)):
        if time.time() > time_limit:
            break
        h, c5 = _slsqp_pass(h, n, mi, 1e-14, t)
    h, c5 = _nm_polish(h, n)
    return h, c5

def _structured_perturb(h, rng, n):
    T = n / 2.0
    cands = []
    cands.append(h.copy())
    cands.append(1.0 - h)
    rev = h[::-1].copy()
    cands.append(_normalize(rev, T))
    shifted = np.roll(h, rng.randint(1, n // 3))
    cands.append(_normalize(shifted, T))
    for eps in [0.05, 0.1, 0.2]:
        noisy = h + rng.randn(n) * eps
        cands.append(_normalize(noisy, T))
    for c in cands:
        yield c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, TypeError):
        n0 = 70
        h0 = np.full(n0, 0.5)

    rng = np.random.RandomState(seed)
    best_h = _normalize(h0, n0 / 2.0).copy()
    dx0 = 2.0 / n0
    best_c5 = compute_c5(best_h, dx0)

    # Phase 1: Annealed optimization from parent
    h_a, c5_a = _anneal_optimize(best_h.copy(), n0, deadline - 300)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a

    # Phase 2: Multiple structured perturbations + optimize
    n_perturb = 12
    for i in range(n_perturb):
        if time.time() > deadline - 200:
            break
        h_p = _structured_perturb(best_h, rng, n0).__next__() if i < len(list(_structured_perturb(best_h, rng, n0))) else None
        if h_p is None:
            break
        h_opt, c5_opt = _slsqp_pass(h_p, n0, 1500, 1e-14, 0.005)
        h_nm, c5_nm = _nm_polish(h_opt, n0)
        if c5_nm < best_c5:
            best_c5, best_h = c5_nm, h_nm

    # Phase 3: Pure random restarts
    n_rand = 20
    for i in range(n_rand):
        if time.time() > deadline - 120:
            break
        h_r = rng.random(n0)
        h_r = _normalize(h_r, n0 / 2.0)
        h_opt, c5_opt = _slsqp_pass(h_r, n0, 1000, 1e-13, 0.01)
        h_nm, c5_nm = _nm_polish(h_opt, n0)
        if c5_nm < best_c5:
            best_c5, best_h = c5_nm, h_nm

    # Phase 4: Multi-resolution (upsample to 2n, optimize, downsample)
    if time.time() < deadline - 100:
        n_hi = 2 * n0
        x_lo = np.linspace(0, 2, n0, endpoint=False)
        x_hi = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_hi, x_lo, best_h, period=2.0)
        h_hi = _normalize(h_hi, n_hi / 2.0)
        h_hi_opt, c5_hi = _slsqp_pass(h_hi, n_hi, 2000, 1e-14, 0.005)
        c5_hi_eval = compute_c5(h_hi_opt, 2.0 / n_hi)
        if c5_hi_eval < best_c5:
            best_c5, best_h = c5_hi_eval, h_hi_opt
        else:
            # Try downsampling back
            x_lo_res = np.linspace(0, 2, n0, endpoint=False)
            x_hi_res = np.linspace(0, 2, n_hi, endpoint=False)
            h_lo = np.interp(x_lo_res, x_hi_res, h_hi_opt, period=2.0)
            h_lo = _normalize(h_lo, n0 / 2.0)
            h_lo_opt, c5_lo = _nm_polish(h_lo, n0)
            if c5_lo < best_c5:
                best_c5, best_h = c5_lo, h_lo_opt

    # Phase 5: Final deep polish
    if time.time() < deadline - 15:
        n_f = len(best_h)
        h_pol, c5_pol = _anneal_optimize(best_h.copy(), n_f, deadline - 10)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol

    # Final normalization and validation
    n_f = len(best_h)
    T = n_f / 2.0
    best_h = _normalize(best_h, T)
    best_c5 = compute_c5(best_h, 2.0 / n_f)

    return best_h, best_c5, n_f
