_dx = 1.0
_target = 35.0
_n_pts = 70
_t_sm = 0.02

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _obj_sharp(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(c) * _dx)

def _jac_sharp(x):
    n = _n_pts
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    ks = np.arange(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    v2 = (ks + s >= 0) & (ks + s < n)
    g = np.zeros(n)
    g[v1] += (1.0 - x[ks[v1] - s])
    g[v2] += (-x[ks[v2] + s])
    return _dx * g

def _obj_smooth(x):
    n = _n_pts
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(c)
    w = np.exp((c - cm) / _t_sm)
    return _dx * (cm + _t_sm * np.log(np.sum(w)))

def _jac_smooth(x):
    n = _n_pts
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(c)
    w = np.exp((c - cm) / _t_sm)
    w /= np.sum(w)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        m = n - 1 + s
        wm = w[m]
        if wm < 1e-14:
            continue
        if s >= 0:
            grad[:n - s] += wm * (1.0 - x[s:])
            grad[s:] += wm * (-x[:n - s])
        else:
            ss = -s
            grad[ss:] += wm * (1.0 - x[:n - ss])
            grad[:n - ss] += wm * (-x[ss:])
    return _dx * grad

    return h

            grad[:n - ss] += (-h[ss:])
        grad *= dx

    return h
