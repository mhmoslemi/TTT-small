Alternative: Do coordinate descent on the "free" parameters, and project at the end. Or: change h[i] and h[j] simultaneously (one up, one down) to maintain the constraint.

Better approach: **Paired coordinate descent**. Pick two coordinates i and j, increase h[i] by δ and decrease h[j] by δ (or vice versa). This maintains the sum constraint.

For a pair (i, j) and perturbation δ:
- h[i] → h[i] + δ
- h[j] → h[j] - δ
- All other h unchanged

The correlation changes to: corr_new(o) = corr(o) + δ * (effect of +δ at i) - δ * (effect of +δ at j)

This is still linear in δ for each lag, so the objective max_o (corr(o) + δ * delta_o) is convex in δ.

We can find the optimal δ by ternary search on a feasible interval (limited by [0,1] bounds).

This is promising! Let me implement this.

For the effect of changing h[i] by δ:
- In the h array: position i changes by δ
- In the (1-h) array: position i changes by -δ
- corr(o) = sum_x h[x] * (1-h[x+o])
  - Terms with x=i: h[i+δ]*(1-h[i+o]) → change by δ*(1-h[i+o])
  - Terms with x+o=i, i.e., x=i-o: h[i-o]*(1-h[i+δ]) → change by -δ*h[i-o]
  - Total change: δ*((1-h[i+o]) - h[i-o]) [with boundary handling]

So for pair (i,j) with perturbation δ (h[i]+=δ, h[j]-=δ):
delta_corr(o) = δ * [(1-h[i+o]) - h[i-o] - (1-h[j+o]) + h[j-o]]

And we minimize max_o (corr(o) + δ * delta_corr(o)) over feasible δ.

This is a max of linear functions of δ, minimized over an interval. Can be solved by:
1. Finding the δ where the active constraints cross
2. Or simply ternary search (since it's convex)

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        if abs(diff) < 1e-14:
            break

                diff += s

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                s = min(diff, 1.0 - h[i])
                if s <= 1e-15:
                    continue
                h[i] += s
                diff -= s
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                s = min(-diff, h[i])
                if s <= 1e-15:
                    continue
                h[i] -= s
                diff += s
                if abs(diff) < 1e-14:
                    break
    return h

def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -300, 300)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se
    obj = (mc + np.log(se) / T) * dx
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-8:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return obj, grad * dx

def adam_optimize(h, target, lrs, temps, max_iters_total, t_start, deadline):
    n = len(h)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    for T in temps:
        if time.time() > deadline:
            break
        lr = lrs
        for it in range(max_iters_total):
            if time.time() > deadline:
                break
            obj, grad = softmax_obj_grad(h, T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-16:
                h = h_new
    return h

def fine_polish(h, target, n_rounds, deadline):
    """Very fine local search with small random perturbations guided by gradient."""
    n = len(h)
    c5_cur = compute_c5(h)
    rng = np.random.default_rng(12345)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        obj, grad = softmax_obj_grad(h, 1.0)
        # Try small step in -grad direction with noise
        for attempt in range(8):
            scale = 0.02 * (0.7 ** attempt)
            step = -grad + scale * 0.01 * rng.standard_normal(n)
            h_try = project_constraint(h + scale * step, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-16:
                h = h_try
                c5_cur = c5_try
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.default_rng(seed)

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Adam with moderate temperatures (explore)
    if time.time() < deadline:
        h_p1 = h_best.copy()
        h_p1 = adam_optimize(h_p1, target, lr=0.3,
                            temps=[80, 50, 30, 20],
                            max_iters_total=800, t_start=t0, deadline=deadline)
        c5_p1 = compute_c5(h_p1)
        if c5_p1 < c5_best:
            c5_best = c5_p1
            h_best = h_p1

    # Phase 2: Adam with low temperatures (exploit)
    if time.time() < deadline:
        h_p2 = h_best.copy()
        h_p2 = adam_optimize(h_p2, target, lr=0.15,
                            temps=[15, 10, 7, 5, 3, 2],
                            max_iters_total=600, t_start=t0, deadline=deadline)
        c5_p2 = compute_c5(h_p2)
        if c5_p2 < c5_best:
            c5_best = c5_p2
            h_best = h_p2

    # Phase 3: Very fine Adam
    if time.time() < deadline:
        h_p3 = h_best.copy()
        h_p3 = adam_optimize(h_p3, target, lr=0.05,
                            temps=[1.5, 1.0, 0.7, 0.5, 0.3],
                            max_iters_total=2000, t_start=t0, deadline=deadline)
        c5_p3 = compute_c5(h_p3)
        if c5_p3 < c5_best:
            c5_best = c5_p3
            h_best = h_p3

    # Phase 4: Fine polish (gradient-guided small steps)
    if time.time() < deadline - 30:
        h_best, c5_pol = fine_polish(h_best, target, n_rounds=2000, deadline=deadline - 30)
        if c5_pol < c5_best:
            c5_best = c5_pol

    # Phase 5: Try interpolation to slightly different n for potential improvement
    for n_try in [250, 300, 320, 200]:
        if time.time() > deadline - 60:
            break
        if n_try == n:
            continue
        target_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_constraint(h_try, target_try)
        h_try = adam_optimize(h_try, target_try, lr=0.1,

            target = target_try

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
