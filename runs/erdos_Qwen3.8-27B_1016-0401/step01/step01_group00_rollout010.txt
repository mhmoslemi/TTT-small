from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_value(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T)

def soft_max_weights(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    return e / np.sum(e)

def compute_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    w = soft_max_weights(corr, T)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        s = i - center
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] += -wi * h[:n - s]
        else:
            p = -s
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    return grad * dx

def project_feasible(x, target_sum):
    """Project x onto {h: 0≤h_i≤1, Σh_i = target_sum}"""
    x = x.copy()
    for _ in range(100):
        # Find lambda such that sum(clip(x + lambda, 0, 1)) = target_sum
        lo, hi = -max(x.max(), 0) - 2, -min(x.min(), 0) + 2
        for _ in range(60):
            mid = (lo + hi) / 2
            proj = np.clip(x + mid, 0, 1)
            s = proj.sum()
            if s > target_sum:
                hi = mid
            else:
                lo = mid
        x = np.clip(x + (lo + hi) / 2, 0, 1)
        if abs(x.sum() - target_sum) < 1e-12:
            break
    return x

def nesterov_gradient(h, n, T, lr, momentum, target_sum, n_iters):
    """Nesterov accelerated projected gradient descent"""
    x = h.copy()
    v = np.zeros(n)
    for it in range(n_iters):
        # Nesterov step
        y = x + momentum * v
        g = compute_grad(y, n, T)
        new_x = project_feasible(y - lr * g, target_sum)
        v = new_x - x
        x = new_x
    return x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_obj(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def softmax_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    w = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        s = i - center
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] += -wi * h[:n - s]
        else:
            p = -s
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    return grad * dx

def project_feasible(x, target):
    x = x.copy()

        lo, hi = -x.max() - 1, -x.min() + 1
        for _ in range(80):
