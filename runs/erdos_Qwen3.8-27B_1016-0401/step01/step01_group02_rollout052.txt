import time

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _proj(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        mask = np.ones(n, dtype=bool)
        if d > 0:
            mask = h < 1.0  # can only decrease
        else:
            mask = h > 0.0  # can only increase
        if not mask.any():
            break
        adj = d / mask.sum()
        h[mask] -= adj
        np.clip(h, 0.0, 1.0, out=h)
    # Final exact adjustment
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        if msk.sum() > 0:
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _solve_lp(C, n):
    """Solve min t s.t. C@x <= t, sum x = n/2, 0<=x<=1."""
    A_ub = np.hstack([C, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    bounds = [(0.0, 1.0)]*n + [(None, None)]
    res = linprog(np.zeros(n+1), A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _alt_lp(h0, n, n_iter=100, alpha=0.5, tmax=60.0):
    """Alternating LP to minimize max correlate(h, 1-h)."""
    t0 = time.time()
    h = _proj(h0, n).copy()
    g = 1.0 - h
    best = h.copy()
    bestc = _c5(h)
    
    for it in range(n_iter):
        if time.time() - t0 > tmax:
            break
        
        # Solve for h given g
        k_max = n - 1
        C = np.zeros((2*n-1, n))
        for k_idx in range(2*n-1):
            k = k_idx - k_max
            lo = max(0, k)
            hi = min(n-1, n-1+k)
            if lo <= hi:
                C[k_idx, lo:hi+1] = g[lo-k:hi-k+1]
        h_new = _solve_lp(C, n)
        if h_new is not None:
            h_new = _proj(h_new, n)
            h = (1-alpha)*h + alpha*h_new
            h = _proj(h, n)
            g = 1.0 - h
        
        # Solve for g given h
        D = np.zeros((2*n-1, n))
        for k_idx in range(2*n-1):
            k = k_idx - k_max
            lo = max(0, -k)
            hi = min(n-1, n-1-k)
            if lo <= hi:
                D[k_idx, lo:hi+1] = h[lo+k:hi+k+1]
        g_new = _solve_lp(D, n)
        if g_new is not None:
            g_new = _proj(g_new, n)
            g = (1-alpha)*g + alpha*g_new
            h = 1.0 - g
            h = _proj(h, n)
            g = 1.0 - h
        
        c = _c5(h)
        if c < bestc:
            best, bestc = h.copy(), c
    
    return best, bestc

import numpy as np
from scipy.optimize import linprog
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] -= d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-13:
        m = (h < 1.0) if r > 0 else (h > 0.0)
        if m.any():
            h[m] += r / m.sum()
    return np.clip(h, 0.0, 1.0)

def _solve_lp(C, n):
    A = np.hstack([C, -np.eye(2*n-1)])
    b = np.zeros(2*n-1)
    Aeq = np.zeros((1, n+1))
    Aeq[0, :n] = 1.0
    beq = np.array([0.5*n])
    bounds = [(0.0, 1.0)]*n + [(-np.inf, np.inf)]
    r = linprog(np.zeros(n+1), A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                bounds=bounds, method='highs')
    return r.x[:n] if r.success else None

def _alt_lp(h0, n, iters=100, alpha=0.5, tmax=30.0):
    t0 = time.time()
    h = _proj(h0, n).copy()
    best, bestc = h.copy(), _c5(h)
    for _ in range(iters):
        if time.time() - t0 > tmax:
            break
        g = 1.0 - h
        # LP for h
        C = np.zeros((2*n-1, n))
        for ki in range(2*n-1):
            k = ki - (n-1)
            lo, hi = max(0, k), min(n-1, n-1+k)
            if lo <= hi:
                C[ki, lo:hi+1] = g[lo-k:hi-k+1]
        hnew = _solve_lp(C, n)
        if hnew is not None:
            h = _proj((1-alpha)*h + alpha*hnew, n)
        # LP for g
        D = np.zeros((2*n-1, n))
        for ki in range(2*n-1):
            k = ki - (n-1)

import numpy as np
from scipy.optimize import linprog
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] -= d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        m = (h < 1.0) if r > 0 else (h > 0.0)
        if m.any():
            h[m] += r / m.sum()
    return np.clip(h, 0.0, 1.0)

def _solve_lp(C, n):
    M = 2 * n - 1
    A = np.hstack([C, -np.eye(M)])
    b = np.zeros(M)
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    beq = np.array([0.5 * n])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    try:
        r = linprog(np.zeros(n + 1), A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                    bounds=bounds, method='highs')
        if r.success:
            return r.x[:n]
    except Exception:
        pass
    return None

def _alt_lp(h0, n, iters=150, alpha=0.6, tmax=40.0):
    t0 = time.time()
    h = _proj(h0, n).copy()
    best, bestc = h.copy(), _c5(h)
    for it in range(iters):
        if time.time() - t0 > tmax:
            break
        g = 1.0 - h
        C = np.zeros((2 * n - 1, n))
        for ki in range(2 * n - 1):
            k = ki - (n - 1)
            lo, hi = max(0, k), min(n - 1, n - 1 + k)
            if lo <= hi:
                C[ki, lo:hi + 1] = g[lo - k:hi - k + 1]
        hnew = _solve_lp(C, n)
        if hnew is not None:
            h = _proj((1 - alpha) * h + alpha * hnew, n)
        D = np.zeros((2 * n - 1, n))
        for ki in range(2 * n - 1):
            k = ki - (n - 1)
            lo, hi = max(0, -k), min(n - 1, n - 1 - k)
            if lo <= hi:
                D[ki, lo:hi + 1] = h[lo + k:hi + k + 1]
        gnew = _solve_lp(D, n)
        if gnew is not None:
            g = _proj((1 - alpha) * g + alpha * gnew, n)
            h = _proj(1.0 - g, n)
        c = _c5(h)
        if c < bestc:
            best, bestc = h.copy(), c
    return best, bestc

def _local_search(h0, n, tmax=100.0):
    t0 = time.time()
    h = _proj(h0, n).copy()
    c_cur = _c5(h)
    best, best_c = h.copy(), c_cur
    tgt = 0.5 * n
    n_moves = 0
    for epoch in range(200):
        if time.time() - t0 > tmax:
            break
        T = max(0.003, 0.08 * (0.96 ** epoch))
        step = T * 4.0
        for _ in range(8000):
            if time.time() - t0 > tmax:
                break
            n_moves += 1
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            dv = np.random.uniform(-step, step)
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
            s = h2.sum() - tgt
