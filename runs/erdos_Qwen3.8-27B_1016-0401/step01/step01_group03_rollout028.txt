def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj < 1.0 - 1e-10) if diff > 0 else (proj > 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def lse_objective(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = beta * (corr - mx)
    shifted -= np.max(shifted)  # numerical stability
    logsum = np.log(np.sum(np.exp(shifted))) + np.max(shifted) / beta  # hmm
    # Actually: log(sum(exp(beta*(c-M)))) = beta*(max(c)-M) + log(sum(exp(beta*(c-M-max(c)+M))))
    # = log(sum(exp(beta*(c - mx)))) since mx = max(c)
    # = log(sum(exp(shifted))) where shifted = beta*(c - mx), max(shifted) = 0
    return dx * (mx + logsum / beta), corr

def smooth_obj_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = beta * (corr - mx)  # max is 0
    exps = np.exp(shifted)
    sum_exp = np.sum(exps)
    obj = dx * (mx + np.log(sum_exp) / beta)
    w = dx * exps / sum_exp  # weight for each lag
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m >= 0:
            # c[k] = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
            jmax = n - m
            # d c[k] / d h[j] = (1-h[j+m]) for j in [0, jmax-1]
            #                    -h[j-m] for j in [m, n-1]
            idx1 = np.arange(jmax)
            grad[idx1] += wk * (1.0 - h[idx1 + m])
            idx2 = np.arange(m, n)
            grad[idx2] -= wk * h[idx2 - m]

            #                    -h[j+p] for j in [0, n-1-p]
            idx1 = np.arange(p, n)
            grad[idx1] += wk * (1.0 - h[idx1 - p])
            idx2 = np.arange(n - p)
            grad[idx2] -= wk * h[idx2 + p]
    return obj, grad
