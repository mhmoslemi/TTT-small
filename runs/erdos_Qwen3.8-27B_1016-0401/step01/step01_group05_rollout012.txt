        return g
    def cons_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def anneal(h_init, n, target, rng, deadline, t0=0.005, t_end=0.00005):
    h_curr = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    c5_curr = best_c5
    T = t0
    nsteps = 50000
    cool = (t_end / t0) ** (1.0 / nsteps)
    for it in range(nsteps):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        s = 0.02 * (0.5 + 0.5 * T / t0)
        d = rng.uniform(-s, s)
        h_try = h_curr.copy()
        h_try[i] = np.clip(h_try[i] + d, 0, 1)
        h_try[j] = np.clip(h_try[j] - d, 0, 1)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_curr:
            h_curr = h_try
            c5_curr = c5_try
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                if it % 5000 == 0 and it > 0 and time.time() < deadline - 40:
                    h_s = run_slsqp(best_h, n, target, 2000, maxiter=80)
                    c5_s = compute_c5(h_s)
                    if c5_s < best_c5:
                        best_c5, best_h = c5_s, h_s
                        c5_curr = c5_s
        else:
            ratio = (c5_try - c5_curr) / max(T, 1e-12)
            if ratio < 50 and rng.random() < np.exp(-ratio):
                h_curr = h_try
                c5_curr = c5_try
        T *= cool
    return best_h, best_c5

def fine_search(h, n, target, rng, deadline, max_iters=80000):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)
    step = 0.003
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 10000 == 9000:
            step *= 0.65
        i, j = rng.choice(n, size=2, replace=False)
        d = rng.uniform(-step, step)
        h_try = h_curr.copy()
        h_try[i] = np.clip(h_try[i] + d, 0, 1)
        h_try[j] = np.clip(h_try[j] - d, 0, 1)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_curr:
            h_curr, c5_curr = h_try, c5_try
    return h_curr, c5_curr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5

    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, n, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha (original n)
    for a in [50, 100, 200, 500, 1000, 3000, 8000, 20000]:
        if time.time() > deadline - 250:
            break
        h_new = run_slsqp(best_h, n, target, a, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 2: Explore other n values and starting points
    n_candidates = [60, 80, 96, 100, 120, 140]
    for n2 in n_candidates:
        if time.time() > deadline - 300:
            break
        t2 = n2 / 2.0
        starts = [np.ones(n2) * 0.5, rng.uniform(0, 1, n2)]
        for hs in starts:
            if time.time() > deadline - 200:
                break
            hs = make_feasible(hs, n2, t2)
            for a in [200, 1000, 5000]:
                if time.time() > deadline - 150:
                    break
                hs = run_slsqp(hs, n2, t2, a, maxiter=200)
            c5_2 = compute_c5(hs)
            if c5_2 < best_c5:
                best_c5, best_h, n, target = c5_2, hs, n2, t2

    # Phase 3: Simulated annealing for global exploration
    if time.time() < deadline - 120:
        h_sa, c5_sa = anneal(best_h, n, target, rng, deadline - 100)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa
        # SLSQP refinement after annealing
        for a in [3000, 10000]:
            if time.time() > deadline - 60:
                break
            h_new = run_slsqp(best_h, n, target, a, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new

    # Phase 4: Second annealing round from best (different seed)
    if time.time() < deadline - 80:
        rng2 = np.random.RandomState(seed + 12345)
        h_sa2, c5_sa2 = anneal(best_h, n, target, rng2, deadline - 60, t0=0.003, t_end=0.00003)
        if c5_sa2 < best_c5:
            best_c5, best_h = c5_sa2, h_sa2

    # Phase 5: Fine local search
    if time.time() < deadline - 15:
        h_ls, c5_ls = fine_search(best_h, n, target, rng, deadline - 5)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 4:
        h_final = run_slsqp(best_h, n, target, 10000, maxiter=300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final

    best_h = make_feasible(best_h, n, target)
    return best_h.tolist(), compute_c5(best_h), n
