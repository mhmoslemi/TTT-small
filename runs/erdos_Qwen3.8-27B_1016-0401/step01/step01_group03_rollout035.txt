def reweighting_optimize(h0, dx, target_sum, t_limit, max_iters=2000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Initialize weights
    corr = np.correlate(h, 1-h, mode='full')
    c_vals = corr * dx
    w = np.ones(2*n-1) / (2*n-1)

        c_vals = np.correlate(h, 1-h, mode='full') * dx
        max_c = np.max(c_vals)
        if max_c > 0:
            w = np.exp(c_vals / (max_c * 0.3 + 1e-10))  # soft weighting
            w /= np.sum(w)
        
        # Solve: min sum_s w_s * corr[s] subject to constraints
        # corr[s] = sum_i h[i+shift_s]*(1-h[i]) where shift_s = s-(n-1)
        # This is quadratic in h
        
        # Use projected gradient
        for inner in range(50):
            # Compute gradient
            grad = np.zeros(n)
            for s_idx in range(2*n-1):
                shift = s_idx - (n-1)
                ws = w[s_idx]
                if ws < 1e-10:
                    continue
                # c(shift) = sum_{valid i} h[i+shift]*(1-h[i])
                if shift >= 0:
                    # i from 0 to n-1-shift
                    # dc/dh[j]:
                    # For j in [shift, n-1]: h[j] appears as h[i+shift] with i=j-shift
                    #   contribution: w * (1-h[j-shift])

                        grad[j] += ws * (1 - h[j-shift])
                    for j in range(0, n-shift):
                        grad[j] -= ws * h[j+shift]
                else:

                    for j in range(0, n-p):

                    for j in range(p, n):
                        grad[j] -= ws * h[j-p]
            
            h = project_feasible(h - 0.01 * grad, target_sum)

        if c5_new < best_c5:

            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(np.minimum(beta * shifted, 50))
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad


def optimize_continuous(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [50.0, 200.0, 800.0, 3000.0, 8000.0]:
        if time.time() > t_limit:
            break
        lr = 1.0
        stall = 0
        for it in range(80000):
            if time.time() > t_limit:
                break
            if it % 200 == 0:
                _, grad = smooth_obj_grad(h, dx, beta)
            else:
                _, grad = smooth_obj_grad(h, dx, beta)
            h_new = project_feasible(h - lr * grad, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                if lr < 3.0:
                    lr *= 1.03
                stall = 0
            else:
                stall += 1
                lr *= 0.95
            h = h_new
            if stall > 2000 or lr < 1e-8:
                break
    return best_h, best_c5


def to_binary(h):
    n = len(h)
    ts = n // 2
    b = (h > 0.5).astype(float)
    oc = int(np.sum(b))
    if oc > ts:
        idxs = np.where(b == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            b[i] = 0.0
    elif oc < ts:
        idxs = np.where(b == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            b[i] = 1.0
    return b

            h[a] = 0; h[b] = 1

            h[a] = 1; h[b] = 0
        cur_c5 = compute_c5(h, dx)

            best = h.copy()
        if it % 5000 == 0:

def resample(h_old, n_old, n_new):

    x_new = np.linspace(0.0, 2.0, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return h_interp
