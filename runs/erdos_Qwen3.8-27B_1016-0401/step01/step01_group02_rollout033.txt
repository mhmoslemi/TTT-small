import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sm(h, dx, sc):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    return (m + np.log(np.exp(sc * (c - m)).sum()) / sc) * dx

def _sm_grad(h, n, dx, sc):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    w = np.exp(sc * (c - m))
    w /= w.sum()
    # A[m] = w[m+n-1], B[m] = w[m+n-1]+w[-m+n-1]
    # T1[j] = sum(w[n-1-j : 2n-1-j])
    wc = np.concatenate([[0.0], np.cumsum(w)])
    T1 = wc[2*n-1-np.arange(n)] - wc[n-1-np.arange(n)]
    # B_full[k] = w[k] + w[2n-2-k]
    Bf = w + w[::-1]
    T2 = np.zeros(n)
    for j in range(n):
        T2[j] = np.dot(h, Bf[n-1-j:2*n-1-j])
    return dx * (T1 - T2)

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _opt_slsqp(h0, n, dx, sc, mi):
    def f(x):
        x = np.clip(x, 0.0, 1.0)
        return _sm(x, dx, sc)
    def g(x):
        x = np.clip(x, 0.0, 1.0)
        return _sm_grad(x, n, dx, sc)
    def ceq(x):
        return x.sum() - 0.5 * n
    def ceqj(x):
        return np.ones(n)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': ceqj}]
    try:
        r = minimize(f, h0, method='SLSQP', jac=g, bounds=bds,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-16})
        return _proj(r.x, n)
    except:
        return _proj(h0, n)

def _opt_lBfgs(h0, n, dx, sc, mi, pen):
    def f(x):
        x = np.clip(x, 0.0, 1.0)
        return _sm(x, dx, sc) + pen * (x.sum() - 0.5*n)**2
    def g(x):
        x = np.clip(x, 0.0, 1.0)
        return _sm_grad(x, n, dx, sc) + 2*pen*(x.sum()-0.5*n)*np.ones(n)
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0, method='L-BFGS-B', jac=g, bounds=bds,
                     options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-10})
        return _proj(r.x, n)
    except:
        return _proj(h0, n)

def _frac_ls(h0, n, tmax, niter, lo, hi, T=0.02):

    cw = h0.copy()

    best = h0.copy()
