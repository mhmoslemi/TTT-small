import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def sm_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    cg = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[cg + o]
        if w < 1e-11:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return grad * dx

def fix(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
            c = int(m.sum())
            if c == 0:
                break
            h[m] += d / c
        else:
            m = h > 1e-15
            c = int(m.sum())
            if c == 0:
                break
            h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                s = min(d, 1.0 - h[i])
                h[i] += s
                d -= s
                if d < 1e-15:
                    break
        else:
            d = -d
            order = np.argsort(h)
            for i in order:
                s = min(d, h[i])
                h[i] -= s
                d -= s
                if d < 1e-15:
                    break
    return h

def pg_optimize(h, target, temps, steps, t_limit, lr_base=1.0):
    n = len(h)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for idx, T in enumerate(temps):
        lr = lr_base / (0.5 + T * 0.02)
        for step in range(steps):
            if time.time() > t_limit:
                break
            grad = sm_grad(h, T)
            gn = np.linalg.norm(grad)
            if gn < 1e-12:
                break
            h_try = np.clip(h - lr * grad, 0.0, 1.0)
            h_try = fix(h_try, target)
            # Backtracking
            v0 = sm_obj(h, T)
            v1 = sm_obj(h_try, T)
            if v1 > v0:
                h_try = np.clip(h - 0.5*lr*grad, 0.0, 1.0)
                h_try = fix(h_try, target)
                v1 = sm_obj(h_try, T)
                if v1 > v0:
                    h_try = np.clip(h - 0.25*lr*grad, 0.0, 1.0)
                    h_try = fix(h_try, target)
            h = h_try
            if step % 200 == 199 or step == steps - 1:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    h_best = h.copy()
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def et():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Annealed PG at parent resolution
    temps1 = [150, 100, 70, 50, 35, 25, 18, 13, 10, 7, 5, 4, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
    h_best, c5_best = pg_optimize(h_best, target, temps1, 600, t0 + budget_s - 400, lr_base=2.0)

    # Phase 2: Complement and perturbations
    candidates = [h_best.copy(), 1.0 - h_best]
    for _ in range(3):
        cands_noise = h_best + rng.normal(0, 0.15, n)
        candidates.append(fix(cands_noise, target))
    for _ in range(3):
        idx = rng.integers(0, n, size=n//5)
        cands_flip = h_best.copy()
        cands_flip[idx] = 1.0 - cands_flip[idx]
        candidates.append(fix(cands_flip, target))

    for hc in candidates:
        if et() > budget_s - 350:
            break
        hc = fix(hc, target)
        temps2 = [40, 25, 15, 10, 7, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
        h_c, c5_c = pg_optimize(hc, target, temps2, 500, t0 + budget_s - 300, lr_base=2.0)
        if c5_c < c5_best:
            c5_best = c5_c
            h_best = h_c

    # Phase 3: Multi-resolution refinement
    for n_hi in [420, 560]:
        if et() > budget_s - 200:
            break
        n_old = len(h_best)
        if n_hi <= n_old:
            continue
        target_hi = n_hi / 2.0
        x_old = np.arange(n_old) * (2.0 / n_old)
        x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = fix(h_hi, target_hi)
        temps3 = [80, 50, 35, 25, 18, 13, 10, 7, 5, 4, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3, 0.15, 0.1]
        h_hi, c5_hi = pg_optimize(h_hi, target_hi, temps3, 400, t0 + budget_s - 80, lr_base=2.0)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi

    # Phase 4: Final polish at best resolution
    n_f = len(h_best)
    target_f = n_f / 2.0
    temps4 = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
    h_best, c5_best = pg_optimize(h_best, target_f, temps4, 1000, t0 + budget_s - 10, lr_base=1.0)

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
