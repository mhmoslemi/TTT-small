    return float(np.max(c) * dx)

def grad_active(h, dx, s):
    """Gradient of the single bilinear form at shift s."""
    n = len(h)
    ks = np.arange(n)
    grad = np.zeros(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    grad[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])
    return dx * grad

def grad_smooth_max(h, dx, tau):
    """Gradient of smooth max approximation."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        if w[m] < 1e-12:
            continue
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += w[m] * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += w[m] * (-h[ks[v2] + s])
    return dx * grad

def project_simplex(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0, 1)
    # Binary search for threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

def subgradient_method(h, dx, n_iters, target_sum, initial_lr=0.05, momentum=0.9):
    """Subgradient descent with momentum on smooth max."""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        tau = max(0.01, 0.05 * (1 - 0.001 * it))
        s_star = int(np.argmax(c)) - (n - 1)
        g = grad_active(h, dx, s_star)
        
        lr = initial_lr / (1 + 0.01 * it)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_simplex(h_new, target_sum)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

The objective is F(h) = max_s f_s(h) where f_s(h) = dx * sum_i h[i]*(1-h[i+s]).

The subdifferential of F at h (when s* is the unique maximizer) is {∇f_{s*}(h)}.
When multiple s are tied, the subdifferential is the convex hull of their gradients.

The smooth max F_τ(h) = τ * log(sum exp(f_s/τ)) has gradient:
∇F_τ = sum_s w_s * ∇f_s where w_s = exp(f_s/τ) / sum_j exp(f_j/τ)

This is always well-defined and gives a descent direction (it's a convex function... wait, is F_τ convex?).

f_s(h) = dx * sum_i h[i]*(1-h[i+s]) = dx * (sum_i h[i] - sum_i h[i]*h[i+s])

The second term sum_i h[i]*h[i+s] is bilinear, not convex. So f_s is not convex, and neither is F_τ.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def grad_single(h, dx, s):
    n = len(h)
    ks = np.arange(n)
    g = np.zeros(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    g[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    g[v2] += (-h[ks[v2] + s])
    return dx * g

def smooth_max_obj(h, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    return dx * (cm + tau * np.log(np.sum(np.exp((c - cm) / dx / tau))))

def smooth_max_grad(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    log_w = (c / dx - cm / dx) / tau
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    g = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        g[v1] += w[m] * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        g[v2] += w[m] * (-h[ks[v2] + s])
    return dx * g

def smooth_obj_with_constraint(h, dx, tau, target_sum, mu):
    return smooth_max_obj(h, dx, tau) + mu * (np.sum(h) - target_sum) ** 2

    g = smooth_max_grad(h, dx, tau)
    g += 2 * mu * (np.sum(h) - target_sum) * np.ones_like(h)
    return g

        s_star = int(np.argmax(c)) - (n - 1)
        g = grad_single(h, dx, s_star)
        lr = 0.02 / (1 + 0.005 * it)
        v = momentum * v - lr * g
        h_new = h + v

        hn, cn = _slsqp_exact(best_h, n0, 2000, 1e-15)
        hn, cn = normalize(hn, tgt0), compute_c5(normalize(hn, tgt0), dx0)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 3: Subgradient momentum
    if time.time() - t0 < budget_s - 120:
        hs, cs = subgrad_mom(best_h, dx0, tgt0, 4000, lr0=0.05)
        hs, cs = _slsqp(hs, n0, 800, 1e-14, 0.008)
        hs, cs = normalize(hs, tgt0), compute_c5(normalize(hs, tgt0), dx0)
        if cs < best_c5:
            best_c5, best_h = cs, hs

    # Phase 4: DE global search
    _P_DX, _P_TAU, _P_TARGET, _P_N = dx0, 0.05, tgt0, n0
    if time.time() - t0 < budget_s - 100:
        try:
            de_r = differential_evolution(
                _de_obj, [(0.0, 1.0)] * n0, seed=seed,
                maxiter=80, popsize=25, tol=1e-8,
                mutation=(0.3, 1.0), recombination=0.7,
                polish=False, init='latinhypercube')
            hd = normalize(de_r.x, tgt0)
            cd = compute_c5(hd, dx0)
            if cd < best_c5:
                best_c5, best_h = cd, hd
            hd, cd = _slsqp(hd, n0, 800, 1e-13, 0.01)
            hd, cd = normalize(hd, tgt0), compute_c5(normalize(hd, tgt0), dx0)
            if cd < best_c5:
                best_c5, best_h = cd, hd
        except Exception:
            pass

    # Phase 5: Random restarts with subgradient + SLSQP
    for i in range(35):
        if time.time() - t0 > budget_s - 80:
            break
        hr = rng.random(n0)
        hr, cr = subgrad_mom(hr, dx0, tgt0, 500, lr0=0.04)
        hr, cr = _slsqp(hr, n0, 500, 1e-13, 0.015)
        hr, cr = normalize(hr, tgt0), compute_c5(normalize(hr, tgt0), dx0)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 6: Multi-resolution
    for n_alt in [100, 140, 50, 200]:
        if time.time() - t0 > budget_s - 50:
            break
        dxa = 2.0 / n_alt
        tga = n_alt / 2.0
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xa = np.linspace(0, 2, n_alt, endpoint=False)
        hi = np.interp(xa, xo, np.concatenate([best_h, [best_h[0]]]))
        hi = normalize(hi, tga)
        ha, ca = _slsqp(hi, n_alt, 600, 1e-13, 0.02)
        ha, ca = normalize(ha, tga), compute_c5(normalize(ha, tga), dxa)
        if ca < best_c5:
            best_c5, best_h = ca, ha
        if time.time() - t0 > budget_s - 40:
            break
        hr2 = rng.random(n_alt)
        hr2, cr2 = subgrad_mom(hr2, dxa, tga, 400)
        hr2, cr2 = _slsqp(hr2, n_alt, 400, 1e-12, 0.015)
        hr2, cr2 = normalize(hr2, tga), compute_c5(normalize(hr2, tga), dxa)
        if cr2 < best_c5:
            best_c5, best_h = cr2, hr2

    # Phase 7: Final polish
    if time.time() - t0 < budget_s - 10:
        nf = len(best_h)
        dxf = 2.0 / nf
        hp, cp = _slsqp(best_h, nf, 3000, 1e-15, 0.003)
        hp, cp = normalize(hp, nf / 2.0), compute_c5(normalize(hp, nf / 2.0), dxf)
        if cp < best_c5:
            best_c5, best_h = cp, hp
        if time.time() - t0 < budget_s - 5:
            hp2, cp2 = _slsqp_exact(best_h, nf, 1000, 1e-15)
            hp2, cp2 = normalize(hp2, nf / 2.0), compute_c5(normalize(hp2, nf / 2.0), dxf)
            if cp2 < best_c5:
                best_c5, best_h = cp2, hp2

    nf = len(best_h)
    dxf = 2.0 / nf
    best_h = normalize(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h, dxf)
    return best_h, best_c5, nf
