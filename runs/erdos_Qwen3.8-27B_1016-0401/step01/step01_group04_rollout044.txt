def to_binary(h, S):
    S_int = int(round(S))
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S_int:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - S_int):
            h_b[one_idx[order[i]]] = 0
    elif ones < S_int:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(S_int - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def sa_diverse(h, n, S, rng, t_end, best_c5, n_iter=600000):
    S_int = int(round(S))
    h_sa = to_binary(h, S).copy()
    c5_sa = compute_c5(h_sa)
    best = h_sa.copy()
    best_c5_local = min(c5_sa, best_c5)
    temp = 0.005
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 8000 == 0 and it > 0:
            temp *= 0.994
        
        mt = rng.randint(5)
        if mt == 0:
            i, j = rng.randint(n, size=2)
            if h_sa[i] == h_sa[j]: continue
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            c5_new = compute_c5(h_sa)
            d = c5_new - c5_sa
            if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-15)):
                c5_sa = c5_new
                if c5_new < best_c5_local:
                    best_c5_local = c5_new; best = h_sa.copy()
            else:
                h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        elif mt == 1:
            oi = np.where(h_sa == 1)[0]
            if len(oi) == 0: continue
            i = oi[rng.randint(len(oi))]
            j = i + rng.choice([-1, 1])
            if 0 <= j < n and h_sa[j] == 0:
                h_sa[i], h_sa[j] = 0, 1
                c5_new = compute_c5(h_sa)
                d = c5_new - c5_sa
                if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-15)):
                    c5_sa = c5_new
                    if c5_new < best_c5_local:
                        best_c5_local = c5_new; best = h_sa.copy()
                else:
                    h_sa[i], h_sa[j] = 1, 0
        elif mt == 2:
            i1, j1, i2, j2 = rng.choice(n, size=4, replace=False)
            if h_sa[i1] == h_sa[j1] or h_sa[i2] == h_sa[j2]: continue
            h_sa[i1], h_sa[j1] = h_sa[j1], h_sa[i1]
            h_sa[i2], h_sa[j2] = h_sa[j2], h_sa[i2]
            c5_new = compute_c5(h_sa)
            d = c5_new - c5_sa
            if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-15)):
                c5_sa = c5_new
                if c5_new < best_c5_local:
                    best_c5_local = c5_new; best = h_sa.copy()
            else:
                h_sa[i1], h_sa[j1] = h_sa[j1], h_sa[i1]
                h_sa[i2], h_sa[j2] = h_sa[j2], h_sa[i2]
        elif mt == 3:
            ln = rng.randint(2, 12)
            lo = rng.randint(0, n - ln)
            h_sa[lo:lo+ln] = h_sa[lo:lo+ln][::-1]
            c5_new = compute_c5(h_sa)
            d = c5_new - c5_sa
            if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-15)):
                c5_sa = c5_new
                if c5_new < best_c5_local:
                    best_c5_local = c5_new; best = h_sa.copy()
            else:
                h_sa[lo:lo+ln] = h_sa[lo:lo+ln][::-1]
        else:
            # Long-range swap: swap a 1 with a far 0
            oi = np.where(h_sa == 1)[0]
            zi = np.where(h_sa == 0)[0]
            if len(oi) == 0 or len(zi) == 0: continue
            i = oi[rng.randint(len(oi))]
            j = zi[rng.randint(len(zi))]
            if abs(i - j) < n // 3: continue
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            c5_new = compute_c5(h_sa)
            d = c5_new - c5_sa
            if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-15)):
                c5_sa = c5_new
                if c5_new < best_c5_local:
                    best_c5_local = c5_new; best = h_sa.copy()
            else:
                h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
    
    return best, best_c5_local

def local_search(h, n, S, best_c5, t_deadline, max_sweeps=200):
    h_l = h.copy()
    c5_l = compute_c5(h_l)
    best = h_l.copy()
    best_c5_local = min(c5_l, best_c5)
    for sweep in range(max_sweeps):
        if time.time() > t_deadline: break
        improved = False
        oi = list(np.where(h_l == 1)[0])
        zi = list(np.where(h_l == 0)[0])
        for i in oi:
            if time.time() > t_deadline: break
            for j in zi:
                h_l[i], h_l[j] = h_l[j], h_l[i]
                c5_new = compute_c5(h_l)
                if c5_new < best_c5_local - 1e-14:
                    best_c5_local = c5_new
                    best = h_l.copy()
                    improved = True
                    zi.remove(j)
                    oi.remove(i)
                    oi.append(j)
                    zi.append(i)
                else:
                    h_l[i], h_l[j] = h_l[j], h_l[i]
        if not improved: break
    return best, best_c5_local

def continuous_refine(h, n, S, t_end, rng):
    S_val = float(S)
    best_h = h.astype(float).copy()
    best_c5 = compute_c5(best_h)
    
    for T in [50, 200, 1000, 5000, 20000, 100000]:
        if time.time() > t_end: break
        # Perturb
        h_try = best_h + rng.normal(0, 0.05, n)
        h_try = normalize_h(h_try, S_val)
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            mx = corr.max()
            sm = mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
            pen = 10.0 * (x.sum() - S_val) ** 2
            return sm + pen
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_try, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-16})
            h_try = normalize_h(res.x, S_val)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        except Exception:
            pass
    
    return best_h, best_c5
