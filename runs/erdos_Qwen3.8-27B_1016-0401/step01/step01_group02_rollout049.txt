            hi = mid
        elif slope < -1e-14:
            lo = mid
        else:

    t_opt = (lo + hi) / 2
    return t_opt

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            break
        h[free] -= d / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _coord_opt_j(j, h, n, corr):
    k_idx = np.arange(2 * n - 1) - (n - 1)
    mk_ik = (j + k_idx >= 0) & (j + k_idx < n)
    mk_jk = (j - k_idx >= 0) & (j - k_idx < n)
    b = np.ones(2 * n - 1)
    if mk_ik.any():
        b[mk_ik] -= h[j + k_idx[mk_ik]]
    if mk_jk.any():
        b[mk_jk] -= h[j - k_idx[mk_jk]]
    a = corr - h[j] * b
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = a + mid * b
        m = vals.max()
        active = vals >= m - 1e-10
        if not active.any():
            active = np.zeros(len(vals), dtype=bool)
            active[int(np.argmax(vals))] = True
        bmin = b[active].min()
        bmax = b[active].max()
        if bmin > 1e-12:
            hi = mid
        elif bmax < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def _coord_descent(h, n, max_passes=80, tol=1e-12, seed=0):
    rng = np.random.RandomState(seed)
    h = _proj(h.copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h)
    for p in range(max_passes):
        corr = np.correlate(h, 1.0 - h, mode="full")
        order = rng.permutation(n)
        moved = 0
        for j in order:
            t_opt = _coord_opt_j(j, h, n, corr)
            if abs(t_opt - h[j]) > 1e-11:
                h[j] = t_opt
                moved += 1
                corr = np.correlate(h, 1.0 - h, mode="full")
        h = _proj(h, n)
        c5_now = _c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if moved == 0:
            break
    return best_h, best_c5

def _slsqp_opt(h0, n, dx, sc, maxiter):
    def sfun(x):
        hh = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hh, 1.0 - hh, mode="full")
        m = corr.max()
        return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx

    def ceq(x):
        return x.sum() - 0.5 * n

    def ceqj(x):
        return np.ones(len(x))

    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': ceqj}]
    try:
        r = minimize(sfun, h0, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _local_search(h0, n, tmax, nit, seed=0):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    cw = h0.copy()
    best = h0.copy()
    bestc = _c5(cw)
    c5w = bestc
    for _ in range(nit):
        if time.time() - t0 > tmax:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dv = (rng.random() - 0.5) * rng.uniform(0.05, 0.5)
        h2 = cw.copy()
        h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
        h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        h2 = _proj(h2, n)
        c5t = _c5(h2)
        if c5t < bestc - 1e-14:
            bestc = c5t
            best = h2.copy()
            cw = h2
            c5w = c5t
        elif c5t < c5w + 0.005 and (rng.random() < 0.2 or c5t < c5w):
            cw = h2
            c5w = c5t
        else:
            cw = best.copy()
            c5w = bestc
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5

    n = len(h)
    h = _proj(h, n)
    bh = h.copy()
    bc = _c5(bh)

    rem = budget_s - (time.time() - ts)
    if rem > 20:
        h1, c1 = _coord_descent(bh, n, max_passes=60, tol=1e-12, seed=seed)
        if c1 < bc:
            bh, bc = h1, c1

    rem = budget_s - (time.time() - ts)
    if rem > 40:
        for sc in [50.0, 120.0, 250.0]:
            h2 = _slsqp_opt(bh, n, 2.0 / n, sc, 1500)
            c2 = _c5(h2)
            if c2 < bc:
                bh, bc = h2, c2

    rem = budget_s - (time.time() - ts)
    if rem > 20:
        h3, c3 = _coord_descent(bh, n, max_passes=40, tol=1e-13, seed=seed + 10)
        if c3 < bc:
            bh, bc = h3, c3

    rem = budget_s - (time.time() - ts)
    if rem > 50:
        t4 = min(rem * 0.12, 50.0)
        h4, c4 = _local_search(bh, n, t4, 400000, seed=seed + 20)
        if c4 < bc:
            bh, bc = h4, c4

    # Upscale
    n2 = 2 * n
    rem = budget_s - (time.time() - ts)
    if n2 <= 400 and rem > 100:
        h5 = _proj(np.repeat(bh, 2), n2)
        h5, c5v = _coord_descent(h5, n2, max_passes=40, tol=1e-12, seed=seed + 30)
        if c5v < bc:
            bh, bc, n = h5, c5v, n2

        rem = budget_s - (time.time() - ts)
        if rem > 50:
            for sc in [100.0, 200.0]:
                h6 = _slsqp_opt(bh, n, 2.0 / n, sc, 1500)
                c6 = _c5(h6)
                if c6 < bc:
                    bh, bc = h6, c6

        rem = budget_s - (time.time() - ts)
        if rem > 40:
            h7, c7 = _coord_descent(bh, n, max_passes=30, tol=1e-13, seed=seed + 40)
            if c7 < bc:
                bh, bc = h7, c7

    # Try alternative starting point (symmetric perturbation)
    rem = budget_s - (time.time() - ts)
    if rem > 80:
        rng_alt = np.random.RandomState(seed + 99)
        h_alt = bh.copy()
        pert = rng_alt.normal(0, 0.05, n)
        h_alt = _proj(h_alt + pert, n)
        h_alt, c_alt = _coord_descent(h_alt, n, max_passes=40, tol=1e-12, seed=seed + 50)
        if c_alt < bc:
            bh, bc = h_alt, c_alt
            rem = budget_s - (time.time() - ts)
            if rem > 40:
                h_alt2, c_alt2 = _slsqp_opt(bh, n, 2.0 / n, 150.0, 1000)
                if c_alt2 < bc:
                    bh, bc = h_alt2, c_alt2

    # Final local search
    rem = budget_s - (time.time() - ts)
    if rem > 30:
        t6 = min(rem * 0.45, 150.0)
        h8, c8 = _local_search(bh, n, t6, 500000, seed=seed + 60)
        if c8 < bc:
            bh, bc = h8, c8

    # Final coord descent polish
    rem = budget_s - (time.time() - ts)
    if rem > 20:
        h9, c9 = _coord_descent(bh, n, max_passes=20, tol=1e-13, seed=seed + 70)
        if c9 < bc:
            bh, bc = h9, c9

    bh = _proj(bh, n)
    fc = _c5(bh)
    return (bh, fc, len(bh))
