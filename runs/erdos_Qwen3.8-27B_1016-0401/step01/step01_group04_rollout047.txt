def local_search_best(h, max_iters=200, t_deadline=None):
    n = len(h)
    h_ls = h.copy()
    best_val = float(compute_corr(h_ls).max())
    for it in range(max_iters):
        if t_deadline and time.time() > t_deadline:
            break
        best_swap_val = best_val
        best_i, best_j = -1, -1
        ones = np.where(h_ls == 1)[0]
        zeros = np.where(h_ls == 0)[0]
        for i in ones:
            for j in zeros:
                h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
                val = float(compute_corr(h_ls).max())
                h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
                if val < best_swap_val - 1e-15:
                    best_swap_val = val
                    best_i, best_j = i, j
        if best_i >= 0:
            h_ls[best_i], h_ls[best_j] = h_ls[best_j], h_ls[best_i]
            best_val = best_swap_val
        else:
            break
    return h_ls, best_val * 2.0 / n

def sa_binary(h, rng, n_iter, t_deadline, t0=0.08, t1=0.002):
    n = len(h)
    h_sa = h.copy()
    cur_val = float(compute_corr(h_sa).max())
    best_h = h_sa.copy()
    best_val = cur_val
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        T = t0 + (t1 - t0) * it / max(n_iter, 1)
        i, j = rng.randint(n, size=2)
        if h_sa[i] == h_sa[j]:
            continue
        h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        new_val = float(compute_corr(h_sa).max())
        delta = new_val - cur_val
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_sa.copy()
        else:
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
    return best_h, best_val * 2.0 / n

def smooth_max(corr, T):
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def slsqp_refine(h, n, S, T, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max(compute_corr(x), T)
    def obj_grad(x):
        corr = compute_corr(x)
        mx = corr.max()
        w = T * np.exp(T * (corr - mx)) / np.sum(np.exp(T * (corr - mx)))
        g = np.zeros(n)
        for k in range(len(corr)):
            s = k - (n - 1)
            for m in range(n):
                if 0 <= m + s < n:
                    g[m] += w[k] * (1.0 - x[m + s])
                if 0 <= m - s < n:
                    g[m] += w[k] * x[m - s]
        return g
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - S, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=con, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_parent = random_binary(rng, 70)

    n_parent = len(h_parent)
    S_parent = n_parent / 2.0
    if abs(h_parent.sum() - S_parent) < 0.01:
        h_bin_parent = make_binary(h_parent, n_parent)
    else:
        h_bin_parent = random_binary(rng, n_parent)

    best_h = h_bin_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Local search on parent
    t1_end = t_start + budget_s * 0.12
    h_ls, c5_ls = local_search_best(h_bin_parent, max_iters=100, t_deadline=t1_end)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls.copy(), c5_ls

    # Phase 2: Multi-start with different n
    n_candidates = list(set([n_parent, 64, 72, 80, 76]))
    n_candidates.sort(key=lambda x: abs(x - n_parent))
    n_candidates = [n for n in n_candidates if n % 2 == 0 and 32 <= n <= 100]

    t_phase2_end = t_start + budget_s * 0.65
    for n in n_candidates:
        if time.time() > t_phase2_end:
            break
        S = n // 2
        n_restarts = min(6, max(2, int((t_phase2_end - time.time()) / 12)))
        for r in range(n_restarts):
            if time.time() > t_phase2_end - 3:
                break
            if n == n_parent and r == 0:
                h_init = best_h.copy() if len(best_h) == n else random_binary(rng, n)
            else:
                h_init = random_binary(rng, n)
            t_ls = min(time.time() + 12, t_phase2_end - 8)
            h_ls2, c5_vals = local_search_best(h_init, max_iters=60, t_deadline=t_ls)
            if c5_vals < best_c5:
                best_h, best_c5 = h_ls2.copy(), c5_vals
            t_sa = min(time.time() + 8, t_phase2_end - 4)
            h_sa, c5_sa = sa_binary(h_ls2, rng, 25000, t_sa)
            if c5_sa < best_c5:
                best_h, best_c5 = h_sa.copy(), c5_sa
            t_ls3 = min(time.time() + 8, t_phase2_end - 1)
            h_ls3, c5_ls3 = local_search_best(h_sa, max_iters=50, t_deadline=t_ls3)
            if c5_ls3 < best_c5:
                best_h, best_c5 = h_ls3.copy(), c5_ls3

    # Phase 3: SA from best with longer run
    t_phase3_end = t_start + budget_s * 0.80
    if time.time() < t_phase3_end:
        n = len(best_h)
        h_sa2, c5_sa2 = sa_binary(best_h, rng, 50000, t_phase3_end, t0=0.06, t1=0.003)
        if c5_sa2 < best_c5:
            best_h, best_c5 = h_sa2.copy(), c5_sa2
        t_ls4 = min(time.time() + 10, t_start + budget_s * 0.85)
        h_ls4, c5_ls4 = local_search_best(best_h, max_iters=80, t_deadline=t_ls4)
        if c5_ls4 < best_c5:
            best_h, best_c5 = h_ls4.copy(), c5_ls4

    # Phase 4: Continuous refinement
    t_phase4_end = t_start + budget_s * 0.93
    if time.time() < t_phase4_end:
        n = len(best_h)
        S = n / 2.0
        h_cont = best_h.astype(float) + rng.normal(0, 0.2, n)
        h_cont = np.clip(h_cont, 0, 1)
        if h_cont.sum() > 1e-6:
            h_cont = h_cont / h_cont.sum() * S
            h_cont = np.clip(h_cont, 0, 1)
        for T in [30, 150, 800]:
            if time.time() > t_phase4_end:
                break
            h_new = slsqp_refine(h_cont, n, S, T, maxiter=150)
            if abs(h_new.sum() - S) > 0.05:
                h_new = make_binary(h_new, n).astype(float)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
            h_cont = h_new

    # Phase 5: Final binary polish
    t_final = deadline - 2
    if time.time() < t_final:
        n = len(best_h)
        h_bin = make_binary(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin.copy(), c5_bin
        h_pol, c5_pol = local_search_best(best_h, max_iters=60, t_deadline=t_final)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol.copy(), c5_pol

    # Ensure validity
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    S = n / 2.0
    if abs(best_h.sum() - S) > 0.01:
        best_h = make_binary(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
