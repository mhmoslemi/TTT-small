def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _soft_obj(h, dx, alpha):
    c = _corr(h)
    m = c.max()
    return (m + np.log(np.exp(alpha * (c - m)).sum()) / alpha) * dx

def _soft_grad(h, dx, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        kk = k - (n - 1)
        # term 1: (1-h[j-kk]) for valid j
        jlo = max(0, kk)
        jhi = min(n - 1, kk + n - 1)
        if jlo <= jhi:
            g[jlo:jhi+1] += wk * (1.0 - h[jlo-kk : jhi-kk+1])
        # term 2: -h[j+kk] for valid j
        jlo2 = max(0, -kk)
        jhi2 = min(n - 1, n - 1 - kk)
        if jlo2 <= jhi2:
            g[jlo2:jhi2+1] -= wk * h[jlo2+kk : jhi2+kk+1]
    return g * dx

def _proj(h, n):

        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h
