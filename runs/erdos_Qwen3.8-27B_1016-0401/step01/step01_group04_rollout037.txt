def normalize_sum(h, S):
    """Ensure sum h = S, clip to [0,1]."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 1e-10:
        h = h * (S / s)
    h = np.clip(h, 0, 1)
    # Fix sum exactly
    diff = S - h.sum()
    if abs(diff) > 0:

        idx = np.where((h > 0.01) & (h < 0.99))[0]

            h = np.clip(h, 0, 1)

    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()

    shifted -= shifted.max()

                g -= h[md]
            md2 = m - d
            if 0 <= md2 < n:
                g -= h[md2]
            grad[m] += pk * g
    return grad

import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = np.log(np.sum(np.exp(T * (corr - mx))))
    return mx + s / T

def proj_sum(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(len(h), S / len(h))
    h = h * (S / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = S - s
        if abs(diff) < 1e-13:
            break
        free = (h > 0.005) & (h < 0.995)
        if free.sum() == 0:
            free = (h > 1e-4) & (h < 1 - 1e-4)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, S):
    hb = (h >= 0.5).astype(float)
    ones = int(hb.sum())
    if ones > S:
        idx = np.where(hb == 1)[0]
        order = np.argsort(h[idx][idx == 1 - 1]) if False else np.argsort(h[idx])
        for i in range(ones - S):
            hb[idx[order[i]]] = 0
    elif ones < S:
        idx = np.where(hb == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(S - ones):
            hb[idx[order[i]]] = 1
    return hb

def slsqp_opt(h_init, S, T, maxiter=500):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    def obj(x):
        return smooth_max_val(x, T)
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h_init

def greedy_ls(h, max_sweeps=80, t_end=None):
    h = h.copy().astype(float)
    n = len(h)
    for _ in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        cm = np.correlate(h, 1 - h, mode='full').max()
        bv, bp = cm, None
        for i in oi:
            for j in zi:
                h[i], h[j] = 0.0, 1.0
                v = np.correlate(h, 1 - h, mode='full').max()
                h[i], h[j] = 1.0, 0.0
                if v < bv:
                    bv, bp = v, (i, j)
        if bp is None:
            break
        h[bp[0]], h[bp[1]] = 0.0, 1.0
    return h, compute_c5(h)

def sa_binary(h, rng, t_end, t0_val=0.06, cool=0.99997):
    h = h.copy().astype(float)
    n = len(h)
    cm = np.correlate(h, 1 - h, mode='full').max()
    bh, bv = h.copy(), cm
    temp = t0_val

        best_c5, best_h, best_n = c5s, hsa.copy(), n
    # Polish SA result
    hsa2, c5s2 = greedy_ls(hsa, max_sweeps=30, t_end=t3)
    if c5s2 < best_c5:
        best_c5, best_h, best_n = c5s2, hsa2.copy(), n

    # Phase 4: Try n=100 with continuous optimization
    t4 = t0 + budget_s * 0.86
    if time.time() < t4:
        n4, S4 = 100, 50
        for r in range(2):
            if time.time() >= t4:
                break
            h4b = np.zeros(n4)
            h4b[rng.choice(n4, S4, replace=False)] = 1.0
            h4 = proj_sum(h4b + 0.25 * rng.randn(n4), S4)
            for T in [20, 100, 500]:
                if time.time() >= t4:
                    break
                h4n = slsqp_opt(h4, S4, T, maxiter=400)
                h4n = proj_sum(h4n, S4)
                c54 = compute_c5(h4n)
                if c54 < best_c5:
                    best_c5, best_h, best_n = c54, h4n.copy(), n4
                h4 = h4n

    # Phase 5: Final binary SA + polish on parent n
    t5 = t0 + budget_s * 0.94
    if time.time() < t5:
        h5 = best_h.copy().if np.issubdtype(best_h.dtype, np.floating) else h_bin.copy()
        h5b = make_binary(h5, S)
        hsa5, c5_5 = sa_binary(h5b, rng, t5, t0_val=0.04, cool=0.99999)
        if c5_5 < best_c5:
            best_c5, best_h, best_n = c5_5, hsa5.copy(), n
        hsa5, c5_5 = greedy_ls(hsa5, max_sweeps=20, t_end=t5)
        if c5_5 < best_c5:
            best_c5, best_h, best_n = c5_5, hsa5.copy(), n

    # Final: one more SLSQP polish on best continuous
    tf = t0 + budget_s * 0.97
    if best_h is not None and np.any((best_h > 0.01) & (best_h < 0.99)):
        if time.time() < tf:
            hfin = proj_sum(best_h + 0.02 * rng.randn(len(best_h)), len(best_h) // 2)
            for T in [500, 2000]:
                if time.time() >= tf:
                    break
                hf2 = slsqp_opt(hfin, len(best_h) // 2, T, maxiter=300)
                hf2 = proj_sum(hf2, len(best_h) // 2)
                c5f = compute_c5(hf2)
                if c5f < best_c5:
                    best_c5, best_h = c5f, hf2.copy()
                hfin = hf2

    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
