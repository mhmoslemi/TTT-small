    """Find best swap among n_trials random pairs"""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, compute_c5(h)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for _ in range(n_trials):
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        if p == q:
            continue
        h[p], h[q] = h[q], h[p]
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[p], h[q] = h[q], h[p]
    
    return best_h, best_c5

Let me think about how to vectorize the swap evaluation:

For a fixed set of "1" positions P and "0" positions Z, I want to evaluate all |P|×|Z| swaps. For each swap (p,q), the new correlation is corr + A[p] + B[q] (as derived above). The new C5 is max(corr + A[p] + B[q]) * dx.

I can compute this for all pairs at once using matrix operations!

Let's define:
- corr: shape (2n-1,)
- A: shape (n, 2n-1), where A[p] is the delta vector for flipping position p from 1 to 0
- B: shape (n, 2n-1), where B[q] is the delta vector for flipping position q from 0 to 1

For swap (p,q): new_corr = corr + A[p] + B[q], new_C5 = max(new_corr) * dx

I can compute this as:

That's a |P|×|Z|×(2n-1) tensor = 35×35×139 ≈ 170K values. The max over k gives a 35×35 matrix of C5 values. This is very fast in numpy!

```python
def compute_all_swap_c5s(h, n):
    """For binary h, compute C5 for all possible single swaps"""
    corr = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # A[p][k] = delta for flipping h[p] from 1 to 0
    # A[p][k] = -I[n-1-p <= k <= 2n-2-p]*(1-h[p+k-n+1]) + I[p <= k <= 2n-2]*h[p-k+n-1]
    # B[q][k] = +I[n-1-q <= k <= 2n-2-q]*(1-h[q+k-n+1]) - I[q <= k <= 2n-2]*h[q-k+n-1]
    
    # Precompute for all positions
    A = np.zeros((n, 2*n-1))
    B = np.zeros((n, 2*n-1))
    
    hh = h  # binary
    for i in range(n):
        # Term 1: -I[n-1-i <= k <= 2n-2-i]*(1-h[i+k-n+1])
        k_lo1 = max(0, n-1-i)
        k_hi1 = min(2*n-2, 2*n-2-i)
        if k_lo1 <= k_hi1:
            k_range = np.arange(k_lo1, k_hi1+1)
            idx = i + k_range - n + 1  # partner indices
            A[i, k_range] = -(1 - hh[idx])
        
        # Term 2: +I[i <= k <= 2n-2]*h[i-k+n-1]
        k_lo2 = max(0, i)
        k_hi2 = min(2*n-2, 2*n-2)  # i <= k
        if k_lo2 <= k_hi2:
            k_range2 = np.arange(k_lo2, k_hi2+1)
            idx2 = i - k_range2 + n - 1
            # Need 0 <= idx2 < n, i.e., 0 <= i-k+n-1 < n
            # i-k+n-1 >= 0 -> k <= i+n-1 (always true for k <= 2n-2, i <= n-1)
            # i-k+n-1 <= n-1 -> i-k <= 0 -> k >= i (already ensured)
            A[i, k_range2] += hh[idx2]
        
        # B[i][k] = +I[n-1-i <= k <= 2n-2-i]*(1-h[i+k-n+1]) - I[i <= k]*h[i-k+n-1]
        if k_lo1 <= k_hi1:
            B[i, k_range] = (1 - hh[idx])
        if k_lo2 <= k_hi2:
            B[i, k_range2] -= hh[idx2]
    
    # For all pairs (p in ones, q in zeros):
    # new_C5[p,q] = max(corr + A[p] + B[q]) * dx
    # Use broadcasting
    dx = 2.0 / n
    
    # A[ones] has shape (len(ones), 2n-1)
    # B[zeros] has shape (len(zeros), 2n-1)
    # Need max over k of corr[k] + A[p,k] + B[q,k]
    
    # This is: for each p, q: max_k(corr[k] + A[ones[p],k] + B[zeros[q],k])
    # = max_k(corr[k]) + cross terms... no, that doesn't factor.
    
    # Use broadcasting: corr[None,None,:] + A[ones][:,None,:] + B[zeros][None,:,:]
    # Shape: (len(ones), len(zeros), 2n-1)
    # That's 35 * 35 * 139 = 170K floats. Fine!
    
    c5_matrix = np.max(corr[np.newaxis, np.newaxis, :] + A[ones][:, np.newaxis, :] + B[zeros][np.newaxis, :, :], axis=2) * dx
    
    return c5_matrix, corr, new_corr not needed
```
