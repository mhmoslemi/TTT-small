def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _sobj(h, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    dx = 2.0 / len(h)
    return float((m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx)

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _mk(n):
    ii = np.arange(n)[:, None]
    jj = np.arange(n)[None, :]
    kv = ii - jj + n - 1
    vd = (kv >= 0) & (kv < 2 * n - 1)
    return kv, vd

def _gr(h, sc, kv, vd):
    omh = 1.0 - h
    corr = np.correlate(h, omh, "full")
    w = np.exp(sc * (corr - corr.max()))
    w /= w.sum()
    W = np.where(vd, w[kv], 0.0)
    return (2.0 / len(h)) * (W @ omh - W.T @ h)

        improved = False

            hn = _proj(h - alpha * g, n)
            fn = _sobj(hn, sc)

                h = hn
                improved = True
                break
            alpha *= 0.5
        if not improved:
            alpha = alpha * 0.1
            h = _proj(h - alpha * g, n)
        ct = _c5(h)
        if ct < bc:
            best = h.copy()
            bc = ct
    return best, bc
