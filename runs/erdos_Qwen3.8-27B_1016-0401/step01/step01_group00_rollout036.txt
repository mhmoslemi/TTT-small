    # Find h in [0,1]^n with sum(h) = S, minimizing ||h - v||^2
    # Solution: h_i = clip(v_i - tau, 0, 1) for some tau

    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / n
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] -= w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
    return val, grad * dx

def project_box_sum(v, S):
    lo = v.min() - S
    hi = v.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def adam_soft(h, T, iters, lr, decay, dl):
    n = len(h)
    S = 0.5 * n
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    vv = np.zeros(n)
    bh = h.copy()
    bc = compute_c5(h)
    for it in range(iters):
        if time.time() > dl:
            break
        _, g = soft_max_obj_grad(h, T)
        m = b1 * m + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g

        vh = vv / (1 - b2 ** (it + 1))

        c = compute_c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
    return bh, bc
