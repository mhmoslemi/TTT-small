def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_obj(h, n, sc):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx

def _smooth_grad(h, n, sc):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(sc * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wm = w[n - 1 + lag]
        if abs(wm) < 1e-15:
            continue
        if lag >= 0:
            grad[:n-lag] += wm * (1.0 - h[lag:])
            grad[lag:] -= wm * h[:n-lag]
        else:
            mm = -lag
            grad[mm:] += wm * (1.0 - h[:n-mm])
            grad[:n-mm] -= wm * h[mm:]
    grad *= dx
    return grad

def _project(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0, 1)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    p = (lo + hi) / 2
    h = np.clip(h - p, 0, 1)
    r = tgt - h.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += r / len(idx)
            np.clip(h, 0, 1, out=h)
    return h

def _lbfgs(h, n, sc, maxiter, tdeadline):
    """L-BFGS-B with penalty for sum constraint"""
    def obj(x):
        hx = np.clip(x, 0, 1)
        return _smooth_obj(hx, n, sc) + 100.0 * (hx.sum() - 0.5*n)**2
    
    def grad(x):
        hx = np.clip(x, 0, 1)
        g = _smooth_grad(hx, n, sc)
        g += 200.0 * (hx.sum() - 0.5*n)
        g[x < 0] = 0
        g[x > 1] = 0
        return g
    
    bds = [(0, 1)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _project(res.x, n)
    except:
        return h

def _proj_grad(h, n, sc, maxiter, tdeadline):
    """Projected gradient with fixed step / backtracking"""
    dx = 2.0 / n
    alpha = dx / sc  # initial step size
    fcur = _smooth_obj(h, n, sc)
    for it in range(maxiter):
        if time.time() > tdeadline:
            break
        g = _smooth_grad(h, n, sc)
        # Ensure descent direction after projection
        cg = np.dot(g, h - _project(h, n))
        if cg < -1e-12:
            h = _project(h, n)
        # Backtracking
        for _b in range(50):
            h_new = _project(h - alpha * g, n)
            f_new = _smooth_obj(h_new, n, sc)
            if f_new <= fcur - 1e-4 * alpha * np.dot(g, h - h_new):
                h = h_new
                fcur = f_new
                break
            alpha *= 0.5
        else:
            break
    return h

def _local_search(h, n, tdeadline, n_iter):
    """Random 2-point local search"""
    t0 = time.time()
    tgt = 0.5 * n
    best = h.copy()
    bestc = _c5(best)
    cur = h.copy()
    curc = bestc
    rng = np.random.default_rng(999)
    for _ in range(n_iter):
        if time.time() - t0 > (tdeadline - time.time() + t0) * 0.95:
            break
        if time.time() > tdeadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        dv = (2*rng.random() - 1) * rng.uniform(0.02, 0.25)
        h2 = cur.copy()
        h2[i] = np.clip(h2[i] + dv, 0, 1)
        h2[j] = np.clip(h2[j] - dv, 0, 1)
        # Fix sum
        d = h2.sum() - tgt
        if abs(d) > 1e-14:
            h2 -= d / n
            np.clip(h2, 0, 1, out=h2)
        c = _c5(h2)
        if c < bestc:
            best, bestc = h2, c
            cur, curc = h2, c
        elif c < curc + 0.01 and rng.random() < 0.2:
            cur, curc = h2, c
        else:
            cur = best.copy()
            curc = bestc
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()
    
    def _timeup(frac=0.0):
        return ts + budget_s * (1.0 - frac)

        td = min(ts + budget_s * 0.25, _timeup(0.05))
        h1 = _lbfgs(bh, bn, sc, 500, td)
        c1 = _c5(h1)
        if c1 < bc:
            bh, bc = h1, c1
    
    # Phase 2: Projected gradient
    td = ts + budget_s * 0.35
    for sc in [50.0, 100.0, 200.0]:
        h2 = _proj_grad(bh, bn, sc, 2000, td)
        c2 = _c5(h2)
        if c2 < bc:
            bh, bc = h2, c2
    
    # Phase 3: Local search
    td = ts + budget_s * 0.50
    hl, cl = _local_search(bh, bn, td, 200000)
    if cl < bc:
        bh, bc = hl, cl
    
    # Phase 4: Upscale to 2*bn
    n2 = 2 * bn
    # Cubic interpolation for upscaling
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 2, bn, endpoint=False) + 1.0/bn
    x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0/n2
    f_interp = interp1d(x_old, bh, kind='cubic', fill_value='extrapolate')
    h2 = np.clip(f_interp(x_new), 0, 1)
    h2 = _project(h2, n2)
    
    td = ts + budget_s * 0.70
    for sc in [30.0, 80.0, 150.0]:
        h3 = _lbfgs(h2, n2, sc, 400, td)
        c3 = _c5(h3)
        if c3 < _c5(h2):
            h2 = h3
    
    td = ts + budget_s * 0.80
    h3 = _proj_grad(h2, n2, 100, 1500, td)
    c3 = _c5(h3)
    if c3 < bc:
        bh, bc, bn = h3, c3, n2
    
    # Phase 5: Local search at higher res
    td = ts + budget_s * 0.90
    hl, cl = _local_search(bh, bn, td, 300000)
    if cl < bc:
        bh, bc = hl, cl
    
    # Phase 6: Final L-BFGS-B polish
    rem = budget_s - (time.time() - ts)
    if rem > 30:
        td = ts + budget_s * 0.97
        h4 = _lbfgs(bh, bn, 200, 300, td)
        c4 = _c5(h4)
        if c4 < bc:
            bh, bc = h4, c4
    
    # Final check
    fc = _c5(bh)
    if fc < bc:
        bc = fc
    return (bh, fc, len(bh))

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_val(h, n, sc):

    corr = np.correlate(h, 1.0 - h, mode="full")

    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx

def _smooth_grad(h, n, sc, thresh=1e-8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(sc * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    lo = max(0, -(n - 1))
    for lag in range(lo, n):
        idx = n - 1 + lag
        wm = w[idx]
        if wm < thresh:
            continue
        if lag >= 0:
            grad[:n - lag] += wm * (1.0 - h[lag:])
            if n - lag > 0:
                grad[lag:] -= wm * h[:n - lag]

            grad[mm:] += wm * (1.0 - h[:n - mm])
            if n - mm > 0:
