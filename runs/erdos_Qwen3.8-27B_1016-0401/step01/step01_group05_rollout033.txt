def slsqp_refine(h, n, target, alpha, maxiter=200):

        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

        return np.sum(x) - target

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})

    except:
        return h.copy()

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = np.clip(alpha * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

def _sum_constraint(x, target):
    return np.sum(x) - target

def slsqp_refine(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP', args=(alpha,),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def optimize_coord(h, n, target, i, top_k):
    """Optimize a single coordinate to minimize max of top-k lags."""
    c = compute_all_corr(h)
    top_idx = np.argsort(c)[::-1][:top_k]
    h_old = h[i]

    intercepts = np.zeros(top_k)

        d = k - (n - 1)
        g = 0.0
        if 0 <= i + d <= n - 1:
            g += (1.0 - h[i + d])
        if 0 <= i - d <= n - 1:
            g -= h[i - d]
        slopes[idx] = g
        intercepts[idx] = c[k] - h_old * g
    
    # Also consider lags rank top_k+1 to top_k*2 for safety
    rest_count = min(top_k, 2*n-1-top_k)
    rest_idx = np.argsort(c)[top_k:top_k+rest_count]
    
    candidates = [0.0, 1.0]
    for a in range(top_k):
        for b in range(a+1, top_k):
            ds = slopes[a] - slopes[b]
            if abs(ds) < 1e-15:

            t = (intercepts[b] - intercepts[a]) / ds
            if -0.01 < t < 1.01:
                candidates.append(t)
    
    best_t = h_old
    best_val = np.inf
    for t in candidates:
        t = max(0.0, min(1.0, t))
        val = np.max(intercepts + t * slopes)
        if rest_count > 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = np.clip(alpha * (corr - m), -60, 60)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

def _sum_constraint(x, target):
    return np.sum(x) - target

def slsqp_refine(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP', args=(alpha,),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def optimize_coord(h, n, target, i, top_k):
    c = compute_all_corr(h)
    top_idx = np.argsort(c)[::-1][:top_k]
    h_old = h[i]
    slopes = np.zeros(top_k)
    intercepts = np.zeros(top_k)
    for idx, k in enumerate(top_idx):
        d = k - (n - 1)
        g = 0.0
        if 0 <= i + d <= n - 1:
            g += (1.0 - h[i + d])
        if 0 <= i - d <= n - 1:
            g -= h[i - d]
        slopes[idx] = g
        intercepts[idx] = c[k] - h_old * g
    rest_count = min(top_k, 2 * n - 1 - top_k)
    rest_idx = np.argsort(c)[top_k:top_k + rest_count]
    candidates = [0.0, 1.0, h_old]
    for a in range(top_k):
        for b in range(a + 1, top_k):
            ds = slopes[a] - slopes[b]
            if abs(ds) > 1e-15:
                t = (intercepts[b] - intercepts[a]) / ds
                if -0.01 < t < 1.01:
                    candidates.append(t)
    best_t = h_old
    best_val = np.inf
    for t in candidates:
        t = max(0.0, min(1.0, t))
        val = np.max(intercepts + t * slopes)
        if rest_count > 0:
            for idx, k in enumerate(rest_idx):
                d = k - (n - 1)
                g = 0.0
                if 0 <= i + d <= n - 1:
                    g += (1.0 - h[i + d])
                if 0 <= i - d <= n - 1:
                    g -= h[i - d]
                rv = c[k] - h_old * g + t * g
                if rv > val:
                    val = rv
        if val < best_val:
            best_val = val
            best_t = t
    h[i] = best_t
    h = project_feasible(h, n, target)
    return h, compute_c5(h)

def coord_descent(h, n, target, rng, n_rounds, deadline, top_k):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    order = np.arange(n)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        rng.shuffle(order)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            h_new, c5_new = optimize_coord(best_h.copy(), n, target, i, top_k)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    return best_h, best_c5

def compute_subgradient_vec(h, n, top_idx):
    sg = np.zeros(n)
    for k in top_idx:
        d = k - (n - 1)
        if d == 0:
            sg += 1.0 - 2.0 * h
        elif d > 0:
            sg[:n - d] += (1.0 - h[d:])
            sg[d:] -= h[:n - d]
        else:
            d = -d
            sg[d:] += (1.0 - h[:n - d])
            sg[:n - d] -= h[d:]
    return sg / len(top_idx)

def sa_optimize(h, n, target, rng, max_iters, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = h.copy()
    curr_c5 = best_c5
    temp = 0.004
    lr = 0.006
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 2000 == 0 and it > 0:
            temp *= 0.87
            lr *= 0.94
        c = compute_all_corr(h_curr)
        top_idx = np.argsort(c)[::-1][:5]
        sg = compute_subgradient_vec(h_curr, n, top_idx)
        h_try = h_curr - lr * sg + rng.normal(0, temp, n)
        h_try = project_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        delta = c5_try - curr_c5
        if delta < 0 or (temp > 1e-10 and rng.random() < np.exp(-delta / temp)):
            h_curr = h_try
            curr_c5 = c5_try
            if c5_try < best_c5:
                best_h = h_curr.copy()
                best_c5 = c5_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:

            best_c5 = c5_new

    if time.time() < deadline - 150:

        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 3: Simulated annealing with subgradient
    if time.time() < deadline - 100:
        h_sa, c5_sa = sa_optimize(best_h, n, target, rng, 50000, deadline - 100)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa

    # Phase 4: More coordinate descent with wider top-k
    if time.time() < deadline - 60:
        h_cd, c5_cd = coord_descent(best_h, n, target, rng, 60, deadline - 60, top_k=16)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 5: Final SLSQP polish

        for alpha in [5000, 20000]:
            if time.time() > deadline - 15:

            h_new = slsqp_refine(best_h, n, target, alpha, maxiter=300)
            c5_new = compute_c5(h_new)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
