def _block_opt(h, n, m=3, iters=10000):
    # Coordinate descent on blocks of m consecutive variables
    tgt = 0.5 * n
    best = h.copy()
    bestc = _c5(best)
    for it in range(iters):
        start = np.random.randint(0, n - m + 1)
        # Extract block
        block = best[start:start+m].copy()
        # The rest is fixed
        # We want to minimize c5 by adjusting the block
        # Subject to sum(block) = sum(best[start:start+m]) (to maintain total sum)
        block_sum = block.sum()
        # Try a few perturbations
        for _ in range(5):
            # Random perturbation
            pert = np.random.randn(m) * 0.1
            new_block = np.clip(block + pert, 0, 1)
            # Adjust to maintain sum
            adj = block_sum - new_block.sum()
            new_block += adj / m
            new_block = np.clip(new_block, 0, 1)
            # Re-sum adjust
            new_h = best.copy()
            new_h[start:start+m] = new_block
            # Fix sum
            ds = new_h.sum() - tgt
            if abs(ds) > 1e-12:
                # Adjust a random non-block element
                other = np.random.choice([i for i in range(n) if i < start or i >= start+m])
                new_h[other] -= ds
                new_h[other] = np.clip(new_h[other], 0, 1)
            c = _c5(new_h)
            if c < bestc:
                best = new_h
                bestc = c
                block = best[start:start+m].copy()
    return best, bestc

import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GDX = 0.0
_GSC = 50.0

def _c5(h):
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / len(h)))

def _sum_eq(x):
    return x.sum() - 0.5 * _GN

def _sum_eq_jac(x):
    return np.ones(_GN)

def _soft_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = _GSC * (corr - m)
    return (m + np.log(np.exp(s).sum()) / _GSC) * _GDX

def _proj_sum(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _do_slsqp(h0, scale, maxit):
    global _GN, _GDX, _GSC
    n = len(h0)
    _GN, _GDX, _GSC = n, 2.0 / n, scale
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        r = minimize(_soft_obj, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-12})
        return _proj_sum(r.x, n)
    except Exception:
        return _proj_sum(h0, n)

def _sweep_improve(h, n, tmax, delta_list):
    t0 = time.time()
    tgt = 0.5 * n
    best = h.copy()
    bc = _c5(best)
    onsg = np.ones(n)
    for it in range(2000000):
        if time.time() - t0 > tmax:
            break
        for j in range(n):
            ov = best[j]
            for dv in delta_list:
                nv = ov + dv
                if nv < 0.0 or nv > 1.0:
                    continue
                h2 = best.copy()
                h2[j] = nv
                ds = tgt - h2.sum()
                if abs(ds) > 1e-12:
                    oth = np.arange(n)
                    oth = oth[oth != j]
                    if ds > 0:
                        k = oth[np.argmin(h2[oth])]
                    else:
                        k = oth[np.argmax(h2[oth])]
                    h2[k] += ds
                    if h2[k] < 0.0 or h2[k] > 1.0:
                        continue
                c = _c5(h2)
                if c < bc - 1e-14:
                    best = h2
                    bc = c
                    ov = nv
    return best, bc

def _pair_swap(h, n, tmax, niters):
    t0 = time.time()
    tgt = 0.5 * n
    best = h.copy()
    bc = _c5(best)
    rng = np.random
    for it in range(niters):
        if time.time() - t0 > tmax:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sgn = 1.0 if rng.random() < 0.5 else -1.0
        dv = sgn * rng.uniform(0.02, 0.25)
        h2 = best.copy()
        h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
        h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        ds = tgt - h2.sum()
        if abs(ds) > 1e-12:
            h2 -= ds / n
            np.clip(h2, 0.0, 1.0, out=h2)
        c = _c5(h2)
        if c < bc - 1e-14:
            best = h2
            bc = c
    return best, bc

def _block_refine(h, n, tmax, step):
    t0 = time.time()
    tgt = 0.5 * n
    best = h.copy()
    bc = _c5(best)
    rng = np.random
    for it in range(500000):
        if time.time() - t0 > tmax:
            break
        s = rng.randint(0, n - step + 1)
        blk = best[s:s+step].copy()
        blks = blk.sum()
        for _ in range(4):
            pert = rng.randn(step) * 0.06
            nb = np.clip(blk + pert, 0.0, 1.0)
            nb += (blks - nb.sum()) / step
            nb = np.clip(nb, 0.0, 1.0)
            h2 = best.copy()
            h2[s:s+step] = nb
            ds = tgt - h2.sum()
            if abs(ds) > 1e-12:
                oth = np.arange(n)
                oth = oth[(oth < s) | (oth >= s + step)]
                oth = oth[oth != -1]
                if len(oth) == 0:
                    continue
                oth = oth[rng.randint(len(oth))]
                h2[oth] += ds
                if h2[oth] < 0.0 or h2[oth] > 1.0:
                    continue
            c = _c5(h2)
            if c < bc - 1e-14:
                best = h2
                bc = c
                blk = best[s:s+step].copy()
    return best, bc
