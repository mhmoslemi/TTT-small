            h, cur_c = h2, c
        elif np.random.random() < 0.1 and c < cur_c + 0.02:

        else:

            cur_c = best_c

    """Push values towards 0/1 while maintaining constraint"""
    tgt = 0.5 * n
    h = h.copy()
    for _ in range(iters):

        for i in range(n):

                h2 = h.copy()
                h2[i] = min(1.0, h[i] + 0.01)
                h2 += (tgt - h2.sum()) / n

                h2[i] = max(0.0, h[i] - 0.01)

                np.clip(h2, 0.0, 1.0, out=h2)
            if _c5(h2) < _c5(h):
                h = h2
    return _project(h, n)

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_obj(x, scale, n, dx):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(scale * (corr - m)).sum()) / scale) * dx

def _smooth_grad(x, scale, n, dx):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(scale * (corr - m))
    w /= w.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for ki in active:
        wv = w[ki]
        k = int(ki) - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            g[js:je] += wv * (1.0 - h[js-k:je-k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            g[js2:je2] -= wv * h[js2+k:je2+k]
    return g * dx

def _project(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)

        d = h.sum() - tgt
        if abs(d) < 1e-15:

        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
