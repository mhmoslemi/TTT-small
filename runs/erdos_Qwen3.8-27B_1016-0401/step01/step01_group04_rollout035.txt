def c5_after_swap(corr, h, i, j, n):
    """Compute new max correlation after swapping h[i]:1→0, h[j]:0→1.
    corr is the current correlation vector (length 2n-1), indexed so that
    corr[m] corresponds to shift s = m - (n-1).
    """
    # For shift s = m - (n-1):
    # Δc[s] = (h[i-s] - h[i+s]) + (h[j+s] - h[j-s])
    # where out-of-range indices contribute 0.
    
    new_corr = corr.copy()
    for m in range(2 * n - 1):
        s = m - (n - 1)
        dc = 0.0
        # Term from i (1→0)
        if 0 <= i - s < n:
            dc += h[i - s]
        if 0 <= i + s < n:
            dc -= h[i + s]
        # Term from j (0→1)
        if 0 <= j + s < n:
            dc += h[j + s]
        if 0 <= j - s < n:
            dc -= h[j - s]
        new_corr[m] += dc
    
    return float(new_corr.max())
