_G_N = 0
_G_DX = 0.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_max(corr, sc):
    m = corr.max()
    if m < 0:
        return 0.0
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return _smooth_max(corr, sc) * dx

    return _sopt_obj(np.clip(x, 0.0, 1.0), _G_DX, _G_SCALE)

def _ceq(x):
    return x.sum() - 0.5 * _G_N

def _ceqj(x):
    return np.ones(len(x))

def _proj(h, n):
    tgt = 0.5 * n

            np.clip(h, 0.0, 1.0, out=h)

        # Convert to actual shift
        # In full correlate of length-n arrays, index k_star corresponds to shift (k_star - (n-1))
        shift = k_star - (n - 1)
        # Find indices where cw[i] is high and cw[i+shift] is low (or vice versa)
        # Try to reduce the product at the bottleneck
        h2 = cw.copy()
        # Small perturbation biased toward reducing the bottleneck

                if j >= n:
                    j = np.random.randint(n)
            else:
                j = i + shift
                if j < 0:
                    j = np.random.randint(n)
            # Move h[i] toward 1-h[j] (to reduce h[i](1-h[j]))
            target = 1.0 - h2[j]
            step = np.clip(0.1 * (target - h2[i]), -0.1, 0.1)
            h2[i] = np.clip(h2[i] + step, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - step, 0.0, 1.0)  # Compensate
        h2 = _proj(h2, n)
        c5t = _c5(h2)
        if c5t < bestc - 1e-14:

            cw, c5w = h2, c5t
        elif np.random.random() < 0.1 and c5t < c5w + 0.005:
            cw, c5w = h2, c5t
        else:

            c5w = bestc
    return best, bestc

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_DX = 0.0
_G_SCALE = 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_max(corr, sc):
    m = corr.max()
    if m <= 0:
        return 0.0
    w = np.exp(sc * (corr - m))
    return (m + np.log(w.sum()) / sc)

def _sopt_obj(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return _smooth_max(corr, sc) * dx

def _f_slsqp(x):
    return _sopt_obj(np.clip(x, 0.0, 1.0), _G_DX, _G_SCALE)

def _ceq(x):
    return x.sum() - 0.5 * _G_N

def _ceqj(x):
    return np.ones(len(x))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _slsqp(h0, n, dx, sc, mi):
    global _G_N, _G_DX, _G_SCALE
    _G_N, _G_DX, _G_SCALE = n, dx, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_f_slsqp, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-15, 'disp': False})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _upscale(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return _proj(h_new, n_new)

def _directed_local(h0, n, tmax, nit):
    t0 = time.time()
    tgt = 0.5 * n
    cx = h0.copy()
    c5x = _c5(cx)
    best = h0.copy()
    bestc = c5x
    for it in range(nit):
        if time.time() - t0 > tmax:
            break
        corr = np.correlate(cx, 1.0 - cx, mode="full")
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        h2 = cx.copy()
        # Directed: reduce product at bottleneck shift
        n_try = 4
        idx = np.random.randint(0, n, n_try)
        for i in idx:
            j = i + shift
            if j < 0 or j >= n:
                j = (i + abs(shift) + np.random.randint(0, 5)) % n
                if j == i:
                    j = (i + 1) % n
            target = 1.0 - h2[j]
            step = 0.08 * (target - h2[i])
            step = np.clip(step, -0.08, 0.08)
            h2[i] = h2[i] + step
            h2[j] = h2[j] - step
        h2 = _proj(h2, n)
        c5t = _c5(h2)
        if c5t < bestc - 1e-15:
            best, bestc = h2, c5t
            cx, c5x = h2, c5t
        elif c5t < c5x + 0.003 and np.random.random() < 0.2:
            cx, c5x = h2, c5t
        else:
            cx = best.copy()
            c5x = bestc
    return best, bestc

def _spectral_local(h0, n, tmax, nit):
    t0 = time.time()
    cx = h0.copy()
    c5x = _c5(cx)
    best = h0.copy()
    bestc = c5x

def _pair_local(h0, n, tmax, nit):

    tgt = 0.5 * n
    cx = h0.copy()
    c5x = _c5(cx)
    best = h0.copy()
    bestc = c5x
    for it in range(nit):
        if time.time() - t0 > tmax:
            break

            cx, c5x = h2, c5t
        else:
            cx = best.copy()
            c5x = bestc
    return best, bestc

def _multi_slsqp(h0, n, rem):
    dx = 2.0 / n
    best = h0.copy()
    bestc = _c5(best)
    for sc in [40.0, 80.0, 150.0, 250.0]:

        h1 = _slsqp(best, n, dx, sc, 2500)

        if c1 < bestc:
            best, bestc = h1, c1
        rem = rem - (2.0 / n) * 0  # SLSQP time not tracked precisely
    return best, bestc
