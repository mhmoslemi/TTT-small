_CONST_N = 70

    grad[v2] += (-h[ks[v2] + s])

    return compute_c5(x, _CUR_DX)

    return grad_c5(x, _CUR_DX)

    cm = np.max(c)

    ks = np.arange(n)

    """Fix sum to exactly T while maintaining [0,1] bounds."""
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) < 1e-12:

        h = np.clip(h, 0.0, 1.0)

        v2 = (ks + s >= 0) & (ks + s < n)
        g[v2] += -wm * x[ks[v2] + s]
    return _CUR_DX * g

def _norm(h, T):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(len(h), T / len(h))
    return np.clip(h * T / s, 0.0, 1.0)

def _solve(f, g, x0, bnds, cons, iters, ftol, t_end):
    if time.time() > t_end:
        return x0
    try:
        r = minimize(f, x0, method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': iters, 'ftol': ftol})
        return r.x
    except Exception:
        return x0

def _full_opt(h0, n, t_end):
    global _CUR_DX, _CONST_T, _CONST_N, _SMOOTH_T
    _CUR_DX = 2.0 / n
    _CONST_T = n / 2.0
    _CONST_N = n
    T = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    x = _norm(h0, T)
    best = x.copy()
    best_c = compute_c5(x, _CUR_DX)
    temps = [0.04, 0.02, 0.01, 0.005, 0.002]
    for tmp in temps:
        if time.time() > t_end - 10:
            break
        _SMOOTH_T = tmp
        x = _solve(_obj_sm, _jac_sm, x, bnds, cons, 500, 1e-12, t_end)
        x = _norm(x, T)
        c = compute_c5(x, _CUR_DX)
        if c < best_c:
            best_c, best = c, x.copy()
    if time.time() < t_end - 5:
        _SMOOTH_T = 0.001
        x = _solve(_obj_sm, _jac_sm, best, bnds, cons, 400, 1e-13, t_end)
        x = _norm(x, T)
        c = compute_c5(x, _CUR_DX)
        if c < best_c:
            best_c, best = c, x.copy()
    if time.time() < t_end:
        x = _solve(_obj, _jac, best, bnds, cons, 600, 1e-14, t_end)
        x = _norm(x, T)
        c = compute_c5(x, _CUR_DX)
        if c < best_c:
            best_c, best = c, x.copy()
    return best, best_c

def _perturb_reopt(h_best, n, rng, t_end):
    global _CUR_DX, _CONST_T, _CONST_N, _SMOOTH_T
    _CUR_DX = 2.0 / n
    _CONST_T = n / 2.0
    _CONST_N = n
    T = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    best = h_best.copy()
    best_c = compute_c5(best, _CUR_DX)
    scales = [0.04, 0.02, 0.01, 0.005]
    for sc in scales:
        if time.time() > t_end - 10:
            break
        for _ in range(4):
            if time.time() > t_end - 10:
                break
            h_p = best + rng.normal(0, sc, n)
            _SMOOTH_T = 0.005
            x = _solve(_obj_sm, _jac_sm, h_p, bnds, cons, 300, 1e-12, t_end)
            x = _norm(x, T)
            x = _solve(_obj, _jac, x, bnds, cons, 300, 1e-13, t_end)
            x = _norm(x, T)
            c = compute_c5(x, _CUR_DX)
            if c < best_c:
                best_c, best = c, x.copy()
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _CUR_DX, _CONST_T, _CONST_N, _SMOOTH_T
    t0 = time.time()
    t_end = t0 + budget_s - 20

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, TypeError):
        n0 = 70
        h0 = np.full(n0, 0.5)

    rng = np.random.RandomState(seed)
    best_h = _norm(h0, n0 / 2.0).copy()
    best_c5 = compute_c5(best_h, 2.0 / n0)

    # Phase 1: Full anneal from initial
    h_opt, c5_opt = _full_opt(best_h, n0, t_end)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt

    # Phase 2: Directed perturbations around best
    h_p, c5_p = _perturb_reopt(best_h, n0, rng, t_end)
    if c5_p < best_c5:
        best_c5, best_h = c5_p, h_p

    # Phase 3: Diversified restarts
    for i in range(60):
        if time.time() > t_end - 60:
            break
        r = rng.random()
        if r < 0.25:
            h_r = rng.random(n0)
        elif r < 0.4:
            blk = max(1, n0 // 14)
            h_r = np.zeros(n0)
            h_r[::2 * blk: ::2 * blk + blk] = 1.0
            h_r += rng.normal(0, 0.2, n0)
        elif r < 0.55:
            freq = rng.randint(2, 8)
            h_r = 0.5 + 0.35 * np.cos(2 * np.pi * freq * np.linspace(0, 1, n0))
            h_r += rng.normal(0, 0.08, n0)
        elif r < 0.7:
            h_r = best_h + rng.normal(0, 0.1, n0)
        elif r < 0.85:
            h_r = 0.5 + 0.3 * rng.randn(n0)
        else:
            h_r = np.clip(np.cumsum(rng.randn(n0)) / n0 * 4 + 0.3, 0, 1)

        h_r2, c5_r2 = _full_opt(h_r, n0, t_end)
        if c5_r2 < best_c5:
            best_c5, best_h = c5_r2, h_r2

        # Occasionally do a quick perturb-reopt on the newly found best
        if c5_r2 < best_c5 + 0.001 and time.time() < t_end - 30:
            if time.time() < t_end - 60:
                hp, cp = _perturb_reopt(best_h, n0, rng, t_end)
                if cp < best_c5:
                    best_c5, best_h = cp, hp

    # Phase 4: Try other grid sizes via interpolation
    for n_alt in [100, 50, 140]:
        if time.time() > t_end - 40:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        xn = np.linspace(0, 2, n_alt, endpoint=False) + 1.0 / n_alt
        h_proj = np.interp(xn, xo, best_h)
        h_alt, c5_alt = _full_opt(h_proj, n_alt, t_end)
        if c5_alt < best_c5:
            best_c5, best_h = c5_alt, h_alt

    # Phase 5: Final deep polish
    if time.time() < t_end - 5:
        n_f = len(best_h)
        _CUR_DX = 2.0 / n_f
        _CONST_T = n_f / 2.0
        _CONST_N = n_f
        T = n_f / 2.0
        bnds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        for tmp in [0.003, 0.001, 0.0005]:
            if time.time() > t_end - 3:
                break
            _SMOOTH_T = tmp
            x = _solve(_obj_sm, _jac_sm, best_h, bnds, cons, 3000, 1e-15, t_end)
            x = _norm(x, T)
            c = compute_c5(x, _CUR_DX)
            if c < best_c5:
                best_c5, best_h = c, x
        if time.time() < t_end - 2:
            x = _solve(_obj, _jac, best_h, bnds, cons, 5000, 1e-15, t_end)
            x = _norm(x, T)
            c = compute_c5(x, _CUR_DX)
            if c < best_c5:
                best_c5, best_h = c, x

    # Final fix: ensure sum is exactly n/2
    n_f = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = n_f / 2.0 - best_h.sum()
    best_h += diff / n_f
    best_h = np.clip(best_h, 0.0, 1.0)
    # Second correction for clipping
    diff2 = n_f / 2.0 - best_h.sum()
    free = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
    if free.sum() > 0 and abs(diff2) > 1e-10:
        best_h[free] += diff2 / free.sum()
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, 2.0 / n_f)

    return best_h, best_c5, n_f
