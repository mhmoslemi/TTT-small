c_f[s] = sum_{t=0}^{n-1-s} h[t] * (1-h[t+s])

For the "backward" part (s = -shift > 0, i.e., k = n-1-s):
c_b[s] = sum_{t=0}^{n-1-s} h[t+s] * (1-h[t])

Now, dc_f[s]/dh[j]:
= sum_{t=0}^{n-1-s} d/dh[j] [h[t]*(1-h[t+s])]
= (1-h[j+s]) if j+s < n (the h[t] term, t=j)
  + (-h[j-s]) if j-s ≥ 0 (the -(h[t+s]) term, t+s=j, t=j-s)
So: dc_f[s]/dh[j] = (1-h[j+s])*1_{j+s<n} + (-h[j-s])*1_{j≥s}

dc_b[s]/dh[j]:
= sum_{t=0}^{n-1-s} d/dh[j] [h[t+s]*(1-h[t])]
= (1-h[j-s]) if j-s ≥ 0 (the h[t+s] term, t+s=j, t=j-s)
  + (-h[j+s]) if j+s < n (the -(h[t]) term, t=j)
So: dc_b[s]/dh[j] = (1-h[j-s])*1_{j≥s} + (-h[j+s])*1_{j+s<n}

For s=0: c_f[0] = c_b[0] = sum h[t]*(1-h[t]), dc/dh[j] = 1-2h[j]
This is consistent: (1-h[j]) + (-h[j]) = 1-2h[j]. ✓

Now the full gradient:
d obj/dh[j] = dx * scale * sum_k weights[k] * dc[k]/dh[j]

= dx * scale * [weights[n-1]*(1-2h[j]) + sum_{s=1}^{n-1} weights[n-1+s]*((1-h[j+s])*1_{j+s<n} - h[j-s]*1_{j≥s}) + sum_{s=1}^{n-1} weights[n-1-s]*((1-h[j-s])*1_{j≥s} - h[j+s]*1_{j+s<n})]

This is O(n^2) to compute all n components, which is fine.

```python
def _soft_obj_grad(h, scale):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")  # 2n-1
    m = corr.max()
    exp_c = np.exp(scale * (corr - m))
    logsum = m + np.log(exp_c.sum()) / scale
    obj = logsum * dx
    w = exp_c / exp_c.sum()  # length 2n-1
    
    grad = np.zeros(n)
    # s=0 term (index n-1)
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    # Forward terms (s=1..n-1), indices n-1+s
    for s in range(1, n):
        ws = w[n-1+s]
        if ws < 1e-15:
            continue
        # (1-h[j+s]) for j=0..n-1-s
        grad[:n-s] += ws * (1.0 - h[s:])
        # -h[j-s] for j=s..n-1
        grad[s:] -= ws * h[:n-s]
        
    # Backward terms (s=1..n-1), indices n-1-s
    for s in range(1, n):
        ws = w[n-1-s]
        if ws < 1e-15:
            continue
        # (1-h[j-s]) for j=s..n-1
        grad[s:] += ws * (1.0 - h[:n-s])
        # -h[j+s] for j=0..n-1-s
        grad[:n-s] -= ws * h[s:]
    
    grad *= scale * dx
    return obj, grad
```

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj_simplex(h, target):
    """Exact Euclidean projection onto {0<=x<=1, sum(x)=target}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_softmax(h, scale):
    """Gradient of soft log-sum-exp approximation to max(corr)*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exp_c = np.exp(scale * (corr - m))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        wf = w[n-1+s]
        if wf > 1e-15:
            grad[:n-s] += wf * (1.0 - h[s:])
            grad[s:] -= wf * h[:n-s]
        wb = w[n-1-s]
        if wb > 1e-15:
            grad[s:] += wb * (1.0 - h[:n-s])
            grad[:n-s] -= wb * h[s:]
    return grad * (scale * dx)

def _accel_grad(h0, scale_start, scale_end, niter, tmax):
    """Accelerated projected gradient with annealing scale"""
    t0 = time.time()
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    h_prev = h.copy()
    mu = 0.7  # momentum
    best = h0.copy()
    bestc = _c5(h0)
    
    for i in range(niter):
        if time.time() - t0 > tmax:
            break
        # Anneal scale
        frac = i / max(niter - 1, 1)
        scale = scale_start * (scale_end / scale_start) ** frac
        
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        g = _grad_softmax(y, scale)
        
        # Adaptive step size (Barzilai-Borwein style)
        # Use a fixed step that decreases over time
        lr = 0.5 / (scale * (2.0 / n) * n)  # rough Lipschitz-based
        lr = min(lr * 0.5, 0.01)
        
        h_new = _proj_simplex(y - lr * g, target)
        h_prev = h
        h = h_new
        
        c = _c5(h)
        if c < bestc:
            best, bestc = h.copy(), c
    
    return best, bestc

    """Gradient-guided local search"""
    t0 = time.time()
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    c = _c5(h)
    best = h.copy()
    bestc = c

            break

        g = _grad_softmax(h, 100.0)
        # Pick coordinate with largest negative gradient component (most beneficial to increase)
        # or move a pair in the gradient direction

        jdx = np.argmax(g)

        h2[idx] = np.clip(h2[idx] + step, 0.0, 1.0)

            h, c = h2, c2
        elif np.random.random() < 0.1:

            h = h2

        lr = lr_base * (scales[0] / scale)
        lr = min(lr, 0.01 / (1.0 + 0.001 * i))
        h_new = _proj_simplex(y - lr * g, target)
        h_prev = h
        h = h_new
        if i % 50 == 0:
            c = _c5(h)
            if c < bestc:
                best, bestc = h.copy(), c
    c = _c5(h)
    if c < bestc:
        best, bestc = h.copy(), c
    return best, bestc

def _grad_local(h0, tmax, niter, rng):
    t0 = time.time()
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    c = _c5(h)
    best = h.copy()
    bestc = c
    scale = 200.0
    for i in range(niter):
        if time.time() - t0 > tmax:
            break
        g = _grad_softmax(h, scale)
        idx = np.argmin(g)
        jdx = np.argmax(g)
        if idx == jdx:
            continue
        step = rng.uniform(0.01, 0.12)
        h2 = h.copy()
        h2[idx] = np.clip(h2[idx] + step, 0.0, 1.0)
        h2[jdx] = np.clip(h2[jdx] - step, 0.0, 1.0)
        h2 = _proj_simplex(h2, target)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2.copy(), c2
            h, c = h2, c2
        elif c2 < c + 0.005 and rng.random() < 0.08:
            h = h2
    return best, bestc

def _random_local(h0, tmax, niter, rng):
    t0 = time.time()
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    c = _c5(h)
    best = h.copy()
    bestc = c
    for i in range(niter):
        if time.time() - t0 > tmax:
            break
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        if rng.random() < 0.5:
            i1, i2 = i2, i1
        step = rng.uniform(0.005, 0.2)
        h2 = h.copy()
        h2[i1] = np.clip(h2[i1] + step, 0.0, 1.0)
        h2[i2] = np.clip(h2[i2] - step, 0.0, 1.0)
        h2 = _proj_simplex(h2, target)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2.copy(), c2
            h = h2
        elif c2 < c + 0.01 and rng.random() < 0.12:
            h = h2
    return best, bestc

def _upscale(h, factor):
    n = len(h)
    n2 = n * factor
    h2 = np.repeat(h, factor).astype(float)
    kernel = np.ones(factor) / factor
    h2_smooth = np.convolve(h2, kernel, mode='same')
    h2 = 0.5 * h2 + 0.5 * h2_smooth
    return _proj_simplex(h2, 0.5 * n2)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(90) * 0.5
        h[:45] = 1.0
        h[45:] = 0.0

    n = len(h)
    bh = h.copy()
    bc = _c5(bh)

    # Phase 1: Accelerated gradient with annealing (coarse)
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        scales = [20, 40, 80, 150]
        h1, c1 = _accel_grad(bh, scales, 6000, min(rem * 0.12, 80.0))
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2: Gradient-guided local search
    rem = budget_s - (time.time() - ts)
    if rem > 50:
        h2, c2 = _grad_local(bh, min(rem * 0.12, 100.0), 800000, rng)
        if c2 < bc:
            bh, bc = h2, c2

    # Phase 3: More accelerated gradient (sharper)
    rem = budget_s - (time.time() - ts)
    if rem > 60:
        scales = [80, 150, 300]
        h3, c3 = _accel_grad(bh, scales, 8000, min(rem * 0.2, 150.0))
        if c3 < bc:
            bh, bc = h3, c3

    # Phase 4: Upscale 2x and optimize
    n2 = n * 2
    rem = budget_s - (time.time() - ts)
    if n2 <= 720 and rem > 150:
        h4 = _upscale(bh, 2)
        h4, c4 = _accel_grad(h4, [30, 60, 120, 250], 10000, min(rem * 0.25, 250.0))
        rem4 = budget_s - (time.time() - ts)
        if rem4 > 40:
            h4, c4 = _grad_local(h4, min(rem4 * 0.2, 100.0), 600000, rng)
        if c4 < bc:
            bh, bc, n = h4, c4, n2

    # Phase 5: Random local search for diversity
    rem = budget_s - (time.time() - ts)
    if rem > 60:
        h5, c5 = _random_local(bh, min(rem * 0.15, 120.0), 1000000, rng)
        if c5 < bc:
            bh, bc = h5, c5

    # Phase 6: Final sharp gradient + local polish
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        scales = [150, 300, 500]
        h6, c6 = _accel_grad(bh, scales, 6000, min(rem * 0.25, 150.0))
        if c6 < bc:
            bh, bc = h6, c5 if False else h6, c6
        rem6 = budget_s - (time.time() - ts)
        if rem6 > 20:
            h7, c7 = _grad_local(bh if bc < c6 else h6, min(rem6 * 0.5, 120.0), 800000, rng)
            if c7 < bc:
                bh, bc = h7, c7

    # Final: ensure feasibility and compute exact C5
    bh = _proj_simplex(bh, 0.5 * len(bh))
    # Verify sum constraint
    assert abs(bh.sum() - 0.5 * len(bh)) < 1e-8, f"Sum={bh.sum()}, target={0.5*len(bh)}"
    fc = _c5(bh)
    return (bh, fc, len(bh))
