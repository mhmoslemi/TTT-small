import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0 - 1e-12
        else:
            mask = proj > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            adj = diff / cnt
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    return np.clip(proj, 0.0, 1.0)

def grad_soft_max(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp  # weights for each lag, sums to 1
    
    grad = np.zeros(n)
    # For each lag m in [0, 2n-2], m-(n-1) is the actual shift
    for i in range(2 * n - 1):
        wi = w[i] * dx  # already includes dx
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            # f = sum_{j=0}^{n-m-1} h[j+m]*(1-h[j])
            # df/dh[k] = [k>=m]*(1-h[k-m]) - [k<=n-m-1]*h[k+m]
            if m == 0:
                grad += wi * (1 - 2*h)
            else:
                lo1 = m
                hi1 = n
                grad[lo1:hi1] += wi * (1.0 - h[0:hi1-lo1])
                lo2 = 0
                hi2 = n - m
                grad[lo2:hi2] -= wi * h[hi2:hi2+m]
        else:
            p = -m
            # f = sum_{j=0}^{n-p-1} h[j]*(1-h[j+p])
            # df/dh[k] = [k<=n-p-1]*(1-h[k+p]) - [k>=p]*h[k-p]
            if p == 0:
                grad += wi * (1 - 2*h)
            else:
                lo1 = 0
                hi1 = n - p
                grad[lo1:hi1] += wi * (1.0 - h[hi1:hi1+p])
                lo2 = p
                hi2 = n
                grad[lo2:hi2] -= wi * h[0:hi2-lo2]
    return obj, grad

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0 - 1e-12
        else:
            mask = proj > 1e-12
        cnt = int(mask.sum())
        if cnt > 0:
            proj[mask] = np.clip(proj[mask] + diff / cnt, 0.0, 1.0)
    return np.clip(proj, 0.0, 1.0)

def smooth_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            if m == 0:
                grad += wi * (1.0 - 2.0 * h)
            else:
                grad[m:] += wi * (1.0 - h[:n - m])
                grad[:n - m] -= wi * h[m:]
        else:
            p = -m
            if p == 0:
                grad += wi * (1.0 - 2.0 * h)
            else:
                grad[:n - p] += wi * (1.0 - h[p:])
                grad[p:] -= wi * h[:n - p]
    return obj, grad

def pg_optimize(h0, dx, ts, t_limit, beta_lo=50.0, beta_hi=5000.0, n_max_iter=30000):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 1.0
    stall = 0
    for it in range(n_max_iter):
        if time.time() > t_limit:
            break
        frac = min(1.0, it / max(1, n_max_iter * 0.3))
        beta = beta_lo + (beta_hi - beta_lo) * (frac ** 0.6)
        _obj, grad = smooth_grad(h, dx, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_try = project_feasible(h - lr * grad, ts)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            lr = min(lr * 1.03, 50.0)
            stall = 0
        else:
            h = h_try
            stall += 1
            lr *= 0.9
            if stall > 500 or lr < 1e-8:
                lr = max(lr, 0.5)
                stall = 0
                h = project_feasible(best_h + 0.01 * np.random.randn(n), ts)
        if stall > 3000:
            break
    return best_h, best_c5

def interp_resample(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    x_s = np.linspace(0.0, 2.0, n_src)
    x_d = np.linspace(0.0, 2.0, n_dst)
    h_new = np.interp(x_d, x_s, h_s for h_s in [h_src])
    return project_feasible(h_new, n_dst / 2.0)

def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    h += rng.uniform(-0.05, 0.05, n)
    return project_feasible(h, n / 2.0)

def perturb(h, ts, rng, n_pert=3):
    h_new = h.copy()

    h_new[idx] = np.clip(h_new[idx] + rng.uniform(-0.3, 0.3, n_pert), 0.0, 1.0)
    return project_feasible(h_new, ts)
