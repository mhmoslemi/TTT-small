def smooth_gradient(h, dx, temp):
    """Gradient of smoothed max objective (log-sum-exp)"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    w = np.exp((c - c_max) / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    ks = np.arange(n)
    grad = np.zeros(n)
    for m in range(2 * n - 1):

        if abs(w[m]) < 1e-15:

        v2 = (ks - s >= 0) & (ks - s < n)
        if v2.any():
            grad[v2] -= wm * h[ks[v2] - s]
    return dx * grad

def grad_sharp(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(c.argmax())
    s = idx - (n - 1)
    ks = np.arange(n)
    grad = np.zeros(n)
    v1 = (ks + s >= 0) & (ks + s < n)
    grad[v1] += (1.0 - h[ks[v1] + s])
    v2 = (ks - s >= 0) & (ks - s < n)
    grad[v2] -= h[ks[v2] - s]
    return dx * grad

def obj_smooth(x, dx, temp):
    c = np.correlate(x, 1.0 - h_ref[0], mode='full')
    cmax = c.max()
    return dx * (cmax + temp * np.log(np.sum(np.exp((c - cmax) / temp))))

def adam_optimize(h_init, n, dx, T, max_steps, lr_init, temp_init, temp_end, time_limit_fn):
    h = project_feasible(h_init.copy(), T)
    m_adv = np.zeros(n)
    v_adv = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stagnation = 0
    max_stag = 400
    for step in range(max_steps):
        if step % 50 == 0 and time_limit_fn():
            break
        frac = step / max(1, max_steps)
        temp = temp_init * (temp_end / temp_init) ** frac
        lr = lr_init * (1.0 - 0.8 * frac)
        if temp > 0.003:
            g = grad_smooth(h, dx, temp)
        else:
            g = grad_sharp(h, dx)
        t_s = step + 1
        m_adv = b1 * m_adv + (1.0 - b1) * g
        v_adv = b2 * v_adv + (1.0 - b2) * g * g
        mc = m_adv / (1.0 - b1 ** t_s)
        vc = v_adv / (1.0 - b2 ** t_s)
        h = h - lr * mc / (np.sqrt(vc) + eps)
        h = project_feasible(h, T)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > max_stag:
            lr = 0.3 * lr
            lr_init = 0.3 * lr_init
            max_stag = max(50, int(0.6 * max_stag))
            stagnation = 0
            if max_stag <= 50:
                break
    return best_h, best_c5

def _cons_eq_h(x, T):
    return x.sum() - T

def _cons_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h_init, n, dx, T, max_iter=2000):
    h0 = project_feasible(h_init.copy(), T)
    bounds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    def obj_fn(x):
        return compute_c5(x, dx)

    def jac_fn(x):
        return grad_sharp(x, dx)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - T, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj_fn, h0, method='SLSQP', jac=jac_fn, bounds=bounds,
                     constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-18})
        cx = np.clip(r.x, 0, 1)
        c5 = compute_c5(cx, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cx.copy()
    except Exception:
        pass

    for temp in [0.05, 0.02, 0.01, 0.005]:
        def obj_s(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = c.max()
            return dx * (cm + temp * np.log(np.sum(np.exp((c - cm) / temp))))
        def jac_s(x):
            return grad_smooth(x, dx, temp)
        try:
            r2 = minimize(obj_s, best_h, method='SLSQP', jac=jac_s, bounds=bounds,
                         constraints=cons, options={'maxiter': max(iter_n, 500), 'ftol': 1e-16})
            cx2 = np.clip(r2.x, 0, 1)
            c5s = compute_c5(cx2, dx)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = cx2.copy()
        except Exception:
            break
    return project_feasible(best_h, T), best_c5

def make_globals_n(n):
    return 2.0 / n, n / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    def time_up(thresh):
        return (time.time() - t0) > (budget_s - thresh)

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    except (NameError, TypeError):
        n = 70
        h_init = np.full(n, 0.5)

    dx = 2.0 / n
    T = n / 2.0
    rng = np.random.RandomState(seed)

    best_h = project_feasible(h_init.copy(), T)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Adam from initial, n points
    h1, c51 = adam_optimize(h_init, n, dx, T, 8000, 0.04, 0.12, 0.0005, lambda: time_up(200))
    if c51 < best_c5:
        best_c5, best_h = c51, h1

    # Phase 2: SLSQP polish
    h2, c52 = slsqp_optimize(best_h, n, dx, T, 3000)
    if c52 < best_c5:
        best_c5, best_h = c52, h2

    # Phase 3: Adam again from SLSQP result (different landscape)
    h3, c53 = adam_optimize(best_h, n, dx, T, 5000, 0.02, 0.04, 0.0003, lambda: time_up(150))
    if c53 < best_c5:
        best_c5, best_h = c53, h3
    h4, c54 = slsqp_optimize(best_h, n, dx, T, 2000)
    if c54 < best_c5:
        best_c5, best_h = c54, h4

    # Phase 4: Structured restarts
    n_restarts = 40
    for i in range(n_restarts):
        if time_up(100):
            break
        r = i
        if r % 4 == 0:
            h_r = best_h + 0.03 * rng.randn(n)
        elif r % 4 == 1:
            h_r = best_h.copy()
            bs = rng.randint(4, min(15, n // 3))
            st = rng.randint(0, n - bs)
            h_r[st:st + bs] = 1.0 - h_r[st:st + bs]
        elif r % 4 == 2:
            freq = rng.uniform(2, 8)
            amp = rng.uniform(0.05, 0.15)
            h_r = best_h + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
        else:
            h_r = 0.7 * best_h + 0.3 * rng.random(n)

        h_r = project_feasible(h_r, T)
        h5, c55 = adam_optimize(h_r, n, dx, T, 4000, 0.03, 0.08, 0.001, lambda: time_up(50))
        if c55 < best_c5:
            best_c5, best_h = c55, h5
            h6, c56 = slsqp_optimize(best_h, n, dx, T, 1500)
            if c56 < best_c5:
                best_c5, best_h = c56, h6

    # Phase 5: Try alternate n
    for n_alt in [100, 140, 50, 200]:
        if time_up(80):
            break
        dx_a = 2.0 / n_alt
        T_a = n_alt / 2.0
        for i in range(10):
            if time_up(40):
                break
            if i == 0:
                h_a = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_h)), best_h)
            elif i == 1:
                h_a = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, n), h_init)
            else:
                h_a = 0.5 + 0.25 * rng.randn(n_alt)
            h_a = project_feasible(h_a, T_a)
            h7, c57 = adam_optimize(h_a, n_alt, dx_a, T_a, 4000, 0.025, 0.06, 0.001, lambda: time_up(20))
            if c57 < best_c5:
                best_c5, best_h = c57, h7
                h8, c58 = slsqp_optimize(best_h, n_alt, dx_a, T_a, 1500)
                if c58 < best_c5:
                    best_c5, best_h = c58, h8
                break

    # Phase 6: Final SLSQP polish on best
    if time_up(10):
        final_n = len(best_h)
        final_dx = 2.0 / final_n
        final_T = final_n / 2.0
        best_h = project_feasible(best_h, final_T)
        best_c5 = compute_c5(best_h, final_dx)
        return best_h, best_c5, final_n

    fin_n = len(best_h)
    fin_dx = 2.0 / fin_n
    fin_T = fin_n / 2.0
    h_fin, c5_fin = slsqp_optimize(best_h, fin_n, fin_dx, fin_T, 5000)
    if c5_fin < best_c5:
        best_c5, best_h = c5_fin, h_fin

    best_h = project_feasible(best_h, fin_T)
    best_c5 = compute_c5(best_h, fin_dx)
    return best_h, best_c5, fin_n
