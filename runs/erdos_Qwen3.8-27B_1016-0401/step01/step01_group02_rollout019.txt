            i = np.random.randint(n)
            j = np.random.randint(n)
            k = np.random.randint(n)
            dv = np.random.uniform(0.005, 0.08)
            h2 = cw.copy()
            h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - dv * 0.7, 0.0, 1.0)
            h2[k] = np.clip(h2[k] - dv * 0.3, 0.0, 1.0)
        else:
            h2 = cw.copy()
            i = np.random.randint(n)
            w = np.random.randint(2, 5)
            lo = max(0, i - w)
            hi = min(n, i + w + 1)
            h2[lo:hi] = np.clip(h2[lo:hi] + np.random.normal(0, 0.03, hi - lo), 0.0, 1.0)
        ds = h2.sum() - tgt
        if abs(ds) > 1e-10:
            h2 -= ds / n
            np.clip(h2, 0.0, 1.0, out=h2)
        c5t = _c5(h2)
        if c5t < bestc:
            best, bestc = h2.copy(), c5t
            cw, c5w = h2, c5t
        elif np.random.random() < 0.12 and c5t < c5w + 0.008:
            cw, c5w = h2, c5t
        else:
            cw = best.copy()
            c5w = bestc
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5

    bh = h.copy()
    bc = _c5(bh)
    bn = len(bh)

    # Phase 1: Quick SLSQP to polish
    for sc in [40.0, 80.0]:
        rem = budget_s - (time.time() - ts)
        if rem < 30:
            break
        h1 = _slsqp(bh, bn, 2.0 / bn, sc, 1500)
        c1 = _c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2: Subgradient descent
    rem = budget_s - (time.time() - ts)
    if rem > 80:
        t_sg = min(rem * 0.2, 120.0)
        h2, c2 = _subgrad_desc(bh, bn, t_sg, init_lr=0.06)
        if c2 < bc:
            bh, bc = h2, c2

    # Phase 3: Local search
    rem = budget_s - (time.time() - ts)
    if rem > 60:
        hl, cl = _local_search(bh, bn, min(rem * 0.25, 200.0), 600000)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 4: Subgradient again at sharper focus
    rem = budget_s - (time.time() - ts)
    if rem > 60:
        h3, c3 = _subgrad_desc(bh, bn, min(rem * 0.15, 80.0), init_lr=0.04)
        if c3 < bc:
            bh, bc = h3, c3
        hl, cl = _local_search(bh, bn, min(rem * 0.2, 150.0), 500000)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 5: Upscale to 360
    n3 = 2 * bn
    rem = budget_s - (time.time() - ts)
    if n3 <= 400 and rem > 200:
        h4 = _proj(np.repeat(bh, 2), n3)
        for sc in [80.0, 150.0]:
            hh = _slsqp(h4, n3, 2.0 / n3, sc, 3000)
            cc = _c5(hh)
            if cc < _c5(h4):
                h4 = hh
        h4, c4 = _subgrad_desc(h4, n3, min(rem * 0.15, 100.0), init_lr=0.05)
        if c4 < bc:
            bh, bc, bn = h4, c4, n3
        elif _c5(h4) < bc:
            bh, bc, bn = h4, _c5(h4), n3

    # Phase 6: More local search
    rem = budget_s - (time.time() - ts)
    if rem > 50:
        hl, cl = _local_search(bh, bn, min(rem * 0.3, 250.0), 1000000)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 7: Final subgradient + polish
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        h5, c5s = _subgrad_desc(bh, bn, min(rem * 0.15, 60.0), init_lr=0.03)
        if c5s < bc:
            bh, bc = h5, c5s
        hl, cl = _local_search(bh, bn, min(rem * 0.4, 150.0), 600000)
        if cl < bc:
            bh, bc = hl, cl

    # Phase 8: Attempt upscale again if at n=180 and time allows
    rem = budget_s - (time.time() - ts)
    if bn == 180 and rem > 180:
        n4 = 2 * bn
        h6 = _proj(np.repeat(bh, 2), n4)
        hh = _slsqp(h6, n4, 2.0 / n4, 120.0, 3000)
        c6 = _c5(hh)
        hh, c6s = _subgrad_desc(hh, n4, min(rem * 0.2, 80.0), init_lr=0.04)
        c6s = _c5(hh)
        if c6s < bc:
            bh, bc, bn = hh, c6s, n4
            hl, cl = _local_search(bh, bn, min(rem * 0.25, 100.0), 400000)
            if cl < bc:
                bh, bc = hl, cl

    # Final SLSQP polish
    rem = budget_s - (time.time() - ts)
    if rem > 15:
        h7 = _slsqp(bh, bn, 2.0 / bn, 100.0, 1000)
        c7 = _c5(h7)
        if c7 < bc:
            bh, bc = h7, c7

    fc = _c5(bh)
    return (bh, fc, len(bh))
