import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = corr_vec(h)
    m = np.max(c)
    v = alpha * (c - m)
    v = np.clip(v, -50, 50)
    return (m + np.log(np.sum(np.exp(v))) / alpha) * dx

def _sm_obj(x, alpha, target):
    return soft_max_c5(x, alpha)

def _sm_con(x, target):
    return np.sum(x) - target

def slsqp_step(h, n, target, alpha, maxiter=100):
    try:
        res = minimize(_sm_obj, h.copy(), args=(alpha, target),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sm_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return make_feasible(res.x, target)
    except Exception:
        return h.copy()

def subgrad_step_multi(h, target, lr, K=5):
    n = len(h)
    c = corr_vec(h)
    m = np.max(c)
    thresh = m - 0.01
    active = np.where(c >= thresh)[0]
    if len(active) > K:
        active = np.argsort(c)[-K:]
    grad = np.zeros(n)
    for k in active:
        off = k - (n - 1)
        for p in range(n):

                grad[p] += (1.0 - h[p + off])
            if 0 <= p - off < n:
                grad[p] -= h[p - off]
    grad /= max(1, len(active))
    h_new = h - lr * grad
    return make_feasible(h_new, target)
