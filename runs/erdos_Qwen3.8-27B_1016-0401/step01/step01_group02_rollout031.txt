from scipy.optimize import minimize
import time

_G_n = 0
_G_dx = 0.0
_G_sc = 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def _project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(3000):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sm_f(x):
    hc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    mx = corr.max()
    return (mx + np.log(np.exp(_G_sc * (corr - mx)).sum()) / _G_sc) * _G_dx

def _sm_eq(x):
    return x.sum() - 0.5 * _G_n

def _sm_eqj(x):
    return np.ones(len(x))

def _slsqp(h0, n, sc, mi):
    global _G_n, _G_dx, _G_sc
    _G_n, _G_dx, _G_sc = n, 2.0 / n, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq, 'jac': _sm_eqj}]
    try:
        r = minimize(_sm_f, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-16})
        return _project(r.x, n)
    except Exception:
        return h0.copy()

def _local_search(h0, n, t_budget, n_iter, rng):
    t0 = time.time()
    h = h0.copy()
    best = h0.copy()
    best_c = _c5(best)
    cur_c = best_c
    tgt = 0.5 * n
    n2 = len(h)
    for it in range(n_iter):
        if time.time() - t0 > t_budget:
            break
        fr = it / n_iter
        step = (0.06 * (1.0 - fr) + 0.004) * rng.uniform(0.4, 2.0)
        mt = rng.random()
        h2 = h.copy()
        if mt < 0.60:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            h2[i] = np.clip(h2[i] + sgn * step, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - sgn * step, 0.0, 1.0)
        elif mt < 0.78:
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            h2[i], h2[j] = h2[j], h2[i]
        elif mt < 0.88:
            L = int(rng.integers(3, min(13, n // 4)))
            s = int(rng.integers(0, n - L))
            h2[s:s+L] = 1.0 - h2[s:s+L]
        else:
            i = rng.integers(n)
            h2[i] = 1.0 - h2[i]
        ds = h2.sum() - tgt
        if abs(ds) > 1e-14:
            h2 -= ds / n
            np.clip(h2, 0.0, 1.0, out=h2)
        c2 = _c5(h2)
        if c2 < best_c:
            best = h2.copy()
            best_c = c2
            h = h2
            cur_c = c2
        elif rng.random() < 0.12 * (1.0 - fr):
            h = h2
            cur_c = c2
        else:
            h = best.copy()
            cur_c = best_c
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5

    n = len(h)
    h = _project(h, n)
    best = h.copy()
    best_c = _c5(best)

    # Phase 1: SLSQP with multiple scales
    for sc in [20.0, 40.0, 70.0, 120.0]:
        if time.time() - t_start > budget_s * 0.13:
            break
        ht = _slsqp(best, n, sc, 4000)
        ct = _c5(ht)
        if ct < best_c:
            best, best_c = ht, ct

    # Phase 2: Local search at n
    rem = budget_s - (time.time() - t_start)
    if rem > 40:
        hl, cl = _local_search(best, n, min(rem * 0.18, 130), 250000, rng)
        if cl < best_c:
            best, best_c = hl, cl
        # Quick SLSQP polish
        ht = _slsqp(best, n, 80.0, 2000)
        ct = _c5(ht)
        if ct < best_c:
            best, best_c = ht, ct

    # Phase 3: Upscale to 2n
    n2 = 2 * n
    rem = budget_s - (time.time() - t_start)
    if rem > 250 and n2 <= 720:
        h2 = _project(np.repeat(best, 2), n2)
        c2_base = _c5(h2)
        for sc in [100.0, 180.0]:
            ht = _slsqp(h2, n2, sc, 5000)
            if _c5(ht) < c2_base:
                h2 = ht
                c2_base = _c5(h2)
        if c2_base < best_c:
            best, best_c, n = h2, c2_base, n2
        rem = budget_s - (time.time() - t_start)
        if rem > 80:
            hl, cl = _local_search(best, n, min(rem * 0.22, 180), 350000, rng)
            if cl < best_c:
                best, best_c = hl, cl
            ht = _slsqp(best, n, 150.0, 3000)
            ct = _c5(ht)
            if ct < best_c:
                best, best_c = ht, ct

    # Phase 4: Perturbation + re-optimization cycle
    rem = budget_s - (time.time() - t_start)
    if rem > 100:
        for _ in range(3):
            if time.time() - t_start > budget_s * 0.85:
                break
            hp = best + rng.normal(0, 0.08, n)
            hp = _project(hp, n)
            ht = _slsqp(hp, n, 60.0, 3000)
            ct = _c5(ht)
            if ct < best_c:
                best, best_c = ht, ct
            rem = budget_s - (time.time() - t_start)
            if rem < 30:
                break
            hl, cl = _local_search(best, n, min(rem * 0.15, 80), 150000, rng)
            if cl < best_c:
                best, best_c = hl, ct if cl < ct else hl, cl

    # Phase 5: Final intensive local search
    rem = budget_s - (time.time() - t_start)
    if rem > 20:
        hl, cl = _local_search(best, n, rem * 0.85, 600000, rng)
        if cl < best_c:
            best, best_c = hl, cl

    # Final polish with SLSQP
    rem = budget_s - (time.time() - t_start)
    if rem > 5:
        ht = _slsqp(best, n, 200.0, 2000)
        ct = _c5(ht)
        if ct < best_c:
            best = ht

    best = _project(best, n)
    final_c = _c5(best)
    return (best, final_c, len(best))
