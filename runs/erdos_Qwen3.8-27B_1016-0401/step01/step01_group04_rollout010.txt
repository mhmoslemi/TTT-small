def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max(corr, T):
    mx = corr.max()
    if T < 1e6:
        shifted = corr - mx
        shifted[shifted > 50.0/T] = 50.0/T  # prevent overflow
        return mx + np.log(np.sum(np.exp(T * shifted))) / T
    return mx

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(-np.abs(h - 0.5))[0:int(abs(diff))+1]
        for i in idx:
            h[i] = np.clip(h[i] + np.sign(diff), 0, 1)

        return objective_continuous(x, T)

            xm = x.copy(); xm[i] -= eps

    sa_h, sa_c5 = binary_sa_short(best_h, n, S, rng, 100000, t_sa_end, best_c5)
    if sa_c5 < best_c5:
        best_c5 = sa_c5
        best_h = sa_h.copy()

    # Phase 4: Final extended local search
    t_final_end = t0 + budget_s * 0.96
    best_h, best_c5 = local_search(best_h, n, S, max_sweeps=40, t_deadline=t_final_end)

    # Phase 5: Quick SA + LS cycle
    if time.time() < t0 + budget_s * 0.98:
        sa_h, sa_c5 = binary_sa_short(best_h, n, S, rng, 30000, t0 + budget_s * 0.97, best_c5)
        if sa_c5 < best_c5:
            best_c5 = sa_c5
            best_h = sa_h.copy()
        best_h, best_c5 = local_search(best_h, n, S, max_sweeps=10, t_deadline=t0 + budget_s * 0.99)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
