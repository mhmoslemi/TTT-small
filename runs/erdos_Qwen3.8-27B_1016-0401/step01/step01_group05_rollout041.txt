```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        adjust = diff / cnt
        h[mask] += adjust
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmax(h) if diff > 0 else np.argmin(h)
        # distribute among all to fix
        h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = alpha * (corr - m)
    vals = np.clip(vals, -50, 50)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

        return soft_max_c5(x, alpha)

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, target)
    except Exception:
        return h
```

```python
    # Phase 5: Final SLSQP
    if time.time() < deadline - 2:
        for alpha in [800, 1500, 3000]:
            if time.time() > deadline - 2:
                break
            h_new = slsqp_step(best_h, n, target, alpha, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = make_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h.tolist(), best_c5, n
```
