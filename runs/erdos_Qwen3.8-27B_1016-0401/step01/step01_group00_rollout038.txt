        if w_neg > 1e-15:

            grad += w_neg * (t1 - t2)
    
    # s=0
    grad += sw[center] * (1.0 - 2.0 * x)
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr_array(h):
    return np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += diff * room / tot
        else:
            cap = h
            tot = cap.sum()
            if tot < 1e-15:
                break
            h += diff * cap / tot
    return np.clip(h, 0.0, 1.0)

def binary_project(h, seed=0):
    n = len(h)
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    diff = target - int(h_b.sum())
    if diff > 0:
        zeros = np.where(h_b == 0)[0]
        rng = np.random.default_rng(seed)
        h_b[rng.choice(zeros, size=diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        rng = np.random.default_rng(seed)
        h_b[rng.choice(ones, size=-diff, replace=False)] = 0.0
    return h_b

def swap_search(h, seed, time_limit, init_T=4.0, cooling=0.9999):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.astype(float).copy()
    C = get_corr_array(h)
    C_max = float(C.max())
    best_C_max = C_max
    best_h = h.copy()
    s_vals = np.arange(-(n - 1), n)
    ones_idx = np.where(h == 1)[0].copy()
    zeros_idx = np.where(h == 0)[0].copy()
    n1 = n // 2
    T = init_T
    T_min = 0.002
    t0 = time.time()
    no_imp = 0
    while time.time() - t0 < time_limit:
        if no_imp > 4000:
            T = init_T * 0.5
            no_imp = 0
        i = ones_idx[rng.integers(n1)]
        j = zeros_idx[rng.integers(n1)]
        idx1 = i + s_vals
        v_i = np.zeros(2 * n - 1)
        m1 = (idx1 >= 0) & (idx1 < n)
        v_i[m1] = h[idx1[m1]]
        idx2 = i - s_vals
        m2 = (idx2 >= 0) & (idx2 < n)
        v_i[m2] += h[idx2[m2]]
        idx3 = j + s_vals
        v_j = np.zeros(2 * n - 1)
        m3 = (idx3 >= 0) & (idx3 < n)
        v_j[m3] = h[idx3[m3]]
        idx4 = j - s_vals
        m4 = (idx4 >= 0) & (idx4 < n)
        v_j[m4] += h[idx4[m4]]
        delta = v_i - v_j
        new_max = float((C + delta).max())
        if new_max <= C_max or rng.random() < np.exp(-(new_max - C_max) / T):
            h[i] = 0.0
            h[j] = 1.0
            C += delta
            C_max = new_max
            pi = np.nonzero(ones_idx == i)[0][0]
            pj = np.nonzero(zeros_idx == j)[0][0]
            ones_idx[pi] = j
            zeros_idx[pj] = i
            if C_max < best_C_max - 1e-10:
                best_C_max = C_max
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        T = max(T * cooling, T_min)
    return best_h, best_C_max * 2.0 / n

def softobj(x, n, T):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = corr.max()
    shifted = T * (corr - mc)
    ms = shifted.max()
    val = mc + (ms + np.log(np.sum(np.exp(shifted - ms)))) / T
    return val * (2.0 / n)

def softgrad(x, n, T):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = corr.max()
    shifted = T * (corr - mc)
    ms = shifted.max()
    e = np.exp(shifted - ms)
    sw = e / e.sum()
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    if sw[center] > 1e-15:
        grad += sw[center] * (1.0 - 2.0 * x)
    for o in range(1, n):
        wp = sw[center + o]
        wn = sw[center - o]
        if wp > 1e-14 or wn > 1e-14:
            t1 = np.zeros(n)
            t1[:n - o] = 1.0 - x[:n - o]
            t2 = np.zeros(n)
            t2[o:] = x[:n - o]
            if wp > 1e-14:
                grad += wp * (t1 - t2)
            if wn > 1e-14:
                t3 = np.zeros(n)
                t3[o:] = 1.0 - x[:n - o]
                t4 = np.zeros(n)
                t4[:n - o] = x[o:]
                grad += wn * (t3 - t4)
    return grad * dx

def grad_refine(h, target, n, time_limit, T_list=(80, 200, 500)):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    best = fix_sum(h.copy(), target)
    best_c5 = compute_c5(best)
    for T in T_list:
        if time.time() - t0 > time_limit:
            break
        def obj(x):
            return softobj(x, n, T)
        def jct(x):
            return softgrad(x, n, T)
        try:
            res = minimize(obj, best, method='SLSQP', jac=jct,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.fun < 100:
                h_new = fix_sum(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best = h_new
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.zeros(280)
        h_start[::2] = 1.0
    n = len(h_start)
    target = n / 2.0
    h_start = fix_sum(h_start, target)

    h_bin = binary_project(h_start, seed)
    h_best, c5_best = swap_search(h_bin, seed, 300, init_T=4.0)

    if elapsed() < budget_s - 350:
        h_r = (np.random.default_rng(seed + 1).random(n) >= 0.5).astype(float)
        h_r = binary_project(h_r, seed + 1)
        h_t, c5_t = swap_search(h_r, seed + 1, 100, init_T=6.0)
        if c5_t < c5_best:
            h_best, c5_best = h_t, c5_t

    if elapsed() < budget_s - 250:
        for s2 in [seed + 10, seed + 20, seed + 30]:
            if elapsed() > budget_s - 150:
                break
            h_t2, c5_t2 = swap_search(binary_project(h_best, s2), s2, 40, init_T=3.0)
            if c5_t2 < c5_best:
                h_best, c5_best = h_t2, c5_t2

    if elapsed() < budget_s - 200:
        h_g, c5_g = grad_refine(h_best, target, n, 120, T_list=(60, 150, 400))
        if c5_g < c5_best:
            h_best, c5_best = h_g, c5_g

    if elapsed() < budget_s - 100:
        h_b2 = binary_project(h_best, seed + 50)
        h_s2, c5_s2 = swap_search(h_b2, seed + 50, 60, init_T=3.0)
        if c5_s2 < c5_best:
            h_best, c5_best = h_s2, c5_s2
            if elapsed() < budget_s - 50:
                h_g3, c5_g3 = grad_refine(h_best, target, n, 40, T_list=(100, 300))
                if c5_g3 < c5_best:
                    h_best, c5_best = h_g3, c5_g3

    if elapsed() < budget_s - 60:
        n_hi = n * 2
        target_hi = n_hi / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = fix_sum(h_hi, target_hi)
        h_hi_b = binary_project(h_hi, seed + 99)
        h_hi_s, c5_hi = swap_search(h_hi_b, seed + 99, 40, init_T=3.0)
        if c5_hi < c5_best:
            h_best, c5_best = h_hi_s, c5_hi

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
