    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def gd_anneal(h_init, n, temps, lrs, iters, deadline):
    dx = 2.0 / n
    T = n / 2.0
    h = project_feasible(h_init.copy(), T)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for pi in range(len(temps)):
        t = temps[pi]
        lr = lrs[pi]
        for it in range(iters[pi]):
            if time.time() > deadline:
                return best_h, best_c5
            grad = grad_smooth_max(h, t)
            step = lr * dx
            h_new = project_feasible(h - step * grad, T)
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def slsqp_polish(h_init, n, max_iter, deadline):
    dx = 2.0 / n
    T = n / 2.0
    h = project_feasible(h_init.copy(), T)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for t in [0.08, 0.03, 0.01, 0.004]:
        if time.time() > deadline - 3:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            return dx * (cm + t * np.log(np.sum(np.exp((c - cm) / t))))
        def jac(x):
            return dx * grad_smooth_max(x, t)
        def cf(x):
            return x.sum() - T
        def cj(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            r = minimize(obj, h, method='SLSQP', jac=jac,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': max_iter, 'ftol': 1e-15})
            c5 = compute_c5(r.x, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = r.x.copy()
            h = project_feasible(r.x, T)
        except Exception:
            pass
    return best_h, best_c5

def perturb_local(h, rng, n_frac=0.15, amp=0.1):
    n = len(h)
    k = max(2, int(n * n_frac))
    idx = rng.choice(n, size=k, replace=False)
    h_new = h.copy()
    h_new[idx] += rng.uniform(-amp, amp, size=k)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, TypeError):
        n0 = 70
        h0 = np.full(n0, 0.5)

    T0 = n0 / 2.0
    dx0 = 2.0 / n0
    h0p = project_feasible(h0, T0)
    best_h = h0p.copy()
    best_c5 = compute_c5(h0p, dx0)
    rng = np.random.RandomState(seed)

    # Phase 1: GD anneal from parent
    h1, c1 = gd_anneal(h0p, n0,
                       [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                       [0.03, 0.02, 0.012, 0.007, 0.004, 0.002, 0.001],
                       [300, 300, 400, 400, 500, 500, 600], deadline)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    h2, c2 = slsqp_polish(best_h, n0, 500, deadline)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: Perturbation + re-optimize (escape local minima)
    for i in range(15):
        if time.time() > deadline - 80:
            break
        hp = perturb_local(best_h, rng)
        hp, cp = gd_anneal(hp, n0, [0.1, 0.03, 0.01],
                          [0.01, 0.004, 0.002],
                          [300, 400, 500], deadline)
        if cp < best_c5:
            best_h, best_c5 = hp, cp
            hs, cs = slsqp_polish(hp, n0, 300, deadline)
            if cs < best_c5:
                best_h, best_c5 = hs, cs

    # Phase 4: Random restarts
    for i in range(30):
        if time.time() > deadline - 100:
            break
        h_r = rng.random(n0)
        hr, cr = gd_anneal(h_r, n0, [0.3, 0.1, 0.03, 0.01],
                          [0.03, 0.015, 0.006, 0.002],
                          [200, 300, 400, 500], deadline)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
            hs, cs = slsqp_polish(hr, n0, 300, deadline)
            if cs < best_c5:
                best_h, best_c5 = hs, cs

    # Phase 5: Structured initializations
    x = np.linspace(0, 2, n0, endpoint=False)
    for freq, amp in [(2.0, 0.35), (3.0, 0.3), (4.0, 0.25), (5.0, 0.2)]:
        if time.time() > deadline - 60:
            break
        hs_init = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0)
        hs_init = project_feasible(hs_init, T0)
        hs, cs = gd_anneal(hs_init, n0, [0.2, 0.05, 0.015, 0.005],
                          [0.02, 0.008, 0.003, 0.001],
                          [200, 300, 400, 400], deadline)
        if cs < best_c5:
            best_h, best_c5 = hs, cs
            hs2, cs2 = slsqp_polish(hs, n0, 300, deadline)
            if cs2 < best_c5:
                best_h, best_c5 = hs2, cs2

    # Phase 6: Alt grid refinement
    for na in [50, 100, 140, 200]:
        if time.time() > deadline - 40:
            break
        xo = np.linspace(0, 2, n0, endpoint=False)
        xn = np.linspace(0, 2, na, endpoint=False)
        hu = project_feasible(np.interp(xn, xo, best_h), na / 2.0)
        ha, ca = gd_anneal(hu, na, [0.2, 0.06, 0.02, 0.006],
                          [0.015, 0.006, 0.002, 0.001],
                          [200, 300, 400, 400], deadline)
        if ca < best_c5:
            best_h, best_c5 = ha, ca
            hs, cs = slsqp_polish(ha, na, 400, deadline)
            if cs < best_c5:
                best_h, best_c5 = hs, cs

    # Final SLSQP polish
    if time.time() < deadline - 8:
        nf = len(best_h)
        hf, cf = slsqp_polish(best_h, nf, 1500, deadline)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    nf = len(best_h)
    best_h = project_feasible(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / nf)
    return best_h, best_c5, nf
