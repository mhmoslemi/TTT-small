    dx = 2.0 / n

        if abs(diff) < 1e-14:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    target_local = target
    alpha_local = alpha

    h_best = h_curr.copy()

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[np.where(mask)[0]] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, target, alpha, maxiter=200):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    
    def con(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n, target)
    except:
        return h.copy()

def continuous_local_search(h, n, target, rng, deadline, max_iters=50000):
    """Adaptive 2-coordinate local search."""
    best_c5 = compute_c5(h)
    h_curr = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-14) if diff > 0 else (h > 1e-14)
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, target, alpha, maxiter=200):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def fix_binary(h, n, target):
    h = (h >= 0.5).astype(float)
    ones = np.where(h == 1)[0]
    while len(ones) > target:
        ones = np.where(h == 1)[0]
        h[ones[len(ones) - 1]] = 0
    zeros = np.where(h == 0)[0]
    while len(zeros) > 0 and np.sum(h) < target:
        zeros = np.where(h == 0)[0]
        if len(zeros) == 0:
            break
        h[zeros[0]] = 1
    return h

def binary_local_search(h0, n, rng, deadline, max_iters=500000):
    target = n // 2
    h = fix_binary(h0, n, target)
    one_list = np.where(h == 1)[0].tolist()
    zero_list = np.where(h == 0)[0].tolist()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 7 == 0 and len(one_list) >= 2 and len(zero_list) >= 2:
            idx1 = rng.randint(len(one_list))
            idx2 = rng.randint(len(one_list) - 1)
            if idx2 >= idx1:
                idx2 += 1
            j1i = rng.randint(len(zero_list))
            j2i = rng.randint(len(zero_list) - 1)
            if j2i >= j1i:
                j2i += 1
            i1, i2 = one_list[idx1], one_list[idx2]
            j1, j2 = zero_list[j1i], zero_list[j2i]
            h[i1], h[i2], h[j1], h[j2] = h[j1], h[j2], h[i1], h[i2]
            one_list[idx1], one_list[idx2] = j1, j2
            zero_list[j1i], zero_list[j2i] = i1, i2
            new_c5 = compute_c5(h)
            if new_c5 <= curr_c5:
                curr_c5 = new_c5
                no_improve = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
            else:
                h[i1], h[i2], h[j1], h[j2] = h[j2], h[j1], h[i2], h[i1]
                one_list[idx1], one_list[idx2] = i1, i2
                zero_list[j1i], zero_list[j2i] = j1, j2
                no_improve += 1
        else:
            i_idx = rng.randint(len(one_list))
            j_idx = rng.randint(len(zero_list))
            i, j = one_list[i_idx], zero_list[j_idx]
            h[i], h[j] = h[j], h[i]
            one_list[i_idx], zero_list[j_idx] = j, i
            new_c5 = compute_c5(h)
            if new_c5 <= curr_c5:
                curr_c5 = new_c5
                no_improve = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
            else:
                h[i], h[j] = h[j], h[i]
                one_list[i_idx], zero_list[j_idx] = i, j
                no_improve += 1
        if no_improve > 800:
            for _ in range(4):
                a_idx = rng.randint(len(one_list))
                b_idx = rng.randint(len(zero_list))
                a, b = one_list[a_idx], zero_list[b_idx]
                h[a], h[b] = h[b], h[a]
                one_list[a_idx], zero_list[b_idx] = b, a
            curr_c5 = compute_c5(h)
            no_improve = 0
    return h, best_c5

def greedy_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    positions = list(range(n))
    rng.shuffle(positions)
    for _ in range(target):
        best_pos, best_cost = None, 1e9
        for pos in positions:
            h[pos] = 1.0
            c5 = compute_c5(h)
            h[pos] = 0.0
            if c5 < best_cost:
                best_cost = c5
                best_pos = pos
        if best_pos is not None:
            h[best_pos] = 1.0
            positions.remove(best_pos)
    return h

def continuous_sa(h0, n, target, rng, t_start, t_end):
    h_curr = h0.copy()
    score_curr = compute_c5(h_curr)
    h_best = h_curr.copy()
    score_best = score_curr
    T = 0.008
    T_min = 1e-5
    cooling = 0.9997
    step_base = 0.04
    no_improve = 0
    while time.time() < t_end and T > T_min:
        k = rng.randint(2, 5)
        idx = rng.choice(n, size=k, replace=False)
        amp = step_base * (0.5 + 0.5 * T / 0.008)
        delta = rng.normal(0, amp, size=k)
        h_try = h_curr.copy()
        h_try[idx] += delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n, target)
        score_try = compute_c5(h_try)
        if score_try < score_curr or rng.random() < np.exp(-(score_try - score_curr) / max(T, 1e-12)):
            h_curr = h_try
            score_curr = score_try
            no_improve = 0
            if score_try < score_best:
                h_best = h_try.copy()
                score_best = score_try
        else:
            no_improve += 1
            if no_improve > 3000:
                T *= 0.5
                no_improve = 0
        T *= cooling
    return h_best, score_best

def continuous_hill_climb(h0, n, target, rng, deadline, max_iters=80000):
    h_curr = h0.copy()
    best_c5 = compute_c5(h_curr)
    step = 0.06
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (i + 1) % n
        delta = rng.uniform(-step, step)
        h_try = h_curr.copy()
        h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
        h_try[j] = np.clip(h_try[j] - delta, 0.0, 1.0)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_curr = h_try
            best_c5 = c5_try
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                step *= 0.6
                no_improve = 0
                if step < 0.002:
                    break
    return h_curr, best_c5

def make_structured_binary(n):
    h = np.zeros(n)
    target = n // 2
    if n % 4 == 0:
        h[0::4] = 1.0
        h[1::4] = 1.0
    elif n % 2 == 0:
        h[0::2] = 1.0
    else:
        h[0::2] = 1.0
    h = fix_binary(h, n, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, n, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search from multiple starts on parent n
    t_bin_end = min(deadline - 400, t0 + 350)
    starts = []
    starts.append(fix_binary(h_init, n, n // 2))
    starts.append(greedy_binary(n, rng))
    starts.append(make_structured_binary(n))
    for _ in range(4):
        hb = np.zeros(n)
        hb[rng.permutation(n)[:n // 2]] = 1.0
        starts.append(hb)
    for s in starts:
        if time.time() > t_bin_end:
            break
        h_b, c5_b = binary_local_search(s, n, rng, t_bin_end, max_iters=300000)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b
    # Extra binary search from best
    if time.time() < t_bin_end + 50:
        h_b2, c5_b2 = binary_local_search(best_h, n, rng, min(t_bin_end + 50, deadline - 350), max_iters=400000)
        if c5_b2 < best_c5:
            best_h = h_b2
            best_c5 = c5_b2

    # Phase 2: SLSQP continuous refinement
    t_sl_end = min(deadline - 250, time.time() + 180)
    h_cont = best_h.copy()
    for alpha in [30, 100, 300, 800, 1500]:
        if time.time() > t_sl_end:
            break
        h_new = slsqp_optimize(h_cont, n, target, alpha, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        h_cont = h_new

    # Phase 3: Continuous SA + hill climb
    if time.time() < deadline - 120:
        h_sa, c5_sa = continuous_sa(best_h, n, target, rng, time.time(), min(deadline - 100, time.time() + 100))
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa
    if time.time() < deadline - 50:
        h_hc, c5_hc = continuous_hill_climb(best_h, n, target, rng, deadline - 30, max_iters=60000)
        if c5_hc < best_c5:
            best_h = h_hc
            best_c5 = c5_hc

    # Phase 4: Try other n values
    for n_try in [56, 64, 80, 96]:
        if time.time() > deadline - 40:
            break
        t_arg = n_try / 2.0
        h_try = greedy_binary(n_try, rng)
        h_try, c5_try = binary_local_search(h_try, n_try, rng, min(deadline - 20, time.time() + 50), max_iters=100000)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
        if time.time() < deadline - 25:
            h_try2 = slsqp_optimize(h_try, n_try, t_arg, 400, maxiter=150)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < best_c5:
                best_h = h_try2
                best_c5 = c5_try2

    # Phase 5: Final polish
    n_final = len(best_h)
    t_fin = n_final / 2.0
    if time.time() < deadline - 8:
        h_fin = slsqp_optimize(best_h, n_final, t_fin, 3000, maxiter=150)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin
    best_h = make_feasible(best_h, n_final, t_fin)
    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n_final
