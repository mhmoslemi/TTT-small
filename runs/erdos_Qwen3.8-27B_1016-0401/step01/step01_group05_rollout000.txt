    n = _G['n']

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    a, nn = _G['alpha'], _G['n']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    v = np.clip(a * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(v))) / a) * (2.0 / nn)

def _con(x):
    return np.sum(x) - _G['target']

def slsqp(h, n, target, alpha, mi=200):
    _G['alpha'], _G['n'], _G['target'] = alpha, n, target
    try:
        r = minimize(_obj, h.copy(), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        return make_feasible(r.x, target)
    except Exception:
        return h.copy()

def annealing(h_start, n, target, rng, deadline):
    hc = h_start.copy()
    cc5 = compute_c5(hc)
    bc5 = cc5
    bh = hc.copy()
    ss = 0.07
    T = 0.03
    last_imp = 0
    it = 0
    while time.time() < deadline - 5:
        it += 1
        i, j = rng.randint(n), rng.randint(n)
        if j == i:
            j = (i + 1) % n
        d = rng.normal(0.0, ss)
        ht = hc.copy()
        nv_i = ht[i] + d
        nv_j = ht[j] - d
        ht[i] = 1.0 if nv_i > 1.0 else (0.0 if nv_i < 0.0 else nv_i)
        ht[j] = 1.0 if nv_j > 1.0 else (0.0 if nv_j < 0.0 else nv_j)
        sf = target - ht.sum()
        if abs(sf) > 1e-10:
            m2 = (ht > 1e-8) & (ht < 1.0 - 1e-8)
            cn = np.count_nonzero(m2)
            if cn > 0:
                ht[m2] += sf / cn
                ht = np.clip(ht, 0.0, 1.0)
        tc5 = compute_c5(ht)
        dif = tc5 - cc5
        if dif < 0.0 or rng.random() < np.exp(-dif / T):
            hc, cc5 = ht, tc5
            if cc5 < bc5:
                bc5, bh = cc5, hc.copy()
                last_imp = it
        if it % 2000 == 0:
            ss *= 0.992
            T *= 0.997
            if ss < 0.003:
                ss = 0.003
            if T < 5e-6:
                T = 5e-6
        if it % 25000 == 0 and time.time() < deadline - 30:
            al = min(200 + (it // 25000) * 80, 3000)
            h1 = slsqp(hc, n, target, al, 80)
            c1 = compute_c5(h1)
            if c1 < bc5:
                bc5, bh, hc, cc5 = c1, h1, h1.copy(), c1
                last_imp = it
        if it - last_imp > 60000 and it > 15000:
            hc = bh.copy() + rng.normal(0.0, 0.012, n)
            hc = make_feasible(hc, target)
            cc5 = compute_c5(hc)
            T, ss = 0.025, 0.04
            last_imp = it
    return bh, bc5
