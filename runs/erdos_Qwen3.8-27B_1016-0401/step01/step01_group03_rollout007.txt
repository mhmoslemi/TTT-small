from functools import partial
import time

def compute_c5(h, dx):

    return np.max(corr) * dx

def compute_c5_and_argmax(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    kmax = np.argmax(corr)
    return corr[kmax] * dx, kmax

def fix_count(h, target):
    """Fix the number of ones to exactly target."""
    n = len(h)
    s = int(np.round(h.sum()))
    if s > target:
        idx = np.where(h > 0.5)[0]
        for i in idx[:s-target]:
            h[i] = 0.0
    elif s < target:
        idx = np.where(h < 0.5)[0]
        for i in idx[:target-s]:
            h[i] = 1.0
    return h

def swap1_local_search(h, dx, t_limit):
    """Exhaustive 1-swap until no improvement."""
    n = len(h)
    c5 = compute_c5(h, dx)
    while time.time() < t_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for oi in ones:
            for zi in zeros:
                h[oi], h[zi] = 0.0, 1.0
                c = compute_c5(h, dx)
                if c < c5 - 1e-15:
                    c5 = c
                    improved = True
                    break
                h[zi], h[oi] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return h, c5

def swap2_local_search(h, dx, t_limit):

    n = len(h)
    c5 = compute_c5(h, dx)

        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])

            for i2 in range(i1+1, len(ones)):
                for j1 in range(len(zeros)):

                        h[o1]=0; h[o2]=0; h[z1]=1; h[z2]=1

def block_move_search(h, dx, t_limit):
    """Try shifting contiguous blocks of 1s."""
    n = len(h)
    c5 = compute_c5(h, dx)
    while time.time() < t_limit:
        improved = False
        # Find contiguous blocks of 1s
        blocks = []
        i = 0
        while i < n:
            if h[i] == 1:
                j = i
                while j < n and h[j] == 1:
                    j += 1
                blocks.append((i, j-1))
                i = j
            else:
                i += 1
        for (s, e) in blocks:
            length = e - s + 1
            for shift in range(-n//2, n//2+1):
                if shift == 0:
                    continue
                ns, ne = s+shift, e+shift
                if ns < 0 or ne >= n:
                    continue
                # Check if the target positions are all 0 (except overlap with source)
                for p in range(ns, ne+1):
                    if 0 <= p < n and h[p] == 1 and not (s <= p <= e):

def smooth_optimize(h, dx, target, t_limit):
    """Smooth optimization with log-sum-exp."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    def obj_grad(hv, beta):
        corr = np.correlate(hv, 1.0-hv, mode='full')
        max_c = np.max(corr)
