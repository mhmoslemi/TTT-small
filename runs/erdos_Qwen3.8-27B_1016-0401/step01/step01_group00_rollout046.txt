def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(x, target):
    """Project x onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(x, 0.0, 1.0).copy()
    # Bisection on lambda to satisfy sum constraint
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def softmax_gradient(corr, T, n):
    """Compute gradient of soft-max(corr) w.r.t. h"""
    dx = 2.0 / n
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    w = e / np.sum(e)
    
    grad = np.zeros(n)

            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0, 1)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):

    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= np.max(sh)
    e = np.exp(sh)

    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]

            grad[:n-o] -= wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def lse_obj(corr, T, n):
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def lse_grad(corr, h, T, n):
    dx = 2.0 / n
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = sh - np.max(sh)
    e = np.exp(sh)
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for idx in nz:
        o = idx - center
        wi = w[idx]
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n - o])
            grad[:n - o] -= wi * h[o:]
        else:
            p = -o
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def topk_subgrad(corr, h, K, n):
    dx = 2.0 / n
    top_idx = np.argsort(corr)[-K:]
    center = n - 1
    g = np.zeros(n)
    for idx in top_idx:
        o = int(idx) - center
        if o == 0:
            g += (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g * dx / K

def slsqp_phase(h, n, target, bounds, T, maxiter):
    def obj(x):
        return lse_obj(np.correlate(x, 1.0 - x, mode="full"), T, n)
    def jac(x):
        return lse_grad(np.correlate(x, 1.0 - x, mode="full"), x, T, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def subgrad_polish(h, n, target, n_steps, K=4):
    dx = 2.0 / n
    c5_local = compute_c5(h)
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = topk_subgrad(corr, h, K, n)
        lr = 0.05 / (1.0 + it * 0.1)
        h_cand = project_feasible(h - lr * g, target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < c5_local:
            c5_local = c5_cand
            h = h_cand
        else:
            lr *= 0.5
            h_cand = project_feasible(h - lr * g, target)
            h = h_cand
    return h, c5_local

def round_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    order = np.argsort(h)[::-1]
    k_full = int(target)
    h_bin[order[:k_full]] = 1.0
    if k_full < n:
        h_bin[order[k_full]] = target - k_full
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t0

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = (np.arange(100) % 2 == 0).astype(float)

    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Continuation SLSQP with decreasing T
    schedule = [(200, 200), (100, 300), (50, 400), (25, 500),
                (15, 600), (10, 800), (7, 1000), (5, 1200),
                (3, 1500), (2, 2000), (1.5, 2500)]
    for T, mit in schedule:
        if elapsed() > budget_s - 200:
            break
        h = slsqp_phase(h, n, target, bounds, T, mit)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 2: Subgradient polish with top-K
    h = h_best.copy()
    for K_val in [6, 4, 3, 2]:
        if elapsed() > budget_s - 100:
            break
        h, c5_local = subgrad_polish(h, n, target, 200, K=K_val)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()

    # Phase 3: More SLSQP at very low T from best
    if elapsed() < budget_s - 80:
        for T in [1.2, 1.0, 0.8]:
            if elapsed() > budget_s - 50:
                break
            h = slsqp_phase(h_best, n, target, bounds, T, 3000)
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()

    # Phase 4: Binary rounding
    h_bin = round_binary(h_best, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()

    # Phase 5: Gradient-guided perturbations
    for it in range(500):
        if elapsed() > budget_s - 10:
            break
        if it % 50 == 0:
            scale = 0.01 + 0.03 * (it // 50) * 0.5
        else:
            scale = 0.02
        noise = rng.normal(0, scale, n)
        h_p = project_feasible(h_best + noise, target)
        c5_p = compute_c5(h_p)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()
            # Local SLSQP polish on this new point
            if elapsed() < budget_s - 60:
                h = slsqp_phase(h_best, n, target, bounds, 3, 500)
                c5_p2 = compute_c5(h)
                if c5_p2 < c5_best:
                    c5_best = c5_p2
                    h_best = h.copy()

    # Phase 6: Try lower resolution if time permits
    if elapsed() < budget_s - 60 and n > 140:
        n_lo = n // 2
        target_lo = n_lo / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_new, x_old, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        c5_lo = compute_c5(h_lo)
        if c5_lo < c5_best:
            c5_best = c5_lo
            h_best = h_lo.copy()
        # Optimize at lower res
        bounds_lo = [(0.0, 1.0)] * n_lo
        for T in [50, 20, 10, 5, 3]:
            if elapsed() > budget_s - 20:
                break
            h_lo = slsqp_phase(h_lo, n_lo, target_lo, bounds_lo, T, 500)
            c5_lo = compute_c5(h_lo)
            if c5_lo < c5_best:
                c5_best = c5_lo
                h_best = h_lo.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
