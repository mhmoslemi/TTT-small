    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-x[ks[v2] + s])
    return _G_DX * grad

    return _G_DX * (c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau))))
