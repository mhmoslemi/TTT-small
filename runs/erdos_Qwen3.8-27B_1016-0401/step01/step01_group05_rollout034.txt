```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_sweep_descent(h, n, target, deadline):
    """Full coordinate sweep: for each i, try multiple values, pick best."""
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            old_val = h[i]
            best_local = old_val
            best_c5_local = best_c5
            # Try a range of values
            candidates = np.linspace(0, 1, 21)
            for v in candidates:
                if abs(v - old_val) < 1e-6:
                    continue
                h[i] = v
                h = make_feasible(h, n, target)
                c5 = compute_c5(h)
                if c5 < best_c5_local - 1e-12:
                    best_c5_local = c5
                    best_local = v
            if best_local != old_val:

                h = make_feasible(h, n, target)

    return h, best_c5
```

```python
def soft_max_gradient(h, a, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Compute weights
    a_corr = a * corr
    a_corr -= np.max(a_corr)  # numerical stability
    w = np.exp(a_corr)
    w /= np.sum(w)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        L = k - (n - 1)
        if L == 0:
            g += w[k] * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += w[k] * (1.0 - h[:n-L])
            g[:n-L] -= w[k] * h[L:]
        else:
            m = -L
            g[:n-m] += w[k] * (1.0 - h[m:])
            g[m:] -= w[k] * h[:n-m]
    return g * dx
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = h < (1.0 - 1e-12) if diff > 0 else h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_gradient(h, a):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    ac = a * corr
    ac -= ac.max()
    w = np.exp(ac)
    w /= w.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n-L])

            m = -L
            g[:n-m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n-m]
    return g * dx

def pair_local_search(h, n, target, rng, deadline, n_trials=50000):

        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        delta = rng.uniform(-step, step)
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h[i] += delta
        h[j] -= delta
        c5 = compute_c5(h)
        if c5 > best_c5:
            h[i] -= delta
            h[j] += delta
        else:
            best_c5 = c5
        if it % 20000 == 10000:
            step *= 0.7
    return h, best_c5

def coord_sweep(h, n, target, deadline):
    """Sweep each coordinate to its best value."""
    h = h.copy()
    best_c5 = compute_c5(h)
    vals = np.concatenate([[0.0, 1.0], np.linspace(0.02, 0.98, 24)])
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            best_c5_i = best_c5
            for v in vals:
                if abs(v - old) < 1e-6:
                    continue
                h[i] = v
                h_tmp = make_feasible(h, n, target)
                c5 = compute_c5(h_tmp)
                if c5 < best_c5_i - 1e-10:
                    best_c5_i = c5
                    best_v = v
                    h = h_tmp.copy()
            # Restore h[i] and apply best
            if abs(best_v - old) > 1e-6:
                h[i] = best_v
                h = make_feasible(h, n, target)
                best_c5 = best_c5_i
                improved = True
            else:
                h[i] = old
                h = make_feasible(h, n, target)
    return h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def make_feasible(h, target):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_gradient(h, a):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    ac = a * corr - a * corr.max()
    w = np.exp(ac)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            m = -L
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n - m]
    return g * dx
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def make_feasible(h, target):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_gradient(h, a):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    ac = a * (corr - corr.max())
    w = np.exp(np.clip(ac, -60, 0))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            m = -L
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n - m]
    return g * dx

def grad_descent(h, target, deadline, a0=120.0, a1=45.0):
    h = make_feasible(h, target)
    n = len(h)
    best_c5 = compute_c5(h)
    alpha = 1.0
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(1.0, it / 4000.0)
        a = a0 + (a1 - a0) * frac
        g = soft_max_gradient(h, a)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = g / gn
        ok = False
        at = alpha
        for _ in range(50):
            h_new = np.clip(h - at * d, 0.0, 1.0)
            h_new = make_feasible(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-15:
                h, best_c5 = h_new, c5
                ok = True
                alpha = min(5.0, at * 1.1)
                break
            at *= 0.5
            if at < 1e-9:
                break
        if ok:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                rng2 = np.random.RandomState(1000 + it)
                h = make_feasible(h + rng2.randn(n) * 0.006, target)
                best_c5 = compute_c5(h)
                alpha = 0.3
                stall = 0
            else:
                alpha *= 0.6
                if alpha < 1e-8:
                    stall = 61
    return h, best_c5

def pair_search(h, target, rng, deadline, nmax=100000):
    h = h.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    step = 0.1
    for it in range(nmax):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        delta = rng.uniform(-step, step)
        if not (0.0 <= h[i] + delta <= 1.0 and 0.0 <= h[j] - delta <= 1.0):
            continue
        h[i] += delta
        h[j] -= delta
        c5 = compute_c5(h)
        if c5 > best_c5:
            h[i] -= delta
            h[j] += delta
        else:
            best_c5 = c5
        if it % 12000 == 6000:
            step *= 0.7
    return h, best_c5

def interp_to(h_src, n_new):
    x_s = np.linspace(0.0, 2.0, len(h_src))
    x_n = np.linspace(0.0, 2.0, n_new)
    return np.clip(np.interp(x_n, x_s, h_src), 0.0, 1.0)

def cosine_start(n, freq):
    i = np.arange(n, dtype=float)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * i / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(70) * 0.5
    n = len(h0)
    target = n / 2.0
    h0 = make_feasible(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient descent from parent
    d = min(deadline - 250, t0 + 280)
    h1, c1 = grad_descent(h0, target, d, a0=150, a1=50)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Pair search
    d = min(deadline - 200, d + 130)
    h2, c2 = pair_search(best_h, target, rng, d, 70000)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: Gradient descent round 2
    d = min(deadline - 150, d + 180)
    h3, c3 = grad_descent(best_h, target, d, a0=180, a1=60)
    if c3 < best_c5:
        best_h, best_c5 = h3, c3

    # Phase 4: Pair search round 2
    d = min(deadline - 80, d + 100)
    h4, c4 = pair_search(best_h, target, rng, d, 50000)
    if c4 < best_c5:
        best_h, best_c5 = h4, c4

    # Phase 5: Try alternative starting points and resolutions
    alts = []
    for na in [56, 84, 100, 126]:
        alts.append((interp_to(best_h, na), na / 2.0))
    # Cosine starts at parent resolution
    for fq in [1, 2, 3]:
        alts.append((make_feasible(cosine_start(n, fq), target), target))
    # Perturbed parent
    alts.append((make_feasible(h0 + rng.randn(n) * 0.05, target), target))

    for ha, ta in alts:
        if time.time() > deadline - 40:
            break
        d = min(deadline - 25, time.time() + 60)
        ha, ca = grad_descent(ha, ta, d, a0=130, a1=50)
        d = min(deadline - 15, d + 40)
        ha, ca = pair_search(ha, ta, rng, d, 30000)
        if ca < best_c5:
            best_h, best_c5 = ha, ca

    # Phase 6: Final polish on best solution
    nf = len(best_h)
    tf = nf / 2.0
    d = deadline - 10
    best_h, best_c5 = grad_descent(best_h, tf, d, a0=250, a1=100)
    best_h, best_c5 = pair_search(best_h, tf, rng, d, 25000)

    best_h = make_feasible(best_h, nf / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, nf
```
