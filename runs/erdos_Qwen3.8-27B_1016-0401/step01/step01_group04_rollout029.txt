    ones = int(h_b.sum())

        order = np.argsort(h[one_idx])
        for i in range(ones - target_ones):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(arr, T):
    mx = arr.max()
    if T < 1e-6:
        return mx
    shifted = np.clip(T * (arr - mx), 0.0, 50.0)
    return mx + np.log1p(np.sum(np.exp(shifted) - 1.0)) / T

def slsqp_opt(h0, n, T, maxiter=500, ftol=1e-14):
    S = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(c, T)
    
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x, 0.0, 1.0)
        # Fix sum
        diff = S - h_new.sum()
        if abs(diff) > 1e-10:
            h_new += diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    except:
        return h0.copy()

def to_binary(h, S):
    t = int(S)
    h_b = (h >= 0.5).astype(float)
    cnt = int(h_b.sum())
    if cnt > t:
        idx = np.where(h_b == 1)[0]
        ord_ = np.argsort(h[idx])
        for i in range(cnt - t):
            h_b[idx[ord_[i]]] = 0
    elif cnt < t:
        idx = np.where(h_b == 0)[0]
        ord_ = np.argsort(-h[idx])
        for i in range(t - cnt):
            h_b[idx[ord_[i]]] = 1
    return h_b

def sa_binary(h0, n, S, rng, n_iter, t_limit, best_c5):

    c5 = compute_c5(h)

    best_c = min(c5, best_c5)
    temp = 0.004
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        if it % 5000 == 0 and it > 0:
            temp *= 0.97
        # Choose move type
        r = rng.rand()
        if r < 0.4:  # single swap
            i, j = rng.randint(n, size=2)
            if h[i] != h[j]:
                h[i], h[j] = h[j], h[i]
        elif r < 0.7:  # block swap (length 2-4)
            L = rng.randint(2, 5)
            i = rng.randint(n - L + 1)
            j = rng.randint(n - L + 1)
            if abs(i - j) < L:
                continue
            # Check that blocks are valid (one has ones, other has zeros)
            blk_i = h[i:i+L].copy()
            blk_j = h[j:j+L].copy()
            h[i:i+L] = blk_j
            h[j:j+L] = blk_i
        else:
            continue
        
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-12)):
            c5 = c5_new
            if c5 < best_c:
                best_c = c5
                best = h.copy()
        else:
            # Undo
            if r < 0.4:
                h[i], h[j] = h[j], h[i]
            else:
                h[i:i+L] = blk_i
                h[j:j+L] = blk_j
    return best, best_c

def hill_climb(h0, n, S, best_c5, max_sweeps=100, t_limit=None):
    h = h0.copy()
    c5 = compute_c5(h)
    best = h.copy()
    best_c = c5
    for sweep in range(max_sweeps):
        if t_limit and time.time() > t_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if t_limit and time.time() > t_limit:
                break
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best, best_c
