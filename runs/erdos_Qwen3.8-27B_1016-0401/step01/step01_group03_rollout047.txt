```python
def swap_pair_move(h, n, rng):
    """Swap a random 1 with a random 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    i = rng.integers(len(ones))
    j = rng.integers(len(zeros))
    h[ones[i]] = 0
```

```python
def coord_sweep(h, n):
    """One sweep of coordinate descent."""
    for j in range(n):
        old = h[j]
        # Compute alpha_k and beta_k for each lag k
        # f_k = sum_i h[i]*(1-h[i+k-(n-1)])
        # Terms with h[j]: 
        #   i=j: h[j]*(1-h[j+k-n+1])
        #   i=j-k+n-1: h[i]*(1-h[j])
        alphas = np.zeros(2*n-1)
        betas = np.zeros(2*n-1)
        corr = np.correlate(h, 1.0-h, mode='full')
        for k in range(2*n-1):
            d = k - (n-1)
            # alpha: coefficient of h[j] in f_k
            alpha = 0.0
            if 0 <= j + d < n:
                alpha += (1.0 - h[j+d])
            if 0 <= j - d < n:
                alpha -= h[j-d]
            # beta: f_k with h[j] set to 0
            # = corr[k] - alpha * old
            alphas[k] = alpha
            betas[k] = corr[k] - alpha * old
        
        # Solve: min_{x in [0,1]} max_k (alphas[k]*x + betas[k])
        # Use scipy minimize_scalar
        from scipy.optimize import minimize_scalar
        def subf(x):
            return np.max(alphas * x + betas)
        res = minimize_scalar(subf, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
        h[j] = res.x
    return h
```

For max of lines, the minimum is at the point where the upper envelope transitions from decreasing to increasing. This happens at the intersection of the two lines that form the "kink" at the minimum.

For n=46 and 91 lines, finding the exact kink requires computing the upper envelope, which is O(m log m) for m lines. But in practice, the max is usually achieved by just 2-3 lines, so we can approximate.

```python
def solve_1d_maxlines(alphas, betas):
    """Solve min_{x in [0,1]} max_k (alphas[k]*x + betas[k])."""
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid1 = lo + (hi - lo) / 3
        mid2 = hi - (hi - lo) / 3
        if np.max(alphas * mid1 + betas) <= np.max(alphas * mid2 + betas):
            hi = mid2
        else:
            lo = mid1
    return (lo + hi) / 2
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def solve_1d_maxlines(alphas, betas):
    lo, hi = 0.0, 1.0
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(alphas * m1 + betas) <= np.max(alphas * m2 + betas):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def coord_sweep(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    two_n_minus_1 = 2 * n - 1
    for j in range(n):
        old = h[j]
        if old < 1e-14 or old > 1.0 - 1e-14:
            pass
        alphas = np.zeros(two_n_minus_1)
        for k in range(two_n_minus_1):
            d = k - (n - 1)
            a = 0.0
            if 0 <= j + d < n:
                a += (1.0 - h[j + d])
            if 0 <= j - d < n:
                a -= h[j - d]
            alphas[k] = a
        betas = corr - alphas * old
        h[j] = solve_1d_maxlines(alphas, betas)
    return h


def coord_descend(h0, n, t_limit):
    dx = 2.0 / n
    ts = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stall = 0
    for it in range(5000000):
        if time.time() > t_limit:
            break
        h = coord_sweep(h, n)
        h = project_feasible(h, ts)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
    return best_h, best_c5


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_c = np.exp(beta * (corr - m))
    s = np.sum(exp_c)
    obj = dx * (m + np.log(s) / beta)
    w = dx * exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        for i in range(n):
            v = 0.0
            if 0 <= i + d < n:
                v += (1.0 - h[i + d])
            if 0 <= i - d < n:
                v -= h[i - d]
            grad[i] += wk * v
    return obj, grad


def _cons_fun(x, ts):
    return np.sum(x) - ts


def slsqp_optimize(h0, dx, ts, t_limit):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': lambda x: np.ones(len(x))}]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [200.0, 1000.0, 5000.0]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta), jac=True,
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5
```
