    dx = 2.0 / n
    steps = [0.012, 0.006, 0.003, 0.0015, 0.0008]
    no_improve = 0
    while time.time() < deadline:
        corr = compute_corr(best_h)
        top_k = np.argsort(corr)[-4:][::-1]
        improved = False
        for k_idx in top_k:
            s = k_idx - (n - 1)
            grad = np.zeros(n)
            for m in range(n):
                if 0 <= m - s < n:
                    grad[m] += (1.0 - best_h[m - s])
                if 0 <= m + s < n:
                    grad[m] -= best_h[m + s]
            for d in steps:
                pairs_found = 0
                for i in range(n):
                    if pairs_found >= 3:
                        break
                    for j in range(i + 1, n):
                        if pairs_found >= 3:
                            break
                        gi, gj = grad[i], grad[j]
                        if gi > gj + 1e-5:
                            hi, hj = best_h[i] + d, best_h[j] - d
                        elif gj > gi + 1e-5:
                            hi, hj = best_h[i] - d, best_h[j] + d
                        else:
                            continue
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h_try = best_h.copy()
                            h_try[i] = hi
                            h_try[j] = hj
                            c5_try = compute_c5(h_try)
                            if c5_try < best_c5 - 1e-12:
                                best_h = h_try
                                best_c5 = c5_try
                                improved = True
                                pairs_found += 1
                                break
                if not improved:
                    break
            if improved:
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > 30:
            break
    return best_h, best_c5

def sa_optimize(h, n, target, rng, deadline, T0=0.002):
    """Simulated annealing with 2-3 coordinate block moves."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = T0
    T_min = 5e-6
    cooling = 0.99985
    amp = 0.025
    stall = 0
    while time.time() < deadline:
        block = 2 if rng.random() < 0.75 else 3
        idx = rng.choice(n, block, replace=False)
        deltas = rng.normal(0, amp, block)
        h_try = cur_h.copy()
        h_try[idx] += deltas
        h_try = make_feasible(h_try, n, target)
        new_c5 = compute_c5(h_try)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = h_try, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_try.copy(), new_c5
                stall = 0
            else:
                stall += 1
        elif rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, T_min)):
            cur_h, cur_c5 = h_try, new_c5
            stall += 1
        else:
            stall += 1
        if stall > 8000:
            cur_h = best_h.copy()
            cur_c5 = best_c5
            stall = 0
            amp *= 0.75
        T = max(T * cooling, T_min)
    return best_h, best_c5

def fine_coord_descent(h, n, target, deadline, step=0.003):
    """Fine coordinate descent from both directions."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for d in (step, -step):
                val = best_h[i] + d
                if 0 <= val <= 1:
                    h_try = best_h.copy()
                    h_try[i] = val
                    h_try = make_feasible(h_try, n, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-11:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
        if not improved:
            step *= 0.5
            if step < 5e-5:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 4

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = float(n) / 2.0
    h_init = make_feasible(h_init, n, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup (45s)
    for alpha in [40, 120, 400, 1000]:
        if time.time() > t0 + 45:
            break
        h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=250, ftol=1e-13)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Directed pair descent (120s)
    if time.time() < t0 + 165:
        h_p, c5_p = directed_pair_descent(best_h, n, target, t0 + 165)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    # Phase 3: SLSQP refine (60s)
    if time.time() < t0 + 225:
        for alpha in [600, 1500, 3000]:
            if time.time() > t0 + 225:
                break
            h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=200, ftol=1e-14)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 4: Simulated annealing (350s)
    if time.time() < t0 + 575:
        h_sa, c5_sa = sa_optimize(best_h, n, target, rng, t0 + 575, T0=0.0018)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa

    # Phase 5: SLSQP + pair descent from SA result (100s)
    if time.time() < t0 + 675:
        h_new = slsqp_optimize(best_h, n, target, 2000, maxiter=200, ftol=1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        if time.time() < t0 + 675:
            h_p2, c5_p2 = directed_pair_descent(best_h, n, target, t0 + 675)
            if c5_p2 < best_c5:
                best_h, best_c5 = h_p2, c5_sa if c5_sa < c5_p2 else c5_p2
                best_h = h_p2

    # Phase 6: Alternative starts (150s)
    for trial in range(3):
        if time.time() > t0 + 825:
            break
        if trial == 0:
            h_alt = (rng.random(n) < 0.5).astype(float)
            h_alt = make_feasible(h_alt, n, target)
        elif trial == 1:
            h_alt = np.linspace(1.0, 0.0, n)
            h_alt += rng.normal(0, 0.05, n)
            h_alt = make_feasible(h_alt, n, target)
        else:
            h_alt = rng.uniform(0.2, 0.8, n)
            h_alt = make_feasible(h_alt, n, target)
        for alpha in [60, 300, 1200]:
            if time.time() > t0 + 825:
                break
            h_alt = slsqp_optimize(h_alt, n, target, alpha, maxiter=150)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt, c5_alt

    # Phase 7: Final fine coordinate descent (80s)
    if time.time() < deadline - 20:
        h_fd, c5_fd = fine_coord_descent(best_h, n, target, deadline - 20, step=0.004)
        if c5_fd < best_c5:
            best_h, best_c5 = h_fd, c5_fd

    # Phase 8: Final SLSQP polish
    if time.time() < deadline - 3:
        h_new = slsqp_optimize(best_h, n, target, 4000, maxiter=150, ftol=1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
