- h[i] appears in c_k as: h[i]*(1-h[i+δ_k]) - h[i-δ_k]*h[i] + ... 
- Actually it's: terms where i is the "h" position: h[i]*(1-h[i+δ_k])
- Terms where i is the "1-h" position: h[i-δ_k]*(1-h[i]) = h[i-δ_k] - h[i-δ_k]*h[i]

            new_c5 = compute_c5(h.astype(float), dx)
            dE = new_c5 - cur_c5
            if dE < 0 or np.random.random() < np.exp(-dE / T):
                cur_c5 = new_c5
                # Update lists: a moved from zeros to ones, b from ones to zeros
                # Find b in ones_list, replace with a
                # Find a in zeros_list, replace with b
                # This is O(n) but n=46 so fine
                i_b = ones_list.index(b)
                i_a = zeros_list.index(a)
                ones_list[i_b] = a
                zeros_list[i_a] = b
                if new_c5 < best_c5 - 1e-14:
                    best = h.copy()
                    best_c5 = new_c5
                    no_impr = 0
                else:
                    no_impr += 1
            else:
                h[a], h[b] = 0, 1
                no_impr += 1
            T *= 0.99998
            if T < 1e-9:
                T = 1e-9
            if no_impr > 80000:
                break
    return best.astype(float), best_c5

     0 ≤ h_i ≤ 1

The problem is c_k = sum_i h_i(1-h_{i+δ_k}) which is bilinear (not linear). So it's not a LP.

This is still not linear in h (the product h_i * h_{i+δ_k} appears with a minus sign: c_k = sum h_i - sum h_i * h_{i+δ_k}).

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ec = np.exp(beta * (corr - mc))
    se = np.sum(ec)
    obj = dx * (mc + np.log(se) / beta)
    w = dx * ec / se
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        d = m - (n - 1)
        if d >= 0:
            idx = np.arange(n - d)
            grad[idx] += wm * (1.0 - h[idx + d])
            grad[idx + d] -= wm * h[idx]
        else:
            p = -d
            idx = np.arange(n - p)
            grad[idx + p] += wm * (1.0 - h[idx])
            grad[idx] -= wm * h[idx + p]
    return obj, grad

def project_feasible(h, target):
    lo, hi = -0.5, 1.5
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(p)
    if abs(diff) > 1e-12:
        if diff > 0:
            m = p < 1.0 - 1e-10
        else:
            m = p > 1e-10
        cnt = int(m.sum())
        if cnt > 0:
            p[m] += diff / cnt
            p = np.clip(p, 0.0, 1.0)
    return p

def gd_optimize(h0, dx, ts, t_limit, beta0=40.0):
    n = len(h0)
    h = project_feasible(h0.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    beta = beta0
    lr = 0.01
    mom = 0.85
    v = np.zeros(n)
    stall = 0
    for it in range(20000000):
        if time.time() > t_limit:
            break
        f, g = lse_grad(h, dx, beta)
        v = mom * v - lr * g
        h_new = project_feasible(h + v, ts)
        f_new, _ = lse_grad(h_new, dx, beta)
        if f_new > f + 1e-14:
            lr *= 0.6
            if lr < 1e-11:
                lr = 0.01
                beta = min(beta * 4.0, 50000.0)
                v = np.zeros(n)
            h = project_feasible(h + v, ts)
        else:
            h = h_new
            c5 = compute_c5(h, dx)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.5)
            else:
                stall += 1
        if stall > 3000:
            beta = min(beta * 3.0, 50000.0)
            lr = 0.005
            stall = 0
            v = np.zeros(n)
    return best_h, best_c5

def sa_binary(h0, dx, t_limit):
    n = len(h0)
    ts = n // 2
    h = (h0 > 0.5).astype(np.int8).copy()
    oc = int(h.sum())
    if oc > ts:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:oc - ts]] = 0
    elif oc < ts:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:ts - oc]] = 1
    hf = h.astype(float)
    best = h.copy()
    best_c5 = compute_c5(hf, dx)
    for restart in range(30):
        if time.time() > t_limit:

        h = best.copy()
        hf = h.astype(float)
        ones_list = np.where(h == 1)[0].tolist()
        zeros_list = np.where(h == 0)[0].tolist()
        T = best_c5 * 0.03 + 0.003
        cur_c5 = best_c5
        no_impr = 0
        for it in range(5000000):
            if time.time() > t_limit:

            a = zeros_list[np.random.randint(len(zeros_list))]
            b = ones_list[np.random.randint(len(ones_list))]

def coord_refine(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0.copy(), ts)
    best_c5 = compute_c5(h, dx)
    for _outer in range(200):

            break
        improved = False
        for i in range(n):
            if time.time() > t_limit:

            old = h[i]

                if abs(trial - old) < 1e-6:

                h[i] = trial

                if c5 < best_c5 - 1e-13:
                    h = h_p
                    best_c5 = c5
                    improved = True
                    break
            if not improved and h[i] != old:
