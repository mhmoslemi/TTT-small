```python
def project_onto_feasible(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase: add to elements not at 1
            mask = h < 1.0
            if not mask.any():
                break
            add = diff / mask.sum()
            h[mask] = np.clip(h[mask] + add, 0, 1)
        else:
            # Need to decrease: subtract from elements not at 0
            mask = h > 0.0
            if not mask.any():
                break
            sub = -diff / mask.sum()
            h[mask] = np.clip(h[mask] - sub, 0, 1)
    return h
```

```python
        return mc * (2.0 / n)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-11:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def proj_gradient(h, n, target, T, max_iter=3000):
    best_h = h.copy()
    best_f = soft_max_obj(h, n, T)
    lr = 0.02
    stuck = 0
    for it in range(max_iter):
        grad = soft_max_grad(h, n, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        f_curr = soft_max_obj(h, n, T)
        lr_e = lr
        for _ in range(30):
            h_trial = project_feasible(h - lr_e * grad, target)
            f_trial = soft_max_obj(h_trial, n, T)
            if f_trial < f_curr - 0.5 * lr_e * np.dot(grad, h_trial - h):
                break
            lr_e *= 0.5
        else:
            break
        h = h_trial
        lr = max(lr_e, 0.002)
        f_now = soft_max_obj(h, n, T)
        if f_now < best_f - 1e-15:
            best_f = f_now
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                lr = lr * 2.0 + 0.005
                stuck = 0
    return best_h, best_f

def subgrad_polish(h, n, target, steps=500):
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = 0.05
    for it in range(steps):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr)
        o = idx - (n - 1)
        g = np.zeros(n)
        if o == 0:
            g = 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] -= h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
        h_trial = project_feasible(h - lr * g, target)
        h = h_trial
        c = compute_c5(h)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        lr *= 0.995
    return best_h, best_c

def sim_anneal(h, n, target, T_sa, steps=200, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_f = soft_max_obj(h, n, T_sa)
    h_cur = h.copy()
    f_cur = best_f
    for _ in range(steps):
        mask = rng.random(n) < 0.3
        delta = rng.normal(0, 0.3, n) * mask
        h_trial = project_feasible(h_cur + delta, target)
        f_trial = soft_max_obj(h_trial, n, T_sa)
        if f_trial < f_cur or rng.random() < np.exp(-(f_trial - f_cur) / (T_sa * 0.01)):
            h_cur = h_trial
            f_cur = f_trial
            if f_trial < best_f:
                best_f = f_trial
                best_h = h_trial.copy()
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Projected gradient at fine temperature (n=280)
    for T in [200, 100, 50, 30, 20, 15]:
        if elapsed() > budget_s - 300:
            break
        h_val, f = proj_gradient(h_best, n, target, T, max_iter=2000)
        c = compute_c5(h_val)
        if c < c5_best:
            c5_best = c
            h_best = h_val.copy()

    # Phase 2: Simulated annealing for diversity
    for T_sa in [50, 30, 15, 8, 4]:
        if elapsed() > budget_s - 250:
            break
        h_val, _ = sim_anneal(h_best, n, target, T_sa, steps=300, rng=rng)
        h_val, f = proj_gradient(h_val, n, target, 20, max_iter=1000)
        c = compute_c5(h_val)
        if c < c5_best:
            c5_best = c
            h_best = h_val.copy()

    # Phase 3: Subgradient polishing at current resolution
    if elapsed() < budget_s - 200:
        h_val, c = subgrad_polish(h_best, n, target, steps=800)
        if c < c5_best:
            c5_best = c
            h_best = h_val.copy()

    # Phase 4: Multi-resolution exploration
    for n_try in [140, 70]:
        if elapsed() > budget_s - 150:
            break
        target_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_coarse = np.interp(x_new, x_old, h_best)
        h_coarse = project_feasible(h_coarse, target_try)
        for T in [30, 15, 8]:
            if elapsed() > budget_s - 100:
                break
            h_val, _ = proj_gradient(h_coarse, n_try, target_try, T, max_iter=1500)
            x_up = np.linspace(0, 2.0, n, endpoint=False)
            h_up = np.interp(x_up, x_new, h_val)
            h_up = project_feasible(h_up, target)
            c = compute_c5(h_up)
            if c < c5_best:
                c5_best = c
                h_best = h_up.copy()

    # Phase 5: Random restarts at full resolution
    n_restarts = 20
    for r in range(n_restarts):
        if elapsed() > budget_s - 80:
            break
        if r % 4 == 0:
            h_init = (rng.random(n) >= 0.5).astype(float)
        elif r % 4 == 1:
            h_init = np.zeros(n)
            h_init[::2] = 1.0
            h_init = np.roll(h_init, int(rng.integers(-5, 6)))
        elif r % 4 == 2:
            h_init = h_best + rng.normal(0, 0.15, n)
        else:
            h_init = (rng.random(n) > 0.45).astype(float)
        h_init = project_feasible(h_init, target)
        T = [40, 20, 10, 8][r % 4]
        h_val, _ = proj_gradient(h_init, n, target, T, max_iter=800)
        h_val, c = subgrad_polish(h_val, n, target, steps=200)
        if c < c5_best:
            c5_best = c
            h_best = h_val.copy()

    # Phase 6: Final fine-tuning with very low temperature
    if elapsed() < budget_s - 20:
        for T in [10, 6, 4, 3]:
            if elapsed() > budget_s - 15:
                break
            h_val, _ = proj_gradient(h_best, n, target, T, max_iter=1500)
            c = compute_c5(h_val)
            if c < c5_best:
                c5_best = c
                h_best = h_val.copy()

    # Final subgradient polish
    if elapsed() < budget_s - 10:
        h_best, c5_best = subgrad_polish(h_best, n, target, steps=500)

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
```
