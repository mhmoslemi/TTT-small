def grad_corr(h, corr, sw, n):
    """Compute gradient of weighted sum of correlations."""
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    return grad * dx

    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add: add to smallest values first
            idx = np.argsort(h)
            for i in idx:
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
                if abs(diff) < 1e-12:
                    break
        else:
            # Need to subtract: subtract from largest values first
            idx = np.argsort(h)[::-1]
            for i in idx:
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
                if abs(diff) < 1e-12:
