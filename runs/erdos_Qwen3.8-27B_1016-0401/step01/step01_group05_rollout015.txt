import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    h = h.copy().astype(float)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        cnt = np.count_nonzero(mask)
        h[mask] += diff / cnt
    return h

def block_descent(h, target, n, block_sizes, n_passes, deadline):
    """Cyclic block coordinate descent."""
    best_c5 = compute_c5(h)
    for _ in range(n_passes):
        improved = False
        for bs in block_sizes:
            if time.time() > deadline:
                return h, best_c5
            n_starts = n - bs + 1
            for start in range(0, n_starts, 2):
                if time.time() > deadline:
                    return h, best_c5
                idx = np.arange(start, start + bs)
                x0 = h[idx].copy()
                h_base = h.copy()
                
                # Use SLSQP for small blocks
                def obj(p):
                    h_t = h_base.copy()
                    h_t[idx] = np.clip(p, 0, 1)
                    h_t = project_feasible(h_t, target)
                    return compute_c5(h_t)
                
                try:
                    res = minimize(obj, x0, method='L-BFGS-B',
                                  bounds=[(0,1)]*bs,
                                  options={'maxiter': 30, 'gtol': 1e-10})
                    if res.fun < best_c5 - 1e-12:
                        h[idx] = np.clip(res.x, 0, 1)
                        h = project_feasible(h, target)
                        best_c5 = compute_c5(h)
                        improved = True
                except Exception:
                    pass
    return h, best_c5

def subgradient_descent(h, target, n, deadline):
    """Subgradient method on the max correlation."""
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    step = 0.1
    for iteration in range(5000):
        if time.time() > deadline:
            break
        corr = compute_corr(h_curr)
        c5 = np.max(corr)
        # Find active lags (within 1% of max)
        threshold = c5 * 0.99
        active = np.where(corr >= threshold)[0]
        
        # Compute subgradient (average of gradients at active lags)
        grad = np.zeros(n)
        for k_full in active:
            k = k_full - (n - 1)  # actual lag
            for i in range(n):
                # d/d h[i] of sum_j h[j]*(1-h[j+k])
                # Term 1: h[i] appears as h[j] when j=i, contributes (1-h[i+k])
                j1 = i + k
                if 0 <= j1 < n:
                    grad[i] += (1.0 - h_curr[j1])
                # Term 2: h[i] appears as h[j+k] when j+k=i, i.e., j=i-k, contributes -h[j]
                j2 = i - k
                if 0 <= j2 < n:
                    grad[i] += -h_curr[j2]
        grad /= len(active)
        
        # Step
        h_new = h_curr - step * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            h_curr = h_new.copy()
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.95
            if step < 0.001:
                step = 0.001
    
    return h_curr, best_c5

def slsqp_soft_max(h, target, n, alpha, maxiter):
    """SLSQP with soft-max surrogate."""
    dx = 2.0 / n
    
    def obj(x):
        x = np.clip(x, 0, 1)
        # Enforce sum constraint via penalty
        sum_pen = (np.sum(x) - target) ** 2 * 100
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = alpha * (corr - m)
        vals = np.clip(vals, -50, 50)
        smax = m + np.log(np.sum(np.exp(vals))) / alpha
        return smax * dx + sum_pen
    
    def grad(x):
        # Numerical gradient
        g = np.zeros(n)
        eps = 1e-7
        for i in range(n):
            xp = x.copy(); xp[i] += eps
            xm = x.copy(); xm[i] -= eps
            g[i] = (obj(xp) - obj(xm)) / (2*eps)
        return g
    
    h_new = h.copy()
    # Use SLSQP
    h_clipped = np.clip(h_new, 0, 1)
    
    def obj_slsqp(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = alpha * (corr - m)
        vals = np.clip(vals, -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    
    def sum_con(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj_slsqp, h_clipped, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type': 'eq', 'fun': sum_con}],
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_feasible(res.x, target)
    except Exception:
        h_new = project_feasible(h, target)
    
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    h = h.copy().astype(float)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        cnt = np.count_nonzero(mask)
        h[mask] += diff / cnt
    return h

def block_descent(h, target, n, block_sizes, n_passes, deadline):
    best_c5 = compute_c5(h)
    for _ in range(n_passes):
        for bs in block_sizes:
            if time.time() > deadline:
                break
            n_starts = max(1, n - bs + 1)
            step_starts = max(1, n_starts // 25)
            for start in range(0, n_starts, step_starts):
                if time.time() > deadline:
                    break
                idx = np.arange(start, start + bs)
                x0 = h[idx].copy()
                h_base = h.copy()
                dx = 2.0 / n
                def obj_block(p):
                    h_t = h_base.copy()
                    h_t[idx] = np.clip(p, 0.0, 1.0)
                    h_t = project_feasible(h_t, target)
                    corr = np.correlate(h_t, 1.0 - h_t, mode='full')
                    return float(np.max(corr)) * dx
                try:
                    res = minimize(obj_block, x0, method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * bs,
                                  options={'maxiter': 40, 'gtol': 1e-10, 'maxls': 10})
                    if res.fun < best_c5 - 1e-13:
                        h[idx] = np.clip(res.x, 0.0, 1.0)
                        h = project_feasible(h, target)
                        best_c5 = compute_c5(h)
                except Exception:
                    pass
    return h, best_c5

def subgradient_descent(h, target, n, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    step = 0.08
    patience = 0
    for iteration in range(20000):
        if time.time() > deadline:
            break
        corr = compute_corr(h_curr)
        c5_val = np.max(corr)
        threshold = c5_val * 0.995
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for m in active:
            k = int(m) - (n - 1)
            for i in range(n):
                j_minus = i - k
                if 0 <= j_minus < n:
                    grad[i] += (1.0 - h_curr[j_minus])
                j_plus = i + k
                if 0 <= j_plus < n:
                    grad[i] += (-h_curr[j_plus])
        grad /= float(len(active))
        h_new = h_curr - step * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            h_curr = h_new
            patience = 0
            step = min(step * 1.03, 0.25)
        else:
            patience += 1
            step = max(step * 0.97, 0.005)
            if patience > 5000:
                break
    return h_curr, best_c5

def slsqp_soft_max(h, target, n, alpha, maxiter):
    dx = 2.0 / n
    def obj_slsqp(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = alpha * (corr - m)
        vals = np.clip(vals, -60.0, 60.0)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def sum_con(x):
        return np.sum(x) - target
    h_clip = np.clip(h, 0.0, 1.0)
    try:
        res = minimize(obj_slsqp, h_clip, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con}],
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_feasible(res.x, target)
    except Exception:
        h_new = project_feasible(h, target)
    return h_new

def local_search(h, target, n, rng, max_iters, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    step = 0.06
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 3000 == 0 and it > 0:
            step *= 0.85
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        delta = rng.uniform(-step, step)
        h_try = h_curr.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_curr = h_try
            best_c5 = c5_try
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10000:
                break
    return h_curr, best_c5

def simulated_annealing_blocks(h, target, n, rng, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    T = 0.02
    T_min = 0.001
    block_size = 4
    n_evals = 0
    for it in range(50000):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            T = max(T * 0.95, T_min)
        start = rng.randint(0, max(1, n - block_size + 1))
        idx = np.arange(start, start + block_size)
        h_try = h_curr.copy()
        h_try[idx] += rng.uniform(-T * 3, T * 3, block_size)
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        delta_c = c5_try - best_c5
        if delta_c < 0 or rng.uniform() < np.exp(-delta_c / T):
            h_curr = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
        n_evals += 1
    return h_curr, best_c5

def try_alternative_start(n, rng, target):
    h = np.zeros(n)
    pattern = rng.permutation(n)
    h[pattern[:n//2]] = 1.0
    h += rng.uniform(-0.1, 0.1, n)
    h = project_feasible(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(70) * 0.5
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP soft-max (quick initial polish)
    for alpha in [50, 200, 600]:
        if time.time() > deadline - budget_s * 0.5:
            break
        h_new = slsqp_soft_max(best_h, target, n, alpha, 80)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Block coordinate descent
    dl2 = min(deadline, t0 + budget_s * 0.3)
    best_h, best_c5 = block_descent(best_h, target, n, [4, 6, 8], 4, dl2)

    # Phase 3: Subgradient descent
    dl3 = min(deadline, t0 + budget_s * 0.45)
    h_sg, c5_sg = subgradient_descent(best_h, target, n, dl3)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # Phase 4: Simulated annealing on blocks
    dl4 = min(deadline, t0 + budget_s * 0.65)
    h_sa, c5_sa = simulated_annealing_blocks(best_h, target, n, rng, dl4)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # Phase 5: SLSQP refinement
    for alpha in [400, 1000, 3000]:
        if time.time() > deadline - 100:
            break
        h_new = slsqp_soft_max(best_h, target, n, alpha, 120)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Local search polish
    dl6 = min(deadline - 10, t0 + budget_s * 0.9)
    best_h, best_c5 = local_search(best_h, target, n, rng, 40000, dl6)

    # Phase 7: Final block descent
    if time.time() < deadline - 10:
        best_h, best_c5 = block_descent(best_h, target, n, [3, 5], 2, deadline - 10)
        best_h, best_c5 = local_search(best_h, target, n, rng, 10000, deadline - 3)

    # Phase 8: Try alternative starting point if time remains
    if time.time() < deadline - 40:
        h_alt = try_alternative_start(n, rng, target)
        for alpha in [30, 100, 400]:
            if time.time() > deadline - 20:

            h_alt = slsqp_soft_max(h_alt, target, n, alpha, 80)
        h_alt, c5_alt = block_descent(h_alt, target, n, [4, 6], 2, deadline - 10)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt, c5_alt

    # Final validation
    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
