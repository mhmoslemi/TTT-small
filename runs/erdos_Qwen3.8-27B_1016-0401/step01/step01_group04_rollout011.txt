def sa_swaps(h, n, rng, n_iter, deadline):
    h_sa = h.copy()
    mc = float(np.correlate(h_sa, 1.0 - h_sa, mode='full').max())
    best_h = h_sa.copy(); best_mc = mc
    ol = np.where(h_sa == 1)[0].tolist()
    zl = np.where(h_sa == 0)[0].tolist()
    temp = 2.5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it > 0 and it % 4000 == 0:
            temp *= 0.975
        a = ol[rng.randint(len(ol))]
        b = zl[rng.randint(len(zl))]
        h_sa[a] = 0.0; h_sa[b] = 1.0
        nm = float(np.correlate(h_sa, 1.0 - h_sa, mode='full').max())
        d = nm - mc
        if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-10)):
            mc = nm
            ol[a] = b; zl[b] = a
            if nm < best_mc:
                best_mc = nm; best_h = h_sa.copy()
        else:
            h_sa[a] = 1.0; h_sa[b] = 0.0
    return best_h, best_mc

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    shifted = np.clip(T * (corr - mx), -50, 50)
    return mx + np.log(np.sum(np.exp(shifted))) / T

def slsqp_refine(h, n, S, T, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - S},)
    def obj(x):
        return smooth_max_obj(x, T)
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = np.clip(res.x, 0, 1)
        x += (S - x.sum()) / n
        x = np.clip(x, 0, 1)
        return x
    except Exception:
        return h.copy()

def normalize_h(h, S):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 1e-15:
        h = h * (S / s)
    h = np.clip(h, 0, 1)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.97

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.zeros(70); h0[rng.choice(70, 35, replace=False)] = 1.0

    n = len(h0)
    S = n // 2
    h_bin = to_binary(h0, S)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()

    # Phase 1: Aggressive binary optimization on initial n
    h_g, mc_g = greedy_swaps(h_bin, n, max_rounds=300)
    c5_g = mc_g * 2.0 / n
    if c5_g < best_c5:
        best_c5 = c5_g; best_h = h_g.copy()

    dl_sa = t0 + budget_s * 0.40
    h_sa, mc_sa = sa_swaps(h_g, n, rng, 800000, dl_sa)
    c5_sa = mc_sa * 2.0 / n
    if c5_sa < best_c5:
        best_c5 = c5_sa; best_h = h_sa.copy()

    h_g2, mc_g2 = greedy_swaps(best_h, n, max_rounds=200)
    c5_g2 = mc_g2 * 2.0 / n
    if c5_g2 < best_c5:
        best_c5 = c5_g2; best_h = h_g2.copy()

    # Extra SA from best
    dl_sa2 = t0 + budget_s * 0.55
    h_sa2, mc_sa2 = sa_swaps(best_h, n, rng, 400000, dl_sa2)
    c5_sa2 = mc_sa2 * 2.0 / n
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2; best_h = h_sa2.copy()
    h_g3, mc_g3 = greedy_swaps(best_h, n, max_rounds=100)
    c5_g3 = mc_g3 * 2.0 / n
    if c5_g3 < best_c5:
        best_c5 = c5_g3; best_h = h_g3.copy()

    # Phase 2: Continuous SLSQP refinement
    dl_slsqp = t0 + budget_s * 0.75
    h_cont = best_h.astype(float) + rng.normal(0, 0.05, n)
    h_cont = normalize_h(h_cont, S)
    for T in [15, 80, 400, 2000, 8000]:
        if time.time() > dl_slsqp:
            break
        h_new = slsqp_refine(h_cont, n, S, T, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new; best_h = h_new.copy()
        h_cont = h_new

    # Phase 3: Binary conversion of continuous result
    if time.time() < t0 + budget_s * 0.80:
        h_b3 = to_binary(best_h, S)
        c5_b3 = compute_c5(h_b3)
        if c5_b3 < best_c5:
            best_c5 = c5_b3; best_h = h_b3.copy()
        h_g4, mc_g4 = greedy_swaps(h_b3, n, max_rounds=150)
        c5_g4 = mc_g4 * 2.0 / n
        if c5_g4 < best_c5:
            best_c5 = c5_g4; best_h = h_g4.copy()

    # Phase 4: Multi-restart and different n
    dl_restart = t0 + budget_s * 0.93
    for n_try in [72, 68, 74, 66]:
        if time.time() > dl_restart:
            break
        S_try = n_try // 2
        for restart in range(4):
            if time.time() > dl_restart:
                break
            h_r = np.zeros(n_try)
            h_r[rng.choice(n_try, S_try, replace=False)] = 1.0
            h_rg, mc_rg = greedy_swaps(h_r, n_try, max_rounds=80)
            c5_rg = mc_rg * 2.0 / n_try
            if c5_rg < best_c5:
                best_c5 = c5_rg; best_h = h_rg.copy()
            h_rs, mc_rs = sa_swaps(h_rg, n_try, rng, 100000, min(dl_restart, t0 + budget_s * 0.93))
            c5_rs = mc_rs * 2.0 / n_try
            if c5_rs < best_c5:
                best_c5 = c5_rs; best_h = h_rs.copy()
            h_rg2, mc_rg2 = greedy_swaps(best_h if best_h.shape[0] == n_try else h_rs,
                                         n_try if best_h.shape[0] == n_try else len(h_rs),
                                         max_rounds=50)

    # Final: one more greedy on best
    if time.time() < deadline:
        n_final = len(best_h)
        h_gf, mc_gf = greedy_swaps(best_h, n_final, max_rounds=100)
        c5_gf = mc_gf * 2.0 / n_final
        if c5_gf < best_c5:
            best_c5 = c5_gf; best_h = h_gf.copy()

    best_h = np.clip(best_h, 0, 1)
    # Ensure exact sum
    S_final = len(best_h) // 2
    if best_h.sum() != S_final:
        best_h = to_binary(best_h, S_final)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
