def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_array(h):
    return np.correlate(h, 1 - h, mode="full")

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)
        remaining = diff
        for i in order:
            if remaining > 0:
                s = min(remaining, 1.0 - h[i])
            else:
                s = min(-remaining, h[i])
            h[i] += s
            remaining = target - h.sum()
            if abs(remaining) < 1e-14:
                break
    return h

def sm_obj(h, n, T):
    corr = compute_corr_array(h)
    mc = np.max(corr)
    s = T * (corr - mc)
    ms = np.max(s)
    val = mc + (ms + np.log(np.sum(np.exp(s - ms)))) / T
    return val * (2.0 / n)

def sm_grad(h, n, T):
    corr = compute_corr_array(h)
    mc = np.max(corr)
    s = T * (corr - mc)
    ms = np.max(s)
    e = np.exp(s - ms)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    return grad * dx

    """Coordinate descent: try setting each h[i] to 0, 1, or keep."""

            # Try 0 and 1

                if c5 < best_c5 - 1e-12:

                    old_val = v
                h[i] = old_val
            h[i] = old_val
            if old_val != h[i]:
                improved = True
        if not improved:
            break
    return h
