def project_onto_simplex_box(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_step(h, target, alpha, maxiter, deadline):
    n = len(h)
    pen = 500.0
    def obj(x):
        return soft_max_c5(x, alpha) + pen * (np.sum(x) - target) ** 2
    def obj_grad(x):
        f = soft_max_c5(x, alpha) + pen * (np.sum(x) - target) ** 2
        # Numerical gradient is too slow; use finite diff via L-BFGS-B internal
        return None
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    if time.time() > deadline:
        return h
    return project_onto_simplex_box(res.x, target)

def subgrad_descent(h, target, n_iters, deadline, lr0=0.03):
    h = h.copy()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        g = subgrad_active(h, topk=3, alpha=30)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        lr = lr0 / (1.0 + 0.0005 * it)
        h = h - lr * g
        h = project_onto_simplex_box(h, target)
    return h

def multi_coord_search(h, target, rng, n_iters, deadline, n_coords=6, step0=0.08):
    h = h.copy()
    best = compute_c5(h)
    step = step0
    n = len(h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        idx = rng.choice(n, size=n_coords, replace=False)
        delta = rng.uniform(-step, step, size=n_coords)
        delta -= delta.sum() / n_coords
        h_try = h.copy()
        h_try[idx] += delta
        h_try = project_onto_simplex_box(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best - 1e-12:
            h = h_try
            best = c5_try
            step = min(step * 1.05, 0.15)
        else:
            step *= 0.995
    return h, best

def anneal_polish(h, target, rng, deadline):
    """Simulated-annealing-style multi-coordinate moves"""
    h = h.copy()
    best = compute_c5(h)
    n = len(h)
    temp = 0.05
    for it in range(5000):
        if time.time() > deadline:
            break
        temp *= 0.999
        nc = rng.randint(2, 11)
        idx = rng.choice(n, size=nc, replace=False)
        delta = rng.randn(nc) * 0.04
        delta -= delta.sum() / nc
        h_try = h.copy()
        h_try[idx] += delta
        h_try = project_onto_simplex_box(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best or (rng.random() < np.exp(-(c5_try - best) / max(temp, 1e-10))):
            h = h_try
            if c5_try < best:
                best = c5_try
    return h, best

def optimize_at_n(h_init, n, target, deadline, rng):
    h = project_onto_simplex_box(h_init, target)
    c5 = compute_c5(h)
    # Phase 1: L-BFGS-B with increasing alpha
    for alpha in [30, 80, 200, 500]:
        rem = deadline - time.time()
        if rem < 15:
            break
        h2 = lbfgsb_step(h, target, alpha, maxiter=150, deadline=deadline - 10)
        h2 = project_onto_simplex_box(h2, target)
        c52 = compute_c5(h2)
        if c52 < c5:
            h = h2
            c5 = c52
    # Phase 2: Subgradient on soft-active set
    if time.time() < deadline - 40:
        h2 = subgrad_descent(h, target, 2000, deadline - 30, lr0=0.02)
        c52 = compute_c5(h2)
        if c52 < c5:
            h = h2
            c5 = c52
    # Phase 3: Multi-coordinate local search
    if time.time() < deadline - 25:
        h, c5 = multi_coord_search(h, target, rng, 4000, deadline - 15, n_coords=6)
    # Phase 4: Anneal polish
    if time.time() < deadline - 10:
        h2, c52 = anneal_polish(h, target, rng, deadline - 8)
        if c52 < c5:
            h = h2
            c5 = c52
    # Phase 5: Final L-BFGS-B polish
    if time.time() < deadline - 8:
        h2 = lbfgsb_step(h, target, 1000, maxiter=80, deadline=deadline - 5)
        h2 = project_onto_simplex_box(h2, target)
        c52 = compute_c5(h2)
        if c52 < c5:
            h = h2
            c5 = c52
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(70) * 0.5

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Try multiple resolutions
    for n in [100, 140, 200]:
        if time.time() > deadline - 120:
            break
        target = n / 2.0
        # Interpolate
        x_old = np.linspace(0, 2, len(h_parent))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_old, h_parent)
        # Also try best_h if it's different
        if len(best_h) != len(h_parent):
            x_best = np.linspace(0, 2, len(best_h))
            h_interp2 = np.interp(x_new, x_best, best_h)
        else:
            h_interp2 = h_interp.copy()

        # Optimize from interpolated parent
        h1, c51 = optimize_at_n(h_interp, n, target, deadline - 30, rng)
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1.copy()

        # Optimize from interpolated best (if different source)
        if time.time() < deadline - 60:
            h2, c52 = optimize_at_n(h_interp2, n, target, deadline - 20, rng)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()

    # Try a structured starting point: piecewise constant with 10 blocks
    if time.time() < deadline - 80:
        n = 140
        target = n / 2.0
        # Create a block structure: alternate high/low in a non-trivial pattern
        block_size = n // 10
        h_struct = np.zeros(n)
        pattern = [0.8, 0.3, 0.7, 0.2, 0.9, 0.1, 0.6, 0.4, 0.85, 0.25]
        for bi in range(10):
            start = bi * block_size
            end = start + block_size if bi < 9 else n
            h_struct[start:end] = pattern[bi]
        h_struct = project_onto_simplex_box(h_struct, target)
        h3, c53 = optimize_at_n(h_struct, n, target, deadline - 15, rng)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()

    # Final: polish the best solution at its own resolution one more time
    n_best = len(best_h)
    target_best = n_best / 2.0
    if time.time() < deadline - 20:
        h_fin, c5_fin = multi_coord_search(best_h, target_best, rng, 3000, deadline - 10, n_coords=5)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Validate
    best_h = project_onto_simplex_box(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h.tolist(), best_c5, len(best_h)
