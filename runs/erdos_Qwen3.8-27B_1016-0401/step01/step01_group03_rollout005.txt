def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    s = np.sum(np.exp(beta * (corr - max_c)))
    return dx * (max_c + np.log(s) / beta), corr

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c

    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:

            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.exp(0.0) if sum_exp == 0 else np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            grad[m:] += wi * (1.0 - h[:n - m])
            grad[:n - m] -= wi * h[m:]
        else:
            p = -m
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)


def _cons_fun(x, ts):
    return np.sum(x) - ts


def _cons_jac(x):
    return np.ones(len(x))


def optimize_smooth(h0, dx, ts, t_limit, betas=None):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    if betas is None:
        betas = [500.0, 3000.0, 15000.0]
    for beta in betas:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    # Adam projected gradient
    beta = betas[-1]
    lr = 0.5
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    stall = 0
    for it in range(100000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        m_buf = 0.9 * m_buf + 0.1 * grad
        v_buf = 0.999 * v_buf + 0.001 * grad * grad
        step = lr * m_buf / (np.sqrt(v_buf) + 1e-8)
        h_new = project_feasible(h - step, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 8.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.93
        h = h_new
        if stall > 3000 or lr < 1e-9:
            break
    return best_h, best_c5


def block_swap_sa(h0, dx, t_limit, max_block=6, T_start=0.05):
    n = len(h0)
    ts = n / 2.0
    best = h0.copy()
    best_c5 = compute_c5(best, dx)
    hb = (best > 0.5).astype(float)
    oc = int(np.sum(hb))
    if oc > int(ts):
        idxs = np.where(hb == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - int(ts)]:
            hb[i] = 0.0
    elif oc < int(ts):
        idxs = np.where(hb == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:int(ts) - oc]:
            hb[i] = 1.0
    if compute_c5(hb, dx) < best_c5:
        best = hb.copy()
        best_c5 = compute_c5(best, dx)
    T = T_start
    for it in range(80000):
        if time.time() > t_limit:
            break
        h = best.copy()
        L = np.random.randint(1, max_block + 1)
        p = np.random.randint(0, n - L)
        for _ in range(10):
            q = np.random.randint(0, n - L)
            if q < p + L or q + L <= p:
                break
        h[p:p + L], h[q:q + L] = h[q:q + L].copy(), h[p:p + L].copy()
        h = project_feasible(h, ts)
        c5_new = compute_c5(h, dx)
        if c5_new < best_c5 or np.random.random() < np.exp(-min(100.0, (c5_new - best_c5) / max(T, 1e-8))):
            best = h
            best_c5 = c5_new
        T = max(1e-5, T * 0.9995)
    return best, best_c5


def make_init(n, method='fourier'):
    if method == 'sinusoidal':
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * x / 2 + np.random.uniform(0, 6.28))
    elif method == 'multi_sin':
        x = np.linspace(0, 2, n)
        h = np.zeros(n)
        for k in range(1, 5):
            h += np.random.uniform(0.2, 0.5) * np.sin(2 * np.pi * k * x / 2 + np.random.uniform(0, 6.28))
        h = 0.5 + 0.3 * h
    elif method == 'fourier':
        h = np.zeros(n)
        for k in range(1, min(n // 2, 30)):
            h += (np.random.uniform(-1, 1) / k) * np.cos(2 * np.pi * k * np.arange(n) / n + np.random.uniform(0, 6.28))
        h = 0.5 + 0.4 * h
    else:
        h = np.random.uniform(0, 1, n)
    return project_feasible(h, n / 2.0)
