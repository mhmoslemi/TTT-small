    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj < 1.0) & (proj > 0.0)
        if mask.sum() == 0:
            mask = (proj < 1.0) if diff > 0 else (proj > 0.0)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)
