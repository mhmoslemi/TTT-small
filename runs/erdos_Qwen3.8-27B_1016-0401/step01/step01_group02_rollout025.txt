def compute_grad(h, w):
    """Compute gradient of smooth max of overlaps.
    h: array of length n
    w: weights for each shift (length 2n-1), should sum to 1
    """
    n = len(h)
    grad = np.zeros(n)
    # For each shift s (from -(n-1) to n-1), weight w[s+n-1]:
    for k in range(2*n-1):
        s = k - (n-1)
        wk = w[k]
        if wk < 1e-15:
            continue
        # overlap[s] = sum_i h[i]*(1-h[i+s]) for valid i
        # d/dh[i] = (1-h[i+s])*[0<=i+s<n] - h[i-s]*[0<=i-s<n]
        if s >= 0:
            # i+s valid for i in [0, n-s-1]
            # (1-h[i+s]) term for i in [0, n-s-1]
            grad[:n-s] += wk * (1 - h[s:])
            # h[i-s] term for i in [s, n-1], i.e., h[i-s] for i-s in [0, n-s-1]
            grad[s:] -= wk * h[:n-s]
        else:
            ss = -s  # positive
            # i+s valid for i in [ss, n-1]
            # (1-h[i+s]) term for i in [ss, n-1]
            grad[ss:] += wk * (1 - h[:n-ss])
            # h[i-s] = h[i+ss] term for i in [0, n-ss-1]
            grad[:n-ss] -= wk * h[ss:]
    return grad

import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_c5(h, scale):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    dx = 2.0 / n
    return (m + np.log(np.exp(scale * (corr - m)).sum()) / scale) * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        cnt = msk.sum()
        if cnt > 0:
            h[msk] += r / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_smooth(h, scale):
    """Gradient of smooth max of correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exps = np.exp(scale * (corr - m))
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            valid_n = n - s
            if valid_n > 0:
                grad[:valid_n] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:valid_n]
        else:
            ss = -s
            valid_n = n - ss
            if valid_n > 0:
                grad[ss:] += wk * (1.0 - h[:valid_n])
                grad[:valid_n] -= wk * h[ss:]
    return grad

def _grad_descent(h0, n, n_iter, scale, lr_init, tmax):
    """Projected gradient descent with momentum."""
    t0 = time.time()
    h = h0.copy()
    v = np.zeros(n)

    best = h0.copy()
    bestc = _c5(best)
    
    for it in range(n_iter):
        if time.time() - t0 > tmax:
            break
        g = _grad_smooth(h, scale)
        v = beta * v + (1 - beta) * g
        h_new = h - lr * v
        h_new = _proj(h_new, n)
        c_new = _c5(h_new)
        if c_new < bestc:
            best, bestc = h_new.copy(), c_new
        h = h_new
        # Decay learning rate
        lr = lr_init * (1.0 / (1.0 + 0.01 * it))
    
    return best, bestc

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        cnt = msk.sum()
        if cnt > 0:
            h[msk] += r / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_smooth(h, scale):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exps = np.exp(scale * (corr - m))
    ss = exps.sum()
    if ss < 1e-300:
        return np.zeros(n)
    w = exps / ss
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        sv = k - (n - 1)
        j1a = max(0, sv)
        j1b = min(n, sv + n)
        if j1a < j1b:
            grad[j1a:j1b] += wk * (1.0 - h[j1a - sv:j1b - sv])
        j2a = max(0, -sv)
        j2b = min(n, n - sv)
        if j2a < j2b:
            grad[j2a:j2b] -= wk * h[j2a + sv:j2b + sv]
    return grad

def _gd(h0, n, iters, scale, lr, tmax):
    t0 = time.time()
    h = h0.copy()
    v = np.zeros(n)
    mom = 0.9
    best = h0.copy()
    bc = _c5(best)
    stall = 0
    for it in range(iters):
        if time.time() - t0 > tmax:
            break
        g = _grad_smooth(h, scale)
        v = mom * v + g
        h_new = _proj(h - lr * v, n)
        c = _c5(h_new)
        if c < bc - 1e-13:
            best = h_new.copy()
            bc = c
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 40:
            h = _proj(h + 0.02 * np.random.randn(n), n)
            v = np.zeros(n)
            stall = 0
        lr *= 0.9995
    return best, bc

def _local_g(h0, n, tmax, nit):
    t0 = time.time()
    tgt = 0.5 * n
    h = h0.copy()
    best = h0.copy()
    bc = _c5(best)
    g = _grad_smooth(h, 100.0)
    idx = np.argsort(g)
    q = max(4, n // 8)
    for it in range(nit):
        if time.time() - t0 > tmax:
            break
        if it % 200 == 0:
            g = _grad_smooth(h, 100.0)
            idx = np.argsort(g)
        i = idx[np.random.randint(q)]
        j = idx[np.random.randint(n - q, n)]
        if i == j:

        dv = np.random.uniform(0.002, 0.06)
        h[i] = np.clip(h[i] + dv, 0.0, 1.0)
        h[j] = np.clip(h[j] - dv, 0.0, 1.0)
        ds = h.sum() - tgt
        if abs(ds) > 1e-12:

        c = _c5(h)

            bc = c
        else:

    return best, bc

def _local_r(h0, n, tmax, nit, step_lo, step_hi):
    t0 = time.time()
    tgt = 0.5 * n
    h = h0.copy()
    best = h0.copy()

        h[i] = np.clip(h[i] + dv, 0.0, 1.0)
        h[j] = np.clip(h[j] - dv, 0.0, 1.0)

def _upscale(h, factor=2):

    nn = n * factor
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    hn = np.interp(xn, xo, h)
    return _proj(hn, nn)
