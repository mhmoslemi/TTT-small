import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_box_sum(x, t):
    x = np.clip(x, 0.0, 1.0).copy()
    for _ in range(60):
        diff = t - x.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-12
            if not mask.any():
                break
            x[mask] = np.minimum(x[mask] + diff / mask.sum(), 1.0)
        else:
            mask = x > 1e-12
            if not mask.any():
                break
            x[mask] = np.maximum(x[mask] + diff / mask.sum(), 0.0)
    return np.clip(x, 0.0, 1.0)

def adam_steps(h, T, n_steps, lr, m_arr, v_arr, target, start_t=1, check_interval=500):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_local = compute_c5(h)
    bhs = h.copy()
    for it in range(n_steps):
        g = compute_grad(h, T)
        t = start_t + it
        m_arr[:] = b1 * m_arr + (1 - b1) * g
        v_arr[:] = b2 * v_arr + (1 - b2) * g * g
        mh = m_arr / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        h = project_box_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
        if (it + 1) % check_interval == 0:
            c5 = compute_c5(h)
            if c5 < best_local:
                best_local = c5
                bhs = h.copy()
    return bhs, best_local

def binary_round(h, target):
    b = (h >= 0.5).astype(float)
    diff = int(round(target - b.sum()))
    if diff > 0:
        zeros = np.where(b == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for i in range(min(diff, len(zeros))):
            b[zeros[i]] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for i in range(min(-diff, len(ones))):
            b[ones[i]] = 0.0
    return b

def binary_local_search(h, max_rounds=80, budget_fn=None):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng(12345)
    c5 = compute_c5(h)
    h = h.copy()
    for r in range(max_rounds):
        if budget_fn and budget_fn():
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_tries = min(5000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            nc = compute_c5(h)
            if nc < c5 - 1e-14:
                c5 = nc
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, c5
