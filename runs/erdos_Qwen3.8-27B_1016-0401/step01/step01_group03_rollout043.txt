def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj < 1.0 - 1e-10) if diff > 0 else (proj > 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -50.0 / max(beta, 1.0), 0.0)
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            grad[m:n] += wi * (1.0 - h[:n - m])
            grad[:n - m] -= wi * h[m:]
        else:
            p = -m
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad

            best_h = h_start.copy()
        t_sub = min(t3, time.time() + max(8, (t3 - time.time()) * 0.35))
        h_r, c5_r = optimize_smooth(h_start, dx_use, ts_use, t_sub)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_use = best_n
            dx_use = 2.0 / n_use
            ts_use = n_use / 2.0

    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
