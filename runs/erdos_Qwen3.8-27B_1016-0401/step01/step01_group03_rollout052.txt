Let me also think about the problem structure. The objective is:
min_h max_k sum_i h[i](1-h[i+k_mod])

For the "full" correlation (non-circular), at shift k:
- If k ≥ 0: R_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
- If k < 0: R_k = sum_{i=-k}^{n-1} h[i](1-h[i+k])

The number of terms decreases with |k|. So the max is likely achieved at small |k| values (where there are more terms).

Key observation: At shift 0, R_0 = sum h[i](1-h[i]) = sum h[i] - sum h[i]² = n/2 - sum h[i]².

To minimize R_0, we want to maximize sum h[i]², which means making h as "peaked" as possible (binary). But this conflicts with minimizing R_k for other k.

For a uniform h = 0.5 everywhere (which satisfies sum = n/2):
R_k = (n-|k|) * 0.25 for all k
Max R = (n-1) * 0.25
C₅ = (n-1) * 0.25 * 2/n = (n-1)/(2n) ≈ 0.5 for large n

So uniform is bad (C₅ ≈ 0.5). We need to structure h to reduce the max overlap.

For binary h: R_0 = 0, but R_k for small k can be large.

The optimal is somewhere in between: h takes values that balance the overlaps across all shifts.

Let me think about a "water-filling" approach:
- We want all R_k to be as equal as possible (at their minimum possible value)
- This is like: minimize max_k R_k subject to sum h = n/2, 0 ≤ h ≤ 1

This is a constrained optimization problem. The KKT conditions would give us the optimal h.

For the KKT: at optimum, either h[i] = 0, h[i] = 1, or the gradient condition holds:

where lambda_k ≥ 0 and lambda_k > 0 only for k at the "active" shifts (where R_k = max R).

This is complex to solve analytically but gives insight: the optimal h is determined by the subgradient of the max at the active shifts.

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(beta * shifted)
    sum_exp = np.sum(exps)
    obj = dx * (cmax + np.log(sum_exp) / beta)
    w = dx * exps / sum_exp
    grad = np.zeros(n)
    for m in range(n):
        wp = w[n - 1 + m]
        if wp > 1e-18:
            j = np.arange(n - m)
            grad[j] += wp * (1.0 - h[j + m])
            grad[j + m] -= wp * h[j]
        if m > 0:
            wm = w[n - 1 - m]
            if wm > 1e-18:
                j = np.arange(m, n)
                grad[j] += wm * (1.0 - h[j - m])
                grad[j - m] -= wm * h[j]
    return obj, grad

def smooth_opt(h0, dx, ts, t_limit, rng):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    betas = [30.0, 100.0, 300.0, 1000.0, 3000.0, 10000.0]
    for beta in betas:
        if time.time() > t_limit:
            break
        lr = 0.5
        momentum = 0.8
        v = np.zeros(n)
        stall = 0
        while time.time() < t_limit:
            _, grad = smooth_obj_grad(h, dx, beta)
            v = momentum * v - lr * grad
            h_new = project_feasible(h + v, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                lr = min(lr * 1.03, 5.0)
                stall = 0
            else:
                stall += 1
                lr *= 0.94
            h = h_new
            if stall > 300 or lr < 1e-9:
                break
        h = best_h.copy()
    
    return best_h, best_c5

def binary_1swap(b, dx, t_limit, rng):

    best_c5 = compute_c5(b, dx)
    improved = True
    while improved and time.time() < t_limit:

        zeros = np.where(b == 0)[0]
        ones = np.where(b == 1)[0]

            if time.time() > t_limit:
                return b, best_c5
            for oi in ones:
                b[zi] = 1.0
                b[oi] = 0.0
                c5_new = compute_c5(b, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                b[zi] = 0.0
                b[oi] = 1.0
            if improved:
                break
    return b, best_c5

def simulated_anneal(b, dx, t_limit, rng):
    n = len(b)
    best_b = b.copy()
    best_c5 = compute_c5(b, dx)
    curr = b.copy()
    curr_c5 = best_c5
    start = time.time()
    total = t_limit
    while time.time() < start + total:
        progress = (time.time() - start) / total
        T = 0.3 * (0.001) ** progress
        zeros = np.where(curr == 0)[0]
        ones = np.where(curr == 1)[0]
        zi = rng.choice(zeros)
        oi = rng.choice(ones)
        curr[zi] = 1.0
        curr[oi] = 0.0
        new_c5 = compute_c5(curr, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_b = curr.copy()
        else:
            curr[zi] = 0.0
            curr[oi] = 1.0
    return best_b, best_c5

def to_binary(h, ts, rng):
    b = (h > 0.5).astype(float)
    oc = int(np.sum(b))
    if oc > ts:
        idxs = np.where(b == 1)[0]
        rng.shuffle(idxs)
        for i in idxs[:oc - ts]:
            b[i] = 0.0
    elif oc < ts:
        idxs = np.where(b == 0)[0]
        rng.shuffle(idxs)
        for i in idxs[:ts - oc]:
            b[i] = 1.0
    return b
