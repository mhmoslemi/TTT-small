import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - S):
            h_b[one_idx[order[i]]] = 0
    elif ones < S:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(S - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def hill_climb_best_swap(h, n, S, rng, t_deadline, max_sweeps=100000, patience=2):
    h_wc = h.copy()
    c5 = compute_c5(h_wc)
    best_h = h_wc.copy()
    best_c5 = c5
    
    consecutive_no_improve = 0
    restarts = 0
    max_restarts = 200

            break

        zero_idx = np.where(h_wc == 0)[0]
        
        best_delta = 0.0
        best_i, best_j = -1, -1
        
        for i in one_idx:
            h_wc[i] = 0.0
            for j in zero_idx:
                h_wc[j] = 1.0
                new_c5 = compute_c5(h_wc)
                delta = c5 - new_c5
                if delta > best_delta + 1e-15:
                    best_delta = delta
                    best_i, best_j = i, j
                h_wc[j] = 0.0
            h_wc[i] = 1.0

            consecutive_no_improve += 1
            if consecutive_no_improve > patience:
                # Perturbation
                n_flip = rng.randint(2, min(5, len(one_idx)))

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - S):
            h_b[one_idx[order[i]]] = 0
    elif ones < S:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(S - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def hill_climb_best(h, n, S, rng, t_deadline, max_sweeps=100000, patience=2, max_restarts=300):
    h_wc = h.copy()
    c5 = compute_c5(h_wc)
    best_h = h_wc.copy()
    best_c5 = c5
    no_improve = 0
    restarts = 0

    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h_wc == 1)[0]
        zero_idx = np.where(h_wc == 0)[0]
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in one_idx:
            h_wc[i] = 0.0
            for j in zero_idx:
                h_wc[j] = 1.0
                nc = compute_c5(h_wc)
                d = c5 - nc
                if d > best_delta + 1e-15:
                    best_delta = d
                    best_i, best_j = i, j
                h_wc[j] = 0.0
            h_wc[i] = 1.0
        if best_i < 0:
            no_improve += 1
            if no_improve > patience:
                nf = rng.randint(2, 5)
                op = np.where(h_wc == 1)[0]
                zp = np.where(h_wc == 0)[0]
                for _ in range(nf):
                    ii = op[rng.randint(len(op))]
                    jj = zp[rng.randint(len(zp))]
                    h_wc[ii], h_wc[jj] = 0.0, 1.0
                c5 = compute_c5(h_wc)
                no_improve = 0
                restarts += 1
                if restarts > max_restarts:
                    break
            continue
        h_wc[best_i] = 0.0
        h_wc[best_j] = 1.0
        c5 -= best_delta
        no_improve = 0
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_wc.copy()
    return best_h, best_c5

def double_swap_hill(h, n, S, rng, t_deadline, max_sweeps=50000):
    h_wc = h.copy()
    c5 = compute_c5(h_wc)
    best_h = h_wc.copy()
    best_c5 = c5
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h_wc == 1)[0]
        zero_idx = np.where(h_wc == 0)[0]
        best_delta = 0.0
        best_pair = None
        n1, n0 = len(one_idx), len(zero_idx)
        for a in range(min(n1, 20)):
            i1 = one_idx[a]
            i2 = one_idx[(a + 1) % n1]
            h_wc[i1] = 0.0
            h_wc[i2] = 0.0
            for b in range(min(n0, 20)):
                j1 = zero_idx[b]
                j2 = zero_idx[(b + 1) % n0]
                h_wc[j1] = 1.0
                h_wc[j2] = 1.0
                nc = compute_c5(h_wc)
                d = c5 - nc
                if d > best_delta + 1e-15:
                    best_delta = d
                    best_pair = (i1, i2, j1, j2)
                h_wc[j1] = 0.0
                h_wc[j2] = 0.0
            h_wc[i1] = 1.0
            h_wc[i2] = 1.0
        if best_pair is None:
            no_improve += 1
            if no_improve > 3:
                op = np.where(h_wc == 1)[0]
                zp = np.where(h_wc == 0)[0]
                for _ in range(3):
                    ii = op[rng.randint(len(op))]
                    jj = zp[rng.randint(len(zp))]
                    h_wc[ii], h_wc[jj] = 0.0, 1.0
                c5 = compute_c5(h_wc)
                no_improve = 0
                if no_improve > 300:
                    break
            continue
        i1, i2, j1, j2 = best_pair
        h_wc[i1] = 0.0
        h_wc[i2] = 0.0
        h_wc[j1] = 1.0
        h_wc[j2] = 1.0
        c5 -= best_delta
        no_improve = 0
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_wc.copy()
    return best_h, best_c5
