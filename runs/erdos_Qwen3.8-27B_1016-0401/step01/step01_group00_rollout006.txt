import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def softmax_obj(corr, dx, T):
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T
    return val * dx

def softmax_weights(corr, dx, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    return e / np.sum(e)

def compute_gradient(h, w, n):
    """Compute gradient of soft-max objective given weights w (length 2n-1)."""
    # w[k] corresponds to lag k-(n-1)
    # grad_i = sum_k w[k] * [(1-h[i+k])[valid] - h[i-k][valid]]
    center = n - 1
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - center
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            # h[i+k] valid for i in [0, n-k-1]
            # h[i] * (1-h[i+k]): term from h[i] is (1-h[i+k]), term from h[i+k] is -h[i]
            # So for position i: contribution is (1-h[i+k]) for i < n-k, minus h[i-k] for i >= k
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] += -wk * h[:n-k]
        else:
            p = -k
            # h[i+k] = h[i-p], valid for i >= p
            # h[i-k] = h[i+p], valid for i < n-p
            grad[p:] += wk * (1.0 - h[:n-p])
            grad[:n-p] += -wk * h[p:]
    return grad

import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def softmax_weights(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    return e / np.sum(e)

def compute_gradient(h, w, n):
    center = n - 1
    grad = np.zeros(n)
    wk0 = w[center]
    if wk0 > 1e-14:
        grad += wk0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[center + k]
        wm = w[center - k]
        ww = wp + wm
        if ww < 1e-14:
            continue
        grad[:n - k] += ww * (1.0 - h[k:])
        grad[k:] += -ww * h[:n - k]
    return grad

def project_simplex(h, target):
    lo = min(float(h.min()) - 1.0, -1.0)
    hi = max(float(h.max()) + 1.0, 1.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def adam_optimize(h, n, target, T, n_steps, lr, t_start):
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    for step in range(n_steps):
        st = t_start + step
        if step % 400 == 0 and step > 0:
            lr *= 0.7
        _, corr = compute_c5_full(h)
        w = softmax_weights(corr, T)
        grad = compute_gradient(h, w, n)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** (st + 1))
        v_hat = v / (1.0 - b2 ** (st + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_simplex(h, target)
    return h, t_start + n_steps

def topk_optimize(h, n, target, K, n_steps, lr):

    t_count = 0

        _, corr = compute_c5_full(h)
        top_idx = np.argsort(corr)[-K:]
        top_corr = corr[top_idx]
        softmax_w = np.exp(T * (corr - np.max(corr))) / np.sum(np.exp(T * (corr - np.max(corr))))
        w_boosted = softmax_w.copy()
        w_boosted[top_idx] *= 5.0
        w_boosted /= w_boosted.sum()
        grad = compute_gradient(h, w_boosted, n)
        t_count += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad

        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_simplex(h, target)
    return h
