    t_start = time.time()
    h = h0.copy()
    tgt = 0.5 * n
    best = h.copy()
    best_c = _c5(best)
    
    for s in range(nsteps):
        if time.time() - t_start > tmax:
            break
        T = T_start * (T_end / T_start) ** (s / max(nsteps-1, 1))
        
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
        # Softmax weights
        corr_max = corr.max()
        weights = np.exp(T * (corr - corr_max))
        weights /= weights.sum()
        
        # Compute gradient: sum_k weights[k] * dC_k/dh
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-10:
                continue
            shift = k_idx - (n - 1)
            # dC(shift)/dh[i] = (1-h[i+shift])_{valid} - h[i-shift]_{valid}
            if shift >= 0:

                g[:n-shift] += weights[k_idx] * (1.0 - h[shift:])
                # h[i-shift] valid for i >= shift
                g[shift:] -= weights[k_idx] * h[:n-shift]
            else:
                s2 = -shift
                # h[i+shift] = h[i-s2] valid for i >= s2

        if c_new < best_c:
            best, best_c = h_new.copy(), c_new
        
        h = h_new
    
    return best, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _smooth_obj(x, n, sc):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * (2.0 / n)

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            break
        h[free] -= d / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0 - 1e-12) if r > 0 else (h > 1e-12)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

_G = {}

def _sobj(x):
    return _smooth_obj(x, _G['n'], _G['sc'])

def _ceq(x):
    return x.sum() - _G['tgt']

def _ceqj(x):
    return np.ones(_G['n'])

def _slsqp(h0, n, sc, mi):
    _G['n'] = n
    _G['sc'] = sc
    _G['tgt'] = 0.5 * n
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_sobj, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': mi, 'ftol': 1e-16})
        return _proj(r.x, n)
    except:
        return _proj(h0, n)

def _grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(len(corr)):
        ws = w[idx]
        if ws < 1e-12:
            continue
        s = idx - (n - 1)
        if s >= 0:
            g[:n-s] += ws * (1.0 - h[s:])
            g[s:] -= ws * h[:n-s]
        else:
            s2 = -s
            g[s2:] += ws * (1.0 - h[:n-s2])
            g[:n-s2] -= ws * h[s2:]
    return g * (2.0 / n)

def _pgd(h0, n, T0, T1, nsteps, alpha0, tmax):
    h = h0.copy()
    best = h.copy()
    bc = _c5(best)
    tst = time.time()
    for s in range(nsteps):
        if time.time() - tst > tmax:
            break
        T = T0 * (T1 / T0) ** (s / max(nsteps - 1, 1))
        g = _grad(h, n, T)
        alpha = alpha0 * (1.0 / (1.0 + 0.01 * s))
        h_new = h - alpha * g
        h_new = _proj(h_new, n)
        c = _c5(h_new)
        if c < bc:
            best, bc = h_new.copy(), c
        h = h_new
    return best, bc

def _local_search(h0, n, tmax, nit):
    h = h0.copy()
    best = h.copy()
    bc = _c5(best)
    wc = bc
    t0 = time.time()
    tgt = 0.5 * n
    for it in range(nit):

        # Choose move type

        if r < 0.3:

            idx = int(corr.argmax())
            s = idx - (n - 1)
            lo = max(0, -s)
