import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_vals(h):
    return np.correlate(h, 1.0 - h, mode="full")

def lse_val(corr, T):
    mc = np.max(corr)
    s = T * (corr - mc)
    sm = np.max(s)
    return mc + (sm + np.log(np.sum(np.exp(s - sm)))) / T

def soft_obj_p(x, T, target, mu):
    n = len(x)
    return lse_val(corr_vals(x), T) * (2.0 / n) + mu * (x.sum() - target) ** 2

def soft_grad_p(x, T, target, mu):
    n = len(x)
    corr = corr_vals(x)
    mc = np.max(corr)
    s = T * (corr - mc)
    sm = np.max(s)
    e = np.exp(s - sm)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o] * dx
        if w < 1e-13:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    grad += 2.0 * mu * (x.sum() - target)
    return grad
