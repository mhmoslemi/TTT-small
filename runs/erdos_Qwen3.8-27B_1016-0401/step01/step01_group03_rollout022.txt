    return np.clip(proj, 0.0, 1.0)

def _cons_fun(x, ts):
    return np.sum(x) - ts

def _cons_jac(x):
    return np.ones(len(x))

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(np.clip(beta * shifted, -700, 700))
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

def optimize_continuous(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for beta in [200.0, 1000.0, 5000.0, 20000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    # Fine-tuning with projected gradient
    beta = 20000.0
    lr = 1.0
    stall = 0
    for it in range(100000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 20.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.95
        h = h_new
        if stall > 500 or lr < 1e-8:
            break
    
    return best_h, best_c5

def sa_binary(h0, dx, ts, t_limit, seed=42):
    n = len(h0)
    state = (h0 > 0.5).astype(float).copy()
    # Fix count
    oc = int(np.sum(state))

        perm = rng.permutation(len(idxs))
        for i in idxs[perm[:oc - ts_int]]:
            state[i] = 0.0
    elif oc < ts_int:
        idxs = np.where(state == 0)[0]
        perm = rng.permutation(len(idxs))
        for i in idxs[perm[:ts_int - oc]]:
            state[i] = 1.0
    
    cur_c5 = compute_c5(state, dx)
    best_state = state.copy()
    best_c5 = cur_c5
    
    T = 0.005
    T_min = 0.00001
    total_iters = max(1, int((t_limit - time.time()) / 0.001))  # rough estimate
    cool_rate = (T_min / T) ** (1.0 / max(total_iters, 1))
    
    iter_count = 0
    while time.time() < t_limit:
        iter_count += 1
        zeros = np.where(state == 0)[0]
        ones = np.where(state == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Choose move type: 70% single swap, 30% double swap
        if len(zeros) >= 2 and len(ones) >= 2 and rng.random() < 0.3:
            # Double swap
            zi1 = zeros[rng.integers(len(zeros))]
            zi2 = zeros[rng.integers(len(zeros))]
            oi1 = ones[rng.integers(len(ones))]
            oi2 = ones[rng.integers(len(ones))]
            if len(set([zi1, zi2, oi1, oi2])) < 4:
                T *= cool_rate
                continue
            saved = state.copy()
            state[zi1] = 1.0; state[zi2] = 1.0
            state[oi1] = 0.0; state[oi2] = 0.0
            new_c5 = compute_c5(state, dx)
        else:
            # Single swap
            saved = state.copy()
            zi = zeros[rng.integers(len(zeros))]
            oi = ones[rng.integers(len(ones))]
            state[zi] = 1.0; state[oi] = 0.0
            new_c5 = compute_c5(state, dx)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_state = state.copy()
        else:
            delta = new_c5 - cur_c5
            if delta < 1e-15:
                pass  # accept neutral
            elif rng.random() < np.exp(-delta / max(T, 1e-15)):
                cur_c5 = new_c5
            else:
                state = saved
        
        T *= cool_rate
    
    # Deterministic local improvement from best
    improved = True
    while improved and time.time() < t_limit:
        improved = False
        state = best_state.copy()
        zeros = np.where(state == 0)[0]
        ones = np.where(state == 1)[0]
        for zi in zeros:
            if time.time() > t_limit:
                break
            for oi in ones:
                state[zi] = 1.0; state[oi] = 0.0
                c5_new = compute_c5(state, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_state = state.copy()
                    improved = True
                    break
                state[zi] = 0.0; state[oi] = 1.0
            if improved:
                break
    
    return best_state, best_c5

        elif oc < ts_i:
            idxs = np.where(h_try == 0)[0]; p = rng_main.permutation(len(idxs))
            for i in idxs[p[:ts_i - oc]]: h_try[i] = 1.0
        t_sub = min(t3, time.time() + max(12, (t3 - time.time()) * 0.22))
        h_r, c5_r = sa_binary(h_try, dx_t, ts_t, t_sub, seed=seed + n_try * 7)
        if c5_r < best_c5:
            best_c5 = c5_r; best_h = h_r.copy(); best_n = n_try
    # Phase 4: final refinement (18%)
    t4 = t0 + budget_s * 0.98
    n_use = best_n; dx_use = 2.0 / n_use; ts_use = n_use / 2.0
    h_f, c5_f = sa_binary(best_h, dx_use, ts_use, t4, seed=seed + 9999)
    if c5_f < best_c5:
        best_c5 = c5_f; best_h = h_f.copy()
    if time.time() < t4 + 10:
        h_c, c5_c = optimize_continuous(best_h, dx_use, ts_use, time.time() + 12)
        if c5_c < best_c5:
            best_c5 = c5_c; best_h = h_c.copy()
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
