import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    obj = (mc + np.log(np.sum(e)) / T) * dx
    
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    return obj, grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add: add to entries with most room
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, room[i])
                h[i] += s
                diff -= s
        else:
            # Need to subtract: remove from entries with most value
            order = np.argsort(-h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

    """Local search: identify worst shift, perturb the contributing pair."""
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    
    for it in range(n_iter):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget:
                break
        corr = np.correlate(h_curr, 1 - h_curr, mode="full")
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        
        if p == 0:
            # Zero shift: F_0 = sum h(m)(1-h(m))
            # Reduce by pushing h toward 0 or 1
            vals = h_curr * (1 - h_curr)
            idx = np.argmax(vals)
            if h_curr[idx] > 0.5:
                h_curr[idx] = max(0, h_curr[idx] - 0.1)
            else:
                h_curr[idx] = min(1, h_curr[idx] + 0.1)
        else:
            if s > 0:
                contrib = h_curr[p:] * (1 - h_curr[:n-p])
            else:
                contrib = h_curr[:n-p] * (1 - h_curr[p:])
            idx = np.argmax(contrib)
            if s > 0:
                j1 = idx + p  # h[j1] is the "h" part
                j2 = idx      # (1-h[j2]) is the "(1-h)" part
            else:
                j1 = idx      # h[j1] is the "h" part
                j2 = idx + p  # (1-h[j2]) is the "(1-h)" part

            h_try[j1] -= delta
            h_try[j2] += delta

        if c5_try < best_c5:
            best_c5 = c5_try
            h_curr = h_try

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def soft_max_jac(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(-(1.0 - h))
            for i in order:
                if diff <= 0:
                    break
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
        else:
            order = np.argsort(-h)
            for i in order:
                if diff >= 0:
                    break
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def two_opt_search(h, target, n_iter=1000, t0=None, budget=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    h_curr = h.copy()
    stall = 0
    delta = 0.1
    for it in range(n_iter):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget:
                break
        if stall > 300:
            delta *= 0.6
            stall = 0
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        p = abs(s)
        h_try = h_curr.copy()
        if p == 0:
            vals = h_curr * (1 - h_curr)
            idx = np.argmax(vals)
            if h_curr[idx] > 0.5:
                h_try[idx] -= delta
            else:
                h_try[idx] += delta
        else:
            if s > 0:
                contrib = h_curr[p:] * (1 - h_curr[:n - p])
                idx = np.argmax(contrib)
                j1, j2 = idx + p, idx
            else:
                contrib = h_curr[:n - p] * (1 - h_curr[p:])
                idx = np.argmax(contrib)
                j1, j2 = idx, idx + p
            h_try[j1] -= delta
            h_try[j2] += delta
        h_try = np.clip(h_try, 0, 1)
        h_try = fix_constraint(h_try, target)
        corr_try = np.correlate(h_try, 1 - h_try, mode="full")
        c5_try = float(np.max(corr_try) * dx)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            h_curr = h_try
            corr = corr_try
            stall = 0
        else:
            stall += 1
    return h_curr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(sum_eq_fun, target=target),
             'jac': partial(sum_eq_jac, target=target)}]

    # Phase 1: Temperature annealing with SLSQP
    temps = [300, 150, 80, 50, 30, 20, 12, 8, 6, 4, 3, 2, 1.5, 1]
    for T in temps:
        if elapsed() > budget_s - 350:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 4000, 'ftol': 1e-16})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 2: 2-opt local search
    h_ls, c5_ls = two_opt_search(h_best, target, n_iter=3000, t0=t0, budget=budget_s - 280)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 3: Random restarts with SLSQP + local search
    n_restarts = 0
    while elapsed() < budget_s - 200:
        n_restarts += 1
        r = n_restarts
        if r % 5 == 1:
            h_init = h_best + rng.normal(0, 0.08, n)
        elif r % 5 == 2:
            h_init = h_best + rng.normal(0, 0.04, n)
        elif r % 5 == 3:
            h_init = h_best.copy()
            start = int(rng.integers(0, max(1, n - 30)))
            length = int(rng.integers(5, 30))
            h_init[start:start + length] = 1.0 - h_init[start:start + length]
        elif r % 5 == 4:
            h_init = h_best.copy()
            for _ in range(int(rng.integers(3, 8))):
                start = int(rng.integers(0, max(1, n - 10)))
                length = int(rng.integers(2, 8))
                h_init[start:start + length] = 1.0 - h_init[start:start + length]
        else:
            h_init = (rng.random(n) >= 0.5).astype(float)

        h_init = fix_constraint(h_init, target)
        T = 25
        try:
            res = minimize(soft_max_obj, h_init, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1500, 'ftol': 1e-14})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass
        # Short local search from current best
        h_ls2, c5_ls2 = two_opt_search(h_best, target, n_iter=500, t0=t0, budget=budget_s - 150)
        if c5_ls2 < c5_best:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 4: Fine-tune SLSQP at low temperature
    for T in [4, 2, 1.2, 0.8]:
        if elapsed() > budget_s - 40:
            break
        try:
            res = minimize(soft_max_obj, h_best, method='SLSQP',
                          jac=soft_max_jac, args=(T,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-17})
            if res.fun < 50:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best - 1e-12:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Final 2-opt
    h_ls3, c5_ls3 = two_opt_search(h_best, target, n_iter=1000, t0=t0, budget=budget_s - 5)
    if c5_ls3 < c5_best:
        c5_best = c5_ls3
        h_best = h_ls3

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
