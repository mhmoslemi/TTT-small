```
def solve_1d_lp(corr, a, delta_min, delta_max, n_lines):
    # Minimize max_l (a[l]*delta + corr[l]) over delta in [delta_min, delta_max]
    # Binary search: find delta where subgradient contains 0
    lo, hi = delta_min, delta_max
    for _ in range(50):  # 50 iterations of bisection gives ~1e-15 precision
        mid = (lo + hi) / 2
        vals = a * mid + corr
        active = vals >= np.max(vals) - 1e-15
        subgrad_min = np.min(a[active])
        subgrad_max = np.max(a[active])
        if subgrad_min > 0:  # need to decrease delta
            hi = mid
        elif subgrad_max < 0:  # need to increase delta
            lo = mid
        else:  # 0 is in subdifferential
            break
    return (lo + hi) / 2
```

```python
def pair_coeff(h, i, j, n):
    """Compute a[m] for all correlation indices m=0..2n-2 for swap (i,j)."""
    a = np.zeros(2 * n - 1)
    # h[i] as first factor: l from -i to n-1-i, m = l + n-1
    m1_lo = n - 1 - i

        l = idx - (n - 1)
        a[idx] += 1.0 - h[i + l]
    # h[i] as second factor: l from i-n+1 to i, m = l + n-1
    m2_lo = i  # l=i-n+1 → m = i
    m2_hi = i + n - 1  # l=i → m = i+n-1
    if m2_lo <= m2_hi:
        idx = np.arange(m2_lo, m2_hi + 1)

    if dhi - dlo < 1e-15:
```

```python
def _slsqp_jac(x, dx):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    t = 0.01
    cmax = c.max()
    w = np.exp((c - cmax) / t)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        ks = np.arange(n)
        v1 = (ks - s >= 0) & (ks - s < n)
        g = np.zeros(n)
        g[v1] += 1.0 - x[ks[v1] - s]
        v2 = (ks + s >= 0) & (ks + s < n)
        g[v2] += -x[ks[v2] + s]
        grad += w[m] * g
    return dx * grad

def _slsqp_cons(x, T, n):
    return float(np.sum(x) - T)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    T = n / 2.0
    h = normalize_h(h0, T)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Phase 1: Quick SLSQP to improve starting point
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: _slsqp_cons(x, T, n),
                 'jac': lambda x: np.ones(n)}]
        r = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                     jac=lambda x: _slsqp_jac(x, dx),
                     bounds=bounds, constraints=cons,
                     options={'maxiter': 300, 'ftol': 1e-12})
        h_s = normalize_h(r.x, T)
        c5_s = compute_c5(h_s, dx)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    except Exception:
        pass

    # Phase 2: Swap-based descent (main optimization)
    h_w = best_h.copy()
    corr = compute_corr(h_w)
    for phase in range(40):
        remaining = budget_s - (time.time() - t0)
        if remaining < 60:
            break
        # Systematic sweep
        t_phase = time.time()
        systematic_sweep(h_w, corr, n, min(remaining * 0.3, 30.0))
        # Random sweep
        random_sweep(h_w, corr, n, 30000, rng, min(remaining * 0.4, 20.0))
        # Recompute corr to avoid drift
        corr = compute_corr(h_w)
        c5_now = float(corr.max() * dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h_w.copy()
        # Periodic reflection averaging
        if phase % 5 == 4:
            h_ref = (h_w + h_w[::-1]) / 2.0
            h_ref = normalize_h(h_ref, T)
            c5_ref = compute_c5(h_ref, dx)
            if c5_ref < c5_now - 1e-8:
                h_w = h_ref.copy()
                corr = compute_corr(h_w)

    # Phase 3: Random restarts with swap descent
    remaining = budget_s - (time.time() - t0)
    n_restarts = max(1, int(remaining / 40))
    for ri in range(n_restarts):
        if time.time() - t0 > budget_s - 30:
            break
        h_r = rng.random(n)
        h_r = normalize_h(h_r, T)
        # Mix with best
        alpha = rng.uniform(0.3, 0.7)
        h_r = alpha * best_h + (1 - alpha) * h_r
        h_r = normalize_h(h_r, T)
        corr_r = compute_corr(h_r)
        t_r = time.time()
        t_budget = min(remaining / (n_restarts - ri), 35.0)
        systematic_sweep(h_r, corr_r, n, t_budget * 0.4)
        random_sweep(h_r, corr_r, n, 15000, rng, t_budget * 0.5)
        corr_r = compute_corr(h_r)
        c5_r = float(corr_r.max() * dx)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Try n=100 by upsampling
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        n2 = 100
        dx2 = 2.0 / n2
        T2 = n2 / 2.0
        h_up = np.interp(np.linspace(0, 2, n2, endpoint=False),
                         np.linspace(0, 2, n, endpoint=False),
                         best_h)
        h_up = normalize_h(h_up, T2)
        corr_up = compute_corr(h_up)
        t_up = min(remaining * 0.5, 60.0)
        systematic_sweep(h_up, corr_up, n2, t_up * 0.4)
        random_sweep(h_up, corr_up, n2, 20000, rng, t_up * 0.5)
        corr_up = compute_corr(h_up)
        c5_up = float(corr_up.max() * dx2)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
            n = n2
            dx = dx2

    # Final polish on best
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        T_f = n_f / 2.0
        h_f = best_h.copy()
        corr_f = compute_corr(h_f)
        random_sweep(h_f, corr_f, n_f, 50000, rng, min(remaining - 10, 30.0))
        corr_f = compute_corr(h_f)
        c5_f = float(corr_f.max() * dx_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Final normalization
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    best_h = normalize_h(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, n_f
```
