def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max(corr, T):
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def l-bfgs_refine(h, n, S, T, maxiter=500):

    def obj(x):
        h_clip = np.clip(x, 0, 1)
        corr = np.correlate(h_clip, 1.0 - h_clip, mode='full')
        penalty = (h_clip.sum() - S) ** 2
        return smooth_max(corr, T) + 100.0 * penalty
    
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   options={'maxiter': maxiter, 'gtol': 1e-10})
    return np.clip(res.x, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max(corr, T):
    mx = corr.max()
    if T < 1e-10:
        return mx
    return float(mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T)

def lbfgs_optimize(h, n, S, T, maxiter=300):
    def obj(x):
        hc = np.clip(x, 0, 1)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        sm = smooth_max(corr, T)
        penalty = 500.0 * (hc.sum() - S) ** 2
        return sm + penalty
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max(corr, T):
    mx = corr.max()
    if T < 1e-10:
        return mx
    shifted = T * (corr - mx)
    return float(mx + np.log(np.sum(np.exp(shifted))) / T)

def lbfgs_opt(h, n, S, T, maxiter=200):
    def obj(x):
        hc = np.clip(x, 0, 1)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        return smooth_max(corr, T) + 500.0 * (hc.sum() - S) ** 2
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
    return np.clip(res.x, 0, 1)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def hill_climb(h, n, S, max_sweeps=500, t_deadline=None, rng=None):
    hc = h.copy()
    c5 = compute_c5(hc)
    best_h = hc.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if t_deadline and time.time() > t_deadline:
            break
        improved = False
        one_idx = np.where(hc == 1)[0].tolist()
        zero_idx = np.where(hc == 0)[0].tolist()
        if rng:
            rng.shuffle(one_idx)
            rng.shuffle(zero_idx)
        else:
            np.random.shuffle(one_idx)
            np.random.shuffle(zero_idx)
        for i in one_idx:
            if t_deadline and time.time() > t_deadline:
                break
            for j in zero_idx:
                hc[i], hc[j] = hc[j], hc[i]
                c5_new = compute_c5(hc)
                if c5_new < c5 - 1e-12:
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc.copy()
                    improved = True
                    zero_idx.remove(j)
                    one_idx.remove(i)
                    one_idx.append(j)
                    zero_idx.append(i)
                    break
                else:
                    hc[i], hc[j] = hc[j], hc[i]
        if not improved:
            break
    return best_h, best_c5

def sa(h, n, S, rng, n_iter, t_deadline, best_c5):
    hs = h.copy()
    c5s = compute_c5(hs)
    best_h = hs.copy()
    best_c5 = min(c5s, best_c5)
    temp = 0.0015
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        if it % 4000 == 0 and it > 0:
            temp *= 0.95
        i, j = rng.randint(n, size=2)
        if hs[i] == hs[j]:
            continue
        hs[i], hs[j] = hs[j], hs[i]
        c5_new = compute_c5(hs)
        d = c5_new - c5s
        if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-12)):
            c5s = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = hs.copy()
        else:
            hs[i], hs[j] = hs[j], hs[i]
    return best_h, best_c5

def kick_rescue(h, n, S, rng, best_c5, t_deadline, n_kicks=3):
    for _ in range(n_kicks):
        if time.time() > t_deadline:
            break
        hk = h.copy()
        for _ in range(3):
            i, j = rng.randint(n, size=2)
            if hk[i] != hk[j]:
                hk[i], hk[j] = hk[j], hk[i]
        hk2, c5k = hill_climb(hk, n, S, max_sweeps=100, t_deadline=t_deadline, rng=rng)
        if c5k < best_c5:
            h, best_c5 = hk2, c5k
    return h, best_c5

    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = None

    n_cands = []
    if h0 is not None:
        n_cands.append(len(h0))
    for nt in [100, 140]:
        if nt not in n_cands:
            n_cands.append(nt)

    # Time allocation: n=70 gets 35%, n=100 gets 35%, n=140 gets 25%
    total_time = budget_s * 0.85
    if len(n_cands) <= 1:
        alloc = {n_cands[0]: total_time}
    elif len(n_cands) == 2:
        alloc = {n_cands[0]: total_time * 0.45, n_cands[1]: total_time * 0.55}
    else:
        alloc = {n_cands[0]: total_time * 0.35, n_cands[1]: total_time * 0.35,
                 n_cands[2]: total_time * 0.30}

    t_cursor = t0
    for n in n_cands:
        S = n / 2.0
        t_end = t_cursor + alloc[n]
        n_starts = 5 if n >= 100 else 3

        for si in range(n_starts):
            if time.time() > t_end:
                break

            if si == 0 and h0 is not None and n == len(h0):
                h_cur = h0.copy()
                c5_cur = compute_c5(h_cur)
                if c5_cur < best_c5:
                    best_c5, best_h, best_n = c5_cur, h_cur.copy(), n
                # L-BFGS for continuous refinement
                h_cont = h_cur.astype(float)
                for T in [200, 2000, 10000]:
                    if time.time() > t_end * 0.5:
                        break
                    h_new = lbfgs_opt(h_cont, n, S, T, maxiter=150)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5, best_h, best_n = c5_new, h_new.copy(), n
                    h_cont = h_new
                h_work = to_binary(h_cont, S)
            else:
                h_cur = random_binary(n, rng)
                c5_cur = compute_c5(h_cur)
                if c5_cur < best_c5:
                    best_c5, best_h, best_n = c5_cur, h_cur.copy(), n
                h_work = h_cur.copy()

            c5_w = compute_c5(h_work)
            if c5_w < best_c5:
                best_c5, best_h, best_n = c5_w, h_work.copy(), n

            # SA phase
            t_sa_end = min(t_end * 0.75, time.time() + alloc[n] / (n_starts * 3))
            h_sa, c5_sa = sa(h_work, n, S, rng, 60000, t_sa_end, best_c5)
            if c5_sa < best_c5:
                best_c5, best_h, best_n = c5_sa, h_sa.copy(), n

            # Hill climb
            t_hc = min(t_end * 0.9, time.time() + alloc[n] / (n_starts * 4))
            h_hc, c5_hc = hill_climb(h_sa if c5_sa <= c5_w else h_work, n, S,
                                      max_sweeps=400, t_deadline=t_hc, rng=rng)
            if c5_hc < best_c5:
                best_c5, best_h, best_n = c5_hc, h_hc.copy(), n

            # Kick & rescue
            if time.time() < t_end * 0.95:
                h_kr, c5_kr = kick_rescue(h_hc, n, S, rng, best_c5, t_end, n_kicks=4)
                if c5_kr < best_c5:
                    best_c5, best_h, best_n = c5_kr, h_kr.copy(), n

        t_cursor = time.time()

    # Final polish on best solution found
    if best_h is not None and time.time() - t0 < budget_s * 0.92:
        S_f = best_n / 2.0
        h_bin = to_binary(best_h, S_f)
        c5_b = compute_c5(h_bin)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_bin
        h_f, c5_f = hill_climb(best_h, best_n, S_f, max_sweeps=300,
                                t_deadline=t0 + budget_s * 0.95, rng=rng)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        h_kr, c5_kr = kick_rescue(best_h, best_n, S_f, rng, best_c5,
                                   t0 + budget_s * 0.97, n_kicks=5)
        if c5_kr < best_c5:
            best_c5, best_h = c5_kr, h_kr

    best_h = np.clip(best_h, 0.0, 1.0)
    S_check = best_n / 2.0
    if abs(best_h.sum() - S_check) > 1e-8:
        best_h += (S_check - best_h.sum()) / best_n
        best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
