        for it in range(100000):
            if time.time() > t_limit:
                break
            _, grad = smooth_obj_grad(h, dx, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = lr
            improved = False
            for _ in range(40):
                h_new = project_feasible(h - step * grad, ts)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-15:
                    improved = True
                    break
                step *= 0.5
            if improved:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.02, 1.0)
                stall = 0
            else:
                stall += 1
                lr *= 0.9
            if stall > 500:
                break
    
    return best_h, best_c5

def local_perturb_search(h0, dx, ts, t_limit):
    """Try small perturbations at individual points"""
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for it in range(100000):
        if time.time() > t_limit:
            break
        # Pick a random index and try small perturbation
        i = np.random.randint(n)
        for delta in [0.01, 0.001, -0.01, -0.001, 0.05, -0.05]:
            h_new = h.copy()
            h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_new = project_feasible(h_new, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                break
    
    # Try pairwise perturbations
    for it in range(50000):
        if time.time() > t_limit:
            break
        i, j = np.random.choice(n, 2, replace=False)
        for di in [0.02, -0.02, 0.05, -0.05]:
            for dj in [0.02, -0.02, 0.05, -0.05]:
                h_new = h.copy()
                h_new[i] = np.clip(h[i] + di, 0.0, 1.0)
                h_new[j] = np.clip(h[j] + dj, 0.0, 1.0)
                h_new = project_feasible(h_new, ts)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    break
    
    return best_h, best_c5
