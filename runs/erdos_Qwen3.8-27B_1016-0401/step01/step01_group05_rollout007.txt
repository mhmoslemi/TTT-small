    n = len(h)

    return float(np.max(corr)) * dx

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.clip(alpha * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
