            best_h, best_c5 = h_sl, c5_sl

    # Phase 5: Final targeted local search
    h_ls2, c5_ls2 = targeted_2coord(best_h, target, rng, 40000, t0 + budget_s * 0.85)
    if c5_ls2 < best_c5:
        best_h, best_c5 = h_ls2, c5_ls2

    # Phase 6: Try alternative n values
    n_default = best_n
    for n_try in [50, 60, 80, 100]:
        if time.time() > deadline - 15:
            break
        amt_left = deadline - time.time()
        if amt_left < 20:
            break
        target_try = n_try / 2.0
        h_try = try_structured_start(n_try, rng)
        ph_d = min(deadline - 10, time.time() + min(80, amt_left * 0.6))
        h_opt, c5_opt = topk_gradient_descent(h_try, target_try, 3000, ph_d, top_k=3)
        if time.time() < deadline - 10:
            h_opt2, c5_opt2 = targeted_2coord(h_opt, target_try, rng, 10000, deadline - 8)
            if c5_opt2 < c5_opt:
                h_opt, c5_opt = h_opt2, c5_opt2
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try

    # Final: one more quick gradient pass on best
    if time.time() < deadline - 3:
        h_final, c5_final = topk_gradient_descent(best_h, best_n / 2.0, 2000, deadline - 1, top_k=3, lr_init=0.005)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    best_h = make_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h.tolist(), final_c5, best_n
