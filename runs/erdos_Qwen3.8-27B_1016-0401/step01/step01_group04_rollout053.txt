    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def get_c5_and_bottleneck(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    return float(corr.max() * dx), k_star, corr

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:

        order = np.argsort(-h[zero_idx])

    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def normalized_l2(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)

    diff = S - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):

def to_binary(h, n):
