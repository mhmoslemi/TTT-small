    bc5 = compute_c5(h, dx)
    t = t0
    lr = lr0
    ks = np.arange(n)
    for step in range(steps):
        if time.time() - tstart > tbudget:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if t > 1e-7:
            cm = np.max(c)
            ev = np.exp((c - cm) / t)
            se = np.sum(ev)
            w = ev / se
            g = np.zeros(n)
            for m in range(2 * n - 1):
                if w[m] < 1e-9:
                    continue
                s = m - (n - 1)
                v1 = (ks - s >= 0) & (ks - s < n)
                if v1.any():
                    g[v1] += w[m] * (1.0 - h[ks[v1] - s])
                v2 = (ks + s >= 0) & (ks + s < n)
                if v2.any():
                    g[v2] += w[m] * (-h[ks[v2] + s])
            g *= dx
        else:
            idx = int(np.argmax(c))
            s = idx - (n - 1)
            g = np.zeros(n)
            v1 = (ks - s >= 0) & (ks - s < n)
            g[v1] += (1.0 - h[ks[v1] - s])
            v2 = (ks + s >= 0) & (ks + s < n)
            g[v2] += (-h[ks[v2] + s])
            g *= dx
        h = project_feasible(h - lr * g, ts)
        c5n = compute_c5(h, dx)
        if c5n < bc5:
            bc5 = c5n
            bh = h.copy()
        t = max(t * t_decay, 1e-7)
        if step % 20 == 0:
            lr = max(lr * 0.97, 0.002)
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    global _G_DX, _G_N, _G_T, _G_ST
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = np.full(70, 0.5)
    n0 = len(h0)
    best_h = project_feasible(h0.copy(), n0 / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n0)

    _G_N, _G_DX, _G_T = n0, 2.0 / n0, n0 / 2.0

    # Phase 1: Annealing at n0
    h_a, c5_a = anneal(best_h, n0, 350, 0.08, 0.955, 0.015, t0, budget_s - 200)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a

    # Phase 2: SLSQP smooth
    if rem() > 80:
        _G_ST = 0.025
        bnds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
        try:
            r = minimize(_obj_smooth, best_h, method='SLSQP', jac=_jac_smooth,
                         bounds=bnds, constraints=cons, options={'maxiter': 600, 'ftol': 1e-14})
            c5r = compute_c5(r.x, _G_DX)
            if c5r < best_c5:
                best_c5, best_h = c5r, r.x.copy()
        except Exception:
            pass

    # Phase 3: SLSQP sharp (longer)
    if rem() > 60:
        _G_ST = 1e-9
        try:
            r = minimize(_obj_sharp, best_h, method='SLSQP', jac=_jac_sharp,
                         bounds=bnds, constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
            c5r = compute_c5(r.x, _G_DX)
            if c5r < best_c5:
                best_c5, best_h = c5r, r.x.copy()
        except Exception:
            pass

    # Phase 4: Larger grids with interpolation
    for n_alt in [100, 140, 200, 280]:
        if rem() < 60:
            break
        dx_a = 2.0 / n_alt
        x_old = np.arange(n0) * (2.0 / n0)
        x_new = np.arange(n_alt) * dx_a
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_alt / 2.0)
        _G_N, _G_DX, _G_T = n_alt, dx_a, n_alt / 2.0

        h_a2, c5_a2 = anneal(h_interp, n_alt, 120, 0.04, 0.94, 0.008, t0, budget_s - 50)
        if c5_a2 < best_c5:
            best_c5, best_h = c5_a2, h_a2

        if rem() > 40:
            _G_ST = 0.015
            bnds_a = [(0.0, 1.0)] * n_alt
            cons_a = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
            try:
                r = minimize(_obj_smooth, best_h, method='SLSQP', jac=_jac_smooth,
                             bounds=bnds_a, constraints=cons_a, options={'maxiter': 400, 'ftol': 1e-14})
                c5r = compute_c5(r.x, dx_a)
                if c5r < best_c5:
                    best_c5, best_h = c5r, r.x.copy()
            except Exception:
                pass
            if rem() > 30:
                _G_ST = 1e-9
                try:
                    r = minimize(_obj_sharp, best_h, method='SLSQP', jac=_jac_sharp,
                                 bounds=bnds_a, constraints=cons_a, options={'maxiter': 1500, 'ftol': 1e-16})
                    c5r = compute_c5(r.x, dx_a)
                    if c5r < best_c5:
                        best_c5, best_h = c5r, r.x.copy()
                except Exception:
                    pass

    # Phase 5: Random restarts at n0
    _G_N, _G_DX, _G_T = n0, 2.0 / n0, n0 / 2.0
    for i in range(10):
        if rem() < 50:
            break
        h_r = project_feasible(rng.random(n0), n0 / 2.0)
        h_r2, c5_r2 = anneal(h_r, n0, 150, 0.06, 0.94, 0.012, t0, budget_s - 30)
        if c5_r2 < best_c5:
            best_c5, best_h = c5_r2, h_r2

    # Phase 6: Final polish on best
    if rem() > 20:
        nf = len(best_h)
        _G_N, _G_DX, _G_T = nf, 2.0 / nf, nf / 2.0
        _G_ST = 1e-9
        bnds_f = [(0.0, 1.0)] * nf
        cons_f = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
        try:
            r = minimize(_obj_sharp, best_h, method='SLSQP', jac=_jac_sharp,
                         bounds=bnds_f, constraints=cons_f, options={'maxiter': 5000, 'ftol': 1e-18})
            c5r = compute_c5(r.x, _G_DX)
            if c5r < best_c5:
                best_c5, best_h = c5r, r.x.copy()
        except Exception:
            pass
        # Tiny annealing nudge
        if rem() > 10:
            h_n, c5_n = anneal(best_h, nf, 50, 1e-7, 0.9, 0.003, t0, budget_s - 2)
            if c5_n < best_c5:
                best_c5, best_h = c5_n, h_n

    nf = len(best_h)
    dx_f = 2.0 / nf
    best_h = project_feasible(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, nf
