            pw2 = perturb(best_h, rng, target0, 0.15)
            random_pair_sweep(pw2, rng, 8, phase2_end)
            c5_p2 = compute_c5(pw2)
            if c5_p2 < best_c5:
                best_h = pw2
                best_c5 = c5_p2

    # Phase 3: Binary push + re-optimize
    if time.time() < deadline - 80:
        h_bin = push_to_binary(best_h, target0)
        all_pairs_sweep(h_bin, deadline - 80)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
        # Also try a fresh binary random start
        h_rand = np.zeros(n0)
        h_rand[rng.choice(n0, size=int(target0), replace=False)] = 1.0
        all_pairs_sweep(h_rand, deadline - 80)
        c5_rand = compute_c5(h_rand)
        if c5_rand < best_c5:
            best_h = h_rand
            best_c5 = c5_rand

    # Phase 4: Try different n values
    for n_try in [50, 60, 80, 100, 40]:
        if time.time() > deadline - 50:
            break
        target_t = n_try / 2.0
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = make_feasible(h_try, target_t)
        all_pairs_sweep(h_try, deadline - 50)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            best_n = n_try

    # Phase 5: Final deep polish on best solution
    if time.time() < deadline - 5:
        all_pairs_sweep(best_h, deadline - 5, max_passes=3)

    best_h = make_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5

    return best_h.tolist(), final_c5, best_n
