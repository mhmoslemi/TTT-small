        # Try to avoid duplicates
        while h[int(pos) % n] == 1 and pos < n * 2:
            pos += 0.1
        h[int(pos) % n] = 1.0
    # Fix if we have wrong count
    while h.sum() < S:
        zeros = np.where(h == 0)[0]
        h[zeros[0]] = 1

        h[ones[-1]] = 0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    mx = corr.max()
    # Soft max
    return mx + np.log(np.exp(T * (corr - mx)).sum()) / T

from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def to_binary(h, S):
    target = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def random_binary(n, S, rng):
    h = np.zeros(n)
    h[rng.choice(n, S, replace=False)] = 1.0
    return h

def best_improvement_hc(h, n, S, max_sweeps=5000, t_deadline=None):
    h_hc = h.copy()
    c5 = compute_c5(h_hc)
    best_h = h_hc.copy()
    best_c5 = c5
    for _ in range(max_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        one_idx = np.where(h_hc == 1)[0]
        zero_idx = np.where(h_hc == 0)[0]
        best_delta = 1e-15
        best_i, best_j = -1, -1
        for i in one_idx:
            for j in zero_idx:
                h_hc[i], h_hc[j] = 0.0, 1.0
                c5_new = compute_c5(h_hc)
                delta = c5 - c5_new
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h_hc[i], h_hc[j] = 1.0, 0.0
        if best_i >= 0:
            h_hc[best_i], h_hc[best_j] = 0.0, 1.0
            c5 -= best_delta
            best_h = h_hc.copy()
            best_c5 = c5

def solve_lp(n, S):
    """Solve the LP relaxation to get a starting point and lower bound."""
    dx = 2.0 / n
    h_var = cp.Variable(n, bounds=(0, 1))
    t_var = cp.Variable()
    
    # x variables for all pairs
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    x_vars = {(i, j): cp.Variable() for i, j in pairs}
    
    # x_{ij} linearization
    constraints = []
    for i, j in pairs:
        x = x_vars[(i, j)]
        constraints += [x <= h_var[i], x <= h_var[j], x >= h_var[i] + h_var[j] - 1, x >= 0]
    
    # Correlation constraints
    for k in range(1, n):
        # Lag +k: sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
        # = sum_{n=0}^{n-1-k} h[n+k] - sum_{n=0}^{n-1-k} x[n, n+k]
        expr_plus = sum(h_var[n+k] - x_vars[(n, n+k)] for n in range(n-k))
        constraints.append(expr_plus * dx <= t_var)
        
        # Lag -k: sum_{n=k}^{n-1} h[n-k]*(1-h[n])
        # = sum_{n=k}^{n-1} h[n-k] - sum_{n=k}^{n-1} x[n-k, n]
        expr_minus = sum(h_var[n-k] - x_vars[(n-k, n)] for n in range(k, n))
        constraints.append(expr_minus * dx <= t_var)
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == S)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            try:
                prob.solve(solver=cp.GLPK)
            except:
                return None, None
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None
    
    h_lp = h_var.value
    if h_lp is None:
        return None, None
    lb = t_var.value
    return h_lp, lb
