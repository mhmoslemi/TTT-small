    n_candidates = [48, 50, 52, 96, 100, 44, 46]
    # Prioritize by potential
    n_candidates.sort(key=lambda x: -min(9 / (2 * x) if x % 2 == 0 else 9.5 / (2 * x), 0.38))

    for n_t in n_candidates:
        if time.time() > t_phase2_end or rem() < 40:
            break
        dx_t = 2.0 / n_t
        ts_t = n_t / 2.0

        # Build multiple initializations
        inits = []
        # Interpolated from best
        x_old = np.linspace(0, 2, len(bh))
        x_new = np.linspace(0, 2, n_t)
        inits.append(proj_sum(np.interp(x_new, x_old, bh), ts_t))
        # Legendre
        inits.append(legendre_seq(n_t))
        # Random
        inits.append(proj_sum(np.random.uniform(0, 1, n_t), ts_t))

        best_h_t, best_c_t = None, 1e9
        time_per = max(10, (t_phase2_end - time.time()) / max(1, len(n_candidates)))

        for hi in inits:
            t_sub = min(t_phase2_end, time.time() + time_per * 0.4)
            ho, co = smooth_opt(hi, dx_t, ts_t, t_sub)
            if co < best_c_t:
                best_c_t, best_h_t = co, ho

        # Binary optimization
        if rem() > 30 and best_h_t is not None:
            t_bin = min(t_phase2_end, time.time() + max(15, rem() * 0.4))
            hb, cb = bin_opt(best_h_t, dx_t, t_bin, seed=n_t + seed)
            if cb < best_c_t:
                best_c_t, best_h_t = cb, hb
            # Also try binary from random
            t_bin2 = min(t_phase2_end, time.time() + max(10, rem() * 0.2))
            hr = (np.random.uniform(0, 1, n_t) > 0.5).astype(float)
            hr = to_bin(hr, n_t // 2)
            hbr, cbr = bin_opt(hr, dx_t, t_bin2, seed=n_t + seed + 99)
            if cbr < best_c_t:
                best_c_t, best_h_t = cbr, hbr

        if best_h_t is not None and best_c_t < bc:
            bc, bh, bn = best_c_t, best_h_t.copy(), n_t

    # Phase 3: Refine best (15%)
    t3_end = t0 + budget_s * 0.92
    dx_u, ts_u = 2.0 / bn, bn / 2.0

    # More SA restarts on best binary
    if bn <= 100 and time.time() < t3_end:
        h_b0 = to_bin(bh, bn // 2)
        t_sub = min(t3_end, time.time() + rem() * 0.5)
        hb, cb = bin_opt(h_b0, dx_u, t_sub, seed=seed + 777)
        if cb < bc:
            bc, bh = cb, hb

    # Random restarts at best n
    for _ in range(4):
        if time.time() > t3_end:
            break
        h_r = proj_sum(np.random.uniform(0, 1, bn), ts_u)
        t_sub = min(t3_end, time.time() + max(5, rem() * 0.25))
        hr, cr = smooth_opt(h_r, dx_u, ts_u, t_sub)
        if cr < bc:
            bc, bh = cr, hr
        if rem() < 10:
            break

    # Final
    bh = np.clip(bh, 0, 1)
    bh = proj_sum(bh, bn / 2.0)
    bc = compute_c5(bh, 2.0 / bn)
    return bh, bc, bn
