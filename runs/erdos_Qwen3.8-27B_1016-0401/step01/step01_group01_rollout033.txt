import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, t, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    if t < 1.0:
        return c_max
    e = np.exp(t * (c - c_max))
    return c_max + np.log(e.sum()) / t

def smooth_grad(h, t, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    if t < 1.0:
        idx = int(np.argmax(c))
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    else:
        e = np.exp(t * (c - c_max))
        w = e / e.sum()
    grad = np.zeros(n)
    ks = np.arange(n)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += w[idx] * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += w[idx] * (-h[ks[v2] + s])
    return dx * grad

def sharp_grad(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    ks = np.arange(n)
    g = np.zeros(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    g[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    g[v2] += (-h[ks[v2] + s])
    return dx * g

def _slsqp_obj_sm(x, n, dx, t):
    return smooth_obj(x, t, dx)

def _slsqp_grad_sm(x, n, dx, t):
    return smooth_grad(x, t, dx)

def _slsqp_obj_sharp(x, n, dx):
    return compute_c5(x, dx)

def _slsqp_grad_sharp(x, n, dx):
    return sharp_grad(x, dx)

def _cons_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, dx, max_iter=2000, ftol=1e-14, t_val=None):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_init = project_feasible(h, n)
    try:
        if t_val is not None and t_val > 1:
            r = minimize(_slsqp_obj_sm, h_init, method='SLSQP',
                         jac=_slsqp_grad_sm, args=(n, dx, t_val),
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}],
                         options={'maxiter': max_iter, 'ftol': ftol})
        else:
            r = minimize(_slsqp_obj_sharp, h_init, method='SLSQP',
                         jac=_slsqp_grad_sharp, args=(n, dx),
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}],
                         options={'maxiter': max_iter, 'ftol': ftol})
        h_out = project_feasible(r.x, n)
        c5 = compute_c5(h_out, dx)
        if c5 < compute_c5(h_init, dx):
            return h_out, c5
    except Exception:
        pass
    # Also try smooth polish on h_init
    try:
        r2 = minimize(_slsqp_obj_sm, h_init, method='SLSQP',
                      jac=_slsqp_grad_sm, args=(n, dx, 50.0),
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}],
                      options={'maxiter': max_iter // 2, 'ftol': ftol})
        h_out2 = project_feasible(r2.x, n)
        c5_2 = compute_c5(h_out2, dx)
        if c5_2 < compute_c5(h_init, dx):
            return h_out2, c5_2
    except Exception:
        pass
    return h_init, compute_c5(h_init, dx)

def pgd_optimize(h, n, dx, t_start, t_end, n_iter, lr, momentum):
    h_cur = h.astype(float).copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur, dx)
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        t = t_start * (t_end / t_start) ** frac
        g = smooth_grad(h_cur, t, dx)
        v = momentum * v + g
        h_new = project_feasible(h_cur - lr * v, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_new)

def downsample(h, n_small):
    n_big = len(h)
    x_big = np.linspace(0.0, 2.0, n_big, endpoint=False) + 1.0 / n_big
    x_small = np.linspace(0.0, 2.0, n_small, endpoint=False) + 1.0 / n_small
    h_s = np.interp(x_small, x_big, h)
    return project_feasible(h_s, n_small)
