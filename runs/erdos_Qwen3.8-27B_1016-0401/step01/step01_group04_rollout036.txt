    return np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S

def to_binary(h, S):
    target = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def slsqp_refine(h, T, maxiter=200):
    n = len(h)
    S = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = corr.max()
        return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return normalize_h(res.x, S)
    except:
        return h.copy()

    """Iterated Local Climb for binary sequences."""
    h = h.copy()
    dx = 2.0 / n
    c5 = float(corr_arr(h).max() * dx)
    best_h = h.copy()
    best_c5 = c5
    
    for rnd in range(max_rounds):
        if time.time() > t_deadline:
            break
        # Hill climb
        improved = True
        while improved:
            if time.time() > t_deadline:
                return best_h, best_c5
            improved = False
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
            # Randomize order for diversity
            rng.shuffle(one_idx)
            rng.shuffle(zero_idx)
            for i in one_idx:
                for j in zero_idx:
                    h[i], h[j] = h[j], h[i]
                    new_c5 = float(corr_arr(h).max() * dx)
                    if new_c5 < c5 - 1e-15:
                        c5 = new_c5
                        improved = True
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                    else:
                        h[i], h[j] = h[j], h[i]
        
        # Perturb: random swaps
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        for _ in range(perturb):
            i = one_idx[rng.randint(len(one_idx))]
            j = zero_idx[rng.randint(len(zero_idx))]
            h[i], h[j] = h[j], h[i]
        c5 = float(corr_arr(h).max() * dx)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, T, maxiter=200):
    n = len(h)
    S = n / 2.0
    def obj(x):
        return smooth_max_val(x, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()

def perturb_and_optimize(h, rng, S, T, n_perturb, maxiter=200):

    h_p[idx] += rng.normal(0, 0.08, n_perturb)
    h_p = normalize_h(h_p, S)
    return slsqp_step(h_p, T, maxiter=maxiter)
