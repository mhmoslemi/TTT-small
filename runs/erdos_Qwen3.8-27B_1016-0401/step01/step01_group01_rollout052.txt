def _lbgfbs_optimize(h_init, dx, T, t_list, penalty, max_iter=200):
    n = len(h_init)
    h = h_init.copy()
    
    for t in t_list:
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            sm = cm + t * np.log(np.sum(np.exp((c - cm) / t)))
            return dx * sm + penalty * (np.sum(x) - T)**2
        
        def grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            w = np.exp((c - cm) / t)
            w /= np.sum(w)
            # Compute gradient of smooth max
            n_x = len(x)
            S = np.arange(2*n_x - 1) - (n_x - 1)
            J = np.arange(n_x)
            # For each j and m: shift s_m
            # dc_m/dx_j = (1-x[j+s_m])*[valid] - x[j-s_m]*[valid]
            # Use vectorized: 
            JS_p = J[:, None] + S[None, :]  # (n, 2n-1)
            JS_m = J[:, None] - S[None, :]  # (n, 2n-1)
            vp = (JS_p >= 0) & (JS_p < n_x)
            vm = (JS_m >= 0) & (JS_m < n_x)
            # term_p[j,m] = vp[j,m] * (1 - x[JS_p[j,m]])
            # term_m[j,m] = vm[j,m] * x[JS_m[j,m]]
            xp = np.zeros_like(JS_p, dtype=float)
            xp[vp] = 1.0 - x[JS_p[vp].astype(int)]
            xm = np.zeros_like(JS_m, dtype=float)
            xm[vm] = x[JS_m[vm].astype(int)]
            g_sm = (w[None, :] * (xp - xm)).sum(axis=1)
            g_pen = 2 * penalty * (np.sum(x) - T)
            return dx * g_sm + g_pen
        
        bounds = [(0.0, 1.0)] * n
        r = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds,
                     options={'maxiter': max_iter, 'ftol': 1e-14})
        h = r.x
        # Project to feasible

        s = h.sum()
        if s > 1e-10:

            h = np.clip(h, 0, 1)

def _simulated_annealing(h_init, dx, T, n_iter, t_start, t_end):
    h = h_init.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_cost = np.max(c) * dx
    
    n = len(h)
    S = np.arange(2*n - 1) - (n - 1)
    
    for it in range(n_iter):
        frac = it / n_iter
        temp = t_start * (t_end / t_start) ** frac  # geometric cooling
        eps = 0.01
        
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = np.random.uniform(-eps, eps)
        
        hi_new = h[i] + d
        hj_new = h[j] - d
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        
        # Compute delta c
        # For index i:
        di = np.zeros(2*n-1)
        ip = i + S
        ip_valid = (ip >= 0) & (ip < n)
        di[ip_valid] += (1.0 - h[ip[ip_valid].astype(int)])
        im = i - S
        im_valid = (im >= 0) & (im < n)
        di[im_valid] += (-h[im[im_valid].astype(int)])
        
        # For index j:
        dj = np.zeros(2*n-1)
        jp = j + S
        jp_valid = (jp >= 0) & (jp < n)
        dj[jp_valid] += (1.0 - h[jp[jp_valid].astype(int)])
        jm = j - S
        jm_valid = (jm >= 0) & (jm < n)
        dj[jm_valid] += (-h[jm[jm_valid].astype(int)])
        
        dc = d * (di - dj)
        new_c = c + dc
        new_cost = np.max(new_c) * dx
        delta_cost = new_cost - current_cost
        
        if delta_cost < 0 or np.random.rand() < np.exp(-delta_cost / max(temp, 1e-10)):
            h[i] = hi_new
            h[j] = hj_new
            c = new_c
            current_cost = new_cost
    
    return h, current_cost

import numpy as np
from scipy.optimize import minimize

_G_DX = 1.0
_G_T = 35.0
_G_N = 70
_G_TS = 0.1
_G_PEN = 100.0

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, T):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h = h * T / s
    else:
        h = np.full_like(h, T / len(h))
    return np.clip(h, 0.0, 1.0)

def _g_obj(x):
    n = _G_N
    c = np.correlate(x, 1.0 - x, mode='full')
    t = _G_TS
    cm = np.max(c)
    val = _G_DX * (cm + t * np.log(np.sum(np.exp((c - cm) / t))))
    val += _G_PEN * (x.sum() - _G_T) ** 2
    return val

def _g_grad(x):
    n = _G_N
    c = np.correlate(x, 1.0 - x, mode='full')
    t = _G_TS
    cm = np.max(c)
    w = np.exp((c - cm) / t)
    w /= w.sum()
    S = np.arange(2 * n - 1) - (n - 1)
    J = np.arange(n)
    JS_p = J[:, None] + S[None, :]
    JS_m = J[:, None] - S[None, :]
    vp = (JS_p >= 0) & (JS_p < n)
    vm = (JS_m >= 0) & (JS_m < n)
    xc = np.clip(JS_p, 0, n - 1)
    xm_ = np.clip(JS_m, 0, n - 1)
    xp = np.where(vp, 1.0 - x[xc], 0.0)
    xm = np.where(vm, x[xm_], 0.0)
    g_sm = (w[None, :] * (xp - xm)).sum(axis=1)
    return _G_DX * g_sm + 2.0 * _G_PEN * (x.sum() - _G_T)

def lbgfsb_opt(h_init, dx, T, t_sched, pen=200.0, maxit=250):
    global _G_DX, _G_T, _G_N, _G_TS, _G_PEN
    _G_DX = dx
    _G_T = T
    _G_N = len(h_init)
    _G_PEN = pen
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * _G_N
    for t in t_sched:
        _G_TS = t
        try:
            r = minimize(_g_obj, h, method='L-BFGS-B', jac=_g_grad,
                         bounds=bounds, options={'maxiter': maxit, 'ftol': 1e-15})
            h = project_feasible(r.x, T)
        except Exception:
            pass
    return h

def sa_opt(h_init, dx, T, max_steps, seed=42, t_start=0.004, t_end=0.00003, eps=0.003, deadline=None):
    import time
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(c))
    best_max = cur_max
    best_h = h.copy()
    nm1 = 2 * n - 1
    S = np.arange(nm1) - (n - 1)
    pp = []
    mp = []
    for i in range(n):
        pv = (i + S >= 0) & (i + S < n)
        pp.append((np.where(pv)[0], (i + S)[pv]))
        mv = (i - S >= 0) & (i - S < n)
        mp.append((np.where(mv)[0], (i - S)[mv]))
    di_buf = np.zeros(nm1)
    dj_buf = np.zeros(nm1)
    dc_buf = np.zeros(nm1)
    step = 0
    while step < max_steps:
        if deadline is not None and step % 5000 == 0 and time.time() > deadline:
            break
        frac = step / max(max_steps, 1)
        temp = t_start * (t_end / t_start) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            step += 1
            continue
        d = rng.uniform(-eps, eps)
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            step += 1
            continue
        pms, phs = pp[i]
        di_buf[:] = 0.0
        di_buf[pms] = (1.0 - h[phs])
        mms, mhs = mp[i]
        di_buf[mms] += (-h[mhs])
        pms, phs = pp[j]
        dj_buf[:] = 0.0
        dj_buf[pms] = (1.0 - h[phs])
        mms, mhs = mp[j]
        dj_buf[mms] += (-h[mhs])
        dc_buf[:] = d * (di_buf - dj_buf)
        new_c = c + dc_buf
        new_max = float(np.max(new_c))
        delta = (new_max - cur_max) * dx
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-12)):
            h[i] += d
            h[j] -= d
            c = new_c
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        step += 1
    best_h = project_feasible(best_h, T)
    return best_h, compute_c5(best_h, dx)

def _make_periodic_init(rng, n, n_periods):
    period = n // n_periods
    if period < 3 or n_periods * period != n:
        return project_feasible(rng.random(n), n / 2.0)
    base = rng.random(period)
    base *= (n / 2.0) / (n_periods * base.sum())
    base = np.clip(base, 0, 1)
    h = np.tile(base, n_periods)
    return project_feasible(h, n / 2.0)
