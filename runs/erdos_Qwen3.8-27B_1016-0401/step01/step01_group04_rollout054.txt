        second = i + idx - n + 1  # = i+k-n+1

    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = compute_corrs(x)
        mx = corr.max()
        return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
    constraint = {'type': 'eq', 'fun': lambda x: x.sum() - S}
    try:

                       constraints=constraint, options={'maxiter': maxiter, 'ftol': 1e-12})

        h_rand[ones_positions] = 1.0

                                          t_deadline=t0 + per_start_time * (s + 1) + budget_s * 0.05)
        if c5_g < best_c5:
            best_h = h_g.copy()
            best_c5 = c5_g

            h_try[i], h_try[j] = 0.0, 1.0

            best_c5 = c5_g
