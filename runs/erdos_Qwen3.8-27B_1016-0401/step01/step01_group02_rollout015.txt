def _smooth_obj_and_grad(x, sc):
    n = len(x)
    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # log-sum-exp
    m = corr.max()
    lse = m + np.log(np.exp(sc * (corr - m)).sum()) / sc
    # gradient: softmax weights
    w = np.exp(sc * (corr - m)) / np.exp(sc * (corr - m)).sum()
    # gradient of lse w.r.t. h

    grad = np.zeros(n)

        h3, c3 = _local_search(bh, rem)
        if c3 < bc:
            bh, bc = h3, c3

    # Final ceil
    fc = _c5(bh)
    return (bh, fc, len(bh))
