Actually, let me reconsider the smooth optimization. The log-sum-exp objective is:
f(h) = dx * (max_c + log(Σ exp(beta*(c - max_c))) / beta)

For large beta, this approximates dx * max_c. The gradient is:
∂f/∂h[i] = dx * Σ_m w_m * ∂c_m/∂h[i]

where w_m = exp(beta*(c_m - max_c)) / Σ exp(beta*(c_m - max_c))

The gradient computation requires knowing which h[i] appears in c_m, which depends on the structure of the correlation.

For n=46, this is all very manageable. Let me just use the same smooth optimization as before but with more aggressive settings.

2. For each, run aggressive binary local search (1-swaps, 2-swaps, random perturbations)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj < 1.0 - 1e-10) if diff > 0 else (proj > 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def make_binary(h, ts):
    h = np.array(h, dtype=float)
    h = (h > 0.5).astype(float)
    oc = int(np.sum(h))
    if oc > ts:
        idxs = np.random.permutation(np.where(h == 1)[0])[:oc - ts]
        h[idxs] = 0.0
    elif oc < ts:
        idxs = np.random.permutation(np.where(h == 0)[0])[:ts - oc]
        h[idxs] = 1.0
    return h

def thue_morse_seq(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

def _cons_fun(x, ts):
    return np.sum(x) - ts

def _cons_jac(x):
    return np.ones(len(x))

def optimize_smooth(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for beta in [200.0, 1000.0, 5000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': lambda x: _cons_fun(x, ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def all_swaps_search(h, dx, t_limit):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    while time.time() < t_limit:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_gain = 1e-14
        best_pair = None
        for zi in zeros:
            for oj in ones:
                h[zi] = 1.0
                h[oj] = 0.0
                g = best_c5 - compute_c5(h, dx)
                if g > best_gain:
                    best_gain = g
                    best_pair = (int(zi), int(oj))
                h[zi] = 0.0
                h[oj] = 1.0
        if best_pair is not None:
            h[best_pair[0]] = 1.0
            h[best_pair[1]] = 0.0
            best_c5 -= best_gain
        else:
            break
    return h.copy(), best_c5

def two_swaps_search(h, dx, t_limit):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    stall = 0
    while time.time() < t_limit and stall < 50:
        zeros = np.where(h == 0)[0].tolist()
        ones = np.where(h == 1)[0].tolist()
        improved = False
        for i1 in range(len(zeros)):
            if improved: break
            for i2 in range(i1 + 1, len(zeros)):
                if improved: break
                for j1 in range(len(ones)):
                    if improved: break
                    for j2 in range(j1 + 1, len(ones)):
                        if time.time() > t_limit:
                            return h.copy(), best_c5
                        z1, z2, o1, o2 = zeros[i1], zeros[i2], ones[j1], ones[j2]
                        h[z1] = h[z2] = 1.0
                        h[o1] = h[o2] = 0.0
                        c5n = compute_c5(h, dx)
                        if c5n < best_c5 - 1e-14:
                            best_c5 = c5n
                            improved = True
                            break
                        h[z1] = h[z2] = 0.0
                        h[o1] = h[o2] = 1.0
        if improved:
            stall = 0
        else:
            stall += 1
    return h.copy(), best_c5

def aggressive_search(h0, dx, t_limit, rng):

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
```

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj < 1.0 - 1e-10) if diff > 0 else (proj > 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def make_binary(h, ts):
    h = np.array(h, dtype=float)
    h = (h > 0.5).astype(float)

        idxs = np.random.permutation(np.where(h == 1)[0])[:oc - ts]
        h[idxs] = 0.0

        h[idxs] = 1.0

    corr = np.correlate(h, 1.0 - h, mode='full')

    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
