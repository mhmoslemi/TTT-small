```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    max_s = np.max(shifted)
    e = np.exp(shifted - max_s)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) + max_s) / T * (2.0 / n)
    
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    grad *= dx
    return obj, grad

def project_feasible(z, target):
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            s = min(abs(diff), (1-h[i]) if diff>0 else h[i])
            h[i] += s if diff > 0 else -s
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def pg_optimize(h, n, target, T, max_iter, t_deadline):
    """Projected gradient with momentum."""
    h_c = h.copy()
    m = np.zeros(n)
    lr = 2.0 / (T * n) * 50  # heuristic
    lr = min(lr, 0.5)
    beta = 0.9
    obj_c, _ = smooth_obj_and_grad(h_c, n, T)
    
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        _, g = smooth_obj_and_grad(h_c, n, T)
        m = beta * m + (1-beta) * g
        h_new = project_feasible(h_c - lr * m, target)
        obj_new, _ = smooth_obj_and_grad(h_new, n, T)
        
        if obj_new < obj_c - 1e-15:
            h_c = h_new
            obj_c = obj_new
            # Occasionally increase lr
            if it % 200 == 199:
                lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return h_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 30
    rng = np.random.default_rng(seed)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Annealed projected gradient
    h_curr = h_best.copy()
    T_schedule = [300, 150, 75, 40, 20, 10, 5, 2, 1]
    iters_per_T = [2000, 3000, 3000, 3000, 3000, 3000, 3000, 3000, 5000]
    
    for T, ni in zip(T_schedule, iters_per_T):
        if time.time() > deadline - 120:
            break
        h_curr = pg_optimize(h_curr, n, target, T, ni, deadline - 120)
        c5_new = compute_c5(h_curr)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()
    
    # Phase 2: Random restarts (few, to explore)
    n_restarts = 0
    while time.time() < deadline - 60 and n_restarts < 10:
        n_restarts += 1
        r = n_restarts
        if r % 3 == 1:
            h_init = (rng.random(n) >= 0.5).astype(float)
        elif r % 3 == 2:
            h_init = h_best + rng.normal(0, 0.15, n)
        else:
            h_init = h_best + rng.normal(0, 0.05, n)
        h_init = project_feasible(h_init, target)
        
        for T in [50, 20, 5, 1]:
            if time.time() > deadline - 60:
                break
            h_init = pg_optimize(h_init, n, target, T, 2000, deadline - 60)
        c5_new = compute_c5(h_init)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_init.copy()
    
    # Final: ensure constraint is met
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)
    
    return (h_best, c5_best, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    ms = np.max(shifted)
    e = np.exp(shifted - ms)
    se = e.sum()
    sw = e / se
    obj = (mc + (ms + np.log(se)) / T) * (2.0 / n)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    grad *= dx
    return obj, grad

def project_feasible(z, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        for i in idx:
            step = min(abs(diff), (1.0 - h[i]) if diff > 0 else h[i])
            h[i] += step if diff > 0 else -step
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def pg_optimize(h, n, target, T, max_iter, t_deadline):
    h_c = h.copy()
    obj_c, g_c = smooth_obj_and_grad(h_c, n, T)
    m = np.zeros(n)
    lr = 8.0 / max(T, 1)
    lr = min(lr, 3.0)
    beta = 0.85
    stall = 0
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        m = beta * m + (1.0 - beta) * g_c
        h_new = project_feasible(h_c - lr * m, target)
        obj_new, g_new = smooth_obj_and_grad(h_new, n, T)
        if obj_new < obj_c - 1e-16:
            h_c = h_new
            obj_c = obj_new
            g_c = g_new
            stall = 0
            if it % 300 == 299:
                lr = min(lr * 1.15, 5.0)
        else:
            lr *= 0.65
            stall += 1
            if stall > 300 or lr < 1e-8:
                break
    return h_c

def soprogram_restart_finite_differences(h, n, target, T, steps, t_deadline):
    h_c = h.copy()
    obj_c, _ = smooth_obj_and_grad(h_c, n, T)
    lr = 2.0 / max(T, 1)
    m = np.zeros(n)
    beta = 0.8
    for it in range(steps):
        if time.time() > t_deadline:
            break
        _, g_c = smooth_obj_and_grad(h_c, n, T)
        m = beta * m + (1.0 - beta) * g_c
        h_new = project_feasible(h_c - lr * m, target)
        obj_new, g_new = smooth_obj_and_grad(h_new, n, T)
        if obj_new < obj_c:
            h_c = h_new
            obj_c = obj_new
        else:
            lr *= 0.6
            if lr < 1e-8:
                break
    return h_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 30
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Main annealed projected gradient
    h_curr = h_best.copy()
    T_list = [400, 200, 100, 50, 25, 12, 6, 3, 1.5, 1, 0.7]
    iters_list = [1500, 2500, 2500, 2500, 2500, 2500, 2500, 2500, 2500, 2500, 3000]
    for T, ni in zip(T_list, iters_list):
        if time.time() > deadline - 150:
            break
        h_curr = pg_optimize(h_curr, n, target, T, ni, deadline - 150)
        c5_new = compute_c5(h_curr)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    # Phase 2: SLSQP polish on best (few iterations to squeeze out)
    if time.time() < deadline - 100:
        bounds = [(0.0, 1.0)] * n
        def _obj_s(x):
            o, _ = smooth_obj_and_grad(x, n, 2.0)
            return o
        def _jac_s(x):
            _, g = smooth_obj_and_grad(x, n, 2.0)
            return g
        def _con_f(x):
            return x.sum() - target
        cons_s = [{'type': 'eq', 'fun': _con_f, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(_obj_s, h_best.copy(), method='SLSQP', jac=_jac_s,
                           bounds=bounds, constraints=cons_s,
                           options={'maxiter': 300, 'ftol': 1e-15})
            if res.fun < 100:
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: Diverse restarts at lower resolution (n=140)
    if time.time() < deadline - 80:
        n_lo = 140
        target_lo = n_lo / 2.0
        x_best = np.arange(n) * (2.0 / n)
        x_lo = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_lo, x_best, h_best)
        h_lo = project_feasible(h_lo, target_lo)
        c5_lo = compute_c5(h_lo)

        for T in [50, 20, 8, 3, 1]:
            if time.time() > deadline - 60:
                break
            h_lo = pg_optimize(h_lo, n_lo, target_lo, T, 3000, deadline - 60)
        c5_lo = compute_c5(h_lo)

        # Interpolate back to n and compare
        x_new = np.arange(n) * (2.0 / n)
        h_interp = np.interp(x_new, np.arange(n_lo) * (2.0 / n_lo), h_lo)
        h_interp = project_feasible(h_interp, target)
        c5_interp = compute_c5(h_interp)

        if c5_interp < c5_best:
            c5_best = c5_interp
            h_best = h_interp
            # Refine at full resolution
            for T in [10, 4, 1.5, 1]:
                if time.time() > deadline - 40:
                    break
                h_best = pg_optimize(h_best, n, target, T, 2000, deadline - 40)
                c5_try = compute_c5(h_best)
                if c5_try < c5_best:
                    c5_best = c5_try
        elif c5_lo < 0.3809:
            # Even if interpolation didn't help, the n_lo solution is candidate
            pass

    # Phase 4: Random restarts at full resolution
    n_restarts = 0
    while time.time() < deadline - 30 and n_restarts < 8:
        n_restarts += 1
        r = n_restarts
        if r % 4 == 1:
            h_init = (rng.random(n) >= 0.5).astype(float)
        elif r % 4 == 2:
            h_init = h_best + rng.normal(0, 0.2, n)
        elif r % 4 == 3:
            h_init = h_best.copy()
            perm = rng.permutation(n)
            h_init[perm[:30]] += 0.5
        else:
            h_init = h_best + rng.normal(0, 0.06, n)
        h_init = project_feasible(h_init, target)
        for T in [80, 30, 10, 3, 1]:
            if time.time() > deadline - 30:
                break
            h_init = pg_optimize(h_init, n, target, T, 2000, deadline - 30)
        c5_new = compute_c5(h_init)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_init.copy()

    # Final constraint check
    h_best = project_feasible(h_best, target)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    if c5_final >= c5_best:
        c5_final = compute_c5(project_feasible(h_best, target))
    return (h_best, c5_final, len(h_best))
```
