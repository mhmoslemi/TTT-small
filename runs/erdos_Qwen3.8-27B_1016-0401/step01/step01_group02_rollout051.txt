def _proj(h, n):
    """Project onto {0≤h≤1, sum(h)=n/2}"""
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-14:

        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)

            np.clip(h, 0.0, 1.0, out=h)

            np.clip(h, 0.0, 1.0, out=h)

def _sg(h, T):

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = c.max()
    e = np.exp(T * (c - cm))
    p = e / e.sum()
    val = cm + np.log(e.sum()) / T
    b = 1.0 - h
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for tau in range(-(n-1), n):
        pi = p[tau + (n-1)]
        if pi < 1e-16:
            continue
        if tau >= 0:
            T1[:n-tau] += pi * b[tau:]
            T2[tau:] += pi * h[:n-tau]
        else:
            s = -tau
            T1[s:] += pi * b[:n-s]
            T2[:n-s] += pi * h[s:]
    return val, (T1 - T2)

    except Exception:

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
