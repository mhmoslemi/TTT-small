def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    i_lo = max(0, m - n + 1)
    i_hi = min(n - 1, m)
    g = np.zeros(n)
    i_vals = np.arange(i_lo, i_hi + 1)
    j_vals = m - i_vals
    g[i_vals] += (1.0 - h[j_vals])
    g[j_vals] += (-h[i_vals])

def active_grad(h, top_k, sharpness):
    n = len(h)
    corr = compute_corr(h)
    idx = np.argsort(corr)[-top_k:]
    vals = corr[idx]
    sm = np.max(vals)
    w = np.exp(sharpness * (vals - sm))
    w /= w.sum()
    grad = np.zeros(n)
    for m, wi in zip(idx, w):
        grad += wi * grad_shift(h, m, n)
    return grad, corr, idx

                    if h_try[j] < -1e-6 or h_try[j] > 1.0 + 1e-6:
                        continue
                    h_try = project_box_sum(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5 - 1e-10:
                        h = h_try
                        c5 = c5_try
                        improved = True
                        break
        if not improved:
            break
    return h, c5

def make_sin_start(n, freq, target):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x / n)
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, target)
    return h

def make_block_start(n, target, nblocks=4, rng=None):
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(nblocks):
        start = b * block_size
        end = start + block_size
        h[start:end] = 1.0 if b % 2 == 0 else 0.0
    h = project_box_sum(h, target)
    if rng is not None:
        h = h + rng.uniform(-0.1, 0.1, n)
        h = project_box_sum(h, target)
    return h

def slsqp_final(h, target, maxiter=200):
    global _G_TARGET
    _G_TARGET = target
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = 100.0 * (corr - m)
        vals = np.clip(vals, -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / 100.0) * (2.0 / n)
    def con_fun(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_box_sum(res.x, target)
        return h_new
    except Exception:
        return h.copy()

_G_TARGET = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(70) * 0.5
    n_parent = len(h_parent)
    target_p = n_parent / 2.0
    h_parent = project_box_sum(h_parent, target_p)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    def try_optimize(h0, n, target, dl, label=""):
        nonlocal best_h, best_c5, best_n
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_h = h0.copy()
            best_c5 = c5_0
            best_n = n
        h1 = project_box_sum(h0, target)
        h1, c5_1 = pgd_optimize(h1, target, dl, top_k=8, sharpness=20, lr0=0.006)
        if c5_1 < best_c5:
            best_h = h1.copy()
            best_c5 = c5_1
            best_n = n
        h2 = slsqp_final(h1, target, maxiter=300)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
            best_n = n
        return best_h, best_c5

    # Phase 1: Optimize parent at native resolution
    if time.time() < deadline - 100:
        try_optimize(h_parent, n_parent, target_p, min(deadline - 200, t0 + 150))

    # Phase 2: Higher resolution (interpolate parent)
    for n_hi in [100, 140]:
        if time.time() > deadline - 80:
            break
        x_old = np.arange(n_parent)
        x_new = np.linspace(x_old[0], x_old[-1], n_hi)
        h_interp = np.interp(x_new, x_old, h_parent)
        target_hi = n_hi / 2.0
        h_interp = project_box_sum(h_interp, target_hi)
        try_optimize(h_interp, n_hi, target_hi, min(deadline - 60, t0 + 300))

    # Phase 3: Structured starting points
    n_try = best_n if best_n >= 70 else 70
    target_try = n_try / 2.0
    if time.time() < deadline - 60:
        for freq in [1, 2, 3]:
            if time.time() > deadline - 50:
                break
            h_s = make_sin_start(n_try, freq, target_try)
            try_optimize(h_s, n_try, target_try, min(deadline - 40, t0 + 450))

    # Phase 4: Block patterns
    if time.time() < deadline - 50:
        for nb in [2, 3, 5]:
            if time.time() > deadline - 45:
                break
            h_b = make_block_start(n_try, target_try, nb, rng)
            try_optimize(h_b, n_try, target_try, min(deadline - 35, t0 + 550))

    # Phase 5: Coordinate descent polish on best
    if time.time() < deadline - 15:
        h_pol, c5_pol = coord_descent_polish(best_h, best_n / 2.0, 5, deadline - 8)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol

    # Phase 6: Final SLSQP + PGD refinement
    if time.time() < deadline - 5:
        h_fin = slsqp_final(best_h, best_n / 2.0, maxiter=400)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin
        h_fin2, c5_fin2 = pgd_optimize(best_h, best_n / 2.0, deadline - 2,
                                       top_k=12, sharpness=60, lr0=0.003)
        if c5_fin2 < best_c5:
            best_h = h_fin2
            best_c5 = c5_fin2

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, best_n
