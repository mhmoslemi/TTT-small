Objective function: min max_{k} [∫h(x)(1-h(x+k))dx]

Implementation details:
- We first optimize the entire vector using CMA-ES with bounds [0,1] and exact sum constraint
- Repair function to enforce ∑h_i = n/2
- Transition from global to local search using scheduled restarts
- Final polish using simulated annealing to escape any remaining local optima

3. Implementation procedure:
a) Setup parents and repair function
```python
import numpy as np
from scipy.optimize import cma
TARGET_SUM = 0.0  # Will be set to n/2

def repair_constraint(h):
    '''Enforce 0 ≤ h[i] ≤ 1 and ∑h[i] = n/2'''
    h = np.array(h, dtype=float)
    n = len(h)
    # Enforce bounds
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    # If exact sum, return
    if np.isclose(total, n/2):
        return h
    # Otherwise adjust to meet sum constraint
    diff = n/2 - total
    # Random selection of indices for adjustment
    num_indices = min(100, len(h))
    indices = np.random.choice(len(h), num_indices, replace=False)
    increments = np.ones(num_indices) * diff / num_indices
    h[indices] += increments
    return np.clip(h, 0.0, 1.0)
    
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def objective_function(x):
    h_adjusted = repair_constraint(x)
    return compute_c5(h_adjusted)
```

```python
def simulated_annealing(h, n, target_sum, temperature=10.0, cooling_rate=0.995,
                       max_iter=50000, initial_temp=10.0):
    current = np.array(h, dtype=float)
    current_energy = objective_function(current)
    best = current.copy()
    best_energy = current_energy
    
    T = initial_temp
    for i in range(max_iter):
        # Two-point mutation
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.normal(0, temperature, 2)
        new1 = np.clip(np.exp(-0.5*delta[0]**2) * current[i], 0.0, 1.0)
        new2 = np.clip(np.exp(-0.5*delta[1]**2) * current[j], 0.0, 1.0)
        
        new_state = current.copy()
        new_state[i] = new1
        new_state[j] = new2
        new_state = repair_constraint(new_state)
        new_energy = objective_function(new_state)
        
        # Metropolis criterion
        if new_energy < best_energy or np.random.random() < np.exp((new_energy - best_energy)/best_energy):
            best = new_state.copy()
            best_energy = new_energy
            
        # Schedule temperature reduction
        T = cooling_rate * T
        if T < 0.01 or np.random.random() < 0.8:
            T = initial_temp * 0.8  # Periodic reset to maintain exploration
    
    return best.tolist()
```
