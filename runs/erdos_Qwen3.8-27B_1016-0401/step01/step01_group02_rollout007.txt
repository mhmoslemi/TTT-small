s.t. sum(h) = n/2, 0 <= h <= 1

Gradient: g[j] = h[j-k] + h[j+k] (with boundary: if j-k < 0, that term is 0; if j+k >= n, that term is 0)
Hessian: H[j,m] = 1 if |j-m| = k (i.e., m = j+k or m = j-k)

Newton step: h_new = h - H^{-1} * g (but H is not positive definite, it's indefinite)

Actually, since we're maximizing, we want to follow the positive direction. The Hessian here is the adjacency of a k-skip graph, which is not positive definite. So Newton might not work directly.

Let me just use gradient ascent on Q_k (which is equivalent to gradient descent on corr_k):
h_new = h + alpha * g_k  (where g_k[j] = h[j-k] + h[j+k])
Then project.

This is O(n) per step! Very fast.

**My final algorithm:**

1. Start from initial_h_values
2. SLSQP with soft-max (to get a good starting point)
3. **Cycling projected gradient**: 
   - For each iteration cycle:
     - Find top K active lags
     - For each active lag k:
       - Compute subgradient of corr_k
       - Take a small gradient descent step (projected)
   - Periodically evaluate and track best
4. Fine-tune with local search
5. Optionally upscale

import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_GN = 0
_GDX = 0.0
_GSC = 50.0

def eval_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] -= d / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-13:
        msk = (h < 1.0 - 1e-10) if r > 0 else (h > 1e-10)
        cnt = msk.sum()
        if cnt > 0:
            h[msk] += r / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _slsqp_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.exp(_GSC * (corr - m)).sum()) / _GSC) * _GDX

def _slsqp_eq(x):
    return x.sum() - 0.5 * _GN

def _slsqp_eqjac(x):
    return np.ones(len(x))

def do_slsqp(h0, n, dx, scales, maxiter):
    global _GN, _GDX, _GSC
    best = h0.copy()
    bestc = eval_c5(best)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eqjac}]
    for sc in scales:
        _GN, _GDX, _GSC = n, dx, sc
        try:
            res = minimize(_slsqp_obj, best, method='SLSQP', bounds=bds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c_new = eval_c5(h_new)
            if c_new < bestc:
                best, bestc = h_new, c_new
        except Exception:
            pass
    return best, bestc

def subgrad_corr_k(h, k):
    """Subgradient of corr_k w.r.t. h. k is the actual lag (can be negative)."""
    n = len(h)
    g = np.zeros(n)
    # corr_k = sum_i h[i+k] * (1-h[i]) for valid i
    # dc_k/dh[j] = (1-h[j-k]) if 0<=j-k<n  minus  h[j+k] if 0<=j+k<n
    # Wait let me re-derive.
    # np.correlate(h, 1-h, mode='full')
    # c[idx] where idx = n-1+k is sum_i h[i+k]*b[i] with b = 1-h
    # So c_k = sum_i h[i+k] * (1-h[i])
    
    # dc_k/dh[j]:
    # h[j] appears in h[i+k] when i+k=j => i=j-k: contributes (1-h[j-k])
    # h[j] appears in h[i] when i=j: contributes -h[j+k] (derivative of (1-h[i]))
    # So dc_k/dh[j] = (1-h[j-k])*I[0<=j-k<n] - h[j+k]*I[0<=j+k<n]
    
    for j in range(n):
        val = 0.0
        jm = j - k  # for (1-h[j-k])
        if 0 <= jm < n:
            val += (1.0 - h[jm])
        jp = j + k  # for h[j+k]
        if 0 <= jp < n:
            val -= h[jp]
        g[j] = val
    return g

# Vectorized version
def subgrad_corr_k_vec(h, k):
    """Vectorized subgradient of corr_k."""
    n = len(h)
    g = np.zeros(n)
    # (1-h[j-k]) for valid j-k
    lo1 = max(0, k)  # j >= k for j-k >= 0
    hi1 = min(n, n + k)  # j-k < n => j < n+k
    if hi1 > lo1:
        g[lo1:hi1] += (1.0 - h[lo1 - k:hi1 - k])
    # -h[j+k] for valid j+k
    lo2 = max(0, -k)  # j >= -k for j+k >= 0
    hi2 = min(n, n - k)  # j+k < n => j < n-k
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 + k:hi2 + k]
    return g

def cycling_gradient(h, n, tmax, niter, topK, alpha0, alpha_decay, rng, t0, t_deadline):
    """Cycling projected subgradient on top-K active lags."""
    tgt = 0.5 * n
    h = h.copy()
    c_cur = eval_c5(h)
    h_best = h.copy()
    c_best = c_cur
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    for it in range(niter):
        if time.time() > t_deadline:
            break
        
        # Find top K lags (recompute every 100 steps for efficiency)
        if it % 100 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_idx = np.argsort(corr)[-topK:]
        
        # Average subgradient over top K lags (weighted by correlation value)
        w = np.exp(20.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        
        g_avg = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k_actual = int(idx) - (n - 1)
            g_avg += wi * subgrad_corr_k_vec(h, k_actual)
        
        # Step size (Polyak-like)
        alpha = alpha0 / (1.0 + alpha_decay * it)
        
        # Gradient descent step (we want to minimize max corr, so descend on each active corr)
        h_try = h - alpha * g_avg
        h_try = project_feasible(h_try, n)
        
        c_try = eval_c5(h_try)
        if c_try < c_best:
            h_best, c_best = h_try.copy(), c_try
            h = h_try
        elif c_try < c_cur + 0.005:
            if rng.random() < 0.4:
                h, c_cur = h_try, c_try
        else:
            h = h_best.copy()  # revert
            c_cur = c_best
    
    return h_best, c_best
