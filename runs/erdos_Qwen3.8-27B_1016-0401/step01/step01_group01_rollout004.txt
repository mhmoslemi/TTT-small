                c5 = compute_c5(xn1, dx1)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = xn1.copy()
                    n0 = n1
                    dx = dx1
                    T = T1
            except Exception:
                pass

    # Final polish at best resolution
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    T_f = n_f / 2.0
    if _rem() > 20:
        try:
            xn = _slsqp(best_h, dx_f, T_f, 5000, sharp=True)
            c5 = compute_c5(xn, dx_f)
            if c5 < best_c5:
                best_c5, best_h = c5, xn.copy()
        except Exception:
            pass

    # Feasibility fix
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = T_f - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 0.001) & (best_h < 0.999)
        nf = int(np.sum(free))
        if nf > 0:
            best_h[free] += diff / nf
        else:
            best_h += diff / n_f
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, n_f
