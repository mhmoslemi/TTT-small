import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        nv = n - s
        g[:nv] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:nv]
    else:
        s2 = -s
        nv = n - s2
        if s2 > 0:
            g[s2:] += (1.0 - h[:nv])
        g[:nv] -= h[s2:]
    return g

def smooth_obj(h, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    return dx * (cm + t * np.log(np.sum(np.exp((c - cm) / t))))

def smooth_grad(h, dx, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    w = np.exp((c - cm) / t)
    ws = np.sum(w)
    g = np.zeros(n)
    for m in range(len(c)):
        if w[m] > 1e-15:
            s = m - (n - 1)
            g += (w[m] / ws) * grad_shift(h, s)
    return dx * g

def project_simplex_box(h, T):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > T:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_step(h, dx, T, alpha, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_k]
    w = c[idx].astype(float)
    w = np.maximum(w - w[-1] + 1.0, 0.1)
    w /= w.sum()
    g = np.zeros(n)
    for i, m in enumerate(idx):
        g += w[i] * grad_shift(h, m - (n - 1))
    g *= dx
    return project_simplex_box(h - alpha * g, T)

    T = n / 2.0
    dx = 2.0 / n
    h = project_simplex_box(h_init, T).copy()
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return x.sum() - T
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    temps = [0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005]
    for temp in temps:
        if time.time() > t0_budget:
            break
        def sobj(x):
            return smooth_obj(x, dx, temp)
        def sjac(x):
            return smooth_grad(x, dx, temp)
        try:

            hc = project_simplex_box(r.x, T)
