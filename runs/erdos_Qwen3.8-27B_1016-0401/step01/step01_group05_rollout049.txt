import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        cnt = np.count_nonzero(mask)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = alpha * (corr - m)
    vals = np.clip(vals, -50, 50)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

def slsqp_optimize(h, target, alpha, maxiter=200):
    n = len(h)
    def obj(x):
        return soft_max_c5(x, alpha)
    def constraint(x):
        return np.sum(x) - target
    try:
        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return make_feasible(res.x, target)
    except Exception:
        return h.copy()

def simulated_annealing(h_init, target, rng, deadline, t0=0.5, cool_rate=0.9997, max_steps=200000):
    n = len(h_init)
    h_curr = h_init.copy()
    s_curr = compute_c5(h_curr)
    h_best = h_curr.copy()
    s_best = s_curr
    T = t0
    step_size = 0.05
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        if step % 1000 == 0:
            T *= cool_rate
            step_size = max(0.01, 0.05 * (0.95 ** (step // 5000)))

        s_try = compute_c5(h_try)

                s_best = s_try

                if rng.random() < ratio:
                    h_curr = h_try
                    s_curr = s_try
    
    return h_best, s_best

def make_feasible(h, target):

            break

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')

        return h.copy()
