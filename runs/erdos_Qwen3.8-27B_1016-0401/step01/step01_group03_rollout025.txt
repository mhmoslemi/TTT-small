        j = zeros_idx[rng.integers(len(zeros_idx))]

        h[j] = 1.0

        delta = new_e - cur_e

                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_mask[i] = True
            ones_mask[j] = False

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def compute_c5(h, dx):
    return compute_max_corr(h) * dx

def make_binary(h0, target_ones, rng):
    h = (h0 > 0.5).astype(float).copy()
    oc = int(round(np.sum(h)))
    if oc > target_ones:
        idxs = np.where(h == 1)[0].copy()
        rng.shuffle(idxs)
        for i in idxs[:oc - target_ones]:
            h[i] = 0.0
    elif oc < target_ones:
        idxs = np.where(h == 0)[0].copy()
        rng.shuffle(idxs)
        for i in idxs[:target_ones - oc]:
            h[i] = 1.0
    return h
