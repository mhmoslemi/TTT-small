These are NOT the same in general (different linear parts), but the quadratic parts are the same.

Gradient for k = +p (p > 0):
c = Σ_{i=0}^{n-1-p} h[i+p] - h[i+p]h[i]
∂c/∂h[j]:
- If j is in [p, n-1]: appears as h[j] in the first sum with coeff 1, and in the quadratic with coeff -h[j-p]
- If j is in [0, n-1-p]: appears in -h[j+p]h[j] with coeff -h[j+p]
So: ∂c/∂h[j] = [j≥p] - h[j-p]*[j≥p] - h[j+p]*[j≤n-1-p]
= (1-h[j-p])*[j≥p] - h[j+p]*[j≤n-1-p]

Gradient for k = -p (p > 0):
c = Σ_{j=0}^{n-1-p} h[j] - h[j]h[j+p]
∂c/∂h[j]:
- If j is in [0, n-1-p]: appears as h[j] with coeff 1, and in -h[j]h[j+p] with coeff -h[j+p]
- If j is in [p, n-1]: appears in -h[j-p]h[j] with coeff -h[j-p]
So: ∂c/∂h[j] = [j≤n-1-p] - h[j+p]*[j≤n-1-p] - h[j-p]*[j≥p]
= (1-h[j+p])*[j≤n-1-p] - h[j-p]*[j≥p]

For k=0:

∂c/∂h[j] = 1 - 2h[j]

    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += (1.0 - x[:n-o])
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += (1.0 - x[p:])
        g[p:] += -x[:n-p]
    return g * dx

def make_cons():
    return [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Active-set cycling (top lags) - 40% budget
    p1_end = t0 + budget_s * 0.40
    for outer in range(60):
        if elapsed() > p1_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        center = n - 1
        top_k = min(5, 2*n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        for lag in top_idx:
            if elapsed() > p1_end:
                break
            _TARGET = target
            cons = make_cons()
            try:
                res = minimize(lag_obj, h_best, args=(lag,), method='SLSQP',
                              jac=lambda x, l=lag: lag_grad(x, l),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 800, 'ftol': 1e-16})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 2: Soft-max at increasing T - 20% budget
    p2_end = t0 + budget_s * 0.60
    for T in [80, 160, 320, 640, 1280, 2560]:
        if elapsed() > p2_end:
            break
        _TARGET = target
        cons = make_cons()
        try:
            res = minimize(sm_obj, h_best, args=(T,), method='SLSQP',
                          jac=lambda x, t=T: sm_grad(x, t),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Subgradient Nesterov polish - 15% budget
    p3_end = t0 + budget_s * 0.75
    h_sg = h_best.copy()
    h_mom = h_best.copy()
    L = 50.0
    center = n - 1
    for it in range(5000):
        if elapsed() > p3_end:
            break
        corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        worst = np.argmax(corr)
        o = worst - center
        g = np.zeros(n)
        if o == 0:
            g[:] = (1.0 - 2.0 * h_sg)
        elif o > 0:
            g[o:] += (1.0 - h_sg[:n-o])
            g[:n-o] += -h_sg[o:]
        else:
            p = -o
            g[:n-p] += (1.0 - h_sg[p:])
            g[p:] += -h_sg[:n-p]
        g *= 2.0 / n
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        h_new_sg = h_mom - (L / t_k) * g
        h_new_sg = fix_constraint(h_new_sg, target)
        h_mom = h_new_sg + ((L - 1.0) / t_k) * (h_new_sg - h_sg)
        h_mom = np.clip(h_mom, 0.0, 1.0)
        h_sg = h_new_sg
        L += 1.0
        if it % 500 == 0:
            c5_sg = compute_c5(h_sg)
            if c5_sg < c5_best:
                c5_best = c5_sg
                h_best = h_sg.copy()

    # Phase 4: Alternate n values - 15% budget
    p4_end = t0 + budget_s * 0.90
    for n_alt in [252, 308, 350, 420]:
        if elapsed() > p4_end:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        _TARGET = target_alt
        cons_alt = make_cons()
        for outer in range(20):
            if elapsed() > p4_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = np.argmax(corr_a)
            try:
                res = minimize(lag_obj, h_alt, args=(worst_a,), method='SLSQP',
                              jac=lambda x, l=worst_a: lag_grad(x, l),
                              bounds=bounds_alt, constraints=cons_alt,
                              options={'maxiter': 500, 'ftol': 1e-15})
                h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-16:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        _TARGET = len(h_alt) / 2.0
        cons_alt2 = make_cons()
        try:
            res = minimize(sm_obj, h_alt, args=(400,), method='SLSQP',
                          jac=lambda x: sm_grad(x, 400),
                          bounds=bounds_alt, constraints=cons_alt2,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt = h_new
        except Exception:
            pass
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt

    # Phase 5: Final active-set polish on best - 10% budget
    p5_end = t0 + budget_s - 10.0
    n_f = len(h_best)
    target_f = n_f / 2.0
    _TARGET = target_f
    bounds_f = [(0.0, 1.0)] * n_f
    for _ in range(20):
        if elapsed() > p5_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = np.argmax(corr_f)
        cons_f = make_cons()
        try:
            res = minimize(lag_obj, h_best, args=(worst_f,), method='SLSQP',
                          jac=lambda x, l=worst_f: lag_grad(x, l),
                          bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 2000, 'ftol': 1e-17})
            h_new = fix_constraint(np.clip(res.x, 0, 1), target_f)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
