def compute_c5(h):

def make_binary(h, target):

    return h_b

def sa_loop(h, n, target, rng, deadline, temp0, cool):
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    temp = temp0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        # Ensure h[i]=1, h[j]=0
        if h[i] < h[j]:
            i, j = j, i
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        d = new_c5 - c5
        if d < 0 or rng.rand() < np.exp(-d / max(temp, 1e-12)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        temp *= cool
    return best_h, best_c5

    best_h = h.copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(h, T):
    """Compute smooth max value and its gradient w.r.t. h."""
    n = len(h)
    corr = corr_vec(h)
    mx = corr.max()
    shifted = T * (corr - mx)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum  # softmax weights
    smooth_val = mx + np.log(w_sum) / T
    
    # Gradient: sum_k w_k * dcorr_k/dh
    # dcorr_k/dh[i] for shift s = k-(n-1):
    # = [0<=i-s<n ? (1-h[i-s]) : 0] - [0<=i+s<n ? h[i+s] : 0]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        s = k - (n - 1)
        if s >= 0:
            # (1-h[i-s]) for i in [s, n-1]
            i_lo, i_hi = s, n
            grad[i_lo:i_hi] += w[k] * (1.0 - h[:n-s])
            # -h[i+s] for i in [0, n-1-s]
            grad[:n-s] -= w[k] * h[s:n]
        else:
            ss = -s
            # (1-h[i+ss]) for i in [0, n-1-ss]
            grad[:n-ss] += w[k] * (1.0 - h[ss:n])
            # -h[i-ss] for i in [ss, n-1]
            grad[ss:n] -= w[k] * h[:n-ss]
    
    return smooth_val, grad * T

def project(h, S):
    """Project h to satisfy 0<=h<=1 and sum(h)=S."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - S) > 1e-10:
        # Add uniform adjustment
        h += (S - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
        # If still off, do a more careful projection
        if abs(h.sum() - S) > 1e-8:
            # Water-filling
            diff = S - h.sum()
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_descent(h, n, S, T, lr, max_iter, deadline):
    """Projected gradient descent on smooth max."""
    h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, g = smooth_max_grad(h, T)
        h_new = h - lr * g
        h = project(h_new, S)
    return h
