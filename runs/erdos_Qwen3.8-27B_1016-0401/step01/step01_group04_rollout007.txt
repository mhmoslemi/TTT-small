def make_binary(h, S):
    """Convert continuous h to binary with exactly int(S) ones."""
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])[:ones - target]
        h_b[idx[order]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])[:target - ones]
        h_b[idx[order]] = 1
    return h_b

    """First-improvement local search for binary. Returns (best_h, best_c5)."""
    n = len(h)
    h_ls = h.astype(float).copy()
    c5 = compute_c5(h_ls)
    best_h = h_ls.copy()
    best_c5 = c5

            break

            if time.time() > t_deadline:
                break

                    break

            break

        h_cur = h_pert

    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return float(mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T)

    def obj(x):
        return smooth_max_eval(x, T)
    def cons_fun(x):
        return x.sum() - S
    def cons_grad(x):
        return np.ones(n)

        h_out = np.clip(res.x, 0, 1)
        # Fix sum
        diff = S - h_out.sum()
        h_out += diff / n
        h_out = np.clip(h_out, 0, 1)
        return h_out
    except:
        return h.copy()

            t_sub = min(time.time() + budget_s * 0.04, t_p3)
            h_ls, c5_ls = ls_binary(h_r, t_sub, max_rounds=5000)
            if c5_ls < best_c5:
                best_h = h_ls
                best_c5 = c5_ls
            if time.time() < t_p3 - 10:
                h_ils2, c5_ils2 = ils_binary(h_ls, rng, t_p3, n_perturb=2)
                if c5_ils2 < best_c5:
                    best_h = h_ils2
                    best_c5 = c5_ils2

    # Phase 4: Final SLSQP refinement on best solution
    if time.time() < t0 + budget_s * 0.95:
        h_cont = best_h.astype(float).copy()
        n_c = len(h_cont)
        S_c = n_c / 2.0
        if np.allclose(h_cont, (h_cont >= 0.5).astype(float)):
            h_cont = h_cont + rng.normal(0, 0.08, n_c)
            h_cont = np.clip(h_cont, 0, 1)
            h_cont += (S_c - h_cont.sum()) / n_c
            h_cont = np.clip(h_cont, 0, 1)
        for T in [200, 800, 3000]:
            if time.time() > t0 + budget_s * 0.96:
                break
            h_new = slsqp_optimize(h_cont, n_c, S_c, T, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_cont = h_new

    # Phase 5: Check binary version of final
    if time.time() < t0 + budget_s * 0.985:
        h_fb = make_binary(best_h, len(best_h) / 2)
        c5_fb = compute_c5(h_fb)
        if c5_fb < best_c5:
            best_h = h_fb
            best_c5 = c5_fb
        # Quick final LS
        t_fin = t0 + budget_s * 0.99
        h_fin, c5_fin = ls_binary(best_h, t_fin, max_rounds=2000)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    target_sum = n_out / 2
    if abs(best_h.sum() - target_sum) > 1e-10:
        best_h = make_binary(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_out)
