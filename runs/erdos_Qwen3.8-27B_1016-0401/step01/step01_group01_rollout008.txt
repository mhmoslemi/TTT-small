    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    return dx * (c_max + t * np.log(np.sum(np.exp((c - c_max) / t))))

def smooth_gradient(h, dx, t):

    w /= np.sum(w)

    grad = np.zeros(n)

        grad[j] = dx * np.dot(w[j:j+n], u)
    return grad

    lo, hi = -1.0, 2.0  # lambda range

    def obj(x):
        return smooth_obj(x, dx, t)
    def jac(x):
        return smooth_grad(x, dx, t)
    def cons_fun(x):
        return x.sum() - target_sum
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_init.copy()
    best_x = h_init.copy()
    best_f = compute_c5(best_x, dx)
    try:
        r = minimize(obj, h_cur, method='SLSQP', jac=jac,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        c = compute_c5(r.x, dx)
        if c < best_f:
            best_x, best_f = r.x.copy(), c
    except Exception:
        pass
    return project_feasible(best_x, target_sum), best_f

def proj_grad(h_init, dx, t_schedule, target_sum, max_iter, alpha0=0.05):
    n = len(h_init)
    h = h_init.copy()
    alpha = alpha0
    for it in range(max_iter):
        t = t_schedule[min(it // max(1, max_iter // len(t_schedule)), len(t_schedule) - 1)]
        grad = smooth_grad(h, dx, t)
        grad -= grad.mean()
        f_val = smooth_obj(h, dx, t)
        improved = False
        for _ in range(40):
            h_trial = project_feasible(h - alpha * grad, target_sum)
            f_trial = smooth_obj(h_trial, dx, t)
            if f_trial < f_val - 1e-15:
                h = h_trial
                f_val = f_trial
                improved = True
                break
            alpha *= 0.5
        if not improved:
            alpha *= 0.5
        else:
            alpha = min(alpha * 1.02, 0.2)
    return project_feasible(h, target_sum)

def coord_descent(h_init, dx, target_sum, n_sweeps=50):
    n = len(h_init)
    h = h_init.copy()
    c5 = compute_c5(h, dx)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            old_val = h[j]
            for lo, hi in [(0.0, 1.0)]:
                best_v = old_val
                best_c = c5
                for _ in range(30):
                    if hi - lo < 1e-7:
                        break
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    h[j] = m1
                    h = project_feasible(h, target_sum)
                    c1 = compute_c5(h, dx)
                    h[j] = m2
                    h = project_feasible(h, target_sum)
                    c2 = compute_c5(h, dx)
                    if c1 < c2:
                        hi = m2
                        if c1 < best_c:
                            best_c, best_v = c1, m1
                    else:
                        lo = m1
                        if c2 < best_c:
                            best_c, best_v = c2, m2
                h[j] = best_v
                h = project_feasible(h, target_sum)
                hc = compute_c5(h, dx)
                if hc < c5 - 1e-12:
                    c5 = hc
                    improved = True
                else:
                    h[j] = old_val
                    h = project_feasible(h, target_sum)
        if not improved:
            break
    return h, c5

def upscale(h_in, n_out):
    n_in = len(h_in)
    if n_in == n_out:
        return h_in.copy()
    x_in = np.linspace(0, 2, n_in, endpoint=False) + 1.0 / n_in
    x_out = np.linspace(0, 2, n_out, endpoint=False) + 1.0 / n_out
    h_out = np.interp(x_out, x_in, h_in)
    return h_out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def time_left():
        return budget_s - elapsed()
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except (NameError, TypeError):
        n = 70
        h0 = np.full(n, 0.5)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP with multiple t at native resolution
    for t_val in [0.1, 0.05, 0.02, 0.01]:
        if time_left() < 120:
            break
        h_try, c = slsqp_smooth(best_h, dx, t_val, target_sum, 1500, 1e-13)
        if c < best_c5:
            best_c5, best_h = c, project_feasible(h_try, target_sum)

    # Phase 2: Projected gradient with annealing
    if time_left() > 100:
        t_sched = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005] * 2
        h_pg = proj_grad(best_h, dx, t_sched, target_sum, 600, alpha0=0.03)
        c = compute_c5(h_pg, dx)
        if c < best_c5:
            best_c5, best_h = c, h_pg

    # Phase 3: Coordinate descent polish
    if time_left() > 80:
        h_cd, c_cd = coord_descent(best_h, dx, target_sum, n_sweeps=40)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd

    # Phase 4: Try higher resolution (2x)
    n2 = n * 2
    dx2 = 2.0 / n2
    ts2 = n2 / 2.0
    if time_left() > 200:
        h2 = upscale(best_h, n2)
        h2 = project_feasible(h2, ts2)
        for t_val in [0.1, 0.03, 0.01]:
            if time_left() < 80:
                break
            h2, c = slsqp_smooth(h2, dx2, t_val, ts2, 1200, 1e-13)
            if c < best_c5:
                best_c5, best_h = c, project_feasible(h2, ts2)
                if time_left() < 120:
                    break
        if time_left() > 100:
            h2 = proj_grad(h2, dx2, [0.2, 0.08, 0.03, 0.01] * 2, ts2, 400, 0.02)
            c = compute_c5(h2, dx2)
            if c < best_c5:
                best_c5, best_h = c, h2
        if time_left() > 60:
            h2, c_cd = coord_descent(h2, dx2, ts2, n_sweeps=25)
            if c_cd < best_c5:
                best_c5, best_h = c_cd, h2

    # Phase 5: Random restarts with structured perturbations
    n_restarts = 20
    for i in range(n_restarts):
        if time_left() < 60:
            break
        # Perturb best solution
        h_r = best_h.copy()
        mask = rng.random(len(h_r)) < 0.3
        h_r[mask] += rng.normal(0, 0.15, mask.sum())
        h_r = project_feasible(h_r, target_sum)
        h_r, c = slsqp_smooth(h_r, dx, 0.02, target_sum, 600, 1e-12)
        if c < best_c5:
            best_c5, best_h = c, h_r
        if time_left() < 30:
            break

    # Phase 6: Final polish
    if time_left() > 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        h_f, c = slsqp_smooth(h_f, dx_f, 0.005, ts_f, 5000, 1e-15)
        if c < best_c5:
            best_c5, best_h = c, h_f
        h_f, c_cd = coord_descent(best_h, dx_f, ts_f, n_sweeps=60)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_f

    # Final feasibility check
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    best_h = project_feasible(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, n_f
