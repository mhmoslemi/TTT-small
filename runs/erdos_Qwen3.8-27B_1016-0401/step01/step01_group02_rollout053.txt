    def _penalty_obj(x):
        x_c = np.clip(x, 0.0, 1.0)
        base = _soft_obj(x_c, s)
        penalty = 1000.0 * (x_c.sum() - 0.5 * n) ** 2
        return base + penalty

    def _penalty_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        g = _soft_grad(x_c, s)
        g += 2000.0 * (x_c.sum() - 0.5 * n)
        return g

    try:
        res = minimize(_penalty_obj, h0, jac=_penalty_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxfun': 80000})
        if time.time() - t0 < tmax:
            return _feasible(res.x, n)
    except Exception:
        pass
    return _feasible(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(180) * 0.5
    n0 = len(h0)
    h0 = _feasible(h0, n0)
    best_h = h0.copy()
    best_c = _c5(h0)

    def _rem():
        return budget_s - (time.time() - t_start)

    # Phase 1: PGD with moderate scales
    if _rem() > 100:
        tmax1 = min(_rem() * 0.3, 300.0)
        h1, c1 = _pgd(best_h, n0, tmax1, [10.0, 30.0, 60.0])
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: L-BFGS with penalty
    rem = _rem()
    if rem > 80:
        h2 = _lbfgs_pass(best_h, n0, 40.0, min(rem * 0.25, 200.0))
        c2 = _c5(h2)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Annealing
    rem = _rem()
    if rem > 50:
        tmax3 = min(rem * 0.25, 180.0)
        h3, c3 = _anneal(best_h, n0, tmax3, 1500000)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: PGD with higher scales
    rem = _rem()
    if rem > 100:
        tmax4 = min(rem * 0.35, 300.0)
        h4, c4 = _pgd(best_h, n0, tmax4, [60.0, 120.0, 250.0])
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Coordinate descent
    rem = _rem()
    if rem > 40:
        tmax5 = min(rem * 0.25, 150.0)
        h5, c5 = _coord_opt(best_h, n0, tmax5, 50000)
        if c5 < best_c:
            best_h, best_c = h5, c4 if False else c5

    # Phase 6: Shake + reoptimize
    rem = _rem()
    if rem > 120:
        for trial in range(5):
            if _rem() < 60:
                break
            shake = best_h + np.random.normal(0, 0.03, n0)
            shake = _feasible(shake, n0)
            th = min(_rem() * 0.2, 120.0)
            hs, cs = _pgd(shake, n0, th, [80.0, 150.0])
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 7: Final anneal
    rem = _rem()
    if rem > 30:
        tmax7 = min(rem * 0.4, 200.0)
        h7, c7 = _anneal(best_h, n0, tmax7, 2000000)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Final verification
    best_h = _feasible(best_h, n0)
    final_c = _c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return (best_h, best_c, n0)
