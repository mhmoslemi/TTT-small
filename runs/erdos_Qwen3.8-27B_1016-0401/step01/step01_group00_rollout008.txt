def corr_and_grad(h, n, T):
    """Compute soft-max objective and gradient."""
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se  # softmax weights over lags
    obj = (mc + np.log(se) / T) * (2.0 / n)
    
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[:n-o] += w * (1.0 - h[o:])  # d/dh[j] of h[j](1-h[j+o])

        else:

            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]

    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def _sm_obj(x):
    T, mu = _STATE['T'], _STATE['mu']
    n = _STATE['n']
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -500, 0)
    e = np.exp(sh)
    se = np.sum(e)
    obj = (mc + np.log(se) / T) * (2.0 / n)
    if mu > 0:
        obj += mu * (np.sum(x) - n / 2.0) ** 2
    return obj

def _sm_grad(x):
    T, mu = _STATE['T'], _STATE['mu']
    n = _STATE['n']
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -500, 0)
    e = np.exp(sh)
    se = np.sum(e)
    sw = e / se
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[:n - o] += w * (1.0 - x[o:])

        else:

            grad[p:] -= w * x[:n - p]
    grad *= dx

    return grad
