def two_coord_descent(h, n, target, rng, max_rounds, deadline, grid_size=80):
    dx = 2.0 / n
    h_curr = h.copy().astype(float)
    current_c5 = compute_c5(h_curr)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        perm = rng.permutation(n)
        
        for pi in range(n - 1):
            if time.time() > deadline:
                break
            i = int(perm[pi])
            j = int(perm[(pi + 1) % n])
            if i == j:
                continue
            
            c = target - (np.sum(h_curr) - h_curr[i] - h_curr[j])
            t_lo = max(0.0, c - 1.0)
            t_hi = min(1.0, c)
            if t_lo >= t_hi - 1e-12:
                continue
            
            old_i = h_curr[i]
            old_j = h_curr[j]
            
            ts = np.linspace(t_lo, t_hi, grid_size)
            best_t = old_i
            best_val = current_c5
            
            for tv in ts:
                h_curr[i] = tv
                h_curr[j] = c - tv
                val = compute_c5(h_curr)
                if val < best_val:
                    best_val = val
                    best_t = tv
            
            h_curr[i] = best_t
            h_curr[j] = c - best_t
            h_curr = np.clip(h_curr, 0.0, 1.0)
            
            if best_val < current_c5 - 1e-14:
                current_c5 = best_val
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 50:
                # Perturb to escape local optimum
                idx = rng.randint(n)
                h_curr[idx] += rng.uniform(-0.05, 0.05)
                h_curr = make_feasible(h_curr, n, target)
                current_c5 = compute_c5(h_curr)
                no_improve = 0
        else:
            no_improve = 0
    
    h_curr = make_feasible(h_curr, n, target)
    return h_curr, compute_c5(h_curr)

def slsqp_optimize(h, n, target, alpha, maxiter=200, deadline=None):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    
    def constraint(x):
        return np.sum(x) - target
    
    try:
        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = make_feasible(res.x, n, target)
        return h_new
    except Exception:
        return h.copy()

def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    x_old = np.linspace(0, 2, n + 1)[:-1] + 1.0/n
    x_new = np.linspace(0, 2, n_new + 1)[:-1] + 1.0/n_new
    h_new = np.interp(x_new, np.linspace(0, 2, n + 1)[:-1] + 1.0/n, h, period=2)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

def subgradient_step(h, n, target, step_size, rng, deadline):
    dx = 2.0 / n
    h_curr = h.copy().astype(float)
    for _ in range(500):
        if time.time() > deadline:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        m = np.max(corr)
        active = np.where(corr > m - 0.01 * m)[0]
        grad = np.zeros(n)
        for ki in active:
            lag = ki - (n - 1)
            for idx in range(n):
                p1 = idx + lag
                if 0 <= p1 < n:
                    grad[idx] += (1.0 - h_curr[p1])
                p2 = idx - lag
                if 0 <= p2 < n:
                    grad[idx] -= h_curr[p2]
        grad = -grad * dx / len(active)
