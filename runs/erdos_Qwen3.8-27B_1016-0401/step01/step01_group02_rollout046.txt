import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def _lse(h, dx, scale):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.exp(scale * (corr - m)).sum()) / scale) * dx

def _lse_grad(h, dx, scale):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(scale * (corr - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - h[lag:n])
            grad[lag:] -= wk * h[:n - lag]
        else:
            l = -lag
            grad[l:] += wk * (1.0 - h[:n - l])
            grad[:n - l] -= wk * h[l:]
    
    return grad * dx * scale

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _pgd(h0, n, dx, max_iter, t0, tmax):
    h = h0.copy()
    best = h.copy()
    bc = _c5(best)
    for scale in [30.0, 60.0, 120.0, 250.0]:
        step = 1.0 / (scale * dx * 2 * n)  # Lipschitz-ish bound
        for it in range(max_iter):
            if time.time() - t0 > tmax:

            g = _lse_grad(h, dx, scale)

                if _lse(h_new, dx, scale) < _lse(h, dx, scale) + 1e-12:

                step_local *= 0.5
            h = h_new
            c = _c5(h)

            if c_try < bc:
                bc = c_try
                best = h2.copy()
                h, c_cur = h2, c_try
            elif c_try < c_cur or rng.random() < np.exp(-(c_try - c_cur) / max(temp, 1e-8)):
                h, c_cur = h2, c_try
            else:
                h[:] = best
                c_cur = bc
        temp *= 0.97
        if temp < 0.0005:
            temp = 0.0005
    return best, bc

def _block_search(h0, n, t0, tmax, seed):
    rng = np.random.RandomState(seed)
    tgt = 0.5 * n
    best = h0.copy()
    bc = _c5(best)
    h = h0.copy()
    c_cur = bc
    bs = max(3, n // 40)
    while time.time() - t0 < tmax:
        for _ in range(500):
            i = rng.randint(0, n - bs)
            j = rng.randint(0, n - bs)
            if abs(i - j) < bs:
                continue
            sgn = rng.choice([1.0, -1.0])
            dv = sgn * rng.uniform(0.01, 0.1)
            h2 = h.copy()
            h2[i:i+bs] = np.clip(h2[i:i+bs] + dv, 0.0, 1.0)
            h2[j:j+bs] = np.clip(h2[j:j+bs] - dv, 0.0, 1.0)
            ds = h2.sum() - tgt
            if abs(ds) > 1e-14:
                h2 -= ds / n
                np.clip(h2, 0.0, 1.0, out=h2)
            c_try = _c5(h2)
            if c_try < bc:
                bc = c_try
                best = h2.copy()
                h, c_cur = h2, c_try
            elif c_try < c_cur + 0.002:
                if rng.random() < 0.3:
                    h, c_cur = h2, c_try
            else:
                h[:] = best
                c_cur = bc
    return best, bc

def _optimize_resolution(h0, n, dx, t0, tmax, seed):
    bh = h0.copy()
    bc = _c5(bh)
    rem = tmax - (time.time() - t0)
    for s, pen, mi in [(15, 30, 400), (40, 80, 400), (80, 200, 300), (150, 500, 250)]:
        if time.time() - t0 > tmax - 30:
            break
        hh = _lbfgsb(bh, n, dx, s, pen, mi, t0, tmax - 15)
        cc = _c5(hh)
        if cc < bc:
            bh, bc = hh, cc
    rem = tmax - (time.time() - t0)
    if rem > 20:
        hh, cc = _local_search(bh, n, t0, t0 + rem * 0.5, seed)
        if cc < bc:
            bh, bc = hh, cc
    rem = tmax - (time.time() - t0)
    if rem > 15:
        hh, cc = _block_search(bh, n, t0, t0 + rem * 0.8, seed + 1)
        if cc < bc:
            bh, bc = hh, cc
    return bh, bc

def _sinusoid(n, freq, amp, phase):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    # Get initial values
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = _proj(_sinusoid(180, 3, 0.3, 0.0), 180)

    n0 = len(h_init)
    dx0 = 2.0 / n0
    bh = h_init.copy()
    bc = _c5(bh)

    # Phase 1: optimize at initial resolution
    t_phase1 = min(t_start + budget_s * 0.45, t_end - 60)
    bh, bc = _optimize_resolution(bh, n0, dx0, t_start, t_phase1, seed)

    # Try an alternative start: sinusoidal pattern
    rem = t_end - (time.time() - t_start)
    if rem > 60 and n0 <= 200:
        h_alt = _proj(_sinusoid(n0, 5, 0.35, np.pi / 3), n0)
        c_alt = _c5(h_alt)
        t_alt_end = min(t_start + budget_s * 0.60, t_end - 50)
        h_alt, c_alt = _optimize_resolution(h_alt, n0, dx0, t_start, t_alt_end, seed + 100)
        if c_alt < bc:
            bh, bc = h_alt, c_alt

    # Phase 2: upscale to 2*n0
    rem = t_end - (time.time() - t_start)
    n2 = 2 * n0
    if n2 <= 720 and rem > 120:
        h_up = _proj(np.repeat(bh, 2), n2)
        t_phase2 = min(t_start + budget_s * 0.80, t_end - 80)
        h_up, c_up = _optimize_resolution(h_up, n2, 2.0 / n2, t_start, t_phase2, seed + 200)
        if c_up < bc:
            bh, bc, n0 = h_up, c_up, n2

    # Phase 3: final polish
    rem = t_end - (time.time() - t_start)
    if rem > 10:
        hh, cc = _local_search(bh, n0, t_start, t_start + rem * 0.7, seed + 500)
        if cc < bc:
            bh, bc = hh, cc
    rem = t_end - (time.time() - t_start)
    if rem > 10:
        hh, cc = _block_search(bh, n0, t_start, t_start + rem * 0.8, seed + 600)
        if cc < bc:
            bh, bc = hh, cc

    # Final micro-polish
    rem = t_end - (time.time() - t_start)
    if rem > 5:
        hh, cc = _local_search(bh, n0, t_start, t_start + rem * 0.8, seed + 700)
        if cc < bc:
            bh, bc = hh, cc

    bc = _c5(bh)
    return (bh, bc, len(bh))
