def _waterfill(h, n, tmax, seed):

    t0 = time.time()

        # corr[k] = sum_i h[i] * (1-h[i + k - (n-1)])
        # d(corr[k])/d(h[i]) = (1-h[i+k-(n-1)]) - h[i] * delta(i == j + k - (n-1))
        # This is complex. Let me use a numerical approach.
        
        # Perturb to reduce the peak
        # The peak is at max_k. We want to reduce the overlap at that lag.
        # corr[max_k] = sum_i h[i] * (1-h[i + max_k - (n-1)])
        # To reduce this, we want to make h[i] small where (1-h[i+lag]) is large,
        # or make (1-h[i+lag]) small where h[i] is large.

        # For negative lag: h[i] overlaps with 1-h[i+lag] where i+lag might be negative

        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                contrib[i] = h[i] * (1 - h[j])

        # But maintain sum constraint
        step = 0.05
        h2 = h.copy()
        # Reduce top contributors
        top_n = max(1, n // 10)
        top_idx = np.argsort(contrib)[-top_n:]
        h2[top_idx] = np.clip(h2[top_idx] - step, 0, 1)

Let me trace through the projection more carefully:
- Start with h in R^n
- Want: 0 ≤ h[i] ≤ 1 and sum(h) = n/2

Algorithm:
1. Clip to [0,1]
2. Compute deficit = n/2 - sum(h)
3. If deficit > 0: distribute deficit to elements that are < 1
4. If deficit < 0: remove |deficit| from elements that are > 0
5. Repeat until convergence

The issue is that after distributing, some elements might hit 0 or 1, requiring redistribution. The iterative approach handles this.

A more efficient projection:
1. Sort h
2. Find the threshold using bisection
3. Apply: h[i] = clip(h[i] + t, 0, 1) where t is chosen so sum = n/2

This is the "water-filling" algorithm, which is O(n log n) per iteration.

Let me implement this:

```python
def _project(h, n):
    tgt = 0.5 * n
    h = h.astype(float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-13:
            break
        # Bisection on t: find t such that sum(clip(h + t, 0, 1)) = tgt
        # clip(h+t, 0, 1) is monotone in t
        lo, hi = -h.max() - 1, 1 - h.min() + 1
        for _ in range(60):
            mid = (lo + hi) / 2
            s_mid = np.clip(h + mid, 0, 1).sum()
            if s_mid < tgt:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2, 0, 1)
    r = tgt - h.sum()
    if abs(r) > 1e-13:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
```

                h2[s1:s1 + bl] = np.clip(h2[s1:s1 + bl] + dv, 0.0, 1.0)
                h2[s2:s2 + bl] = np.clip(h2[s2:s2 + bl] - dv, 0.0, 1.0)
        h2 = _project(h2, n)
        fc2 = _c5(h2)
        df = fc2 - fc
        if df < 0 or (T > Tf and rng.random() < np.exp(-df / T)):
            h, fc = h2, fc2
            if fc2 < bestc:
                best, bestc = h2.copy(), fc2
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 8000:
            h = _project(best + rng.normal(0.0, 0.02, n), n)
            fc = _c5(h)
            stall = 0
    return best, bestc

def _upscale(h, nn):
    xo = np.linspace(0.0, 1.0, len(h))
    xn = np.linspace(0.0, 1.0, nn)
    return _project(np.interp(xn, xo, h), nn)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5
    n = len(h)
    h = _project(h, n)
    bh = h.copy()
    bhc = _c5(bh)
    bn = n

    # Phase 1: SLSQP at base resolution with multiple scales
    if budget_s - (time.time() - t0) > 60:
        bh, bhc = _slsqp(bh, bn, 2.0 / bn, [25.0, 50.0, 100.0, 200.0], 2500)

    # Phase 2: Anneal at base
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        ah, ac = _anneal(bh, bn, min(rem * 0.1, 45.0), seed)
        if ac < bhc:
            bh, bhc = ah, ac

    # Phase 3: Upscale to 2n, optimize
    n2 = 2 * bn
    rem = budget_s - (time.time() - t0)
    if n2 <= 720 and rem > 200:
        h2 = _upscale(bh, n2)
        c2 = _c5(h2)
        rem = budget_s - (time.time() - t0)
        if rem > 120:
            h2, c2 = _slsqp(h2, n2, 2.0 / n2, [50.0, 100.0, 200.0], 2500)
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            h2a, c2a = _anneal(h2, n2, min(rem * 0.3, 130.0), seed + 100)
            if c2a < c2:
                h2, c2 = h2a, c2a
        if c2 < bhc:
            bh, bhc, bn = h2, c2, n2

    # Phase 4: Upscale to 4*original, optimize
    n3 = 2 * bn
    rem = budget_s - (time.time() - t0)
    if n3 <= 720 and rem > 280:
        h3 = _upscale(bh, n3)
        c3 = _c5(h3)
        rem = budget_s - (time.time() - t0)
        if rem > 200:
            h3, c3 = _slsqp(h3, n3, 2.0 / n3, [80.0, 150.0], 2000)
        rem = budget_s - (time.time() - t0)
        if rem > 80:
            h3a, c3a = _anneal(h3, n3, min(rem * 0.35, 180.0), seed + 200)
            if c3a < c3:
                h3, c3 = h3a, c3a
        if c3 < bhc:
            bh, bhc, bn = h3, c3, n3

    # Phase 5: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        bh, bhc = _slsqp(bh, bn, 2.0 / bn, [100.0, 200.0], 1500)

    # Phase 6: Final anneal polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        af, cf = _anneal(bh, bn, min(rem * 0.55, 60.0), seed + 300)
        if cf < bhc:
            bh, bhc = af, cf

    bh = _project(bh, bn)
    fc = _c5(bh)
    return (bh, fc, bn)
