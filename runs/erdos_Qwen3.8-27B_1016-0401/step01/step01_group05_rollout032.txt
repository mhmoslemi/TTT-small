        hp = np.array(initial_h_values, dtype=float)
    else:
        hp = np.ones(70) * 0.5

    n = 200
    target = n / 2.0
    xp = np.linspace(0, 2, len(hp), endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    h0 = np.interp(xn, xp, hp, period=2.0)
    h0 = project_to_constraint(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Gradient optimization with increasing alpha
    for alpha in [20, 40, 80, 160, 320, 640, 1280]:
        if time.time() > deadline - 120:
            break
        lr = 3.0 / alpha
        for mu in [0.9, 0.7]:
            if time.time() > deadline - 120:
                break
            h_new, c5_new = pg_optimize(best_h, target, alpha, lr, mu, 400, deadline - 120)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Coordinate search
    if time.time() < deadline - 60:
        h_cd, c5_cd = coord_search(best_h, target, 80000, deadline - 60, rng)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Phase 3: High-alpha refinement
    if time.time() < deadline - 40:
        for alpha in [800, 1600, 3200]:
            if time.time() > deadline - 40:
                break
            lr = 1.5 / alpha
            h_new, c5_new = pg_optimize(best_h, target, alpha, lr, 0.5, 300, deadline - 40)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Alternative structured starts
    starts = [
        0.5 + 0.25 * np.cos(2.0 * np.pi * xn),
        np.where(xn < 1.0, 0.65, 0.35),
        0.5 + 0.15 * np.sin(4.0 * np.pi * xn) + 0.1 * np.cos(6.0 * np.pi * xn),
    ]
    for st in starts:
        if time.time() > deadline - 80:
            break
        h_alt = project_to_constraint(st, target)
        for alpha in [40, 150, 400]:
            if time.time() > deadline - 80:
                break
            lr = 3.0 / alpha
            h_alt, _ = pg_optimize(h_alt, target, alpha, lr, 0.9, 200, deadline - 80)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h = h_alt
            best_c5 = c5_alt

    # Phase 5: Final coordinate polish
    if time.time() < deadline - 10:
        h_cd, c5_cd = coord_search(best_h, target, 60000, deadline - 5, rng)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    best_h = project_to_constraint(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), final_c5, n
