import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h = h - d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] = h[msk] + r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_soft(h, S):
    """Returns (gradient, soft_objective) for soft-C5."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exp_s = np.exp(S * (corr - m))
    w = exp_s / exp_s.sum()
    obj = (m + np.log(exp_s.sum()) / S) * (2.0 / n)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        mk = k - (n - 1)
        # Term 1: (1-h[j-mk]) for j in [max(0,mk), min(n-1, n-1+mk)]
        j1_lo = max(0, mk)
        j1_hi = min(n - 1, n - 1 + mk)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += w[k] * (1.0 - h[j1_lo - mk:j1_hi - mk + 1])
        # Term 2: -h[j+mk] for j in [max(0,-mk), min(n-1, n-1-mk)]
        j2_lo = max(0, -mk)
        j2_hi = min(n - 1, n - 1 - mk)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += -w[k] * h[j2_lo + mk:j2_hi + mk + 1]
    return dx * grad, obj

def _pgd(h0, n, S, t_max, lr_init=0.03):
    """Projected gradient descent on soft-C5."""
    t0 = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = _c5(best)
    lr = lr_init
    grad, obj = _grad_soft(h, S)
    stale = 0
    
    while time.time() - t0 < t_max:
        # Backtracking
        for _ in range(10):
            h_new = h - lr * grad
            h_new = _proj(h_new, n)
            _, obj_new = _grad_soft(h_new, S)
            if obj_new < obj - 1e-12:
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-10)
            h_new = _proj(h - lr * grad, n)
        
        h = h_new
        grad, obj = _grad_soft(h, S)
        lr = min(lr * 1.02, 0.3)
        
        c_h = _c5(h)
        if c_h < best_c:
            best_c = c_h
            best = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 500:
                # Perturb and restart
                h = _proj(h + np.random.randn(n) * 0.05, n)
                stale = 0
    
    return best, best_c

def _local_search(h, n, t_max, block_sizes=(3, 5, 8)):
    """Block-based local search with simulated annealing."""
    t0 = time.time()
    best = h.copy()
    best_c = _c5(best)
    cur = h.copy()
    cur_c = best_c
    temp = 0.01
    
    while time.time() - t0 < t_max:
        B = int(block_sizes[np.random.randint(len(block_sizes))])
        if B >= n:
            B = n - 1
        start = int(np.random.randint(0, n - B + 1))
        idx = np.arange(start, start + B)
        
        h_new = cur.copy()
        h_new[idx] += np.random.randn(B) * 0.08
        h_new = _proj(h_new, n)
        c_new = _c5(h_new)
        
        if c_new < cur_c:
            cur, cur_c = h_new, c_new
            if c_new < best_c:
                best, best_c = h_new.copy(), c_new
        elif np.random.random() < np.exp(-(c_new - cur_c) / temp):
            cur, cur_c = h_new, c_new
        
        temp *= 0.9998
    
    return best, best_c

def _upscale(h, n_new):

    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        if r > 0:
            msk = h < 1.0
        else:
            msk = h > 0.0
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_obj(h, S):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exp_s = np.exp(S * (corr - m))
    wsum = exp_s.sum()
    w = exp_s / wsum
    obj = (m + np.log(wsum) / S) * (2.0 / n)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        mk = k - (n - 1)
        j1lo = max(0, mk)
        j1hi = min(n - 1, n - 1 + mk)
        if j1lo <= j1hi:
            grad[j1lo:j1hi + 1] += wk * (1.0 - h[j1lo - mk:j1hi - mk + 1])
        j2lo = max(0, -mk)
        j2hi = min(n - 1, n - 1 - mk)
        if j2lo <= j2hi:
            grad[j2lo:j2hi + 1] += -wk * h[j2lo + mk:j2hi + mk + 1]
    return dx * grad, obj

def _pgd(h0, n, S, t_max, lr_init=0.05):
    t0 = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = _c5(best)
    lr = lr_init
    grad, obj = _grad_obj(h, S)
    no_improve = 0
    
    while time.time() - t0 < t_max:
        # Backtracking line search
        for _ in range(8):
            step = h - lr * grad
            step = _proj(step, n)
            _, obj_new = _grad_obj(step, S)
            if obj_new < obj - 1e-13:
                break
            lr *= 0.5
        else:
            step = _proj(h - lr * grad, n)
            _, obj_new = _grad_obj(step, S)
        
        h = step
        obj = obj_new
        lr = min(lr * 1.03, 0.4)
        grad, obj = _grad_obj(h, S)
        
        c_now = _c5(h)
        if c_now < best_c - 1e-9:
            best = h.copy()
            best_c = c_now
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                h = _proj(h + 0.05 * np.random.randn(n), n)
                grad, obj = _grad_obj(h, S)
                lr = lr_init
                no_improve = 0
    
    if _c5(h) < best_c:
        best = h.copy()
        best_c = _c5(h)
    return best, best_c

def _local_search(h, n, t_max):
    t0 = time.time()
    best = h.copy()
    best_c = _c5(best)
    cur = h.copy()
    cur_c = best_c
    temp = 0.008

    return best, best_c

                cur, cur_c = h2, c2
                improved = True
                continue
            h2 = cur.copy()
            h2[i] = max(0.0, h2[i] - step)
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < cur_c:
                cur, cur_c = h2, c2
                improved = True
        if cur_c < best_c:
            best, best_c = cur.copy(), cur_c
        if not improved:
            step *= 0.6
            stagnation += 1
            if stagnation > 4 or step < 5e-6:
                break
        else:
            stagnation = 0
    return best, best_c

def _upscale(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return _proj(np.interp(x_new, x_old, h.ravel(), period=1.0), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    ts = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(180, 0.5)
    n = len(h)
    h = _proj(h, n)
    bh = h.copy()
    bc = _c5(bh)
    bn = n

    # Phase 1: PGD S=20 (smooth landscape)
    rem = budget_s - (time.time() - ts)
    if rem > 120:
        t1 = min(rem * 0.18, 170.0)
        r = _pgd(bh, bn, S=20, t_max=t1, lr_init=0.06, momentum=0.3)
        if r[1] < bc:
            bh, bc = r

    # Phase 2: PGD S=50
    rem = budget_s - (time.time() - ts)
    if rem > 120:
        t2 = min(rem * 0.18, 180.0)
        r = _pgd(bh, bn, S=50, t_max=t2, lr_init=0.04, momentum=0.5)
        if r[1] < bc:
            bh, bc = r

    # Phase 3: PGD S=100 (sharper focus on max)
    rem = budget_s - (time.time() - ts)
    if rem > 100:
        t3 = min(rem * 0.15, 160.0)
        r = _pgd(bh, bn, S=100, t_max=t3, lr_init=0.03, momentum=0.6)
        if r[1] < bc:
            bh, bc = r

    # Phase 4: Local search
    rem = budget_s - (time.time() - ts)
    if rem > 130:
        t4 = min(rem * 0.2, 250.0)
        r = _local_search(bh, bn, t_max=t4, base_amp=0.07)
        if r[1] < bc:
            bh, bc = r

    # Phase 5: Coordinate descent polish
    rem = budget_s - (time.time() - ts)
    if rem > 80:
        t5 = min(rem * 0.15, 120.0)
        r = _coord_descent(bh, bn, t_max=t5)
        if r[1] < bc:
            bh, bc = r

    # Phase 6: Upscale to 2n
    rem = budget_s - (time.time() - ts)
    n2 = 2 * bn
    if n2 <= 720 and rem > 320:
        h_u = _upscale(bh, n2)
        cu = _c5(h_u)
        t6 = min(rem * 0.14, 120.0)
        h_u, cu = _pgd(h_u, n2, S=40, t_max=t6, lr_init=0.05, momentum=0.4)
        rem = budget_s - (time.time() - ts)
        if rem > 100:
            t6b = min(rem * 0.12, 100.0)
            h_u, cu = _local_search(h_u, n2, t_max=t6b, base_amp=0.06)
            rem = budget_s - (time.time() - ts)
            if rem > 50:
                t6c = min(rem * 0.12, 80.0)
                h_u, cu = _coord_descent(h_u, n2, t_max=t6c)
        if cu < bc:
            bh, bc, bn = h_u, cu, n2

    # Phase 7: Final PGD high-S
    rem = budget_s - (time.time() - ts)
    if rem > 70:
        t7 = min(rem * 0.22, 180.0)
        r = _pgd(bh, bn, S=150, t_max=t7, lr_init=0.02, momentum=0.7)
        if r[1] < bc:
            bh, bc = r

    # Phase 8: Final local search
    rem = budget_s - (time.time() - ts)
    if rem > 40:
        t8 = min(rem * 0.5, 200.0)
        r = _local_search(bh, bn, t_max=t8, base_amp=0.04)
        if r[1] < bc:
            bh, bc = r

    # Final coordinate descent
    rem = budget_s - (time.time() - ts)
    if rem > 20:
        r = _coord_descent(bh, bn, t_max=min(rem * 0.5, 80.0))
        if r[1] < bc:
            bh, bc = r

    bh = _proj(bh, len(bh))
    fc = _c5(bh)
    return (bh, fc, len(bh))
