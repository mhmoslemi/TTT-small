```python
def nag_optimize(h0, n, target, alpha_steps, iters_per_alpha, deadline):
    h = h0.copy()
    h_prev = h.copy()
    t_nest = 1.0
    dx = 2.0 / n
    
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for alpha in alpha_steps:
        t_nest = 1.0
        h = best_h.copy()
        h_prev = h.copy()
        lr = 0.5 / (n * dx)  # initial learning rate, will adapt
        
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            # Nesterov momentum
            momentum = (t_nest - 1) / (t_nest + 1)
            y = h + momentum * (h - h_prev)
            y = np.clip(y, 0, 1)
            
            # Compute soft-max and gradient at y
            corr_y = np.correlate(y, 1.0 - y, mode='full')
            grad = soft_max_grad(corr_y, y, alpha) * dx
            
            # Gradient step
            y_new = y - lr * grad
            h_new = project_sum(y_new, target)
            
            # Check if objective decreased
            val_new = soft_max_value(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * dx
            val_old = soft_max_value(np.correlate(h, 1.0 - h, mode='full'), alpha) * dx
            
            if val_new > val_old:
                lr *= 0.5
                if lr < 1e-10:
                    break
                # Redo with smaller lr
                y_new = y - lr * grad
                h_new = project_sum(y_new, target)
            else:
                lr = min(lr * 1.05, 1.0/(alpha*dx))  # slight increase
                h_prev = h
                h = h_new
                t_nest = (1 + np.sqrt(1 + 4*t_nest**2)) / 2
                
                # Track best
                true_val = compute_c5(h)
                if true_val < best_val:
                    best_h = h.copy()
                    best_val = true_val
    
    return best_h, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_val(corr, alpha):
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -60, 60)
    return m + np.log(np.sum(np.exp(v))) / alpha

def sm_grad(corr, h, alpha):
    n = len(h)
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -60, 60)
    w = np.exp(v)
    w /= np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for s in range(2 * n - 1):
        shift = s - (n - 1)
        ws = w[s]
        if ws < 1e-14:
            continue
        if shift >= 0:
            grad[:n-shift] += ws * omh[shift:]
            grad[shift:] -= ws * h[:n-shift]
        else:
            sa = -shift
            grad[sa:] += ws * omh[:n-sa]
            grad[:n-sa] -= ws * h[sa:]
    return grad

def sub_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    max_idx = np.where(np.abs(corr - m) < 1e-12)[0]
    # Average gradient over all maximizers
    grad = np.zeros(n)
    omh = 1.0 - h
    for idx in max_idx:
        shift = idx - (n - 1)
        if shift >= 0:
            grad[:n-shift] += omh[shift:]
            grad[shift:] -= h[:n-shift]
        else:
            sa = -shift
            grad[sa:] += omh[:n-sa]
            grad[:n-sa] -= h[sa:]
    grad /= len(max_idx)
    return grad, m

def nag_loop(h0, n, target, alphas, iters, deadline, dx):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for alpha in alphas:
        h = best_h.copy()
        h_prev = h.copy()
        t_nest = 1.0
        lr = 1.0 / (alpha * dx * n)
        
        for it in range(iters):
            if time.time() > deadline:
                break
            mom = (t_nest - 1.0) / (t_nest + 1.0)
            y = h + mom * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            
            corr_y = np.correlate(y, 1.0 - y, mode='full')
            g = sm_grad(corr_y, y, alpha) * dx
            
            y_new = y - lr * g
            h_new = project_sum(y_new, target)
            
            # Backtracking
            val_new = soft_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * dx
            val_ref = soft_max_val(corr_y, alpha) * dx
            tries = 0
            while val_new > val_ref and tries < 10:
                lr *= 0.5
                y_new = y - lr * g
                h_new = project_sum(y_new, target)
                val_new = soft_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * dx
                tries += 1
            
            h_prev = h
            h = h_new
            t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            
            tv = compute_c5(h)
            if tv < best_c5:
                best_h = h.copy()
                best_c5 = tv
        
        # Update best after each alpha
        tv = compute_c5(h)
        if tv < best_c5:
            best_h = h.copy()
            best_c5 = tv
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_val(corr, alpha):
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -60, 60)
    return m + np.log(np.sum(np.exp(v))) / alpha

def sm_grad(corr, h, alpha):
    n = len(h)
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -60, 60)
    w = np.exp(v)
    w /= np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for s in range(2 * n - 1):
        shift = s - (n - 1)
        ws = w[s]
        if ws < 1e-14:
            continue
        if shift >= 0:
            grad[:n - shift] += ws * omh[shift:]
            grad[shift:] -= ws * h[:n - shift]
        else:
            sa = -shift
            grad[sa:] += ws * omh[:n - sa]
            grad[:n - sa] -= ws * h[sa:]
    return grad

def true_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    tol = 1e-10 * m
    max_idx = np.where(np.abs(corr - m) <= tol)[0]
    # Weight maximizers by proximity to max
    weights = np.clip(m - corr[max_idx], 0, None)
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(max_idx))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    omh = 1.0 - h
    for idx, wt in zip(max_idx, weights):
        shift = idx - (n - 1)
        if shift >= 0:
            grad[:n - shift] += wt * omh[shift:]
            grad[shift:] -= wt * h[:n - shift]
        else:
            sa = -shift
            grad[sa:] += wt * omh[:n - sa]
            grad[:n - sa] -= wt * h[sa:]
    return grad, m

def nag_loop(h0, n, target, alphas, iters, deadline, dx):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for alpha in alphas:
        h = best_h.copy()
        h_prev = h.copy()
        t_nest = 1.0
        lr = 1.0 / (alpha * dx * n * 0.5)
        for it in range(iters):
            if time.time() > deadline:
                return best_h, best_c5
            mom = (t_nest - 1.0) / (t_nest + 1.0)
            y = h + mom * (h - h_prev)
            corr_y = np.correlate(y, 1.0 - y, mode='full')
            g = sm_grad(corr_y, y, alpha) * dx
            y_new = y - lr * g
            h_new = project_sum(y_new, target)
            val_new = soft_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * dx
            val_ref = soft_max_val(corr_y, alpha) * dx
            for _ in range(12):
                if val_new <= val_ref:
                    break
                lr *= 0.5
                y_new = y - lr * g
                h_new = project_sum(y_new, target)
                val_new = soft_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * dx
            h_prev = h
            h = h_new
            t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            tv = compute_c5(h)
            if tv < best_c5:
                best_h = h.copy()
                best_c5 = tv
        tv = compute_c5(h)
        if tv < best_c5:
            best_h = h.copy()
            best_c5 = tv
    return best_h, best_c5

def subg_loop(h0, n, target, iters, deadline, dx, best_c5_init):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lo_bound = best_c5 - 0.3
    for it in range(iters):
        if time.time() > deadline:
            break
        g, m = true_subgrad(h)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        f_cur = compute_c5(h)
        step = min(0.5 / gn, 0.1 * (f_cur - lo_bound) / max(gn * gn, 1e-14))
        h_new = project_sum(h - step * g, target)
        f_new = compute_c5(h_new)
        if f_new < best_c5:
            best_h = h_new.copy()
            best_c5 = f_new
            lo_bound = best_c5 - 0.2
        if it % 500 == 0:
            lo_bound = best_c5 - 0.15
    return best_h, best_c5

def targeted_local_search(h0, n, target, rng, iters, deadline):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.1
    for it in range(iters):
        if time.time() > deadline:
            break
        if it % 2000 == 2000:
            step *= 0.8
        corr = np.correlate(h, 1.0 - h, mode='full')
        s_star = np.argmax(corr)
        shift = s_star - (n - 1)
        i = rng.randint(n)
        if shift >= 0:
            j = i + shift
            if j >= n:
                j = i - shift if i - shift >= 0 else i
        else:
            j = i + shift
            if j < 0:
                j = i - shift if i - shift < n else i
        if j == i:
            j = (i + 1) % n
        delta = rng.uniform(-step, step)
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_sum(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_h = h.copy()
            best_c5 = c5_try
        elif it % 5000 == 4000:
            idx = rng.choice(n, size=5, replace=False)
            h[idx] += rng.uniform(-0.05, 0.05, 5)
            h = project_sum(h, target)
    return best_h, best_c5

def random_label_search(h0, n, target, rng, iters, deadline):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.08
    for it in range(iters):
        if time.time() > deadline:
            break
        if it % 3000 == 3000:
            step *= 0.85
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        delta = rng.uniform(-step, step)
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_sum(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_h = h.copy()
            best_c5 = c5_try
    return best_h, best_c5

def slsqp_polish(h0, n, target, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_val(c, alpha) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return sm_grad(c, x, alpha) * dx
    def con_fun(x):
        return np.sum(x) - target
    def con_grad(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_grad}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < deadline:
            h_new = project_sum(res.x, target)
            return h_new
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h_init = project_sum(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: NAG with increasing alpha
    alphas = [5, 15, 40, 100, 250, 600, 1200, 2500]
    iters_per = max(100, int((deadline - t0) * 0.06 / len(alphas)))
    best_h, best_c5 = nag_loop(best_h, n, target, alphas, iters_per, deadline - 60, dx)

    # Phase 2: True subgradient
    if time.time() < deadline - 80:
        sub_iters = max(500, int((deadline - 80 - time.time()) * 2))
        h_sg, c5_sg = subg_loop(best_h, n, target, sub_iters, deadline - 80, dx, best_c5)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 3: NAG refinement from subgradient result
    if time.time() < deadline - 60:
        best_h, best_c5 = nag_loop(best_h, n, target, [500, 1500, 3000], 800, deadline - 60, dx)

    # Phase 4: Targeted local search
    if time.time() < deadline - 40:
        ls_iters = max(5000, int((deadline - 40 - time.time()) * 3))
        h_ls, c5_ls = targeted_local_search(best_h, n, target, rng, ls_iters, deadline - 40)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 5: SLSQP polish
    if time.time() < deadline - 20:
        for a in [1000, 3000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_polish(best_h, n, target, a, 200, deadline - 15)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 6: Final random label search
    if time.time() < deadline - 10:
        rl_iters = max(3000, int((deadline - 10 - time.time()) * 4))
        h_rl, c5_rl = random_label_search(best_h, n, target, rng, rl_iters, deadline - 5)
        if c5_rl < best_c5:
            best_h, best_c5 = h_rl, c5_rl

    # Phase 7: Try alternate n for reconstruction
    for n_alt in [50, 60, 80, 90, 110]:
        if time.time() > deadline - 30:
            break
        tgt_alt = n_alt / 2.0
        h_alt = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, n), best_h)
        h_alt = project_sum(h_alt, tgt_alt)
        dx_alt = 2.0 / n_alt
        h_alt, c5_alt = nag_loop(h_alt, n_alt, tgt_alt, [20, 80, 300, 800], 600, deadline - 20, dx_alt)
        if c5_alt < best_c5:
            best_h, best_c5, n = h_alt, c5_alt, n_alt

    final_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    return best_h.tolist(), best_c5, len(best_h)
```
