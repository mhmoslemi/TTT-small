def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * S / s
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-10:
        idx = np.argsort(h)
        if diff > 0:
            for i in range(len(idx)):
                t = min(diff, 1.0 - h[idx[i]])
                h[idx[i]] += t
                diff -= t
                if abs(diff) < 1e-12:
                    break
        else:
            for i in range(len(idx)-1, -1, -1):
                t = max(diff, -h[idx[i]])
                h[idx[i]] += t
                diff -= t
                if abs(diff) < 1e-12:

    return np.clip(h, 0.0, 1.0)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        oidx = np.where(h_b == 1)[0]
        order = np.argsort(h[oidx])
        for i in range(ones - target):
            h_b[oidx[order[i]]] = 0

        order = np.argsort(-h[zIndex])
        for i in range(target - ones):

    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    if T < 5:
        return float(mx)
    return float(mx + np.log(sum(np.exp(T * (corr - mx)))) / T)

        return soft_max(x, T) + p

        return normalize_h(r.x, S)

    bc5 = compute_c5(hc)
    bh = hc.copy()
    dx = 2.0 / n
    for sw in range(max_sweeps):
        if t_dead and time.time() > t_dead:
            break
        oi = np.where(hc == 1)[0]
        zi = np.where(hc == 0)[0]
        bv, bi, bj = bc5, -1, -1
        for i in oi:
            for j in zi:
                hc[i], hc[j] = 0.0, 1.0
                cn = float(np.correlate(hc, 1.0 - hc, mode='full').max() * dx)
                hc[i], hc[j] = 1.0, 0.0
                if cn < bv - 1e-13:
                    bv, bi, bj = cn, int(i), int(j)
        if bi >= 0:
            hc[bi], hc[bj] = 0.0, 1.0
            bc5, bh = bv, hc.copy()
        else:
            break
    return bh, bc5

def doiswap_search(h, n, S, max_sweeps=5, t_dead=None):
    """2-swap local search: swap two 1s with two 0s."""
    hc = h.astype(float).copy()
    bc5 = compute_c5(hc)
    bh = hc.copy()
    dx = 2.0 / n
    for sw in range(max_sweeps):
        if t_dead and time.time() > t_dead:
            break
        oi = np.where(hc == 1)[0]
        zi = np.where(hc == 0)[0]
        found = False
        bv = bc5
        bswap = None
        for i1 in range(len(oi)):
            for i2 in range(i1 + 1, len(oi)):
                for j1 in range(len(zi)):
                    for j2 in range(j1 + 1, len(zi)):
                        a, b, c, d = int(oi[i1]), int(oi[i2]), int(zi[j1]), int(zi[j2])
                        hc[a], hc[b], hc[c], hc[d] = 0, 0, 1, 1
                        cn = float(np.correlate(hc, 1.0 - hc, mode='full').max() * dx)
                        hc[a], hc[b], hc[c], hc[d] = 1, 1, 0, 0
                        if cn < bv - 1e-13:
                            bv = cn
                            bswap = (a, b, c, d)
                            found = True
        if found:
            a, b, c, d = bswap
            hc[a], hc[b], hc[c], hc[d] = 0, 0, 1, 1
            bc5, bh, found = bv, hc.copy(), True
        if not found:
            break
    return bh, bc5

def sa_search(h, n, S, rng, n_iter, t_dead, bci):
    hc = h.astype(float).copy()
    cc = compute_c5(hc)
    bc = min(cc, bci)
    bh = hc.copy()
    dx = 2.0 / n
    temp = 0.0012
    for it in range(n_iter):
        if time.time() > t_dead:
            break
        if it % 5000 == 0 and it > 0:
            temp *= 0.98
        oi = np.where(hc == 1)[0]
        zi = np.where(hc == 0)[0]
        i = int(oi[rng.randint(len(oi))])
        j = int(zi[rng.randint(len(zi))])
        hc[i], hc[j] = 0.0, 1.0
        cn = float(np.correlate(hc, 1.0 - hc, mode='full').max() * dx)
        d = cn - cc
        if d <= 0 or (temp > 1e-10 and rng.rand() < np.exp(-d / temp)):
            cc = cn
            if cn < bc:
                bc, bh = cn, hc.copy()
        else:
            hc[i], hc[j] = 1.0, 0.0
    return bh, bc
