import numpy as np
from scipy.optimize import minimize

_DX = 1.0
_T = 35.0
_N = 70
_CUR_T = 0.02

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    return _DX * (cmax + _CUR_T * np.log(np.sum(np.exp((c - cmax) / _CUR_T))))

def smooth_grad(x):
    n = _N
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    t = _CUR_T
    w = np.exp((c - cmax) / t)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros(len(c)); w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m]
        if wm < 1e-15:
            continue
        v1 = (ks - s >= 0) & (ks - s < n)
        if v1.any():
            grad[v1] += wm * (1.0 - x[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        if v2.any():
            grad[v2] += wm * (-x[ks[v2] + s])
    return _DX * grad

def cons_fun(x):
    return float(np.sum(x) - _T)

def cons_jac(x):
    return np.ones(_N)

def project_bs(h, T):
    h = np.asarray(h, dtype=float)
    lo = float(h.max()) + 1.0
    hi = float(h.min()) - 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > T:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def run_slsqp(h_init, max_iter, ftol):
    n = _N
    dx = _DX
    T = _T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_try = project_bs(h_init, T)
    try:
        r = minimize(smooth_obj, h_try, method='SLSQP', jac=smooth_grad,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        if np.isfinite(r.fun):
            return project_bs(r.x, T)
    except Exception:
        pass
    return h_try

def anneal(h_init, t_stages, iters, ftol):
    global _CUR_T
    n = _N
    dx = _DX
    T = _T
    best = h_init.copy()
    best_v = compute_c5(best, dx)
    for t in t_stages:
        _CUR_T = t
        h_new = run_slsqp(best, iters, ftol)
        v = compute_c5(h_new, dx)
        if v < best_v:
            best_v = v
            best = h_new
    return best, best_v

def projected_grad(h, t, n_iter, step):
    global _CUR_T, _X, _T, _N
    _CUR_T = t
    dx = _DX
    T = _T
    h = h.copy()
    v_prev = compute_c5(h, dx)
    momentum = h.copy()
    for it in range(n_iter):
        g = smooth_grad(h, )
        momentum = 0.9 * momentum - step * g
        h_new = project_bs(h + momentum, T)
        v = compute_c5(h_new, dx)
        if v < v_prev:
            h = h_new
            v_prev = v
        else:
            step *= 0.5
            h = project_bs(h - 0.5 * step * g, T)
    return h, compute_c5(h, dx)

def binary_init(n, T, rng):
    h = np.zeros(n)
    p = rng.randint(0, 4)
    if p == 0:
        idx = rng.choice(n, size=int(T), replace=False)
        h[idx] = 1.0
    elif p == 1:
        per = max(1, n // 6)
        for i in range(0, n, per):
            if rng.random() < T / n:
                h[i:min(i + per, n)] = 1.0
    elif p == 2:
        h = np.tile([1.0, 0.0], n // 2 + 1)[:n]
    else:
        h = (np.sin(np.linspace(0, 4 * np.pi, n)) > 0.3).astype(float)
    return project_bs(h, T)

def run(seed=42, budget_s=1100, **kwargs):
    global _DX, _T, _N, _CUR_T
    import time
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float)

    T_parent = n_parent / 2.0
    h_parent = project_bs(h_parent, T_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_parent)

    def elapsed():
        return time.time() - t0

    # Phase 1: Annealed SLSQP from parent at native size
    _N = n_parent; _DX = 2.0 / n_parent; _T = T_parent
    t_stages = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    h_a, c5_a = anneal(best_h, t_stages, [200]*len(t_stages), 1e-14)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a

    # Phase 2: Binary-like restarts at parent size
    for bi in range(20):
        if elapsed() > budget_s - 200:
            break
        h_b = binary_init(n_parent, T_parent, rng)
        h_b, c5_b = anneal(h_b, [0.3, 0.1, 0.04, 0.01, 0.003], [300]*5, 1e-13)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b

    # Phase 3: Multi-resolution exploration
    for n_try in [n_parent, 50, 90, 110, 140]:
        if elapsed() > budget_s - 120:
            break
        _N = n_try; _DX = 2.0 / n_try; _T = n_try / 2.0
        if n_try == n_parent:
            h_init = best_h.copy()
        else:
            xo = np.linspace(0, 2, n_parent, endpoint=False) + 1.0 / n_parent
            xn = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_init = np.interp(xn, xo, h_parent)
            h_init = project_bs(h_init, n_try / 2.0)
        h_r, c5_r = anneal(h_init, [0.4, 0.15, 0.06, 0.02, 0.008, 0.003], [250]*6, 1e-14)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r
        # A couple random restarts at this size

            if elapsed() > budget_s - 80:
                break
            h_rr = rng.random(n_try)
            h_rr, c5_rr = anneal(h_rr, [0.3, 0.1, 0.04, 0.01], [250]*4, 1e-13)
            if c5_rr < best_c5:
                best_c5, best_h = c5_rr, h_rr

    # Phase 4: Projected gradient refinement on best
    if elapsed() < budget_s - 60:
        n_f = len(best_h)
        _N = n_f; _DX = 2.0 / n_f; _T = n_f / 2.0
        _CUR_T = 0.02
        h_pg, c5_pg = projected_grad(best_h, 0.02, 2000, 0.05)
        if c5_pg < best_c5:
            best_c5, best_h = c5_pg, h_pg

    # Phase 5: Final sharp polish
    if elapsed() < budget_s - 15:
        n_f = len(best_h)
        _N = n_f; _DX = 2.0 / n_f; _T = n_f / 2.0
        for t_fin in [0.005, 0.002, 0.001, 0.0005]:
            if elapsed() > budget_s - 8:
                break
            h_fin, c5_fin = anneal(best_h, [t_fin], [1500], 1e-15)
            if c5_fin < best_c5:
                best_c5, best_h = c5_fin, h_fin

    # Final cleanup
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    T_f = n_f / 2.0
    best_h = project_bs(best_h, T_f)
    d = T_f - best_h.sum()
    if abs(d) > 1e-12:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        if mask.sum() > 0:
            best_h[mask] += d / mask.sum()
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, n_f
