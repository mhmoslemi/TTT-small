def _pgd(h, n, steps, scale0, scale1, tmax):
    t_start = time.time()
    dx = 2.0 / n
    tgt = 0.5 * n
    best = h.copy()
    best_c = _c5(best)
    
    lr = 0.1  # learning rate
    for it in range(steps):
        if time.time() - t_start > tmax:
            break
        frac = it / max(1, steps - 1)
        scale = scale0 * (scale1 / scale0) ** frac
        
        # Compute soft-max objective and gradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = corr.max()
        w = np.exp(scale * (corr - m))
        w /= w.sum()
        
        # Gradient
        grad = np.zeros(n)
        for j in range(n):
            s1 = 0.0
            for t in range(n - j):
                s1 += w[n - 1 - j + t] * (1.0 - h[t])
            s2 = 0.0
            for t in range(min(n, n + j)):  # t from 0 to min(n-1, n-1) but k = n-1+j-t must be valid (< 2n-1)
                # k = n-1+j-t, need 0 <= k <= 2n-2
                # 0 <= n-1+j-t → t <= n-1+j (always true for t < n)
                # n-1+j-t <= 2n-2 → t >= j-1
                lo = max(0, j - 1)
                hi = n  # exclusive
                for t in range(lo, hi):
                    s2 += w[n - 1 + j - t] * h[t]
            grad[j] = scale * dx * (s1 - s2)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = _proj(h_new, n)
        
        # Check if it improved
        c_new = _c5(h_new)
        if c_new < best_c:
            best = h_new.copy()
            best_c = c_new
        
        h = h_new
    
    return best, best_c

    gh = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        s1 = np.dot(w[n - 1 - j: 2 * n - 1 - j], gh[:n - j])
        lo = max(0, j - 1)
        s2 = np.dot(w[j: j + n - lo][::-1], h[lo:])
        grad[j] = s1 - s2
    return grad * scale * dx

def _pgd(h0, n, tmax, sc0, sc1, lr0):
    t_start = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = _c5(h)
    lr = lr0
    it = 0
    total_it = max(100, int(tmax * 300))
    while time.time() - t_start < tmax:
        it += 1
        frac = min(1.0, it / total_it)
        scale = sc0 * (sc1 / sc0) ** frac
        g = _grad_c5(h, n, scale)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        alpha = lr
        accepted = False
        for _ in range(15):
            h_new = _proj(h - alpha * g, n)
            c_new = _c5(h_new)
            c_cur = _c5(h)
            if c_new < c_cur - 1e-15:
                h = h_new
                accepted = True
                if c_new < best_c:
                    best, best_c = h.copy(), c_new
                break
            alpha *= 0.5
        if not accepted:
            lr = max(lr * 0.5, 0.001)
        else:
            lr = min(lr * 1.02, 0.8)
    return best, best_c

def _local_search(h0, n, tmax):
    t_start = time.time()
    h = h0.copy()
    best = h.copy()
    best_c = _c5(h)
    cur_c = best_c
    temp = 0.005
    no_improve = 0
    while time.time() - t_start < tmax:
        temp = max(temp * 0.9997, 1e-6)
        r = np.random.random()
        if r < 0.25:
            i = np.random.randint(n)
            dv = np.random.uniform(-0.08, 0.08)
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
        elif r < 0.5:
            blen = np.random.randint(3, max(4, n // 20))
            i0 = np.random.randint(0, n - blen)
            amp = np.random.uniform(0.01, 0.06)
            h2 = h.copy()
            h2[i0:i0 + blen] = np.clip(h2[i0:i0 + blen] + amp * np.random.randn(blen), 0.0, 1.0)
        elif r < 0.8:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            dv = np.random.uniform(0.02, 0.15)
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        else:
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr))
            s = k_star - (n - 1)
            i = np.random.randint(n)
            h2 = h.copy()
            if 0 <= i + s < n:
                if h[i] > h[i + s]:
                    dv = np.random.uniform(0.03, 0.12)
                    h2[i] = np.clip(h2[i] - dv, 0.0, 1.0)
                    h2[i + s] = np.clip(h2[i + s] + dv, 0.0, 1.0)
                else:
                    dv = np.random.uniform(0.03, 0.12)
                    h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
                    h2[i + s] = np.clip(h2[i + s] - dv, 0.0, 1.0)
            else:
                j2 = np.random.randint(n)
                if j2 != i:
                    dv = np.random.uniform(0.02, 0.1)
                    h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
                    h2[j2] = np.clip(h2[j2] - dv, 0.0, 1.0)
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 <= cur_c or np.random.random() < np.exp(-(c2 - cur_c) / max(temp, 1e-10)):
            h = h2
            cur_c = c2
            no_improve = 0
            if c2 < best_c - 1e-12:
                best, best_c = h.copy(), c2
        else:
            no_improve += 1
            if no_improve > 5000:
                h = best.copy()
                cur_c = best_c
                no_improve = 0
                temp = max(temp * 5, 0.005)
    return best, best_c

def _slsqp_fine(h0, n, tmax):
    t_start = time.time()
    dx = 2.0 / n
    tgt = 0.5 * n
    best = h0.copy()
    best_c = _c5(best)
    for sc in [80.0, 150.0, 250.0]:
        if time.time() - t_start > tmax:
            break
        def _obj(x):
            hc = np.clip(x, 0.0, 1.0)
            corr = np.correlate(hc, 1.0 - hc, mode="full")
            m = corr.max()
            return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(_obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h1 = _proj(res.x, n)
            c1 = _c5(h1)
            if c1 < best_c:
                best, best_c = h1, c1
        except Exception:
            pass
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(180) * 0.5
    n = len(h)
    bh = _proj(h, n)
    bc = _c5(bh)
    bn = n

    # Phase 1: Projected gradient descent, moderate scale
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        h1, c1 = _pgd(bh, n, min(rem * 0.28, 310), 20, 100, 0.4)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2: Local search with targeted moves
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        h2, c2 = _local_search(bh, n, min(rem * 0.18, 170))
        if c2 < bc:
            bh, bc = h2, c2

    # Phase 3: Fine SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h3, c3 = _slsqp_fine(bh, n, min(rem * 0.15, 130))
        if c3 < bc:
            bh, bc = h3, c3

    # Phase 4: Another gradient pass at high scale
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h4, c4 = _pgd(bh, n, min(rem * 0.15, 130), 60, 250, 0.15)
        if c4 < bc:
            bh, bc = h4, c3 if c4 > c3 else h4, c4
        if c4 < bc:
            bh, bc = h4, c4

    # Phase 5: Try upscaling
    rem = budget_s - (time.time() - t0)
    n2 = 2 * n
    if rem > 200 and n2 <= 720:
        h5 = _proj(np.kron(bh, np.ones(2)), n2)
        # Smooth the upsample: average neighbors
        h5s = h5.copy()
        for _ in range(3):
            h5s = 0.5 * (h5s + np.roll(h5s, 1))
            h5s[0] = 0.5 * (h5s[0] + h5s[-1])
        h5s = _proj(h5s, n2)
        c5a = _c5(h5s)
        h5b, c5b = _pgd(h5s, n2, min(rem * 0.25, 200), 30, 120, 0.3)
        if c5b < bc:
            bh, bc, bn = h5b, c5b, n2
        # Local search at higher res
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            h5c, c5c = _local_search(bh, bn, min(rem * 0.25, 150))
            if c5c < bc:
                bh, bc = h5c, c5c

    # Phase 6: Final local search at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h6, c6 = _local_search(bh, bn, min(rem * 0.5, 200))
        if c6 < bc:
            bh, bc = h6, c6

    # Final projection and evaluation
    bh = _proj(bh, bn)
    fc = _c5(bh)
    return (bh, fc, bn)
