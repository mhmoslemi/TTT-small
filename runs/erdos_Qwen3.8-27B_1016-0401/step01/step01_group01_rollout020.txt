                x_new = (np.arange(n_alt) + 0.5) * (2.0 / n_alt)
                h_r = np.interp(x_new, x_old, 1.0 - best_h)
            else:
                h_r = rng.random(n_alt)
            h_r, c5_r = optimize_smooth(h_r, t_sched_coarse, 300)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    # Phase 4: Final polish on best
    if _elapsed() < budget_s - 15:
        n_f = len(best_h)
        _G['n'] = n_f
        _G['dx'] = 2.0 / n_f
        _G['T'] = n_f / 2.0
        best_h = coord_refine(best_h, 8)
        best_h = normalize(best_h, n_f / 2.0)
        c5_pol = compute_c5(best_h, 2.0 / n_f)
        if c5_pol < best_c5:
            best_c5 = c5_pol
        try:
            bounds = [(0.0, 1.0)] * n_f
            r = minimize(sharp_obj, best_h, method='SLSQP', jac=sharp_grad,
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}],
                         options={'maxiter': 5000, 'ftol': 1e-16})
            h_s = normalize(r.x, n_f / 2.0)
            c5s = compute_c5(h_s, 2.0 / n_f)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h_s
        except Exception:
            pass

    # Final feasibility fix
    n_f = len(best_h)
    best_h = np.clip(best_h, 0, 1)
    d = n_f / 2.0 - best_h.sum()
    if abs(d) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        if mask.sum() > 0:
            best_h[mask] += d / mask.sum()
        else:
            best_h += d / n_f
        best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h, 2.0 / n_f)
    return best_h, best_c5, n_f
