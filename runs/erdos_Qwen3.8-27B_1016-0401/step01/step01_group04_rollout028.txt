from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def slsqp_refine(h, n, S, T, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    try:
        res = minimize(lambda x: smooth_max_obj(x, T), h.copy(), method='SLSQP',
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for sweep in range(max_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        improved = False
        for i in range(n):

            for j in range(n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_b[idx[order[i]]] = 1
    return h_b

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x, S):
    return x.sum() - S

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def slsqp_refine(h, n, S, T, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: _sum_constraint(x, S)}]

        res = minimize(lambda x: smooth_max_obj(x, T), h.copy(), method='SLSQP',
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()
