corr[m] = sum_j a[j] * b[j+m-(n-1)] where a=h, b=1-h, valid when 0≤j<n and 0≤j+m-n+1<n

So corr[m] = sum_j h[j] * (1-h[j+m-n+1]) for max(0, n-1-m) ≤ j ≤ min(n-1, n-1+m)

d(corr[m])/d(h[i]) = sum over contributing j:
- If j = i and i+m-n+1 is valid: (1 - h[i+m-n+1])
- If j+m-n+1 = i, i.e., j = i-m+n-1, and j is valid: h[j] * d(1-h[i])/dh[i] = -h[i-m+n-1]

So: g[i] = (1-h[i+m-n+1])*1[valid] - h[i-m+n-1]*1[valid]

Grad = dx * sum_k w_k * g_k where w_k are the softmax weights

    top_k = min(15, len(corr))
    idx = np.argpartition(corr, -top_k)[-top_k:]

    w = np.exp(50.0 * (vc - m))
    w /= w.sum()
    
    g = np.zeros(n)
    for ki, k in enumerate(idx):
        # For each i, g[i] += w[ki] * (1-h[i+k-n+1]) if 0<=i+k-n+1<n
        # and g[i] -= w[ki] * h[i-k+n-1] if 0<=i-k+n-1<n
        # i+k-n+1 in [0,n) means i in [n-1-k, n-1-k+n) = [n-1-k, 2n-1-k)
        # Combined with i in [0,n): i in [max(0,n-1-k), min(n, 2n-1-k))
        
        # First term: for i where a = i+k-n+1 is valid
        a_lo = max(0, n - 1 - k)  # i + k - n + 1 >= 0
        a_hi = min(n, n + 1 + k)  # i + k - n + 1 < n => i < n-1+k => i <= n-2+k
        # Actually: i+k-n+1 >= 0 => i >= n-1-k; i+k-n+1 < n => i < 2n-1-k
        # So i in [max(0, n-1-k), min(n, 2n-1-k))
        i_lo = max(0, n - 1 - k)
        i_hi = min(n, 2 * n - 1 - k)
        if i_lo < i_hi:

            a_vals = idx_a + k - n + 1
            g[i_lo:i_hi] += w[ki] * (1.0 - h[a_vals])
        
        # Second term: for i where b = i-k+n-1 is valid
        # i-k+n-1 >= 0 => i >= k-n+1; i-k+n-1 < n => i < k+1

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0 - 1e-12) if r > 0 else (h > 1e-12)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

_G = {}

def _obj_slsqp(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(_G['tk'], len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    vc = corr[idx]
    m = vc.max()
    return (m + np.log(np.exp(_G['sc'] * (vc - m)).sum()) / _G['sc']) * _G['dx']

def _con_fun(x):
    return x.sum() - _G['tgt']

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h0, n, dx, sc, tk, max_iter):
    global _G
    _G = {'dx': dx, 'sc': sc, 'tk': tk, 'tgt': 0.5 * n}
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        r = minimize(_obj_slsqp, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _grad_step(h, n, step, tk=15, sc=50.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(tk, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    vc = corr[idx]
    m = vc.max()
    w = np.exp(sc * (vc - m))
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(k):
        kk = int(idx[ki])
        wl = w[ki]
        i_lo = max(0, n - 1 - kk)
        i_hi = min(n, 2 * n - 1 - kk)
        if i_lo < i_hi:
            a_vals = np.arange(i_lo, i_hi) + kk - n + 1
            g[i_lo:i_hi] += wl * (1.0 - h[a_vals])
        i_lo2 = max(0, kk - n + 1)
        i_hi2 = min(n, kk + 1)
        if i_lo2 < i_hi2:
            b_vals = np.arange(i_lo2, i_hi2) - kk + n - 1
            g[i_lo2:i_hi2] -= wl * h[b_vals]
    return np.clip(h - step * g, 0.0, 1.0)

            h2[i] = np.clip(h2[i] + amt, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0 - 1e-12) if r > 0 else (h > 1e-12)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

_G = {}

def _obj_slsqp(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(_G['tk'], len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    vc = corr[idx]
    m = vc.max()
    return (m + np.log(np.exp(_G['sc'] * (vc - m)).sum()) / _G['sc']) * _G['dx']

def _con_fun(x):
    return x.sum() - _G['tgt']

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h0, n, dx, sc, tk, max_iter):
    global _G
    _G = {'dx': dx, 'sc': sc, 'tk': tk, 'tgt': 0.5 * n}
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        r = minimize(_obj_slsqp, h0, method='SLSQP', bounds=bds,
                     constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _grad_step(h, n, step, tk=15, sc=50.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(tk, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    vc = corr[idx]
    m = vc.max()
    w = np.exp(sc * (vc - m))
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(k):
        kk = int(idx[ki])
        wl = w[ki]
        i_lo = max(0, n - 1 - kk)
        i_hi = min(n, 2 * n - 1 - kk)
        if i_lo < i_hi:
            a_vals = np.arange(i_lo, i_hi) + kk - n + 1
            g[i_lo:i_hi] += wl * (1.0 - h[a_vals])
        i_lo2 = max(0, kk - n + 1)
        i_hi2 = min(n, kk + 1)
        if i_lo2 < i_hi2:
            b_vals = np.arange(i_lo2, i_hi2) - kk + n - 1
            g[i_lo2:i_hi2] -= wl * h[b_vals]
    return np.clip(h - step * g, 0.0, 1.0)

def _local_search(h0, n, budget, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    curr = h0.copy()
    curr_c = _c5(curr)
    best = h0.copy()
    best_c = curr_c
    T = 0.008
    while time.time() - t0 < budget:
        frac = (time.time() - t0) / budget
        r = rng.random()
        if r < 0.30:
            base = 0.04 * (1.0 - frac) + 0.005
            step = base * (0.5 + rng.random())
            h2 = _grad_step(curr, n, step)
        elif r < 0.55:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            amt = rng.uniform(0.02, 0.4) * rng.choice([-1.0, 1.0])
            h2 = curr.copy()
            h2[i] = np.clip(h2[i] + amt, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - amt, 0.0, 1.0)
        elif r < 0.75:
            blen = int(rng.integers(2, max(3, n // 8)))
            i = int(rng.integers(0, n - blen))
            j = int(rng.integers(0, n - blen))
            if abs(i - j) < blen + 1:
                continue
            h2 = curr.copy()
            t1 = h2[i:i+blen].copy()
            h2[i:i+blen] = h2[j:j+blen]
            h2[j:j+blen] = t1
        elif r < 0.90:
            blen = int(rng.integers(2, max(3, n // 12)))
            i = int(rng.integers(0, n - blen))
            shift = int(rng.integers(-n // 8, n // 8 + 1))
            if shift == 0:
                continue
            j = i + shift
            if j < 0 or j + blen > n:
                continue
            h2 = curr.copy()
            block = curr[i:i+blen].copy()
            if shift > 0:
                h2[i:j+blen] = np.concatenate([curr[i+blen:j+blen], block])
            else:
                h2[j:i+blen] = np.concatenate([block, curr[j:i]])
        else:
            width = int(rng.integers(4, max(5, n // 5)))
            i = int(rng.integers(0, max(1, n - width)))
            h2 = curr.copy()
            h2[i:i+width] = np.clip(h2[i:i+width] + rng.normal(0, 0.08, size=width), 0.0, 1.0)
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 < best_c:

            curr, curr_c = h2, c2

        elif rng.random() < min(1.0, np.exp(-(c2 - curr_c) / max(T, 1e-9))):
            curr, curr_c = h2, c2
        T = max(0.001, T * 0.9995)
    return best, best_c

def _alt_start(n, seed):
    rng = np.random.RandomState(seed)
    b = (rng.random(n) < 0.5).astype(float)
    ones = int(b.sum())
    need = n // 2
    if ones > need:
        idx = np.where(b == 1)[0]
        b[rng.choice(idx, ones - need, replace=False)] = 0
    elif ones < need:
        idx = np.where(b == 0)[0]
        b[rng.choice(idx, need - ones, replace=False)] = 1
    return _proj(b, n)
