def project_onto_box_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0).copy()
    # Iteratively adjust to hit target
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add to elements not at 1
            room = 1.0 - h
            room_idx = room > 1e-12
            if not np.any(room_idx):
                break
            # Distribute proportionally to room
            add = diff * room[room_idx] / room[room_idx].sum()
            h[room_idx] += add
        else:
            # Need to subtract from elements not at 0
            space = h
            space_idx = space > 1e-12
            if not np.any(space_idx):
                break
            sub = (-diff) * space[space_idx] / space[space_idx].sum()
            h[space_idx] -= sub
    return np.clip(h, 0.0, 1.0)

def pgm_step(h, target, step_size):
    """One projected subgradient step for minimizing max correlation"""
    corr = compute_correlations(h)
    k_max = np.argmax(corr)
    center = len(h) - 1
    o = k_max - center  # shift
    
    # Gradient of correlation at shift o
    grad = np.zeros(len(h))
    n = len(h)
    if o == 0:

def project_box_sum(z, target):
    """Project z onto {0<=y<=1, sum(y)=target}"""
    n = len(z)
    lo, hi = -2.0, 2.0  # bounds on mu
    for _ in range(100):
        mu = (lo + hi) / 2
        y = np.clip(z - mu, 0, 1)
        s = y.sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    y = np.clip(z - (lo+hi)/2, 0, 1)
    # Tiny correction
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        # Fine-tune
        idx = np.argsort(y)[::-1] if diff > 0 else np.argsort(y)
        for i in idx:
            if diff > 0:

                y[i] += s
            else:
                s = min(-diff, y[i])
                y[i] -= s
            diff = target - y.sum()

import numpy as np
from scipy.optimize import minimize
import time

_CON_TARGET = 0.0
_SM_T = 1000.0

def _sum_eq_fun(x):
    return x.sum() - _CON_TARGET

def _sum_eq_jac(x):
    return np.ones(len(x))

def _sm_obj(x):
    corr = get_corr(x)
    mc = corr.max()
    dx = 2.0 / len(x)
    shifted = _SM_T * (corr - mc)
    logsum = np.log(np.exp(shifted - shifted.max()).sum() + 1e-300) / _SM_T
    return (mc + logsum) * dx

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]
    return g

def project_box_sum(z, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mu = 0.5 * (lo + hi)
        if np.clip(z - mu, 0.0, 1.0).sum() > target:
            lo = mu
        else:
            hi = mu
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        idx = np.argsort(1.0 - y) if diff > 0 else np.argsort(y)
        for i in idx:
            s = min(diff, 1.0 - y[i]) if diff > 0 else min(-diff, y[i])
            y[i] += s
            diff = target - y.sum()
            if abs(diff) < 1e-11:
                break
    return y

def subgradient_phase(h, target, max_time, rng):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    t = 0
    batch = 2000
    while time.time() - t_start < max_time:
        M = min(1 + t // 8000, 40)
        alpha = 0.025 * max(0.02, 1.0 - t / 300000.0)
        for _ in range(batch):
            corr = get_corr(h)
            center = n - 1
            top_M = np.argsort(corr)[-M:]
            cmax = corr[top_M].max()
            weights = np.exp(4.0 * (corr[top_M] - cmax))
            ws = weights.sum()
            if ws < 1e-30:
                weights = np.ones(M) / M
            else:
                weights /= ws
            grad = np.zeros(n)
            for idx, w in zip(top_M, weights):
                if w > 1e-8:
                    grad += w * grad_shift(h, idx - center)
            h = project_box_sum(h - alpha * grad, target)
            t += 1
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_phase(h, target, max_time):
    global _CON_TARGET, _SM_T
    _CON_TARGET = target
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq_fun, 'jac': _sum_eq_jac}]
    T_list = [200, 500, 1000, 2000, 4000, 8000]
    for T in T_list:
        if time.time() - t_start > max_time:
            break
        _SM_T = float(T)
        try:
            res = minimize(_sm_obj, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-14})
            if res.fun < 100:
                h_new = project_box_sum(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new
                    h = h_new
        except Exception:
            pass
    return best_h, best_c5

def binary_swap_phase(h, target, max_time, rng):
    n = int(target * 2)
    hb = h.copy()
    ones = np.where(hb > 0.5)[0]
    zeros = np.where(hb <= 0.5)[0]
    if len(ones) > n // 2:
        for i in ones[n // 2:]:
            hb[i] = 0.0
    while len(ones) < n // 2 and len(zeros) > 0:
        j = zeros.pop()
        hb[j] = 1.0
        ones = np.where(hb > 0.5)[0]
        zeros = np.where(hb <= 0.5)[0]
    if hb.sum() != target:
        return h, compute_c5(h)
    best_h = hb.copy()
    best_c5 = compute_c5(hb)
    T = 0.005
    t_start = time.time()
    while time.time() - t_start < max_time:
        for _ in range(500):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if hb[i] == hb[j]:
                continue
            hb[i], hb[j] = hb[j], hb[i]
            nc5 = compute_c5(hb)
            if nc5 <= best_c5:
                best_c5 = nc5
                best_h = hb.copy()
            elif nc5 - best_c5 < T and rng.random() < np.exp(-(nc5 - best_c5) / T):
                pass
            else:
                hb[i], hb[j] = hb[j], hb[i]
        T *= 0.995
        if T < 1e-6:
            T = 1e-6
    return best_h, best_c5

    h_new, c5 = slsqp_phase(h_new, target, min(budget_remaining * 0.3, 60))
    return h_new, c5
