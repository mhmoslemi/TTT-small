def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.clip(alpha * (corr - m), -50, 50)
    exp_vals = np.exp(vals)
    prob = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if prob[k] < 1e-12:
            continue
        offset = k - (n - 1)
        j_min = max(0, -offset)
        j_max = min(n - 1, n - 1 - offset)
        for j in range(j_min, j_max + 1):
            grad[j] += prob[k] * (1.0 - h[j + offset])
            grad[j + offset] -= prob[k] * h[j]
    return grad * dx

        # Update

        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n, target)

                momentum *= 0.5
            if patience > 500:

                patience = 0
                lr *= 0.7
                momentum = np.zeros(n)
    return h_best, best
