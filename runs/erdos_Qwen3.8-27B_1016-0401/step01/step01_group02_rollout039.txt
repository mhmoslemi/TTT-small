def _smooth_grad(h, scale, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # softmax weights
    ac = corr - corr.max()
    w = np.exp(scale * ac)
    w /= w.sum()
    # gradient
    g = np.zeros(n)
    for k in range(2*n-1):
        shift = k - (n-1)
        jmin = max(0, -shift)
        jmax = min(n-1, n-1-shift)
        wk = w[k]
        if wk < 1e-10:
            continue
        for j in range(jmin, jmax+1):
            # ∂/∂h[j+shift] of h[j+shift]*(1-h[j]) = 1-h[j]
            g[j+shift] += wk * (1.0 - h[j])
            # ∂/∂h[j] of h[j+shift]*(1-h[j]) = -h[j+shift]
            g[j] -= wk * h[j+shift]
    return g * dx

- Solve the resulting convex problem
- Iterate

The linearization of h[j+shift]*(1-h[j]) around h0:

import numpy as np
from scipy.optimize import linprog
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _project(h, n):
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-13:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-13:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sccp_solve(h, n, dx, iters, tmax):
    t0 = time.time()
    h = h.copy()
    best = h.copy()
    bestc = _c5(h)
    for it in range(iters):
        if time.time() - t0 > tmax:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        A_ub = np.zeros((2 * n - 1, n + 1))
        b_ub = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            shift = k - (n - 1)
            jmin = max(0, -shift)
            jmax = min(n - 1, n - 1 - shift)
            if jmin > jmax:
                A_ub[k, -1] = -1.0 / dx
                continue
            omh = 1.0 - h[jmin:jmax + 1]
            hv = h[jmin + shift:jmax + shift + 1]
            A_ub[k, jmin + shift:jmax + shift + 1] += omh
            A_ub[k, jmin:jmax + 1] -= hv
            f0 = np.dot(hv, omh)
            g_at_h0 = np.dot(omh, hv) - np.dot(hv, h[jmin:jmax + 1])
            b_ub[k] = -(f0 - g_at_h0) * dx
            A_ub[k, -1] = -1.0 / dx
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.5 * n])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        except Exception:
            break
        if not res.success:
            break
        h_new = _project(res.x[:n], n)
        cn = _c5(h_new)
        if cn < bestc:
            best, bestc = h_new, cn
            h = h_new
        else:
            h = 0.5 * h + 0.5 * h_new
            h = _project(h, n)
    return best, bestc

def _smooth_grad(h, scale, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    ac = corr - corr.max()
    w = np.exp(scale * ac)
    w /= w.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        shift = k - (n - 1)
        jmin = max(0, -shift)
        jmax = min(n - 1, n - 1 - shift)
        if jmin > jmax:
            continue
        gv = omh[jmin:jmax + 1] * wk
        g[jmin + shift:jmax + shift + 1] += gv
        g[jmin:jmax + 1] -= h[jmin + shift:jmax + shift + 1] * wk
    return g * dx

def _pgd_solve(h, n, dx, scale, iters, tmax):
    h = h.copy()
    t0 = time.time()
    best = h.copy()
    bestc = _c5(h)
    alpha = 2.0 / (scale * dx * 4.0)
    for _ in range(iters):
        if time.time() - t0 > tmax:
            break
        g = _smooth_grad(h, scale, dx)
        cg = _c5(h)
        for _ in range(25):
            h_new = _project(h - alpha * g, n)
            cn = _c5(h_new)
            if cn <= cg + 1e-15:
                break
            alpha *= 0.5
        h = h_new
        if cn < bestc:
            best, bestc = h.copy(), cn
        alpha = min(alpha * 1.08, 4.0 / (scale * dx))
    return best, bestc

def _subgrad_top(h, n, dx, topk, iters, tmax):
    h = h.copy()
    t0 = time.time()
    best = h.copy()
    bestc = _c5(h)
    alpha = 0.5 / dx
    for _ in range(iters):
        if time.time() - t0 > tmax:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-topk:]
        g = np.zeros(n)
        for k in idx:
            shift = k - (n - 1)
            jmin = max(0, -shift)
            jmax = min(n - 1, n - 1 - shift)
            if jmin > jmax:
                continue
            g[jmin + shift:jmax + shift + 1] += (1.0 - h[jmin:jmax + 1])
            g[jmin:jmax + 1] -= h[jmin + shift:jmax + shift + 1]
        g /= topk
        g *= dx
        h = _project(h - alpha * g, n)
        cn = _c5(h)
        if cn < bestc:
            best, bestc = h.copy(), cn
        alpha *= 0.998
    return best, bestc

def _sharpen(h, n, lam):
    h2 = h.copy()
    up = h2 > 0.5
    dn = h2 < 0.5
    h2[up] = np.minimum(1.0, h2[up] + lam)
    h2[dn] = np.maximum(0.0, h2[dn] - lam)
    return _project(h2, n)

def _local_search(h, n, tmax, nit):
    t0 = time.time()
    tgt = 0.5 * n
    best = h.copy()
    bestc = _c5(h)
    cur = h.copy()
    curc = bestc
    for _ in range(nit):
        if time.time() - t0 > tmax:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        dv = np.random.uniform(0.01, 0.3) * (1.0 if np.random.random() < 0.5 else -1.0)
        h2 = cur.copy()
        h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
        h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        ds = h2.sum() - tgt
        if abs(ds) > 1e-12:
            h2 -= ds / n
            np.clip(h2, 0.0, 1.0, out=h2)
        c2 = _c5(h2)
        if c2 < bestc:
            best, bestc = h2, c2
            cur, curc = h2, c2
        elif np.random.random() < 0.12 and c2 < curc + 0.015:
            cur, curc = h2, c2
        else:
            cur, curc = best.copy(), bestc
    return best, bestc
