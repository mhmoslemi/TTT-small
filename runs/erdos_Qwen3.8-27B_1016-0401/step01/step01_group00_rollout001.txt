    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    s = h.sum() - n / 2.0
    grad *= dx
    grad += 2.0 * pen * s
    return grad

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def slsqp_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return grad * dx

def make_block_init(n, rng, n_blocks):
    h = np.zeros(n)
    block_size = n // (2 * n_blocks)
    for i in range(2 * n_blocks):
        start = i * block_size
        end = start + block_size if i < 2 * n_blocks - 1 else n
        h[start:end] = 1.0 if i % 2 == 0 else 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.ones(140, dtype=float)
        h_best[1::2] = 0.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B annealing at original resolution
    temps = [500, 200, 100, 50, 30, 20, 12, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5]
    pen = 5.0
    bounds = [(0.0, 1.0)] * n
    for T in temps:
        if elapsed() > budget_s - 350:
            break
        try:
            res = minimize(lse_obj, h_best.copy(), method='L-BFGS-B',
                          jac=lse_grad, args=(n, T, pen),
                          bounds=bounds,
                          options={'maxiter': 600, 'ftol': 1e-15, 'gtol': 1e-10})
            if res.fun < 1000:
                h_new = project_feasible(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
            if T > 50:
                pen *= 3.0
        except Exception:
            pass

    # Phase 2: Multi-restart with varied structures
    n_restarts = 0
    while elapsed() < budget_s - 300:
        n_restarts += 1
        r = n_restarts
        if r % 7 == 1:
            h_init = (rng.random(n) >= 0.5).astype(float)
        elif r % 7 == 2:
            h_init = np.zeros(n)
            h_init[::2] = 1.0
            h_init = np.roll(h_init, int(rng.integers(-10, 11)))
        elif r % 7 == 3:
            h_init = h_best + rng.normal(0, 0.25, n)
        elif r % 7 == 4:
            h_init = make_block_init(n, rng, int(rng.integers(3, 9)))
        elif r % 7 == 5:
            h_init = h_best.copy()
            idx = rng.choice(n, size=n//4, replace=False)
            h_init[idx] = 1.0 - h_best[idx]
        elif r % 7 == 6:
            h_init = make_block_init(n, rng, int(rng.integers(5, 15)))
            h_init += rng.normal(0, 0.1, n)
        else:
            h_init = (rng.random(n) > 0.5 - rng.random()*0.1).astype(float)

        h_init = project_feasible(h_init, target)
        # Quick anneal for restart
        for T in [80, 25, 10, 5, 2]:
            if elapsed() > budget_s - 300:
                break
            try:
                res = minimize(lse_obj, h_init.copy(), method='L-BFGS-B',
                              jac=lse_grad, args=(n, T, pen),
                              bounds=bounds,
                              options={'maxiter': 300, 'ftol': 1e-13, 'gtol': 1e-9})
                if res.fun < 1000:
                    h_new = project_feasible(res.x.copy(), target)
                    h_init = h_new
            except Exception:
                pass

        c5 = compute_c5(h_init)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_init

    # Phase 3: Coarse-to-fine at half resolution
    n2 = n // 2
    if elapsed() < budget_s - 200:
        target2 = n2 / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h_best)
        h2 = project_feasible(h2, target2)
        c5_2 = compute_c5(h2)
        bounds2 = [(0.0, 1.0)] * n2
        for T in [40, 15, 8, 4, 2, 1]:
            if elapsed() > budget_s - 100:
                break
            try:
                res = minimize(lse_obj, h2.copy(), method='L-BFGS-B',
                              jac=lse_grad, args=(n2, T, pen),
                              bounds=bounds2,
                              options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
                if res.fun < 1000:
                    h2_new = project_feasible(res.x.copy(), target2)
                    c5_new = compute_c5(h2_new)
                    if c5_new < c5_2:
                        c5_2 = c5_new
                        h2 = h2_new
            except Exception:
                pass
        h_up = np.interp(x_old, x_new, h2)
        h_up = project_feasible(h_up, target)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            c5_best = c5_up
            h_best = h_up

    # Phase 4: SLSQP polish for constraint accuracy
    if elapsed() < budget_s - 30:
        eq_fun = np.array([h_best.sum() - target])
        for T in [0.5, 0.3, 0.2]:
            if elapsed() > budget_s - 10:
                break
            try:
                cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac,
                         'args': (target,)}]
                res = minimize(slsqp_obj, h_best.copy(), method='SLSQP',
                              jac=slsqp_grad, args=(n, T),
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 800, 'ftol': 1e-15})
                if res.fun < 1000:
                    h_new = project_feasible(res.x.copy(), target)
                    c5 = compute_c5(h_new)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = h_new
            except Exception:
                pass

    h_best = project_feasible(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))

def slsqp_eq_fun(h, target):
    return np.array([h.sum() - target])

def slsqp_eq_jac(h, target):
    return np.array([np.ones(len(h))])
