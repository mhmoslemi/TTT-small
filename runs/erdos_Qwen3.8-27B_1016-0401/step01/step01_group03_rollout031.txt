def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    proj = np.clip(h - mu, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-8
        else:
            mask = proj > 1e-8
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def topk_obj_grad(h, dx, k, lam):
    """Penalize top-k correlations with lambda * sum(top_k)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices
    topk_idx = np.argsort(corr)[-k:]
    obj = lam * np.sum(corr[topk_idx]) * dx
    
    grad = np.zeros(n)
    for i in topk_idx:
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += lam * dx * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= lam * dx * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += lam * dx * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= lam * dx * h[idx2 - p]
    return obj, grad
