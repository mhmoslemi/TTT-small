    lo, hi = -1.0, 1.0

        if s2 > target_sum:

        else:

    if abs(diff) < 1e-10:

    # Distribute remaining among non-saturated entries

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_set(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_vec(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if abs(w) < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:n - s])
        else:
            ss = -s
            grad[:n - ss] += w * (-h[ss:])
            grad[ss:] += w * (1.0 - h[:n - ss])
    return dx * grad

def smooth_smooth_obj(h, dx, temp, lam):
    n = len(h)
    c = get_corr(h)
    cmax = c.max()
    vals = np.exp((c - cmax) / temp)
    obj = dx * (cmax + temp * np.log(vals.sum()))
    obj += lam * (h.sum() - n / 2.0) ** 2
    return obj

def smooth_smooth_jac(h, dx, temp, lam):
    n = len(h)
    c = get_corr(h)
    cmax = c.max()
    w = np.exp((c - cmax) / temp)
    w /= w.sum()
    g = grad_vec(h, w, dx)
    g += 2.0 * lam * (h.sum() - n / 2.0) * np.ones(n)
    return g

def lbfgs_opt(h_init, dx, n, temp, lam, max_iter):
    def obj(x):
        return smooth_smooth_obj(x, dx, temp, lam)
    def jac(x):
        return smooth_smooth_jac(x, dx, temp, lam)
    try:
        r = minimize(obj, h_init, method='L-BFGS-B', jac=jac,
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_set(r.x, n / 2.0)
    except Exception:
        return h_init

def slsqp_opt(h_init, dx, n, temp, max_iter):
    def obj(x):
        return smooth_smooth_obj(x, dx, temp, 0.0)
    def jac(x):
        return smooth_smooth_jac(x, dx, temp, 0.0)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h_init, method='SLSQP', jac=jac,
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': 1e-15})
        return project_set(r.x, n / 2.0)
    except Exception:
        return h_init
