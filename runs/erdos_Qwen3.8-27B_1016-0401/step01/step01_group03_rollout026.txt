def compute_c5(h, dx):

    corr = np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n_ones):
    """Convert to binary with exactly n_ones ones"""
    h = h.copy()
    oc = int(np.round(np.sum(h)))
    h = (h > 0.5).astype(float)
    oc = int(np.sum(h))
    if oc > n_ones:

    elif oc < n_ones:
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:n_ones - oc]:
            h[i] = 1.0
    return h

def greedy_local_search(h, dx, t_limit, max_rounds=500):
    """Try all single swaps, accept best improvement each round"""
    n = len(h)
    best_c5 = compute_c5(h, dx)
    best_score = int(np.max(np.correlate(h, 1.0-h, mode='full')))
    
    for r in range(max_rounds):
        if time.time() > t_limit:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_delta = 0
        best_move = None
        
        for zi in zeros:
            h[zi] = 1.0
            corr = np.correlate(h, 1.0-h, mode='full')
            current_max = np.max(corr)
            # For each 1 to turn off
            for oi in ones:
                if oi == zi:
                    continue
                h[oi] = 0.0
                sc = int(np.max(np.correlate(h, 1.0-h, mode='full')))
                h[oi] = 1.0
                delta = best_score - sc
                if delta > best_delta:
                    best_delta = delta
                    best_move = (zi, oi, sc)
            h[zi] = 0.0
        
        if best_move is None:
            break
        zi, oi, new_score = best_move
        h[zi] = 1.0
        h[oi] = 0.0
        best_score = new_score
        best_c5 = new_score * dx
    
    return h, best_c5

            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
        if time.time() > t_limit:
            break
    beta = 50000.0
    lr = 0.1
    stall = 0
    for it in range(500000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 10.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.97
        h = h_new
        if stall > 3000 or lr < 1e-8:
            break
    return best_h, best_c5

def max_corr_int(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.max(corr))

def to_binary(h, n_ones, rng):
    h = (h > 0.5).astype(float).copy()
    oc = int(np.sum(h))
    if oc > n_ones:
        idxs = np.where(h == 1)[0]
        rng.shuffle(idxs)
        h[idxs[:oc - n_ones]] = 0.0
    elif oc < n_ones:
        idxs = np.where(h == 0)[0]
        rng.shuffle(idxs)
        h[idxs[:n_ones - oc]] = 1.0
    return h

def binary_sa(h0, t_limit, T0, cooling, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_sc = max_corr_int(h)
    T = T0
    T_min = 1e-6
    while time.time() < t_limit and T > T_min:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        zi = zeros[rng.randint(len(zeros))]
        oi = ones[rng.randint(len(ones))]
        h[zi] = 1.0
        h[oi] = 0.0
        sc = max_corr_int(h)
        d = sc - best_sc
        if d <= 0 or rng.random() < np.exp(-float(d) / T):
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()
        else:
            h[zi] = 0.0
            h[oi] = 1.0
        T *= cooling
    return best_h, best_sc

def binary_local_search(h, t_limit, max_rounds=300):
    n = len(h)
    best_sc = max_corr_int(h)
    for _ in range(max_rounds):
        if time.time() > t_limit:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_d = 0
        best_mv = None
        for zi in zeros:
            h[zi] = 1.0
            for oi in ones:
                if oi == zi:
                    continue
                h[oi] = 0.0
                sc = max_corr_int(h)
                h[oi] = 1.0
                d = best_sc - sc
                if d > best_d:
                    best_d = d
                    best_mv = (zi, oi, sc)
            h[zi] = 0.0
        if best_mv is None:
            break
        zi, oi, ns = best_mv
        h[zi] = 1.0
        h[oi] = 0.0
        best_sc = ns
    return h, best_sc

def continuous_sa(h0, dx, ts, t_limit, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    T = 0.05
    T_min = 1e-5
    cooling = 0.9997
    while time.time() < t_limit and T > T_min:
        scale = rng.uniform(0.01, 0.3)
        h_pert = h + rng.normal(0, scale, n)
        h_new = project_feasible(h_pert, ts)
        c5_new = compute_c5(h_new, dx)
        d = c5_new - best_c5
        if d <= 0 or rng.random() < np.exp(-d / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        T *= cooling
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_end = start + budget_s - 15

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0

    best_h = project_feasible(h_init, ts_init).copy()
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    # Phase 1: Smooth optimization on initial (15% time)
    t1 = start + budget_s * 0.15
    h1, c5_1 = smooth_optimize(best_h, dx_init, ts_init, t1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Continuous SA + smooth refine (35% time)
    t2 = start + budget_s * 0.50
    n = n_init
    dx = 2.0 / n
    ts = n / 2.0
    n_restarts = 12
    for i in range(n_restarts):
        if time.time() > t2:
            break
        t_sub = min(t2, time.time() + (t2 - time.time()) / (n_restarts - i))
        if i % 3 == 0:
            h_start = best_h + rng.normal(0, 0.15, n)
        elif i % 3 == 1:
            h_start = rng.uniform(0, 1, n)
        else:
            hb = to_binary(h_init, n // 2, rng)
            h_start = hb + rng.normal(0, 0.1, n)
        h_cs, c5_cs = continuous_sa(h_start, dx, ts, t_sub * 0.6, seed + i * 7919)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs
        if time.time() < t_sub:
            t_ref = min(t_sub, time.time() + (t_sub - time.time()) * 0.7)
            h_ref, c5_ref = smooth_optimize(h_cs, dx, ts, t_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref

    # Phase 3: Binary SA to find diverse good structures (20% time)
    t3 = start + budget_s * 0.70
    n_ones = n // 2
    h_bin_best = to_binary(best_h, n_ones, rng)
    sc_bin_best = max_corr_int(h_bin_best)
    c5_bin_best = sc_bin_best * dx
    n_sa = 15
    for i in range(n_sa):
        if time.time() > t3:
            break
        t_sub = min(t3, time.time() + (t3 - time.time()) / (n_sa - i))
        if i == 0:
            h_start_b = h_bin_best.copy()
        elif i % 2 == 0:
            h_start_b = rng.randint(0, 2, n).astype(float)
            h_start_b = to_binary(h_start_b, n_ones, rng)
        else:
            h_start_b = h_bin_best.copy()
            for _ in range(rng.randint(3, 8)):
                z = np.where(h_start_b == 0)[0]
                o = np.where(h_start_b == 1)[0]
                h_start_b[z[rng.randint(len(z))]] = 1.0
                h_start_b[o[rng.randint(len(o))]] = 0.0
        T0 = rng.uniform(0.5, 2.0)
        cool = rng.uniform(0.9992, 0.9999)
        h_sa, sc_sa = binary_sa(h_start_b, t_sub, T0, cool, seed + 1000 + i * 31)
        if sc_sa < sc_bin_best:
            sc_bin_best = sc_sa
            h_bin_best = h_sa
            c5_bin = sc_sa * dx
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_sa.copy()

    # Polish binary with local search
    if time.time() < t3 + 5:
        h_pol, sc_pol = binary_local_search(h_bin_best.copy(), min(t3 + 5, time.time() + 20), 200)
        if sc_pol < sc_bin_best:
            sc_bin_best = sc_pol
            h_bin_best = h_pol
            c5_pol = sc_pol * dx
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_bin_best.copy()

    # Phase 4: Refine best binary in continuous space (15% time)
    t4 = start + budget_s * 0.85
    for h_seed in [best_h, h_bin_best, to_binary(h_init, n_ones, rng)]:
        if time.time() > t4:
            break
        t_sub = min(t4, time.time() + (t4 - time.time()) / 3)
        h_cont = project_feasible(h_seed, ts).copy()
        h_r, c5_r = smooth_optimize(h_cont, dx, ts, t_sub)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 5: Try nearby n values (10% time)
    t5 = t_end
    for n_try in [n + 2, n - 2, n + 4, n - 4]:
        if n_try < 4 or n_try % 2 != 0 or time.time() > t5:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_interp, ts_t)
        t_sub = min(t5, time.time() + max(5, (t5 - time.time()) * 0.25))
        h_ot, c5_ot = smooth_optimize(h_try, dx_t, ts_t, t_sub)
        if c5_ot < best_c5:
            best_c5 = c5_ot
            best_h = h_ot
            best_n = n_try

    # Final polish
    if time.time() < t_end:
        dx_f = 2.0 / best_n
        ts_f = best_n / 2.0
        h_fin, c5_fin = smooth_optimize(best_h, dx_f, ts_f,
                                         min(t_end, time.time() + 20))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
