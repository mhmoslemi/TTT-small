```python
def project_to_polytope(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection for lambda
    lo, hi = -1.0, 2.0  # range for lambda
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Tiny correction
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        # Adjust non-bound entries
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.any(mask):
            proj[mask] += diff / np.count_nonzero(mask)
            proj = np.clip(proj, 0.0, 1.0)
    return proj
```

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_polytope(h, target):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_for_lag(h, m):
    """Gradient of overlap_m w.r.t. h, where m >= 0.
    For even index (lag +m): sum h[i+m]*(1-h[i])
    For odd index (lag -m): sum h[i]*(1-h[i+m])
    """
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        # Returns gradient for the +m lag (correlate index n-1+m)
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def grad_for_neg_lag(h, m):
    """Gradient of overlap_{-m} = sum h[i]*(1-h[i+m]) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def soft_gradient(h, alpha, top_k=5):
    """Compute soft-max weighted gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    g = np.zeros(n)
    center = n - 1
    # Use top_k lags for efficiency
    if top_k is not None and top_k < 2*n-1:
        top_idx = np.argsort(w)[-top_k:]
    else:
        top_idx = np.where(w > 1e-6)[0]
    
    for idx in top_idx:
        wi = w[idx]
        if wi < 1e-8:
            continue
        m = abs(idx - center)
        if idx >= center:
            g += wi * grad_for_lag(h, m)
        else:
            g += wi * grad_for_neg_lag(h, m)
    
    return g, max_c

def slsqp_optimize(h, n, target, alpha, maxiter=200):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * (2.0/n)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -50, 50)
        w = np.exp(shifted)
        s = np.sum(w)
        probs = w / s  # softmax weights
        # gradient of log-sum-exp part
        g = np.zeros(n)
        center = n - 1
        for idx in range(2*n - 1):
            if probs[idx] < 1e-8:
                continue
            m_lag = abs(idx - center)
            if idx >= center:
                g += probs[idx] * grad_for_lag(x, m_lag)
            else:
                g += probs[idx] * grad_for_neg_lag(x, m_lag)
        return g * (2.0/n)
    
    def constraint(x):
        return np.sum(x) - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(
            obj, h.copy(), jac=obj_grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
        )
        h_new = project_to_polytope(res.x, target)
        return h_new
    except Exception:
        return h.copy()

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

    proj = np.clip(h - lam, 0.0, 1.0)

    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[m:] += (1.0 - h[:n-m])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_polytope(h, target):
    lo, hi = -1.5, 2.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_lag_pos(h, m):
    n = len(h)
    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[m:] += (1.0 - h[:n - m])
    g[:n - m] -= h[m:]
    return g

def grad_lag_neg(h, m):
    n = len(h)
    if m == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[:n - m] += (1.0 - h[m:])
    g[m:] -= h[:n - m]
    return g

def soft_grad(h, alpha, top_k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    w = np.exp(np.clip(alpha * (corr - max_c), -50, 0))
    sw = np.sum(w)
    if sw < 1e-300:
        w[np.argmax(corr)] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    center = n - 1
    idx = np.where(w > 1e-6)[0]
    for i in idx:
        m = abs(i - center)
        if i >= center:
            g += w[i] * grad_lag_pos(h, m)
        else:
            g += w[i] * grad_lag_neg(h, m)
    return g, max_c * (2.0 / n)

def slsqp_refine(h, n, target, alpha, maxiter=250):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -50, 0)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * (2.0 / n)
    def con_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_to_polytope(res.x, target)
    except Exception:
        return h.copy()

def sa_loop(h, n, target, t_end, alpha0, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_h = h.copy()
    curr_c5 = best_c5
    momentum = np.zeros(n)
    lr0 = 0.008
    decay = 5e-6
    beta = 0.75
    T0 = 0.004
    T1 = 5e-7
    max_steps = 800000
    for t in range(max_steps):
        if time.time() > t_end:
            break
        frac = min(t / max_steps, 1.0)
        lr = lr0 / (1.0 + t * decay)
        T = T0 * (T1 / T0) ** frac
        alpha = alpha0 + 100.0 * frac
        g, _ = soft_grad(curr_h, alpha, top_k=8)
        momentum = beta * momentum - lr * g
        h_trial = curr_h + momentum
        h_trial = project_to_polytope(h_trial, target)
        c5_trial = compute_c5(h_trial)
        if c5_trial < curr_c5:
            curr_h = h_trial
            curr_c5 = c5_trial
        elif c5_trial - curr_c5 < T * 5.0:
            if np.log(rng.random() + 1e-30) > -(c5_trial - curr_c5) / T:
                curr_h = h_trial
                curr_c5 = c5_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        if t % 20000 == 0 and t > 0:
            h_new = slsqp_refine(best_h, n, target, 100 + int(500 * frac), maxiter=120)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                curr_h = h_new
                curr_c5 = c5_new
    return best_h, best_c5, curr_h

def coord_descent(h, n, target, rng, max_iters, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for it in range(max_iters):

            break

        k = rng.randint(n - 1)
        if k >= j:
            k += 1
        hj, hk = best_h[j], best_h[k]
        lo_d = max(-hj, hk - 1.0)
        hi_d = min(1.0 - hj, hk)
        if hi_d - lo_d < 1e-8:
            continue
        deltas = np.linspace(lo_d, hi_d, 9)
        best_d, best_v = 0.0, best_c5
        for d in deltas:
            if abs(d) < 1e-10:
                continue
            try_h = best_h.copy()
            try_h[j] += d
            try_h[k] -= d
            try_h = project_to_polytope(try_h, target)
            v = compute_c5(try_h)
            if v < best_v:
                best_v = v
                best_d = d
        if best_d != 0.0:
            best_h[j] += best_d
            best_h[k] -= best_d
            best_h = project_to_polytope(best_h, target)
            best_c5 = best_v
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            kick = rng.choice(n, size=4, replace=False)
            best_h[kick] += rng.uniform(-0.02, 0.02, size=4)
            best_h = project_to_polytope(best_h, target)
            best_c5 = compute_c5(best_h)
            stall = 0
    return best_h, best_c5
