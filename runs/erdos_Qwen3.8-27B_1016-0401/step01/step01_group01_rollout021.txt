def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[:n - s] += (1.0 - h[s:])
            g[s:] += -h[:n - s]
    else:
        ss = -s
        if n - ss > 0:
            g[ss:] += (1.0 - h[:n - ss])
            g[:n - ss] += -h[ss:]
    return g

        s = (n - 1) - m
        if w[m] > 1e-10:
            grad += w[m] * _grad_shift(x, s)
    return grad

        h = np.full_like(h, T / len(h))
