    """Project h onto {x : lo <= x_i <= hi, sum(x) = S}"""
    # First clip to bounds
    h = np.clip(h, lo, hi)
    s = h.sum()
    if abs(s - S) < 1e-12:
        return h
    # Find lambda such that sum(clamp(h - lambda, lo, hi)) = S
    lo_lam, hi_lam = lo - hi, hi - lo
    for _ in range(100):
        lam = (lo_lam + hi_lam) / 2
        h_new = np.clip(h - lam, lo, hi)
        s_new = h_new.sum()
        if s_new > S:
            lo_lam = lam
        else:
            hi_lam = lam
    h_new = np.clip(h - (lo_lam + hi_lam) / 2, lo, hi)
    # Exact adjustment for remaining
    diff = S - h_new.sum()
    # Add diff to elements that aren't at bounds
    free = (h_new > lo + 1e-10) & (h_new < hi - 1e-10)
    if free.sum() > 0:
        h_new[free] += diff / free.sum()
        h_new = np.clip(h_new, lo, hi)
    return h_new

    mx = corr.max()
    s = T * (corr - mx)
    return mx + np.log(np.sum(np.exp(s))) / T

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx)
    p = np.exp(s) / np.exp(s).sum()

    grad = np.zeros(n)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx)
    return float(mx + np.log(np.sum(np.exp(s))) / T)

def smooth_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx)
    s_max = s.max()
    exp_s = np.exp(s - s_max)
    p = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s_shift = k - (n - 1)
        if s_shift >= 0:
            L = n - s_shift
            grad[s_shift:] += pk * (1.0 - h[:L])
            grad[:L] -= pk * h[s_shift:]
        else:
            ss = -s_shift
            L = n - ss
            grad[:L] += pk * (1.0 - h[ss:])
            grad[ss:] -= pk * h[:L]
    return grad

def project_sum(h, S):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)

        else:

    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.any():

    return np.clip(h, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx)
    return float(mx + np.log(np.sum(np.exp(s))) / T)

def smooth_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx) - T * (corr - mx).max()
    e = np.exp(s)
    p = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            L = n - sh
            grad[sh:] += pk * (1.0 - h[:L])
            grad[:L] -= pk * h[sh:]
        else:
            ss = -sh
            L = n - ss
            grad[:L] += pk * (1.0 - h[ss:])
            grad[ss:] -= pk * h[:L]
    return grad

def project_sum(h, S):
