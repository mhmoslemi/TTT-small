def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _smooth_obj(h, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    return dx * (c_max + temp * np.log(np.sum(np.exp((c - c_max) / temp))))

def _smooth_grad(h, dx, temp):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    w = np.exp((c - c_max) / temp)
    w_sum = w.sum()
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m] / w_sum
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += wm * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += wm * (-h[ks[v2] + s])
    return dx * grad

def _sharp_grad(h, dx):
    n = len(h)

    idx = int(np.argmax(c))
