import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, n):
    S = int(n / 2)
    h_b = np.zeros(n)
    order = np.argsort(-h)
    h_b[order[:S]] = 1
    return h_b

def sa_binary(h_init, n, rng, n_iter, t_end, c5_init, temp0, cool_rate):
    h = h_init.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = min(c5, c5_init)
    temp = temp0
    for it in range(n_iter):
        if time.time() >= t_end:
            break
        if it > 0 and it % 4000 == 0:
            temp *= cool_rate
        r = rng.rand()
        if r < 0.65:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5
            if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-15)):
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
        else:
            i1, j1 = rng.randint(n, 2)
            i2, j2 = rng.randint(n, 2)
            if h[i1] == h[j1] or h[i2] == h[j2]:
                continue
            if len(set([i1, j1, i2, j2])) < 4:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
            c5_new = compute_c5(h)
            delta = c5_new - c5
            if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-15)):
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5

            else:

                h[i2], h[j2] = h[j2], h[i2]

def hill_climb(h_init, n, t_end, best_c5_init):
    h = h_init.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = min(c5, best_c5_init)
    while time.time() < t_end:
        improved = False
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        best_move_c5 = best_c5
        best_move = None
        for i in one_idx:
            if time.time() > t_end:
                break
            for j in zero_idx:
                h[i] = 0
                h[j] = 1
                c5_new = compute_c5(h)
                if c5_new < best_move_c5 - 1e-15:
                    best_move_c5 = c5_new
                    best_move = (i, j)
                h[i] = 1
                h[j] = 0
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            c5 = best_move_c5
            best_c5 = c5
            best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c5

def targeted_local(h_init, n, t_end, best_c5_init):
    """Focus on reducing the worst correlation lag."""
    h = h_init.copy()
    c5 = compute_c5(h)
    best_h = h.copy()

def slsqp_smooth(h0, n, S, T, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = corr.max()
        shifted = T * (corr - mx)
        return (mx + np.log(np.sum(np.exp(shifted))) / T) * dx
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = corr.max()
        p = T * (corr - mx)
        p -= p.max()
        w = np.exp(p) / T
        w_sum = w.sum()
        g = np.zeros(2 * n - 1)
        g[:] = -w * dx / w_sum
        for k in range(n):
            idx_pos = n - 1 + k
            idx_neg = n - 1 - k
            if idx_pos < 2 * n - 1:
                g[idx_pos] += w[idx_pos] * dx / w_sum
            if idx_neg >= 0:
                g[idx_neg] += w[idx_neg] * dx / w_sum
        gx = np.zeros(n)
        for k in range(n):
            idx_pos = n - 1 + k
            if idx_pos < 2 * n - 1:
                gx[:n - k] += -w[idx_pos] * dx / w_sum
                gx[k:] += -w[idx_pos] * dx / w_sum
            idx_neg = n - 1 - k
            if idx_neg >= 0:
                gx[n - k:] += -w[idx_neg] * dx / w_sum
                gx[:k] += -w[idx_neg] * dx / w_sum
        return gx
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - S}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = np.clip(res.x, 0, 1)
        s = h_new.sum()
        if s > 1e-15:
            h_new = h_new * S / s
        h_new = np.clip(h_new, 0, 1)
        return h_new
    except Exception:
        return h0.copy()
