def grad_at_shift(h, s):
    """Gradient of f_s(h) = sum_i h[i]*(1-h[i+s]) at the given shift s"""
    n = len(h)
    grad = np.zeros(n)
    ks = np.arange(n)
    # f_s = sum_{i: 0<=i<n, 0<=i+s<n} h[i]*(1-h[i+s])

    v1 = (ks - s >= 0) & (ks - s < n)
    grad[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])
    return grad

    idx = int(np.argmax(c))
    s = idx - (n - 1)
    return dx * grad_at_shift(h, s)

        h_o, c5_o = slsqp_sharp(h_r, n0, dx0, T0, 1000, 1e-13)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o
        if time_ok(t0, budget_s, 50):
            h_o, c5_o = slsqp_smooth(h_o, n0, dx0, T0, 0.03, 400, 1e-13)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o
            h_o, c5_o = slsqp_sharp(best_h, n0, dx0, T0, 800, 1e-14)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o

    # Phase 6: Alternate grid sizes
    for n_alt in [63, 84, 105]:
        if not time_ok(t0, budget_s, 40):
            break
        dx_a = 2.0 / n_alt
        T_a = n_alt / 2.0
        x_old = (np.arange(n0) + 0.5) * dx0
        x_new = (np.arange(n_alt) + 0.5) * dx_a
        h_int = np.interp(x_new, x_old, best_h, left=best_h[0], right=best_h[-1])
        h_int = normalize(h_int, T_a)
        h_o, c5_o = slsqp_smooth(h_int, n_alt, dx_a, T_a, 0.05, 600, 1e-12)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o
        h_o, c5_o = slsqp_sharp(best_h, n_alt, dx_a, T_a, 1200, 1e-13)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 7: Final intensive polish on best
    if time_ok(t0, budget_s, 30):
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        T_f = n_f / 2.0
        for _ in range(4):
            if not time_ok(t0, budget_s, 20):
                break
            h_o, c5_o = slsqp_smooth(best_h, n_f, dx_f, T_f, 0.005, 500, 1e-14)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o
            h_o, c5_o = slsqp_sharp(best_h, n_f, dx_f, T_f, 2000, 1e-15)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o
        if time_ok(t0, budget_s, 15):
            h_o, c5_o = proj_grad(best_h, n_f, dx_f, T_f, 0.01, 0.001, 150, 0.005, t0, budget_s)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o
            h_o, c5_o = slsqp_sharp(best_h, n_f, dx_f, T_f, 1500, 1e-15)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_o

    # Finalize
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    T_f = n_f / 2.0
    best_h = project_feasible(best_h, T_f)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, n_f
