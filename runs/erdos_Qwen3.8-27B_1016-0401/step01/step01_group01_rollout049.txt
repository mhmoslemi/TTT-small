        h_a, c5_a = do_optimize(best_h.copy(), n0, 1000, 1e-14, True, T_s)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a
    if time.time() > t_end - 300:
        return best_h, best_c5, len(best_h)

    # Phase 3: Shake and reoptimize (many iterations, decreasing eps)
    n_shakes = 150
    for i in range(n_shakes):
        if time.time() > t_end - 150:
            break
        frac = i / float(n_shakes)
        eps = 0.20 * (1.0 - frac) + 0.005
        h_shake = best_h + rng.randn(n0) * eps
        h_shake = project_feasible(h_shake, n0)
        h_s, c5_s = do_optimize(h_shake, n0, 500, 1e-13)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s
    if time.time() > t_end - 120:
        return best_h, best_c5, len(best_h)

    # Phase 4: Structured perturbations
    for i in range(40):
        if time.time() > t_end - 100:
            break
        h_pert = best_h.copy()
        kind = i % 4
        if kind == 0:
            freq = rng.randint(2, max(3, n0 // 3))
            phase = rng.uniform(0, 2.0 * np.pi)
            amp = rng.uniform(0.02, 0.12)
            h_pert += amp * np.sin(2.0 * np.pi * freq * np.arange(n0) / n0 + phase)
        elif kind == 1:
            center = rng.randint(0, n0)
            width = float(rng.randint(3, max(4, n0 // 4)))
            amp = rng.uniform(-0.15, 0.15)
            dist = np.minimum(np.abs(np.arange(n0) - center), n0 - np.abs(np.arange(n0) - center))
            h_pert += amp * np.exp(-0.5 * (dist / width) ** 2)
        elif kind == 2:
            n_flip = rng.randint(max(2, n0 // 10), max(3, n0 // 3))
            idx = rng.choice(n0, size=n_flip, replace=False)
            h_pert[idx] += rng.uniform(-0.25, 0.25, size=n_flip)
        else:
            h_pert = h_pert[rng.permutation(n0)].copy()
            h_pert = project_feasible(h_pert, n0)
        h_pert = project_feasible(h_pert, n0)
        h_f, c5_f = do_optimize(h_pert, n0, 600, 1e-13)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
    if time.time() > t_end - 80:
        return best_h, best_c5, len(best_h)

    # Phase 5: Other grid sizes
    for n_alt in [50, 80, 90, 100, 120, 140, 160, 200]:
        if time.time() > t_end - 50:
            break
        x_new = np.linspace(0, 2, n_alt)
        x_old = np.linspace(0, 2, len(best_h))
        h_proj = np.interp(x_new, x_old, best_h)
        h_proj = project_feasible(h_proj, n_alt)
        h_alt, c5_alt = do_optimize(h_proj, n_alt, 1200, 1e-14, True, 0.01)
        if c5_alt < best_c5:
            best_c5, best_h = c5_alt, h_alt
    if time.time() > t_end - 30:
        return best_h, best_c5, len(best_h)

    # Phase 6: Final high-precision polish (multiple passes)
    n_f = len(best_h)
    for _ in range(3):
        if time.time() > t_end - 10:
            break
        h_pol, c5_pol = do_optimize(best_h, n_f, 8000, 1e-18, True, 0.002)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol

    # Final cleanup
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
