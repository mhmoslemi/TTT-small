        if rng.random() < 0.65:
            zi = zeros[rng.randint(len(zeros))]
            oi = ones[rng.randint(len(ones))]
            if zi == oi:
                continue
            h[zi] = 1.0
            h[oi] = 0.0
            nc = compute_c5(h, dx)
            d = nc - curr_c5
            if d < 0 or rng.random() < np.exp(-d / T):
                curr_c5 = nc
                if nc < best_c5 - 1e-15:
                    best_c5 = nc
                    best = h.copy()
            else:
                h[zi] = 0.0
                h[oi] = 1.0
        else:
            if len(zeros) < 2 or len(ones) < 2:
                continue
            zs = rng.choice(len(zeros), 2, replace=False)
            os_ = rng.choice(len(ones), 2, replace=False)
            z1, z2 = zeros[zs[0]], zeros[zs[1]]
            o1, o2 = ones[os_[0]], ones[os_[1]]
            h[z1] = 1.0; h[z2] = 1.0; h[o1] = 0.0; h[o2] = 0.0
            nc = compute_c5(h, dx)
            d = nc - curr_c5
            if d < 0 or rng.random() < np.exp(-d / T):
                curr_c5 = nc
                if nc < best_c5 - 1e-15:
                    best_c5 = nc
                    best = h.copy()
            else:
                h[z1] = 0.0; h[z2] = 0.0; h[o1] = 1.0; h[o2] = 1.0
        T *= cool
    return best, best_c5


def hill_climb_2swap(h0, dx, t_limit, max_rounds=500):
    n = len(h0)
    h = (h0 > 0.5).astype(float).copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for zi in zeros:
            if time.time() > t_limit:
                return h, best_c5
            for oi in ones:
                h[zi] = 1.0
                h[oi] = 0.0
                nc = compute_c5(h, dx)
                if nc < best_c5 - 1e-15:
                    best_c5 = nc
                    improved = True
                    break
                if nc < best_c5:
                    pass
                h[zi] = 0.0
                h[oi] = 1.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5


def frac_refine(h0, dx, t_limit, beta_start=2000.0, beta_end=8000.0):
    n = len(h0)
    ts = n / 2.0
    h = project_feasible(h0, ts).copy()
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    beta = beta_start
    lr = 0.3
    for it in range(500000):
        if time.time() > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        shifted = corr - max_c
        exp_c = np.exp(np.clip(beta * shifted, -500, 500))
        sum_exp = np.sum(exp_c)
        w = dx * exp_c / sum_exp
        grad = np.zeros(n)
        for i in range(2 * n - 1):
            wi = w[i]
            if wi < 1e-15:
                continue
            m = i - (n - 1)
            if m >= 0:
                idx1 = np.arange(m, n)
                grad[idx1] += wi * (1.0 - h[idx1 - m])
                idx2 = np.arange(n - m)
                grad[idx2] -= wi * h[idx2 + m]
            else:
                p = -m
                idx1 = np.arange(n - p)
                grad[idx1] += wi * (1.0 - h[idx1 + p])
                idx2 = np.arange(p, n)
                grad[idx2] -= wi * h[idx2 - p]
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best = h_new.copy()
            lr = min(lr * 1.04, 3.0)
        else:
            lr *= 0.93
        h = h_new
        if it % 2000 == 0:
            beta = min(beta * 1.5, beta_end)
        if lr < 1e-8:
            break
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0

    best_h = project_feasible(h_init, ts_init)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    t1 = start_time + budget_s * 0.25
    t2 = start_time + budget_s * 0.45
    t3 = start_time + budget_s * 0.70
    t4 = start_time + budget_s * 0.92

    # Phase 1: SA on binary version of initial
    h_bin = (best_h > 0.5).astype(float)
    h_sa, c5_sa = sa_binary(h_bin, dx_init, t1, seed=seed, T0=0.003)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Hill climb after SA
    t_hc = min(t2, time.time() + 30)
    h_hc, c5_hc = hill_climb_2swap(best_h, dx_init, t_hc)
    if c5_hc < best_c5:
        best_c5 = c5_hc
        best_h = h_hc.copy()

    # Phase 2: Fractional refinement on n=46
    t_fr1 = min(t2, time.time() + max(15, (t2 - time.time()) * 0.6))
    h_fr, c5_fr = frac_refine(best_h, dx_init, t_fr1, beta_start=3000.0)
    if c5_fr < best_c5:
        best_c5 = c5_fr
        best_h = h_fr.copy()

    # Phase 3: Try other resolutions
    for n_try in [48, 92, 96, 192]:
        if time.time() > t3:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_try = (h_interp > 0.5).astype(float)
        oc = int(np.sum(h_try))
        if oc > int(ts_t):
            idxs = np.where(h_try == 1)[0]
            np.random.shuffle(idxs)
            for i in idxs[:oc - int(ts_t)]:
                h_try[i] = 0.0
        elif oc < int(ts_t):
            idxs = np.where(h_try == 0)[0]
            np.random.shuffle(idxs)
            for i in idxs[:int(ts_t) - oc]:
                h_try[i] = 1.0

        t_sa2 = min(t3, time.time() + max(5, (t3 - time.time()) * 0.25))
        h_b2, c5_b2 = sa_binary(h_try, dx_t, t_sa2, seed=seed + n_try, T0=0.002)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = n_try

        t_fr2 = min(t3, time.time() + max(5, (t3 - time.time()) * 0.2))
        dxb = 2.0 / best_n
        h_f2, c5_f2 = frac_refine(best_h, dxb, t_fr2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
            best_n = n_try

    # Phase 4: Multi-start SA + refine on best resolution
    dx_b = 2.0 / best_n
    ts_b = best_n / 2.0
    np.random.seed(seed + 777)
    while time.time() < t4:
        remaining = t4 - time.time()
        if remaining < 5:
            break
        t_sa3 = min(t4, time.time() + min(20, remaining * 0.4))
        h_r = (np.random.random(best_n) > 0.5).astype(float)
        oc = int(np.sum(h_r))
        if oc > int(ts_b):
            idxs = np.where(h_r == 1)[0]
            np.random.shuffle(idxs)
            for i in idxs[:oc - int(ts_b)]:
                h_r[i] = 0.0
        elif oc < int(ts_b):
            idxs = np.where(h_r == 0)[0]
            np.random.shuffle(idxs)
            for i in idxs[:int(ts_b) - oc]:
                h_r[i] = 1.0
        h_r2, c5_r2 = sa_binary(h_r, dx_b, t_sa3, seed=np.random.randint(100000), T0=0.004)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()

        t_fr3 = min(t4, time.time() + min(10, (t4 - time.time()) * 0.3))
        h_f3, c5_f3 = frac_refine(best_h, dx_b, t_fr3, beta_start=5000.0, beta_end=10000.0)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()

    # Final polish
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, best_n
