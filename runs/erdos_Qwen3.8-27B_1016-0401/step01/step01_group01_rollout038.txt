import time

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_set(h, T):
    """Project onto {x: 0<=x[i]<=1, sum(x)=T} using bisection"""
    n = len(h)
    lo = -h.max() - 1
    hi = 1 - h.min() + 1
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0, 1))
        if s < T:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5*(lo+hi), 0, 1)

def coordinate_sweep(h, T, dx, lam=0.1, max_passes=20):
    """Gauss-Seidel coordinate descent on the max-correlation + penalty objective"""
    n = len(h)
    h = h.copy()
    s_vals = np.arange(-(n-1), n)  # all lags
    
    for _ in range(max_passes):
        changed = False
        for j in range(n):
            hj = h[j]
            # Compute C_s and D_s
            # D_s = (1-h[j+s])*[0<=j+s<n] - h[j-s]*[0<=j-s<n]
            js = j + s_vals
            jm = j - s_vals
            a_mask = (js >= 0) & (js < n)
            b_mask = (jm >= 0) & (jm < n)
            # a_s = (1 - h[j+s]) where valid, 0 otherwise
            a_val = np.where(a_mask, 1.0 - h[js], 0.0)
            b_val = np.where(b_mask, h[jm], 0.0)
            D_s = a_val - b_val
            
            # C_s = full_correlation_s - a_val * h[j] - b_val * (1-h[j])
            # But a_val here is (1-h[j+s]) or 0, and the actual contribution to c_s is:
            #   a_val * h[j]  (from i=j term)
            #   b_val * (1-h[j])  (from i=j-s term)
            # So C_s = c_s - a_val*h[j] - b_val*(1-h[j])
            # I need c_s for all s... let me compute the full correlation
            c_full = np.correlate(h, 1.0 - h, mode='full')
            C_s = c_full - a_val * hj - b_val * (1.0 - hj)
            
            S_other = np.sum(h) - hj
            
            # Minimize f(t) = dx * max(C_s + D_s*t) + lam*(S_other + t - T)^2
            def f(t):
                objs = C_s + D_s * t
                c5_part = dx * np.max(objs)
                pen = lam * (S_other + t - T) ** 2
                return c5_part + pen
            
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-10})
            if res.success and abs(res.x - hj) > 1e-12:
                h[j] = res.x
                changed = True
        # Gentle projection
        h = project_onto_set(h, T)
        if not changed:
            break
    return h

            b_m = (jm>=0)&(jm<n)

                    c += D * delta  # update correlation incrementally
        h = project_blk(h, T)
        # Re-sync correlation after projection
        c = np.correlate(h, 1.0-h, mode='full')

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

# Globals for 1D objective
_GC = None
_GD = None
_GDX = 0.0
_GLAM = 0.0
_GSO = 0.0
_GT = 0.0

def _f1d(t):
    return _GDX * np.max(_GC + _GD * t) + _GLAM * (_GSO + t - _GT) ** 2

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_blk(h, T):
    n = len(h)
    lo = -h.max() - 0.5
    hi = 1.0 - h.min() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < T:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def coord_sweep(h, T, dx, lam, n_passes=5):
    global _GC, _GD, _GDX, _GLAM, _GSO, _GT
    n = len(h)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    s_vals = np.arange(-(n - 1), n)
    _GDX = dx
    _GLAM = lam
    _GT = T
    
    for _ in range(n_passes):
        for j in range(n):
            hj = h[j]
            js = j + s_vals
            jm = j - s_vals
            a_m = (js >= 0) & (js < n)
            b_m = (jm >= 0) & (jm < n)
            a_v = np.where(a_m, 1.0 - h[js], 0.0)
            b_v = np.where(b_m, h[jm], 0.0)
            D = a_v - b_v
            C = c - D * hj
            _GC = C
            _GD = D
            _GSO = np.sum(h) - hj
            res = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                   options={'xatol': 1e-12})
            if res.success:
                delta = res.x - hj
                if abs(delta) > 1e-14:
                    h[j] = res.x
                    c += D * delta
        h = project_blk(h, T)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h

def smooth_grad(h, T, dx, tau):
    """Gradient of smooth max (log-sum-exp) of correlation * dx"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    w = np.exp((c - c_max) / tau)
    w /= np.sum(w)
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        a_m = (ks + s >= 0) & (ks + s < n)
        b_m = (ks - s >= 0) & (ks - s < n)
        # From the correlation: c_m = sum_i h[i]*(1-h[i+s])
        # dc_m/dh[j] = (1-h[j+s])*[valid] - h[j-s]*[valid]
        a_v = np.where(a_m, 1.0 - h[ks + s], 0.0) if np.any(a_m) else np.zeros(n)
        b_v = np.where(b_m, h[ks - s], 0.0) if np.any(b_m) else np.zeros(n)
        grad += w[m] * (a_v - b_v)
    return dx * grad

def proj_grad_descent(h, T, dx, tau, n_steps=2000, alpha_init=0.01):
    """Projected gradient descent on smooth objective"""
    h = h.copy()

        h_new = h - alpha * g
        h_new = project_blk(h_new, T)

        c5_new = compute_c5(h_new, dx)
