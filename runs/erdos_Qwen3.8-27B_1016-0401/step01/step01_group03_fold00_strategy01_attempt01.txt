                indices = np.where(mask)
                avg_shift = diff / min(len(indices), 10)
                for idx in indices[:5]:
                    h[idx] = np.clip(h[idx] + avg_shift, 0.5, 1)
        s_adjusted = np.sum(h)
        adjustment_needed = abs(target_sum - s_adjusted)
        max_iter -= 1
        
    s_adjusted = np.sum(h)
    if abs(target_sum - s_adjusted) > 1e-5:
        # Last resort: random find if constraint cannot be strictly met
        x_span = (target_sum - s_adjusted) / (len(h) * 1.0)
        np.add(h, x_span, out=h, casting='same_kind')
        h = np.clip(h, 0, 1)
    
    return h

class SolutionPool:
    def __init__(self, n_init, t_budget=1100, parent=None):
        self.n_init = n_init
        self.budget = t_budget
        self.pool = []
        self.best = None
        self.time_budget = t_budget
        self.timer_start = time.time()
        self.active_candidates = deque([(np.array(initial_h_values) if parent is None else parent, 0.0, n_init)])
        self.current_n = n_init
        self.dx = 2.0 / n_init
        self.feasible = [1 if x > 0.5 else 0 for x in initial_h_values]
        self.level = 0
        self.max_level = 4
        self.targets = [46, 92, 184, 368]
        self.current_res = self.targets[0]
        self.function_evals = 0
        self.mutations = [0, 0.3, 0.1, 0.2]  # Mutation rates at each level

    def _compute_objective(self, h, dx=0.0):
        try:
            corr = np.correlate(h, 1.0-h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        except:
            avg = np.mean(h)
            h_fix = np.clip(h, 0, 1)
            corr = np.correlate(h_fix, 1.0-h_fix, mode='full')
            return np.max(corr) * dx

    def _objective(self, h):
        # This is the evaluation function used in optimization
        return self._compute_objective(h, self.dx)

    def _smooth_perturbation(self, h, severity=1.0):
        # Gradient-like perturbation pattern
        n = len(h)
        w = np.exp(-0.01 * np.random.randn(n)) * severity
        w = np.clip(w, 0.1, 2.0)
        return h + w*(np.random.randn(n) + 0.5)

    def _pool_elitism(self):
        # Keep top 20% of all solutions in all levels
        if not self.pool:
            return
            
        all_solutions = []
        for sol in self.pool:
            try:
                score = self._compute_objective(sol[0], 2.0/sol[1])
                all_solutions.append((sol[0].copy(), score, sol[1]))
            except:
                continue
            
        all_solutions.sort(key=lambda x: x[1])
        top_20 = int(0.2 * len(all_solutions))
        self.pool = all_solutions[:top_20]

    def run_metropolis(self):
        while time.time() < self.time_budget:
            self._pool_elitism()
            
            # Sample k solutions from the pool
            sample_n = min(10, len(self.pool))
            candidates = [sol[0] for sol in self.pool[:sample_n]]
            
            # Try gradient-based refinement with 3 candidates
            refined_sols = []
            for cand in candidates:
                try:
                    # Smooth refinforcement
                    result = minimize(
                        self._objective, cand, 
                        method='L-BFGS-B', 
                        options={'maxiter': 500},
                        bounds=[(0,1)]*len(cand),
                        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2}
                    )
                    if result.success and 'x' in result and self._objective(result.x) < self._objective(cand):
                        refined_sols.append((result.x, self._objective(result.x)))
                except:
                    continue
            
            # Pattern-driven explorative mutations
            directions = self._pool_elitism()
            while len(refined_sols) < 3 and time.time() < self.time_budget:
                current_sol = self.pool[-1][0] if self.pool else initial_h_values
                # Genetic crossover-like mutation
                crossed = []
                if len(self.pool) >= 5 and self.level < self.max_level:
                    parent1 = self.pool[0][0]
                    parent2 = self.pool[-1][0]
                    n = len(parent1)
                    child = np.zeros(n)
                    child[:n//2] = parent1[:n//2]
                    child[n//2:] = parent2[n//2:]
                    crossed.append((child, self._objective(child)))
                # 
                # Standard mutation with adaptive rate
                mutated = current_sol * (1 + 0.3*np.random.randn(len(current_sol)))
                mutated = balance_integral(mutated, len(mutated)/2)
                crossed.append((mutated, self._objective(mutated)))
                
                refined_sols.extend(crossed)
                
                # Score and rank candidates for expansion
                refined_sols.sort(key=lambda x: x[1])
                
                # Check for budget exhaustion
                time_remaining = self.time_budget - time.time()
                if time_remaining < 10:  # Early termination
                    break
                    
                # Level up if improvement found and two checks pass
                if len(refined_sols) >= 5 and refined_sols[-1][1] < refined_sols[0][1] - 1e-3:
                    if self.level < 3:
                        self.level += 1
                        self.current_res = self.targets[self.level]
                        self.dx = 2.0 / self.current_res
                    # Project best solution to new resolution
                    best_sol = balance_integral(self.pool[-1][0], self.current_res/2)
                    best_score = self._compute_objective(best_sol, self.dx)
                    self.pool.append((best_sol, best_score, self.current_res))
                    self.active_candidates.append((best_sol, 0.0, self.current_res))
                    # Time cost for resolution upgrade: proportional to n^2
                    time_remaining -= 10 * self.current_res**0.5
        
            # Update best candidate overall
            if not self.best or refined_sols[0][1] < self.best[1]:
                self.best = refined_sols[0]
            
    def return_best(self):
        if not self.active_candidates:
            return None, None, None
            
        best_sols = []
        for cand in self.active_candidates:
            try:
                score = self._compute_objective(cand[0], 2.0/cand[1])
                best_sols.append((score, cand[0], cand[1]))
            except:
                continue
                
        best_sols.sort(key=lambda x: x[0])
        return best_sols[0][1] if best_sols else initial_h_values, best_sols[0][0] if best_sols else 1.0, 2.0 / (2.0 / len(initial_h_values)) * 2

def run_solution_pool(seed=42, time_budget=1100):
    start_time = time.time()
    solver = SolutionPool(46, time_budget, None)
    
    solver.timer_start = start_time
    
    # Parallel objectives balanced over resolutions
    n_active = min(4, len(initial_h_values))
    active_sols = [np.array(initial_h_values[i], dtype=float) for i in range(min(n_active, len(initial_h_values)))]
    
    # Initial exploration across multiple resolutions
    for i, sol in enumerate(active_sols):
        candidate = balance_integral(sol, len(sol)/2)
        score = compute_c5(candidate, 2.0/len(candidate))
        solver.active_candidates.append((candidate, 2.0/len(candidate), len(candidate)))
        if i == 0:
            solver.best = (candidate, score, len(candidate))
    
    # Gradual escalation strategy
    while time.time() < time_budget:
        solver.run_metropolis()
        current_best, best_score = solver.best
        
        # Check for global convergence
        score_improvement = np.log10(solver.best[1]) < abs(time.time() - start_time/100)
        if score_improvement < 1e-4 or time.time() > time_budget * 0.9:
            break
            
    return solver.return_best()
