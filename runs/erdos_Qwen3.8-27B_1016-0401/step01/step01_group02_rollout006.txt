import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    tgt = 0.5 * n
    h = h.copy()
    # Bisection on water level
    lo, hi = -2.0, 2.0

def _smooth_max_obj(h, s):
    """Smooth max of corr * dx, with temperature parameter s."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return float(m + np.log(np.exp(s * (corr - m)).sum()) / s)
