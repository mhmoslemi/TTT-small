import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, target, alpha, maxiter=200):
    n = len(h)
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -50, 50)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    
    def cons(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return make_feasible(res.x, target)
    except Exception:
        return h.copy()

    curr_c5 = best_c5
    T = t0

    stall_counter = 0
    
    while time.time() < deadline:
        # Adaptive step size
        step = step_base * np.sqrt(T / t0)
        
        # Pick two coordinates

                best_c5 = c5_try

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(corr.max())
        shifted = np.clip(alpha * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

        return float(x.sum() - target)

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
