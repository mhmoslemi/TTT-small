def _f_and_g(x, s, lam):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    val = _smooth_val(h, s) * (2.0/n) + lam*(h.sum()-0.5*n)**2
    g = _grad_obj(h, s) * (2.0/n) + 2.0*lam*(h.sum()-0.5*n)*np.ones(n)
    return val, g

def _proj_sum(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not free.any():
            break
        h[free] -= d
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] = np.clip(h[msk] + r / msk.sum(), 0.0, 1.0)
    return h

def _lbfgs(h0, s, lam, maxiter, ftol=1e-16):
    def f(x):
        return _f_and_g(x, s, lam)[0]
    def fgy(x):
        v, g = _f_and_g(x, s, lam)
        # Project gradient: zero out where at boundary
        g = g.copy()
        g[x <= 1e-12] = max(g[x <= 1e-12], 0)
        g[x >= 1.0-1e-12] = min(g[x >= 1.0-1e-12], 0)
        return v, g
    bds = [(0.0, 1.0)] * len(h0)
    res = minimize(fgy, h0, method='L-BFGS-B', jac=True, bounds=bds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12, 'maxls': 50})
    return _proj_sum(res.x, len(h0))
