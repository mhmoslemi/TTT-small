            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

def projected_grad_descent(h0, dx, ts, t_limit, beta):

    h = project_feasible(h0, ts).copy()

    lr = 1.0
    stall = 0
    while time.time() < t_limit:
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)

            stall = 0
        else:
            stall += 1
            lr *= 0.93
        h = h_new
        if stall > 3000 or lr < 1e-8:

    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10

import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)


def fix_binary(h, n):
    h = (h > 0.5).astype(float).copy()
    oc = int(np.sum(h))
    target = n // 2
    if oc > target:
        idxs = np.where(h == 1)[0].copy()
        np.random.shuffle(idxs)
        for i in idxs[:oc - target]:
            h[i] = 0.0
    elif oc < target:
        idxs = np.where(h == 0)[0].copy()
        np.random.shuffle(idxs)
        for i in idxs[:target - oc]:
            h[i] = 1.0
    return h


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(np.clip(beta * shifted, -60, 60))
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-16:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad


def grad_descent_momentum(h0, dx, ts, t_limit, beta, gamma=0.85):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.3
    stall = 0
    while time.time() < t_limit:
        _, grad = smooth_obj_grad(h, dx, beta)
        v = gamma * v + (1.0 - gamma) * grad
        h_new = project_feasible(h - lr * v, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 8.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.94
        h = h_new
        if stall > 1500 or lr < 1e-10:
            if time.time() < t_limit - 3.0:
                h = best_h + np.random.randn(n) * 0.02
                h = project_feasible(h, ts)
                v = np.zeros(n)
                lr = 0.3
                stall = 0
            else:
                break
    return best_h, best_c5


def local_search_1swap(h_in, dx, t_limit):
    n = len(h_in)
    h = fix_binary(h_in, n).copy()
    best_c5 = compute_c5(h, dx)

                    improved = True

            if improved:
