import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return grad * dx

def project_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h0, target, T, max_iter, step_init):
    n = len(h0)
    h = h0.copy()
    zh = h.copy()
    t_k = 1.0
    step = step_init
    for _ in range(max_iter):
        y = h + ((t_k - 1.0) / (t_k + 2.0)) * (h - zh)
        g = soft_max_grad(y, T)
        zh = project_box_sum(y - step * g, target)
        if np.linalg.norm(zh - h) < 1e-13 * (1.0 + np.linalg.norm(h)):
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = zh
        t_k = t_new
    return h

def lbfgsb_optimize(h0, n, target, T, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return soft_max_obj(x, T) + pen * (x.sum() - target) ** 2
    def grad(x):
        return soft_max_grad(x, T) + 2.0 * pen * (x.sum() - target) * ones
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14})
        return project_box_sum(res.x, target)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project_box_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with progressive cooling
    for T in [30, 60, 120, 250, 500, 1000]:
        if elapsed() > budget_s - 250:
            break
        for pen in [1.0, 10.0]:
            h_new = lbfgsb_optimize(h_best, n, target, T, pen, 3000)
            h_new = project_box_sum(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 2: FISTA with cooling (faster per-iteration, more iterations)
    for T in [20, 50, 100, 300, 600]:
        if elapsed() > budget_s - 200:
            break
        step = min(0.4, 3.0 / (T + 2.0))
        h_new = fista_optimize(h_best, target, T, 4000, step)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Perturbation + short L-BFGS-B (escaping local minima)
    n_perturb = 0
    while elapsed() < budget_s - 120:
        n_perturb += 1
        scale = max(0.03, 0.15 / (1.0 + 0.05 * n_perturb))
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = project_box_sum(h_pert, target)
        T = 80 + rng.integers(0, 300)
        h_new = lbfgsb_optimize(h_pert, n, target, T, 10.0, 800)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            scale = max(0.03, 0.15 / (1.0 + 0.05 * 0.3 * n_perturb))

    # Phase 4: FISTA local search with small perturbations
    while elapsed() < budget_s - 60:
        h_pert = h_best + rng.normal(0, 0.04, n)
        h_pert = project_box_sum(h_pert, target)
        T = 200 + rng.integers(0, 400)
        step = min(0.2, 2.0 / (T + 2.0))
        h_new = fista_optimize(h_pert, target, T, 3000, step)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try different resolutions
    for n_try in [140, 70, 210, 350]:
        if elapsed() > budget_s - 30:
            break
        t_try = n_try / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_box_sum(h_try, t_try)
        c5_try = compute_c5(h_try)
        best_local = h_try
        best_try_c5 = c5_try
        for T in [50, 150, 400]:
            if elapsed() > budget_s - 10:
                break
            h_t = lbfgsb_optimize(best_local, n_try, t_try, T, 10.0, 2000)
            h_t = project_box_sum(h_t, t_try)
            c5_t = compute_c5(h_t)
            if c5_t < best_try_c5:
                best_try_c5 = c5_t
                best_local = h_t
        if best_try_c5 < c5_best:
            c5_best = best_try_c5
            h_best = best_local
            n = n_try
            target = t_try

    # Final polish with L-BFGS-B at low temperature
    if elapsed() < budget_s - 5:
        h_new = lbfgsb_optimize(h_best, n, target, 30, 1.0, 2000)
        h_new = project_box_sum(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
