import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    n = len(h)
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def make_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.0
    # adjust if needed
    target = n // 2
    ones = int(h.sum())
    if ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:ones - target]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - ones]] = 1
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_blocks(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = rng.randint(2, max(3, n // 6))
        if rng.rand() > 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    target = n // 2
    ones = int(h.sum())
    if ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for i in range(ones - target):
            h[idx[i]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for i in range(target - ones):
            h[idx[i]] = 1
    return h

def exhaustive_climb(h, t_deadline, rng, max_rounds=5000, max_kicks=15):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    current_c5 = best_c5
    kicks_done = 0

    for rnd in range(max_rounds):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h_cur == 1)[0]
        zero_idx = np.where(h_cur == 0)[0]
        best_new_c5 = current_c5
        best_swap = None

        for i in one_idx:
            for j in zero_idx:
                h_cur[i], h_cur[j] = 0.0, 1.0
                c5_new = compute_c5(h_cur)
                h_cur[i], h_cur[j] = 1.0, 0.0
                if c5_new < best_new_c5 - 1e-15:
                    best_new_c5 = c5_new
                    best_swap = (i, j)

        if best_swap is None:
            if kicks_done >= max_kicks:
                break
            kicks_done += 1
            kick_sz = rng.randint(2, min(5, len(one_idx)))
            oc = one_idx.copy()
            zc = zero_idx.copy()
            for _ in range(kick_sz):
                ii = oc[rng.randint(len(oc))]
                jj = zc[rng.randint(len(zc))]
                h_cur[ii], h_cur[jj] = h_cur[jj], h_cur[ii]
                oc = np.where(h_cur == 1)[0]
                zc = np.where(h_cur == 0)[0]
            current_c5 = compute_c5(h_cur)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_cur.copy()
            continue

        i, j = best_swap
        h_cur[i], h_cur[j] = 0.0, 1.0
        current_c5 = best_new_c5
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_cur.copy()

    return best_h, best_c5

def slsqp_refine(h, S, t_deadline):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = corr.max()
        T = 300.0
        return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 600, 'ftol': 1e-14})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()
