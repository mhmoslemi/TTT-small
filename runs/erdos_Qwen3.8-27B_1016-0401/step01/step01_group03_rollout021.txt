    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    grid = np.linspace(0.0, 1.0, grid_sz)
    for sweep in range(200000):
        if time.time() > t_limit:
            break
        changed = False
        for j in range(n):
            save = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            delta = c1 - c0
            h[j] = save
            best_v = save
            best_c = np.max(c0 + save * delta)
            for v in grid:
                c = np.max(c0 + v * delta)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if abs(best_v - save) > 1e-12:
                changed = True
        h = project_feasible(h, ts)
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        if not changed:
            break
    return best_h, best_c5

def proj_grad(h, dx, ts, t_limit, beta=5000.0, max_iter=500000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 1.0
    stall = 0
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        _, grad = smooth_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 5.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.92
        h = h_new
        if stall > 200 or lr < 1e-8:
            break
    return best_h, best_c5

def to_binary(h, ts):
    n = len(h)
    b = (h > 0.5).astype(float)
    oc = int(np.sum(b))
    if oc > int(ts):
        idxs = np.where(b == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - int(ts)]:
            b[i] = 0.0
    elif oc < int(ts):
        idxs = np.where(b == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:int(ts) - oc]:
            b[i] = 1.0
    return b

def binary_search(h0, dx, t_limit):
    n = len(h0)
    ts = n // 2
    h = to_binary(h0, ts).copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(300):
        if time.time() > t_limit:
            break
        improved = False
        zeros = list(np.where(h == 0)[0])
        ones = list(np.where(h == 1)[0])
        for zi in zeros:
            if time.time() > t_limit:
                return h, best_c5
            for oi in ones:
                h[zi] = 1.0
                h[oi] = 0.0
                c5n = compute_c5(h, dx)
                if c5n < best_c5 - 1e-14:
                    best_c5 = c5n
                    improved = True
                    zeros = list(np.where(h == 0)[0])
                    ones = list(np.where(h == 1)[0])
                    break
                h[zi] = 0.0
                h[oi] = 1.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def optimize_at_n(h0, n, t_limit, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    if n == len(h0):
        h = project_feasible(h0, ts)
    else:
        x_old = np.linspace(0.0, 2.0, len(h0))
        x_new = np.linspace(0.0, 2.0, n)
        h = np.interp(x_new, x_old, h0)
        h = project_feasible(h, ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t_sub = time.time() + (t_limit - time.time()) * 0.3
    h1, c51 = proj_grad(h, dx, ts, t_sub, beta=3000.0)
    if c51 < best_c5:
        best_c5, best_h = c51, h1
    t_sub = time.time() + (t_limit - time.time()) * 0.3
    h2, c52 = coord_descent(best_h, dx, ts, t_sub)
    if c52 < best_c5:
        best_c5, best_h = c52, h2
    h3, c53 = proj_grad(best_h, dx, ts, t_sub, beta=15000.0)
    if c53 < best_c5:
        best_c5, best_h = c53, h3
    t_sub = time.time() + min(60.0, (t_limit - time.time()) * 0.25)
    if time.time() < t_limit:
        h4, c54 = binary_search(best_h, dx, t_sub)
        if c54 < best_c5:
            best_c5, best_h = c54, h4
        h5, c55 = coord_descent(best_h, dx, ts, t_sub)
        if c55 < best_c5:
            best_c5, best_h = c55, h5
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = project_feasible(h_init, n_init / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n_init)
    best_n = n_init

    t_phase1 = start + budget_s * 0.45
    h1, c51, n1 = optimize_at_n(h_init, n_init, t_phase1, seed)
    if c51 < best_c5:
        best_c5, best_h, best_n = c51, h1, n1

    t_phase2 = start + budget_s * 0.75
    for n_try in [48, 92, 96, 194]:
        if time.time() > t_phase2:
            break
        t_sub = min(t_phase2, time.time() + (t_phase2 - time.time()) * 0.4)
        h2, c52, n2 = optimize_at_n(best_h, n_try, t_sub, seed + n_try)
        if c52 < best_c5:
            best_c5, best_h, best_n = c52, h2, n2

    t_phase3 = start + budget_s * 0.90
    for restart in range(6):
        if time.time() > t_phase3:
            break
        dx_r = 2.0 / best_n
        ts_r = best_n / 2.0
        hr = np.random.uniform(0.0, 1.0, best_n)
        hr = project_feasible(hr, ts_r)
        t_sub = min(t_phase3, time.time() + max(20.0, (t_phase3 - time.time()) * 0.3))
        h3, c53, n3 = optimize_at_n(hr, best_n, t_sub, seed + 100 + restart)
        if c53 < best_c5:
            best_c5, best_h, best_n = c53, h3, n3

    t_final = start + budget_s * 0.97
    h_fin, c5_fin = proj_grad(best_h, 2.0 / best_n, best_n / 2.0, t_final, beta=20000.0)
    if c5_fin < best_c5:
        best_c5, best_h = c5_fin, h_fin
    h_fin2, c5_fin2 = coord_descent(best_h, 2.0 / best_n, best_n / 2.0, t_final, grid_sz=401)
    if c5_fin2 < best_c5:
        best_c5, best_h = c5_fin2, h_fin2

    dx_f = 2.0 / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
