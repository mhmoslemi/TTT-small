    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _soft_max(corr, alpha):
    m = corr.max()
    return (m + np.log(np.exp(alpha * (corr - m)).sum()) / alpha)

def _objective(y, alpha, beta, n, dx):

            np.clip(h, 0.0, 1.0, out=h)
