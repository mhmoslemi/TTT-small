```python
import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0
_ALPHA = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.clip(alpha * (corr - m), -100, 100)
    return (m + np.log(np.sum(np.exp(w))) / alpha) * dx

def sum_constraint(x):
    return np.sum(x) - _TARGET_SUM

def soft_max_obj(x):
    return soft_max_c5(x, _ALPHA)

def slsqp_solve(h, n, target, alpha, maxiter):
    global _TARGET_SUM, _ALPHA
    _TARGET_SUM = target
    _ALPHA = alpha
    try:
        res = minimize(soft_max_obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_constraint}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, target)
    except Exception:
        return h.copy()

def fix_sum(cur, target):
    diff = target - cur.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = cur < 1.0 - 1e-13
        else:
            mask = cur > 1e-13
        c = mask.sum()
        if c > 0:
            cur[mask] += diff / c
            np.clip(cur, 0.0, 1.0, out=cur)

def anneal(h, n, target, rng, deadline, max_steps, T0, slsqp_every):
    bh = h.copy()
    bc = compute_c5(h)
    cur = h.copy()
    cc = bc
    T = T0
    T_min = 1e-9
    cool = (T_min / T) ** (1.0 / max(1, max_steps))
    sigma = 0.035
    for st in range(max_steps):
        if time.time() > deadline:
            break
        if st > 0 and st % slsqp_every == 0 and time.time() < deadline - 80:
            a = min(300 + st // 50, 5000)
            hs = slsqp_solve(bh, n, target, a, 80)
            cs = compute_c5(hs)
            if cs < bc - 1e-12:
                bc, bh = cs, hs
                cur, cc = hs.copy(), cs
            T = max(T, 0.05)
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        d = rng.normal(0, sigma)
        oi, oj = cur[i], cur[j]
        cur[i] = oi + d
        cur[j] = oj - d
        np.clip(cur, 0.0, 1.0, out=cur)
        fix_sum(cur, target)
        cn = compute_c5(cur)
        if cn <= cc or rng.random() < np.exp(max(-(cn - cc) / T, -50)):
            cc = cn
            if cn < bc:
                bc, bh = cn, cur.copy()
        else:
            cur[i], cur[j] = oi, oj
            fix_sum(cur, target)
        T = max(T * cool, T_min)
        if st > 0 and st % 30000 == 0:
            sigma *= 0.93
    return bh, bc

def polish(h, n, target, rng, deadline, max_iters):
    bh = h.copy()
    bc = compute_c5(h)
    step = 0.015
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 8000 == 0:
            step *= 0.9
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        d = rng.uniform(-step, step)
        th = bh.copy()
        th[i] += d
        th[j] -= d
        np.clip(th, 0.0, 1.0, out=th)
        th = make_feasible(th, target)
        c = compute_c5(th)
        if c < bc:
            bh, bc = th, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dl = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    n = len(h0)
    tgt = n / 2.0
    h0 = make_feasible(h0, tgt)
    bh, bc = h0.copy(), compute_c5(h0)

    # Phase 1: SLSQP warmup with increasing alpha
    for a in [30, 80, 200, 500]:
        if time.time() > dl - 700:
            break
        h1 = slsqp_solve(bh, n, tgt, a, 200)
        c1 = compute_c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 2A: Main simulated annealing (hot exploration)
    bh, bc = anneal(bh, n, tgt, rng, dl - 350, 500000, 0.25, 5000)

    # Phase 2B: Second annealing (cooler refinement)
    if time.time() < dl - 180:
        bh, bc = anneal(bh, n, tgt, rng, dl - 180, 300000, 0.08, 5000)

    # Phase 3: SLSQP refinement with high alpha
    for a in [1000, 2000, 4000, 8000]:
        if time.time() > dl - 25:
            break
        h1 = slsqp_solve(bh, n, tgt, a, 150)
        c1 = compute_c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Phase 4: Greedy local search polish
    bh, bc = polish(bh, n, tgt, rng, dl - 5, 120000)

    # Phase 5: Final SLSQP
    if time.time() < dl - 2:
        h1 = slsqp_solve(bh, n, tgt, 10000, 80)
        c1 = compute_c5(h1)
        if c1 < bc:
            bh, bc = h1, c1

    # Final validation
    bh = make_feasible(bh, tgt)
    fc = compute_c5(bh)
    if fc < bc:
        bc = fc
    return bh.tolist(), bc, n
```
