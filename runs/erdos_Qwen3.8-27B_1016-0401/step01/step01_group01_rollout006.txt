            nn = len(x)
            ks = np.arange(nn)
            grad = np.zeros(nn)
            for m in range(2 * nn - 1):
                if w[m] < 1e-15:
                    continue
                sh = m - (nn - 1)
                g = np.zeros(nn)
                v1 = (ks - sh >= 0) & (ks - sh < nn)
                g[v1] += (1.0 - x[ks[v1] - sh])
                v2 = (ks + sh >= 0) & (ks + sh < nn)
                g[v2] -= x[ks[v2] + sh]
                grad += w[m] * g
            return _dx * grad
        try:
            r = minimize(obj_w, h, jac=jac_w, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': iters_per_round, 'ftol': 1e-14})
            h = np.clip(r.x, 0, 1)
            h = _norm(h, T)
        except Exception:
            pass
        c_full = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(c_full)
        if c_max > 1e-15:
            w = w * np.exp(beta * c_full / max(c_max, 1e-10))
            w /= w.sum()
        c_h = compute_c5(h, _dx)
        if c_h < bc:
            bc, best = c_h, h.copy()
    return _norm(best, T), bc

def run(seed=42, budget_s=1100, **kwargs):
    global _dx, _n_pts, _T_t, _t_sm
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = np.full(70, 0.5)
    n0 = len(h0)
    h0 = _norm(h0, n0 / 2.0)

    best_h, best_c = h0.copy(), compute_c5(h0, 2.0 / n0)
    best_n = n0

    def try_update(h_new, c_new, n_new):
        nonlocal best_h, best_c, best_n
        if c_new < best_c - 1e-10:
            best_h, best_c, best_n = h_new, c_new, n_new

    # Phase 1: SLSQP from parent at n0
    if elapsed() < budget_s - 60:
        h_s, c_s = _run_slsqp(h0, n0, 3000, 0.05)
        try_update(h_s, c_s, n0)

    # Phase 2: Binary SA from parent
    if elapsed() < budget_s - 60:
        hb = (h0 > 0.5).astype(float)
        hn = n0 / 2.0
        if abs(hb.sum() - hn) > 0.01:
            idx = np.argsort(h0)
            while hb.sum() < hn - 0.01:
                hb[idx[int(hb.sum())]] = 1.0 if hb[idx[int(hb.sum())]] == 0 else 0
                idx = idx[1:] if hb.sum() < hn else idx
            # simpler: just force
            ones_needed = int(hn)
            topk = np.argsort(-h0)[:ones_needed]
            hb = np.zeros(n0)
            hb[topk] = 1.0
        c50 = 2.0 / n0
        for sa_seed in range(8):
            if elapsed() > budget_s - 30:
                break
            hb_r, c5r = _bin_sa(hb, n0, c50, 40000, seed + sa_seed)
            try_update(hb_r, c5r, n0)
            # Soften and re-optimize
            if elapsed() < budget_s - 30:
                hs, cs = _soften_and_opt(hb_r, n0, 500, 0.03)
                try_update(hs, cs, n0)

    # Phase 3: Iterative reweighting at n0
    if elapsed() < budget_s - 100:
        h_rw, c_rw = _iter_reweight(best_h, n0, 10, 500)
        try_update(h_rw, c_rw, n0)

    # Phase 4: Alternate grid sizes
    for n_alt in [50, 90, 100, 140]:
        if elapsed() > budget_s - 40:
            break
        # Project parent
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_alt)
        h_proj = np.interp(x_new, x_old, best_h)
        h_proj = _norm(h_proj, n_alt / 2.0)
        # SLSQP
        if elapsed() < budget_s - 30:
            h_p, c_p = _run_slsqp(h_proj, n_alt, 1000, 0.05)
            try_update(h_p, c_p, n_alt)
        # Binary SA at this size
        if elapsed() < budget_s - 30:
            hb2 = np.zeros(n_alt)
            topk2 = np.argsort(-h_proj)[:n_alt // 2]
            hb2[topk2] = 1.0
            c5a = 2.0 / n_alt
            hb_r2, c_r2 = _bin_sa(hb2, n_alt, c5a, 30000, seed + 100 + n_alt)
            try_update(hb_r2, c_r2, n_alt)
            if elapsed() < budget_s - 30:
                hs2, cs2 = _soften_and_opt(hb_r2, n_alt, 400, 0.03)
                try_update(hs2, cs2, n_alt)
        # Random init
        if elapsed() < budget_s - 30:
            hr = rng.random(n_alt)
            hr, cr = _run_slsqp(hr, n_alt, 800, 0.05)
            try_update(hr, cr, n_alt)

    # Phase 5: More SA restarts at best n
    if elapsed() < budget_s - 20:
        bn = best_n
        dx_b = 2.0 / bn
        for sa_s in range(15):
            if elapsed() > budget_s - 10:
                break
            # perturb best
            h_pert = best_h.copy()
            k = max(1, bn // 14)
            indices = rng.choice(bn, size=k * 2, replace=False)
            for ii in range(0, k * 2, 2):
                h_pert[indices[ii]], h_pert[indices[ii + 1]] = h_pert[indices[ii + 1]], h_pert[indices[ii]]
            h_pert = _norm(h_pert, bn / 2.0)
            hb3 = (h_pert > 0.5).astype(float)
            hn3 = bn / 2.0
            if hb3.sum() != hn3:
                diffs = h_pert - 0.5
                order = np.argsort(-np.abs(diffs))
                ones = np.sum(hb3 == 1)
                if ones < hn3:
                    for idx2 in order:
                        if hb3.sum() >= hn3:
                            break
                        if hb3[int(idx2)] == 0:
                            hb3[int(idx2)] = 1.0
                elif ones > hn3:
                    for idx2 in order:
                        if hb3.sum() <= hn3:
                            break
                        if hb3[int(idx2)] == 1:
                            hb3[int(idx2)] = 0.0
            hb_r3, c_r3 = _bin_sa(hb3, bn, dx_b, 30000, seed + 200 + sa_s)
            try_update(hb_r3, c_r3, bn)

    # Phase 6: Final polish
    if elapsed() < budget_s - 5:
        h_pol, c_pol = _run_slsqp(best_h, best_n, 2000, 0.02)
        try_update(h_pol, c_pol, best_n)

    # Final normalization
    bn = len(best_h)
    dx_f = 2.0 / bn
    best_h = np.clip(best_h, 0, 1)
    d = bn / 2.0 - best_h.sum()
    if abs(d) > 1e-12:
        # Adjust smallest/magnitude
        if d > 0:
            order = np.argsort(-best_h)
            for i in order:
                a = min(best_h[i] + d / max(1, len(order)), 1.0) - best_h[i]
                best_h[i] += a
                d -= a
                if abs(d) < 1e-12:
                    break
        else:
            order = np.argsort(best_h)
            for i in order:
                a = max(best_h[i + 0] + d / max(1, len(order)), 0.0) - best_h[i]
                best_h[i] += a
                d -= a
                if abs(d) < 1e-12:
                    break
    best_h = np.clip(best_h, 0, 1)
    # Simple fix: adjust uniform
    d2 = bn / 2.0 - best_h.sum()
    if abs(d2) > 1e-9:
        adj = np.full(bn, d2 / bn)
        best_h = np.clip(best_h + adj, 0, 1)
    best_c = compute_c5(best_h, dx_f)

    return best_h, best_c, bn
