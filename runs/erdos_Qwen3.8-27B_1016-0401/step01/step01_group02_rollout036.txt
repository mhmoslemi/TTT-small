```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sm_grad(h, s, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    ec = np.exp(s * (corr - m))
    w = ec / ec.sum()
    omh = 1.0 - h
    idx = np.arange(n)
    grad = np.zeros(n)
    for j in range(n):
        ia = n - 1 + j - idx
        ib = n - 1 + idx - j
        grad[j] = np.dot(w[ia], omh) - np.dot(w[ib], h)
    grad *= dx * s
    fval = (m + np.log(ec.sum()) / s) * dx
    return grad, fval

def _adam(h0, n, tmax, lr0=0.02, sc=50.0):
    dx = 2.0 / n
    h = h0.copy()
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h0.copy()
    best_c = _c5(h0)
    t0 = time.time()
    
    while time.time() - t0 < tmax:
        t += 1
        # Adaptive scale
        s = sc * (1.0 + 0.5 * np.sin(0.01 * t))
        grad, fval = _sm_grad(h, s, dx)
        
        lr = lr0 / (1.0 + 0.01 * t)
        mv = b1 * mv + (1 - b1) * grad
        vv = b2 * vv + (1 - b2) * grad * grad
        mh = mv / (1 - b1**t)

        c = _c5(h_new)
        if c < best_c:

            best_h = h_new.copy()
        h = h_new
```

```python
def _subgrad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = corr.argmax()
    shift = k_star - (n - 1)
    omh = 1.0 - h
    idx = np.arange(n)
    # d(corr[k_star])/dh[j] = (1-h[j-shift])*[0<=j-shift<=n-1] - h[j+shift]*[0<=j+shift<=n-1]
    grad = np.zeros(n)
    valid1 = (idx - shift >= 0) & (idx - shift < n)
    valid2 = (idx + shift >= 0) & (idx + shift < n)
    grad[valid1] += omh[idx[valid1] - shift]
    grad[valid2] -= h[idx[valid2] + shift]
    grad *= dx
    return grad, corr.max() * dx
```

```python
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-12:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sm_grad(h, s, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    ec = np.exp(s * (corr - m))
    w = ec / ec.sum()
    omh = 1.0 - h
    idx = np.arange(n)
    grad = np.zeros(n)
    for j in range(n):
        ia = n - 1 + j - idx
        ib = n - 1 + idx - j
        grad[j] = np.dot(w[ia], omh) - np.dot(w[ib], h)
    grad *= dx * s
    fval = (m + np.log(ec.sum()) / s) * dx
    return grad, fval

def _subgrad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = corr.argmax()

    grad = np.zeros(n)
    v1 = idx - sh
    v2 = idx + sh

    m2 = (v2 >= 0) & (v2 < n)

    grad[m2] -= h[v2[m2]]
```
