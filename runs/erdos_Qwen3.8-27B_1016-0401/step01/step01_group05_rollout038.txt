import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_obj_grad(alpha, penalty, target):
    def obj_and_grad(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -400, 400)
        exps = np.exp(shifted)
        s = np.sum(exps)
        f = (m + np.log(s) / alpha) * dx
        f += penalty * (np.sum(h) - target) ** 2
        
        p = exps / s
        # Gradient
        cHp = np.correlate(h, p, 'full')
        cph = np.convolve(p, h, 'full')
        A = cHp[n-1:2*n-1][::-1]
        B = cph[n-1:2*n-1]
        grad = (1.0 - A - B) * dx + 2.0 * penalty * (np.sum(h) - target)
        return f, grad
    return obj_and_grad

def lbfgs_optimize(h, alpha, penalty, maxiter=500, **opts):
    n = len(h)
    target = n / 2.0
    obj_grad = make_obj_grad(alpha, penalty, target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h.copy(), method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8,
                      'ftol': 1e-14, 'maxfun': maxiter * 5})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def coord_descent(h, n, target, rng, n_iters, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    step = 0.1
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 2000 == 0 and it > 0:
            step *= 0.9
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        delta = rng.uniform(-step, step)
        h_try = h_curr.copy()
        h_try[i] = np.clip(h_try[i] + delta, 0, 1)
        h_try[j] = np.clip(h_try[j] - delta, 0, 1)
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_curr = h_try
            best_c5 = c5_try
    return h_curr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(70) * 0.5
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    # Try multiple resolutions
    for n in [140, 200, 300]:
        if time.time() > deadline - 120:
            break
        # Interpolate
        x_old = np.linspace(0, 2, len(h_parent), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_parent, period=2)
        h_interp = project_feasible(h_interp, n / 2.0)
        
        c5_interp = compute_c5(h_interp)
        
        # Optimize with continuation
        for alpha in [20, 50, 100, 200, 400, 800]:
            if time.time() > deadline - 60:
                break
            penalty = 100.0
            h_new = lbfgs_optimize(h_interp, alpha, penalty, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < c5_interp:
                h_interp = h_new
                c5_interp = c5_new
        
        # Local search
        if time.time() < deadline - 30:
            h_ls, c5_ls = coord_descent(h_interp, n, n//2, rng, 20000, deadline - 20)
            if c5_ls < c5_interp:
                h_interp = h_ls
                c5_interp = c5_ls
        
        if c5_interp < best_c5:
            best_h = h_interp
            best_c5 = c5_interp
    
    # Final polish at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 20:
        for alpha in [500, 1000]:
            if time.time() > deadline - 15:
                break
            h_new = lbfgs_optimize(best_h, alpha, 200.0, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha, penalty, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400, 400)
    exps = np.exp(shifted)
    s = np.sum(exps)
    f = (m + np.log(s) / alpha) * dx
    f += penalty * (np.sum(h) - target) ** 2
    
    p = exps / s
    cHp = np.correlate(h, p, 'full')
    cph = np.convolve(h, p, 'full')
    A = cHp[n-1:2*n-1][::-1]
    B = cph[0:n]
    grad = (1.0 - A - B) * dx + 2.0 * penalty * (np.sum(h) - target)
    return f, grad

def lbfgs_step(h, alpha, penalty, maxiter=400):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_obj_grad(x, alpha, penalty, target),
            h.copy(), method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-16, 'maxfun': maxiter*10}
        )
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

    s = np.sum(exps)
    f = (m + np.log(s) / alpha) * dx
    f += penalty * (np.sum(h) - target) ** 2
    p = exps / s
    cHp = np.correlate(h, p, 'full')
    cph = np.convolve(h, p, 'full')
    A = cHp[n - 1:2 * n - 1][::-1]
    B = cph[0:n]
    grad = (1.0 - A - B) * dx + 2.0 * penalty * (np.sum(h) - target)
    return f, grad

def _obj_wrapper(x):
    return smooth_obj_grad(x, _g_alpha, _g_penalty, _g_target)

def lbfgs_step(h, alpha, penalty, maxiter=400):
    global _g_alpha, _g_penalty, _g_target
    n = len(h)
    target = n / 2.0
    _g_alpha = alpha
    _g_penalty = penalty
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_wrapper, h.copy(), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-9,
                                'ftol': 1e-16, 'maxfun': maxiter * 8})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def coord_search(h, n, target, rng, deadline, max_iters=40000):
    best_c5 = compute_c5(h)
    h_c = h.copy()
    step = 0.08
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            step *= 0.87
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (i + 1) % n
        d = rng.uniform(-step, step)
        h_t = h_c.copy()
        h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
        h_t[j] = np.clip(h_t[j] - d, 0.0, 1.0)
        h_t = project_feasible(h_t, target)
        c = compute_c5(h_t)
        if c < best_c5:
            h_c = h_t
            best_c5 = c
    return h_c, best_c5

def interp_to(h_src, n):
    x_old = np.arange(len(h_src)) * (2.0 / len(h_src))
    x_new = np.arange(n) * (2.0 / n)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.asarray(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(70) * 0.5

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Multi-resolution optimization
    for n in [140, 200, 300, 500]:
        if time.time() > deadline - 180:
            break
        h_n = interp_to(h_parent, n)
        h_n = project_feasible(h_n, n / 2.0)
        c5_cur = compute_c5(h_n)

        # Continuation L-BFGS
        alphas = [25, 60, 150, 400, 1000] if n <= 200 else [40, 100, 300, 800]
        for alpha in alphas:
            if time.time() > deadline - 60:
                break
            pen = 150.0 if n <= 200 else 300.0
            h_n = lbfgs_step(h_n, alpha, pen, maxiter=500)
            c5_new = compute_c5(h_n)
            c5_cur = min(c5_cur, c5_new) if c5_new < c5_cur else c5_cur
            if c5_new < c5_cur + 1e-8:
                c5_cur = c5_new

        # Coordinate descent polish
        if time.time() < deadline - 20:
            h_n, c5_cur = coord_search(h_n, n, n // 2, rng, deadline - 15,
                                       max_iters=30000 if n <= 200 else 20000)

        if c5_cur < best_c5:
            best_h = h_n.copy()
            best_c5 = c5_cur

    # Try alternative starts: smooth bump + parent average
    if time.time() < deadline - 120:
        n_alt = 200
        x = np.arange(n_alt) * (2.0 / n_alt)
        h_bump = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        h_bump = h_bump / np.sum(h_bump) * (n_alt / 2.0)
        h_bump = project_feasible(h_bump, n_alt / 2.0)
        h_avg = 0.5 * interp_to(h_parent, n_alt) + 0.5 * h_bump
        h_avg = project_feasible(h_avg, n_alt / 2.0)
        for h_start, tag in [(h_bump, 'bump'), (h_avg, 'avg')]:
            if time.time() > deadline - 40:
                break
            h_try = h_start.copy()
            for alpha in [50, 200, 600]:
                if time.time() > deadline - 30:
                    break
                h_try = lbfgs_step(h_try, alpha, 200.0, maxiter=400)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    # Final L-BFGS polish
    if time.time() < deadline - 25:
        n_b = len(best_h)
        for alpha in [1500, 3000]:
            if time.time() > deadline - 12:
                break
            best_h = lbfgs_step(best_h, alpha, 500.0, maxiter=300)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5:
                best_c5 = c5_new

    # Final coord search
    if time.time() < deadline - 5:
        n_b = len(best_h)
        best_h, best_c5 = coord_search(best_h, n_b, n_b // 2, rng,
                                       deadline - 2, max_iters=15000)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, len(best_h)
