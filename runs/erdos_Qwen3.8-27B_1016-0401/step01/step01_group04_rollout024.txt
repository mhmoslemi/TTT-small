    c5_cur = compute_c5(h_hc)
    best_h = h_hc.copy()
    best_c5_loc = min(c5_cur, best_c5)
    
    one_idx = np.where(h_hc == 1)[0]
    n_imp = 0

        improved = False
        # Try all single moves: for each 1, try moving to each 0
        for i in one_idx:
            if time.time() > t_deadline:
                return best_h, best_c5_loc
            zero_idx = np.where(h_hc == 0)[0]
            for j in zero_idx:
                h_hc[i], h_hc[j] = h_hc[j], h_hc[i]
                c5_new = compute_c5(h_hc)
                if c5_new < best_c5_loc - 1e-14:
                    best_c5_loc = c5_new
                    c5_cur = c5_new
                    best_h = h_hc.copy()
                    improved = True
                    n_imp += 1
                else:
                    h_hc[i], h_hc[j] = h_hc[j], h_hc[i]
        if not improved:
            break
    
    return best_h, best_c5_loc

    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    def obj(x):

        m = c.max()
        return m + np.log(np.sum(np.exp(T * (c - m)))) / T
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

    except:
        return h.copy()
