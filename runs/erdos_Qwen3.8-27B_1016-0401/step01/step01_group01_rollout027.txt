def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, T):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > T:
            lo = tau
        else:
            hi = tau
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

_G = {}

def _obj_pp(h):
    c = np.maximum(np.correlate(h, 1.0 - h, mode='full') * _G['dx'], 0.0)
    p = _G['p']
    return float(np.sum(c ** p) ** (1.0 / p))

def _jac_pp(h):
    n = len(h)
    dx = _G['dx']
    p = _G['p']
    c = np.maximum(np.correlate(h, 1.0 - h, mode='full') * dx, 0.0)
    base = float(np.sum(c ** p))
    if base < 1e-30:
        return np.zeros(n)
    wp1 = c ** (p - 1)
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        w = wp1[m]
        if w < 1e-20:

        v1 = (ks - s >= 0) & (ks - s < n)
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v1] += w * (1.0 - h[ks[v1] - s])
        grad[v2] -= w * h[ks[v2] + s]
    return dx * base ** (1.0 / p - 1.0) * grad

        h_r = _coord_descent(h_r, 60)
        c5_r = compute_c5(h_r, 2.0 / n0)
        if c5_r < best_c5:
            h_r = _slsqp_max(h_r, 1000, 1e-13)
            h_r = _coord_descent(h_r, 80)
            c5_r = compute_c5(h_r, 2.0 / n0)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    # Phase 4: Finer grids
    for n_alt in [100, 140]:
        if time.time() - t0 > budget_s - 80:
            break
        _G['n'] = n_alt
        _G['dx'] = 2.0 / n_alt
        _G['T'] = n_alt / 2.0
        h_alt = _interp_to(best_h, n_alt)
        h_alt = project_box_sum(h_alt, n_alt / 2.0)
        h_alt = _anneal_p(h_alt, t0, budget_s - 60)
        h_alt = _coord_descent(h_alt, 80)
        c5_alt = compute_c5(h_alt, 2.0 / n_alt)
        if c5_alt < best_c5:
            h_alt = _slsqp_max(h_alt, 2000, 1e-13)
            h_alt = _coord_descent(h_alt, 100)
            c5_alt = compute_c5(h_alt, 2.0 / n_alt)
            if c5_alt < best_c5:
                best_c5, best_h = c5_alt, h_alt

    # Phase 5: Final polish at best grid size
    if time.time() - t0 < budget_s - 15:
        n_f = len(best_h)
        _G['n'] = n_f
        _G['dx'] = 2.0 / n_f
        _G['T'] = n_f / 2.0
        h_f = _anneal_p(best_h, t0, budget_s - 10)
        h_f = _slsqp_max(h_f, 4000, 1e-14)
        h_f = _coord_descent(h_f, 200)
        c5_f = compute_c5(h_f, 2.0 / n_f)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    n_f = len(best_h)
    best_h = project_box_sum(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n_f)
    return best_h, best_c5, n_f
