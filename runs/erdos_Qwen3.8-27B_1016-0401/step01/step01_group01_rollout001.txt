import numpy as np
from scipy.optimize import minimize
import time

_SLSQP_STATE = {}

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, T):
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(h - lam, 0.0, 1.0)
        if proj.sum() > T:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj_vals(h, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    return dx * (cmax + t * np.log(np.sum(np.exp((c - cmax) / t))))

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    w = np.exp((c - cmax) / t)

        v2 = (ks + s_val >= 0) & (ks + s_val < n)
        grad[v2] += -w[m] * h[ks[v2] + s_val]
    return dx * grad

    t = _SLSQP_STATE['t']

    if t < 1e-14:

        grad[v2] += -x[ks[v2] + s_val]

def _cons_fun(x):

def _cons_jac(x):

    _SLSQP_STATE['dx'] = dx
    _SLSQP_STATE['T'] = T
    bounds = [(0.0, 1.0)] * len(h)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h = project_feasible(h, T).copy()
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    # Sharp max
    _SLSQP_STATE['t'] = 0.0
    try:
        r = minimize(_slsqp_obj, h, method='SLSQP', jac=_slsqp_jac,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': 1e-16})
        r.x = project_feasible(r.x, T)
        c5 = compute_c5(r.x, dx)
        if c5 < best_c5:
            best_c5, best = c5, r.x.copy()
    except Exception:
        pass
    # Smooth max polish

    try:
        r2 = minimize(_slsqp_obj, best, method='SLSQP', jac=_slsqp_jac,
