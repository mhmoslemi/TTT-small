import numpy as np
import time

def compute_c5_binary(h):
    """Fast C5 for binary h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return corr.max() * dx

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    ones = int(h_bin.sum())
    if ones > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])
        for i in range(ones - target):
            h_bin[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(target - ones):
            h_bin[idx[order[i]]] = 1
    return h_bin

def fast_max_corr(h):
    """Return max of full correlation of h with 1-h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max()

def hill_climb(h, n, max_iters=5000, t_deadline=None):
    """Aggressive hill climbing with all-pair swap evaluation."""
    h = h.copy()
    n = len(h)
    target = n // 2
    # Ensure correct number of ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_corr = fast_max_corr(h)
    
    for iteration in range(max_iters):
        if t_deadline and time.time() > t_deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                mc = fast_max_corr(h)
                if mc < best_corr:
                    best_corr = mc
                    improved = True
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    return h, best_corr
