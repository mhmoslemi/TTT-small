```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, S):
    n = len(h)
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def best_improvement_hc(h, n, max_sweeps=200, t_deadline=None):
    """Best-improvement hill climbing for binary h."""
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if t_deadline and time.time() > t_deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        best_delta = 1e-12
        best_pair = None
        for i in one_idx:
            for j in zero_idx:
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                delta = c5 - c5_new
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            c5 -= best_delta
            best_h = h.copy()
            best_c5 = c5
        else:
            break
    return best_h, best_c5
```

```python
        order = np.argsort(h[idx])
```
