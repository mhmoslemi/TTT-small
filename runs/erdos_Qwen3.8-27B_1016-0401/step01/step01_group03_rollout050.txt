    # Ensure exactly n/2 ones

        if len(ones) < nsw or len(zeros) < nsw:

        swap_o = np.random.choice(ones, nsw, replace=False)

        current[swap_z] = 1

            if new_val < best_val:

from scipy.optimize import minimize
from functools import partial

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_max_lag_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    return corr[idx], idx

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    # Clip to avoid overflow
    shifted = np.clip(shifted, -20/beta, 0)
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for beta in [beta_start, beta_start*3, beta_start*10, beta_start*30]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_sum, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    # Projected gradient with adaptive step
    beta = beta_start * 30
    lr = 1.0
    stall = 0
    for it in range(200000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 10.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.95
        h = h_new
        if stall > 2000 or lr < 1e-8:
            break
    
    return best_h, best_c5

def sa_binary(h0, dx, t_limit):
    """Simulated annealing on binary sequences with focused moves."""
    n = len(h0)
    target = n // 2
    current = (h0 > 0.5).astype(int).copy()
    # Fix count
    ones = np.where(current == 1)[0]
    if len(ones) > target:
        np.random.shuffle(ones)
        for i in ones[:len(ones)-target]:
            current[i] = 0
    elif len(ones) < target:
        zeros = np.where(current == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target-len(ones)]:
            current[i] = 1
    
    best = current.copy()
    best_h = best.astype(float)
    best_val = compute_c5(best_h, dx)
    curr_val = best_val
    
    T = 1.0
    T_min = 0.001
    cooling = 0.999
    
    start = time.time()
    while time.time() < t_limit and T > T_min:
        # Choose move type
        r = np.random.rand()
        if r < 0.6:
            # Single swap
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            oi = np.random.choice(ones)
            zi = np.random.choice(zeros)
            if oi == zi:
                continue
            current[oi] = 0
            current[zi] = 1
            new_val = compute_c5(current.astype(float), dx)
            delta = new_val - curr_val
            if delta < 0 or np.random.rand() < np.exp(-delta / (T * dx)):
                curr_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = current.copy()
                    best_h = current.astype(float)
            else:
                current[oi] = 1
                current[zi] = 0
        else:
            # Double swap
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            oi = np.random.choice(ones, 2, replace=False)
            zi = np.random.choice(zeros, 2, replace=False)
            current[oi[0]] = 0; current[oi[1]] = 0
            current[zi[0]] = 1; current[zi[1]] = 1
            new_val = compute_c5(current.astype(float), dx)
            delta = new_val - curr_val
            if delta < 0 or np.random.rand() < np.exp(-delta / (T * dx)):
                curr_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = current.copy()
                    best_h = current.astype(float)
            else:
                current[oi[0]] = 1; current[oi[1]] = 1
                current[zi[0]] = 0; current[zi[1]] = 0
        
        T *= cooling
    
    return best_h, best_val

def focused_local_search(h0, dx, t_limit):
    """Targeted local search: always try to reduce the worst lag."""
    n = len(h0)
    target = n // 2
    current = (h0 > 0.5).astype(int).copy()
    ones = np.where(current == 1)[0]
    if len(ones) > target:
        np.random.shuffle(ones)
        for i in ones[:len(ones)-target]:
            current[i] = 0
    elif len(ones) < target:
        zeros = np.where(current == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target-len(ones)]:
            current[i] = 1
    
    best = current.copy()
    best_val = compute_c5(best.astype(float), dx)
    curr_val = best_val
    
    start = time.time()
    no_improve = 0
    while time.time() < t_limit and no_improve < 500:
        # Find worst lag
        corr = np.correlate(current.astype(float), 1.0 - current.astype(float), mode='full')
        worst_k = np.argmax(corr)
        worst_m = worst_k - (n - 1)
        
        # Try to fix: find a swap that reduces the worst lag
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        
        best_delta = 0
        best_move = None
        # Try a subset of swaps
        sample_o = np.random.choice(ones, min(len(ones), 30), replace=False)
        sample_z = np.random.choice(zeros, min(len(zeros), 30), replace=False)
        
        for oi in sample_o:
            for zi in sample_z:
                if oi == zi:
                    continue
                current[oi] = 0
                current[zi] = 1
                c = np.correlate(current.astype(float), 1.0 - current.astype(float), mode='full')
                new_max = np.max(c)
                if new_max < corr[worst_k]:  # reduced the worst
                    current[oi] = 1
                    current[zi] = 0
                    best_delta = new_max - corr[worst_k]
                    best_move = (oi, zi, new_max)
                    break
                current[oi] = 1
                current[zi] = 0
            if best_move:
                break
        
        if best_move:
            oi, zi, new_max = best_move
            current[oi] = 0
            current[zi] = 1
            total_max = compute_c5(current.astype(float), dx)
            if total_max < curr_val - 1e-15:
                curr_val = total_max
                if total_max < best_val:
                    best_val = total_max
                    best = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 50 == 0 and len(ones) > 0 and len(zeros) > 0:
                ones = np.where(current == 1)[0]
                zeros = np.where(current == 0)[0]
                oi = np.random.choice(ones)
                zi = np.random.choice(zeros)
                if oi != zi:
                    current[oi] = 0
                    current[zi] = 1
    
    return best.astype(float), best_val

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -50.0 / beta, 0.0)
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def _cons_fun(x, ts):
    return np.sum(x) - ts


def _cons_jac(x):
    return np.ones(len(x))


def gradient_opt(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [200.0, 800.0, 3000.0, 12000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    beta = 12000.0
    lr = 1.0
    stall = 0
    for _ in range(300000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.04, 15.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.94
        h = h_new
        if stall > 3000 or lr < 1e-8:
            break
    return best_h, best_c5
