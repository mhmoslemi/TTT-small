```python
            g[valid_j_backward] += (-h[sp:])           # h[j+sp] for j in [0, n-1-sp]
        else:
            # s = 0
            # corr = sum_j h[j]*(1-h[j]) = sum_j h[j] - sum_j h[j]^2
            # d/dh[j] = 1 - 2*h[j]
            g = 1.0 - 2.0 * h
        
        grad += w[m] * g
    return dx / T * grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_and_grad(h, dx, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(c)
    obj = dx * (cm + T * np.log(np.sum(np.exp((c - cm) / T))))
    w = np.exp((c - cm) / T)
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-16:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[:n - s] += w[m] * (1.0 - h[s:])
            grad[s:] += -w[m] * h[:n - s]
        elif s < 0:
            p = -s
            grad[p:] += w[m] * (1.0 - h[:n - p])
            grad[:n - p] += -w[m] * h[p:]
        else:
            grad += w[m] * (1.0 - 2.0 * h)
    grad *= dx / T
    return obj, grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def anneal_optimize(h_init, n, t_limit, T_start=0.05, T_end=0.001, n_steps=80):
    dx = 2.0 / n
    T_sum = n / 2.0
    h = project_sum(h_init, T_sum).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    Ts = np.geomspace(T_start, T_end, n_steps)
    for T_cur in Ts:
        if time.time() > t_limit:
            break
        def _o(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            return dx * (cm + T_cur * np.log(np.sum(np.exp((c - cm) / T_cur)))) + 50.0 * (x.sum() - T_sum) ** 2
        def _og(x):
            o, g = smooth_obj_and_grad(x, dx, T_cur)
            g += 100.0 * (x.sum() - T_sum)
            return g
        try:
            r = minimize(_o, h, jac=_og, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-10})
            h_new = project_sum(r.x, T_sum)
            c5v = compute_c5(h_new, dx)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_new.copy()
            h = h_new
        except Exception:
            pass
    # Final SLSQP polish with hard constraint
    if time.time() < t_limit - 5:
        def _sobj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(c)
            return dx * (cm + 0.0005 * np.log(np.sum(np.exp((c - cm) / 0.0005))))
        def _sjac(x):
            o, g = smooth_obj_and_grad(x, dx, 0.0005)
            return g
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - T_sum,
                 'jac': lambda x: np.ones(n)}]
        try:
            r2 = minimize(_sobj, h, jac=_sjac, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-14})
            h_new = project_sum(r2.x, T_sum)
            c5v = compute_c5(h_new, dx)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

    best_h = project_sum(best_h, n_f / 2.0)
```
