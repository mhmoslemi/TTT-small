import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project(h, target):
    """Project h to [0,1]^n with sum = target"""
    h = np.clip(h, 0, 1).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum: increase elements with room
            room = 1 - h
            idx = np.argsort(room)[::-1]
            for i in idx:
                s = min(diff, room[i])
                h[i] += s
                diff -= s
                if diff < 1e-12:
                    break
        else:
            # Need to decrease sum: decrease non-zero elements
            idx = np.argsort(h)[::-1]
            for i in idx:
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
                if diff > -1e-12:
                    break
    return h

                h[i] += s
            else:
                s = min(-diff, h[i])
                h[i] -= s
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break

    corr = get_corr(h)
    mx = corr.max()
    return (mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T) * (2.0 / len(h))

def softmax_grad(h, T):
    n = len(h)
    corr = get_corr(h)
    mx = corr.max()
    e = np.exp(T * (corr - mx))
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n-1), n):

        else:
            p = -o
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
    return grad * (2.0 / n)

def get_corr(h):

def project(h, target):
