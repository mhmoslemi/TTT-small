import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def smooth_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    exp_T = np.exp(T * (corr - mx))
    w = exp_T / exp_T.sum()  # softmax weights, length 2n-1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-10:
            continue
        lag = m - (n - 1)
        wm = w[m]
        # d(corr[m])/dh[i] = (1-h[i-lag])*I(0<=i-lag<n) - h[i+lag]*I(0<=i+lag<n)
        # For term 1: i-lag in [0,n-1] means i in [lag, lag+n-1], intersect [0,n-1]
        # For term 2: i+lag in [0,n-1] means i in [-lag, n-1-lag], intersect [0,n-1]
        # Term 1: add (1-h[i-lag]) to grad[i] for valid i
        lo1, hi1 = max(0, lag), min(n - 1, lag + n - 1)
        if lo1 <= hi1:
            i1 = np.arange(lo1, hi1 + 1)
            j1 = i1 - lag
            grad[i1] += wm * (1.0 - h[j1])
        # Term 2: subtract h[i+lag] from grad[i] for valid i
        lo2, hi2 = max(0, -lag), min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            i2 = np.arange(lo2, hi2 + 1)
            j2 = i2 + lag
            grad[i2] -= wm * h[j2]
    return smooth_max_val(h, T), grad

def project(h, S):
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        # Add/subtract uniformly, then clip and fix
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Fix remaining difference
        diff2 = S - h.sum()
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                idx = np.argsort(1.0 - h)
            else:
                idx = np.argsort(h)
            for i in idx:
                step = min(abs(diff2), abs(1.0 - h[i]) if diff2 < 0 else (1.0 - h[i]))
                if diff2 > 0:
                    step = min(diff2, 1.0 - h[i])
                else:
                    step = min(-diff2, h[i])
                h[i] += step if diff2 > 0 else -step
                diff2 -= step if diff2 > 0 else -step
                if abs(diff2) < 1e-12:
                    break
    return h

def lbfgs_smooth(h, n, S, T, maxiter=500, ftol=1e-15):
    # Use L-BFGS-B with the smooth max objective
    def obj_and_grad(x):
        x_proj = project(x, S)
        val, grad = smooth_max_grad(x_proj, T)
        return val, grad
    res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': 2000})
    return project(res.x, S)

def binary_local_search(h, n, S, t_deadline, max_rounds=500):
    """Try all pairwise swaps, accepting any improvement."""
    h_ls = h.astype(float).copy()
    c5_current = compute_c5(h_ls)
    best_h = h_ls.copy()
    best_c5 = c5_current
    for rnd in range(max_rounds):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h_ls == 1)[0]
        zero_idx = np.where(h_ls == 0)[0]
        improved = False
        for i in one_idx:
            for j in zero_idx:
                h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
                c5_new = compute_c5(h_ls)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_ls.copy()
                    improved = True
                else:
                    h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
        if not improved:
            break
    return best_h, best_c5

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(round(S))
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

            h_curr = h_new

But wait, I can be smarter. Instead of recomputing the full correlation for each swap, I can compute the change in the correlation analytically.

If I swap h[i] from 1 to 0 and h[j] from 0 to 1, the change in corr[k] is:
- Position i was contributing (1-h[n]) to corr[m] for certain n, and h[i] to corr[m] for other n
- The change is: -contribution_of_i + contribution_of_j

Where contribution_of_i to corr[m] = (1-h[i-lag_m]) + h[i+lag_m] (wait, this is the gradient, not the value)

Actually, let me think about it differently. corr[m] = sum_n h[n+lag_m] * (1-h[n]).

If I change h[i] from 1 to 0:
- corr[m] loses h[i]*(1-h[i-lag_m]) from the h[n+lag_m] term (when n+lag_m = i, i.e., n = i-lag_m)
- corr[m] loses h[i+lag_m]*(1-h[i]) from the (1-h[n]) term (when n = i)

Wait, (1-h[i]) was 0 before (h[i]=1), and becomes 1 after. So:
- The h[n+lag_m] term: h[i] changes from 1 to 0, so corr[m] decreases by (1-h[i-lag_m]) if i-lag_m is valid
- The (1-h[n]) term: (1-h[i]) changes from 0 to 1, so corr[m] increases by h[i+lag_m] if i+lag_m is valid

Net change to corr[m] from changing h[i]: -(1-h[i-lag_m])*I(valid) + h[i+lag_m]*I(valid)

Similarly for h[j] changing from 0 to 1:
- h[j] changes from 0 to 1: corr[m] increases by (1-h[j-lag_m]) if valid
- (1-h[j]) changes from 1 to 0: corr[m] decreases by h[j+lag_m] if valid

Net change to corr[m] from changing h[j]: +(1-h[j-lag_m])*I(valid) - h[j+lag_m]*I(valid)

Total change to corr[m] = [-(1-h[i-lag_m]) + h[i+lag_m] + (1-h[j-lag_m]) - h[j+lag_m]] * I(valid terms)

This is O(n) per pair to compute the max change, which is faster than O(n²) for the full correlate.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def smooth_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    exp_T = np.exp(T * (corr - mx))
    w = exp_T / exp_T.sum()
    val = mx + np.log(exp_T.sum()) / T
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        lag = m - (n - 1)
        wm = w[m]
        lo1 = max(0, lag)
        hi1 = min(n - 1, lag + n - 1)
        if lo1 <= hi1:
            i1 = np.arange(lo1, hi1 + 1)
            j1 = i1 - lag
            grad[i1] += wm * (1.0 - h[j1])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            i2 = np.arange(lo2, hi2 + 1)
            j2 = i2 + lag
            grad[i2] -= wm * h[j2]
    return val, grad

def project_sum(h, S):
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff2 = S - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff2 > 0:
                step = min(diff2, 1.0 - h[i])
                h[i] += step
                diff2 -= step
            else:
                step = min(-diff2, h[i])
                h[i] -= step
                diff2 += step
            if abs(diff2) < 1e-13:
                break
    return h

def slsqp_opt(h, n, S, T, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    con_sum = {'type': 'eq', 'fun': lambda x: x.sum() - S}
    def obj_fun(x):
        v, g = smooth_max_grad(x, T)
        return v, g
    try:
        res = minimize(obj_fun, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=[con_sum], options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_sum(res.x, S)
    except Exception:
        return h.copy()

def custom_pg(h, n, S, T, n_iter=2000, lr=0.02):
    h_curr = project_sum(h.copy(), S)
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    for it in range(n_iter):
        val, grad = smooth_max_grad(h_curr, T)
        h_new = h_curr - lr * grad
        h_new = project_sum(h_new, S)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
        if it % 200 == 0:
            lr = max(lr * 0.95, 0.001)
    return best_h, best_c5

def binary_local_search(h, n, S, t_deadline, max_rounds=200, strategy='best'):
    h_ls = h.astype(float).copy()
    best_c5 = compute_c5(h_ls)
    best_h = h_ls.copy()
    for rnd in range(max_rounds):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h_ls == 1)[0]
        zero_idx = np.where(h_ls == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        improved = False
        if strategy == 'best':
            best_delta = 1e10
            best_pair = (-1, -1)
            for i in one_idx:
                for j in zero_idx:
                    h_ls[i], h_ls[j] = 0.0, 1.0
                    c5_tmp = compute_c5(h_ls)
                    h_ls[i], h_ls[j] = 1.0, 0.0
                    delta = c5_tmp - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
            if best_pair[0] >= 0 and best_delta < -1e-14:
                i, j = best_pair
                h_ls[i], h_ls[j] = 0.0, 1.0
                best_c5 += best_delta
                best_h = h_ls.copy()
                improved = True
        else:
            for i in one_idx:
                for j in zero_idx:
                    h_ls[i], h_ls[j] = 0.0, 1.0
                    c5_tmp = compute_c5(h_ls)
                    if c5_tmp < best_c5 - 1e-14:
                        best_c5 = c5_tmp
                        best_h = h_ls.copy()
                        improved = True
                    else:
                        h_ls[i], h_ls[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best_c5

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(round(S))
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def random_binary(n, S, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h_b = np.zeros(n)
    h_b[idx[:int(S)]] = 1.0
    return h_b

    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(70, 0.5)

    n = len(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuous optimization (SLSQP + projected gradient)
    t_phase1 = t0 + budget_s * 0.30
    for restart in range(6):
        if time.time() > t_phase1:
            break
        if restart == 0:
            h_curr = best_h.copy()
        else:
            h_curr = best_h + rng.normal(0, 0.02 * (restart + 1), n)
            h_curr = project_sum(h_curr, S)
        for T in [10, 50, 200, 1000]:
            if time.time() > t_phase1:
                break
            h_new = slsqp_opt(h_curr, n, S, T, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
            h_curr = h_new
        # Projected gradient refinement
        if time.time() < t_phase1:
            h_pg, c5_pg = custom_pg(h_curr, n, S, T=500, n_iter=1000, lr=0.01)
            if c5_pg < best_c5 - 1e-12:
                best_h = h_pg
                best_c5 = c5_pg

    # Phase 2: Binary local search (thorough, all-pairs)
    t_phase2 = t0 + budget_s * 0.75
    h_bin = to_binary(best_h, S)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
    h_ls, c5_ls = binary_local_search(best_h, n, S, t_phase2, max_rounds=150, strategy='best')
    if c5_ls < best_c5:
        best_h = h_ls
        best_c5 = c5_ls

    # Also try first-improvement from the result
    h_ls2, c5_ls2 = binary_local_search(best_h, n, S, t_phase2, max_rounds=100, strategy='first')
    if c5_ls2 < best_c5:
        best_h = h_ls2
        best_c5 = c5_ls2

    # Phase 3: Random binary restarts + local search
    t_phase3 = t0 + budget_s * 0.90
    n_restarts = 20
    for r in range(n_restarts):
        if time.time() > t_phase3:
            break
        h_rand = random_binary(n, S, rng)
        h_rs, c5_rs = binary_local_search(h_rand, n, S, t_phase3, max_rounds=80, strategy='first')
        if c5_rs < best_c5 - 1e-12:
            best_h = h_rs
            best_c5 = c5_rs

    # Phase 4: Final continuous refinement + binary
    t_phase4 = t0 + budget_s * 0.97
    if time.time() < t_phase4:
        h_cont = best_h.astype(float)
        for T in [100, 500, 3000]:
            if time.time() > t_phase4:
                break
            h_new = slsqp_opt(h_cont, n, S, T, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
            h_cont = h_new

    if time.time() < t_phase4:
        h_bin2 = to_binary(best_h, S)
        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < best_c5:
            best_h = h_bin2.copy()
            best_c5 = c5_bin2
        h_final, c5_final = binary_local_search(best_h, n, S, t0 + budget_s * 0.98,
                                                 max_rounds=100, strategy='best')
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
