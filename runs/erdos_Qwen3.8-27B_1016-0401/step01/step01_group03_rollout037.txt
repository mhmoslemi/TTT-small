import time

def smooth_obj_and_grad(h, dx, beta):

    corr = np.correlate(h, 1.0 - h, mode='full')

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    n_betas = 8
    betas = np.geomspace(beta_start, beta_end, n_betas)
    t_per_beta = (t_limit - time.time()) / (n_betas + 1)
    
    for beta in betas:
        t_beta_end = time.time() + max(5, t_per_beta * 1.5)
        
        def obj_fun(x):
            return smooth_obj_and_grad(x, dx, beta)[0]
        
        def grad_fun(x):
            return smooth_obj_and_grad(x, dx, beta)[1]

                              bounds=bnds, options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_feasible(res.x, target)

from scipy.optimize import minimize
import time

_gdx = 0.0
_gbe = 0.0

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _sog(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    e = np.exp(np.clip(beta * (c - mc), -500, 0))
    se = np.sum(e)
    obj = dx * (mc + np.log(se) / beta)
    w = dx * e / se
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - n + 1
        if m >= 0:
            g[m:n] += wi * (1.0 - h[:n - m])
            g[:n - m] -= wi * h[m:n]
        else:
            p = -m
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return obj, g

def _ob(x):
    return _sog(x, _gdx, _gbe)[0]

def _gr(x):
    return _sog(x, _gdx, _gbe)[1]

    for _ in range(60):
        md = 0.5 * (lo + hi)
        p = np.clip(h - md, 0.0, 1.0)
        if np.sum(p) > ts:
            lo = md
        else:
            hi = md
    p = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

        else:
            ms = p > 1e-10

    return p

import numpy as np
from scipy.optimize import minimize
import time

_gdx = 0.0
_gbe = 0.0

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _sog(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    e = np.exp(np.clip(beta * (c - mc), -500, 0))
    se = np.sum(e)
    obj = dx * (mc + np.log(se) / beta)
    w = dx * e / se
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - n + 1
        if m >= 0:
            g[m:n] += wi * (1.0 - h[:n - m])
            g[:n - m] -= wi * h[m:n]
        else:
            p = -m
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return obj, g

def _ob(x):
    return _sog(x, _gdx, _gbe)[0]

def _gr(x):
    return _sog(x, _gdx, _gbe)[1]

def _pr(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        md = 0.5 * (lo + hi)
        p = np.clip(h - md, 0.0, 1.0)
        if np.sum(p) > ts:
            lo = md
        else:
            hi = md
    p = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

        else:
            ms = p > 1e-10
        if ms.sum() > 0:
            p[ms] = np.clip(p[ms] + d / ms.sum(), 0.0, 1.0)
    return p
