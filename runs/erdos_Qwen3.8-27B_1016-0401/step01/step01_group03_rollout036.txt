import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_binary(h, target_sum):
    """Project a real vector to a binary vector with exact sum."""
    n = len(h)
    target = int(target_sum)
    # Sort by value descending, take top 'target' as 1s
    sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:target]] = 1.0
    return result

def greedy_swap_search(h, dx, max_iters=1000, t_limit=None):
    """Greedy local search with swap moves."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    improved = True
    it = 0
    while improved and it < max_iters:
        if t_limit and time.time() > t_limit:
            break
        improved = False
        it += 1
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        for oi in ones:
            for zi in zeros:
                best[oi] = 0.0
                best[zi] = 1.0
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                best[oi] = 1.0
                best[zi] = 0.0
            if improved:
                break
    return best, best_c5

def simulated_annealing(h0, dx, t_start, t_end, n_iter, seed=42):
    """Simulated annealing with swap moves."""
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    
    current_c5 = best_c5
    cool = (t_end / t_start) ** (1.0 / max(n_iter, 1))
    
    for i in range(n_iter):
        if time.time() > t_start + 400:  # absolute time limit
            break
        T = t_start * (cool ** i)
        if T < 1e-6:
            break
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h, dx)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    
    return best, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_binary(h, target):
    n = len(h)
    h = (h > 0.5).astype(float)
    s = int(np.sum(h))
    if s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        for k in range(s - target):
            h[idx[k]] = 0.0
    elif s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        for k in range(target - s):
            h[idx[k]] = 1.0
    return h

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return proj

    return np.sum(x) - ts

def _cons_jac(x):
    return np.ones(len(x))

    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for beta in [200.0, 800.0, 3000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    
    # Gradient descent with projection
    beta = 3000.0
    lr = 0.5
    stall = 0
    for it in range(100000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 3.0)

        else:
            stall += 1
            lr *= 0.95
        h = h_new
        if stall > 500 or lr < 1e-6:
            break
    
    return best_h, best_c5

def greedy_search(h0, dx, max_rounds=50, t_limit=None):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h, dx)

        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

                no_improve += 1
            else:

                h[zi] = 0.0
                no_improve += 1

            # Reheat

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _cons_fun(x, ts):
    return np.sum(x) - ts

def _cons_jac(x):
    return np.ones(len(x))

def optimize_smooth(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [200.0, 800.0, 3000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    beta = 3000.0
    lr = 0.5
    stall = 0
    for it in range(80000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 3.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.94
        h = h_new
        if stall > 600 or lr < 1e-6:
            break
    return best_h, best_c5

def greedy_search(h0, dx, max_rounds=80, t_limit=None):
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(max_rounds):
        if t_limit is not None and time.time() > t_limit:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for oi in ones:
            if t_limit is not None and time.time() > t_limit:
                return h, best_c5
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
                    break
                h[oi] = 1.0
                h[zi] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def sa_search(h0, dx, seed, t_limit, n_iter=200000, t0=0.025, t_end=0.0002):
    rng = np.random.RandomState(seed)
    n = len(h0)
    target = n // 2
    h = fix_binary(h0, target, rng)
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    cool = (t_end / t0) ** (1.0 / max(n_iter, 1))
    no_improve = 0
    reheat_limit = 4000
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        T = t0 * cool ** it
        if T < 1e-7:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if rng.random() < 0.15 and len(ones) >= 2 and len(zeros) >= 2:
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            oi1, oi2 = ones[i1], ones[i2]
            zi1, zi2 = zeros[j1], zeros[j2]
            h[oi1] = 0.0; h[oi2] = 0.0; h[zi1] = 1.0; h[zi2] = 1.0
            new_c5 = compute_c5(h, dx)
            if new_c5 <= current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / T):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[oi1] = 1.0; h[oi2] = 1.0; h[zi1] = 0.0; h[zi2] = 0.0
                no_improve += 1
        else:
            oi = ones[rng.randint(len(ones))]
            zi = zeros[rng.randint(len(zeros))]
            h[oi] = 0.0; h[zi] = 1.0
            new_c5 = compute_c5(h, dx)
            if new_c5 <= current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / T):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[oi] = 1.0; h[zi] = 0.0
                no_improve += 1
        if no_improve > reheat_limit:
            no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0
    best_h = fix_binary(h_init, n_init // 2)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    # Phase 1: Smooth optimization (15% budget)
    t1 = start + budget_s * 0.15
    h_smooth, c5_smooth = optimize_smooth(h_init, dx_init, ts_init, t1)
    h_bin = fix_binary(h_smooth, n_init // 2)
    c5_bin = compute_c5(h_bin, dx_init)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: Multi-resolution smooth + SA (50% budget)
    t2 = start + budget_s * 0.65
    n_tries = [44, 46, 48, 50, 52, 56, 60]
    for n_try in n_tries:
        if time.time() > t2:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, n_init)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, h_init)
        h_try = project_feasible(h_interp, ts_t)
        t_sub = min(t2, time.time() + max(5, (t2 - time.time()) * 0.18))
        h_so, c5_so = optimize_smooth(h_try, dx_t, ts_t, t_sub)
        h_bin_t = fix_binary(h_so, n_try // 2)
        c5_bin_t = compute_c5(h_bin_t, dx_t)
        if c5_bin_t < best_c5:
            best_c5 = c5_bin_t
            best_h = h_bin_t.copy()
            best_n = n_try
        if time.time() < t2 - 10:
            t_sa = min(t2, time.time() + max(8, (t2 - time.time()) * 0.35))
            h_sa, c5_sa = sa_search(h_bin_t, dx_t, seed, t_sa,
                                    n_iter=120000, t0=0.02, t_end=0.0002)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_try

    # Phase 3: SA restarts on best_n (25% budget)
    t3 = start + budget_s * 0.90
    n_use = best_n
    dx_use = 2.0 / n_use
    for restart in range(8):
        if time.time() > t3:
            break
        t_sub = min(t3, time.time() + max(8, (t3 - time.time()) * 0.35))
        if restart == 0:
            h_start = best_h.copy()
        else:
            h_start = best_h.copy()
            ones = np.where(h_start == 1)[0]
            zeros = np.where(h_start == 0)[0]
            n_pert = min(2 + restart // 2, len(ones), len(zeros))
            for _ in range(n_pert):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                h_start[oi] = 0.0
                h_start[zi] = 1.0
        h_sa, c5_sa = sa_search(h_start, dx_use, seed + restart * 7919, t_sub,
                                n_iter=180000, t0=0.018, t_end=0.0001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Final greedy polish (10% budget)
    t4 = start + budget_s * 0.98
    dx_f = 2.0 / best_n
    best_h, best_c5 = greedy_search(best_h, dx_f, max_rounds=100, t_limit=t4)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_binary(best_h, best_n // 2)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
