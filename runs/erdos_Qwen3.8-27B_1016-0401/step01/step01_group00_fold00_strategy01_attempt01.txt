def repair_constraint(h, target_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    diff = target_sum - current_sum
    max_possible = np.sum(1.0 - h_clipped)
    if abs(diff) < 1e-7:
        return h_clipped
    if max_possible < abs(diff):
        # Scale all variables to satisfy the constraint
        if diff > 0:
            h_clipped = np.minimum(h_clipped + min(1.0 - h_clipped, diff/max_possible), 1.0)
        else:
            h_clipped = np.maximum(h_clipped - min(h_clipped, -diff/max_possible), 0.0)
        current_sum = np.sum(h_clipped)
        if abs(diff) > 1e-7:
            direction = np.sign(diff)
            h_step = np.ones_like(h_clipped) * min(np.min(h_clipped if diff>0 else 1.0-h_clipped), abs(diff)/min(current_sum-diff if diff<current_sum else abs(diff)))
            h_clipped = h_clipped + direction * h_step
    return h_clipped
