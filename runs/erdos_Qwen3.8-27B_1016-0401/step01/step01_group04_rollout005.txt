import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def greedy_local_search(h, max_sweeps=10000, t_deadline=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for sweep in range(max_sweeps):
        if t_deadline and time.time() > t_deadline:
            break
        
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        
        best_delta = 0
        best_pair = None
        
        for i in one_idx:
            for j in zero_idx:
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                delta = new_c5 - c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        c5 = c5 + best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

def project_constraint(h, S):
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def greedy_ls_best(h, n, t_deadline, max_sweeps=5000):
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        best_delta = -1e-15
        best_pair = None
        for i in one_idx:
            for j in zero_idx:
                h[i], h[j] = 0.0, 1.0
                new_c5 = compute_c5(h)
                delta = new_c5 - c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[i], h[j] = 1.0, 0.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = 0.0, 1.0
        c5 += best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def greedy_ls_first(h, n, rng, t_deadline, max_sweeps=20000):
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        pairs = np.array(list(zip(one_idx, zero_idx)))
        rng.shuffle(pairs)
        improved = False
        for i, j in pairs:
            if h[i] != 1 or h[j] != 0:
                continue
            h[i], h[j] = 0.0, 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-15:
                c5 = new_c5
                improved = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best_c5

def slsqp_optimize(h, n, S, T, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    obj_val = smooth_max_obj(h, T)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, T),
            h.copy(), method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        if res.fun < obj_val:
            return project_constraint(res.x, S)
    except Exception:
        pass
    return h.copy()

def slsqp_2phase(h, n, S, t_deadline):
    T_schedule = [3, 10, 40, 150, 600, 2500, 10000, 40000]
    h_cur = h.copy()
    for T in T_schedule:
        if time.time() > t_deadline:
            break
        h_new = slsqp_optimize(h_cur, n, S, T, maxiter=500)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h_cur)
        if c5_new < c5_old:
            h_cur = h_new
    return h_cur

def make_spread_binary(n, rng, pattern='golden'):
    h = np.zeros(n)
    target = n // 2
    if pattern == 'golden':
        phi = 0.6180339887498949
        for i in range(target):
            pos = int((i * phi + rng.random() * 0.3) % 1.0 * n)
            if h[pos] == 0:
                h[pos] = 1.0
                i += 1
        # Fill remaining
        attempts = 0
        while h.sum() < target and attempts < n * 10:
            pos = rng.randint(n)
            if h[pos] == 0:
                h[pos] = 1.0
            attempts += 1
    elif pattern == 'prime':
        # Use a simple quadratic residue pattern
        p = target + 1
        while p <= n:
            residues = set()
            for i in range(p):
                r = (i * i) % p
                residues.add(r)
            for r in residues:
                if r < n and h[r] == 0:
                    h[r] = 1.0
            p += 1
        while h.sum() < target:
            pos = rng.randint(n)
            if h[pos] == 0:
                h[pos] = 1.0
    else:
        for i in range(target):
            pos = int((i + 0.5) * n / target)
            h[pos % n] = 1.0
        while h.sum() < target:
            pos = rng.randint(n)
            if h[pos] == 0:
                h[pos] = 1.0
    while h.sum() > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones)] = 0
    return h

        if c5_fin < best_c5:

            best_h = h_fin.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
