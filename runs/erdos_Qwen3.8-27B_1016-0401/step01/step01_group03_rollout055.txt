    es = np.exp(beta * s)
    se = es.sum()
    obj = dx * (mc + np.log(se) / beta)
    w = es / se
    g = np.zeros(n)
    ctr = n - 1
    for k in range(2 * n - 1):

        if wk < 1e-15:
            continue

        if d >= 0:

            g[idx + d] += wk * (1.0 - h[idx])
            g[idx] -= wk * h[idx + d]
        else:
            p = -d
            j = np.arange(n - p)
            g[j] += wk * (1.0 - h[j + p])
            g[j + p] -= wk * h[j]
    return obj, g * dx

import numpy as np
import time

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = ts - p.sum()
    if abs(d) > 1e-14:
        m = (p < 1.0 - 1e-10) if d > 0 else (p > 1e-10)
        if m.sum() > 0:
            p[m] += d / m.sum()
    return np.clip(p, 0.0, 1.0)

def topM_grad(h, dx, M=5, sharp=50.0):
    """Gradient of soft-max of top-M correlation terms."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Sort lags by correlation value, keep top M
    if M >= 2 * n - 1:
        ws = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        order = np.argsort(corr)[::-1]
        top = corr[order[:M]]
        w_top = np.exp(sharp * (top - top[0]))
        w_top /= w_top.sum()
        ws = np.zeros(2 * n - 1)
        for i in range(M):
            ws[order[i]] = w_top[i]
    # Compute gradient
    g = np.zeros(n)
    ctr = n - 1
    for k in range(2 * n - 1):
        wk = ws[k]
        if wk < 1e-14:
            continue
        d = k - ctr
        if d >= 0:
            g[d:] += wk * (1.0 - h[:n - d])
            g[:n - d] -= wk * h[d:]
        else:
            p = -d
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    max_c = corr.max()
    # soft obj ≈ max_c + (log-sum-exp of top M) / sharp
    if M < 2 * n - 1:
        order = np.argsort(corr)[::-1]
        top = corr[order[:M]]
        logse = np.log(np.sum(np.exp(sharp * (top - top[0])))) / sharp
        obj = dx * (max_c + logse)
    else:
        obj = dx * max_c
    return obj, g * dx

import numpy as np
import time


def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = ts - p.sum()
    if abs(d) > 1e-14:
        if d > 0:
            m = p < 1.0 - 1e-10
        else:
            m = p > 1e-10
        if m.sum() > 0:
            p[m] += d / m.sum()
    return np.clip(p, 0.0, 1.0)


def smooth_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    es = np.exp(beta * (corr - mc))
    se = es.sum()
    obj = dx * (mc + np.log(se) / beta)
    w = es / se
    g = np.zeros(n)
    ctr = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - ctr
        if d >= 0:
            g[d:] += wk * (1.0 - h[:n - d])
            g[:n - d] -= wk * h[d:]
        else:
            p = -d
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return obj, g * dx


def optimize_smooth(h0, dx, ts, t_end):
    h = project_feasible(h0.astype(float).copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [20.0, 100.0, 500.0, 2000.0, 10000.0]:
        if time.time() >= t_end:
            break
        lr = 0.3
        for _ in range(50000):
            if time.time() >= t_end:
                break
            _, grad = smooth_grad(h, dx, beta)
            ok = False
            for _t in range(30):
                h_t = project_feasible(h - lr * grad, ts)
                c5_t = compute_c5(h_t, dx)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    h = h_t
                    lr = min(lr * 1.05, 10.0)
                    ok = True
                    break
                lr *= 0.5
            if not ok and lr < 1e-10:
                break
    return best_h, best_c5

    h = (h0 > 0.5).astype(float).copy()
    oc = int(h.sum())
    if oc > ts:
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            h[i] = 0
    elif oc < ts:
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            h[i] = 1
    return h


def binary_steepest(h0, dx, t_end, max_iter=50000):
    n = len(h0)
    h = _make_binary(h0, n // 2)
    best_c5 = compute_c5(h, dx)
    for _ in range(max_iter):
        if time.time() >= t_end:
            break
        z = np.where(h == 0)[0]
        o = np.where(h == 1)[0]
        if len(z) == 0 or len(o) == 0:
            break
        best_imp = 0
        best_s = None
        for zi in z:
            for oi in o:
                h[zi] = 1
                h[oi] = 0
                c5 = compute_c5(h, dx)
                imp = best_c5 - c5
                if imp > best_imp:
                    best_imp = imp
                    best_s = (zi, oi)
                h[zi] = 0
                h[oi] = 1
        if best_s is None:
            break
        h[best_s[0]] = 1
        h[best_s[1]] = 0
        best_c5 = compute_c5(h, dx)
    return h, best_c5

    g = np.zeros(n)
    ctr = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - ctr
        if d >= 0:
            g[d:] += wk * (1.0 - h[:n - d])
            g[:n - d] -= wk * h[d:]
        else:
            p = -d
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return obj, g * dx


def optimize_smooth(h0, dx, ts, t_end):
    h = project_feasible(h0.astype(float).copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [20.0, 100.0, 500.0, 2000.0, 10000.0]:
        if time.time() >= t_end:
            break
        lr = 0.3
        for _ in range(50000):
            if time.time() >= t_end:
                break
            _, grad = smooth_grad(h, dx, beta)
            ok = False
            for _t in range(30):
                h_t = project_feasible(h - lr * grad, ts)
                c5_t = compute_c5(h_t, dx)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    h = h_t
                    lr = min(lr * 1.05, 10.0)
                    ok = True
                    break
                lr *= 0.5
            if not ok and lr < 1e-10:
                break
    return best_h, best_c5


def _make_binary(h0, ts):
    h = (h0 > 0.5).astype(float).copy()
    oc = int(h.sum())
    if oc > ts:
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            h[i] = 0
    elif oc < ts:
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            h[i] = 1
    return h


def binary_steepest(h0, dx, t_end, max_iter=50000):
    n = len(h0)
    h = _make_binary(h0, n // 2)
    best_c5 = compute_c5(h, dx)
    for _ in range(max_iter):
        if time.time() >= t_end:
            break
        z = np.where(h == 0)[0]
        o = np.where(h == 1)[0]
        if len(z) == 0 or len(o) == 0:
            break
        best_imp = 0
        best_s = None
        for zi in z:
            for oi in o:
                h[zi] = 1
                h[oi] = 0
                c5 = compute_c5(h, dx)
                imp = best_c5 - c5
                if imp > best_imp:
                    best_imp = imp
                    best_s = (zi, oi)
                h[zi] = 0
                h[oi] = 1
        if best_s is None:
            break
        h[best_s[0]] = 1
        h[best_s[1]] = 0
        best_c5 = compute_c5(h, dx)
    return h, best_c5


def binary_ilsh(h0, dx, t_end, n_restarts=20, k_pert=3):
    n = len(h0)
    ts = n // 2
    h = _make_binary(h0, ts)
    t_sub = min(t_end, time.time() + max(1.0, (t_end - time.time()) * 0.3))
    best_h, best_c5 = binary_steepest(h, dx, t_sub)
    for _ in range(n_restarts):
        if time.time() >= t_end:
            break
        h_p = best_h.copy()
        for _ in range(k_pert):
            z = np.where(h_p == 0)[0]
            o = np.where(h_p == 1)[0]
            if len(z) == 0 or len(o) == 0:
                break
            h_p[z[np.random.randint(len(z))]] = 1
            h_p[o[np.random.randint(len(o))]] = 0
        t_sub = min(t_end, time.time() + max(0.5, (t_end - time.time()) / (n_restarts + 1)))
        h_r, c5_r = binary_steepest(h_p, dx, t_sub)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
    return best_h, best_c5


def upsample_h(h, n_new):
    xo = np.linspace(0.0, 2.0, len(h))
    xn = np.linspace(0.0, 2.0, n_new)
    return np.interp(xn, xo, h)


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    dx0 = 2.0 / n0
    ts0 = n0 / 2.0
    best_h = project_feasible(h_init.copy(), ts0)
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0
    try:
        t = start + budget_s * 0.15
        h1, c5_1 = optimize_smooth(best_h, dx0, ts0, t)
        if c5_1 < best_c5:
            best_c5, best_h, best_n = c5_1, h1, n0
        t = start + budget_s * 0.35
        hb, c5b = binary_ilsh(h1, dx0, t, n_restarts=40, k_pert=3)
        if c5b < best_c5:
            best_c5, best_h, best_n = c5b, hb, n0
        t = start + budget_s * 0.70
        for n_try in [92, 96, 184, 194, 248]:
            if time.time() >= t:
                break
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            hi = project_feasible(upsample_h(best_h, n_try), ts_t)
            t_sm = min(t, time.time() + budget_s * 0.07)
            ho, c5o = optimize_smooth(hi, dx_t, ts_t, t_sm)
            if c5o < best_c5:
                best_c5, best_h, best_n = c5o, ho, n_try
            if n_try <= 96 and time.time() < t:
                t_b = min(t, time.time() + budget_s * 0.04)
                hb2, c5b2 = binary_ilsh(ho, dx_t, t_b, n_restarts=25, k_pert=3)
                if c5b2 < best_c5:
                    best_c5, best_h, best_n = c5b2, hb2, n_try
        tb = start + budget_s * 0.90
        dxb = 2.0 / best_n
        tsb = best_n / 2.0
        hp, c5p = optimize_smooth(best_h, dxb, tsb, tb)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp
        if best_n <= 100:
            t2 = start + budget_s * 0.96
            hb3, c5b3 = binary_ilsh(best_h, dxb, t2, n_restarts=15, k_pert=2)
            if c5b3 < best_c5:
                best_c5, best_h = c5b3, hb3
    except Exception:
        pass
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
