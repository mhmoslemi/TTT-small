import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    h = np.clip(h, 0.0, 1.0)
    return h

def obj_with_penalty(x, target, penalty=100.0):
    h = np.clip(x, 0.0, 1.0)
    c5 = compute_c5(h)
    penalty_val = penalty * (np.sum(h) - target) ** 2
    return c5 + penalty_val

def slsqp_optimize(h, n, target, alpha=200, maxiter=300):
    dx = 2.0 / n
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        m = np.max(corr)
        vals = alpha * (corr - m)
        vals = np.clip(vals, -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target
    
    try:
        res = minimize(
            objective, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = make_feasible(res.x, n, target)
        return h_new
    except Exception:
        return h.copy()

def local_search(h, n, target, rng, max_iters, deadline):

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(h, target, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -60, 60)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def slsqp(h, n, target, alpha, maxiter):
    def obj(x):
        return soft_obj(x, target, alpha)
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':1e-14})
        return make_feasible(res.x, n, target)
    except:
        return h.copy()

def de_obj(x, target, pen=100.0):
    h = np.clip(x, 0, 1)
    return compute_c5(h) + pen*(np.sum(h)-target)**2

def local_search(h, n, target, rng, max_iters, deadline):

    hc = h.copy()
    step = 0.02
    for it in range(max_iters):

            break
        if it % 20000 == 0:
            step *= 0.92
        i = rng.randint(n)
        j = rng.randint(n-1)
        if j >= i: j += 1
        d = rng.uniform(-step, step)
        hi = hc[i] + d
        hj = hc[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        hc[i], hc[j] = hi, hj
        c = compute_c5(hc)
        if c < best - 1e-12:
            best = c
        else:
            hc[i], hc[j] = hi - d, hj + d  # revert
    return hc, best

    h_best = make_feasible(h_best, n, target)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_TARGET = 0.0
_ALPHA = 100.0
_DE_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(_ALPHA * (corr - m), -60.0, 60.0)
    return (m + np.log(np.sum(np.exp(sh))) / _ALPHA) * dx

def sum_con(x):
    return np.sum(x) - _TARGET

def slsqp(h, n, target, alpha, maxiter):
    global _TARGET, _ALPHA
    _TARGET = target
    _ALPHA = alpha
    try:
        res = minimize(soft_obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def de_objective(x):
    h = np.clip(x, 0.0, 1.0)
    return compute_c5(h) + 200.0 * (np.sum(h) - _DE_TGT) ** 2

def sa_search(h, n, target, rng, max_iters, deadline):
    best_c5 = compute_c5(h)
    hc = h.copy()
    cur = best_c5
    T = 0.004
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 40000 == 0 and it > 0:
            T *= 0.85
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        d = rng.uniform(-0.04, 0.04)
        hi = hc[i] + d
        hj = hc[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        hc[i] = hi
        hc[j] = hj
        nc = compute_c5(hc)
        if nc <= cur or rng.rand() < np.exp(-(nc - cur) / max(T, 1e-10)):
            cur = nc
            if nc < best_c5:
                best_c5 = nc
        else:
            hc[i] -= d
            hc[j] += d
    return hc, best_c5

def greedy_ls(h, n, target, rng, max_iters, deadline):
    best_c5 = compute_c5(h)
    hc = h.copy()
    step = 0.012
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 25000 == 0 and it > 0:
            step *= 0.88
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        d = rng.uniform(-step, step)
        hi = hc[i] + d
        hj = hc[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        hc[i] = hi
        hc[j] = hj
        c = compute_c5(hc)
        if c < best_c5 - 1e-14:
            best_c5 = c
        else:
            hc[i] -= d
            hc[j] += d
    return hc, best_c5

    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.ones(70) * 0.5
    n = len(h_best)
    target = n / 2.0
    _DE_TGT = target
    h_best = make_feasible(h_best, n, target)
    best_c5 = compute_c5(h_best)

    # Phase 1: Differential Evolution for global exploration
    if time.time() < deadline - 250:
        de_res = differential_evolution(
            de_objective, [(0.0, 1.0)] * n,

            tol=1e-9, mutation=(0.5, 1.0), recombination=0.7,
            polish=False, init='latinhypercube', updating='deferred')
        h_de = make_feasible(de_res.x, n, target)
        c5_de = compute_c5(h_de)
        if c5_de < best_c5:
            h_best, best_c5 = h_de, c5_de

    # Phase 2: SLSQP alpha sweep
    for alpha in [60, 120, 250, 500, 1000, 2000]:
        if time.time() > deadline - 120:
            break
        h_new = slsqp(h_best, n, target, alpha, 400)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            h_best, best_c5 = h_new, c5

    # Phase 3: Multi-start with random binary inits
    if time.time() < deadline - 80:

            h_bin = np.zeros(n)

    # Phase 4: Simulated Annealing

        if c5_sa < best_c5:

    # Phase 5: SLSQP refinement after SA
    for alpha in [400, 1200, 2400]:
