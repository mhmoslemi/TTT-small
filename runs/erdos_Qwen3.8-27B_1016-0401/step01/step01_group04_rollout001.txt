import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def greedy_descent(h, n, t_deadline, max_sweeps=500):

    h_g = h.astype(float).copy()
    c5_cur = compute_c5(h_g)
    best_c5 = c5_cur
    best_h = h_g.copy()

            break
        ones = list(np.where(h_g == 1)[0])
        zeros = list(np.where(h_g == 0)[0])

        best_i, best_j = -1, -1
        base_max = np.correlate(h_g, 1.0 - h_g, mode='full').max()

            for j in zeros:
                h_g[i], h_g[j] = h_g[j], h_g[i]
                new_max = np.correlate(h_g, 1.0 - h_g, mode='full').max()
                h_g[i], h_g[j] = h_g[j], h_g[i]
                if new_max < base_max - 1e-12:

                    best_i, best_j = i, j
        
        if best_i >= 0:

            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h_g.copy()
        else:
            break
    
    return best_h, best_c5

    best_h = h_init.astype(float).copy()

    S = n / 2

    for restart in range(n_restarts):
        if time.time() > t_end:
            break
        
        h_sa = best_h.copy()
        c5_sa = compute_c5(h_sa)
        
        # Perturb if not first restart
        if restart > 0:
            for _ in range(restart * 3):
                i, j = rng.randint(n), rng.randint(n)
                if h_sa[i] != h_sa[j]:
                    h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        
        temp = 0.002
        t_cool = 0.97
        for it in range(n_per_restart):
            if time.time() > t_end:
                break
            if it % 3000 == 0 and it > 0:
                temp *= t_cool
            
            i, j = rng.randint(n, size=2)
            if h_sa[i] == h_sa[j]:
                continue
            
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            c5_new = compute_c5(h_sa)

            if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-14)):

                if c5_new < best_c5:

            else:

        if c5_g < best_c5:

def slsqp_refine(h, n, S, T, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - S}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = corr.max()
        return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-13})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()
