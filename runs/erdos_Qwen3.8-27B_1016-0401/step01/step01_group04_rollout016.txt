import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)

    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

    h_bd = h.astype(float).copy()
    c5_cur = compute_c5(h_bd)
    best_h = h_bd.copy()
    best_c5 = c5_cur
    while time.time() < t_deadline:
        ones = np.where(h_bd == 1)[0]
        zeros = np.where(h_bd == 0)[0]
        best_delta = 0
        best_i, best_j = -1, -1
        for i in ones:

                h_bd[i], h_bd[j] = h_bd[j], h_bd[i]

                h_bd[i], h_bd[j] = h_bd[j], h_bd[i]

                    if delta < best_delta:
                        best_delta = delta
                        best_i, best_j = i, j
        if best_i == -1:

        h_bd[best_i], h_bd[best_j] = h_bd[best_j], h_bd[best_i]

import numpy as np
import time
from scipy.optimize import minimize

_CON_S = 0.0
_OBJ_T = 1.0

def _con_fun(x):
    return x.sum() - _CON_S

def _obj_fun(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    shifted = np.clip(_OBJ_T * (corr - mx), -500, 500)
    return float(mx + np.log(np.sum(np.exp(shifted))) / _OBJ_T)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def slsqp_refine(h, n, S, T, maxiter=500):
    global _CON_S, _OBJ_T
    _CON_S = S
    _OBJ_T = T
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _con_fun}]
    try:
        res = minimize(_obj_fun, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-15})
        return normalize_h(res.x, S)
    except Exception:
        return h.copy()

def binary_steepest(h, n, t_deadline, max_rounds=500):
    h_b = h.astype(float).copy()
    c5_cur = compute_c5(h_b)
    best_h = h_b.copy()
    best_c5 = c5_cur
    for rnd in range(max_rounds):
        if time.time() > t_deadline:
            break
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h_b[i], h_b[j] = 0.0, 1.0
                c5_new = compute_c5(h_b)
                h_b[i], h_b[j] = 1.0, 0.0
                d = c5_new - c5_cur
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h_b[best_i], h_b[best_j] = 0.0, 1.0
        c5_cur += best_delta
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_b.copy()
    return best_h, best_c5

def binary_sa(h, n, rng, t_deadline, best_c5):
    h_sa = h.astype(float).copy()
    c5_sa = compute_c5(h_sa)
    best_h = h_sa.copy()
    best_c5_local = min(c5_sa, best_c5)
    temp = 0.12
    it = 0
    while time.time() < t_deadline:
        if it % 8000 == 0 and it > 0:
            temp *= 0.96
        i = rng.randint(n)
        j = rng.randint(n)
        if h_sa[i] == h_sa[j]:
            it += 1
            continue
        h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        c5_new = compute_c5(h_sa)
        delta = c5_new - c5_sa
        if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-12)):
            c5_sa = c5_new
            if c5_new < best_c5_local:
                best_c5_local = c5_new
                best_h = h_sa.copy()
        else:
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        it += 1
    return best_h, best_c5_local

def make_spread_binary(n, S, rng):
    """Create a spread-out binary sequence as initial guess."""
    target = int(S)
    h = np.zeros(n)
    step = n / float(target)
    offsets = rng.uniform(0, 1, target)
    positions = np.sort(((np.arange(target) + offsets) * step).astype(int))
    positions = np.clip(positions, 0, n - 1)
    seen = set()
    count = 0
    for p in positions:
        if p not in seen:
            h[p] = 1.0
            seen.add(p)
            count += 1
    if count < target:
        zeros = np.where(h == 0)[0]
        for i in range(target - count):
            h[zeros[rng.randint(len(zeros))]] = 1.0
    return h
