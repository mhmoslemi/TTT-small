import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(100):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        h = h + d / n
        h = np.clip(h, 0, 1)
    return h

def smooth_obj(h, n, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    return dx * (c_max + temp * np.log(np.sum(np.exp((c - c_max) / temp)) + 1e-300))

def smooth_grad(h, n, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    e = np.exp((c - c_max) / temp)
    se = e.sum()
    w = e / se
    ks = np.arange(n)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        sg = m - (n - 1)
        v1 = (ks - sg >= 0) & (ks - sg < n)
        v2 = (ks + sg >= 0) & (ks + sg < n)
        g = np.zeros(n)
        g[v1] += (1.0 - h[ks[v1] - sg])
        g[v2] += (-h[ks[v2] + sg])
        grad += w[m] * g
    return dx * grad

def optimize_anneal_sl(h_init, n, temps, max_iter_each=300, ftol=1e-12):
    dx = 2.0 / n
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    x = project_feasible(h_init.copy(), n)
    best_x = x.copy()
    best_c5 = compute_c5(x, dx)
    for temp in temps:
        def fun(z):
            return smooth_obj(z, n, dx, temp)
        def jac(z):
            return smooth_grad(z, n, dx, temp)
        def cf(z):
            return z.sum() - target
        def cj(z):
            return np.ones(n)
        try:
            res = minimize(fun, x, jac=jac, method='SLSQP',
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                         options={'maxiter': max_iter_each, 'ftol': ftol})
            xn = project_feasible(res.x, n)
            c5 = compute_c5(xn, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_x = xn.copy()
            x = xn
        except Exception:
            pass
    return best_x, best_c5

def optimize_anneal_lbfgs(h_init, n, temps, max_iter_each=300, pen=200.0):
    dx = 2.0 / n
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n

    best_x = x.copy()
    best_c5 = compute_c5(x, dx)
    for temp in temps:
        def fun(z):
            base = smooth_obj(z, n, dx, temp)
            s = z.sum() - target
            return base + pen * s * s
        def jac(z):
            bg = smooth_grad(z, n, dx, temp)
            s = z.sum() - target
            return bg + 2.0 * pen * s * np.ones(n)
        try:
            res = minimize(fun, x, jac=jac, method='L-BFGS-B',

            xn = project_feasible(res.x, n)
            c5 = compute_c5(xn, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_x = xn.copy()
            x = xn
        except Exception:
            pass
    return best_x, best_c5

def local_refine(h, n, eps=0.01, max_rounds=3):
    dx = 2.0 / n
    target = float(n) / 2.0
    best = project_feasible(h.copy(), n)
    best_c5 = compute_c5(best, dx)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in (eps, -eps):
                    h2 = best.copy()
                    h2[i] = best[i] + d
                    h2[j] = best[j] - d
                    if h2[i] < -1e-12 or h2[i] > 1.0 + 1e-12:
                        continue
                    if h2[j] < -1e-12 or h2[j] > 1.0 + 1e-12:
                        continue
                    c5 = compute_c5(h2, dx)
                    if c5 < best_c5 - 1e-12:
                        best = h2
                        best_c5 = c5
                        improved = True
        if not improved:
            break
    return best, best_c5
