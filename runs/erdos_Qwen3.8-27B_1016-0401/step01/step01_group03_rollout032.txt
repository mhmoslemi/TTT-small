import numpy as np

import time

def compute_c5(h, dx):

    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def smooth_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

        def gw(x):
            xp = project_feasible(x, ts)
            _, g = smooth_c5_grad(xp, dx, beta)
            return g
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(ow, h, jac=gw, method='L-BFGS-B',
                          bounds=bnds, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    bd = coord_max_descent(h, dx, ts, t_limit)
    if bd[1] < best_c5:
        best_h, best_c5 = bd
    return best_h, best_c5

def to_binary(h0, ts):
    h = (h0 > 0.5).astype(float).copy()
    oc = int(np.sum(h))
    if oc > ts:
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            h[i] = 0.0
    elif oc < ts:
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            h[i] = 1.0
    return h

def binary_local_search(h0, dx, t_limit, use_2swap=True, n_2swap_iters=200):
    n = len(h0)
    ts = n // 2
    best = to_binary(h0, ts)
    best_c5 = compute_c5(best, dx)
    improved = True
    iters = 0
    while improved and time.time() < t_limit and iters < 5000:
        improved = False
        iters += 1
        zeros = np.where(best == 0)[0]
        ones = np.where(best == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        for zi in zeros:
            if time.time() > t_limit:
                return best, best_c5
            best[zi] = 1.0
            for oi in ones:
                if oi == zi:
                    continue
                best[oi] = 0.0
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                best[oi] = 1.0
            best[zi] = 0.0
            if improved:
                break
    if use_2swap and time.time() < t_limit:
        for _ in range(n_2swap_iters):
            if time.time() > t_limit:
                break
            improved = False
            zeros = np.where(best == 0)[0]
            ones = np.where(best == 1)[0]
            if len(zeros) < 2 or len(ones) < 2:
                break
            for _ in range(500):
                z2 = np.random.choice(zeros, 2, replace=False)
                o2 = np.random.choice(ones, 2, replace=False)
                best[z2[0]] = 1.0; best[z2[1]] = 1.0
                best[o2[0]] = 0.0; best[o2[1]] = 0.0
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                best[z2[0]] = 0.0; best[z2[1]] = 0.0
                best[o2[0]] = 1.0; best[o2[1]] = 1.0
            if not improved:
                break
    return best, best_c5

def simulated_annealing(h0, dx, t_limit):
    n = len(h0)
    ts = n // 2
    best = to_binary(h0, ts)
    current_c5 = compute_c5(best, dx)
    best_c5 = current_c5
    best_copy = best.copy()
    n_iter = 400000
    t0 = time.time()
    total_time = t_limit - t0
    T = 0.05 * dx
    T_min = 0.001 * dx
    decay = (T_min / T) ** (1.0 / n_iter)
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        if it % 5000 == 0 and it > 0:
            T *= decay
        zeros = np.where(best == 0)[0]
        ones = np.where(best == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        if np.random.rand() < 0.6:
            zi = np.random.choice(zeros)
            oi = np.random.choice(ones)
            best[zi] = 1.0; best[oi] = 0.0
            c5_new = compute_c5(best, dx)
            delta = c5_new - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
                current_c5 = c5_new
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_copy = best.copy()
            else:
                best[zi] = 0.0; best[oi] = 1.0
        else:
            if len(zeros) >= 2 and len(ones) >= 2:
                z2 = np.random.choice(zeros, 2, replace=False)
                o2 = np.random.choice(ones, 2, replace=False)
                best[z2[0]] = 1.0; best[z2[1]] = 1.0
                best[o2[0]] = 0.0; best[o2[1]] = 0.0
                c5_new = compute_c5(best, dx)
                delta = c5_new - current_c5
                if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
                    current_c5 = c5_new
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_copy = best.copy()
                else:
                    best[z2[0]] = 0.0; best[z2[1]] = 0.0
                    best[o2[0]] = 1.0; best[o2[1]] = 1.0
    return best_copy, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0
    best_h = project_feasible(h_init, ts_init)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init
    # Phase 1: smooth + coord descent from initial
    t1 = start_time + budget_s * 0.20
    h1, c5_1 = optimize_smooth(best_h, dx_init, ts_init, t1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    # Phase 2: binary local search
    t2 = start_time + budget_s * 0.38
    h_bin, c5_bin = binary_local_search(h1, dx_init, t2, use_2swap=True, n_2swap_iters=100)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin
    # Phase 3: simulated annealing
    t3 = start_time + budget_s * 0.60
    h_sa, c5_sa = simulated_annealing(best_h, dx_init, t3)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa
    # Phase 4: more local search from SA result
    t4 = start_time + budget_s * 0.72
    h_bl2, c5_bl2 = binary_local_search(best_h, dx_init, t4, use_2swap=True, n_2swap_iters=200)
    if c5_bl2 < best_c5:
        best_c5 = c5_bl2
        best_h = h_bl2
    # Phase 5: multi-start with different n
    t5 = start_time + budget_s * 0.95
    for n_try in [n_init, 48, 52]:
        if time.time() > t5:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        n_restarts = 4 if n_try == n_init else 3
        for r in range(n_restarts):
            if time.time() > t5:
                break
            if r == 0:
                x_old = np.linspace(0.0, 2.0, best_n)
                x_new = np.linspace(0.0, 2.0, n_try)
                h_try = np.interp(x_new, x_old, best_h)
                h_try = project_feasible(h_try, ts_t)
            else:
                h_try = np.random.uniform(0.0, 1.0, n_try)
                h_try = project_feasible(h_try, ts_t)
            t_sub = min(t5, time.time() + max(8, (t5 - time.time()) * 0.12))
            h_opt, c5_opt = optimize_smooth(h_try, dx_t, ts_t, t_sub)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
        if n_try == best_n and time.time() < t5:
            t_b = min(t5, time.time() + max(10, (t5 - time.time()) * 0.15))
            h_b, c5_b = binary_local_search(best_h, 2.0 / best_n, t_b, use_2swap=True)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
    # Final polish with coord descent
    if time.time() < start_time + budget_s * 0.99:
        dx_f = 2.0 / best_n
        h_pol, c5_pol = coord_max_descent(best_h, dx_f, best_n / 2.0, start_time + budget_s * 0.99)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
