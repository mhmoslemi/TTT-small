Let me also think about whether the coordinate descent is correct. The issue is: when I change h_j, the sum constraint is violated. I handle this by calling `enforce` after each pass. But during the pass, the sum is changing. Is this a problem?

For the objective C₅ = max_k Σ h_i(1-h_{i+k}) * dx, the sum constraint doesn't directly appear in the objective. It's a constraint on the feasible set. If I ignore it during CD and project at the end, the CD is optimizing an "unconstrained" version of the problem. The projection might destroy the optimality.

A better approach: after each coordinate update, immediately adjust the sum by distributing the error equally among all other coordinates. This keeps the constraint nearly satisfied at all times.

```python
def cd_pass_constrained(h, n, dx, order):
    target = n / 2.0
    for j in order:
        h[j] = 0.0
        c0 = np.correlate(h, 1-h, mode='full')
        h[j] = 1.0
        c1 = np.correlate(h, 1-h, mode='full')
        d = c1 - c0
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi-lo)/3.0
            m2 = hi - (hi-lo)/3.0
            if np.max(c0+m1*d) < np.max(c0+m2*d):
                hi = m2
            else:
                lo = m1
        new_val = (lo+hi)/2.0
        old_val = h[j]  # this is 1.0 from the c1 computation
        # Actually h[j] was set to 1.0 for c1, so we need to restore
        h[j] = new_val
        # Adjust others to maintain sum
        delta = target - h.sum()
        if j != 0:  # avoid adjusting j itself
            others = np.delete(np.arange(n), j)
            h[others] += delta / (n-1)
            h[others] = np.clip(h[others], 0, 1)
        else:
            # Adjust a different coordinate
            others = np.delete(np.arange(n), j)
            h[others] += delta / (n-1)
            h[others] = np.clip(h[others], 0, 1)
    return h
```

```python
_G_T = 35.0

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')

def enforce(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        if s < 1e-14:
            return np.full(n, 0.5)

            break
    return h

def cd_pass(h, n, dx, order, n_ternary=40):
    for j in order:
        h[j] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[j] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        d = c1 - c0
        lo, hi = 0.0, 1.0
        for _ in range(n_ternary):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0

                hi = m2
            else:
                lo = m1
        h[j] = (lo + hi) / 2.0
    return h

    return float(x.sum() - _G_T)

    return np.ones(_G_N)

    cm = c.max()
    return _G_DX * (cm + _G_TEMP * np.log(np.sum(np.exp((c - cm) / _G_TEMP))))

    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    w = np.exp((c - cm) / _G_TEMP)
    w /= w.sum()
    g = np.zeros(n)

        if s == 0:

            g[s:] += wm * (-x[:n - s])
        else:
            ss = -s
            g[ss:] += wm * (1.0 - x[:n - ss])
            g[:n - ss] += wm * (-x[ss:])
    return _G_DX * g

def slsqp_polish(h, n, temps, max_iter=400):

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    best = h.copy()
    best_c5 = compute_c5(h, _G_DX)
    for t in temps:
        _G_TEMP = t
        try:
            r = minimize(_sobj, best, method='SLSQP', jac=_sgrad,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-15})
            hx = enforce(np.array(r.x), n)
            c5 = compute_c5(hx, _G_DX)
            if c5 < best_c5:
                best_c5, best = c5, hx
        except Exception:
            pass
    return best, best_c5
```
