    dx = 2.0 / len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(64):
        lam = 0.5 * (lo + hi)
        s = np.clip(a - lam, 0.0, 1.0).sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
