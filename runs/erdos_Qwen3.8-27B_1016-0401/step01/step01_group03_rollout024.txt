    def solve_lp(n, dx, initial=None):
        """Solve the LP relaxation for shifts 1..n-1 (both directions)."""
        h = cp.Variable(n)
        # z[i,j] for i < j, representing h[i]*h[j]
        z = {}
        for i in range(n):
            for j in range(i+1, n):
                z[(i,j)] = cp.Variable()
        
        t = cp.Variable()
        
        constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
        
        # McCormick constraints
        for i in range(n):
            for j in range(i+1, n):
                z[(i,j)] <= h[i]
                z[(i,j)] <= h[j]
                z[(i,j)] >= h[i] + h[j] - 1
                z[(i,j)] >= 0
        
        # Shift constraints (m = 1 to n-1)
        for m in range(1, n):
            # Positive shift: sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            # = sum_{i=m}^{n-1} (h[i] - h[i]*h[i-m])
            pos_sum = cp.sum(h[m:]) - cp.sum([z[(i-m, i)] for i in range(m, n)])
            # Negative shift: sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])

        obj = cp.Minimize(t)

        # Start from initial if available
        if initial is not None:

        prob.solve(solver=cp.ECOS, verbose=False)
        return h.value, t.value

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def solve_lp_erdos(n, dx, init=None):
    h = cp.Variable(n)
    nz = n * (n - 1) // 2
    z = cp.Variable(nz)
    t = cp.Variable()
    
    # Build pair index mapping
    pair_to_idx = {}
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            pair_to_idx[(i, j)] = idx
            idx += 1
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    # McCormick
    for (i, j), idx_z in pair_to_idx.items():
        constraints += [z[idx_z] <= h[i], z[idx_z] <= h[j],
                       z[idx_z] >= h[i] + h[j] - 1, z[idx_z] >= 0]
    
    # Shift constraints
    for m in range(1, n):
        z_indices = [pair_to_idx[(k, k + m)] for k in range(n - m)]
        z_sum = cp.sum(z[z_indices])
        constraints.append((cp.sum(h[m:]) - z_sum) * dx <= t)
        constraints.append((cp.sum(h[:n - m]) - z_sum) * dx <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

    prob.solve(solver=cp.ECOS, verbose=False)

    result[order[:target_ones]] = 1.0

def smooth_obj_grad(h, dx, beta):
    n = len(h)

    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(np.clip(beta * shifted, -50, 50))

    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp

        if wi < 1e-15:

            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)

        else:
            mask = proj > 1e-9
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def solve_lp_erdos(n, dx, init=None):
    h = cp.Variable(n)
    nz = n * (n - 1) // 2
    z = cp.Variable(nz)
    t = cp.Variable()
    pair_to_idx = {}
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            pair_to_idx[(i, j)] = idx
            idx += 1
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    for key, iz in pair_to_idx.items():
        i, j = key
        constraints += [z[iz] <= h[i], z[iz] <= h[j], z[iz] >= h[i] + h[j] - 1, z[iz] >= 0]
    for m in range(1, n):
        z_inds = [pair_to_idx[(k, k + m)] for k in range(n - m)]
        z_sum = cp.sum(z[z_inds])
        constraints.append((cp.sum(h[m:]) - z_sum) * dx <= t)
        constraints.append((cp.sum(h[:n - m]) - z_sum) * dx <= t)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    if init is not None:
        h.value = init
    for solver in [cp.ECOS, cp.SCS, cp.CVCANOPY]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(h.value).flatten().copy(), float(t.value)
        except Exception:
            continue
    return None, None


def project_binary(h, target_ones):
    n = len(h)
    order = np.argsort(-h)
    result = np.zeros(n)
    result[order[:target_ones]] = 1.0
    return result


def pairwise_local_search(h, dx, t_limit, max_rounds=2000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    for rnd in range(max_rounds):
        if time.time() > t_limit:
            break
        ones = np.where(best >= 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for oi in ones:
            if improved:
                break
            for zi in zeros:
                best[oi] = 0.0
                best[zi] = 1.0
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    improved = True
                    break
                best[oi] = 1.0
                best[zi] = 0.0
        if not improved:
            break
    return best, best_c5


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = proj < 1.0 - 1e-9
        else:
            mask = proj > 1e-9
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(np.clip(beta * shifted, -50, 50))
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad


def gradient_polish(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [200.0, 800.0, 3000.0, 10000.0]:
        lr = 1.0
        for it in range(3000):
            if time.time() > t_limit:
                return best_h, best_c5
            obj, grad = smooth_obj_grad(h, dx, beta)
            h_new = project_feasible(h - lr * grad, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                lr = min(lr * 1.03, 5.0)
            else:
                lr *= 0.93
            h = h_new
            if lr < 1e-7:
                break
    return best_h, best_c5


def try_resolution(n, dx, h_ref, best_c5, best_h, best_n, t_limit):
    np.random.seed(42 + n)
    x_ref = np.linspace(0, 2, len(h_ref))
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_ref, h_ref)
    h_lp, t_lp = solve_lp_erdos(n, dx, init=h_interp)
    if h_lp is not None:
        h_lp = project_feasible(np.clip(h_lp, 0, 1), n / 2.0)
        c5_val = compute_c5(h_lp, dx)
        if c5_val < best_c5:
            best_c5, best_h, best_n = c5_val, h_lp.copy(), n
        h_bin = project_binary(h_lp, n // 2)
        c5_bin = compute_c5(h_bin, dx)
        if c5_bin < best_c5:
            best_c5, best_h, best_n = c5_bin, h_bin.copy(), n
    if time.time() < t_limit:
        h_start = best_h if best_n == n else (h_bin if h_bin is not None else h_interp)
        h_b = (h_start > 0.5).astype(float)
        oc = int(np.sum(h_b))
        tgt = n // 2
        if oc > tgt:
            idxs = np.where(h_b == 1)[0]
            np.random.shuffle(idxs)
            for i in idxs[:oc - tgt]:
                h_b[i] = 0.0
        elif oc < tgt:
            idxs = np.where(h_b == 0)[0]
            np.random.shuffle(idxs)
            for i in idxs[:tgt - oc]:
                h_b[i] = 1.0
        t_sub = min(t_limit, time.time() + 60)
        h_ls, c5_ls = pairwise_local_search(h_b, dx, t_sub)
        if c5_ls < best_c5:
            best_c5, best_h, best_n = c5_ls, h_ls.copy(), n
    return best_h, best_c5, best_n
