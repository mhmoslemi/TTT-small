    """Project h onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -max(h) - 1, max(h) + 1  # or use 0 and 1
    # Actually, the range of valid lambda is [min(h)-1, max(h)] or similar
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Final fix for exact sum
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # Distribute remaining
        if diff > 0:
            idx = np.argsort(np.clip(h, 0.01, 0.99))[::-1]  # prefer elements not at boundary
        else:
            idx = np.argsort(np.clip(h, 0.01, 0.99))
        for i in idx:

                x[i] += add
                diff -= add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
        diff2 = target - x.sum()
        if abs(diff2) > 1e-12:
            for i in range(len(x)):
                if diff2 > 0:
                    a = min(diff2, 1.0 - x[i])
                else:
                    a = max(diff2, x[i] - 1.0)
                if abs(a) < 1e-15:
                    continue
                x[i] += a
                diff2 -= a
                if abs(diff2) < 1e-12:
                    break
    return x

def softmax_obj_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = float(np.max(corr))
    sh = T * (corr - mc)
    sh = np.clip(sh, -400, 400)
    e = np.exp(sh)
    se = float(e.sum())
    w = e / se
    obj = (mc + np.log(se) / T) * (2.0 / n)
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for m_idx in range(2 * n - 1):
        wi = w[m_idx]
        if wi < 1e-10:
            continue
        m = m_idx - center
        if m >= 0:
            grad[:n - m] += wi * (1.0 - h[m:])
            grad[m:] -= wi * h[:n - m]
        else:
            p = -m
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] -= wi * h[p:]
    return obj, grad * dx

def nesterov_adam(h0, n, target, elapsed_fn, budget_fn, T_init, T_min, lr_init, max_iters):
    h = h0.astype(float).copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    v_nest = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c5_best = compute_c5(h)
    h_best = h.copy()
    stall = 0
    while t < max_iters and elapsed_fn() < budget_fn():
        frac = min(1.0, t / max(t * 2, max_iters))
        T = max(T_min, T_init * (1.0 - frac)**1.5)
        lr = lr_init / (1.0 + 0.002 * t)
        # Nesterov lookahead
        h_look = h - beta1 * v_nest
        h_look = project_feasible(h_look, target)
        _, grad = softmax_obj_grad(h_look, n, T)
        t += 1
        m_arr = beta1 * m_arr + (1 - beta1) * grad
        v_arr = beta2 * v_arr + (1 - beta2) * grad**2
        m_hat = m_arr / (1 - beta1**t)
        v_hat = v_arr / (1 - beta2**t)
        v_nest = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - beta1 * v_nest
        h_new = project_feasible(h_new, target)
        h = h_new
        c5 = compute_c5(h)
        if c5 < c5_best - 1e-7:
            c5_best = c5
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                h = h_best + 0.05 * np.random.default_rng(t).standard_normal(n)
                h = project_feasible(h, target)
                stall = 0
    return h_best, c5_best

def local_polish(h, n, target, time_budget):
    t_start = time.time()
    h = h.copy()
    c5_cur = compute_c5(h)
    rng = np.random.default_rng(99)
    while time.time() - t_start < time_budget:
        improved = False
        for _ in range(30):
            if time.time() - t_start > time_budget:
                break
            i = int(rng.integers(0, n))
            delta = (1 if rng.random() < 0.5 else -1) * rng.uniform(0.005, 0.08)
            h_new = h.copy()
            h_new[i] = float(np.clip(h_new[i] + delta, 0, 1))
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-9:
                h = h_new
                c5_cur = c5_new
                improved = True
        if not improved:
            for _ in range(10):
                if time.time() - t_start > time_budget:
                    break
                i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
                if i == j:
                    continue
                delta = rng.uniform(0.01, 0.04) * (1 if rng.random() < 0.5 else -1)
                h_new = h.copy()
                h_new[i] = float(np.clip(h_new[i] - delta, 0, 1))
                h_new[j] = float(np.clip(h_new[j] + delta, 0, 1))
                h_new = project_feasible(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-9:
                    h = h_new
                    c5_cur = c5_new
                    break
            else:
                break
    return h, c5_cur

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:

        if r == 0:

        elif r == 2:

            if n_ones != n // 2:

    return (h_best, c5_final, n)
