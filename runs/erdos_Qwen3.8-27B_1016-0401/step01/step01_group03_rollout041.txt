import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def soft_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        m = i - (n - 1)
        if m >= 0:
            grad[:n - m] += wi * (1.0 - h[m:])
            grad[m:] -= wi * h[:n - m]
        else:
            p = -m
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] -= wi * h[p:]
    return obj, grad

def _eq_fun(x, ts):
    return np.sum(x) - ts

def _eq_jac(x):
    return np.ones(len(x))

def smooth_optimize(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    for beta in [50.0, 200.0, 1000.0, 5000.0, 20000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (ts,)}]
        bnds = [(0.0, 1.0)] * n
        try:
            def _obj(x):
                o, _ = soft_obj_grad(x, dx, beta)
                return o + 1e6 * (np.sum(x) - ts) ** 2
            def _grad(x):
                _, g = soft_obj_grad(x, dx, beta)
                g = g + 2e6 * (np.sum(x) - ts) * np.ones(n)
                return g
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                           bounds=bnds, options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
        # Also try SLSQP for the last few betas
        if beta >= 1000.0 and time.time() < t_limit:
            try:
                res = minimize(soft_obj_grad, h, args=(dx, beta), jac=True,
                             method='SLSQP', bounds=bnds, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass
    # Projected gradient polish
    beta = 20000.0
    lr = 0.01
    for it in range(20000):
        if time.time() > t_limit:
            break
        _, grad = soft_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.9
        h = h_new
        if lr < 1e-10:
            break
    return best_h, best_c5

def fix_binary(h, target_ones):
    n = len(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    nb = np.zeros(n)
    if len(ones) >= target_ones:
        nb[ones[:target_ones]] = 1.0
    else:
        nb[ones] = 1.0
        nb[zeros[:target_ones - len(ones)]] = 1.0
    return nb

def binary_sa(h0, dx, t_limit):
    n = len(h0)
    ts = n // 2
    best = fix_binary(h0, ts).copy()
    cur = best.copy()
    best_c5 = compute_c5(best, dx)
    cur_c5 = best_c5
    T = 0.01
    T_min = 1e-8
    cooling = 0.9997
    np.random.seed(int(time.time() * 1000) % 100000)
    it = 0
    while time.time() < t_limit:
        it += 1
        ones_mask = cur > 0.5
        zeros_mask = cur <= 0.5
        n_ones = np.sum(ones_mask)
        if n_ones == 0 or n_ones == n:
            break
        oi = np.random.choice(np.where(ones_mask)[0])
        zi = np.random.choice(np.where(zeros_mask)[0])
        if oi == zi:
            continue
        cur[oi] = 0.0
        cur[zi] = 1.0
        new_c5 = compute_c5(cur, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, T_min)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best = cur.copy()
        else:
            cur[oi] = 1.0
            cur[zi] = 0.0
        T *= cooling
        if it % 5000 == 0 and T < T_min:
            break
    return best, best_c5

def binary_greedy(h0, dx, t_limit):
    n = len(h0)
    ts = n // 2
    best = fix_binary(h0, ts).copy()
    best_c5 = compute_c5(best, dx)
    for sweep in range(100):
        if time.time() > t_limit:
            break
        improved = False
        zeros = np.where(best == 0)[0]
        ones = np.where(best == 1)[0]
        for zi in zeros:
            if time.time() > t_limit:
                return best, best_c5
            for oi in ones:
                if zi == oi:
                    continue
                best[zi] = 1.0
                best[oi] = 0.0
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                best[zi] = 0.0
                best[oi] = 1.0
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=0)
