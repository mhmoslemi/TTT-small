from scipy.optimize import minimize
import time

_GN = 70
_GT = 35.0
_GDX = 2.0 / 70
_GTS = 0.05

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _sum_fun(x):
    return float(np.sum(x) - _GT)

def _sum_jac(x):
    return np.ones(_GN)

def smooth_obj(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    return _GDX * (cm + _GTS * np.log(np.sum(np.exp((c - cm) / _GTS))))

def smooth_grad(x):
    n = _GN
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    w = np.exp((c - cm) / _GTS)
    w /= w.sum()
    ks = np.arange(n)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wt = w[m]
        if wt < 1e-15:
            continue
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += wt * (1.0 - x[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += wt * (-x[ks[v2] + s])
    return _GDX * grad

def sharp_obj(x):
    return compute_c5(x, _GDX)

def sharp_grad(x):

    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-x[ks[v2] + s])
    return _GDX * grad

        if mask.sum() > 0:
            h[mask] += d / mask.sum()
    return np.clip(h, 0.0, 1.0)

def _set_globals(n):
    global _GN, _GT, _GDX
    _GN = n
    _GT = n / 2.0
    _GDX = 2.0 / n

def _set_smooth(t):
    global _GTS
    _GTS = t

def _do_smooth_opt(h, n, t_start, t_end, n_stages, max_iter):
    global _GTS
    _set_globals(n)
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_local = h.copy()
    best_c5 = compute_c5(h, dx)
    ts = np.geomspace(t_start, t_end, n_stages)
    for t in ts:
        _set_smooth(t)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _set_globals.__globals__['_sum_jac']}]
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
        try:
            r = minimize(smooth_obj, h, method='SLSQP', jac=smooth_grad,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-14})
            h_new = project_feasible(r.x, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_local = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_local, best_c5

def _do_sharp_opt(h, n, max_iter):

    dx = 2.0 / n
    h = project_feasible(h, n)
    best_local = h.copy()
