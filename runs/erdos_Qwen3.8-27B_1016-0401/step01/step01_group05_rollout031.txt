def c5_gradient(h, alpha):
    """Analytical subgradient of soft-max C5."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h

    exp_v = np.exp(np.clip(alpha * (corr - m), -500, 500))
    w = exp_v / np.sum(exp_v)

        if wk < 1e-14:

        else:

            grad[Lp:] += wk * (1.0 - h[:n-Lp])
            grad[:n-Lp] -= wk * h[Lp:]

        if not np.any(mask):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Phase 1: Continuation PGD from parent
    for alpha in [20, 50, 100, 250, 600, 1500, 4000]:
        if time.time() > deadline - 90:
            break
        h_new, c5_new = opt_pgd_adapt(best_h, alpha, 2.0, 0.35, 12000, deadline - 90)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Alternative initializations at parent resolution
    alts = [
        block_init(n0, 7, 3, rng),
        block_init(n0, 10, 4, rng),
        block_init(n0, 5, 2, rng),
        block_init(n0, 14, 5, rng),
    ]
    x = np.linspace(0, 2, n0)
    for freq in [3, 5, 7]:
        h_harm = 0.5 + 0.4 * np.sin(np.pi * freq * x)
        h_harm += rng.uniform(-0.02, 0.02, n0)
        alts.append(project(h_harm, n0))

    for h_alt in alts:
        if time.time() > deadline - 120:
            break
        h_new, c5_new = opt_pgd_adapt(h_alt, 100, 2.0, 0.3, 8000, deadline - 120)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Higher resolution refinement
    for n_try in [100, 140, 200]:
        if time.time() > deadline - 100:
            break
        h_interp = interp_h(best_h, len(best_h), n_try)
        for alpha in [40, 150, 600]:
            if time.time() > deadline - 100:
                break
            h_new, c5_new = opt_pgd_adapt(h_interp, alpha, 1.5, 0.3, 6000, deadline - 100)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Final PGD polish at current best resolution
    if time.time() < deadline - 30:
        for alpha in [500, 1500, 5000]:
            if time.time() > deadline - 30:
                break
            h_new, c5_new = opt_pgd_adapt(best_h, alpha, 1.0, 0.25, 8000, deadline - 30)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Coordinate descent polish
    if time.time() < deadline - 8:
        h_cd, c5_cd = coord_desc(best_h, rng, 80000, deadline - 4, step0=0.015)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Final validation
    best_h = project(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, len(best_h)
