from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def softmax_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    shifted = T * (corr - mx)
    # Use log-sum-exp trick
    max_shifted = shifted.max()
    return mx + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / T

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h / s * S
    h = np.clip(h, 0.0, 1.0)
    # Fix sum exactly
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        h[int(diff / h.max() * len(h))] += 0  # just clip and hope
        # Better: distribute
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, S):
    S = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(-h[one_idx])  # remove least "confident" ones
        for i in range(ones - S):
            h_b[one_idx[order[i]]] = 0
    elif ones < S:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])  # add most "confident" zeros
        for i in range(S - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def steepest_descent_binary(h, t_deadline, max_rounds=50):
    """Try all swaps, pick the best improving one each round."""
    n = len(h)
    S = int(h.sum())
    h_d = h.copy()
    c5 = compute_c5(h_d)
    best_c5 = c5
    best_h = h_d.copy()
    
    one_idx = np.where(h_d == 1)[0]
    zero_idx = np.where(h_d == 0)[0]
    
    for rnd in range(max_rounds):
        if time.time() > t_deadline:
            break
        best_delta = 0
        best_i, best_j = -1, -1
        for i in one_idx:
            for j in zero_idx:
                h_d[i], h_d[j] = h_d[j], h_d[i]
                c5_new = compute_c5(h_d)
                delta = c5 - c5_new
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h_d[i], h_d[j] = h_d[j], h_d[i]
        
        if best_i < 0:
            break
        h_d[best_i], h_d[best_j] = h_d[best_j], h_d[best_i]
        c5 = c5 - best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_d.copy()
            one_idx = np.where(h_d == 1)[0]
            zero_idx = np.where(h_d == 0)[0]
    
    return best_h, best_c5

def sa_binary(h, rng, t_deadline, start_temp=0.005, cool_rate=0.999):
    """Simulated annealing for binary optimization."""
    n = len(h)
    S = int(h.sum())
    h_sa = h.copy()
    c5 = compute_c5(h_sa)
    best_c5 = c5
    best_h = h_sa.copy()
    temp = start_temp
    one_idx = np.where(h_sa == 1)[0].tolist()
    zero_idx = np.where(h_sa == 0)[0].tolist()
    
    while time.time() < t_deadline:
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        if i == j:
            continue
        h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        c5_new = compute_c5(h_sa)
        delta = c5_new - c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(temp, 1e-12)):
            c5 = c5_new
            # Update one_idx and zero_idx

                zero_idx = np.where(h_sa == 0)[0].tolist()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sa.copy()
        else:
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def softmax_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = T * (corr - mx)
    ms = s.max()
    return float(mx + (ms + np.log(np.sum(np.exp(s - ms)))) / T)

def normalize_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h *= S / s
    h = np.clip(h, 0.0, 1.0)
    h += (S - h.sum()) / len(h)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, S):
    S = int(round(S))
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > S:
        idx = np.where(h_b == 1)[0]
        ord_ = np.argsort(-h[idx])
        h_b[idx[ord_[:ones - S]]] = 0
    elif ones < S:
        idx = np.where(h_b == 0)[0]
        ord_ = np.argsort(-h[idx])
        h_b[idx[ord_[:S - ones]]] = 1
    return h_b

def steepest_desc(h, t_lim, rounds=200):
    h_d = h.copy()
    c5 = compute_c5(h_d)
    b5, bh = c5, h_d.copy()
    for _ in range(rounds):
        if time.time() > t_lim:
            break
        bd, bp = 1e-15, None
        oi = np.where(h_d == 1)[0]
        zi = np.where(h_d == 0)[0]
        for i in oi:
            for j in zi:
                h_d[i], h_d[j] = 0.0, 1.0
                cn = compute_c5(h_d)
                h_d[i], h_d[j] = 1.0, 0.0
                d = c5 - cn
                if d > bd:
                    bd, bp = d, (i, j)
        if bp is None:
            break
        i, j = bp
        h_d[i], h_d[j] = 0.0, 1.0
        c5 -= bd
        if c5 < b5:
            b5, bh = c5, h_d.copy()
    return bh, b5

def sa_binary(h, rng, t_lim):
    h_s = h.copy()
    c5 = compute_c5(h_s)
    b5, bh = c5, h_s.copy()
    temp, cnt = 0.003, 0
    while time.time() < t_lim:
        oi = np.flatwhere(h_s == 1).ravel()
        zi = np.flatwhere(h_s == 0).ravel()
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        h_s[i], h_s[j] = 0.0, 1.0
        cn = compute_c5(h_s)
        d = cn - c5
        if d <= 0 or rng.rand() < np.exp(-d / max(temp, 1e-12)):
            c5 = cn
            if c5 < b5:
                b5, bh = c5, h_s.copy()
        else:
            h_s[i], h_s[j] = 1.0, 0.0
        cnt += 1
        if cnt % 1500 == 0:
            temp *= 0.992
    return bh, b5

def slsqp_refine(h, S, T, maxiter=300):
    n = len(h)
    bnd = [(0.0, 1.0)] * n
    ones_v = np.ones(n)
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - S, 'jac': lambda x: ones_v}]
    try:
        r = minimize(
            lambda x: softmax_obj(x, T), h.copy(), method='SLSQP',
            bounds=bnd, constraints=con,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return np.clip(r.x, 0, 1)
    except Exception:
        return h.copy()

def make_pattern(n, kind, s):
    S = n // 2
    r = np.random.RandomState(s)
    if kind == 0:
        a = (np.sqrt(5.0) - 1.0) / 2.0
        h = ((np.arange(n) * a) % 1.0 < 0.5).astype(float)
    elif kind == 1:
        base = np.concatenate([np.ones(4), np.zeros(3)])
        h = np.tile(base, n // 7 + 1)[:n].copy()
    elif kind == 2:
        base = np.concatenate([np.ones(5), np.zeros(6)])
        h = np.tile(base, n // 11 + 1)[:n].copy()
    elif kind == 3:
        base = np.concatenate([np.ones(6), np.zeros(7)])
        h = np.tile(base, n // 13 + 1)[:n].copy()
    else:
        h = np.zeros(n)
        h[r.choice(n, S, replace=False)] = 1.0
    c = int(h.sum())
    if c > S:
        h[r.choice(np.where(h == 1)[0], c - S, replace=False)] = 0
    elif c < S:
        h[r.choice(np.where(h == 0)[0], S - c, replace=False)] = 1
    return h
