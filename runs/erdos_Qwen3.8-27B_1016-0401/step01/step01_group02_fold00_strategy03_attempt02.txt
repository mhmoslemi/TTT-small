    """Reduce the support around peaks by removing low-value pixels."""
    sorted_indices = np.argsort(h)[::-1]
    extreme_indices = sorted_indices[:n_val*num_remove]
    other_indices = np.delete(sorted_indices, np.arange(5))
    avg_val = np.mean(h[extreme_indices])
    h[other_indices] = np.minimum(avg_val, h[other_indices])
    return h

def project_feasible(h, n):
    """Adjust h to satisfy constraints."""
    dx = 2.0 / n
    # Clip values to [0,1]
    h_clip = np.clip(h, 0, 1)
    # Adjust integral
    ideal_int = 1.0
    current_int = (sum(h_clip) * dx
    diff_int = (ideal_int - current_int) / (2.0 * n)  # proportional adjustment
    h_adjust = h_clip + np.random.normal(0, 0.01, n)  # smooth random perturbation
    adjusted = diff_int * np.sign(0.5 - (sum(h_adjust) * dx - ideal_int))
    h_clip += adjusted
    h_clip = np.clip(h_clip, 0, 1)
    return h_clip

def grid_step(n_base, factor=2.0):
    """Expand grid uniformly."""
    new_n = int(n_base * factor) if n_base < 700 else 720
    return new_n

def step_reduction_optimization(initial_h, n, seed=42, max_iter=1000):
    """Perform optimization with continuous step-size reduction."""
    np.random.seed(seed)
    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_obj = float('inf')
    current_obj = evaluate_objective(current_h, n)
    best_obj = min(best_obj, current_obj)

    for _ in range(max_iter):
        dx_val = 2.0 / n
        ideal_int = 1.0
        current_int = (sum(current_h) * dx_val)
        perturb = (ideal_int - current_int) * 0.1

        # Stochastic gradient of objective
        corr_old = np.correlate(current_h, 1 - current_h, mode='full')
        old_energy = corr_old.max()

        # Random perturbation
        perturbed_h = current_h + np.random.normal(0, 0.2, n)
        perturbed_h = np.clip(perturbed_h, 0, 1)

        # Combine perturbed samples using soft-clipping
        clipped_h = np.clip(current_h + 0.4*(perturbed_h - current_h), 0, 1)
        clipped_h = truncate_to_peak(clipped_h, n, num_remove=4)
        clipped_h = project_feasible(clipped_h, n)

        new_energy = evaluate_objective(clipped_h, n)
        anneal_factor = 0.99

        if new_energy < old_energy * anneal_factor/2.0:
            current_obj = new_energy
            current_h = clipped_h
            if new_energy < best_obj:
                best_obj = new_energy
                best_h = clipped_h.copy()
        
    return best_h, best_obj

def simulated_annealing_optimize(initial_h, n, seed=42, max_steps=500, T=10.0, cooling=0.995):
    """Perform simulated annealing to explore solution space."""
    np.random.seed(seed)
    current = initial_h.copy()
    best = initial_h.copy()
    T_min = 0.1
    current_energy = evaluate_objective(current, n)
    best_energy = current_energy

    for _ in range(max_steps):
        T = T_min + (1 - _/max_steps) * T
        perturbed = current + np.random.normal(0, 0.3, n)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = project_feasible(perturbed, n)
        new_energy = evaluate_objective(perturbed, n)

        if new_energy < current_energy or np.random.random() < np.exp((new_energy - current_energy)/T):
            current = perturbed
            current_energy = new_energy
            if new_energy < best_energy:
                best = perturbed
                best_energy = new_energy

    return best, best_energy

def run(seed=42, budget_s=1100, **kwargs):

        'budget': budget_s
    }

    best_solution = h_initial.copy()
    best_c = float('inf')
    n_list = [n0]

    for size in n_list:
        stage_time = min(budget_s//3, 400) if size == n0 else min(budget_s//3, 300)
        np.random.seed(seed)
        
        # Perform step-wise optimization

    c5_val = evaluate_objective(best_solution, n_list[-1])
    n_points = len(best_solution)
    
    return best_solution, c5_val, n_points
