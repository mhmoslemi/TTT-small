def smooth_obj_fn(h, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    return dx * (c_max + temp * np.log(np.sum(np.exp((c - c_max) / temp))))

def smooth_grad_fn(h, dx, temp):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    w = np.exp((c - c_max) / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        if np.any(v1):
            grad[v1] += w[m] * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        if np.any(v2):
            grad[v2] += w[m] * (-h[ks[v2] + s])
    return dx * grad

def _mk_obj(dx, T, temp):
    def f(x):
        return smooth_obj_fn(x, dx, temp)
    return f

def _mk_jac(dx, temp):
    def f(x):
        return smooth_grad_fn(x, dx, temp)
    return f

def _mk_cons(n, T):
    def f(x):
        return float(np.sum(x) - T)
    def g(x):
        return np.ones(n)
    return f, g

def run(seed=42, budget_s=1100, **kwargs):
    import time
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, TypeError):
        n0 = 70
        h0 = np.full(n0, 0.5)
    best_h = project_constraint(h0.copy(), n0 / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n0)

    # Phase 1: Coordinate descent at n0
    h_cd = coord_descent(best_h, 400, 300, seed)
    c5_cd = compute_c5(h_cd, 2.0 / n0)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd

    # Phase 2: Smooth gradient + coord polish at n0
    if time.time() - t0 < budget_s - 120:
        for temp in [0.05, 0.01, 0.003]:
            if time.time() - t0 > budget_s - 100:
                break
            dx_t = 2.0 / n0
            try:
                cf, cg = _mk_cons(n0, n0 / 2.0)
                r = minimize(_mk_obj(dx_t, n0/2.0, temp), best_h.copy(), method='SLSQP',
                             jac=_mk_jac(dx_t, temp), bounds=[(0,1)]*n0,
                             constraints=[{'type':'eq','fun':cf,'jac':cg}],
                             options={'maxiter':500,'ftol':1e-14})
                h_sm = coord_descent(project_constraint(r.x, n0/2.0), 300, 300, seed+temp)
                c5_sm = compute_c5(h_sm, 2.0/n0)
                if c5_sm < best_c5:
                    best_c5, best_h = c5_sm, h_sm
            except Exception:
                pass

    # Phase 3: Try different grid sizes
    for n_alt in [50, 100, 140, 200]:
        if time.time() - t0 > budget_s - 60:
            break
        dx_a = 2.0 / n_alt
        x_fine = np.linspace(0, 2, n_alt)
        h_interp = np.interp(x_fine, np.linspace(0, 2, len(best_h)), best_h)
        h_interp = project_constraint(h_interp, n_alt / 2.0)
        h_try = coord_descent(h_interp, 250, 250, seed + n_alt)
        c5_try = compute_c5(h_try, dx_a)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        # Random restart
        if time.time() - t0 < budget_s - 40:
            h_rand = project_constraint(rng.random(n_alt), n_alt / 2.0)
            h_try2 = coord_descent(h_rand, 250, 250, seed + n_alt + 1000)
            c5_try2 = compute_c5(h_try2, dx_a)
            if c5_try2 < best_c5:
                best_c5, best_h = c5_try2, h_try2

    # Phase 4: Additional random restarts at n0
    if time.time() - t0 < budget_s - 30:
        h_rand0 = project_constraint(rng.random(n0), n0 / 2.0)
        h_try3 = coord_descent(h_rand0, 400, 300, seed + 7777)
        c5_try3 = compute_c5(h_try3, 2.0 / n0)
        if c5_try3 < best_c5:
            best_c5, best_h = c5_try3, h_try3

    # Phase 5: Final polish
    if time.time() - t0 < budget_s - 10:
        n_f = len(best_h)
        h_final = coord_descent(best_h, 600, 500, seed + 4242)
        c5_final = compute_c5(h_final, 2.0 / n_f)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final
        # SLSQP final
        dx_f = 2.0 / n_f
        try:
            cf, cg = _mk_cons(n_f, n_f / 2.0)
            r = minimize(_mk_obj(dx_f, n_f/2.0, 0.001), best_h.copy(), method='SLSQP',
                         jac=_mk_jac(dx_f, 0.001), bounds=[(0,1)]*n_f,
                         constraints=[{'type':'eq','fun':cf,'jac':cg}],
                         options={'maxiter':500,'ftol':1e-15})
            h_sl = project_constraint(r.x, n_f/2.0)
            c5_sl = compute_c5(h_sl, dx_f)
            if c5_sl < best_c5:
                best_c5, best_h = c5_sl, h_sl
        except Exception:
            pass
        # Final coord descent
        h_final2 = coord_descent(best_h, 400, 500, seed + 9999)
        c5_final2 = compute_c5(h_final2, dx_f)
        if c5_final2 < best_c5:
            best_c5, best_h = c5_final2, h_final2

    n_f = len(best_h)
    best_h = project_constraint(best_h, n_f / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n_f)
    return best_h, best_c5, n_f
