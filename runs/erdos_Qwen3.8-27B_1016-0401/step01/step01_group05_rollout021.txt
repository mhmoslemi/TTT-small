    pop = np.zeros((pop_size, n))
    fit = np.zeros(pop_size)

    pop[0] = h_init
    pop[1] = np.ones(n) * 0.5
    pop[2][:n//2] = 1.0
    pop[3] = np.tile(1.0, n//2 * 2)[:n] * (np.arange(n) % 2 == 0).astype(float)
    # Add periodic patterns with different periods
    for p in [5, 7, 9, 11]:
        idx = min(p, pop_size - 1)
        pat = (np.arange(n) % p < p * 0.5).astype(float)
        if abs(pat.sum() - target) < 1:
            pop[idx] = pat
        else:
            pop[idx] = rng.random(n)

    for i in range(5, pop_size):
        r = rng.random()
        if r < 0.25:
            pop[i] = (rng.random(n) < 0.5).astype(float)
        elif r < 0.5:
            pop[i] = h_init + rng.normal(0, 0.12, n)
        elif r < 0.75:
            pop[i] = rng.random(n)
        else:
            # Shifted parent
            shift = rng.randint(1, n)
            pop[i] = np.roll(h_init, shift)

    for i in range(pop_size):
        pop[i] = make_feasible(pop[i], target)
        fit[i] = compute_c5(pop[i])

    # Update best from initial pop
    mi = np.argmin(fit)
    if fit[mi] < best_c5:
        best_c5 = fit[mi]
        best_h = pop[mi].copy()

    elite_size = 6
    gens = 0
    while time.time() < deadline - 80:
        gens += 1
        new_pop = np.zeros((pop_size, n))
        new_fit = np.zeros(pop_size)

        # Elitism
        elite_idx = np.argsort(fit)[:elite_size]
        for k, idx in enumerate(elite_idx):
            new_pop[k] = pop[idx]
            new_fit[k] = fit[idx]

        # Offspring
        for i in range(elite_size, pop_size):
            p1i = rng.randint(0, pop_size)
            p2i = rng.randint(0, pop_size)
            # Tournament for better diversity
            t1 = rng.randint(0, pop_size, 4)
            t2 = rng.randint(0, pop_size, 4)
            p1i = t1[np.argmin(fit[t1])]
            p2i = t2[np.argmin(fit[t2])]

            c1, c2 = sbx_crossover(pop[p1i], pop[p2i], rng, eta_c=14.0, prob=0.85)
            gf = gens / 500.0
            c1 = mutate_mixed(c1, rng, gf)
            c1 = make_feasible(c1, target)
            new_pop[i] = c1
            new_fit[i] = compute_c5(c1)

            # Occasionally add c2
            if i + 1 < pop_size and rng.random() < 0.3:
                c2m = mutate_mixed(c2, rng, gf)
                c2m = make_feasible(c2m, target)
                new_pop[i + 1] = c2m
                new_fit[i + 1] = compute_c5(c2m)
                i += 1

        pop = new_pop
        fit = new_fit

        mi = np.argmin(fit)
        if fit[mi] < best_c5:
            best_c5 = fit[mi]
            best_h = pop[mi].copy()

        # Periodic local search
        if gens % 15 == 0 and time.time() < deadline - 30:
            h_ls, c5_ls = local_search(best_h, n, target, rng, 1500, deadline - 20)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls
                pop[0] = best_h
                fit[0] = best_c5

    # Final intensive local search
    remaining = deadline - time.time()
    if remaining > 10:
        iters = int(remaining * 3000)
        iters = max(iters, 3000)
        h_ls, c5_ls = local_search(best_h, n, target, rng, iters, deadline - 3)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # One more round of tiny perturbations
    if time.time() < deadline - 2:
        for _ in range(2000):
            if time.time() > deadline - 2:
                break
            if best_c5 >= 0.38086:
                break
            h_try = best_h + rng.normal(0, 0.01, n)
            h_try = make_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    best_h = make_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
