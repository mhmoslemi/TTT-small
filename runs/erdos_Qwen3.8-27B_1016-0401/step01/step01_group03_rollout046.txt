                best[i0] = 0.0
        else:
            if len(ones_cache) >= 2 and len(zeros_cache) >= 2:
                idx_o = np.random.choice(len(ones_cache), 2, replace=False)
                idx_z = np.random.choice(len(zeros_cache), 2, replace=False)
                o1, o2 = ones_cache[idx_o[0]], ones_cache[idx_o[1]]
                z1, z2 = zeros_cache[idx_z[0]], zeros_cache[idx_z[1]]
                best[o1] = 0.0
                best[o2] = 0.0
                best[z1] = 1.0
                best[z2] = 1.0
                new_c5 = compute_c5(best, dx)
                delta = new_c5 - curr_c5
                if delta < 0 or np.random.random() < np.exp(-delta / (T + 1e-10)):
                    curr_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                    ones_cache = np.where(best == 1)[0]
                    zeros_cache = np.where(best == 0)[0]
                else:
                    best[o1] = 1.0
                    best[o2] = 1.0
                    best[z1] = 0.0
                    best[z2] = 0.0
        T *= cooling
    return best, best_c5


def fix_binary_count(h, target):
    cnt = int(np.sum(h))
    if cnt > target:
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:cnt - target]:
            h[i] = 0.0
    elif cnt < target:
        idxs = np.where(h == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:target - cnt]:
            h[i] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0

    best_h = project_feasible(h_init, ts_init)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    # Phase 1: Optimize at initial resolution (25% time)
    t1 = start_time + budget_s * 0.18
    h1, c5_1 = optimize_projected_grad(best_h, dx_init, ts_init, t1, lr_init=1.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    t1b = start_time + budget_s * 0.38
    h1b, c5_1b = optimize_lbfgs(best_h, dx_init, ts_init, t1b)
    if c5_1b < best_c5:
        best_c5 = c5_1b
        best_h = h1b
        best_n = n_init

    # Phase 2: Multi-resolution search (45% time)
    t2 = start_time + budget_s * 0.83
    n_tries = sorted(set([44, 48, 52, 56, 60, 64, 68, 72, 76, 80, 84, 88, 92, 96, 100, 104]))
    n_tries = [n for n in n_tries if n != n_init and n % 2 == 0]

    for n_try in n_tries:
        if time.time() > t2:
            break
        remaining = t2 - time.time()
        if remaining < 15:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0

        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin = (h_interp > 0.5).astype(float)
        h_bin = fix_binary_count(h_bin, int(ts_t))

        # Short SA
        t_sa = min(t2, time.time() + min(15, remaining * 0.15))
        h_sa, c5_sa = sa_binary(h_bin, dx_t, ts_t, t_sa)

        # Continuous from binary
        t_cont = min(t2, time.time() + min(30, remaining * 0.35))
        h_cont, c5_cont = optimize_projected_grad(h_sa, dx_t, ts_t, t_cont, lr_init=0.8)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont
            best_n = n_try

        # Also try from interpolated continuous
        if time.time() < t2:
            h_cont0 = project_feasible(h_interp, ts_t)
            t_cont2 = min(t2, time.time() + min(20, remaining * 0.2))
            h_c2, c5_c2 = optimize_lbfgs(h_cont0, dx_t, ts_t, t_cont2)
            if c5_c2 < best_c5:
                best_c5 = c5_c2
                best_h = h_c2
                best_n = n_try

    # Phase 3: Polish best (12% time)
    t3 = start_time + budget_s * 0.95
    dx_use = 2.0 / best_n
    ts_use = best_n / 2.0

    h_pol, c5_pol = optimize_projected_grad(best_h, dx_use, ts_use, t3, lr_init=0.5)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol

    # Random restarts if time remains
    t4 = start_time + budget_s * 0.98
    restarts = 0
    while time.time() < t4 and restarts < 4:
        restarts += 1
        h_rand = np.random.uniform(0.0, 1.0, best_n)
        h_rand = project_feasible(h_rand, ts_use)
        c5_r = compute_c5(h_rand, dx_use)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_rand.copy()
        t_sub = min(t4, time.time() + max(5, (t4 - time.time()) * 0.3))
        h_r, c5_r2 = optimize_projected_grad(h_rand, dx_use, ts_use, t_sub)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r

    # Final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, best_n
