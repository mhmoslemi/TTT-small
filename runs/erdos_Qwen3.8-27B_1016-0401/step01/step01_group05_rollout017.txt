        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

_g_alpha = 100.0
_g_target = 0.0

def _soft_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = _g_alpha * (corr - m)
    np.clip(vals, -50, 50, out=vals)
    return (m + np.log(np.sum(np.exp(vals))) / _g_alpha) * dx

def _sum_con(x):
    return np.sum(x) - _g_target

def slsqp_refine(h, n, target, alpha, maxiter=100):
    global _g_alpha, _g_target
    _g_alpha = alpha
    _g_target = target
    try:
        res = minimize(_soft_obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_con}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        out = make_feasible(res.x, n, target)
        return out
    except Exception:
        return h.copy()

def sa_loop(h, n, target, rng, deadline, t_init=3.0, cool=0.9999, step0=0.10):

    cur_s = compute_c5(cur)
    best = cur.copy()

    stall = 0
    it = 0
    step = step0

        noise = rng.normal(0, scale, size=len(idx))

def coord_descent(h, n, target, rng, max_passes=5, deadline_check=None):
    cur = h.copy()
    cur_s = compute_c5(cur)
    for p in range(max_passes):
        if deadline_check is not None and time.time() > deadline_check:
            break
        for i in range(n - 1, -1, -1):
            best_d = 0.0
            best_s_i = cur_s
            for j in range(i - 1, -1, -1):
                if j == i:
                    continue
                for delta in [-0.05, -0.02, 0.02, 0.05]:
                    new = cur.copy()
                    new[i] += delta
                    new[j] -= delta
                    if new[i] < 0 or new[i] > 1 or new[j] < 0 or new[j] > 1:
                        continue
                    new = make_feasible(new, n, target)
                    s = compute_c5(new)
                    if s < best_s_i:
                        best_s_i = s
                        best_d = delta
                        best_j = j
            if best_d != 0.0:
                cur[i] += best_d
                cur[best_j] -= best_d
                np.clip(cur, 0.0, 1.0, out=cur)
                cur = make_feasible(cur, n, target)
                cur_s = best_s_i
    return cur, cur_s
