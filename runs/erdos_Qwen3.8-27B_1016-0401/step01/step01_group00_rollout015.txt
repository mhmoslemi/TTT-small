        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def scl_solve(h0, target, T, maxiter, t0, budget):
    global _G_T, _G_TARGET, _G_N
    _G_TARGET = target
    _G_N = len(h0)
    _G_T = T
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scl_fun, 'jac': _scl_jac}]
    try:
        res = minimize(_fobj, h0, method='SLSQP', jac=_fgrad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < 100:
            h = project_constraint(res.x, target)
            return h, compute_c5(h)
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n0 = len(h_best)
    target0 = n0 / 2.0
    h_best = project_constraint(h_best, target0)
    c5_best = compute_c5(h_best)

    # Phase 1: Quick SLSQP at original resolution
    for T in [40, 100]:
        if time.time() - t0 > budget - 500:
            break
        h, c5 = scl_solve(h_best, target0, T, 300, t0, budget - 200)
        if c5 < c5_best:
            c5_best, h_best = c5, h

    # Phase 2: Upsample to n1=560
    n1 = 560
    target1 = n1 / 2.0
    x_old = np.arange(n0) * (2.0 / n0)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    h1 = np.interp(x1, x_old, h_best)
    h1 = project_constraint(h1, target1)

    # Phase 3: Annealed projected gradient at n1
    for T in [15, 40, 80, 160, 320, 600]:
        if time.time() - t0 > budget - 250:
            break
        h1, c5 = pg_solve(h1, target1, T, 2500, 0.05, t0, budget - 80)
        if c5 < c5_best:
            c5_best, h_best = c5, h1.copy()

    # Phase 4: SLSQP fine-tune at n1
    for T in [100, 250, 500]:
        if time.time() - t0 > budget - 120:
            break
        h1, c5 = scl_solve(h1, target1, T, 350, t0, budget - 40)
        if c5 < c5_best:
            c5_best = c5
            h_best = h1.copy()

    # Phase 5: Structured perturbations + reoptimize
    for r in range(40):
        if time.time() - t0 > budget - 40:
            break
        h_p = h1.copy()
        if r % 4 == 0:
            idx = rng.integers(0, n1)
            sz = rng.integers(4, 30)
            for i in range(sz):
                h_p[(idx + i) % n1] = 1.0 - h_p[(idx + i) % n1]
        elif r % 4 == 1:
            h_p = np.clip(h1 + rng.normal(0, 0.05, n1), 0, 1)
        elif r % 4 == 2:
            h_p = (h1 > rng.uniform(0.2, 0.8)).astype(float)
        else:
            shift = rng.integers(1, 15)
            h_p = np.roll(h1, shift)
        h_p = project_constraint(h_p, target1)
        T_p = 100 + r * 5
        h_p, c5_p = pg_solve(h_p, target1, T_p, 800, 0.04, t0, budget - 20)
        if c5_p < c5_best:
            c5_best, h_best = c5_p, h_p.copy()
            h1 = h_p.copy()

    # Phase 6: Higher resolution n2=840 if time allows
    if time.time() - t0 < budget - 100:
        n2 = 840
        target2 = n2 / 2.0
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        nc = len(h_best)
        xc = np.arange(nc) * (2.0 / nc)
        h2 = np.interp(x2, xc, h_best)
        h2 = project_constraint(h2, target2)
        for T in [40, 120, 280]:
            if time.time() - t0 > budget - 30:
                break
            h2, c5_2 = pg_solve(h2, target2, T, 1200, 0.035, t0, budget - 15)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h2
        if time.time() - t0 < budget - 25:
            h2, c5_2 = scl_solve(h2, target2, 200, 150, t0, budget - 5)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h2

    # Phase 7: Final micro-refinement at best resolution
    nf = len(h_best)
    tf = nf / 2.0
    if time.time() - t0 < budget - 15:
        h_best, c5_f = pg_solve(h_best, tf, 800, 500, 0.02, t0, budget - 5)
        if c5_f < c5_best:
            c5_best = c5_f

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
