import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(v, target):
    n = len(v)
    lo = -max(v.max(), 0.0) - 0.2
    hi = -min(v.min(), 0.0) + 0.2
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    return np.clip(v + lam, 0.0, 1.0)

def obj_smooth(x, dx, tau):
    c = corr_vec(x)
    cmax = c.max()
    return dx * (cmax + tau * np.log(np.sum(np.exp((c - cmax) / tau))))

def grad_smooth(x, dx, tau):
    n = len(x)
    c = corr_vec(x)
    cmax = c.max()
    w = np.exp((c - cmax) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        s = m - (n - 1)
        wm = w[m]
        jlo = max(0, -s)
        jhi = min(n, n - s)
        if jlo < jhi:
            grad[jlo:jhi] += wm * (1.0 - x[jlo + s:jhi + s])
        jlo2 = max(0, s)
        jhi2 = min(n, n + s)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] += wm * (-x[jlo2 - s:jhi2 - s])
    return dx * grad

def grad_exact(x, dx):
    n = len(x)
    c = corr_vec(x)
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    grad = np.zeros(n)
    jlo = max(0, -s)
    jhi = min(n, n - s)
    if jlo < jhi:
        grad[jlo:jhi] += (1.0 - x[jlo + s:jhi + s])
    jlo2 = max(0, s)
    jhi2 = min(n, n + s)
    if jlo2 < jhi2:
        grad[jlo2:jhi2] += (-x[jlo2 - s:jhi2 - s])
    return dx * grad

def _mk_obj(dx, tau):
    def f(x):
        return obj_smooth(x, dx, tau)
    return f

def _mk_jac(dx, tau):
    def g(x):
        return grad_smooth(x, dx, tau)
    return g

def slsqp_smooth(h, n, tau, max_iter=200, ftol=1e-14):
    dx = 2.0 / n
    T = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return float(np.sum(x) - T)
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h = project_box_sum(h, T)
    try:
        r = minimize(_mk_obj(dx, tau), h, jac=_mk_jac(dx, tau),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        h = project_box_sum(r.x, T)
        c5 = compute_c5(h, dx)
        return h, c5
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_exact(h, n, max_iter=500):
    dx = 2.0 / n
    T = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return float(np.sum(x) - T)
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h = project_box_sum(h, T)
    def f(x):
        return compute_c5(x, dx)
    try:
        r = minimize(f, h, jac=lambda x: grad_exact(x, dx),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': 1e-15})
        h = project_box_sum(r.x, T)
        c5 = compute_c5(h, dx)
        return h, c5
    except Exception:
        return h, compute_c5(h, dx)

def local_search(h, n, max_sweeps=30, time_limit=None, t0=None):
    dx = 2.0 / n
    T = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        for i in range(n):
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.05, -0.05]:
                h_try = best.copy()
                h_try[i] += delta
                h_try = project_box_sum(h_try, T)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-12:
                    best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            # Try pairwise swaps
            for i in range(n):
                for j in range(i + 1, n):
                    h_try = best.copy()
                    h_try[i], h_try[j] = best[j], best[i]
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_c5 - 1e-12:
                        best = h_try
                        best_c5 = c5_try
                        improved = True
        if not improved:
            break
    return best, best_c5

def perturb_structured(h, rng, idx, mode):
    v = h.copy()
    n = len(v)
    if mode == 0:
        v = v[::-1].copy()
    elif mode == 1:
        v[idx:idx+n//2] += rng.normal(0, 0.05, n//2)
    elif mode == 2:
        perm = rng.permutation(n)
        v = h[perm]
    elif mode == 3:
        shift = rng.randint(1, n//4)
        v = np.roll(h, shift)
    else:
        v += rng.normal(0, 0.03, n)
    return project_box_sum(v, n / 2.0)

def interpolate_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
