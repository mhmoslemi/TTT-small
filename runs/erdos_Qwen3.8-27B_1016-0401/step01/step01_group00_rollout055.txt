def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()  # numerical stability
    val = mc + np.log(np.exp(s).sum()) / T
    return val * (2.0 / n)

def soft_max_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o]
        if wi < 1e-12:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] += -wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)  # least room first
        else:
            idx = np.argsort(h)  # least value first
        for i in idx:
            if diff > 0:
                s = min(diff, 1.0 - h[i])
                if s > 0:
                    h[i] += s
            else:
                s = min(-diff, h[i])
                if s > 0:
                    h[i] -= s
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def grad_topk(h, n, top_k=3):
    """Sum of gradients of top-k lags."""
    corr = np.correlate(h, 1 - h, mode='full')
    top = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for ki in top:
        k = ki - (n - 1)
        if k == 0:
            grad += 1.0 - 2.0 * h
        elif k > 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] += -h[:n-k]
        else:
            p = -k
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] += -h[:n-p]
    return grad

def sgs_step(h, n, lr, top_k=3, target=None):
    """One step of projected subgradient descent on top-k lags."""
    if target is None:
        target = n / 2.0
    g = grad_topk(h, n, top_k)
    h_new = h - lr * g
    return fix_constraint(h_new, target)

def slsqp_run(h, n, T, target, maxiter, bounds, cons, ftol=1e-14):
    def obj(x):
        return soft_max_obj(x, n, T)
    def jac(x):
        return soft_max_grad(x, n, T)
    res = minimize(obj, h, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res

def make_constraints(n, target):
    return [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t,
             'jac': lambda x, n_=n: np.ones(n_)}]

def run(seed=42, budget_s=1100, **kwargs):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def soft_max_obj(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    val = mc + np.log(np.exp(s).sum()) / T
    return val * (2.0 / n)

def soft_max_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[center + o]
        if wi < 1e-12:
            continue
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] += -wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    return grad * dx

                if s > 0: h[i] -= s

        else:
            p = -o
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] += -h[:n-p]

    return h

def grad_topk(h, n, k):
    corr = np.correlate(h, 1 - h, mode='full')
    top = np.argsort(corr)[-k:]
    g = np.zeros(n)
    for ki in top:
        o = ki - (n - 1)
        if o == 0:
            g += 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += (1.0 - h[:n - o])
            g[:n - o] += -h[o:]
        else:
            p = -o
            g[:n - p] += (1.0 - h[p:])
            g[p:] += -h[:n - p]
    return g

def _obj_cb(x):
    return soft_max_obj(x, _gn, _gT)

def _jac_cb(x):
    return soft_max_grad(x, _gn, _gT)

def _consfun(x):
    return x.sum() - _gtarget

def _consjac(x):
    return np.ones(len(x))

def slsqp_opt(hs, n, T, target, maxiter, ftol):
    global _gn, _gT, _gtarget
    _gn, _gT, _gtarget = n, T, target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _consfun, 'jac': _consjac}]
    try:
        r = minimize(_obj_cb, hs.copy(), method='SLSQP', jac=_jac_cb,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.fun < 50:
            return fix_constraint(r.x.copy(), target)
    except Exception:
        pass
    return hs.copy()

def grad_polish(h_best, n, target, c5b, n_steps, lr0, lr_decay, top_k, time_fn, budget_rem):
    lr = lr0
    for i in range(n_steps):
        if time_fn() > budget_rem:
            break
        g = grad_topk(h_best, n, top_k)
        ht = fix_constraint(h_best - lr * g, target)
        c = compute_c5(ht)
        if c < c5b:
            c5b = c
            h_best = ht
        if lr0 > 0.01 and (i + 1) % 400 == 0:
            lr = max(lr0 * 0.05, lr * lr_decay)
    return h_best, c5b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.asarray(initial_h_values, dtype=float).ravel().copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5b = compute_c5(h_best)

    # Phase 1: SLSQP cooling schedule
    for T in [100, 500, 2000, 8000, 32000, 128000]:
        if elapsed() > budget_s - 300:
            break
        hn = slsqp_opt(h_best, n, T, target, 5000, 1e-14)
        c = compute_c5(hn)
        if c < c5b:
            c5b, h_best = c, hn

    # Phase 2: Gradient polish (top-3 active set)
    h_best, c5b = grad_polish(h_best, n, target, c5b, 4000, 0.004, 0.55, 3, elapsed, budget_s - 250)

    # Phase 3: Simulated annealing with local SLSQP
    Tsa = 0.008
    for i in range(300):
        if elapsed() > budget_s - 200:
            break
        sig = 0.015 + 0.075 * (1.0 - i / 300.0)
        ht = h_best + rng.normal(0, sig, n)
        ht = fix_constraint(ht, target)
        ct = compute_c5(ht)
        accept = ct < c5b
        if not accept and ct - c5b < 0.05:
            accept = rng.random() < np.exp(-(ct - c5b) / Tsa)
        if accept:
            if ct < c5b:
                c5b = ct
            h_best = ht
            if elapsed() < budget_s - 80:
                hn = slsqp_opt(h_best, n, 5000, target, 2000, 1e-14)
                c = compute_c5(hn)
                if c < c5b:
                    c5b, h_best = c, hn
        Tsa = max(0.002, Tsa * 0.992)

    # Phase 4: Very low-temperature SLSQP for final convergence
    for T in [100000, 400000, 1000000]:
        if elapsed() > budget_s - 60:
            break
        hn = slsqp_opt(h_best, n, T, target, 8000, 1e-15)
        c = compute_c5(hn)
        if c < c5b:
            c5b, h_best = c, hn

    # Phase 5: Fine gradient polish (top-5, very small lr)
    h_best, c5b = grad_polish(h_best, n, target, c5b, 5000, 0.0008, 0.997, 5, elapsed, budget_s - 30)

    # Phase 6: Try downsampled n/2 for structural insight
    n2 = n // 2
    if n2 >= 50 and elapsed() < budget_s - 100:
        t2 = n2 / 2.0
        xo = np.arange(n) * (2.0 / n)
        xn = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(xn, xo, h_best)
        h2 = fix_constraint(h2, t2)
        for T in [500, 5000, 50000, 500000]:
            if elapsed() > budget_s - 40:
                break
            h2n = slsqp_opt(h2, n2, T, t2, 3000, 1e-14)
            c = compute_c5(h2n)
            if c < c5b:
                c5b, h_best, n = c, h2n, n2
                h2 = h2n
        h_best, c5b = grad_polish(h_best, n, n / 2.0, c5b, 3000, 0.003, 0.998, 3, elapsed, budget_s - 20)

    # Final gradient polish at whatever n we ended up with
    target_final = len(h_best) / 2.0
    h_best, c5b = grad_polish(h_best, len(h_best), target_final, c5b,
                               2000, 0.001, 0.998, 3, elapsed, budget_s - 5)

    c5f = min(compute_c5(h_best), c5b)
    return (h_best, c5f, len(h_best))
