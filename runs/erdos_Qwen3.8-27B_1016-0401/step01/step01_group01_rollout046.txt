def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(v, T):
    n = len(v)
    lo = float(v.min()) - T
    hi = float(v.max())

        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()

        else:

def smooth_obj_grad(h, dx, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    val = dx * (c_max + t * np.log(np.sum(np.exp((c - c_max) / t))))
    w = np.exp((c - c_max) / t)
    w /= np.sum(w)
    grad = np.zeros(n)
    ks = np.arange(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        v1 = (ks - s >= 0) & (ks - s < n)
        grad[v1] += w[m] * (1.0 - h[ks[v1] - s])
        v2 = (ks + s >= 0) & (ks + s < n)
        grad[v2] += w[m] * (-h[ks[v2] + s])
    return val, dx * grad

        v2 = (ks + s >= 0) & (ks + s < n)

        for j in range(n):
            c_full = np.correlate(h, 1.0 - h, mode='full')
            a = c_full.copy()
            b = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                s = m - (n - 1)
                contrib = 0.0
                coeff = 0.0
                if 0 <= j + s < n:
                    contrib += h[j] * (1.0 - h[j + s])
                    coeff += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    contrib += h[j - s] * (1.0 - h[j])
                    coeff += (-h[j - s])
                a[m] = c_full[m] - contrib
                b[m] = coeff
            def obj_1d(x):
                return float(np.max(a + x * b))
            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-14})
                h[j] = res.x
            except Exception:
                pass
            h = project_feasible(h, T)
    return h
