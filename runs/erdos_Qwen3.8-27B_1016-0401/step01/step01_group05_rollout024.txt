```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_grad(x, alpha, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = np.clip(alpha * (corr - cmax), -100, 100)
    exp_s = np.exp(scaled)
    ssum = np.sum(exp_s)
    obj = (cmax + np.log(ssum) / alpha) * dx
    w = exp_s / ssum
    offsets = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)
    idx1 = j[:, None] + offsets[None, :]
    v1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j[:, None] - offsets[None, :]
    v2 = (idx2 >= 0) & (idx2 < n)
    t1 = np.where(v1, w[None, :] * (1.0 - h[np.clip(idx1, 0, n - 1)]), 0.0)
    t2 = np.where(v2, w[None, :] * h[np.clip(idx2, 0, n - 1)], 0.0)
    grad = dx * (t1.sum(axis=1) - t2.sum(axis=1))
    return obj, grad

def lbgfsb_penalty(h0, n, target, alpha, maxiter=200):
    def fun(x):
        h = np.clip(x, 0.0, 1.0)
        obj, _ = soft_max_obj_grad(h, alpha, n)
        diff = np.sum(h) - target
        return obj + 500.0 * diff * diff

    def gfun(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = soft_max_obj_grad(h, alpha, n)
        diff = np.sum(h) - target
        return g + 2.0 * 500.0 * diff

    res = minimize(fun, h0.copy(), jac=gfun, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    return make_feasible(np.clip(res.x, 0, 1), n, target)

def gradient_descent(h0, n, target, alpha, max_steps=500, deadline=None):
    h = h0.copy()
    lr = 0.02
    for step in range(max_steps):
        if deadline and time.time() > deadline:
            break
        obj, grad = soft_max_obj_grad(h, alpha, n)
        diff = np.sum(h) - target
        grad += 2.0 * 200.0 * diff  # penalty gradient
        step_size = lr / (np.max(np.abs(grad)) + 1e-12)
        step_size = min(step_size, 0.05)
        h_new = np.clip(h - step_size * grad, 0.0, 1.0)
        h_new = make_feasible(h_new, n, target)
        obj_new, _ = soft_max_obj_grad(h_new, alpha, n)
        if obj_new < obj:
            h = h_new
            lr *= 1.02
        else:
            lr *= 0.7
        if lr < 1e-6:
            break
    return make_feasible(h, n, target)

def local_search(h0, n, target, rng, max_iters, deadline):
    h = h0.copy()
    best_c5 = compute_c5(h)
    step_init = 0.04
    for it in range(max_iters):
        if time.time() > deadline:
            break
        step = step_init * (0.97 ** (it // 2000))
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        delta = rng.uniform(-step, step)
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_c5 = c5_try
        elif it % 4000 == 2000:
            idx = rng.choice(n, size=4, replace=False)
            h[idx] += rng.uniform(-0.02, 0.02, size=4)
            h = make_feasible(np.clip(h, 0, 1), n, target)
            best_c5 = compute_c5(h)
    return h, best_c5

def gradient_informed_ls(h0, n, target, rng, max_iters, deadline):
    h = h0.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        obj, grad = soft_max_obj_grad(h, 200.0, n)
        top_k = np.argsort(np.abs(grad))[-8:]
        i = top_k[rng.randint(len(top_k))]
        j = top_k[rng.randint(len(top_k))]
        if i == j:
            j = (j + 1) % n
        direction = -np.sign(grad[i] - grad[j])
        step = 0.03 * (1.0 - 0.5 * it / max_iters)
        delta = direction * step
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def make_structured(n, rng):
    x = np.linspace(0, 2, n)
    candidates = []
    c1 = np.full(n, 0.5)
    c2 = 0.5 + 0.4 * np.sin(np.pi * x)
    c3 = 0.5 + 0.35 * np.cos(2 * np.pi * x)
    c4 = np.where(x < 0.5, 1.0, np.where(x < 1.5, 0.0, 1.0))
    c5 = np.clip(1.0 - 1.5 * np.abs(x - 1.0), 0.0, 1.0)
    c6 = np.clip(0.5 + 0.5 * np.sin(3 * np.pi * x + rng.uniform(0, 2 * np.pi)), 0, 1)
    candidates = [np.clip(c, 0, 1) for c in [c1, c2, c3, c4, c5, c6]]
    best_h, best_c5 = None, 999
    for c in candidates:
        c = make_feasible(c, n, n / 2.0)
        c5v = compute_c5(c)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = c
    return best_h

def optimize_h(h0, n, target, deadline):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    alphas = [10, 25, 50, 100, 200, 400, 800, 1600]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        h1 = lbgfsb_penalty(best_h, n, target, a, maxiter=150)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1
        if time.time() > deadline - 30:
            break
        h2 = gradient_descent(best_h, n, target, a, max_steps=300, deadline=deadline - 30)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(70, 0.5)

    results = []

    for n in [70, 100, 140]:
        if time.time() > deadline - 60:
            break
        target = n / 2.0
        if n == len(h_parent):
            h0 = make_feasible(h_parent.copy(), n, target)
        else:
            h_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_parent)), h_parent)
            h0 = make_feasible(h_resampled, n, target)
        h_struct = make_structured(n, rng)
        h0 = h0 if compute_c5(h0) < compute_c5(h_struct) else h_struct

        h_opt, c5_opt = optimize_h(h0, n, target, deadline - 60)
        results.append((c5_opt, h_opt, n, target))

    results.sort(key=lambda x: x[0])
    best_c5, best_h, best_n, best_target = results[0]

    # Local search polish on best
    if time.time() < deadline - 20:
        h_ls, c5_ls = gradient_informed_ls(best_h, best_n, best_target, rng, 20000, deadline - 15)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls
    if time.time() < deadline - 10:
        h_ls2, c5_ls2 = local_search(best_h, best_n, best_target, rng, 30000, deadline - 5)
        if c5_ls2 < best_c5:
            best_h, best_c5 = h_ls2, c5_ls2

    # Final refinement
    if time.time() < deadline - 5:
        h_fin = lbgfsb_penalty(best_h, best_n, best_target, 2000, maxiter=100)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    best_h = make_feasible(best_h, best_n, best_target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, best_n
```
