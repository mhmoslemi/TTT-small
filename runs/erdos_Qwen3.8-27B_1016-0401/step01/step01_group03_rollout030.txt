    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    sh = corr - mc
    ec = np.exp(beta * sh)
    se = ec.sum()

    w = dx * ec / se

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = ts - p.sum()
    if abs(d) > 1e-14:
        if d > 0:
            m = p < 1.0 - 1e-10
        else:
            m = p > 1e-10
        if m.any():
            p[m] += d / m.sum()
    return np.clip(p, 0.0, 1.0)

def _eqf(x, ts):
    return x.sum() - ts

def _eqj(x):
    return np.ones(len(x))

def sm_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    ec = np.exp(beta * (corr - mc))
    se = ec.sum()
    obj = dx * (mc + np.log(se) / beta)
    w = dx * ec / se
    n = len(h)
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - n + 1
        if m >= 0:
            g[m:n] += wi * (1.0 - h[:n-m])
            g[:n-m] -= wi * h[m:]
        else:
            p = -m
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
    return obj, g

def smooth_opt(h0, dx, ts, tl):
    n = len(h0)
    h = project_feasible(h0, ts).astype(float)
    bh = h.copy()
    bc = compute_c5(h, dx)
    cons = [{'type': 'eq', 'fun': partial(_eqf, ts=ts), 'jac': _eqj}]
    bnds = [(0.0, 1.0)] * n
    for bt in [30.0, 100.0, 300.0, 1000.0, 3000.0]:
        if time.time() > tl:
            break
        try:
            r = minimize(sm_grad, h, args=(dx, bt), jac=True, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 2000, 'ftol': 1e-16})
            hn = project_feasible(r.x, ts)
            cn = compute_c5(hn, dx)
            if cn < bc:
                bc, bh, h = cn, hn.copy(), hn.copy()
        except Exception:
            pass
    # Projected gradient with BB step size
    bt = 3000.0
    h_prev = h.copy()
    g_prev, _ = sm_grad(h_prev, dx, bt)
    _, g = sm_grad(h, dx, bt)
    s = h - h_prev
    y = g - g_prev
    sy = s.dot(y)
    yy = y.dot(y)
    lr = sy / yy if yy > 1e-12 else 0.1
    st = 0
    for it in range(200000):
        if time.time() > tl:
            break
        hn = project_feasible(h - lr * g, ts)
        cn = compute_c5(hn, dx)
        h_new, g_new = hn, None
        _, g_new = sm_grad(hn, dx, bt)
        s = hn - h
        y = g_new - g
        sy = s.dot(y)
        yy = y.dot(y)
        if yy > 1e-12:

        if cn < bc - 1e-16:
            bc, bh = cn, hn.copy()
            st = 0
        else:
            st += 1
            lr *= 0.9
        h, g = hn, g_new
        if st > 2000 or lr < 1e-9:
            break
    return bh, bc
