def compute_c5(h, dx):

def c5_from_binary(h):
    n = len(h)

    int_c5 = np.max(corr) * dx

def binary_swap_eval(h, i, j, n):
    """Evaluate change in max correlation when swapping positions i and j."""
    # This is expensive to do incrementally, so just recompute
    h[i], h[j] = h[j], h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    h[i], h[j] = h[j], h[i]
    return max_c

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        else:

        if mask.sum() > 0:

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)

            idx1 = np.arange(m, n)

            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]

            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def smooth_obj_grad(h, dx, beta):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)

    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)

    return np.ones(len(x))

def optimize_smooth(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for beta in [200.0, 1000.0, 5000.0, 20000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    beta = 5000.0
    lr = 0.5
    stall = 0
    for it in range(200000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.03, 3.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.95
        h = h_new
        if stall > 2000 or lr < 1e-8:
            break
    return best_h, best_c5

def binary_sa(h0, n, t_limit, target):
    """SA for binary seq with target max correlation."""
    dx = 2.0 / n
    ts = n // 2
    cur = (h0 > 0.5).astype(float).copy()
    oc = int(np.sum(cur))
    if oc > ts:
        idxs = np.where(cur == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            cur[i] = 0.0
    elif oc < ts:
        idxs = np.where(cur == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            cur[i] = 1.0
    mc, _ = max_corr_val(cur)
    best_mc = mc
    best = cur.copy()
    T = max(3.0, target * 2.0)
    T_min = 0.01
    cool = 0.99985
    for it in range(500000):
        if time.time() > t_limit:
            break
        zeros = np.where(cur == 0)[0]
        ones = np.where(cur == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        if np.random.random() < 0.7:
            zi = zeros[np.random.randint(len(zeros))]
            oi = ones[np.random.randint(len(ones))]
            cur[zi], cur[oi] = 1.0, 0.0
            new_mc, _ = max_corr_val(cur)
            delta = max(0.0, new_mc - target) - max(0.0, mc - target) + 0.001 * (new_mc - mc)
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                mc = new_mc
                if mc < best_mc:
                    best_mc = mc
                    best = cur.copy()
            else:
                cur[zi], cur[oi] = 0.0, 1.0
        else:
            if len(zeros) < 2 or len(ones) < 2:
                continue
            z1, z2 = np.random.choice(len(zeros), 2, replace=False)
            o1, o2 = np.random.choice(len(ones), 2, replace=False)
            zpos = [zeros[z1], zeros[z2]]
            opos = [ones[o1], ones[o2]]
            cur[zpos[0]], cur[zpos[1]], cur[opos[0]], cur[opos[1]] = 1.0, 1.0, 0.0, 0.0
            new_mc, _ = max_corr_val(cur)
            delta = max(0.0, new_mc - target) - max(0.0, mc - target) + 0.001 * (new_mc - mc)
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                mc = new_mc
                if mc < best_mc:
                    best_mc = mc
                    best = cur.copy()
            else:
                cur[zpos[0]], cur[zpos[1]], cur[opos[0]], cur[opos[1]] = 0.0, 0.0, 1.0, 1.0
        T = max(T * cool, T_min)
    return best, best_mc * dx

def binary_greedy_local(h0, n, t_limit):
    """Greedy 1-swap local search."""
    dx = 2.0 / n
    cur = h0.copy()
    mc, _ = max_corr_val(cur)
    for _ in range(200):
        if time.time() > t_limit:
            break
        improved = False
        zeros = np.where(cur == 0)[0]
        ones = np.where(cur == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        for oi in ones:
            if time.time() > t_limit:
                return cur, mc * dx
            for zi in zeros:
                cur[zi], cur[oi] = 1.0, 0.0
                new_mc, _ = max_corr_val(cur)
                if new_mc < mc:
                    mc = new_mc
                    improved = True
                    break
                cur[zi], cur[oi] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return cur, mc * dx

def greedy_construct(n, rng):
    """Greedy left-to-right construction."""
    h = np.zeros(n)
    remaining = n // 2
    for i in range(n):
        if remaining == 0 or remaining == n - i:
            h[i] = 1.0 if remaining == n - i else 0.0
            remaining -= h[i]
            continue
        for val in [1.0, 0.0]:
            if val == 1.0 and remaining == 0:
                continue
            if val == 0.0 and remaining == n - i:
                continue
            h[i] = val
            mc, _ = max_corr_val(h[:i+1])
            if np.random.random() < 0.3 or (val == 1.0 and remaining >= 1):
                break
        if h[i] == 1.0:
            remaining -= 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    ts_init = n_init / 2.0
    
    best_h = project_feasible(h_init, ts_init)
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init
    
    # Phase 1: Continuous optimization (25% time)
    t1 = start_time + budget_s * 0.25
    h1, c5_1 = optimize_smooth(best_h, dx_init, ts_init, t1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
        best_n = n_init
    
    # Phase 2: Binary search at multiple resolutions (55% time)
    t2 = start_time + budget_s * 0.80
    x_old = np.linspace(0.0, 2.0, n_init)
    
    for n_try, tgt in [(48, 9), (50, 9), (52, 9), (96, 18)]:
        if time.time() > t2:
            break
        rem = t2 - time.time()
        if rem < 15:
            break
        dx_t = 2.0 / n_try
        ts_t = n // 2 if False else n_try // 2
        
        # Initial: interpolate
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, h_init)
        h_bin = (h_interp > 0.5).astype(float)
        oc = int(np.sum(h_bin))
        if oc != ts_t:
            if oc > ts_t:
                idxs = np.where(h_bin == 1)[0]
                order = idxs[np.argsort(h_interp[idxs])]
                for i in order[:oc - ts_t]:
                    h_bin[i] = 0.0
            else:
                idxs = np.where(h_bin == 0)[0]
                order = idxs[np.argsort(-h_interp[idxs])]
                for i in order[:ts_t - oc]:
                    h_bin[i] = 1.0
        
        # Greedy local search
        t_ls = min(t2, time.time() + max(5, rem * 0.15))
        h_ls, c5_ls = binary_greedy_local(h_bin, n_try, t_ls)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
            best_n = n_try
        
        # SA from interpolated init
        if time.time() < t2:
            t_sa = min(t2, time.time() + max(8, rem * 0.20))
            h_sa, c5_sa = binary_sa(h_ls, n_try, t_sa, tgt)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
                best_n = n_try
        
        # SA from greedy construct
        if time.time() < t2:
            h_gc = greedy_construct(n_try, rng)
            t_sa2 = min(t2, time.time() + max(8, rem * 0.15))
            h_sa2, c5_sa2 = binary_sa(h_gc, n_try, t_sa2, tgt)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2
                best_n = n_try
        
        # Random restart SA
        if time.time() < t2:
            h_rand = np.zeros(n_try)
            pos = rng.choice(n_try, ts_t, replace=False)
            h_rand[pos] = 1.0
            t_sr = min(t2, time.time() + max(5, rem * 0.12))
            h_sr, c5_sr = binary_sa(h_rand, n_try, t_sr, tgt)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr
                best_n = n_try
        
        # Local polish on best
        if time.time() < t2:
            t_pol = min(t2, time.time() + max(3, rem * 0.08))
            h_pol, c5_pol = binary_greedy_local(best_h if best_n == n_try else h_sa, n_try, t_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
                best_n = n_try
    
    # Phase 3: Continuous refinement on best binary (12% time)
    t3 = start_time + budget_s * 0.94
    if time.time() < t3:
        n_use = best_n
        dx_use = 2.0 / n_use
        ts_use = n_use / 2.0
        h_cont = np.clip(best_h, 0.0, 1.0).astype(float)
        h_opt, c5_opt = optimize_smooth(h_cont, dx_use, ts_use, t3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_use
    
    # Phase 4: Final binary polish (6% time)
    t4 = start_time + budget_s * 0.985
    if time.time() < t4 and best_n <= 100:
        n_use = best_n
        dx_use = 2.0 / n_use
        ts_use = n_use // 2
        h_bin2 = (best_h > 0.5).astype(float)
        oc = int(np.sum(h_bin2))
        if oc != ts_use:
            if oc > ts_use:
                idxs = np.where(h_bin2 == 1)[0]
                rng.shuffle(idxs)
                for i in idxs[:oc - ts_use]:
                    h_bin2[i] = 0.0
            else:
                idxs = np.where(h_bin2 == 0)[0]
                rng.shuffle(idxs)
                for i in idxs[:ts_use - oc]:
                    h_bin2[i] = 1.0
        h_fin, c5_fin = binary_greedy_local(h_bin2, n_use, t4)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
            best_n = n_use
    
    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    
    return best_h, best_c5, best_n
