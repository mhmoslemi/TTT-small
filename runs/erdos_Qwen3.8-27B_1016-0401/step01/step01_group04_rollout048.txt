def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

    return np.correlate(h, 1.0 - h, mode='full').max()

def to_binary(h, S):
    n = len(h)
    h_b = np.zeros(n)
    target = int(S)
    order = np.argsort(-h)
    h_b[order[:target]] = 1

def full_hill_climb(h, t_deadline, max_sweeps=10000):
    n = len(h)
    h_cur = h.astype(float).copy()
    c5 = np.correlate(h_cur, 1.0 - h_cur, mode='full').max()
    best_h = h_cur.copy()
    best_c5 = c5
    sweep = 0
    while sweep < max_sweeps:

            break
        improved = False
        one_idx = np.where(h_cur == 1)[0]
        zero_idx = np.where(h_cur == 0)[0]

def swap_delta(h, p, q):
    n = len(h)
    # Compute delta for all lags
    d_range = np.arange(-(n-1), n)  # all possible offsets
    # h[p+d] for valid p+d
    pd = p + d_range
    hp_d = np.where((pd >= 0) & (pd < n), h[pd], 0)
    # h[p-d] for valid p-d
    pdm = p - d_range
    hp_dm = np.where((pdm >= 0) & (pdm < n), h[pdm], 0)
    # h[q+d] for valid q+d
    qd = q + d_range
    hq_d = np.where((qd >= 0) & (qd < n), h[qd], 0)
    # h[q-d] for valid q-d
    qdm = q - d_range
    hq_dm = np.where((qdm >= 0) & (qdm < n), h[qdm], 0)
    
    # Handle special cases where indices point to p or q
    # h'[p+d] where p+d=q → 1
    mask_pq = (pd == q)
    hp_d[mask_pq] = 1
    # h'[p-d] where p-d=q → 1
    mask_pdm_q = (pdm == q)
    hp_dm[mask_pdm_q] = 1
    # h'[q+d] where q+d=p → 0
    mask_qp = (qd == p)
    hq_d[mask_qp] = 0
    # h'[q-d] where q-d=p → 0
    mask_qdm_p = (qdm == p)
    hq_dm[mask_qdm_p] = 0
    
    delta = hp_d + hp_dm - hq_d - hq_dm
    return delta
