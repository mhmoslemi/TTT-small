def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(np.clip(beta * shifted, -100, 0))
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            grad[m:] += wi * (1.0 - h[:n - m])
            grad[:n - m] -= wi * h[m:]
        else:
            p = -m
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return obj, grad
