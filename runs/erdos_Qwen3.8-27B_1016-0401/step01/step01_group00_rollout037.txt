def project(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target}"""
    h = h.copy()
    # Iterative: clip to [0,1], then adjust
    for _ in range(50):
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add: add to those not at 1
            mask = h < 1.0
            if not mask.any():
                break
            add = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            # Need to subtract: subtract from those not at 0
            mask = h > 0.0
            if not mask.any():
                break
            sub = -diff / mask.sum()
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    # Final fix
    h = np.clip(h, 0, 1)
    # Exact fix for remaining error
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Add/subtract 1/n to all
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def project_sum(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target} using bisection on threshold."""
    h = h.copy()
    s = h.sum()
    n = len(h)
    if abs(s - target) < 1e-14:
        return np.clip(h, 0, 1)
    
    # First clip
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    
    # Water-filling: find threshold tau such that sum(clip(h - tau, 0, 1)) = target
    # clip(h - tau, 0, 1) = max(0, min(1, h - tau))
    # For tau > max(h): result is all 0s → sum = 0
    # For tau < min(h) - 1: result is all 1s → sum = n
    # We need sum = target
    
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi)/2, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    e = np.exp(shifted)
    esum = e.sum()
    sw = e / esum
    center = n - 1
    grad = np.zeros(n)
    w0 = sw[center]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = sw[center + p]
        wm = sw[center - p]
        if wp > 1e-12:
            grad[p:] += wp * (1.0 - h[:n - p])
            grad[:n - p] -= wp * h[p:]
        if wm > 1e-12:
            grad[:n - p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:n - p]
    grad *= dx
    obj = (mc + np.log(esum) / T) * dx
    return obj, grad

def project(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h, target, n_iters, T0, T1, lr, seed=0, max_time=None, t0=None):
    n = len(h)
    h = project(h, target)
    rng = np.random.default_rng(seed)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    
    for it in range(n_iters):
        if max_time is not None and t0 is not None and (it % 10 == 0):
            if time.time() - t0 > max_time:
                break
        T = T0 + (T1 - T0) * (it / max(1, n_iters - 1))
        _, grad = get_grad(h, T)
        m_arr = beta1 * m_arr + (1 - beta1) * grad
        v_arr = beta2 * v_arr + (1 - beta2) * grad * grad
        bc1 = 1.0 - beta1 ** (it + 1)
        bc2 = 1.0 - beta2 ** (it + 1)
        h = project(h - lr * (m_arr / bc1) / (np.sqrt(v_arr / bc2) + eps), target)
        
        if (it + 1) % 300 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 3:
                h = project(best_h + rng.normal(0, 0.03, n), target)
                m_arr = np.zeros(n)
                v_arr = np.zeros(n)
                no_improve = 0
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def exact_refine(h, target, n_iters=2000, lr0=0.005):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        center = n - 1
        o = k_star - center
        grad = np.zeros(n)
        if o == 0:
            grad[:] = 1.0 - 2.0 * h
        elif o > 0:
            grad[o:] += 1.0 - h[:n - o]
            grad[:n - o] -= h[o:]
        else:
            p = -o
            grad[:n - p] += 1.0 - h[p:]
            grad[p:] -= h[:n - p]
        grad *= dx
        lr = lr0 / (1.0 + 0.001 * it)
        h = project(h - lr * grad, target)
        if it % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, target, n_sweeps=5):
    n = len(h)
    best_c5 = compute_c5(h)
    deltas = np.array([-0.04, -0.02, -0.01, 0.01, 0.02, 0.04])
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            l = (j + 1) % n
            best_d = 0.0
            best_val = best_c5
            for d in deltas:
                hj = h[j] + d
                hl = h[l] - d
                if hj < 0 or hj > 1 or hl < 0 or hl > 1:
                    continue
                h[j] = hj
                h[l] = hl
                c5_t = compute_c5(h)
                if c5_t < best_val:
                    best_val = c5_t
                    best_d = d
                h[j] -= d
                h[l] += d
            if best_d != 0.0:
                h[j] += best_d
                h[l] -= best_d
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

            h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h, target, n_iters, T0, T1, lr, seed=0, max_time=None, t0=None):
    n = len(h)
    h = project(h, target)
    rng = np.random.default_rng(seed)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = max(100, n_iters // 30)
    for it in range(n_iters):
        if max_time is not None and t0 is not None and (it % 50 == 0):
            if time.time() - t0 > max_time:
                break
        frac = it / max(1, n_iters - 1)
        T = T0 + (T1 - T0) * frac
        _, grad = get_grad(h, T)
        m_arr = beta1 * m_arr + (1.0 - beta1) * grad
        v_arr = beta2 * v_arr + (1.0 - beta2) * grad * grad
        bc1 = 1.0 - beta1 ** (it + 1)
        bc2 = 1.0 - beta2 ** (it + 1)
        lr_eff = lr * (1.0 - 0.5 * frac)
        h = project(h - lr_eff * (m_arr / bc1) / (np.sqrt(v_arr / bc2) + eps), target)
        if (it + 1) % check_interval == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 4:
                h = project(best_h + rng.normal(0, 0.04, n), target)
                m_arr[:] = 0.0
                v_arr[:] = 0.0
                no_improve = 0
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def exact_refine(h, target, n_iters=5000, lr0=0.003):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        center = n - 1
        o = k_star - center
        grad = np.zeros(n)
        if o == 0:
            grad[:] = 1.0 - 2.0 * h
        elif o > 0:
            grad[o:] += 1.0 - h[:n - o]
            grad[:n - o] -= h[o:]
        else:
            p = -o
            grad[:n - p] += 1.0 - h[p:]
            grad[p:] -= h[:n - p]
        grad *= dx
        lr = lr0 / (1.0 + 0.002 * it)
        h = project(h - lr * grad, target)
        if it % 1000 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, target, n_sweeps=8):
    n = len(h)
    best_c5 = compute_c5(h)
    deltas = np.array([-0.03, -0.015, -0.005, 0.005, 0.015, 0.03])
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            l = (j + 1) % n
            best_d = 0.0
            best_val = best_c5
            for d in deltas:
                hj = h[j] + d
                hl = h[l] - d
                if hj < 0.0 or hj > 1.0 or hl < 0.0 or hl > 1.0:
                    continue
                h[j] = hj
                h[l] = hl
                c5_t = compute_c5(h)
                if c5_t < best_val:
                    best_val = c5_t
                    best_d = d
                h[j] -= d
                h[l] += d
            if best_d != 0.0:
                h[j] += best_d
                h[l] -= best_d
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

def snap_and_reproject(h, target, thresh=0.2):
    h2 = h.copy()
    h2[h2 < thresh] = 0.0
    h2[h2 > 1.0 - thresh] = 1.0
    h2 = project(h2, target)
    return h2

def interpolate_up(h, n_hi):
    n_old = len(h)
    target_hi = n_hi / 2.0
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_hi, endpoint=False)
    h_hi = np.interp(x_new, x_old, h, period=None)
    h_hi = project(h_hi, target_hi)
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = project(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Adam at current resolution
    if elapsed() < budget_s - 400:
        h_best, c5_p1 = adam_optimize(h_best, target, 5000, 5, 200, 0.06,
                                       seed, budget_s - 350, t0)
        if c5_p1 < c5_best:
            c5_best = c5_p1

    # Phase 2: Interpolate to n=560 and optimize
    n_hi = 560
    if elapsed() < budget_s - 300:
        target_hi = n_hi / 2.0
        h_hi = interpolate_up(h_best, n_hi)
        c5_hi = compute_c5(h_hi)
        h_hi, c5_hi = adam_optimize(h_hi, target_hi, 30000, 10, 500, 0.04,
                                     seed + 100, budget_s - 200, t0)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n_hi
            target = target_hi

    # Phase 3: Try binary-ish restart at higher res
    if elapsed() < budget_s - 200:
        n_use = max(n, 560)
        target_use = n_use / 2.0
        if n_use != n:
            h_base = interpolate_up(h_best, n_use)
            target = target_use
            n = n_use
        else:
            h_base = h_best.copy()
        h_bin = (h_base > 0.45).astype(float)
        h_bin = project(h_bin, target)
        h_bin, c5_bin = adam_optimize(h_bin, target, 10000, 20, 300, 0.03,
                                       seed + 200, budget_s - 150, t0)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin

    # Phase 4: Exact max refine
    if elapsed() < budget_s - 100:
        h_best, c5_p4 = exact_refine(h_best, target, 5000)
        if c5_p4 < c5_best:
            c5_best = c5_p4

    # Phase 5: Snap to binary + reoptimize
    if elapsed() < budget_s - 80:
        h_snap = snap_and_reproject(h_best, target, 0.25)
        c5_snap = compute_c5(h_snap)
        if c5_snap < c5_best:
            c5_best = c5_snap
            h_best = h_snap
        h_snap, c5_snap2 = adam_optimize(h_snap, target, 5000, 20, 200, 0.02,
                                          seed + 300, budget_s - 60, t0)
        if c5_snap2 < c5_best:
            c5_best = c5_snap2
            h_best = h_snap

    # Phase 6: Coordinate descent polish
    if elapsed() < budget_s - 20:
        h_best, c5_cd = coord_descent(h_best, target, 10)
        if c5_cd < c5_best:
            c5_best = c5_cd

    # Phase 7: Final fine CD
    if elapsed() < budget_s - 5:
        h_best, c5_cd2 = coord_descent(h_best, target, 5)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = np.clip(h_best, 0.0, 1.0)
    return (h_best, c5_final, len(h_best))
