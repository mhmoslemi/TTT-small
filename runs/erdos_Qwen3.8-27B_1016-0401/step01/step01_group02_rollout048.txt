import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GDX = 0.0
_GSC = 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _softobj(h, dx, sc):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    return (m + np.log(np.exp(sc * (c - m)).sum()) / sc) * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _f_slsqp(x):
    return _softobj(np.clip(x, 0.0, 1.0), _GDX, _GSC)

def _ceq(x):
    return x.sum() - 0.5 * _GN

def _ceqj(x):
    return np.ones(len(x))

def _slsqp(h0, n, dx, sc, mi):
    global _GN, _GDX, _GSC
    _GN, _GDX, _GSC = n, dx, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_f_slsqp, h0, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _c5_grad(h, sc):
    """Gradient of soft-max C5 objective w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    p = e / e.sum()
    grad = np.zeros(n)
    # For each i: grad[i] = dx * sum_kf p[kf] * [(1-h[i-k])*[valid] - h[i+k]*[valid]]
    # where k = kf - (n-1)
    for i in range(n):
        s = 0.0
        for kf in range(2 * n - 1):
            k = kf - (n - 1)
            val = 0.0
            idx1 = i - k
            if 0 <= idx1 < n:
                val += (1.0 - h[idx1])
            idx2 = i + k
            if 0 <= idx2 < n:
                val -= h[idx2]
            if p[kf] > 1e-15:
                s += p[kf] * val
        grad[i] = dx * s
    return grad

def _grad_ls(h, n, tmax, sc, iters=100000):
    """Gradient-guided local search."""
    t0 = time.time()
    tgt = 0.5 * n
    dx = 2.0 / n
    best = h.copy()
    bc = _c5(best)
    cur = h.copy()
    cc = bc
    lr = 0.1
    for it in range(iters):
        if time.time() - t0 > tmax:
            break
        g = _c5_grad(cur, sc)
        # Projected gradient step
        step = cur - lr * g
        step = _proj(step, n)
        sc5 = _c5(step)
        if sc5 < cc:
            cur = step
            cc = sc5
            if cc < bc:
                best = cur.copy()
                bc = cc
        else:
            # Reduce learning rate occasionally
            if it % 100 == 0:
                lr *= 0.95
            # Or do a random pair move as fallback
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                dv = 0.1 * np.random.uniform(0.5, 1.0) * (1.0 if g[i] > g[j] else -1.0)
                trial = cur.copy()
                trial[i] = np.clip(trial[i] + dv, 0.0, 1.0)
                trial[j] = np.clip(trial[j] - dv, 0.0, 1.0)
                trial = _proj(trial, n)
                tc = _c5(trial)
                if tc < cc:
                    cur = trial
                    cc = tc
                    if cc < bc:
                        best = cur.copy()
                        bc = cc
    return best, bc

import numpy as np
from scipy.optimize import minimize
import time

_GN, _GDX, _GSC = 0, 0.0, 50.0

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _softobj(h, dx, sc):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    return (m + np.log(np.exp(sc * (c - m)).sum()) / sc) * dx

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _f_slsqp(x):
    return _softobj(np.clip(x, 0.0, 1.0), _GDX, _GSC)

def _ceq(x):
    return x.sum() - 0.5 * _GN

def _ceqj(x):
    return np.ones(len(x))

def _slsqp(h0, n, dx, sc, mi):
    global _GN, _GDX, _GSC
    _GN, _GDX, _GSC = n, dx, sc
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_f_slsqp, h0, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)

def _grad(h, sc):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    p = e / e.sum()
    oh = 1.0 - h

        if p[kf] < 1e-14:

        lag = kf - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += p[kf] * oh[lo1 - lag:hi1 - lag + 1]
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= p[kf] * h[lo2 + lag:hi2 + lag + 1]
    return g * dx

        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t_k = t_new
        c = _c5(x)
        if c < bc:

            bc = c

    return best, bc
