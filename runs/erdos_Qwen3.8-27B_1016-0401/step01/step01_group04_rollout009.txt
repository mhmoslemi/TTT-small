def normalize_to_target(h, target_sum):
    """Clip and adjust to hit exact target sum."""
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-10:

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def slsqp_smooth(h, n, T, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = corr.max()
        return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return np.clip(res.x, 0, 1)
    except:
        return h.copy()
