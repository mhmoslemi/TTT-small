                if h[zi] != h[oi]:
                    h[zi], h[oi] = h[oi], h[zi]
        cur = compute_c5(h, dx)
        T = t0
        while T > tend:
            if time.time() > t_limit:
                return best_h, best_c5
            zi = rng.randint(n)
            oi = rng.randint(n)
            if h[zi] == h[oi]:
                continue
            h[zi], h[oi] = h[oi], h[zi]
            new_c = compute_c5(h, dx)
            d = new_c - cur
            if d < 0 or rng.random() < math.exp(-d / T):
                cur = new_c
                if cur < best_c5 - 1e-15:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                h[zi], h[oi] = h[oi], h[zi]
            T *= cool
    return best_h, best_c5

def sa_double_swap(h0, dx, t_limit, n_restarts=6, t0=0.12, tend=0.0008, cool=0.997, rng=None):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    if rng is None:
        rng = np.random.RandomState(7)
    for rs in range(n_restarts):
        if time.time() > t_limit:
            break
        h = best_h.copy()
        if rs > 0:
            for _ in range(rs * 3):
                z0, o0 = rng.randint(n), rng.randint(n)
                z1, o1 = rng.randint(n), rng.randint(n)
                ok = (h[z0] == 0 and h[o0] == 1 and h[z1] == 0 and h[o1] == 1)
                if ok and z0 != z1 and o0 != o1:
                    h[z0], h[o0] = 1.0, 0.0
                    h[z1], h[o1] = 1.0, 0.0
        cur = compute_c5(h, dx)
        T = t0
        while T > tend:
            if time.time() > t_limit:
                return best_h, best_c5
            z0, o0 = rng.randint(n), rng.randint(n)
            z1, o1 = rng.randint(n), rng.randint(n)
            if not (h[z0] == 0 and h[o0] == 1 and h[z1] == 0 and h[o1] == 1):
                continue
            if z0 == z1 or o0 == o1 or z0 == o0 or z1 == o1:
                continue
            h[z0], h[o0] = 1.0, 0.0
            h[z1], h[o1] = 1.0, 0.0
            new_c = compute_c5(h, dx)
            d = new_c - cur
            if d < 0 or rng.random() < math.exp(-d / T):
                cur = new_c
                if cur < best_c5 - 1e-15:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                h[z0], h[o0] = 0.0, 1.0
                h[z1], h[o1] = 0.0, 1.0
            T *= cool
    return best_h, best_c5

def greedy_polish(h0, dx, t_limit):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best, dx)
    improved = True
    while improved and time.time() < t_limit:
        improved = False
        zeros = list(np.where(best == 0)[0])
        ones = list(np.where(best == 1)[0])
        for zi in zeros:
            if time.time() > t_limit:
                return best, best_c5
            for oi in ones:
                best[zi], best[oi] = 1.0, 0.0
                c = compute_c5(best, dx)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    improved = True
                    break
                best[zi], best[oi] = 0.0, 1.0
    return best, best_c5

def init_patterns(n, rng):
    pats = []
    phi = (1 + math.sqrt(5)) / 2.0
    pats.append((np.sin(np.arange(n) * phi * math.pi) > 0).astype(float))
    pats.append((np.arange(n) % 2 == 0).astype(float))
    for k in [5, 7, 11, 13]:
        pats.append(((np.arange(n) * k) % (n + 1) < (n + 1) / 2).astype(float))
    pats.append(rng.uniform(0, 1, n).astype(float))
    return pats

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = project_feasible(h_init, n_init / 2.0).copy()
    best_c5 = compute_c5(best_h, 2.0 / n_init)
    best_n = n_init

    res_list = [48, 46, 96, 92, 192]

    for n in res_list:
        if time.time() > t_start + budget_s * 0.92:
            break
        dx = 2.0 / n
        ts = n / 2.0
        t_phase = min(t_start + budget_s * 0.92, time.time() + max(5.0, (t_start + budget_s * 0.92 - time.time()) * 0.25))

        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_old, h_init)
        h_bin = to_binary(h_interp, int(ts))

        inits = [h_bin]
        for p in init_patterns(n, rng)[:4]:
            inits.append(to_binary(p, int(ts)))

        for h_try in inits:
            if time.time() > t_phase:
                break
            c_try = compute_c5(h_try, dx)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
                best_n = n

        t_sa1 = min(t_phase, time.time() + max(3.0, (t_phase - time.time()) * 0.5))
        h_best, c_best = sa_binary(h_bin, dx, t_sa1, n_restarts=12, t0=0.06, tend=0.0003, cool=0.9985, rng=np.random.RandomState(seed + n))
        if c_best < best_c5:
            best_c5 = c_best
            best_h = h_best.copy()
            best_n = n

        if time.time() < t_phase:
            t_sa2 = min(t_phase, time.time() + max(2.0, (t_phase - time.time()) * 0.5))
            h_d, c_d = sa_double_swap(h_best, dx, t_sa2, n_restarts=6, t0=0.1, tend=0.0005, cool=0.997, rng=np.random.RandomState(seed + 100 + n))
            if c_d < best_c5:
                best_c5 = c_d
                best_h = h_d.copy()
                best_n = n

        if time.time() < t_start + budget_s * 0.92 and n <= 96:
            t_pol = min(t_start + budget_s * 0.92, time.time() + 8.0)
            h_g, c_g = greedy_polish(best_h, dx, t_pol)
            if c_g < best_c5:
                best_c5 = c_g
                best_h = h_g.copy()
                best_n = n

    n_use = best_n
    dx_use = 2.0 / n_use
    ts_use = n_use / 2.0
    t_remain = t_start + budget_s * 0.95 - time.time()
    n_extra = max(1, int(t_remain / 15.0))

    for i in range(min(n_extra, 6)):
        if time.time() > t_start + budget_s * 0.95:
            break
        h_rand = rng.uniform(0, 1, n_use)
        h_r = to_binary(h_rand, int(ts_use))
        c_r = compute_c5(h_r, dx_use)
        t_sub = min(t_start + budget_s * 0.95, time.time() + max(3.0, (t_start + budget_s * 0.95 - time.time()) * 0.3))
        h_s, c_s = sa_binary(h_r, dx_use, t_sub, n_restarts=4, t0=0.08, tend=0.001, cool=0.998, rng=np.random.RandomState(seed + 500 + i))
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()
            best_n = n_use

    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    s_cur = np.sum(best_h)
    if abs(s_cur - best_n / 2.0) > 1e-10:
        best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    return best_h, best_c5, best_n
