def _make_periodic(n, p, rng):
    k = max(1, p // 2)
    block = np.zeros(p)
    ones = rng.choice(p, size=k, replace=False)
    block[ones] = 1.0
    h = np.tile(block, n // p + 1)[:n]
    return h

def _make_anti(n, p, rng):
    # Alternating blocks
    half = p // 2
    block = np.concatenate([np.ones(half), np.zeros(p - half)])
    h = np.tile(block, n // p + 1)[:n]
    # Random shift
    shift = int(rng.integers(0, p))
    h = np.roll(h, shift)
    return h
