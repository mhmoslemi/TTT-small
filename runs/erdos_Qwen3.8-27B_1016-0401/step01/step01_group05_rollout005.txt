def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_smooth_max(corr, alpha):
    m = np.max(corr)
    vals = np.clip(alpha * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(vals))) / alpha

def lse_obj(h, target, alpha, reg):
    corr = compute_corr(h)
    sm = lse_smooth_max(corr, alpha)
    # Small regularization to encourage smoothness
    if reg > 0:
        d = np.diff(h)
        reg_term = reg * np.sum(d**2)
    else:
        reg_term = 0
    return sm + reg_term

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_max(corr, alpha):
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(v))) / alpha

def slsqp_opt(h, n, target, alpha, maxiter=300):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return lse_max(c, alpha)

        obj, h.copy(), method='SLSQP',

        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return make_feasible(res.x, n, target)
