def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def topk_smooth(corr, k, T):
    if k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    topk = corr[idx]
    mc = np.max(topk)
    s = (topk - mc) / T
    return mc + T * np.log(np.mean(np.exp(s)))

def topk_smooth_grad(h, n, k, T):

    else:

    wsum = np.sum(exps) / k

            h[i] += s if rem > 0 else -s
            rem -= s if rem > 0 else -s

    return h
