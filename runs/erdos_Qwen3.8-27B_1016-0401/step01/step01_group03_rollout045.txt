def compute_c5(h, dx):

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    proj = np.clip(h - mid, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:

        else:
            mask = proj > 1e-10
        if mask.sum() > 0:

    return np.clip(proj, 0.0, 1.0)

def smooth_max_value_corr(h, dx, beta):

        while time.time() - t_start < (t_limit - time.time()) / (n_restarts - 1 + 1) * 0.8:
            # Random swap
            zi = zeros[np.random.randint(len(zeros))]
            oi = ones[np.random.randint(len(ones))]
            h[zi] = 1.0
            h[oi] = 0.0
            new_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
            delta = new_corr - corr
            if delta <= 0 or np.random.random() < np.exp(-delta / T):
                corr = new_corr
                # Update sets
                zi_idx = np.where(zeros == zi)[0][0]
                zeros[zi_idx] = oi
                oi_idx = np.where(ones == oi)[0][0]
                ones[oi_idx] = zi
            else:
                h[zi] = 0.0
                h[oi] = 1.0
            steps += 1
            if steps % 1000 == 0:
                T *= 0.995
            if T < 0.01:
                T = 0.01
        
        if corr < best_corr:
            best_corr = corr
            best_c5 = corr * dx
            best_h = h.copy()
    
    return best_h, best_c5
