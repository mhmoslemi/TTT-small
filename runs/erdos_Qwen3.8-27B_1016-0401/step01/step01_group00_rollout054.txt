import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_fn(x, n, T, tgt, lam):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = np.clip(T * (corr - mc), -500.0, 500.0)
    se = float(np.sum(np.exp(sh)))
    base = (mc + np.log(se) / T) * (2.0 / n)
    pen = lam * (x.sum() - tgt) ** 2
    return base + pen

def grad_fn(x, n, T, tgt, lam):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = np.clip(T * (corr - mc), -500.0, 500.0)
    e = np.exp(sh)
    sw = e / np.sum(e)
    ctr = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[ctr + o]
        if w < 1e-12:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n - o])
            g[:n - o] += -w * x[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - x[p:])
            g[p:] += -w * x[:n - p]
    g *= dx
    g += 2.0 * lam * (x.sum() - tgt)
    return g

def fix_c(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        order = np.argsort(h)[::-1] if d > 0 else np.argsort(h)
        for i in order:
            r = (1.0 - h[i]) if d > 0 else h[i]
            s = min(abs(d), r)
            if d > 0:
                h[i] += s
            else:
                h[i] -= s
            if abs(tgt - h.sum()) < 1e-15:
                break
    return h

def offset_grad(h, o, n):
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n - o]
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - h[p:]
        g[p:] += -h[:n - p]
    return g

def subgrad_ref(h, tgt, tb):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    c5c = compute_c5(h)
    a = 0.01
    st = 0
    for _ in range(50000):
        if time.time() - t0 > tb:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        ks = int(np.argmax(corr))
        o = ks - (n - 1)
        g = offset_grad(h, o, n)
        ht = fix_c(h - a * g, tgt)
        c5t = compute_c5(ht)
        if c5t < c5c - 1e-15:
            h = ht
            c5c = c5t
            st = 0
            a = min(a * 1.03, 0.02)
        else:
            st += 1
            a *= 0.85
            if st > 300 or a < 1e-6:
                break
    return h, c5c

def pixel_search(h, tgt, tb):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    c5c = compute_c5(h)
    vals = [0.0, 0.5, 1.0]
    while time.time() - t0 < tb:
        imp = False
        for i in range(n):
            orig = h.copy()
            hi = h[i]
            for v in vals:
                if abs(hi - v) < 1e-7:
                    continue
                h[i] = v
                ht = fix_c(h.copy(), tgt)
                c5t = compute_c5(ht)
                if c5t < c5c - 1e-14:
                    h = ht
                    c5c = c5t
                    imp = True
                    break
            if not imp:
                h[:] = orig
        if not imp:
            break
    return h, c5c

def pair_search(h, tgt, tb):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    c5c = compute_c5(h)
    while time.time() - t0 < tb:
        imp = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-7:
                    continue
                orig = (h[i], h[j])
                h[i], h[j] = h[j], h[i]
                c5t = compute_c5(h)
                if c5t < c5c - 1e-14:
                    c5c = c5t
                    imp = True
                else:
                    h[i], h[j] = orig
        if not imp:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0
    n = len(h_best)
    tgt = n / 2.0
    h_best = fix_c(h_best, tgt)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    for T in [3, 5, 10, 20, 40, 80, 150, 250]:
        if time.time() - t0 > budget_s - 200:
            break
        try:
            res = minimize(obj_fn, h_best.copy(), args=(n, T, tgt, 1000.0),
                           method='L-BFGS-B', jac=grad_fn,
                           bounds=bounds,
                           options={'maxiter': 20000, 'ftol': 1e-20})
            h_new = fix_c(res.x, tgt)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h2, c52 = subgrad_ref(h_best, tgt, min(rem - 100, 100))
        if c52 < c5_best:
            c5_best = c52
            h_best = h2

    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h3, c53 = pixel_search(h_best, tgt, min(rem - 60, 120))
        if c53 < c5_best:
            c5_best = c53
            h_best = h3

    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h3p, c53p = pair_search(h_best, tgt, min(rem - 40, 60))
        if c53p < c5_best:
            c5_best = c53p
            h_best = h3p

    for T in [400, 800]:
        if time.time() - t0 > budget_s - 60:
            break
        try:
            res = minimize(obj_fn, h_best.copy(), args=(n, T, tgt, 1000.0),
                           method='L-BFGS-B', jac=grad_fn,
                           bounds=bounds,
                           options={'maxiter': 10000, 'ftol': 1e-20})
            h_new = fix_c(res.x, tgt)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h5, c55 = subgrad_ref(h_best, tgt, min(rem - 15, 25))
        if c55 < c5_best:
            c5_best = c55
            h_best = h5

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        h6, c56 = pixel_search(h_best, tgt, min(rem - 5, 25))
        if c56 < c5_best:
            c5_best = c56
            h_best = h6

    rem = budget_s - (time.time() - t0)

        if c57 < c5_best:
            c5_best = c57
            h_best = h7

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
