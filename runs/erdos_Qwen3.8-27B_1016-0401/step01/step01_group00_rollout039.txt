        + (1-h[j-m]) if j-m in [0,n-1]
        - h[j+m] if j+m in [0,n-1]

  delta[m] = -(1-h[i-m])*[i-m valid] + h[i+m]*[i+m valid] + (1-h[j-m])*[j-m valid] - h[j+m]*[j+m valid]

where [x valid] means 0 ≤ x ≤ n-1.

This is O(1) per m, so O(n) total for all shifts.

                best_c5 = c5_cur
                best_h = h_cur.copy()
    return best_h, best_c5

def binary_local_search(h, target, max_sweeps, time_fn, budget_left):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h < 0.5] = 0.0
    h = proj_feasible(h, target)
    if h.sum() != target:
        return h, compute_c5(h)
    corr = np.correlate(h, 1 - h, mode="full")
    c5_arr = float(np.max(corr) * dx)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    rem = np.zeros((n, 2 * n - 1))
    s_range = np.arange(-(n - 1), n)
    for p in range(n):
        i1 = p - s_range
        i2 = p + s_range
        v1 = (i1 >= 0) & (i1 < n)
        v2 = (i2 >= 0) & (i2 < n)
        rem[p] = -(1.0 - h[np.clip(i1, 0, n - 1)]) * v1 + h[np.clip(i2, 0, n - 1)] * v2
    for _ in range(max_sweeps):
        if time_fn() < 0:
            break
        best_improve = 1e-14
        best_pair = None
        ones_arr = np.array(ones)
        zeros_arr = np.array(zzeros) if False else np.array(zeros)
        cur_max = np.max(corr)
        for ii in range(len(ones_arr)):
            i = ones_arr[ii]
            new_maxs = np.max(corr[None, :] + rem[i][None, :] - rem[zeros_arr], axis=1)
            min_idx = int(np.argmin(new_maxs))
            if new_maxs[min_idx] < cur_max - best_improve:
                best_improve = cur_max - new_maxs[min_idx]
                best_pair = (i, zeros_arr[min_idx])
        if best_pair is None:
            break
        i, j = best_pair
        corr = corr + rem[i] - rem[j]
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        c5_arr = float(np.max(corr) * dx)
    return h, c5_arr

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            cnt = int(m.sum())
            if cnt == 0:
                break
            h[m] += d / cnt
        else:
            m = h > 0.0
            cnt = int(m.sum())
            if cnt == 0:
                break
            h[m] += d / cnt
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = (corr - mc) * T
    e = np.exp(shifted)
    se = float(np.sum(e))
    sw = e / se
    dx = 2.0 / n
    f = (mc + np.log(se) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-9:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] -= w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    return f, grad * dx

def _c_fun(x):
    return x.sum() * 2.0 - len(x)

def _c_jac(x):
    return np.ones(len(x))

def _slsqp_obj(x, n, T):
    return obj_grad(x, n, T)[0]

def _slsqp_jac(x, n, T):
    return obj_grad(x, n, T)[1]

def projected_gradient(h0, n, target, Ts, max_iters, time_fn):
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for T in Ts:
        if time_fn() <= 0:
            break
        mom = np.zeros(n)
        step = 0.05
        g_prev = None
        x_prev = None
        for _ in range(max_iters):
            if time_fn() <= 0:
                break
            f, g = obj_grad(h_cur, n, T)
            g_p = g - g.mean()
            mom = 0.85 * mom + g_p
            if g_prev is not None and x_prev is not None:
                dxv = h_cur - x_prev
                dgv = g_p - g_prev
                sv = float(dxv @ dgv)
                yv = float(dgv @ dgv)
                if sv > 1e-12 and yv > 1e-12:
                    step = float(np.clip(sv / yv, 0.002, 0.2))
            h_new = proj_feasible(h_cur - step * mom, target)
            f_new, _ = obj_grad(h_new, n, T)
            if f_new > f + 0.5 * step * float(mom @ (h_new - h_cur)):
                step *= 0.5
                h_new = proj_feasible(h_cur - step * mom, target)
            x_prev = h_cur
            g_prev = g_p
            h_cur = h_new
            c5c = compute_c5(h_cur)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_cur.copy()
    return best_h, best_c5
