import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1-h, mode="full")) * 2.0/n)

def project_polytope(h, target):
    lo, hi = -h.max(), 1.0 - h.min()
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.clip(h+mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5*(lo+hi), 0, 1)

def _sm_obj(h, n, T, mu):
    c = np.correlate(h, 1-h, mode="full")
    mc = c.max()
    sh = np.clip(T*(c-mc), -50, 50)
    return (mc + np.log(np.exp(sh).sum())/T)*2.0/n + mu*(h.sum()-n/2.0)**2

def _sm_grad(h, n, T, mu):
    c = np.correlate(h, 1-h, mode="full")
    mc = c.max()
    sh = np.clip(T*(c-mc), -50, 50)
    e = np.exp(sh); w = e/e.sum()
    ct = n-1; dx = 2.0/n
    g = np.zeros(n)
    for o in range(-(n-1), n):
        wi = w[ct+o]
        if wi < 1e-13: continue
        if o == 0:
            g += wi*(1.0-2.0*h)
        elif o > 0:
            g[:n-o] += wi*(1.0-h[o:])
            g[o:] -= wi*h[:n-o]
        else:
            p = -o
            g[p:] += wi*(1.0-h[:n-p])
            g[:n-p] -= wi*h[p:]
    g *= dx
    if mu != 0:
        g += 2.0*mu*(h.sum()-n/2.0)
    return g

def _lbfgs(h, n, T, mu, maxit=5000, ftol=1e-16):
    b = [(0.0, 1.0)]*n
    try:
        r = minimize(_sm_obj, h, method='L-BFGS-B', jac=_sm_grad,
                    bounds=b, args=(n, T, mu),
                    options={'maxiter': maxit, 'ftol': ftol, 'gtol': 1e-10})
        return project_polytope(r.x, n/2.0)
    except Exception:
        return h

def _nesterov_polish(h, n, T, steps=2000, lr0=0.5, beta=0.9):
    target = n/2.0
    h = h.copy(); v = np.zeros(n); c5_cur = compute_c5(h)
    for k in range(steps):
        g = _sm_grad(h, n, T, 0.0)
        lr = lr0 / (1.0 + 0.01*k)
        v = beta*v + (1.0-beta)*g
        h_new = project_polytope(h - lr*v, target)
        if np.abs(h_new-h).max() < 1e-14:
            break
        h = h_new
    return h
