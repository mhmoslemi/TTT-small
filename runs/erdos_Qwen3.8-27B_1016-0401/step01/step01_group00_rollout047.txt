import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smax_corr(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx, e / e.sum()

def smax_grad(h, sw, dx):
    n = len(h)
    c = n - 1
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[c + o] * dx
        if w < 1e-11:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[:n - o] += w * (1.0 - h[o:])
            g[o:] -= w * h[:n - o]
        else:
            p = -o
            g[p:] += w * (1.0 - h[:n - p])
            g[:n - p] -= w * h[p:]
    return g

def smax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    obj, sw = smax_corr(h, T)
    g = smax_grad(h, sw, dx)
    return obj, g

def proj(h, S):
    lo, hi = -h.max() - 0.1, 1.0 - h.min() + 0.1
    for _ in range(55):
        m = (lo + hi) / 2.0
        if np.sum(np.clip(h + m, 0.0, 1.0)) < S:
            lo = m
        else:
            hi = m
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def gd(h0, T, n_iter, lr0, t_limit, t_start):
    h = h0.copy()
    n = len(h)
    S = n / 2.0
    lr = lr0
    bobj = smax_corr(h, T)[0]
    stalling = 0
    for it in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        obj, g = smax_obj_grad(h, T)
        hn = proj(h - lr * g, S)
        o_new = smax_corr(hn, T)[0]
        if o_new < bobj - 1e-16:
            h, bobj, stalling = hn, o_new, 0
            lr = min(lr * 1.08, 0.6)
        else:
            hr = lr
            for _ in range(18):
                hn = proj(h - hr * g, S)
                o_new = smax_corr(hn, T)[0]
                if o_new < bobj:
                    h, bobj, stalling = hn, o_new, 0
                    lr = min(hr * 1.1, 0.6)
                    break
                hr *= 0.4
            else:
                stalling += 1
                if stalling > 8:
                    break
                break
    return h

def cd_polish(h, n_rounds, t_limit, t_start):

    S = n / 2.0
    obj = compute_c5(h)
    rng = np.random.default_rng(999)
    for rnd in range(n_rounds):
        if time.time() - t_start > t_limit:
            break
        imp = False
        for j in rng.permutation(n):
            if time.time() - t_start > t_limit:
                break
            for d in [0.04, -0.04, 0.12, -0.12, 0.25, -0.25, 0.5, -0.5]:
                hn = h.copy()
                hn[j] += d
                hn = proj(hn, S)
                o = compute_c5(hn)
                if o < obj - 1e-14:
                    h, obj, imp = hn, o, True
                    break
        if not imp and rnd > 0:
            break
    return h, obj

def downsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = (np.arange(n_new) + 0.5) * n_old / n_new
    return np.interp(x_new, x_old, h)
