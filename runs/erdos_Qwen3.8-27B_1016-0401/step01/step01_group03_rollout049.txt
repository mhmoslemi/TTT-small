    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0/beta, 0.0)
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)

            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

    n = len(h)

    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:

        else:

    best_c5 = compute_c5(h, dx)
    v = np.zeros(n)
    
    betas = np.geomspace(beta_start, beta_end, n_beta_steps)

            break
        lr = 1.0 / (beta * dx + 0.1)  # initial learning rate
        momentum = 0.9
        stall = 0

                return best_h, best_c5
            _, grad = smooth_obj_grad(h, dx, beta)

            v = momentum * v + (1 - momentum) * grad

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = compute_corr(h)
    max_c = np.max(corr)
    shifted = np.clip(corr - max_c, -50.0, 0.0)
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

def project_feasible(h, target_sum):

        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
