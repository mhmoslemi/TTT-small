def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single_lag(h, lag, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx] += (1.0 - h[idx + lag]) * dx
        g[idx + lag] += (-h[idx]) * dx
    else:
        mm = -lag
        idx = np.arange(mm, n)
        g[idx] += (1.0 - h[idx - mm]) * dx
        g[idx - mm] += (-h[idx]) * dx
    return g

def soft_max_grad(h, alpha, top_k=5):

    corr = compute_corr_vals(h)
    # Get top-k lags

    weights = np.exp(alpha * (corr[idx_top] - max_val))
    weights /= weights.sum()
    grad = np.zeros(n)

        lag = ci - (n - 1)
        grad += w * grad_single_lag(h, lag, n)
