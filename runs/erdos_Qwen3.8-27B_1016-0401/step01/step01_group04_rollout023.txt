def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, S):
    h_b = (h >= 0.5).astype(float)
    target = int(S)
    ones = int(h_b.sum())
    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:
        zero_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

def init_pattern(n, seed):

    rng = np.random.RandomState(seed)
    S = n // 2
    # Golden ratio based placement
    golden = (1 + np.sqrt(5)) / 2
    positions = set()
    for k in range(S):
        pos = int(k * golden) % n
        # Ensure uniqueness
        while pos in positions:
            pos = (pos + 1) % n
        positions.add(pos)
    h = np.zeros(n)
    for p in positions:
        h[p] = 1
    return h

def init_spaced(n, seed):

    rng = np.random.RandomState(seed)
    S = n // 2

    jitter = rng.uniform(-n/(2*S), n/(2*S), S)

    positions = np.unique(positions)

    while len(positions) < S:

        positions = np.append(positions, candidates[rng.randint(len(candidates))])
    positions = positions[:S]
    h = np.zeros(n)
    h[positions] = 1
    return h

            perm = rng.permutation(len(one_idx))
            for pi in perm:
                if time.time() > t_deadline:
                    return best_h, best_c5
                a = one_idx[pi]
                for b in zero_idx:
                    h[a], h[b] = h[b], h[a]
                    c5_new = compute_c5(h)
                    if c5_new < c5 - 1e-12:
                        c5 = c5_new
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[a], h[b] = h[b], h[a]
                
        if time.time() >= t_deadline:
            break

        stall_count += 1
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        n_pert = pert_size
        for _ in range(n_pert):
            a = one_idx[rng.randint(len(one_idx))]
            b = zero_idx[rng.randint(len(zero_idx))]
            h[a], h[b] = h[b], h[a]
        c5 = compute_c5(h)

        if stall_count > 20:
            pert_size = min(pert_size + 1, 18)
            stall_count = 0
            if rng.rand() < 0.08:
                for _ in range(5):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if h[a] != h[b]:
                        h[a], h[b] = h[b], h[a]
                c5 = compute_c5(h)

        if c5 <= best_c5 + 1e-12:
            pert_size = max(2, pert_size - 1)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = golden_pattern(70, seed)

    n_init = len(h_init)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    n_list = sorted(set([n_init] + [64, 66, 68, 72, 74, 76, 80, 84, 90]))
    n_list = [n for n in n_list if n % 2 == 0 and 48 <= n <= 120]

    time_ihs = budget_s * 0.55
    time_others = budget_s * 0.30
    time_final = budget_s * 0.10

    # Phase 1: ILS on n_init with multiple starts
    t1_end = t0 + time_ihs
    starts_init = []
    starts_init.append(to_binary(h_init, n_init / 2.0))
    starts_init.append(golden_pattern(n_init, seed + 7))
    starts_init.append(random_binary(n_init, seed + 13))
    starts_init.append(random_binary(n_init, seed + 999))
    starts_init.append(golden_pattern(n_init, seed + 42))

    t_per_start = time_ihs / len(starts_init)
    for si, h_start in enumerate(starts_init):
        t_dl = min(t1_end - 2, t0 + (si + 1) * t_per_start)
        if time.time() > t_dl - 5:
            break
        h_b, c5_b = ils_binary(h_start, n_init, rng, t_dl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_init

    # Phase 2: ILS on other n values
    t2_end = t0 + time_ihs + time_others
    time_each_n = max(10, time_others / max(1, len(n_list) - 1))
    for n in n_list:
        if n == n_init:
            continue
        if time.time() > t2_end - 5:
            break
        h_interp = interpolate_to_n(h_init, n_init, n)
        h_bin = to_binary(h_interp, n / 2.0)
        t_dl = min(t2_end - 2, time.time() + time_each_n)
        h_b, c5_b = ils_binary(h_bin, n, rng, t_dl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
        h_rand = random_binary(n, seed + n * 31)
        t_dl2 = min(t2_end - 2, time.time() + time_each_n * 0.4)
        h_b2, c5_b2 = ils_binary(h_rand, n, rng, t_dl2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = n

    # Phase 3: Final intensive ILS on the best
    if best_h is not None and time.time() < t0 + budget_s * 0.92:
        t_dl_final = t0 + budget_s * 0.95
        h_f, c5_f = ils_binary(best_h, best_n, rng, t_dl_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

        # One more perturbation cycle
        if time.time() < t0 + budget_s * 0.97:
            h_pert = best_h.copy()
            for _ in range(rng.randint(5, 12)):
                a, b = rng.randint(best_n, size=2)
                if h_pert[a] != h_pert[b]:
                    h_pert[a], h_pert[b] = h_pert[b], h_pert[a]
            h_f2, c5_f2 = ils_binary(h_pert, best_n, rng, t0 + budget_s * 0.98)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    if best_h is None:
        best_h = to_binary(h_init, n_init / 2.0)
        best_n = n_init
        best_c5 = compute_c5(best_h)

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
