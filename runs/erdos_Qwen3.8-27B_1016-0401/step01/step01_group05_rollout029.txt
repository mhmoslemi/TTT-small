import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def sbx_crossover(p1, p2, eta=20.0, rng=None):
    """Simulated Binary Crossover"""
    if rng is None:
        rng = np.random
    n = len(p1)
    c1 = p1.copy()
    c2 = p2.copy()
    for i in range(n):
        if rng.random() < 0.5:
            if abs(p1[i] - p2[i]) > 1e-14:
                u = rng.random()
                if u <= 0.5:
                    beta = (2 * u) ** (1.0 / (eta + 1))
                else:
                    beta = (1.0 / (2 * (1 - u))) ** (1.0 / (eta + 1))
                c1[i] = 0.5 * ((1 + beta) * p1[i] + (1 - beta) * p2[i])
                c2[i] = 0.5 * ((1 - beta) * p1[i] + (1 + beta) * p2[i])
    return c1, c2

def polynomial_mutate(h, prob=0.1, eta=20.0, rng=None):
    """Polynomial mutation"""
    if rng is None:
        rng = np.random
    h = h.copy()
    n = len(h)
    for i in range(n):
        if rng.random() < prob:
            u = rng.random()
            if u < 0.5:
                delta = (2 * u) ** (1.0 / (eta + 1)) - 1
            else:
                delta = 1 - (2 * (1 - u)) ** (1.0 / (eta + 1))
            h[i] += delta
    return np.clip(h, 0.0, 1.0)

def tournament_select(pop, fitness, k=3, rng=None):
    if rng is None:
        rng = np.random
    n = len(pop)
    idx = rng.choice(n, size=k, replace=False)
    best = idx[np.argmin(fitness[idx])]
    return pop[best].copy()

def ga_optimize(h_init, n, target, budget_s, deadline):
    pop_size = 150
    eta_c = 20.0
    eta_m = 20.0
    mut_prob = 0.1
    tour_size = 3
    
    rng = np.random.RandomState(42)
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    fitness = np.zeros(pop_size)
    
    # First individual is the initial
    pop[0] = h_init.copy()
    fitness[0] = compute_c5(pop[0])
    
    # Generate diverse initial population
    for i in range(1, pop_size):
        if i == 1:
            # Periodic pattern
            k = rng.choice([2, 3, 4, 5, 7, 8, 10, 11])
            pop[i] = 0.5 + 0.4 * np.sin(2 * np.pi * k * np.arange(n) / n)
        elif i == 2:
            # Block pattern
            pop[i] = np.concatenate([np.ones(n//2) * 0.8, np.zeros(n//2) * 0.2])
            rng.shuffle(pop[i])
        elif i == 3:
            # Another sinusoidal
            k = rng.choice([3, 5, 7, 13])
            phase = rng.uniform(0, 2*np.pi)
            pop[i] = 0.5 + 0.35 * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
        elif i == 4:
            # Ramp
            pop[i] = np.linspace(0.2, 0.8, n)
        else:
            # Random with structure
            freq = rng.choice([1, 2, 3, 4, 5, 6, 7])
            phase = rng.uniform(0, 2*np.pi)
            amp = rng.uniform(0.2, 0.45)
            pop[i] = 0.5 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            # Add some random perturbation
            pop[i] += rng.uniform(-0.05, 0.05, n)
        
        pop[i] = make_feasible(pop[i], n, target)
        fitness[i] = compute_c5(pop[i])
    
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    gen = 0
    while time.time() < deadline:
        gen += 1
        new_pop = np.zeros((pop_size, n))
        new_fit = np.zeros(pop_size)
        
        # Keep best individual
        new_pop[0] = best_h
        new_fit[0] = best_c5
        
        # Generate new individuals
        for i in range(1, pop_size):
            # Selection
            p1 = tournament_select(pop, fitness, tour_size, rng)
            p2 = tournament_select(pop, fitness, tour_size, rng)
            
            # Crossover
            if rng.random() < 0.9:
                c1, c2 = sbx_crossover(p1, p2, eta_c, rng)
            else:
                c1, c2 = p1.copy(), p2.copy()
            
            # Mutation
            c1 = polynomial_mutate(c1, mut_prob, eta_m, rng)
            c2 = polynomial_mutate(c2, mut_prob, eta_m, rng)
            
            # Repair
            c1 = make_feasible(c1, n, target)
            c2 = make_feasible(c2, n, target)
            
            # Fit both
            f1 = compute_c5(c1)
            f2 = compute_c5(c2)
            
            new_pop[i] = c1
            new_fit[i] = f1
            
            # Sometimes keep both (elitism variant)
            if i + 1 < pop_size and rng.random() < 0.3:
                new_pop[i+1] = c2
                new_fit[i+1] = f2
                i += 1
        
        # Elitism: keep top 10
        sorted_idx = np.argsort(fitness)
        for k in range(min(10, pop_size)):
            idx = sorted_idx[k]
            # Find a slot in new_pop to replace with this elite
            # Just replace the worst in new_pop
            worst = np.argmax(new_fit)
            if fitness[idx] < new_fit[worst]:
                new_pop[worst] = pop[idx]
                new_fit[worst] = fitness[idx]
        
        pop = new_pop
        fitness = new_fit
        
        new_best = np.argmin(fitness)
        if fitness[new_best] < best_c5:
            best_c5 = fitness[new_best]
            best_h = pop[new_best].copy()
        
        # Adaptive mutation rate
        if gen % 10 == 0:
            mut_prob *= 0.95
            mut_prob = max(mut_prob, 0.03)
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(curr * dx))

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sbx_crossover(p1, p2, eta, rng):
    n = len(p1)
    c1 = p1.copy()
    c2 = p2.copy()
    for i in range(n):
        if rng.random() < 0.5 and abs(p1[i] - p2[i]) > 1e-14:

            if u <= 0.5:

            c2[i] = 0.5*((1-beta)*p1[i] + (1+beta)*p2[i])

    _A = [alpha]

        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(_A[0] * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(vals))) / _A[0]) * dx

        return np.sum(x) - _T[0]

def slsqp_refine(h, n, target, alpha, maxiter=150):
    dx = 2.0 / n
    n_c = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -60, 60)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -60, 60)
        exp_vals = np.exp(vals)
        softmax = exp_vals / np.sum(exp_vals)
        dc = (softmax * alpha) * dx
        grad = np.zeros(n)
        for k in range(n_c):
            lag = k - (n - 1)
            w = dc[k]
            i_s = max(0, -lag)
            i_e = min(n, n - lag)
            grad[i_s:i_e] += w * (1.0 - x[i_s:i_e])
            grad[i_s:i_e] -= w * x[i_s + lag:i_e + lag] if lag > 0 else w * x[i_s:i_e]
        return grad
    def eq_con(x):
        return np.sum(x) - target
    try:

                       bounds=[(0.0, 1.0)] * n,

                       options={'maxiter': maxiter, 'ftol': 1e-14})

            # Quick SLSQP
            for alpha in [50, 200]:
                if time.time() > deadline - 30:
                    break
                h_alt = slsqp_refine(h_alt, n_alt, target_alt, alpha, maxiter=80)
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                n = n_alt
                target = target_alt

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        h_new = slsqp_refine(best_h, n, target, 3000, maxiter=80)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        if time.time() < deadline - 3:
            h_ls, c5_ls = lag_aware_local(best_h, n, target, rng, 10000, deadline - 2, step=0.015)
            if c5_ls < best_c5:
                best_h = h_ls
                best_c5 = c5_ls

    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
