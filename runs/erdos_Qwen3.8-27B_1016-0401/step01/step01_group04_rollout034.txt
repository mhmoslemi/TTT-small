    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h):
    n = len(h)
    target = n // 2
    # Threshold at 0.5, then adjust count
    scores = h.copy()
    h_b = np.zeros(n)
    top = np.argsort(-scores)[:target]
    h_b[top] = 1.0
    return h_b

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def structured_binary(n, rng, kind=0):
    """Generate structured binary patterns"""
    h = np.zeros(n)
    half = n // 2
    if kind == 0:  # evenly spaced
        pos = np.arange(half) * (n // half)
        h[pos[:half]] = 1.0
    elif kind == 1:  # Thue-Morse like
        t = np.zeros(n, dtype=int)
        for i in range(1, n):
            t[i] = t[i >> 1] ^ 1
        h = (t == 0).astype(float)
        # adjust to have exactly half ones
        if h.sum() > half:
            excess = int(h.sum() - half)
            idx = np.where(h == 1)[0]
            h[rng.choice(idx, excess, replace=False)] = 0
        elif h.sum() < half:
            deficit = int(half - h.sum())
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, deficit, replace=False)] = 1
    else:  # random
        h[rng.choice(n, half, replace=False)] = 1.0
    return h

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def to_binary(h):
    n = len(h)
    t = n // 2
    b = np.zeros(n)
    b[np.argsort(-h)[:t]] = 1.0
    return b

def rand_binary(n, rng):

def greedy_binary(n):
    h = np.zeros(n)
    for _ in range(n // 2):
        best_p, best_v = 0, 1e9
        for p in range(n):

                continue
            h[p] = 1.0
            v = compute_c5(h)
            h[p] = 0.0
            if v < best_v:

        h[best_p] = 1.0
    return h
