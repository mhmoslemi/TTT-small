import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_active_max(h, n):
    """Analytical gradient of the max correlation w.r.t. h (subgradient)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    kstar = int(np.argmax(corr))
    lag = kstar - (n - 1)
    k = abs(lag)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - k):
            grad[i + k] += (1.0 - h[i])
            grad[i] -= h[i + k]
    else:
        for i in range(k, n):
            grad[i] += (1.0 - h[i - k])
            grad[i - k] -= h[i]
    return grad * dx

def projected_step(h, n, target, lr, grad):
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h_new)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h_new < 1.0 - 1e-15
        else:
            mask = h_new > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = diff / cnt
        if diff > 0:
            step = min(step, 1.0 - np.max(h_new[mask]) - 1e-15)
        else:
            step = max(step, -np.min(h_new[mask]) - 1e-15)
        h_new[mask] += step
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def gradient_descent(h, n, target, rng, deadline, max_iters=2000):
    """Line-search gradient descent on the active max correlation."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.05 * n / (n * n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = gradient_active_max(h, n)
        # Adaptive line search
        step_accepted = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = projected_step(h, n, target, trial_lr, grad)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h):
                h = h_try
                step_accepted = True
                if c5_try < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_try
                lr = min(lr * 1.1, 0.1)
                break
        if not step_accepted:
            lr *= 0.6
            if lr < 1e-6:
                break
    return best_h, best_c5

def simulated_annealing(h, n, target, rng, deadline, t0=0.04, cooling=0.9998):
    """Multi-coordinate simulated annealing for global exploration."""
    current = h.copy()
    current_c5 = compute_c5(current)
    best = h.copy()
    best_c5 = current_c5
    T = t0
    stall_count = 0
    best_at_stall = current_c5
    
    while time.time() < deadline:
        # Perturb 2-5 coordinates
        nc = rng.choice([2, 3, 4, 5], p=[0.35, 0.3, 0.2, 0.15])
        idx = rng.choice(n, size=nc, replace=False)
        mags = rng.normal(0, 0.15, size=nc)
        new_h = current.copy()
        new_h[idx] += mags
        new_h = make_feasible(new_h, n, target)
        
        new_c5 = compute_c5(new_h)
        
        if new_c5 <= current_c5:
            current = new_h
            current_c5 = new_c5
            stall_count = 0
        else:
            delta = new_c5 - current_c5
            if delta < T:
                if rng.random() < np.exp(-delta / T):
                    current = new_h
                    current_c5 = new_c5
            else:
                stall_count += 1
        
        if new_c5 < best_c5:
            best = new_h.copy()
            best_c5 = new_c5
        
        T *= cooling
        if T < 0.001:
            T = 0.001

            current = best + rng.normal(0, 0.04, n)
            current = make_feasible(current, n, target)
            current_c5 = compute_c5(current)
            stall_count = 0
            T = max(T, 0.02)

                current_c5 = c5_g
            if c5_g < best_c5:
                best = h_g.copy()
                best_c5 = c5_g
    
    return best, best_c5

def slsqp_refine(h, n, target, alpha, maxiter=150):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -40, 40)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    
    def con(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def structured_init(n, rng, variant=0):
    """Algorithmic structured initial guesses."""
    x = np.linspace(0, 2, n, endpoint=False)
    if variant == 0:
        h = 0.5 + 0.3*np.sin(3*np.pi*x) + 0.15*np.sin(7*np.pi*x)
    elif variant == 1:
        h = 0.5 + 0.25*np.sin(2*np.pi*x) + 0.2*np.cos(5*np.pi*x) + 0.1*np.sin(9*np.pi*x)
    else:
        # Block pattern: alternating blocks
        block = max(n // 10, 2)
        h = np.tile([0.7, 0.3, 0.5, 0.35, 0.65, 0.45, 0.55, 0.4, 0.6, 0.5], 
                    n // 10 + 1)[:n]
        h += rng.normal(0, 0.1, n)
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, n, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

    t_start = time.time()
