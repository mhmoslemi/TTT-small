    return float(np.max(corr) * dx)

        else:

        h = np.clip(h, 0.0, 1.0)

def slsqp_refine(h, n, target, alpha, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -60, 60)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

        return np.sum(x) - target

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, n, target)
    except:
        return h.copy()

                best_c5 = new_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj_alpha(alpha, target):
    def _obj(x):
        n = len(x)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        vals = np.clip(alpha * (corr - m), -60, 60)
        return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx
    def _con(x):
        return np.sum(x) - target
    return _obj, _con

def slsqp_refine(h, n, target, alpha=500, maxiter=200):
    obj, con = soft_obj_alpha(alpha, target)
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def sa_optimize(h, n, target, rng, deadline, T0=0.004, cool_rate=0.99997):
    T = T0
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    step = 0.05
    acc = 0
    tot = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        d = rng.uniform(-step, step)
        vi = cur[i] + d
        vj = cur[j] - d
        if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
            cur[i] = vi
            cur[j] = vj
        else:
            cur[i] = np.clip(vi, 0.0, 1.0)
            cur[j] = np.clip(vj, 0.0, 1.0)
            cur = make_feasible(cur, n, target)
        new_c5 = compute_c5(cur)
        tot += 1
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            acc += 1
            if new_c5 < best_c5:
                best = cur.copy()
                best_c5 = new_c5
        else:
            delta = new_c5 - cur_c5
            if delta < 0.08 and rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                acc += 1
        if tot % 4000 == 0:
            rate = acc / tot
            if rate > 0.55:
                step = min(step * 1.12, 0.18)
            elif rate < 0.12:
                step = max(step * 0.78, 0.004)
            T = max(T * cool_rate, T0 * 5e-5)
            acc = 0
            tot = 0
    return best, best_c5

def make_sinusoidal(n, freq, amp, phase):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + amp * np.sin(freq * np.pi * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, n, n / 2.0)

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    h += rng.uniform(-0.08, 0.08, n)
    return make_feasible(h, n, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(h_init)

    best_h = h_init.copy()

    # Phase 1: SLSQP from parent with increasing alpha
    for alpha in [80, 200, 500, 1200, 3000]:
        if time.time() > deadline - 250:
            break

        c5 = compute_c5(h_new)
        if c5 < best_c5:

            best_c5 = c5

    # Phase 2: Main simulated annealing
    if time.time() < deadline - 100:

            best_h = h_sa
            best_c5 = c5_sa

    # Phase 3: High-alpha SLSQP refinement
    if time.time() < deadline - 35:

            if time.time() > deadline - 30:

            h_new = slsqp_refine(best_h, n, target, alpha, 200)

                best_c5 = c5

    # Phase 4: Diverse starting points

        make_sinusoidal(n, 3, 0.25, 1.0),

        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h = h_alt
            best_c5 = c5_alt
        elif c5_alt < best_c5 + 0.005 and time.time() < deadline - 40:
            h_alt, c5_alt = sa_optimize(h_alt, n, target, rng, min(deadline - 25, time.time() + 35), T0=0.003)
            h_alt = slsqp_refine(h_alt, n, target, 3000, 100)
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_h = h_alt
                best_c5 = c5_alt

    # Phase 5: Final SA polish on best
    if time.time() < deadline - 12:
        h_sa, c5_sa = sa_optimize(best_h, n, target, rng, deadline - 6, T0=0.0015)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa

    # Final SLSQP squeeze
    if time.time() < deadline - 2:
        h_new = slsqp_refine(best_h, n, target, 5000, 100)

            best_c5 = c5

    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n
