    # Find lambda such that sum(clip(h - lambda, 0, 1)) = T
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > T:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h_new = np.clip(h - lam, 0, 1)
    # Fix tiny numerical error
    diff = T - h_new.sum()
    # Distribute among non-boundary entries
    mask = (h_new > 1e-10) & (h_new < 1 - 1e-10)
    if mask.sum() > 0 and abs(diff) > 1e-12:
        h_new[mask] += diff / mask.sum()
        h_new = np.clip(h_new, 0, 1)
    return h_new

def coord_descent(h, n, time_limit, rng, momentum=0.0):
    dx = 2.0 / n
    T = n / 2.0
    h = np.array(h, dtype=float)
    h = project_feasible(h, T)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    v = np.zeros(n)  # momentum
    
    for sweep in range(10000):
        if time.time() > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            # Compute gradient of each corr_k w.r.t. h[i]
            g = np.zeros(2 * n - 1)
            s_range = np.arange(-(n-1), n)
            # For shift s, d(corr_{s+n-1})/d(h[i]) = (1-h[i-s]) if 0<=i-s<n else 0
            #                                        - h[i+s] if 0<=i+s<n else 0
            idx_pos = i - s_range  # = i-s for each s
            valid_pos = (idx_pos >= 0) & (idx_pos < n)
            g[valid_pos] += (1.0 - h[idx_pos[valid_pos]])
            
            idx_neg = i + s_range  # = i+s for each s
            valid_neg = (idx_neg >= 0) & (idx_neg < n)
            g[valid_neg] -= h[idx_neg[valid_neg]]
            
            # 1D minimax: min_delta max_k (corr[k] + delta*g[k])
            # subject to 0 <= h[i]+delta <= 1
            lo = -h[i]
            hi = 1.0 - h[i]
            
            if lo >= hi:
                continue
            
            # Evaluate at boundaries and intersection points
            best_d = 0.0
            best_val = np.max(corr)
            
            # Check boundaries
            for d in [lo, hi]:
                val = np.max(corr + d * g)
                if val < best_val:
                    best_val = val
                    best_d = d
            
            # Check 0
            val0 = np.max(corr)
            if val0 < best_val:
                best_val = val0
                best_d = 0.0
            
            # Check intersection points of top correlations
            top_idx = np.argsort(-corr)[:min(30, 2*n-1)]
            for ai in top_idx:
                for bi in top_idx:
                    if ai == bi:
                        continue
                    if abs(g[bi] - g[ai]) < 1e-14:
                        continue
                    d = (corr[ai] - corr[bi]) / (g[bi] - g[ai])
                    if lo < d < hi:
                        val = np.max(corr + d * g)
                        if val < best_val:
                            best_val = val
                            best_d = d
            
            # Apply with damping
            step = 0.99 * best_d
            if momentum > 0:
                v[i] = momentum * v[i] + (1 - momentum) * step
                step = v[i]
            
            h[i] += step
            # Update correlations (exact for k!=0, approximate for k=0)
            corr += step * g
            # Correct for k=0 (the n-1 index in the full correlation)
            # Actually, for k=0 (index n-1), there's a -step² correction
            # But this is tiny for small steps, skip for now
        
        # Periodic: recompute correlations and project
        if sweep % 5 == 0:
            h = project_feasible(h, T)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_now = np.max(corr) * dx
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
    
    h = project_feasible(h, T)
    return h

def project_to_grid(h_parent, n_target):
    """Interpolate parent solution to target grid size"""
    n_parent = len(h_parent)
    x_parent = np.linspace(0, 2, n_parent, endpoint=False) + 1.0/n_parent
    x_target = np.linspace(0, 2, n_target, endpoint=False) + 1.0/n_target

    # Actually simpler: just interpolate
    h_new = np.interp(x_target, x_parent, h_parent, period=2)
    # Normalize
    T = n_target / 2.0
    h_new = project_feasible(h_new, T)
    return h_new

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, T):
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > T:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = T - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_all(h, n):
    """Compute gradient of all correlations w.r.t. h[i] for all i.
    Returns array of shape (n, 2n-1) where grad[i,k] = d(corr_k)/d(h[i]).
    """
    # grad[i, k+n-1] where s=k-(n-1):
    # = (1-h[i-s]) if 0<=i-s<n - h[i+s] if 0<=i+s<n
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    # i - s: shape (n, 2n-1)
    i_idx = np.arange(n)[:, None]
    diff = i_idx - s[None, :]  # i - s
    ts = i_idx + s[None, :]    # i + s
    
    grad = np.zeros((n, 2*n-1))
    valid1 = (diff >= 0) & (diff < n)
    # For valid1, grad[i,k] += 1 - h[diff[i,k]]
    grad[valid1] = 1.0 - h[np.where(valid1, diff, 0)]
    valid2 = (ts >= 0) & (ts < n)
    grad[valid2] -= h[np.where(valid2, ts, 0)]
    return grad

def solve_1d_minimax(corr, g, lo, hi):
    """Solve min_delta max_k (corr[k] + delta * g[k]) for delta in [lo, hi]."""
    if lo >= hi:
        return 0.0
    
    def f(d):
        return np.max(corr + d * g)
    
    # Candidate points
    cands = [lo, hi, 0.0]
    
    # Pairwise intersections of top lines
    top_n = min(25, len(corr))
    top_idx = np.argsort(-corr)[:top_n]
    for idx in range(top_n):
        for jdx in range(idx+1, top_n):
            a, b = top_idx[idx], top_idx[jdx]
            dg = g[b] - g[a]
            if abs(dg) < 1e-14:
                continue
            d = (corr[a] - corr[b]) / dg
            if lo < d < hi:
                cands.append(d)
    
    best_d = 0.0
    best_v = f(0.0)
    for d in cands:
        v = f(d)
        if v < best_v:
            best_v = v
            best_d = d
    
    # Golden section refinement around best_d
    half = (hi - lo) * 0.05
    gl = max(lo, best_d - half)
    gh = min(hi, best_d + half)
    res = minimize_scalar(f, bounds=(gl, gh), method='bounded', options={'xatol': 1e-12})
    if res.fun < best_v:
        best_d = res.x
        best_v = res.fun
    
    return best_d

def coord_descent(h_init, n, time_limit, rng):
    dx = 2.0 / n
    T = n / 2.0
    h = project_feasible(h_init, T)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    max_sweeps = 5000
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            # Gradient of corr w.r.t. h[i]
            s_arr = np.arange(-(n-1), n)
            g = np.zeros(2*n - 1)
            d1 = i - s_arr
            v1 = (d1 >= 0) & (d1 < n)
            g[v1] += (1.0 - h[d1[v1]])
            d2 = i + s_arr
            v2 = (d2 >= 0) & (d2 < n)
            g[v2] -= h[d2[v2]]
            
            lo = -h[i]
            hi = 1.0 - h[i]
            if lo >= hi - 1e-14:
                continue
            
            best_d = solve_1d_minimax(corr, g, lo, hi)
            # Damping
            step = best_d
            h[i] += step
            # Update corr incrementally (exact for k≠0, approximate for k=0)
            corr += step * g
        
        # Every 3 sweeps: project and recompute
        if sweep % 3 == 0:
            h = project_feasible(h, T)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_now = np.max(corr) * dx
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h.copy()
    
    h = project_feasible(h, T)
    c5_final = compute_c5(h, dx)
    return best_h if c5_final > best_c5 else h

def project_to_grid(h_parent, n_target):
    n_p = len(h_parent)
    if n_p == n_target:
        return h_parent.copy()
    x_p = np.linspace(0, 2, n_p, endpoint=False)
    x_t = np.linspace(0, 2, n_target, endpoint=False)
    # Add periodic endpoint for smooth interpolation

    x_ext = np.concatenate([x_p, [2.0]])
    h_new = np.interp(x_t, x_ext, h_ext)
    return project_feasible(h_new, n_target / 2.0)

def binary_local_search(n, rng, time_limit):
    """Search over binary {0,1} patterns with exactly n/2 ones."""
    T = n // 2
    best_h = None
    best_c5 = 1e9
    dx = 2.0 / n
    
    # Random init
    h = np.zeros(n)
    h[rng.choice(n, T, replace=False)] = 1.0
    
    for _ in range(100000):
        if time.time() > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(c) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try a random swap: pick a 1-position and a 0-position
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        # Evaluate
        c2 = np.correlate(h, 1.0 - h, mode='full')
        c5_2 = np.max(c2) * dx
        if c5_2 >= c5:
            # Revert
            h[j] = 0.0
            h[i] = 1.0
    
    return best_h, best_c5

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, T):
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > T:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def eval_minimax(corr, g, d):
    return float(np.max(corr + d * g))

def solve_1d_minimax(corr, g, lo, hi):
    if lo >= hi:
        return 0.0
    cands = [lo, hi, 0.0]
    top_n = min(30, len(corr))
    top_idx = np.argsort(-corr)[:top_n]
    for ii in range(top_n):
        for jj in range(ii + 1, top_n):
            a, b = int(top_idx[ii]), int(top_idx[jj])
            dg = g[b] - g[a]
            if abs(dg) < 1e-14:
                continue
            d = (corr[a] - corr[b]) / dg
            if lo < d < hi:
                cands.append(d)
    best_d = 0.0
    best_v = eval_minimax(corr, g, 0.0)
    for d in cands:
        v = eval_minimax(corr, g, d)
        if v < best_v:
            best_v = v
            best_d = d
    # Refine with golden section
    half = 0.1 * (hi - lo)
    gl = max(lo, best_d - half)
    gh = min(hi, best_d + half)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = gl, gh
    c = b - gr * (b - a)

    for _ in range(40):

            dpt = c
            fd = fc
            c = b - gr * (b - a)

        else:
            a = c
            c = dpt
            fc = fd
            dpt = a + gr * (b - a)
            fd = eval_minimax(corr, g, dpt)
    best_d = 0.5 * (a + b)
    best_v = eval_minimax(corr, g, best_d)
    return best_d

def coord_descent(h_init, n, time_limit, rng):

    T = n / 2.0
    h = project_feasible(h_init, T)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    s_arr = np.arange(-(n - 1), n)
    for sweep in range(20000):
        if time.time() > time_limit:

        order = rng.permutation(n)

            d1 = i - s_arr
            v1 = (d1 >= 0) & (d1 < n)
            g[v1] = (1.0 - h[d1[v1]])
            d2 = i + s_arr
            v2 = (d2 >= 0) & (d2 < n)
            g[v2] -= h[d2[v2]]
            lo = -h[i]
            hi = 1.0 - h[i]
            if lo >= hi - 1e-14:
                continue
            step = solve_1d_minimax(corr, g, lo, hi)
            h[i] += step

            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_now = float(np.max(corr) * dx)
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h.copy()
    c5f = compute_c5(h, dx)

    h_new = np.interp(x_t, x_ext, h_ext)
