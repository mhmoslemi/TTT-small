def project(h, S):

    d = S - h.sum()
    if abs(d) > 1e-12:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
        d2 = S - h.sum()
        if d2 > 0:
            for i in np.argsort(-h):
                a = min(d2, 1.0 - h[i])
                if a > 0: h[i] += a; d2 -= a
                if d2 <= 1e-12: break

                a = min(-d2, h[i])

        hb[zi[np.argsort(-h[zi])[:t - ow]]] = 1
    return hb
