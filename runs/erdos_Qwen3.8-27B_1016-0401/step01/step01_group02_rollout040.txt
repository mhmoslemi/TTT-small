        # Backtracking line search

            h_new = _proj(h - alpha * g, n)
            f_new = _c5(h_new)
            if f_new < f - 1e-4 * alpha * np.sum(g * (h_new - h)):

            alpha *= rho
            continue

        f_prev = f
        if f < best_f:
            best_f = f
            best_h = h.copy()
    
    return best_h, best_f

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-14:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_smooth(h, sc):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    w = np.exp(sc * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
        else:
            ss = -s
            if ss < n:
                grad[:n - ss] += wk * (1.0 - h[ss:])
                grad[ss:] -= wk * h[:n - ss]
    return grad * dx

def _pgd(h0, sc_lo, sc_hi, tmax, n_iter_cap=2000000):
    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    f = _c5(h)
    best_h = h.copy()
    best_f = f
    alpha = 0.02
    it = 0
    while time.time() - t0 < tmax and it < n_iter_cap:
        prog = (time.time() - t0) / max(tmax, 1e-9)
        sc = sc_lo + (sc_hi - sc_lo) * min(prog * 1.5, 1.0)
        g = _grad_smooth(h, sc)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        accepted = False
        for _ in range(20):
            h_new = _proj(h - alpha * g, n)
            f_new = _c5(h_new)
            if f_new < f + 1e-16:
                h, f = h_new, f_new
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            alpha = max(alpha * 0.5, 1e-6)
            break
        if f < best_f:
            best_f = f
            best_h = h.copy()
        alpha = min(alpha * 1.03, 0.08)
        it += 1
    return best_h, best_f

def _block_ls(h0, n, tmax, nit_max=500000):
    t0 = time.time()
    tgt = 0.5 * n
    best = h0.copy()
    bestc = _c5(best)

    curc = bestc

        if time.time() - t0 > tmax:

def _slsqp_quick(h0, n, dx, sc, maxiter=2000):
    def _f(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        m = corr.max()
        return (m + np.log(np.exp(sc * (corr - m)).sum()) / sc) * dx
    def _ceq(x):
        return x.sum() - 0.5 * n
    def _ceqj(x):
        return np.ones(len(x))
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    try:
        r = minimize(_f, h0, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-13})
        return _proj(r.x, n)
    except Exception:
        return _proj(h0, n)
