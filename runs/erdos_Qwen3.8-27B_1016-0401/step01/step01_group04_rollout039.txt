def corr_max(corr):
    return float(corr.max())

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    # Smooth max
    shifted = T * (corr - mx)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    return mx + np.log(np.sum(np.exp(shifted))) / T

def project_h(h, S):
    """Project h to satisfy 0<=h<=1 and sum=S"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - S) < 1e-12:
        return h
    # Simple projection: add uniform offset
    h = h + (S - s) / len(h)
    h = np.clip(h, 0.0, 1.0)
    # If still off, adjust
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-h)[:int(abs(diff)) + 1]
            h[idx] = np.minimum(1.0, h[idx] + diff / len(idx))
        else:
            idx = np.argsort(h)[:int(abs(diff)) + 1]
            h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return np.clip(h, 0.0, 1.0)

    target = int(S)

    if ones > target:
        one_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(ones - target):
            h_b[one_idx[order[i]]] = 0
    elif ones < target:

        order = np.argsort(-h[zero_idx])
        for i in range(target - ones):
            h_b[zero_idx[order[i]]] = 1
    return h_b

    h[i], h[j] = h[j], h[i]
    c5 = compute_c5(h)

        i, j = best_swap
        h_best[i], h_best[j] = h_best[j], h_best[i]
        c5_best = best_c5
    
    return h_best, c5_best

    mx = corr.max()
    s = T * (corr - mx)

    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    return h_b

                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_feasible(res.x, S)

    diff = S - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

        idx = np.where(h_b == 0)[0]

import numpy as np
from scipy.optimize import minimize
import time

_S_val = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    s = np.clip(T * (corr - mx), -300, 300)
    return mx + np.log(np.sum(np.exp(s))) / T

def project_h(h, S):
    h = np.clip(h, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = S - h.sum()
        if abs(diff2) > 1e-10:
            idx = np.argsort(-h) if diff2 > 0 else np.argsort(h)
            h[idx[:int(abs(diff2))+1]] += diff2 / (int(abs(diff2))+1)
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, S):
    target = int(S)
    h_b = (h >= 0.5).astype(float)
    ones = int(h_b.sum())
    if ones > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for k in range(ones - target):
            h_b[idx[order[k]]] = 0.0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for k in range(target - ones):
            h_b[idx[order[k]]] = 1.0
    return h_b

    h = np.zeros(n)
    used = set()

        p = int((k + 0.5) * n * phi) % n
        while p in used:
            p = (p + 1) % n
        h[p] = 1.0
        used.add(p)

        for i in ones:

                h_cur[j] = 1.0
                c = compute_c5(h_cur)
                if c < best_c5:

                    best_pair = (i, j)
                h_cur[i] = 1.0
                h_cur[j] = 0.0
        if best_pair is None:

        i, j = best_pair
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c5_cur = best_c5
    return h_cur, c5_cur
