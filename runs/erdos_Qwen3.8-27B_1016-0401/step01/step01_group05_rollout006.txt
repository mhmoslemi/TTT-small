    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a1 = rng.uniform(0.25, 0.45)
    a2 = rng.uniform(0.1, 0.3)
    h = 0.5 + a1 * np.cos(2 * np.pi * f1 * x + p1) + a2 * np.cos(2 * np.pi * f2 * x + p2)
    return make_feasible(np.clip(h, 0, 1), n / 2.0)

def simulated_annealing(h, n, target, rng, deadline, max_steps=180000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_h = h.copy()
    curr_c5 = best_c5
    T = 0.04
    T_min = 0.0005
    decay = (T_min / T) ** (1.0 / max(1, max_steps - 1))
    step_size = 0.08
    no_improve = 0
    accept_count = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        n_p = 2 if rng.random() < 0.75 else 3
        idx = rng.choice(n, size=n_p, replace=False)
        deltas = rng.randn(n_p) * step_size
        new_h = curr_h.copy()
        for ii in range(n_p):
            new_h[idx[ii]] = np.clip(new_h[idx[ii]] + deltas[ii], 0.0, 1.0)
        new_h = make_feasible(new_h, target)
        new_c5 = compute_c5(new_h)
        delta_f = new_c5 - curr_c5
        if delta_f < 0 or (T > T_min * 1.5 and rng.random() < np.exp(-delta_f / T)):
            curr_h = new_h
            curr_c5 = new_c5
            accept_count += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= decay
        if step % 8000 == 7999:
            step_size *= 0.92
            if step % 40000 == 39999:
                big_idx = rng.choice(n, size=rng.randint(4, 8), replace=False)
                curr_h[big_idx] += rng.randn(len(big_idx)) * 0.12
                curr_h = make_feasible(curr_h, target)
                curr_c5 = compute_c5(curr_h)
        if no_improve > 25000:
            T = max(T, 0.015)
            step_size = max(step_size, 0.08)
            no_improve = 0
    return best_h, best_c5, curr_h, curr_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = cosine_init(70, rng)
    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B quick refinement
    for alpha in [40, 100, 250]:
        if time.time() > deadline - 180:
            break
        h_new = lbfgs_optimize(best_h, n, target, alpha, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Main simulated annealing
    h_sa, c5_sa, h_sa_end, _ = simulated_annealing(best_h, n, target, rng, deadline - 150, 160000)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa
    if compute_c5(h_sa_end) < best_c5:
        best_h = h_sa_end
        best_c5 = compute_c5(best_h)

    # Phase 3: Multi-start with structured initializations
    for trial in range(6):
        if time.time() > deadline - 80:
            break
        h_alt = cosine_init(n, rng)
        h_alt = lbfgs_optimize(h_alt, n, target, 80, maxiter=200)
        h_alt = lbfgs_optimize(h_alt, n, target, 300, maxiter=200)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt, c5_alt
            # SA from this new best
            h_sa2, c5_sa2, _, _ = simulated_annealing(h_alt, n, target, rng, deadline - 80, 40000)
            if c5_sa2 < best_c5:
                best_h, best_c5 = h_sa2, c5_sa2

    # Phase 4: Final L-BFGS-B + SLSQP polish
    if time.time() < deadline - 20:
        for alpha in [400, 800]:
            if time.time() > deadline - 20:
                break
            h_new = lbfgs_optimize(best_h, n, target, alpha, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
    if time.time() < deadline - 8:
        h_new = slsqp_refine(best_h, n, target, 500, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Short final SA polish
    if time.time() < deadline - 4:
        h_pol, c5_pol, _, _ = simulated_annealing(best_h, n, target, rng, deadline - 2, 25000)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    best_h = make_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h.tolist(), best_c5, n
