```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_auto(h):
    """Compute autocorrelation for all lags."""
    n = len(h)
    return np.correlate(h, h, mode='full')  # length 2n-1, centered at n-1

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_auto_at_lag(h, lag):
    """Gradient of auto[lag] with respect to h. lag is offset from center."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = 2.0 * h
    elif lag > 0:
        # auto at positive lag: sum_{i=0}^{n-1-lag} h[i]*h[i+lag]
        g[:n-lag] += h[lag:]
        g[lag:] += h[:n-lag]
    else:
        lag = -lag
        g[lag:] += h[:n-lag]
        g[:n-lag] += h[lag:]
    return g

def subgradient_ascent(h, n, target, lr0, n_steps, deadline, rng):
    """Subgradient ascent to maximize min auto[k]."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    lr = lr0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Compute autocorrelations
        auto = compute_auto(h_curr)  # length 2n-1, center at n-1
        # Find the minimum (excluding... actually including all lags since C5 uses all)
        # C5 = max(corr(h, 1-h)) * dx = (sum(h) - min(auto)) * dx
        # So we want to maximize min(auto)
        k_star = np.argmin(auto)
        # Gradient of auto[k_star]
        lag = k_star - (n - 1)  # actual lag
        g = grad_auto_at_lag(h_curr, lag)
        # Gradient ascent step (we want to maximize auto[k_star])
        h_try = h_curr + lr * g
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h_curr = h_try
        # Decay learning rate
        lr = lr0 * (0.999 ** step)
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_auto(h):
    return np.correlate(h, h, mode='full')

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_objective(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    vals = alpha * (corr - m)
    vals = np.clip(vals, -500, 0)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

def slsqp_optimize(h, n, target, alpha, maxiter=200):
    try:
        res = minimize(
            lambda x: soft_max_objective(x, n, alpha),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def simulated_annealing(h_init, n, target, T0, T_final, n_iters, deadline, rng):
    h_curr = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    c5_curr = best_c5
    n_total = n_iters
    total_iters = 0
    
    while total_iters < n_total:
        if time.time() > deadline:
            break
        # Temperature
        frac = total_iters / n_total
        T = T0 * (T_final / T0) ** frac
        
        # Perturbation: random 2-5 coordinate change
        n_pert = rng.randint(2, 6)
        idx = rng.choice(n, size=n_pert, replace=False)
        deltas = rng.uniform(-0.1, 0.1, size=n_pert)
        
        h_try = h_curr.copy()
        h_try[idx] += deltas
        h_try = make_feasible(h_try, n, target)
        
        c5_try = compute_c5(h_try)
        delta_c5 = c5_try - c5_curr
        
        if delta_c5 < 0 or rng.uniform() < np.exp(-delta_c5 / max(T, 1e-10)):
            h_curr = h_try
            c5_curr = c5_try
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
        
        total_iters += 1
    
    return best_h, best_c5

def subgradient_method(h_init, n, target, lr0, n_iters, deadline, rng):
    h_curr = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    lr = lr0
    patience = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Compute the max correlation and its lag
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        
        # Gradient of g_{k_star} = sum(h) - auto(lag)
        # = -grad_auto(lag)
        g = np.zeros(n)
        if lag == 0:
            g[:] = -2.0 * h_curr
        elif 0 < abs(lag) < n:
            a = min(abs(lag), n)
            if lag > 0:
                g[:n-a] -= h_curr[a:]
                g[a:] -= h_curr[:n-a]
            else:
                g[a:] -= h_curr[:n-a]
                g[:n-a] -= h_curr[a:]
        
        # Step: h -= lr * g (g is gradient of the max, so -g is descent)
        h_try = h_curr - lr * g
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            patience = 0
        else:
            patience += 1
        
        h_curr = h_try
        lr *= 0.999
        
        # If stuck, take a random kick
        if patience > 200:
            idx = rng.choice(n, size=3, replace=False)
            h_curr[idx] += rng.uniform(-0.05, 0.05, 3)
            h_curr = make_feasible(h_curr, n, target)
            patience = 0
    
    return best_h, best_c5

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:

        if c5_sa < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    vals = alpha * (corr - m)
    vals = np.clip(vals, -500, 0)
    return (m + np.log(np.sum(np.exp(vals))) / alpha) * dx

def slsqp_optimize(h, n, target, alpha, maxiter=300):
    def obj(x):
        return soft_max_obj(x, n, alpha)
    def con_fun(x):
        return x.sum() - target
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n, target)
    except Exception:
        return h.copy()

def subgradient_polish(h_init, n, target, n_iters, deadline, rng, lr_scale=1.0):
    h_curr = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    T = float(n_iters)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        lr = lr_scale * 0.5 / np.sqrt(it + 10)
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = np.zeros(n)
        a = abs(lag)
        if a == 0:
            g[:] = -2.0 * h_curr
        elif a < n:
            if lag > 0:
                g[:n-a] -= h_curr[a:]
                g[a:] -= h_curr[:n-a]
            else:
                g[a:] -= h_curr[:n-a]
                g[:n-a] -= h_curr[a:]
        h_try = h_curr - lr * g
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h_curr = h_try
    return best_h, best_c5

def binary_swap_search(h_init, n, target, n_iters, deadline, rng):
    h_b = (h_init > 0.5).astype(float)
    ones_count = int(h_b.sum())
    target_ones = n // 2
    if ones_count != target_ones:
        idx_sorted = np.argsort(-h_b)
        h_b = np.zeros(n)
        h_b[idx_sorted[:target_ones]] = 1.0
    best_h = h_b.copy()
    best_c5 = compute_c5(best_h)
    h_curr = h_b.copy()
    c5_curr = best_c5
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ones = np.where(h_curr == 1.0)[0]
        zeros = np.where(h_curr == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_try = h_curr.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        c5_try = compute_c5(h_try)
        if c5_try < c5_curr:
            h_curr = h_try
            c5_curr = c5_try
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
    return best_h, best_c5

def simulated_annealing(h_init, n, target, T0, T1, n_iters, deadline, rng):
    h_curr = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    c5_curr = best_c5
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters, 1)
        T = T0 * (T1 / T0) ** frac
        n_pert = 2 + rng.randint(0, 3)
        idx = rng.choice(n, size=n_pert, replace=False)
        deltas = rng.normal(0, 0.05, size=n_pert)
        h_try = h_curr.copy()
        h_try[idx] += deltas
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        dc = c5_try - c5_curr
        if dc < 0 or rng.uniform() < np.exp(-dc / max(T, 1e-12)):
            h_curr = h_try
            c5_curr = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_curr.copy()
    return best_h, best_c5

    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:

    for alpha in [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]:

    # Phase 3: SLSQP refinement after subgradient
