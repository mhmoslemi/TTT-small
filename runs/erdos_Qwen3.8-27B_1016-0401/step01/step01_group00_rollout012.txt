```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_values(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = shifted - np.max(shifted)  # stability
    e = np.exp(shifted)
    val = mc + np.log(np.sum(e)) / T
    weights = e / np.sum(e)
    return val, weights

def soft_max_obj(h, n, T, mu=0):
    corr = np.correlate(h, 1 - h, mode="full")
    val, _ = soft_max_values(corr, T)
    obj = val * (2.0 / n)
    if mu > 0:
        obj += mu * (h.sum() - n/2.0)**2
    return obj

def soft_max_grad(h, n, T, mu=0):
    corr = np.correlate(h, 1 - h, mode="full")
    _, sw = soft_max_values(corr, T)
    
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n-o])
            grad[:n-o] += -w * h[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n-p]
    
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n/2.0)
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum: add to smallest entries
            idx = np.argsort(h)
            for i in idx:
                s = min(diff, 1.0 - h[i])
                h[i] += s
                diff -= s
                if diff <= 1e-12:
                    break
        else:
            # Need to decrease sum: reduce largest entries
            idx = np.argsort(h)[::-1]
            for i in idx:
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
                if diff >= -1e-12:
                    break
    return h
```
