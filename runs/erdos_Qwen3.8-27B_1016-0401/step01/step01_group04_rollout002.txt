    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = min(c5, best_c5_init)

    n = len(h)
    dx = 2.0 / n

    return float(corr.max() * dx)

def smooth_max_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T

    best_h = best_h.astype(float)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_out)
