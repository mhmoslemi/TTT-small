    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            order = np.argsort(-h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 1e-14:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 1e-14:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def topk_soft_obj(h, n, T, K):
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    if K >= len(corr):
        K = len(corr)
    # Get top K values
    topk = np.partition(corr, len(corr) - K)[len(corr) - K:]
    mc = topk.max()
    shifted = T * (topk - mc)
    # Numerically stable log-sum-exp
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val

    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    total = len(corr)
    if K >= total:
        K = total
    # Get top K indices and values
    topk_idx = np.argsort(corr)[-K:]
    topk_vals = corr[topk_idx]
    mc = topk_vals.max()
    shifted = T * (topk_vals - mc)
    e = np.exp(shifted)

        else:
            p = -s
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]

    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(-h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 1e-14:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 1e-14:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def topk_soft_obj(h, n, T, K):
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    total = len(corr)
    K = min(K, total)
    topk = np.partition(corr, total - K)[total - K:]
    mc = topk.max()
    shifted = T * (topk - mc)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def topk_soft_grad(h, n, T, K):
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    total = len(corr)
    K = min(K, total)
    topk_idx = np.argsort(corr)[-K:]
    topk_vals = corr[topk_idx]
    mc = topk_vals.max()
    shifted = T * (topk_vals - mc)
    e = np.exp(shifted)
    w = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(K):
        wi = w[i]
        if wi < 1e-12:
            continue
        s = topk_idx[i] - center
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] += -wi * h[:n - s]
        else:
            p = -s
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    return grad * dx

_G = {'n': 0, 'T': 0, 'K': 0, 'target': 0, 't0': 0, 'limit': 0}

def _g_obj(x):
    return topk_soft_obj(x, _G['n'], _G['T'], _G['K'])

def _g_jac(x):
    return topk_soft_grad(x, _G['n'], _G['T'], _G['K'])

def _g_eqfun(x):
    return x.sum() - _G['target']

def _g_eqjac(x):
    return np.ones(_G['n'])

def _time_cb(xk):
    return (time.time() - _G['t0']) > _G['limit']

def _subgrad_step(h, n, alpha):
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if s == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif s > 0:
        g[:n - s] = (1.0 - h[s:]) * dx
        g[s:] = -h[:n - s] * dx
    else:
        p = -s
        g[p:] = (1.0 - h[:n - p]) * dx
        g[:n - p] = -h[p:] * dx
    h_new = h - alpha * g
    return fix_constraint(h_new, n / 2.0)

def _avg_topk_step(h, n, K, alpha):
    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    topk_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    center = n - 1
    dx = 2.0 / n
    for ji in topk_idx:
        s = ji - center
        if s == 0:
            g += (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += (1.0 - h[s:])
            g[s:] += -h[:n - s]
        else:
            p = -s
            g[p:] += (1.0 - h[:n - p])
            g[:n - p] += -h[p:]
    g *= dx / K
    h_new = h - alpha * g
    return fix_constraint(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient warm start
    for i in range(5000):
        if elapsed() > 80:
            break
        alpha = 0.08 / (1.0 + i * 0.0008)
        h_new = _subgrad_step(h_best, n, alpha)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: SLSQP with top-K soft-max (main refinement)
    _G['n'] = n
    _G['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_eqfun, 'jac': _g_eqjac}]

    configs = [
        (3, 5), (3, 15), (5, 5), (5, 15), (5, 40),
        (10, 10), (10, 30), (20, 20), (20, 60),
        (50, 100), (100, 200),
    ]
    for K, T in configs:
        if elapsed() > budget_s - 450:
            break
        _G['T'] = T
        _G['K'] = min(K, 2 * n - 1)
        _G['t0'] = time.time()
        _G['limit'] = 60
        try:
            res = minimize(_g_obj, h_best, method='SLSQP', jac=_g_jac,
                          bounds=bounds, constraints=cons, callback=_time_cb,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = fix_constraint(res.x.copy(), target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Averaged top-K subgradient (fast local moves)
    for i in range(10000):
        if elapsed() > budget_s - 300:
            break
        K = 3 if i % 3 == 0 else (5 if i % 3 == 1 else 10)
        alpha = 0.04 / (1.0 + i * 0.0003)
        h_new = _avg_topk_step(h_best, n, K, alpha)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Diverse restarts
    n_restarts = 0
    while elapsed() < budget_s - 350:
        n_restarts += 1
        r = n_restarts
        if r % 8 == 1:
            h_init = (rng.random(n) >= 0.5).astype(float)
        elif r % 8 == 2:
            period = int(rng.integers(14, 60))
            half = period // 2
            pattern = np.zeros(period)
            pattern[:half] = 1.0
            h_init = np.tile(pattern, n // period + 1)[:n]
        elif r % 8 == 3:
            h_init = h_best + rng.normal(0, 0.12, n)
        elif r % 8 == 4:
            freq = int(rng.integers(3, 12))
            phase = rng.random() * 2.0
            h_init = ((np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase) > 0).astype(float))
        elif r % 8 == 5:
            n_blocks = int(rng.integers(4, 10))
            h_init = np.zeros(n)
            pos = int(rng.integers(0, 10))
            for _ in range(n_blocks):
                sz = int(rng.integers(8, 35))
                pos = int(rng.integers(0, 5))
                end = min(pos + sz, n)
                h_init[pos:end] = 1.0
                pos = end + int(rng.integers(4, 20))
                if pos >= n:
                    break
        elif r % 8 == 6:
            h_init = np.clip(h_best + rng.normal(0, 0.04, n), 0, 1)
        elif r % 8 == 7:
            h_init = (rng.random(n) > 0.48).astype(float)
        else:
            h_init = (rng.random(n) > 0.42).astype(float)

        h_init = fix_constraint(h_init, target)
        _G['K'] = 5
        _G['T'] = 20
        _G['t0'] = time.time()
        _G['limit'] = 40
        try:
            res = minimize(_g_obj, h_init, method='SLSQP', jac=_g_jac,
                          bounds=bounds, constraints=cons, callback=_time_cb,
                          options={'maxiter': 800, 'ftol': 1e-14})
            h_new = fix_constraint(res.x.copy(), target)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    # Phase 5: Resolution increase
    for scale in [2, 3]:
        n_hi = n * scale
        if n_hi > 900:
            break
        if elapsed() > budget_s - 120:
            break
        target_hi = n_hi / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = fix_constraint(h_hi, target_hi)

        _G['n'] = n_hi
        _G['target'] = target_hi
        bounds_hi = [(0.0, 1.0)] * n_hi
        cons_hi = [{'type': 'eq', 'fun': _g_eqfun, 'jac': _g_eqjac}]

        for K, T in [(5, 15), (10, 40), (3, 8)]:
            if elapsed() > budget_s - 40:
                break
            _G['K'] = min(K, 2 * n_hi - 1)
            _G['T'] = T
            _G['t0'] = time.time()
            _G['limit'] = 80
            try:
                res = minimize(_g_obj, h_hi, method='SLSQP', jac=_g_jac,
                              bounds=bounds_hi, constraints=cons_hi,
                              callback=_time_cb,
                              options={'maxiter': 1500, 'ftol': 1e-15})
                h_new = fix_constraint(res.x.copy(), target_hi)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    h_hi = h_new.copy()
            except Exception:
                pass

    # Phase 6: Final subgradient polish
    n = len(h_best)
    target = n / 2.0
    _G['n'] = n
    _G['target'] = target
    for i in range(8000):
        if elapsed() > budget_s - 5:
            break
        alpha = 0.03 / (1.0 + i * 0.0004)
        K = 3 if i % 2 == 0 else 5
        h_new = _avg_topk_step(h_best, n, K, alpha)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
