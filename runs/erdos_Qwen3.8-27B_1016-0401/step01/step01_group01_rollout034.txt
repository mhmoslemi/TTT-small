        best_h, best_c5 = _targeted_opt(best_h, n_f)
        best_h, best_c5 = _coord_descent(best_h, n_f, n_pass=20, n_grid=60)

    # Phase 8: Ultra-fine coordinate descent
    n_f = len(best_h)
    if time.time() - t0 < budget_s - 3:
        best_h, best_c5 = _coord_descent(best_h, n_f, n_pass=10, n_grid=80)

    # Enforce exact feasibility
    n_f = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n_f / 2.0
    for _ in range(3):
        d = target - best_h.sum()
        if abs(d) < 1e-12:
            break
        mask = (best_h > 0.01) & (best_h < 0.99)
        if mask.sum() > 0:
            best_h[mask] += d / mask.sum()
        else:
            best_h += d / n_f
        best_h = np.clip(best_h, 0.0, 1.0)

    _DX = 2.0 / n_f
    final_c5 = _corr_c5(best_h)

    return best_h, final_c5, n_f
