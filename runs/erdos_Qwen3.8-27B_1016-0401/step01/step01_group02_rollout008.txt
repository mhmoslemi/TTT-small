    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(d) < 1e-14:

        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)

def _subgrad(h, n):
    """Compute subgradient of max_k correlate(h, 1-h, 'full')[k] * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    kmax = np.argmax(corr)
    g = np.zeros(2 * n - 1)

    # = sum_i h[i] * (1 - h[i + k - (n-1)])

    # i.e., n-1-k <= i <= n-1 (intersected with [0, n-1])
    # 
    # Let j = i + k - (n-1), so i = j - k + (n-1)
    # corr[k] = sum_j h[j-k+n-1] * (1-h[j])  where both indices in [0,n-1]
    #
    # d(corr[k])/d(h[m]):
    # Term from h[m] in first factor: sum over j where j-k+n-1 = m -> j = m+k-n+1
    #   (1-h[m+k-n+1]) if m+k-n+1 in [0,n-1]
    # Term from h[m] in (1-h) factor: sum over j where j = m
    #   -h[m-k+n-1] if m-k+n-1 in [0,n-1]
    
    # Simpler: just do it numerically or use the structure
    # corr[k] = sum_{i=max(0,n-1-k)}^{min(n-1, n-1+k)} h[i] * (1-h[i+k-(n-1)])
    
    lo = max(0, n - 1 - kmax)
    hi = min(n - 1, n - 1 + kmax)
    for i in range(lo, hi + 1):
        j = i + kmax - (n - 1)
        # d/dh[i] of h[i]*(1-h[j]) = (1-h[j]) + h[i]*(-d(h[j])/d(h[i]))
        # If j != i: d/dh[i] = (1-h[j]) - h[i]*0 ... wait
        # d/dh[i] h[i]*(1-h[j]) = (1-h[j]) - h[i]*delta(i,j)
        # And d/dh[j] h[i]*(1-h[j]) = -h[i]  (if j != i)
        if j == i:
            g[i] = 1.0 - 2.0 * h[i]
        else:
            g[i] += 1.0 - h[j]  # from h[i] term
            g[j] += -h[i]  # from (1-h[j]) term
    return g * dx

        # Adaptive learning rate
        if it % 100 == 99:

            lr *= 0.95
    return best, bestc

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    kmax = int(np.argmax(corr))
    g = np.zeros(n)
    lo = max(0, n - 1 - kmax)
    hi = min(n - 1, n - 1 + kmax)
    for i in range(lo, hi + 1):
        j = i + kmax - (n - 1)
        if j == i:
            g[i] += 1.0 - 2.0 * h[i]
        else:
            g[i] += 1.0 - h[j]
            g[j] += -h[i]
    return g * dx

def _pgd(h, n, tmax, seed=0):
    rng = np.random.RandomState(seed)
    h = _proj(h.copy(), n)
    best = h.copy()
    bestc = _c5(h)
    t0 = time.time()
    lr = 0.03
    stagnant = 0
    while time.time() - t0 < tmax:
        g = _subgrad(h, n)
        h2 = _proj(h - lr * g, n)
        c = _c5(h2)
        if c < bestc - 1e-15:
            best, bestc = h2.copy(), c
            h = h2
            stagnant = 0
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.92
            stagnant += 1
            if stagnant > 80:
                h = _proj(h + rng.normal(0, 0.008, n), n)
                lr = 0.03
                stagnant = 0
    return best, bestc

def _local_search(h, n, tmax, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best = h.copy()
    bestc = _c5(h)
    t0 = time.time()
    step = 0.06
    while time.time() - t0 < tmax:
        mode = rng.random()
        h2 = h.copy()
        if mode < 0.7:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            dv = sgn * rng.uniform(0.005, step)
            h2[i] = np.clip(h2[i] + dv, 0.0, 1.0)
            h2[j] = np.clip(h2[j] - dv, 0.0, 1.0)
        else:
            bsz = rng.randint(2, 6)
            start = rng.randint(0, max(1, n - bsz))
            delta = rng.uniform(-0.04, 0.04, bsz)
            h2[start:start+bsz] = np.clip(h2[start:start+bsz] + delta, 0.0, 1.0)
        h2 = _proj(h2, n)
        c = _c5(h2)
        if c < bestc:
            best, bestc = h2.copy(), c
            h = h2
            step = min(step * 1.01, 0.15)
        elif c < bestc + 0.003 and rng.random() < 0.08:
            h = h2
        step = max(step * 0.9995, 0.004)
    return best, bestc

def _upscale(h, n2):
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n)
    x_new = np.linspace(0.0, 2.0, n2)
    h2 = np.interp(x_new, x_old, h)
    return _proj(h2, n2)

def run(seed=42, budget_s=1100, **kwargs):
