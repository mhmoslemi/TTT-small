import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    se = np.sum(e)
    sw = e / se
    obj = (mc + np.log(se) / T) * dx
    
    grad = np.zeros(n)
    center = n - 1
    for kk in range(-(n-1), n):
        w = sw[center + kk]
        if w < 1e-13:
            continue
        if kk >= 0:
            grad[:n-kk] += w * omh[kk:]
            grad[kk:] -= w * h[:n-kk]
        else:
            p = -kk
            grad[p:] += w * omh[:n-p]
            grad[:n-p] -= w * h[p:]
    grad *= dx
    return obj, grad

def project_box_sum(h, target):
    """Project h onto {p: 0<=p<=1, sum(p)=target}"""
    n = len(h)
    h = h.copy()
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - p.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining
        if diff > 0:
            mask = p < 1.0
            cnt = mask.sum()
            if cnt > 0:
                p[mask] += diff / cnt
        else:
            mask = p > 0.0
            cnt = mask.sum()
            if cnt > 0:
                p[mask] += diff / cnt
        p = np.clip(p, 0.0, 1.0)
    return p

def nesterov_pg(h0, target, T, n_iter, alpha0, rng=None):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_obj = obj_grad(h0, T)[0]
    
    for it in range(n_iter):
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mu = (t - 1.0) / t_k if t_k > 0 else 0.0
        y = h + mu * (h - h_prev)
        y = project_box_sum(y, target)  # Keep y feasible? Or just clip?
        # Actually for Nesterov, y doesn't need to be feasible, only the step result does
        # Let's not project y, just h
        f, g = obj_grad(y, T)
        alpha = alpha0 / (1.0 + 0.01 * it)  # Simple decay
        h_new = y - alpha * g
        h_new = project_box_sum(h_new, target)
        
        # Check if we improved
        f_new = compute_c5(h_new)
        if f_new < best_obj:
            best_obj = f_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        t = t_k
    return best_h, best_obj

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    se = np.sum(e)
    sw = e / se
    obj = (mc + np.log(se) / T) * dx
    grad = np.zeros(n)
    c = n - 1
    for k in range(-(n - 1), n):
        w = sw[c + k]
        if w < 1e-13:
            continue
        if k >= 0:
            grad[:n - k] += w * omh[k:]
            grad[k:] -= w * h[:n - k]
        else:
            p = -k
            grad[p:] += w * omh[:n - p]
            grad[:n - p] -= w * h[p:]
    grad *= dx
    return obj, grad

def project_box_sum(h, target):
    n = len(h)
    if n == 0:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    p = np.clip(h - lam, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-7:
        if diff > 0:
            m = p < 1.0
        else:
            m = p > 0.0
        cnt = m.sum()
        if cnt > 0:
            p[m] = np.clip(p[m] + diff / cnt, 0.0, 1.0)
    return p
