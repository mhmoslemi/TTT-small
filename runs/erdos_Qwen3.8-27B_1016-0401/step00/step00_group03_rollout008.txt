import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def best_improvement_swap(h, dx, time_limit, rng, max_passes=2000):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    passes = 0
    while passes < max_passes and time.time() < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_val = best_c5
        best_pair = None
        # Use random order to break symmetry
        oi = rng.permutation(len(ones))
        oj = rng.permutation(len(zeros))
        for ii in oi:
            i = ones[ii]
            for jj in oj:
                j = zeros[jj]
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h, dx)
                if c5 < best_val - 1e-12:
                    best_val = c5
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
                if time.time() >= time_limit:
                    break
            if time.time() >= time_limit:
                break
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_val
            passes += 1
        else:
            break
    return h, best_c5
