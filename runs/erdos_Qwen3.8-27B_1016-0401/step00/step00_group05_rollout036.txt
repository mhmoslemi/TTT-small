```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr_array(h):
    """Compute full cross-correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_value(h):
    """Compute C5 for a given h."""
    n = len(h)
    return np.max(_corr_array(h)) * (2.0 / n)

def _grad_smooth_max(h, beta=50.0):
    """Gradient of soft-max of correlations."""
    n = len(h)
    corr = _corr_array(h)
    # Soft-max weights
    corr_shifted = corr - corr.max()
    weights = np.exp(beta * corr_shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        offset = idx - (n - 1)
        if offset == 0:
            grad += w * (1.0 - 2.0 * h)
        elif offset > 0:
            j = offset
            grad[:n-j] += w * (1.0 - h[j:])
            grad[j:] -= w * h[:n-j]
        else:
            j = -offset
            grad[j:] += w * (1.0 - h[:n-j])
            grad[:n-j] -= w * h[j:]
    return grad, corr.max()

def _project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    lo = h.min() - target_sum - 1
    hi = h.max() + 1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        y = np.clip(h - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _optimize(h0, n, target_sum, time_limit):
    """Run projected gradient optimization."""
    h = h0.copy()
    c5 = _c5_value(h)
    best_h = h.copy()
    best_c5 = c5
    dx = 2.0 / n
    
    # Phase 1: Subgradient with soft-max
    beta = 30.0
    lr = 0.05 / n  # small learning rate
    patience = 0
    
    for it in range(20000):
        if time.time() > time_limit:
            break
        
        grad, corr_max = _grad_smooth_max(h, beta)
        
        # Step
        h_new = _project(h - lr * grad, target_sum)
        c5_new = _c5_value(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            patience = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            patience += 1
            # Reduce learning rate
            if patience % 100 == 0:
                lr *= 0.9
            h = h_new  # still move (even if not improving, to escape)
        
        # Increase beta (sharpen)
        if it % 1000 == 500:
            beta *= 1.5
        if it % 500 == 499 and patience > 200:
            # Add noise to escape local min
            h += np.random.randn(n) * 0.005
            h = _project(h, target_sum)
            patience = 0
    
    return best_h, best_c5

def _polish(h, n, target_sum, time_limit):
    """Polish using scipy SLSQP with soft-max objective."""
    
    def obj(x):
        h_l = np.clip(x, 0, 1)
        corr = _corr_array(h_l)
        beta = 100.0
        m = corr.max()
        return np.log(np.sum(np.exp(beta * (corr - m)))) / beta + m
    
    def obj_grad(x):
        h_l = np.clip(x, 0, 1)
        g, _ = _grad_smooth_max(h_l, beta=100.0)
        return g
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
         'jac': lambda x: np.ones(n)},
    ]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-10,
                                'disp': False})
        if res.success or res.fun < _c5_value(h):
            h_out = _project(res.x, target_sum)
            if _c5_value(h_out) < _c5_value(h):
                return h_out
    except:
        pass
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return np.max(_corr(h)) * (2.0 / len(h))

def _grad_sm(h, beta=40.0):
    n = len(h)
    c = _corr(h)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(n - 1):
        if w[idx] < 1e-8:
            continue
        j = n - 1 - idx
        g[:n - int(j)] += w[idx] * (1.0 - h[int(j):])
        g[int(j):] -= w[idx] * h[:n - int(j)]
    if w[n - 1] >= 1e-8:
        g += w[n - 1] * (1.0 - 2.0 * h)
    for idx in range(n, 2 * n - 1):
        if w[idx] < 1e-8:
            continue
        j = idx - n + 1
        g[int(j):] += w[idx] * (1.0 - h[:n - int(j)])
        g[:n - int(j)] -= w[idx] * h[int(j):]
    return g, cm

def _proj(x, S):
    lo, hi = float(x.min()) - S - 1, float(x.max()) + 1
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0, 1).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0, 1)

    except Exception:
        pass
    return h
```
