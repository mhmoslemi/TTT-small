    return h_out

def eval_c5(h):
    """Return (c5_max, c_array) where c = correlate(h,1-h,'full')*dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c *= dx
    return c.max(), c

def grad_smoothed(h, c, dx, alpha):
    """Gradient of log-sum-exp smoothed max of c."""
    n = len(h)
    c_max = c.max()
    shifted = (c - c_max) / alpha
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        for i in range(i_lo, i_hi):
            j = i + s
            grad[j] += w[k] * (1.0 - h[i]) * dx
            grad[i] -= w[k] * h[j] * dx
    return grad

def optimize_phase(h, n, t_end, target_sum, rng):
    """Projected gradient + perturbation optimization."""
    dx = 2.0 / n
    best_c5, c = eval_c5(h)
    alpha = 0.04
    lr = 0.2
    stagnant = 0

    while time.time() < t_end:
        g = grad_smoothed(h, c, dx, alpha)
        g_norm = np.linalg.norm(g)
        success = False

        for _ in range(25):
            if g_norm < 1e-14:
                break
            h_new = project(h - lr * g / g_norm * 0.1, target_sum)
            c5_new, c_new = eval_c5(h_new)
            if c5_new <= c.max() * (1.0 + 1e-7):
                h = h_new
                c = c_new
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    stagnant = 0
                else:
                    stagnant += 1
                success = True
                break
            lr *= 0.6
        if not success:
            stagnant += 3
            alpha = max(0.01, alpha * 0.9)

        # Periodic perturbation to escape local minima
        if stagnant > 8:
            n_pert = max(2, n // 15)
            idx_pert = rng.choice(n, n_pert, replace=False)
            h_pert = h.copy()
            for j in idx_pert:
                h_pert[j] += rng.uniform(-0.2, 0.2)
            h_pert = project(h_pert, target_sum)
            c5_pert, c_pert = eval_c5(h_pert)
            if c5_pert < best_c5:
                h = h_pert
                c = c_pert
                best_c5 = c5_pert
            stagnant = 0
            alpha = min(0.08, alpha * 1.4)
            lr = 0.15

        alpha = max(0.01, alpha * 0.9995)

    return h, best_c5

def local_search_polish(h, n, t_end, best_c5, rng):
    """Random 2-point swap polish."""
    target_sum = n / 2.0
    while time.time() < t_end:
        j1 = rng.randint(n)
        j2 = rng.randint(n)
        if j1 == j2:
            continue
        delta = rng.uniform(-0.15, 0.15)
        h_new = h.copy()
        h_new[j1] += delta
        h_new[j2] -= delta
        h_new = project(h_new, target_sum)
        c5_new, _ = eval_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            j = rng.randint(n)
            h_new2 = h.copy()
            h_new2[j] += rng.choice([-0.08, 0.08, -0.04, 0.04])
            h_new2 = project(h_new2, target_sum)
            c5_new2, _ = eval_c5(h_new2)
            if c5_new2 < best_c5:
                h = h_new2
                best_c5 = c5_new2
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n = 100
        x = np.linspace(0, 2, n, endpoint=False)
        best_h = 0.5 + 0.35 * np.sin(np.pi * x) + 0.15 * np.sin(3 * np.pi * x)
        best_h = np.clip(best_h, 0, 1)
        best_h = project(best_h, n / 2.0)

    n0 = len(best_h)
    _, best_c5 = eval_c5(best_h)

    # Phase 1: Optimize at n0
    t_p1 = t0 + budget_s * 0.30
    best_h, best_c5 = optimize_phase(best_h, n0, t_p1, n0 / 2.0, rng)

    # Phase 2: Try other resolutions
    for n_try in [100, 80, 120, 60, 90, 110]:
        if time.time() - t0 > budget_s * 0.82:
            break
        remaining = budget_s * 0.82 - (time.time() - t0)
        if remaining < 50:
            break
        if n_try == n0:
            h_try = best_h.copy() + rng.randn(n0) * 0.05
        else:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, best_h)
            h_try += rng.randn(n_try) * 0.03
        h_try = project(h_try, n_try / 2.0)
        t_end = time.time() + remaining
        h_try, c5_try = optimize_phase(h_try, n_try, t_end, n_try / 2.0, rng)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            n0 = n_try

    # Phase 3: Polish with coordinate descent
    t_p3 = t0 + budget_s - 10
    if time.time() < t_p3:
        best_h, best_c5 = local_search_polish(best_h, len(best_h), t_p3, best_c5, rng)

    # Final projection and evaluation
    n = len(best_h)
    best_h = project(best_h, n / 2.0)
    final_c5, _ = eval_c5(best_h)
    return best_h, final_c5, n
