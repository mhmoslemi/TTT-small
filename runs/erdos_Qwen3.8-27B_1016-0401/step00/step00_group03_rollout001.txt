import numpy as np
import time

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def ensure_binary_feasible(h, n, rng):
    h = (h > 0.5).astype(float)
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, size=cur - target, replace=False)] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, size=target - cur, replace=False)] = 1
    return h

def steep_descent(h, time_limit=60.0):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5_fast(h)
    cur = h.copy()
    t0 = time.time()

    for _ in range(200000):
        if time.time() - t0 > time_limit:
            break
        c5_cur = compute_c5_fast(cur)
        bi, bj = -1, -1
        best_step = c5_cur
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = compute_c5_fast(cur)
                cur[j] = 0.0
                cur[i] = 1.0
                if c < best_step - 1e-14:
                    best_step = c
                    bi, bj = int(i), int(j)
        if bi == -1:
            break
        cur[bi] = 0.0
        cur[bj] = 1.0
        if best_step < best_c5 - 1e-14:
            best_c5 = best_step
            best = cur.copy()
            patience = 0
        else:
            patience += 1
    return best, best_c5

def simulated_annealing(h, n, rng, time_limit=60.0):
    cur = h.copy()
    best = h.copy()
    c5_cur = compute_c5_fast(cur)
    c5_best = c5_cur
    T = 0.04
    t0 = time.time()
    while time.time() - t0 < time_limit:
        T = max(T * 0.9997, 1e-6)
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i] = 0.0
        cur[j] = 1.0
        c5_new = compute_c5_fast(cur)
        d = c5_new - c5_cur
        if d < 0 or rng.random() < np.exp(-d / T):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                best = cur.copy()
        else:
            cur[j] = 0.0
            cur[i] = 1.0
    return best, c5_best
