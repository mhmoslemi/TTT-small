    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def objective_factory(n_points):
    dx = 2.0 / n_points
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        return smooth_max(c, 50.0)
    return obj

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        parent = np.array(initial_h_values, dtype=float)
    except Exception:
        parent = np.ones(45) * 0.5
    n_parent = len(parent)
    parent_c5 = compute_c5(parent)
    best_h = parent.copy()
    best_c5 = parent_c5

    n_points_list = [45, 60, 80, 100, 120, 150]
    for n_pts in n_points_list:
        if time.time() - t_start > budget_s - 30:
            break
        target_sum = n_pts / 2.0

        # Build initial guesses
        inits = []
        if n_parent != n_pts:
            x_old = np.linspace(0, 2, n_parent, endpoint=False)
            x_new = np.linspace(0, 2, n_pts, endpoint=False)
            base = np.interp(x_new, x_old, parent)
        else:
            base = parent.copy()
        base = project_constraint(base, target_sum)
        inits.append(base)
        inits.append(project_constraint(base + rng.normal(0, 0.08, n_pts), target_sum))
        inits.append(project_constraint(base + rng.normal(0, 0.2, n_pts), target_sum))
        for freq in [2, 3, 5, 7]:
            ph = rng.uniform(0, 2 * np.pi)
            h_s = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n_pts) / n_pts + ph)
            inits.append(project_constraint(h_s, target_sum))
        h_r = rng.uniform(0.1, 0.9, n_pts)
        inits.append(project_constraint(h_r, target_sum))

        # Optimize each initial
        for h0 in inits:
            if time.time() - t_start > budget_s - 15:
                break
            obj_fn = objective_factory(n_pts)
            constraints = [{'type': 'eq', 'fun': lambda x, ts=target_sum: x.sum() - ts}]
            bounds = [(0.0, 1.0)] * n_pts

            # Multi-stage: decreasing beta via re-running
            current = h0
            for stage, it_max in [(300, 200), (100, 100)]:
                if time.time() - t_start > budget_s - 10:
                    break
                res = minimize(obj_fn, current, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': it_max, 'ftol': 1e-12})
                current = project_constraint(res.x, target_sum)

            c5_val = compute_c5(current)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = current.copy()

            # Local polish: try small perturbations around best
            time_left = budget_s - 5 - (time.time() - t_start)
            if time_left > 5 and c5_val < best_c5 + 0.005:
                for _ in range(20):
                    if time.time() - t_start > budget_s - 5:
                        break
                    h_try = current + rng.normal(0, 0.02, n_pts)
                    h_try = project_constraint(h_try, target_sum)
                    c_try = compute_c5(h_try)
                    if c_try < compute_c5(current):
                        current = h_try
                        if c_try < best_c5:
                            best_c5 = c_try
                            best_h = current.copy()

    # Final check
    if best_h is not None:
        best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
