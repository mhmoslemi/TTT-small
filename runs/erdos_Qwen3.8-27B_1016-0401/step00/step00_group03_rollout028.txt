def c5_smooth_and_grad(h, T=30.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    # logsumexp for numerical stability
    shifted = T * corr * dx - np.max(T * corr * dx)
    weights = np.exp(shifted)
    weights /= np.sum(weights)  # softmax weights
    c5 = np.max(corr * dx) + np.log(np.sum(weights)) / T
    # Actually: logsumexp(T*corr*dx)/T = max(T*corr*dx)/T + log(mean(exp(T*corr*dx - max)))/T
    # = max(corr*dx) + (1/T) * log(mean(exp(T*corr*dx - T*max(corr*dx))))
    
    # Gradient
    grad = np.zeros(n)
    # For each lag with significant weight, add its gradient
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-8:
            continue
        k = m - (n - 1)
        if k >= 0:
            # c_k = sum_{i=k}^{n-1} h[i](1-h[i-k])
            # grad: for i in [k, n-1]: (1-h[i-k]), for i in [0, n-1-k]: -h[i+k]
            idx1 = np.arange(k, n)
            grad[idx1] += w * (1 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] -= w * h[idx2 + k]
        else:
            l = -k
            # c_{-l} = sum_{i=0}^{n-1-l} h[i](1-h[i+l])
            # grad: for i in [0, n-1-l]: (1-h[i+l]), for i in [l, n-1]: -h[i-l]
            idx1 = np.arange(0, n - l)
            grad[idx1] += w * (1 - h[idx1 + l])
            idx2 = np.arange(l, n)
            grad[idx2] -= w * h[idx2 - l]
    grad *= dx
    return c5, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_c5_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = corr * dx
    vm = np.max(val)
    sh = tau * (val - vm)
    e = np.exp(sh)
    ws = np.sum(e)
    c5 = vm + np.log(ws) / tau
    wts = e / ws
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = wts[m]
        if w < 1e-9:
            continue
        k = m - (n - 1)
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                idx1 = np.arange(k, n)
                grad[idx1] += w * (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                grad[idx2] -= w * h[idx2 + k]
        else:
            l = -k
            idx1 = np.arange(0, n - l)
            grad[idx1] += w * (1.0 - h[idx1 + l])
            idx2 = np.arange(l, n)
            grad[idx2] -= w * h[idx2 - l]
    grad *= dx
    return c5, grad

def project_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize_start(h0, t_limit):
    n = len(h0)
    target = n / 2.0
    h = np.clip(np.asarray(h0, dtype=float).copy(), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        h[:] = 0.5
    else:
        h = project_sum(h, target)
    best = h.copy()
    best_c5 = compute_c5(best)

    for tau in [300.0, 100.0, 30.0, 10.0, 3.0]:
        if time.time() > t_limit:
            break
        for _ in range(3):
            if time.time() > t_limit:
                break
            def obj(x, t=tau):
                v, g = soft_c5_grad(x, t)
                p = np.sum(x) - target
                return v + 10.0 * p * p, g + 20.0 * p

            try:
                res = minimize(obj, h, jac=True, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10})
                h_new = project_sum(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    best = h_new.copy()
                    h = h_new.copy()
                elif np.sum(np.abs(res.x - h)) > 1e-4:
                    h = project_sum(res.x, target)
            except Exception:
                pass
    return best, best_c5

def try_binary_refine(h_frac, t_limit):
    n = len(h_frac)
    if np.sum(h_frac) != n / 2.0:
        return h_frac, compute_c5(h_frac)
    return h_frac, compute_c5(h_frac)

        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
