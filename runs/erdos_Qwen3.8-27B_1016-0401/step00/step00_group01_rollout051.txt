import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max(arr, temp):
    m = np.max(arr)
    return float(m + np.log(np.sum(np.exp(arr - m))) / temp)


def obj_fn(x, temp):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return smooth_max(c, temp)


def make_qr(p):
    h = np.zeros(p)
    for x in range(1, p):
        h[(x * x) % p] = 1.0
    diff = p / 2.0 - np.sum(h)
    if diff > 1e-12:
        h[np.where(h == 0)[0][0]] = diff
    elif diff < -1e-12:
        h[np.where(h == 1)[0][0]] = 1.0 + diff
    return h


def proj(h, t):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h[mask] = np.clip(h[mask], 0.0, 1.0)
    return h


def local_search(h, t, tlim):

    bc = compute_c5(b)
    n = len(b)
    t0 = time.time()
    imp = True
    while imp and time.time() - t0 < tlim:
        imp = False

                h2 = b.copy()

                    h2 = proj(h2, t)
                    c = compute_c5(h2)
                    if c < bc - 1e-13:
