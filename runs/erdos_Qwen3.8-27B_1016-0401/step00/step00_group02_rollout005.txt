    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    lo, hi = -2.0, 2.0

        proj = np.clip(h - lam, 0.0, 1.0)
        s = np.sum(proj)
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_values(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx
