import numpy as np
import time

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode="full")) * dx)

def compute_softgrad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    bc = beta * c * dx
    bc -= bc.max()
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-8:
            continue
        wk = w[k] * dx
        d = k - (n-1)
        for j in range(n):
            jpd = j + d
            jmd = j - d
            g = 0.0
            if 0 <= jpd < n:
                g += (1 - h[jpd])
            if 0 <= jmd < n:
                g -= h[jmd]
            grad[j] += wk * g
    return grad

    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(x + mid, 0.0, 1.0)) > tgt:
            hi = mid
        else:
            lo = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def _upsample(h, n):
    x_old = np.linspace(0.0, 1.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _optimize(h, n, rng, t_end, lr0=0.08, beta0=25.0):
    h = _proj(h, n)
    best = _c5(h)
    best_snap = h.copy()
    lr = lr0
    beta = beta0
    stall = 0
    while time.time() < t_end:
        val = _c5(h)
        if val < best - 1e-9:
            best = val
            best_snap = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 120:
            h = h + rng.randn(n) * (0.02 + 0.02 * min(stall / 200.0, 1.0))
            h = _proj(h, n)
            stall = 0
            continue
        g = _grad(h, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        improved = False
        for a in (lr, lr * 0.5, lr * 0.25, lr * 0.1):
            ht = _proj(h - a * g, n)
            vtry = _c5(ht)
            if vtry <= val + 1e-9:
                h = ht
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                lr = lr0 * 0.1
        if stall > 40:
            beta = max(5.0, beta * 0.85)
        elif stall < 10:
            beta = min(250.0, beta * 1.05)
    return best_snap, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.full(42, 0.5)
    n0 = len(h0)
    h0 = _proj(h0, n0)

    best_h = h0.copy()
    best_val = _c5(best_h)

    # Phase 1: optimize at original resolution
    frac1 = 0.25
    h1, v1 = _optimize(h0.copy(), n0, rng, t0 + budget_s * frac1)
    if v1 < best_val:
        best_h, best_val = h1, v1

    # Phase 2: try upsampled grids
    for mult in (2, 3):
        n_up = n0 * mult
        if n_up > 400 or time.time() > t_end - 30:
            break
        h_up = _proj(_upsample(best_h, n_up), n_up)
        remain = t_end - time.time()
        if remain < 15:
            break
        h2, v2 = _optimize(h_up, n_up, rng, time.time() + remain * 0.5)
        if v2 < best_val:
            best_h, best_val = h2, v2

    # Phase 3: more iterations at original resolution with different params
    if time.time() < t_end - 20:
        h3, v3 = _optimize(best_h[:n0].copy() if len(best_h) != n0 else best_h.copy(),
                           n0, rng, t_end, lr0=0.03, beta0=80.0)
        if v3 < best_val:
            best_h, best_val = h3, v3

    # Phase 4: random perturbations of best solution
    if time.time() < t_end - 10:
        for _ in range(8):
            if time.time() > t_end - 3:
                break
            h_p = best_h[:n0] + rng.randn(n0) * 0.08
            h_p = _proj(h_p, n0)
            h4, v4 = _optimize(h_p, n0, rng, time.time() + (t_end - time.time()) * 0.1)
            if v4 < best_val:
                best_h, best_val = h4, v4

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return best_h, best_val, len(best_h)
