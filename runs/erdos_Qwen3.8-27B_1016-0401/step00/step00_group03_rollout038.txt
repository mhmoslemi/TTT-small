                    return
                _, grad = smooth_obj_and_grad(h, target_sum, alpha)
                h_new = h - lr * grad
                h_new = project_to_simplex(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                h = h_new
                if it % 100 == 99:
                    lr *= 0.75

    # Phase 1: Optimize from initial
    optimize_from(best_h, iters=500, alphas=(5, 10, 30, 100, 300, 800, 2000))

    # Phase 2: Random binary restarts
    n_restarts = 12
    for r in range(n_restarts):
        if time.time() - start_time > budget_s - 45:
            break
        h_r = np.zeros(n)
        idx = rng.permutation(n)
        h_r[idx[:n // 2]] = 1.0
        h_r = project_to_simplex(h_r, target_sum)
        cr = compute_c5(h_r)
        if cr < best_c5:
            best_c5 = cr
            best_h = h_r.copy()
        optimize_from(h_r, iters=200, alphas=(30, 100, 300, 800), base_lr=0.04)

    # Phase 3: Structured starts (quadratic residues, modular patterns)
    for p in [7, 11, 13, 17, 29, 41, 53]:
        if time.time() - start_time > budget_s - 40:
            break
        h_s = np.zeros(n)
        for i in range(n):
            r2 = (i * i) % p
            h_s[i] = 1.0 if r2 in [0] or ((r2 ** ((p - 1) // 2)) % p == 1) else 0.0
        if h_s.sum() > 0:
            h_s = h_s / h_s.sum() * (n / 2.0)
            h_s = project_to_simplex(h_s, target_sum)
            cr = compute_c5(h_s)
            if cr < best_c5:
                best_c5 = cr
                best_h = h_s.copy()
            optimize_from(h_s, iters=150, alphas=(50, 200, 600), base_lr=0.03)

    # Phase 4: Fine-tune with very high alpha
    if time.time() - start_time < budget_s - 20:
        h_f = best_h.copy()
        for alpha in [500, 1500, 5000]:
            if time.time() - start_time > budget_s - 10:
                break
            lr = 0.005
            for it in range(800):
                if time.time() - start_time > budget_s - 5:
                    break
                _, grad = smooth_obj_and_grad(h_f, target_sum, alpha)
                h_new = h_f - lr * grad
                h_new = project_to_simplex(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                h_f = h_new
                if it % 200 == 199:
                    lr *= 0.7

    # Phase 5: Try higher resolution if time permits
    if time.time() - start_time < budget_s - 80 and n < 200:
        n2 = min(n * 3, 210)
        h_hi = np.interp(np.linspace(0, 2, n2, endpoint=False),
                         np.linspace(0, 2, n, endpoint=False), best_h)
        h_hi = project_to_simplex(h_hi, n2 / 2.0)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi
            n = n2
            target_sum = n / 2.0
            # Optimize at new resolution
            optimize_from(best_h, iters=300, alphas=(30, 100, 300, 800), base_lr=0.02)

    return best_h, best_c5, n
