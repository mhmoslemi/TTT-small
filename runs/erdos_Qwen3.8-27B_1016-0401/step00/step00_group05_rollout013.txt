def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, float(np.max(c))

def project(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_at_k(h, k):
    """Gradient of the k-th element of correlate(h, 1-h, 'full') * 2/n"""
    n = len(h)
    L = k - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if L >= 0:
        m = n - L  # number of valid pairs
        # c_k = sum_{i=0}^{m-1} h[i+L]*(1-h[i])
        # d/dh[j]:
        #   As h[i+L]: (1-h[i]) for i=j-L, valid if 0<=j-L<m, i.e., L<=j<L+m=L+n-L=n
        #   As h[i]: -h[i+L] for i=j, valid if 0<=j<m
        grad[L:L+m] += (1.0 - h[:m])       # h[i+L] contribution
        grad[:m] -= h[L:L+m]               # h[i] contribution
    else:
        M = -L
        m = n - M
        # c_k = sum_{i=M}^{n-1} h[i]*(1-h[i-M])
        grad[M:M+m] += (1.0 - h[:m])     # h[i] contribution  
        grad[:m] -= h[M:M+m]             # h[i-M] contribution
    return grad * dx

def optimize(h0, n, max_iter, starting_lr):
    h = h0.copy().astype(float)
    h = project(h, n)
    best_c5, best_h = np.inf, h.copy()
    lr = starting_lr
    c, c5 = compute_c5(h)
    best_c5, best_h = c5, h.copy()
    stuck = 0
