def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _grad_lag(h, k):

    g = np.zeros(N)
    for j in range(N):
        a = j - k
        b = j + k
        if 0 <= a < N:
            g[j] += 1.0 - h[a]
        if 0 <= b < N:
            g[j] -= h[b]
    return g

def _project(h, tgt):

            lo = m
        else:

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _grad_lag_vec(h, k):
    N = len(h)
    g = np.zeros(N)
    if k >= 0:
        g[k:] += (1.0 - h[:N - k])
        g[:N - k] -= h[k:]
    else:
        s = -k
        g[:N - s] += (1.0 - h[s:])
        g[s:] -= h[:N - s]
    return g


def _project(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        m = (lo + hi) * 0.5
        p = np.clip(h - m, 0.0, 1.0)
        if p.sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def _c5(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))


def _optimize(h0, tgt, t_end, rng):
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    best = h.copy()
    best_c = _c5(h)
    vel = np.zeros(N)
    mu = 0.82
    base_lr = 0.012
    base_tau = 0.05
    corr_len = 2 * N - 1

    for st in range(400000):
        if time.time() > t_end:
            break
        corr = _corr(h)
        c = float(np.max(corr) * dx)
        if c < best_c:
            best_c = c
            best = h.copy()

        tau = max(0.008, base_tau * (1.0 - st / 120000.0))
        s_arr = (corr - corr.max()) / tau
        w = np.exp(s_arr)
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w /= wsum

        g = np.zeros(N)
        for idx in range(corr_len):
            if w[idx] > 0.004:
                k = idx - (N - 1)
                g += w[idx] * _grad_lag_vec(h, k)
        g *= dx

        lr = base_lr / (1.0 + st / 25000.0)
        vel = mu * vel - lr * g
        h = _project(h + vel, tgt)

        if st % 6000 == 5999:
            h = _project(h + rng.normal(0, 0.015, N), tgt)

    return best, best_c
