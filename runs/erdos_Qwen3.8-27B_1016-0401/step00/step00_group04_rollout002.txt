        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            no_improve = 0
            ones_idx[ri] = j
            zeros_idx[rj] = i
        else:
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
            if no_improve > early_stop:
                break
    return h, best_c5


def simulated_annealing(h, time_limit, rng, T_init=0.015, T_min=0.0005, cooling=0.9998):
    h = h.copy().astype(np.float64)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones_idx = np.where(h > 0.5)[0].tolist()
    zeros_idx = np.where(h <= 0.5)[0].tolist()
    start = time.time()
    T = T_init
    while time.time() - start < time_limit:
        if not ones_idx or not zeros_idx:
            break
        ri = int(rng.integers(len(ones_idx)))
        rj = int(rng.integers(len(zeros_idx)))
        i = ones_idx[ri]
        j = zeros_idx[rj]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        delta = c5 - cur_c5
        accept = delta < 0 or (rng.random() < np.exp(-delta / T))
        if accept:
            cur_c5 = c5
            ones_idx[ri] = j
            zeros_idx[rj] = i
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5


def remaining_time(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except (NameError, UnboundLocalError):
        h0 = np.zeros(42)
        h0[::2] = 1.0

    h0 = h0.astype(np.float64)
    # Enforce binary
    h0[h0 > 0.5] = 1.0
    h0[h0 <= 0.5] = 0.0
    # Fix sum if needed
    s = int(np.sum(h0))
    target = len(h0) // 2
    if s != target:
        ones = np.where(h0 == 1.0)[0]
        zeros = np.where(h0 == 0.0)[0]
        if s > target:
            remove = rng.choice(ones, s - target, replace=False)
            h0[remove] = 0.0
        else:
            add = rng.choice(zeros, target - s, replace=False)
            h0[add] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick local search at original grid
    t_rem = remaining_time(start_time, budget_s)
    t1 = min(15.0, t_rem * 0.04)
    if t1 > 2:
        h_s, c5_s = local_search(h0, t1, rng)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
        h0 = h_s

    # Phase 2: Grid doubling sequence
    h_cur = h0.copy()
    for phase in range(6):
        t_rem = remaining_time(start_time, budget_s)
        if t_rem < 25:
            break
        n_next = len(h_cur) * 2
        if n_next > 1500:
            break

        h_new = np.repeat(h_cur, 2).astype(np.float64)

        # Allocate time: scale with grid size but cap
        if phase == 0:
            t_ls = min(t_rem * 0.30, 80.0)
            t_sa = min(t_rem * 0.15, 40.0)
        elif phase == 1:
            t_ls = min(t_rem * 0.30, 200.0)
            t_sa = min(t_rem * 0.15, 60.0)
        elif phase == 2:
            t_ls = min(t_rem * 0.35, 300.0)
            t_sa = min(t_rem * 0.15, 80.0)
        else:
            t_ls = min(t_rem * 0.40, 250.0)
            t_sa = min(t_rem * 0.10, 50.0)

        if t_ls > 3:
            h_new, c5_new = local_search(h_new, t_ls, rng)
        else:
            c5_new = compute_c5(h_new)

        t_rem = remaining_time(start_time, budget_s)
        if t_sa > 3 and t_rem > 10:
            h_sa, c5_sa = simulated_annealing(h_new, t_sa, rng)
            if c5_sa < c5_new:
                h_new, c5_new = h_sa, c5_sa

        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

        h_cur = h_new

    # Phase 3: Direct 4x transfer from best n0 solution
    t_rem = remaining_time(start_time, budget_s)
    if t_rem > 40:
        h_4x = np.repeat(h0, 4).astype(np.float64)
        t_alloc = min(t_rem * 0.30, 120.0)
        h_r, c5_r = local_search(h_4x, t_alloc, rng)
        t_rem = remaining_time(start_time, budget_s)
        if t_rem > 15:
            h_sa2, c5_sa2 = simulated_annealing(h_r, min(t_rem * 0.3, 40.0), rng)
            if c5_sa2 < c5_r:
                h_r, c5_r = h_sa2, c5_sa2
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 4: Random restarts at promising sizes
    t_rem = remaining_time(start_time, budget_s)
    for n_try in [168, 336, 84]:
        t_rem = remaining_time(start_time, budget_s)
        if t_rem < 20:
            break
        h_rand = np.zeros(n_try)
        idx = rng.choice(n_try, n_try // 2, replace=False)
        h_rand[idx] = 1.0
        t_alloc = min(t_rem * 0.25, 80.0)
        h_r, c5_r = local_search(h_rand, t_alloc, rng)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 5: Final polish on best solution
    t_rem = remaining_time(start_time, budget_s)
    if t_rem > 10:
        h_pol, c5_pol = local_search(best_h, min(t_rem * 0.8, 60.0), rng, early_stop=20000)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
