        projected = np.clip(projected, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_constraint(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum}."""
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0, 1)
    # Binary search for shift
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(h + (lo + hi) / 2.0, 0, 1)

    rem = target_sum - proj.sum()
