    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_grad_max(h):
    """Compute gradient of the maximizing correlation component."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    m = idx - (n - 1)  # lag

        grad[:sz] += 1.0 - h[m:]
        grad[m:] -= h[:sz]
    else:
        sz = n + m
        grad[-m:] += 1.0 - h[:sz]
        grad[:sz] -= h[-m:]
    return k_best, grad

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_momentum(h, target, max_iter, t_limit, eta0=1.0, beta=0.85):
    n = len(h)
    eta = eta0 / n
    v = np.zeros(n)
    best_val = evaluate_c5(h)
    stall = 0
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        _, grad = compute_subgrad(h)
        v = beta * v + (1.0 - beta) * grad
        h_new = project_box_sum(h - eta * v, target)
        val = evaluate_c5(h_new)
        if val < best_val - 1e-14:
            h = h_new
            best_val = val
            stall = 0
            eta = min(eta * 1.005, 3.0 / n)
        else:
            stall += 1
            eta *= 0.97
            if stall > 500:
                eta = eta0 / n
                v = np.zeros(n)
                stall = 0
                if eta < 1e-10:
                    break
    return h, best_val

def smooth_obj_fn(x, target, temp):
    n = len(x)
    dx = 2.0 / n
    hx = project_box_sum(x, target)
    corr = np.correlate(hx, 1.0 - hx, mode="full") * dx
    shifted = corr - corr.max()
    return corr.max() + temp * np.log(np.sum(np.exp(shifted / temp)))

def nm_obj_fn(x, target):
    n = len(x)
    if n == 0 or x is None:
        return 1e6
    hx = np.clip(x, 0.0, 1.0)
    if abs(np.sum(hx) - target) > 0.5:
        return 1e6
    return evaluate_c5(hx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_raw = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_raw = None

    if h_raw is None or len(h_raw) == 0:
        n_init = 45
        h_raw = np.ones(n_init) * 0.5

    n = len(h_raw)
    target = float(n) / 2.0
    h0 = project_box_sum(h_raw, target)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Subgradient with momentum (fast, many iterations)
    t_lim = t0 + budget_s * 0.30
    best_h, best_c5 = subgrad_momentum(best_h, target, 50000, t_lim, eta0=1.0)

    # Phase 2: Smoothed L-BFGS-B with annealing temperature
    t_lim = t0 + budget_s * 0.65
    temps = [0.10, 0.05, 0.02, 0.01]
    for tp in temps:
        if time.time() > t_lim:
            break
        if time.time() > t_lim - 5:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_fn, best_h, args=(target, tp),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-14})
            hc = project_box_sum(res.x, target)
            vc = evaluate_c5(hc)
            if vc < best_c5:
                best_c5 = vc
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 3: Nelder-Mead on a perturbed copy (escape local minima)
    t_lim = t0 + budget_s * 0.80
    for trial in range(4):
        if time.time() > t_lim:
            break
        x_start = project_box_sum(best_h + 0.15 * np.random.randn(n), target)
        try:
            res = minimize(nm_obj_fn, x_start, args=(target,),
                           method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-12})
            hc = project_box_sum(np.clip(res.x, 0, 1), target)
            vc = evaluate_c5(hc)
            if vc < best_c5:
                best_c5 = vc
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 4: Fine subgradient polish
    t_lim = t0 + budget_s * 0.95
    best_h, best_c5 = subgrad_momentum(best_h, target, 50000, t_lim, eta0=0.05, beta=0.8)

    # Final projection and evaluation
    best_h = project_box_sum(best_h, target)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n)
