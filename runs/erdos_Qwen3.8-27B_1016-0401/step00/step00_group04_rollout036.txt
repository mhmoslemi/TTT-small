import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        idx = np.argsort(-room, kind='stable')
        rem = diff
        for i in idx:
            if rem <= 1e-14:
                break
            a = min(rem, room[i])
            h[i] += a
            rem -= a
    else:
        idx = np.argsort(h, kind='stable')
        rem = -diff
        for i in idx:
            if rem <= 1e-14:
                break
            s = min(rem, h[i])
            h[i] -= s
            rem -= s
    return h


def grad_lse(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    t = beta * corr
    t -= t.max()
    e = np.exp(t)
    w = e / e.sum()
    offset = n - 1
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12:
            continue
        l = k - offset
        if l >= 0 and l < n:
            g[l:] += wk * (1.0 - h[:n - l])
            g[:n - l] += wk * (-h[l:])
        elif l < 0 and l > -n:
            m = -l
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] += wk * (-h[:n - m])
    return g * dx


def interp_to_size(h_src, n):
    ns = len(h_src)
    if ns == n:
        return h_src.copy()
    xo = np.linspace(0.0, 1.0, ns, endpoint=False)
    xn = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(xn, xo, h_src)


def grad_descent(h, n, dx, target, t_dead, beta_init, lr_init):
    h = project_feasible(h, target)
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    beta = beta_init
    lr = lr_init
    stall = 0
    while time.time() < t_dead and stall < 500:
        g = grad_lse(h, beta, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        step = lr * g
        sn = np.linalg.norm(step)
        if sn > 0.4:
            step *= 0.4 / sn
        h_new = project_feasible(h - step, target)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.7
            stall += 1
            if lr < 1e-9:
                break
        beta = min(beta * 1.004, 250.0)
    return best_h, best_c5


def local_search(h, n, dx, target, t_dead, max_iter=1500):
    h = project_feasible(h, target)
    c5 = evaluate_c5(h)
    for _ in range(max_iter):
        if time.time() > t_dead:
            break
        improved = False
        for i in range(n // 2):
            j = n - 1 - i
            for delta in (0.01, -0.01, 0.03, -0.03, 0.005, -0.005):
                hi = h[i] + delta
                hj = h[j] - delta
                if hi < 0 or hi > 1 or hj < 0 or hj > 1:

                h[i] = hi
                h[j] = hj
                c5_t = evaluate_c5(h)

                    c5 = c5_t
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved:
            break
    return h, c5
