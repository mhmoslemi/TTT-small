def soft_max_obj_and_grad(h, T, dx):
    n = len(h)
    F = np.correlate(h, 1-h, mode='full')
    scale = T * dx
    logits = F / scale
    logits -= np.max(logits)
    exp_logits = np.exp(logits)
    Z = np.sum(exp_logits)
    obj = scale * np.log(Z)
    w = exp_logits / Z

            grad[s:] -= wk * h[:n-s]
        else:
            grad[-s:] += wk * (1.0 - h[:n+s])
            grad[:n+s] -= wk * h[-s:]
    return obj, grad
