    dx = 2.0 / n
    s = np.sum(x)
    return sm * dx + lam * (s - n / 2.0) ** 2


def _gradient(x, alpha, lam, n):
    corr = _compute_corr(x)
    m = np.max(corr)
    ew = np.exp(alpha * (corr - m))
    sw = np.sum(ew)
    w = ew / sw
    dx = 2.0 / n
    nc = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(nc):
        k = idx - (n - 1)
        wi = w[idx] * dx
        if wi < 1e-18:
            continue
        if k >= 0:
            nk = n - k
            grad[k:] += wi * (1.0 - x[:nk])
            grad[:nk] -= wi * x[k:]
        else:
            p = -k
            np_ = n - p
            grad[:np_] += wi * (1.0 - x[p:])
            grad[p:] -= wi * x[:np_]
    s = np.sum(x)
    grad += 2.0 * lam * (s - n / 2.0)
    return grad


def _project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(1000):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        for i in range(n):
            nh = h[i] - diff
            if 0.0 <= nh <= 1.0:
                h[i] = nh
                break
        else:
            if diff > 0:
                order = np.argsort(h)[::-1]
            else:
                order = np.argsort(h)
            for i in order:
                if abs(diff) < 1e-14:
                    break
                if diff > 0:
                    adj = min(diff, h[i])
                else:
                    adj = max(diff, h[i] - 1.0)
                h[i] -= adj
                diff -= adj
    return h


def _local_search_binary(h, n, deadline):
    h = h.copy()
    best_c5 = _compute_c5(h)
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in ones:
            for j in zeros:
                trial = h.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                c5 = _compute_c5(trial)
                if c5 < best_c5 - 1e-14:
                    h = trial
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                break
        if improved:
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return h, best_c5


def _grad_optimize(h, n, bounds, alpha, lam, max_outer, t0, budget_s, margin):
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    target = n / 2.0
    h = best_h.copy()
    for outer in range(max_outer):
        if time.time() - t0 > budget_s - margin:
            break
        for _ in range(3):
            if time.time() - t0 > budget_s - margin:
                break
            res = minimize(
                _objective, h, jac=_gradient,
                args=(alpha, lam, n),
                bounds=bounds, method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14}
            )
            h = _project_sum(np.clip(res.x, 0.0, 1.0), target)
            c5 = _compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        lam = min(lam * 3.0, 1e7)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0
        h0 = h0[rng.permutation(n)]

    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = _project_sum(h0, target)
    best_c5 = _compute_c5(best_h)

    # Phase 1: Gradient descent with moderate smoothing
    h1, c5_1 = _grad_optimize(best_h, n, bounds, 40.0, 50.0, 10, t0, budget_s, 180)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Sharper max, higher penalty
    if time.time() - t0 < budget_s - 90:
        h2, c5_2 = _grad_optimize(best_h, n, bounds, 150.0, 2000.0, 5, t0, budget_s, 90)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Binary rounding + local search
    if time.time() - t0 < budget_s - 30:
        h_bin = (best_h >= 0.5).astype(float)
        h_bin = _project_sum(h_bin, target)
        h_ls, c5_ls = _local_search_binary(h_bin, n, min(time.time() + 60, t0 + budget_s - 10))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # Phase 4: Random restarts
    if time.time() - t0 < budget_s - 40:
        for _ in range(3):
            if time.time() - t0 > budget_s - 40:
                break
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            h_rand = _project_sum(h_rand, target)
            h_r, c5_r = _grad_optimize(h_rand, n, bounds, 60.0, 500.0, 5, t0, budget_s, 40)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
            # Also try local search on this
            if time.time() - t0 < budget_s - 15:
                h_rb = (h_r >= 0.5).astype(float)
                h_rb = _project_sum(h_rb, target)
                h_rs, c5_rs = _local_search_binary(h_rb, n, min(time.time() + 15, t0 + budget_s - 5))
                if c5_rs < best_c5:
                    best_c5 = c5_rs
                    best_h = h_rs

    # Phase 5: Structured pattern (periodic)
    if time.time() - t0 < budget_s - 30:
        # Try pattern: blocks of 1s and 0s with varying period
        for period in [3, 4, 5, 6, 7]:
            if time.time() - t0 > budget_s - 30:
                break
            h_pat = np.zeros(n)
            for i in range(n):
                pos_in_period = i % period
                if pos_in_period < (period + 1) // 2:
                    h_pat[i] = 1.0
            h_pat = _project_sum(h_pat, target)
            h_p, c5_p = _grad_optimize(h_pat, n, bounds, 80.0, 1000.0, 3, t0, budget_s, 30)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    # Final: ensure valid output
    best_h = _project_sum(best_h, target)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, n
