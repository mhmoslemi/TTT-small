import numpy as np
import time

def eval_c5(h, dx):
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(corr) * dx), corr

def try_all_flips(h, dx):
    """Try all valid pair flips, return best (h, c5) or None if no improvement."""
    n = len(h)
    current_c5, _ = eval_c5(h, dx)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_gain = 0
    best_pair = None
    
    for i in ones:
        for j in zeros:
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            new_c5, _ = eval_c5(h, dx)
            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain
                best_pair = (i, j)
            h[i] = 1
            h[j] = 0
    
    if best_pair is not None:
        i, j = best_pair
        h[i] = 0
        h[j] = 1
        return h, current_c5 - best_gain
    return None

def steepest_descent(h, dx, max_iters=1000):
    """Repeated steepest descent until no improvement."""
    for _ in range(max_iters):
        result = try_all_flips(h, dx)
        if result is None:
            break
        h, c5 = result
    return h

    lr = 0.005
    for it in range(n_iters):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        # Gradient of f_k at h
        offset = k_star - (n - 1)
        g = np.zeros(n)
        i_min = max(0, -offset)
        i_max = min(n - 1, n - 1 - offset)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1)
            jdx = idx + offset
            g[idx] += (1.0 - h[jdx])
            g[jdx] -= h[idx]
        g *= dx
        h = h - lr * g
        # Project to [0,1] and sum=target
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        c5 = eval_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 500 == 0 and it > 0:
            lr = max(lr * 0.8, 0.0005)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline_hard = t0 + budget_s

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    # Enforce binary and sum
    h = (h > 0.5).astype(float)
    diff = int(round(n / 2.0 - h.sum()))
    if diff > 0:
        z = np.where(h == 0)[0]
        h[z[:diff]] = 1.0
    elif diff < 0:
        o = np.where(h == 1)[0]
        h[o[:(-diff)]] = 0.0

    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    best_n = n
    print(f"Start: n={n}, C5={best_c5:.6f}")

    # Phase 1: Steepest descent at initial n
    dl1 = t0 + budget_s * 0.15
    h, c5 = steepest_descent(h, dx, max_steps=2000, deadline=dl1)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n
    print(f"P1 SD n={n}: C5={best_c5:.6f} t={time.time()-t0:.1f}s")

    # Phase 2: Random restarts at initial n
    dl2 = t0 + budget_s * 0.35
    n_rounds = max(5, int((dl2 - time.time()) / 1.0))
    h, c5 = random_restarts(h, dx, rng, n_rounds, max_steps=80, deadline=dl2)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n
    print(f"P2 RLS n={n}: C5={best_c5:.6f} t={time.time()-t0:.1f}s")

    # Phase 3: Continuous relaxation at initial n
    dl3 = t0 + budget_s * 0.5
    hc, c5c = continuous_optimize(h, dx, n_iters=20000, deadline=dl3)
    # Round to binary preserving sum
    hr = (hc > 0.5).astype(float)
    diff = int(round(n / 2.0 - hr.sum()))
    if diff > 0:
        z = np.where(hr == 0)[0]
        # Pick zeros closest to 0.5 in hc
        order = np.argsort(-hc[z])
        hr[z[order[:diff]]] = 1.0
    elif diff < 0:
        o = np.where(hr == 1)[0]
        order = np.argsort(hc[o])
        hr[o[order[:(-diff)]]] = 0.0
    hr, c5r = steepest_descent(hr, dx, max_steps=300, deadline=dl3)
    if c5r < best_c5:
        best_c5, best_h, best_n = c5r, hr.copy(), n
    print(f"P3 Cont n={n}: C5={best_c5:.6f} t={time.time()-t0:.1f}s")

    # Phase 4: Grid doubling
    for stage in range(3):
        if time.time() > deadline_hard - 30:
            break
        h_up = upsample2(best_h)
        n_new = len(h_up)
        dx_new = 2.0 / n_new
        print(f"P4 upscale -> n={n_new}")

        dl_sd = t0 + budget_s * (0.5 + 0.15 * (stage + 1))
        if dl_sd < time.time() + 10:
            dl_sd = time.time() + 20
        h_up, c5_up = steepest_descent(h_up, dx_new, max_steps=500, deadline=dl_sd)
        if c5_up < best_c5:
            best_c5, best_h, best_n = c5_up, h_up.copy(), n_new
            print(f"  SD n={n_new}: C5={best_c5:.6f}")

        dl_rr = t0 + budget_s * (0.5 + 0.15 * (stage + 1) + 0.1)
        if dl_rr < time.time() + 10:
            n_rounds2 = 3
        else:
            n_rounds2 = max(3, int((dl_rr - time.time()) / 3.0))
        h_up, c5_up = random_restarts(h_up, dx_new, rng, n_rounds2, max_steps=40, deadline=dl_rr)
        if c5_up < best_c5:
            best_c5, best_h, best_n = c5_up, h_up.copy(), n_new
            print(f"  RLS n={n_new}: C5={best_c5:.6f}")

    # Final verification
    final_c5 = eval_c5(best_h, 2.0 / best_n)
    print(f"FINAL: n={best_n}, C5={final_c5:.6f}, time={time.time()-t0:.1f}s")
    return best_h, final_c5, best_n
