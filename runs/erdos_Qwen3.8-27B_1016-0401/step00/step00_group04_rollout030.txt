The key insight: for each coordinate j, the objective max_k c[k] as a function of h[j] (with other h fixed) is:
- max_k (a_k * h[j] + b_k) where a_k, b_k depend on the other h values
- This is a convex piecewise-linear function of h[j]
- Its minimum over [0,1] is at a breakpoint or endpoint

And with the sum constraint, I need to adjust: when I change h[j] by δ, I need to change another h[j'] by -δ to maintain the sum. This makes it more complex.

Alternative: use a Lagrange multiplier for the sum constraint.
- Augmented objective: max_k c[k] + λ * (sum(h) - n/2)
- For coordinate j: the augmented terms are (a_k + λ) * h[j] + b_k
- Minimize max_k ((a_k + λ) * h[j] + b_k) over h[j] ∈ [0,1]
- After each sweep, bisect on λ to satisfy sum(h) = n/2

This is cleaner! Let me implement this.

For a single coordinate update:
- Given current h and current λ
- For each k, compute a_k = coefficient of h[j] in c[k]
- The augmented coefficient is a_k + λ
- Find x* = argmin_x max_k ((a_k + λ) * x + b_k) over x ∈ [0,1]
- Set h[j] = x*

To find x*: 
- The function f(x) = max_k (α_k * x + β_k) where α_k = a_k + λ, β_k = b_k
- f is convex piecewise-linear
- Minimum is where f'(x-) ≤ 0 and f'(x+) ≥ 0
- Equivalently, at a point where the max is achieved by at least 2 terms with opposite slopes, or at 0/1

Algorithm:
1. Compute all α_k, β_k
2. Find all breakpoints x = (β_l - β_k) / (α_k - α_l) for α_k ≠ α_l
3. Keep those in [0,1]
4. Add 0 and 1
5. Evaluate f at each candidate
6. Return argmin

This is O((2n-1)^2) per coordinate, O(n * (2n-1)^2) per sweep = O(n^3).

Actually, the clearest definition is:
np.correlate(a, v, 'full') is the cross-correlation, defined as:
c[τ] = Σ_n a[n+τ] * conj(v[n])

where τ ranges from -(N-1) to M-1 (M=len(a), N=len(v)). With M=N=n:
τ ranges from -(n-1) to n-1, giving 2n-1 values.

In the output array (index 0 to 2n-2):
output[k] = c[k-(n-1)] = c[k-(n-1)]

So output[n-1] = c[0] = Σ_n a[n] * v[n] (zero-lag correlation)
output[n-1+τ] = c[τ] = Σ_n a[n+τ] * v[n] (positive lag τ)
output[n-1-τ] = c[-τ] = Σ_n a[n-τ] * v[n] (negative lag τ)

With a=h, v=1-h:
output[n-1+τ] = Σ_n h[n+τ] * (1-h[n]) for n where 0≤n≤n-1 and 0≤n+τ≤n-1
= Σ_{n=0}^{n-1-τ} h[n+τ] * (1-h[n]) for τ > 0

And multiplied by dx:
output[n-1+τ] * dx = Σ_{n=0}^{n-1-τ} h[n+τ] * (1-h[n]) * dx

This represents the integral ∫ h(x+τ*dx) * (1-h(x)) dx (approximately).

So the "shift" in the original problem corresponds to τ*dx shift in x.

OK so my understanding is correct. The max of output * dx over all entries is C₅.

Total coefficient: ((1-h[j-τ]) - h[j+τ]) * dx (when both are valid)
or just one of them (when only one is valid).

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def fix_sum(h, n):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = h.copy()
        if diff > 0:
            mask = h < 0.99999
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
            else:
                h += diff / n
        else:
            mask = h > 0.00001
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
            else:
                h += diff / n
    return np.clip(h, 0.1e-15, 1.0 - 1e-15)

def refine_grid(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = (np.arange(old_n) + 0.5) * 2.0 / old_n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, new_n)

def coord_sweep_lam(h, lam, n, dx):
    m = 2 * n - 1
    for j in range(n):
        h0 = h.copy()
        h0[j] = 0.0
        h1 = h.copy()
        h1[j] = 1.0
        c0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
        c1 = np.correlate(h1, 1.0 - h1, mode='full') * dx
        a = c1 - c0
        b = c0
        alpha = a + lam
        beta = b
        candidates = [0.0, 1.0]
        for ki in range(m):
            ai, bi = alpha[ki], beta[ki]
            for li in range(ki + 1, m):
                al = alpha[li]
                da = ai - al
                if abs(da) > 1e-10:
                    xb = (beta[li] - bi) / da
                    if -0.5 <= xb <= 1.5:
                        candidates.append(xb)
        best_x = 0.0
        best_f = np.inf
        for x in candidates:
            xc = min(max(x, 0.0), 1.0)
            fv = np.max(alpha * xc + beta)
            if fv < best_f:
                best_f = fv
                best_x = xc
        h[j] = best_x
    return h

def lagrangian_cd(h_init, n, budget, t0):
    dx = 2.0 / n
    h = fix_sum(h_init, n)
    lo, hi = -5.0, 5.0
    h_curr = h.copy()
    for it in range(500):
        if time.time() - t0 > t0 + budget:
            break
        h_curr = coord_sweep_lam(h_curr, 0.5 * (lo + hi), n, dx)
        s = h_curr.sum() - n / 2.0
        if s > 0:
            hi = 0.5 * (lo + hi)
        else:
            lo = 0.5 * (lo + hi)
        if abs(s) < 1e-10:
            h_curr = fix_sum(h_curr, n)
            # Check for improvement
            cur_val = evaluate_c5(h_curr)
            if cur_val < evaluate_c5(h):
                h = h_curr.copy()
            break
    return fix_sum(h_curr, n)

def slsqp_optimize(h_init, n, budget, t0):
    dx = 2.0 / n
    h_init = fix_sum(h_init, n)
    best_h = h_init.copy()
    best_val = evaluate_c5(best_h)
    m = 2 * n - 1

    def soft_obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        t = 0.005
        shifted = corr / t
        shifted -= shifted.max()
        return t * (shifted.max() + np.log(np.sum(np.exp(shifted))))

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bounds = [(0, 1)] * n
    rng = np.random.default_rng(777)

    for attempt in range(4):
        if time.time() - t0 > t0 + budget:
            break
        if attempt == 0:
            x0 = best_h.copy()
        else:
            x0 = np.clip(best_h + rng.normal(0, 0.08, n), 0, 1)
            x0 = fix_sum(x0, n)
        res = minimize(soft_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 400, 'ftol': 1e-12})
        hx = fix_sum(np.clip(res.x, 0, 1), n)
        v = evaluate_c5(hx)
        if v < best_val:
            best_h = hx
            best_val = v
    return best_h, best_val

def try_binary_improve(h, n, dt=5.0):
    """Try rounding near-binary values and local flips."""
    h = h.copy()
    thresh = 0.5
    h_binary = (h > thresh).astype(float)
    h_binary = fix_sum(np.clip(h_binary, 0.0, 1.0), n)
    c5_bin = evaluate_c5(h_binary)
    c5_cont = evaluate_c5(h)
    if c5_bin < c5_cont:
        return h_binary, c5_bin
    # Try single flips
    best_h = h.copy()
    best_c5 = c5_cont
    for j in range(n):
        if h[j] > 0.9 or h[j] < 0.1:
            continue
        for target in [0.0, 1.0]:
            trial = h.copy()
            trial[j] = target
            trial = fix_sum(trial, n)
            v = evaluate_c5(trial)
            if v < best_c5:
                best_c5 = v
                best_h = trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n0 = 42
        rng = np.random.default_rng(seed)
        h_init = np.zeros(n0)
        h_init[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    n0 = len(h_init)
    best_h = fix_sum(h_init, n0)
    best_c5 = evaluate_c5(best_h)

    def time_left():
        return max(0, budget_s - (time.time() - t0) - 3.0)

    # Phase 1: Coordinate descent at n0
    if time_left() > 30:
        t_sub = min(60, time_left() * 0.3)
        h_cd = lagrangian_cd(best_h, n0, t_sub, time.time())
        c5_cd = evaluate_c5(h_cd)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 2: SLSQP at n0
    if time_left() > 30:
        t_sub = min(60, time_left() * 0.3)
        h_s, c5_s = slsqp_optimize(best_h, n0, t_sub, time.time())
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Binary improvement at n0
    if time_left() > 10:
        h_b, c5_b = try_binary_improve(best_h, n0)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b

    # Phase 4: Refine to 2*n0
    n1 = 2 * n0
    if time_left() > 60 and n1 <= 504:
        h_r = refine_grid(best_h, n1)
        # CD on refined
        if time_left() > 30:
            t_sub = min(80, time_left() * 0.25)
            h_cd2 = lagrangian_cd(h_r, n1, t_sub, time.time())
            c5_2 = evaluate_c5(h_cd2)
            if c5_2 < best_c5:
                best_h, best_c5 = h_cd2, c5_2
        # SLSQP on refined
        if time_left() > 30:
            t_sub = min(60, time_left() * 0.25)
            h_s2, c5_s2 = slsqp_optimize(best_h, n1, t_sub, time.time())
            if c5_s2 < best_c5:
                best_h, best_c5 = h_s2, c5_s2

    # Phase 5: Refine to 4*n0
    n2 = 4 * n0
    if time_left() > 60 and n2 <= 504:
        h_r2 = refine_grid(best_h, n2)
        if time_left() > 30:
            t_sub = min(100, time_left() * 0.3)
            h_cd3 = lagrangian_cd(h_r2, n2, t_sub, time.time())
            c5_3 = evaluate_c5(h_cd3)
            if c5_3 < best_c5:
                best_h, best_c5 = h_cd3, c5_3
        if time_left() > 30:
            t_sub = min(80, time_left() * 0.25)
            h_s3, c5_s3 = slsqp_optimize(best_h, n2, t_sub, time.time())
            if c5_s3 < best_c5:
                best_h, best_c5 = h_s3, c5_s2 if False else c5_s3
                best_h = h_s3

    # Phase 6: Final CD polish on best
    if time_left() > 20:
        ns = len(best_h)
        t_sub = min(40, time_left() * 0.5)
        h_final = lagrangian_cd(best_h, ns, t_sub, time.time())
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    best_h = fix_sum(np.clip(best_h, 0, 1), len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
