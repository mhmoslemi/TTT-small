import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = float(np.max(x))
    shifted = x - m
    return m + float(np.log(np.sum(np.exp(beta * shifted)))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-12:
            continue
        s = mi - (n - 1)
        contrib = np.zeros(n)
        if s >= 0:
            nn = n - s
            if nn > 0:
                contrib[:nn] += (1.0 - h[s:s+nn])
                contrib[s:s+nn] -= h[:nn]
        else:
            s2 = -s
            nn = n - s2
            if nn > 0:
                contrib[s2:s2+nn] += (1.0 - h[:nn])
                contrib[:nn] -= h[s2:s2+nn]
        grad += w[mi] * contrib
    return grad * dx

def slsqp_optimize(h0, n, target_sum, beta, max_iter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max(corr, beta)
    def obj_jac(x):
        return grad_smooth_max_c5(x, beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h0, method='SLSQP', jac=obj_jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_res = project_feasible(res.x, target_sum)
        return h_res
    return h0

def projected_gradient(h, n, target_sum, beta, n_iter, step_init):
    h = project_feasible(h, target_sum)
    step = step_init
    best_local = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        g = grad_smooth_max_c5(h, beta)
        h_new = project_feasible(h - step * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_local = h.copy()
            best_c5 = c5_new
            step = min(step * 1.2, 0.5)
        else:
            step *= 0.5
            if step < 1e-8:
                break
    return best_local, best_c5

def make_random_h(n, rs, spread=0.5, center=0.5):
    h = rs.random(n) * spread + (center - spread / 2.0)
    h = project_feasible(h, n / 2.0)
    return h
