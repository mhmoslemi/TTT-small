    if abs(diff) < 1e-12:
        return h

        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)

        cnt = mask.sum()
