import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def project_constrained(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_new = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        free = (h_new > 1e-9) & (h_new < 1.0 - 1e-9)
        if free.sum() > 0:
            h_new[free] += diff / free.sum()
    return np.clip(h_new, 0.0, 1.0)

def compute_gradient(h, N, active_p):
    grad = np.zeros(N)
    for k in active_p:
        offset = k - N + 1
        if offset >= 0:
            if offset < N:
                grad[offset:] += 1.0 - h[:N - offset]
                grad[:N - offset] -= h[offset:]
        else:
            off = -offset
            if off < N:
                grad[:N - off] += 1.0 - h[off:]
                grad[off:] -= h[:N - off]
    return grad
