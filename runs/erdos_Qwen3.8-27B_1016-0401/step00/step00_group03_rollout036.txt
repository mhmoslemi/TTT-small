import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    onemh = 1.0 - h
    corr = np.correlate(h, onemh, mode='full')
    c5 = np.max(corr) * dx
    active_k = np.argmax(corr)
    lag = active_k - (n - 1)
    grad = np.ones(n)
    # subtract h[j-lag] where valid
    if lag >= 0:
        if lag > 0:
            grad[lag:] -= h[:n-lag]
    else:
        l = -lag
        grad[:n-l] -= h[l:]
    # subtract h[j+lag] where valid
    if lag >= 0:
        if lag > 0:
            grad[:n-lag] -= h[lag:]
    else:
        l = -lag
        grad[l:] -= h[:n-l]
    grad *= dx
    return c5, grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_grad_hard(h):
    """C5 and subgradient at the active lag."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    ak = np.argmax(corr)
    lag = ak - (n - 1)
    g = np.ones(n)
    if lag >= 0:
        if lag > 0:
            g[lag:] -= h[:n - lag]
            g[:n - lag] -= h[lag:]
    else:
        l = -lag
        g[:n - l] -= h[l:]
        g[l:] -= h[:n - l]
    return c5, dx * g

def c5_grad_soft(h, t=200.0):
    """C5 smooth approximation and gradient via log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # log-sum-exp for stability
    m = np.max(corr)
    log_sum = m + np.log(np.sum(np.exp(t * (corr - m))))
    c5 = log_sum / t * dx
    # weights
    p = np.exp(t * (corr - m)) / np.sum(np.exp(t * (corr - m)))
    # gradient: weighted average of subgradients
    g = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-15:
            continue
        lag = k - (n - 1)
        gk = np.ones(n)
        if lag >= 0:
            if lag > 0:
                gk[lag:] -= h[:n - lag]
                gk[:n - lag] -= h[lag:]
        else:
            l = -lag
            gk[:n - l] -= h[l:]
            gk[l:] -= h[:n - l]
        g += p[k] * gk
    return c5, dx * g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx

        g[:n - lag] -= h[lag:]
    elif lag < 0:
        l = -lag
        g[:n - l] -= h[l:]
        g[l:] -= h[:n - l]

    # Wait, for lag=0: g[0:] -= h[:n] and g[:n] -= h[0:], so g = 1 - 2h. Correct!
    return c5, dx * g

def make_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] += diff / m.sum()
    return h

def random_bin(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def random_cont(n, rng):
    return make_feasible(rng.random(n), n)

def interp_init(hp, nt):
    np_ = len(hp)
    if np_ == nt:
        return hp.copy()
    x0 = np.arange(np_) / np_ * 2.0
    x1 = np.arange(nt) / nt * 2.0
    y = np.interp(x1, x0, hp)
    return make_feasible(y, nt)

def run_slsqp(h0, n, maxiter=500, ftol=1e-10):
    h0 = make_feasible(h0, n)
    
    def fun(h):
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)
    
    def jac(h):
        _, g = obj_and_grad(h)
        return g
    
    cons = [{'type': 'eq',
             'fun': lambda h: h.sum() - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    res = minimize(fun, h0, jac=jac, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h = make_feasible(res.x, n)
    return h, compute_c5(h)

        if pn not in ns:
            ns.insert(0, pn)

        if time.time() - t0 > budget_s - 10:
            break
        inits = []
        if parent is not None:
            inits.append(interp_init(parent, n))
        for _ in range(8):
            inits.append(random_bin(n, rng))
        for _ in range(8):
            inits.append(random_cont(n, rng))

                break
            try:

                if c5 < best_c5:
                    best_h, best_c5, best_n = h.copy(), c5, n
            except Exception:
                pass
    
    # Refine best
    if best_h is not None and time.time() - t0 < budget_s - 5:
        try:
            h2, c52 = run_slsqp(best_h, best_n, maxiter=1000, ftol=1e-12)
            if c52 < best_c5:
                best_h, best_c5 = h2, c52
        except Exception:
            pass
    
    if best_h is None:

        best_h = make_feasible(parent.copy() if parent is not None else random_bin(n, rng), n)
        best_c5 = compute_c5(best_h)
        best_n = n
    
    return best_h, best_c5, len(best_h)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_g(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
    else:
        l = -L
        if l < n:
            g[:n - l] += 1.0 - h[l:]
            g[l:] -= h[:n - l]
    return g

def _grad_hard(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    L = int(np.argmax(corr)) - (n - 1)
    return dx * _lag_g(h, L)

def _obj_soft(h, t=400.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(t * (corr - m))
    dx = 2.0 / len(h)
    return (m + np.log(e.sum())) / t * dx

def _grad_soft(h, t=400.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(t * (corr - m))
    p = e / e.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-14:
            continue
        g += p[k] * _lag_g(h, k - (n - 1))
    return dx * g

def _con_fun(h):
    return h.sum() - len(h) / 2.0

def _con_jac(h):
    return np.ones(len(h))

def _feasible(h, n):
    tgt = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        if not m.any():
            break
        h[m] += d / m.sum()
    return h

def _rbin(n, rng):
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def _rcont(n, rng):
    return _feasible(rng.random(n), n)

def _block_init(n, rng):
    h = np.zeros(n)
    nb = max(2, int(rng.integers(2, n // 15 + 1)))
    bs = n // (nb * 2)
    if bs < 1:
        return _rbin(n, rng)
    for i in range(0, n, bs * 2):
        h[i:min(i + bs, n)] = 1.0
    return _feasible(h, n)

def _interp(hp, nt):
    na = len(hp)
    if na == nt:
        return np.array(hp, dtype=float).copy()
    x0 = np.arange(na, dtype=float) / na * 2.0
    x1 = np.arange(nt, dtype=float) / nt * 2.0
    y = np.interp(x1, x0, np.asarray(hp, dtype=float))
    return _feasible(y, nt)

def _run_slsqp(h0, n, maxiter, soft):
    h0 = _feasible(h0, n)
    f, j = (_obj_soft, _grad_soft) if soft else (_c5, _grad_hard)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    r = minimize(f, h0, jac=j, method='SLSQP',
                 bounds=[(0.0, 1.0)] * n, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': 1e-12})
    h = _feasible(r.x, n)
    return h, _c5(h)

def _pen_obj(h, n, lam):
    return _c5(h) + lam * (h.sum() - n / 2.0) ** 2

def _pen_jac(h, n, lam):
    g = _grad_hard(h).copy()
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

def _run_lbfgs(h0, n, maxiter):
    h0 = _feasible(h0, n)
    r = minimize(_pen_obj, h0, args=(n, 200.0), jac=_pen_jac,
                 method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                 options={'maxiter': maxiter, 'ftol': 1e-14, 'maxfun': 8000})
    h = _feasible(r.x, n)
    return h, _c5(h)

def _try_one(h0, n, t0, budget_s):
    if time.time() - t0 > budget_s - 10:
        return None, float('inf')
    if n <= 200:
        try:
            h, c5 = _run_slsqp(h0, n, 500, True)
            if c5 < float('inf'):
                return h, c5
        except Exception:
            pass
        try:
            h, c5 = _run_slsqp(h0, n, 500, False)
            return h, c5
        except Exception:
            pass
    else:
        try:
            h, c5 = _run_lbfgs(h0, n, 2000)
            return h, c5
        except Exception:
            pass
    return None, float('inf')

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    bh, bc = None, float('inf')
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float)
    except NameError:
        pass

    ns = []
    if parent is not None:
        ns.append(len(parent))
    for v in [70, 100, 140, 200, 300, 400]:
        if v not in ns:
            ns.append(v)

    for n in ns:
        if time.time() - t0 > budget_s - 15:
            break
        inits = []
        if parent is not None:
            inits.append(_interp(parent, n))
        n_r = 8 if n <= 200 else 4
        for _ in range(n_r):
            inits.append(_rbin(n, rng))
        for _ in range(n_r):
            inits.append(_rcont(n, rng))
        inits.append(_block_init(n, rng))
        for h0 in inits:
            h, c5 = _try_one(h0, n, t0, budget_s)
            if h is not None and c5 < bc:
                bh, bc = h, c5

    # Refinement: polish best with longer SLSQP
    if bh is not None and time.time() - t0 < budget_s - 10:
        for it in range(3):
            if time.time() - t0 > budget_s - 5:
                break
            try:
                h2, c52 = _run_slsqp(bh, len(bh), 2000, True)
                if c52 < bc:
                    bh, bc = h2, c52
                else:
                    break
            except Exception:
                break

    # Perturbation restarts on best
    if bh is not None and time.time() - t0 < budget_s - 15:
        bn = len(bh)
        for _ in range(10):
            if time.time() - t0 > budget_s - 10:
                break
            pert = _feasible(bh + 0.1 * rng.standard_normal(bn), bn)
            h, c5 = _try_one(pert, bn, t0, budget_s)
            if h is not None and c5 < bc:
                bh, bc = h, c5

    if bh is None:
        n = len(parent) if parent is not None else 70
        bh = _feasible(parent if parent is not None else _rbin(n, rng), n)
        bc = _c5(bh)

    return bh, bc, len(bh)
