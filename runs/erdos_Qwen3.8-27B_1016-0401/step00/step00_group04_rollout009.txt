def _fix_sum(h, target):
    """Fix sum of h to target, maintaining 0<=h<=1."""
    diff = target - h.sum()

        return
    if diff > 0:

        total_slack = slack.sum()
        if total_slack > 0:

            t4r = min(time.time() + (t4 - time.time()) * 0.6, t4)
            h2_opt, s4 = _optimize(h2, n2, target2, dx2, rng, t4r)
            if s4 < best_c5:
                best_c5 = s4
                best_h = h2_opt
                n = n2
            # Quick local search at new resolution
            t4l = min(time.time() + 2.0 * (t4 - time.time()), t4)
            if time.time() < t4l:
                h2_ls, s4b = _local_search(best_h, n, n / 2, 2.0 / n, rng, t4l)
                if s4b < best_c5:
                    best_c5 = s4b
                    best_h = h2_ls

    # Final: ensure correctness
    final_n = len(best_h)
    final_target = final_n / 2
    best_h = _fix_sum(best_h, final_target)
    final_dx = 2.0 / final_n
    final_c5 = _eval(best_h, final_dx)

    return (best_h, final_c5, final_n)
