import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c), c

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(y + mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(c, beta):
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def c5_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode="full")
    c = c_full * dx
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        d = k - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                grad[:L] += w[k] * (1.0 - h[d:])
                grad[d:] -= w[k] * h[:L]
        else:
            da = -d
            L = n - da
            if L > 0:
                grad[da:] += w[k] * (1.0 - h[:L])
                grad[:L] -= w[k] * h[da:]
    grad *= dx
    return grad

def max_subgradient(h, beta=None):
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode="full")
    c = c_full * dx
    grad = np.zeros(n)
    # Use argmax lag (or smooth if beta given)
    if beta is not None:
        return c5_smooth_gradient(h, beta)
    k_star = c.argmax()
    d = k_star - (n - 1)
    if d >= 0:
        L = n - d
        if L > 0:
            grad[:L] += (1.0 - h[d:])
            grad[d:] -= h[:L]
    else:
        da = -d
        L = n - da
        if L > 0:
            grad[da:] += (1.0 - h[:L])
            grad[:L] -= h[da:]
    grad *= dx
    return grad

        grad = c5_smooth_gradient(h, beta)
