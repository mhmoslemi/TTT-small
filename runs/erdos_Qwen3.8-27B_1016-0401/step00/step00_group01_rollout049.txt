```python
        h0[:n // 2] = 1.0
        if n % 2 == 1:
            h0[n // 2] = 0.5

    best_h = project_h(h0, n)
    best_c5 = compute_c5(best_h, n)

    def obj(x):
        return compute_c5(project_h(x, n), n)

    # Phase 1: Nelder-Mead from parent
    res = minimize(obj, best_h, method='Nelder-Mead',
                   options={'maxiter': 20000, 'adaptive': True,
                            'xatol': 1e-9, 'fatol': 1e-11})
    hc = project_h(res.x, n)
    c5c = compute_c5(hc, n)
    if c5c < best_c5:
        best_c5, best_h = c5c, hc

    if time.time() > deadline - 60:
        return best_h, best_c5, n

    # Phase 2: Powell refinement
    res = minimize(obj, best_h, method='Powell',
                   options={'maxiter': 5000, 'xtol': 1e-13, 'ftol': 1e-14,
                            'maxfev': 200000})
    hc = project_h(res.x, n)
    c5c = compute_c5(hc, n)
    if c5c < best_c5:
        best_c5, best_h = c5c, hc

    # Phase 3: Local search
    best_h, best_c5 = local_search(best_h, n, rng, deadline - 30)

    if time.time() > deadline - 20:
        return best_h, best_c5, n

    # Phase 4: Random restarts with local optimization
    for _ in range(500):
        if time.time() > deadline - 10:
            break
        s = rng.choice([0.1, 0.2, 0.3, 0.5, 0.8, 1.2])
        ht = project_h(best_h + rng.randn(n) * s, n)
        r = minimize(obj, ht, method='Nelder-Mead',
                     options={'maxiter': 3000, 'adaptive': True,
                              'xatol': 1e-8, 'fatol': 1e-10})
        hc = project_h(r.x, n)
        c5c = compute_c5(hc, n)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc
            best_h, best_c5 = local_search(best_h, n, rng, deadline - 10)

    # Final polish
    if time.time() < deadline - 1:
        best_h, best_c5 = local_search(best_h, n, rng, deadline)

    return best_h, best_c5, n
```
