def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode="full").max() * (2.0/n)

def hill_climb(h, deadline, rng):
    """Greedy hill climbing with single swaps."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stall = 0
    max_stall = 2000
    
    while time.time() < deadline:
        if stall >= max_stall:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < c5:
            c5 = new_c5
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1
    
    return h, c5

            break

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n)

def make_binary(h, n, rng):
    h = np.round(h).astype(float)
    target = n // 2
    s = int(h.sum())
    if s > target:
        idxs = rng.choice(np.where(h == 1)[0], size=s - target, replace=False)
        h[idxs] = 0
    elif s < target:
        idxs = rng.choice(np.where(h == 0)[0], size=target - s, replace=False)
        h[idxs] = 1
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h

def hill_climb(h, deadline, rng, stall_limit=5000):

    h = h.astype(float).copy()
    c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        if new_c5 < c5 - 1e-12:
            c5 = new_c5
            stall = 0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]

            h[i], h[j] = h[j], h[i]

    return h, c5

    cur_h = h.copy()

        cur_h, c5 = hill_climb(cur_h, deadline, rng)
        if c5 < best_c5:
            best_c5 = c5
