def project_to_polytope(x, target_sum):
    """Project x onto {y: 0<=y_i<=1, sum(y)=target_sum} via bisection on shift."""
    n = len(x)
    if abs(float(x.sum()) - target_sum) < 1e-13:
        return np.clip(x, 0.0, 1.0)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_factory(n, dx, target_sum, beta):
    """Return a smooth approximation of C5 using log-sum-exp."""
    def obj(p):
        h = project_to_polytope(p, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        scaled = (corr - m) * beta
        scaled = np.clip(scaled, -50, 50)
        return m + np.log(np.sum(np.exp(scaled))) / beta - np.log(len(corr)) / beta
    return obj

                best_h = h_new.copy()
        h_sub = subgradient_polish(best_h, dx, target_sum, n, steps=1500, lr=0.002)
        c5_sub = eval_c5(h_sub, dx)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Final feasibility check
    best_h = project_to_polytope(best_h, target_sum)
    best_c5 = eval_c5(best_h, dx)
    return best_h, best_c5, n
