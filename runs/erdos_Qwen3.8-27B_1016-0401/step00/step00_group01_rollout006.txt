def obj_exact(h):
    return compute_c5(h, len(h))

def sum_eq(h):
    return float(h.sum() - len(h) / 2.0)

def sum_eq_jac(h):
    return np.ones(len(h))

def time_remaining(t0, budget):
    return budget - (time.time() - t0)

def try_slsqp(h_start, n, beta, maxiter):
    global _BETA
    _BETA = beta
    bnd = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    res = minimize(obj_smooth, h_start, method='SLSQP', bounds=bnd,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, n)

def try_nm(h_start, n, maxiter):
    res = minimize(obj_exact, h_start, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10,
                            'fatol': 1e-12, 'adaptive': True})
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        n = 47
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)
    c5_best = compute_c5(h0, n)
    h_best = h0.copy()

    # Phase 1: SLSQP with increasing log-sum-exp smoothing
    for beta in [20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]:
        if time_remaining(t0, budget_s) < 35:
            break
        hc = try_slsqp(h0, n, beta, 3000)
        c5c = compute_c5(hc, n)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h0 = hc.copy()

    # Phase 2: Nelder-Mead polish with random restarts
    for trial in range(20):
        if time_remaining(t0, budget_s) < 20:
            break
        if trial == 0:
            h_try = h0.copy()
        elif trial == 1:
            h_try = h0 + rng.randn(n) * 0.01
        else:
            h_try = h0 + rng.randn(n) * (0.03 + 0.008 * trial)
        h_try = project_feasible(h_try, n)
        hc = try_nm(h_try, n, 2000)
        c5c = compute_c5(hc, n)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h0 = hc.copy()

    # Phase 3: SLSQP from best after NM (refetch with high beta)
    if time_remaining(t0, budget_s) > 40:
        hc = try_slsqp(h0, n, 2000.0, 2000)
        c5c = compute_c5(hc, n)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h0 = hc.copy()
        hc = try_nm(h0, n, 3000)
        c5c = compute_c5(hc, n)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h0 = hc.copy()

    # Phase 4: Try different grid sizes via interpolation
    n_list = [47, 49, 51, 53, 55, 43, 45, 41, 39]
    for n_try in n_list:
        if time_remaining(t0, budget_s) < 45:
            break
        if n_try == n:
            continue
        t_ref = np.linspace(0, 1, n)
        t_new = np.linspace(0, 1, n_try)
        h_interp = np.interp(t_new, t_ref, h0)
        h_interp = project_feasible(h_interp, n_try)
        hc = try_slsqp(h_interp, n_try, 200.0, 1500)
        c5c = compute_c5(hc, n_try)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            n = n_try
            h0 = hc.copy()
            hc = try_nm(h0, n_try, 1500)
            c5c = compute_c5(hc, n_try)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()
                h0 = hc.copy()

    # Phase 5: Final SLSQP + NM polish on best
    if time_remaining(t0, budget_s) > 30:
        hc = try_slsqp(h_best, len(h_best), 3000.0, 2000)
        c5c = compute_c5(hc, len(hc))
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
        hc = try_nm(h_best, len(h_best), 2000)
        c5c = compute_c5(hc, len(hc))
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best, len(h_best))
    return (h_best, c5_final, len(h_best))
