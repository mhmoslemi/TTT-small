import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

    c_max = np.max(corr)
    shifted = t * (corr - c_max)
    return c_max + np.log(np.sum(np.exp(shifted))) / t

    corr = np.correlate(h, 1 - h, mode="full")
    obj = soft_max(corr, t) * dx
    penalty = alpha * (np.sum(h) - len(h) / 2.0) ** 2
    return obj + penalty

    shifted = t * (corr - c_max)
    weights = np.exp(shifted) / np.exp(np.max(shifted))
