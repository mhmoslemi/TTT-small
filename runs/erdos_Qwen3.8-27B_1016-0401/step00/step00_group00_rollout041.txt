import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, n):
    S = n / 2.0
    if x.min() >= 0 and x.max() <= 1 and abs(x.sum() - S) < 1e-10:
        return x.copy()
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(x, n, p):
    h = project_feasible(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.abs(corr)
    return float(np.sum(c**p)**(1.0/p) * dx)

def obj_max(x, n):
    h = project_feasible(x, n)
    return compute_c5(h, n)

    t0 = time.time()
    result = minimize(obj_smooth, x0, args=(n, p), method='L-BFGS-B',
                      options={'maxiter': maxiter, 'eps': 1e-8, 'gtol': 1e-10})
    return result.x, result.fun, time.time() - t0
