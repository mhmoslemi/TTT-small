```python
import numpy as np
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_simplex(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    lo = -float(np.max(h)) - 2.0
    hi = 2.0 - float(np.min(h))
    for _ in range(70):
        mid = (lo + hi) / 2.0
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def gradient_softmax(h, weights, dx):
    n = len(h)
    shifts = np.arange(2 * n - 1) - (n - 1)
    t1 = np.zeros(n)
    t2 = np.zeros(n)
    t3 = np.zeros(n)
    for m in range(len(weights)):
        wm = float(weights[m])
        if wm < 1e-11:
            continue
        s = int(shifts[m])
        if s >= 0:
            t1[s:] += wm
            t2[s:] += wm * h[:n - s]
            t3[:n - s] += wm * h[s:]
        else:
            s2 = -s
            t1[:n - s2] += wm
            t2[:n - s2] += wm * h[s2:]
            t3[s2:] += wm * h[:n - s2]
    return (t1 - t2 - t3) * dx

def gradient_argmax(h, k, dx):
    n = len(h)
    shifts = np.arange(2 * n - 1) - (n - 1)
    s = int(shifts[k])
    t1 = np.zeros(n)
    t2 = np.zeros(n)
    t3 = np.zeros(n)
    if s >= 0:
        t1[s:] = 1.0
        t2[s:] = h[:n - s]
        t3[:n - s] = h[s:]
    else:
        s2 = -s
        t1[:n - s2] = 1.0
        t2[:n - s2] = h[s2:]
        t3[s2:] = h[:n - s2]
    return (t1 - t2 - t3) * dx

def optimize(h, n_iter, time_limit, beta_start=10.0, beta_end=500.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project_box_simplex(h, target).copy()
    best_c5 = compute_c5_value(h)
    best_h = h.copy()
    lr = 0.3
    beta = beta_start
    stall = 0
    
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        if it % 20 == 0:
            # Soft-max
            scaled = beta * corr * dx
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
            grad = gradient_softmax(h, w, dx)
        else:
            # Argmax
            k = int(np.argmax(corr))
            grad = gradient_argmax(h, k, dx)
        
        h_new = h - lr * grad
        h_new = project_box_simplex(h_new, target)
        c5_new = compute_c5_value(h_new)
        
        if c5_new < best_c5 - 1e-7:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 1.5)
            stall = 0
        elif c5_new < best_c5:
            h = h_new
            lr = min(lr * 1.02, 1.5)
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 30:
                # Perturb
                h = project_box_simplex(h + 0.2 * np.random.randn(n), target)
                best_c5 = compute_c5_value(h)
                best_h = h.copy()
                lr = 0.3
                stall = 0
        
        # Anneal
        frac = (it + 1) / max(n_iter, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
    
    return best_h, best_c5

def make_cosine_init(n, freq=1.0, phase=0.0, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n)
    h = 0.5 + 0.35 * np.cos(np.pi * freq * x + phase)
    h += 0.08 * rng.randn(n)
    return project_box_simplex(h, n / 2.0)

            h_up = project_box_simplex(h_up, n_new / 2.0)
            t_lim = min(t0 + budget_s - 10, time.time() + remaining * 0.35)
            h2, c5_2 = optimize(h_up, 3000, t_lim)
```
