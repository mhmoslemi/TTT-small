def _c5_raw(h):
    """Return raw correlation (before dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_value(h, n):
    dx = 2.0 / n
    return _c5_raw(h).max() * dx

def _smooth_max_grad(c, T):
    """Return (smooth_max_value, weights) for smooth approximation of max."""
    m = c.max()
    exp_c = np.exp((c - m) / T)
    s = exp_c.sum()
    w = exp_c / s
    return m + T * np.log(s), w

def _project_gradient(grad, n):
    """Project gradient onto sum-zero subspace."""
    return grad - np.mean(grad) * np.ones(n)

def _smooth_grad_descent_step(h, lr, T, n):
    """One step of smooth gradient descent. Returns new h and the raw c5."""
    c = _c5_raw(h)
    _, w = _smooth_max_grad(c, T)
    
    # grad[j] = sum_{r=0}^{n-1} w[j+r] * (1-2*h[r])
    grad = np.zeros(n)
    one_minus_two_h = 1.0 - 2.0 * h
    for j in range(n):
        grad[j] = np.dot(w[j:j+n], one_minus_two_h)
    
    grad = _project_gradient(grad, n)
    h_new = h - lr * grad
    h_new = _enforce_sum(h_new, n / 2.0)
    return h_new, c.max()

def _coordinate_descent(h, n, rng, n_pairs=100):
    """Try random coordinate pair swaps to improve c5."""
    c_best = _c5_raw(h).max()
    for _ in range(n_pairs):
        i, j = rng.choice(n, 2, replace=False)
        # Try various deltas

            if c_new < c_best:
                c_best = c_new
            else:

    final_c5 = _c5_val(best_h, len(best_h))
    return (best_h, final_c5, len(best_h))
