            if tl < 15:
                break
            h_pert = perturbed_copy(best_h, rng, frac=0.12)
            h_pert = ensure_binary(h_pert, n, rng)
            alloc = min(time_left(t0, budget_s) * 0.12, 50.0)
            h3, s3 = sa_search(h_pert, rng, time.time() + alloc, 0.01, 0.99999, 1e-6)
            if s3 < best_s:
                best_s = s3
                best_h = h3

    # Phase 5: One last low-temperature polish if time remains
    tl = time_left(t0, budget_s)
    if tl > 20:
        alloc = min(tl * 0.8, 120.0)
        best_h, best_s = sa_search(best_h, rng, time.time() + alloc, 0.002, 0.999999, 1e-8)

    return best_h, best_s, len(best_h)
