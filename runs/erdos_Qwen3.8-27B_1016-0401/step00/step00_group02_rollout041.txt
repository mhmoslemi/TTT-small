import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_c5(h, dx, temp=0.02):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -50, 50)
    return float((m + temp * np.log(np.sum(np.exp(shifted)))) * dx)

def project_to_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def build_obj_fn(n, dx, target_sum):
    def _obj(z):
        h = np.clip(z, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)

        val = (m + t * np.log(np.sum(np.exp(sh)))) * dx
        pen = 300.0 * (np.sum(h) - target_sum) ** 2
        return val + pen
    return _obj
