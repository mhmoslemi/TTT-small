            rr = -r
            grad[:n - rr] += w * (1.0 - h[rr:])
            grad[rr:] -= w * h[:n - rr]
    return grad


def optimize_h(h_init, n, max_iter, deadline, tau0=2.0, momentum=0.9):
    h = project_h(h_init.copy(), n)
    tau = tau0
    lr = 1.0
    vel = np.zeros(n)
    best_c5 = get_c5(h, n)
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        c5 = corr.max()
        if c5 < best_c5:
            best_c5 = c5
            patience = 0
        else:
            patience += 1
        if patience > 500:
            break
        weights = smooth_max_weights(corr, tau)
        grad = gradient_c5(corr, weights, h, n)
        vel = momentum * vel - lr * grad
        h_new = project_h(h + vel, n)
        c_new = get_c5(h_new, n)
        if c_new <= c5 + 1e-12:
            h = h_new
            lr = min(lr * 1.05, 5.0)
        else:
            lr *= 0.5
            vel *= 0.5
            if lr < 1e-10:
                break
        if it % 300 == 299:
            tau = max(0.3, tau * 0.85)
    return h, best_c5


def make_binary_starter(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) > 0.5).astype(float)
    ones_needed = n // 2
    if n % 2 == 0:
        idx = np.where(h == 1)[0]
        if len(idx) > ones_needed:
            remove = rng.choice(idx, len(idx) - ones_needed, replace=False)
            h[remove] = 0
        elif len(idx) < ones_needed:
            idx0 = np.where(h == 0)[0]
            add = rng.choice(idx0, ones_needed - len(idx), replace=False)
            h[add] = 1
    else:
        h = (rng.rand(n) > 0.5).astype(float)
        h = project_h(h, n)
    return h


def make_block_starter(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 0:
        h[n // 2 - 1] = 0.5
        h[n // 2] = 0.5
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    best_h = None
    best_c5 = 1e10
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    candidates = []
    if h0 is not None:
        candidates.append(('init', h0, len(h0)))
        n_int = 2 * len(h0) - 1
        x_old = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, n_int)
        candidates.append(('interp', np.interp(x_new, x_old, h0), n_int))

    candidates.append(('bin47', make_binary_starter(47, seed), 47))
    candidates.append(('bin94', make_binary_starter(94, seed + 1), 94))
    candidates.append(('blk47', make_block_starter(47), 47))
    candidates.append(('blk94', make_block_starter(94), 94))

    rng = np.random.RandomState(seed + 100)
    for trial in range(4):
        n = 47 + trial * 47
        h_rand = project_h(rng.rand(n), n)
        candidates.append(('rand%d' % n, h_rand, n))

    results = []
    for name, h_cand, n in candidates:
        if time.time() > deadline - 20:
            break
        h_opt, c5_opt = optimize_h(h_cand, n, 4000, deadline, tau0=2.0)
        c5_final = get_c5(h_opt, n)
        results.append((c5_final, h_opt, n, name))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_opt
            best_n = n

    results.sort(key=lambda x: x[0])

    if time.time() < deadline - 30:
        h_refine, c5_refine = optimize_h(best_h, best_n, 10000, deadline, tau0=1.0, momentum=0.7)
        c5_rf = get_c5(h_refine, best_n)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_refine

    if time.time() < deadline - 15:
        for shift_n in [best_n + 1, 2 * best_n - 1]:
            if shift_n > 500:
                break
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, shift_n)
            h_interp = np.interp(x_new, x_old, best_h)
            h_opt2, _ = optimize_h(h_interp, shift_n, 3000, deadline, tau0=1.5)
            c5_2 = get_c5(h_opt2, shift_n)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_opt2
                best_n = shift_n

    best_h = project_h(best_h, best_n)
    best_c5 = get_c5(best_h, best_n)
    return best_h, best_c5, best_n
