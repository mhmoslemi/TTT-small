def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        nm = np.count_nonzero(mask)
        if nm == 0:
            break
        h[mask] += diff / nm
    return np.clip(h, 0.0, 1.0)

def c5_eval(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = corr.max()
    if c_max < 1e-15:
        return 0.0, np.zeros(n)
    shifted = T * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exps = np.exp(shifted)
    val = c_max + np.log(exps.sum()) / T
    p = exps / exps.sum()
    omh = 1.0 - h
    nc = len(corr)
    grad = np.zeros(n)
    for idx in range(nc):
        s = idx - (n - 1)
        wl = p[idx]
        if wl < 1e-15:
            continue
        if s >= 0:
            if s < n:
                grad[s:] += wl * omh[:n-s]
            if n > s:
                grad[:n-s] += wl * (-h[s:])
        else:
            a = -s
            if n > a:
                grad[:n-a] += wl * omh[a:]
            if a < n:
                grad[a:] += wl * (-h[:n-a])
    grad *= dx
    return val, grad

def _obj(x):
    return smooth_c5(x, _SMOOTH_T)[0]

def _jac(x):
    return smooth_c5(x, _SMOOTH_T)[1]

def _conn_fun(x):
    return np.sum(x) - len(x) / 2.0

def _conn_jac(x):
    return np.ones(len(x))

    global _SMOOTH_T
    _SMOOTH_T = T
    n = len(h0)
    cons = [{'type': 'eq', 'fun': _conn_fun, 'jac': _conn_jac}]
    bounds = list(zip(np.zeros(n), np.ones(n)))

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h0

    f_curr = c5_eval(h)
    alpha = 0.05

            if tl() < budget_s * 0.12:
                break
            h_curr = h_init.copy()
            for T in [10.0, 30.0, 80.0]:
                if tl() < budget_s * 0.08:
                    break
                h_opt = slsqp_optimize(h_curr, T, maxiter=120)
                c5_opt = c5_eval(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_curr = h_opt

    # Phase 4: Perturbation + re-optimize on best
    if tl() > budget_s * 0.2:
        n_best = len(best_h)
        for _p in range(3):
            if tl() < budget_s * 0.08:
                break
            pert = best_h + 0.05 * rng.standard_normal(n_best)
            pert = project_feasible(pert, n_best)
            h_opt = slsqp_optimize(pert, 30.0, maxiter=150)
            c5_opt = c5_eval(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Final SLSQP + gradient polish
    if tl() > budget_s * 0.08:
        for T in [50.0, 100.0]:
            if tl() < budget_s * 0.04:
                break
            h_opt = slsqp_optimize(best_h, T, maxiter=200)
            c5_opt = c5_eval(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        if tl() > budget_s * 0.04:
            h_pg, c5_pg = proj_grad_optimize(best_h, n_steps=2000)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = c5_eval(best_h)
    return best_h, best_c5, n_final
