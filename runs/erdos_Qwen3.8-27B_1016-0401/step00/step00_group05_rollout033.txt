import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * 2.0 / n)

def _fake(h, bt, pn):
    n = len(h)
    c = np.correlate(h, 1.0 - h, "full")
    v = c * bt
    m = v.max()
    sm = (m + np.log(np.exp(v - m).sum())) / bt
    return sm * 2.0 / n + pn * (h.sum() - 0.5 * n) * (h.sum() - 0.5 * n)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = 0.5 * n
    for _ in range(200):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

class _O:
    def __init__(self, b, p):
        self.b, self.p = b, p
    def __call__(self, h):
        return _fake(h, self.b, self.p)

def _ls(h, n, rs, it):
    bv = _eval(h)
    hb = h.copy()
    sc = 0.1
    for i in range(it):
        c = sc * max(0.05, 1.0 - 0.4 * i / max(it, 1))
        hn = _proj(hb + c * rs.randn(n), n)
        v = _eval(hn)
        if v < bv:
            bv, hb = v, hn
            sc = max(c, 0.003)
    return hb, bv

def run(seed=42, budget_s=1100, **kw):

    rs = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:

    bc = _eval(h)
    bh = h.copy()
    bn = n
    bd = [(0.0, 1.0)] * n

    # SLSQP annealed
    for b in [5, 15, 40, 100, 200]:
        if time.time() - t0 > budget_s - 100:
            break
        r = minimize(_O(b, 1000), bh, method='SLSQP', bounds=bd,
                     options={'maxiter': 500, 'ftol': 1e-12})

        cr = _eval(hr)
        if cr < bc:

    for i in range(5):

        st = bh if i == 0 else _proj(bh + 0.03 * rs.randn(n), n)
        r = minimize(_O(150, 1000), st, method='Nelder-Mead', bounds=bd,
                     options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11})
        hr = _proj(r.x, n)
        cr = _eval(hr)
        if cr < bc:
            bc, bh = cr, hr

    # Local search
    if time.time() - t0 < budget_s - 30:

        bn = n

        ht = np.interp(np.linspace(0, 1, nt), np.linspace(0, 1, n), bh)

        ct = _eval(ht)
        if nt <= 100 and time.time() - t0 < budget_s - 60:

                         bounds=[(0.0, 1.0)] * nt,
                         options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-11})
            ht = _proj(r.x, nt)
            ct = _eval(ht)
        if ct < bc:
            bc, bh, bn = ct, ht, nt

    # Final polish
    if time.time() - t0 < budget_s - 5:
        bh, bc = _ls(bh, bn, rs, 3000)

    bh = _proj(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)
