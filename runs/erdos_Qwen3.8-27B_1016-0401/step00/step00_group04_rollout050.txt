def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # First term: (1-h[n-1-k+ℓ]) for ℓ in [lo, hi]
        # Second term: -h[ℓ+k-n+1] for ℓ in [lo+n-1-k, hi+n-1-k]
        
        # First term
        idx1 = np.arange(lo, hi + 1)
        partner1 = n - 1 - k + idx1  # = j_{ℓ}
        grad[idx1] += wk * (1.0 - h[partner1])
        
        # Second term
        lo2 = lo + n - 1 - k
        hi2 = hi + n - 1 - k
        # Clamp to [0, n-1]
        lo2c = max(0, lo2)
        hi2c = min(n - 1, hi2)
        if lo2c <= hi2c:
            idx2 = np.arange(lo2c, hi2c + 1)
            partner2 = idx2 + k - n + 1  # = i where j_i = ℓ
            grad[idx2] -= wk * h[partner2]
    
    return grad * dx

import numpy as np
import time
from scipy.optimize import minimize

def project_onto_box_sum(v, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx1 = np.arange(lo, hi + 1)
        partner1 = n - 1 - k + idx1
        grad[idx1] += wk * (1.0 - h[partner1])
        lo2 = max(0, lo + n - 1 - k)
        hi2 = min(n - 1, hi + n - 1 - k)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            partner2 = idx2 + k - n + 1
            grad[idx2] -= wk * h[partner2]
    return grad * dx

def pgd(h0, alpha, lr0, n_iter, decay=0.999, momentum=0.85):
    n = len(h0)
    target = n / 2.0
    h = project_onto_box_sum(h0.copy(), target)
    v = np.zeros(n)
    lr = lr0
    best_h = h.copy()
    best_s = compute_score(h)
    for it in range(n_iter):
        g = smooth_grad(h, alpha)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_onto_box_sum(h_new, target)
        s = compute_score(h_new)
        if s < best_s:
            best_s = s
            best_h = h_new.copy()
        h = h_new
        lr *= decay
    return best_h, best_s

def coordinate_refine(h, n_sweeps=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_s = compute_score(h)
    best_h = h.copy()
    steps = np.array([0.01, -0.01, 0.03, -0.03, 0.05, -0.05, 0.1, -0.1])
    for sw in range(n_sweeps):
        improved = False
        for i in range(n):
            orig = h[i]
            for d in steps:
                val = orig + d
                if val < 0.0 or val > 1.0:
                    continue
                h[i] = val
                s = compute_score(h)
                if s < best_s:
                    best_s = s
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = orig
        h = project_onto_box_sum(h, target)
        s = compute_score(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
            improved = True
        if not improved and sw > 5:
            break
    return best_h, best_s
