import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new / 2.0)

def smooth_optimize(h, n, time_limit):
    t0 = time.time()
    target = n / 2.0
    for t in [40.0, 80.0, 160.0, 400.0]:
        if time.time() - t0 > time_limit:
            break
        def obj(hv):
            corr = np.correlate(hv, 1.0 - hv, mode="full")
            return logsumexp(t * corr) / t
        cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target,
                 'jac': lambda hv: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 400, 'ftol': 1e-18})
            h_cand = fix_sum(np.clip(res.x, 0, 1), target)
            if evaluate_h(h_cand) <= evaluate_h(h) + 1e-14:
                h = h_cand
        except:
            pass
    return h

def hill_climb(h, n, n_iter, time_limit):

    target = n / 2.0

        idx = np.random.choice(n, k, replace=False)
        saved = h[idx].copy()
        h[idx] += np.random.uniform(-0.04, 0.04, k)
        h[idx] = np.clip(h[idx], 0, 1)
        h = fix_sum(h, target)
        c5 = evaluate_h(h)
        if c5 > best_c5 + 1e-12:
            h[idx] = saved
            h = fix_sum(h, target)
        else:
            best_c5 = c5
    return h, evaluate_h(h)

    t0 = time.time()
    target = n // 2
    rng = np.random.RandomState(0)
    h = np.zeros(n)
    h[rng.choice(n, target, replace=False)] = 1.0

    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()

            best = c5
            ones[i], zeros[i] = zeros[i], ones[i]
            if i == j:

            ones = sorted(set(ones + [j] - [i]))
            zeros = sorted(set(zeros + [i] - [j]))
        else:
            h[i], h[j] = 1.0, 0.0
    return h, best
