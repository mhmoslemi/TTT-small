import time

def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target):
    """Project h onto {0≤x≤1, sum(x)=target} using bisection."""
    n = len(h)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def c5_score(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(corr.max() * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    k = int(np.argmax(corr))
    m = k - (n-1)  # lag
    grad = np.zeros(n)
    if m >= 0:
        # f = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        # d/dh[j]:
        #   j in [m, n-1]: +(1-h[j-m])
        #   j in [0, n-1-m]: -h[j+m]
        idx_a = np.arange(m, n)
        grad[idx_a] += (1.0 - h[idx_a - m])
        idx_b = np.arange(0, n - m)
        grad[idx_b] -= h[idx_b + m]
    else:
        jp = -m
        # f = sum_{p=0}^{n-1-jp} h[p]*(1-h[p+jp])
        # d/dh[k]:
        #   k in [0, n-1-jp]: +(1-h[k+jp])
        #   k in [jp, n-1]: -h[k-jp]
        idx_a = np.arange(0, n - jp)
        grad[idx_a] += (1.0 - h[idx_a + jp])
        idx_b = np.arange(jp, n)
        grad[idx_b] -= h[idx_b - jp]
    return dx * grad

def pgd_step(h, target, lr):
    g = c5_gradient(h)
    h_new = h - lr * g
    return project_onto_constraint(h_new, target)

            grad[idx_b] -= h[idx_b - jp]
    return dx * (grad * weights[:, None]).sum(axis=1)

def pgd_optimize(h, target, lr, n_iters, time_limit, use_smoothed=False, tau=50.0):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = c5_score(h)
    cur_c5 = best_c5
    stall = 0
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        if use_smoothed:
            g = smoothed_gradient(h, tau=tau)
        else:
            g = c5_gradient(h)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target)
        c = c5_score(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 200:
            lr *= 0.7
            stall = 0
    return best_h, best_c5

def make_initial(n):
    h = np.full(n, 0.5)
    idx = np.arange(n)
    h[idx % 2 == 0] = 0.8
    h[idx % 2 == 1] = 0.2
    target = n / 2.0
    return project_onto_constraint(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = make_initial(42)

    n = len(h0)
    target = n / 2.0
    h0 = project_onto_constraint(h0, target)

    best_h = h0.copy()
    best_c5 = c5_score(h0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Sharp gradient descent on base grid
    h = h0.copy()
    lr = 0.15
    for phase in range(20):
        if elapsed() > budget_s * 0.35:
            break
        t_lim = min(30.0, budget_s * 0.35 - elapsed())
        if t_lim < 5:
            break
        h, c = pgd_optimize(h, target, lr, 20000, t_lim, use_smoothed=False)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        lr *= 0.75

    # Phase 2: Smoothed optimization (handle flat regions / multi-active lags)
    h = best_h.copy()
    for tau in [50.0, 30.0, 15.0, 8.0]:
        if elapsed() > budget_s * 0.65:
            break
        t_lim = min(40.0, budget_s * 0.65 - elapsed())
        if t_lim < 5:
            break
        lr = 0.05
        for phase in range(5):
            h, c = pgd_optimize(h, target, lr, 15000, t_lim, use_smoothed=True, tau=tau)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
            lr *= 0.7

    # Phase 3: Random restarts
    for rs in range(4):
        if elapsed() > budget_s * 0.8:
            break
        scale = 0.05 * (rs + 1)
        h = best_h + scale * np.random.randn(n)
        h = project_onto_constraint(h, target)
        lr = 0.08
        t_lim = min(20.0, budget_s * 0.8 - elapsed())
        if t_lim < 3:
            break
        h, c = pgd_optimize(h, target, lr, 10000, t_lim, use_smoothed=False)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 4: Grid refinement to 2n
    if elapsed() < budget_s * 0.85:
        n2 = n * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_onto_constraint(h2, target2)
        c2 = c5_score(h2)
        lr = 0.03
        t_lim = min(budget_s * 0.1, budget_s - elapsed() - 20)
        if t_lim > 10:
            h2, c2 = pgd_optimize(h2, target2, lr, 30000, t_lim, use_smoothed=False)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
        # Second pass with smaller lr
        if elapsed() < budget_s * 0.95:
            t_lim2 = min(30.0, budget_s * 0.95 - elapsed())
            if t_lim2 > 5:
                h2 = best_h if len(best_h) == n2 else h2
                h2, c2 = pgd_optimize(h2, target2, 0.01, 20000, t_lim2, use_smoothed=False)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()

    # Phase 5: Final polish on the best grid
    n_f = len(best_h)
    tgt_f = n_f / 2.0
    best_h = project_onto_constraint(best_h, tgt_f)
    t_lim = min(20.0, budget_s - elapsed() - 5)
    if t_lim > 2:
        best_h, c = pgd_optimize(best_h, tgt_f, 0.005, 5000, t_lim, use_smoothed=False)
        if c < best_c5:
            best_c5 = c
            best_h = h  # careful: pgd returns best
    best_h = project_onto_constraint(best_h, n_f / 2.0)
    best_c5 = c5_score(best_h)

    return best_h, best_c5, n_f
