                h = proj(h + np.random.randn(n) * 0.025, ts)
                c5 = c5_eval(h, dx)
                lr = 0.04
                stall = 0
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    run._t_limit = t0 + budget_s - 8
    t_limit = run._t_limit

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h0 = None

    best_h = None
    best_c5 = 1.0

    # Phase 1: Optimize from parent (40% of time)
    if h0 is not None and len(h0) >= 10:
        n0 = len(h0)
        dx0 = 2.0 / n0
        ts0 = n0 / 2.0
        h0 = np.clip(h0, 0, 1)
        h0 = proj(h0, ts0)
        c5_0 = c5_eval(h0, dx0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()

        t_phase1 = t0 + (t_limit - t0) * 0.35
        h_r, c5_r = optimize_soft(h0, n0, t_phase1, 0.04)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

        # Binary local search on parent
        t_phase1b = min(t0 + (t_limit - t0) * 0.42, t_limit - 60)
        h_b, c5_b = try_binary_local(h0, n0, dx0, 2000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = c5_eval(best_h, 2.0 / 70)

    # Phase 2: Try other n values (30% of time)
    n_options = [70, 100, 140]
    n_opt = len(best_h)
    if n_opt not in n_options:
        n_options.insert(0, n_opt)
    seen_n = set()
    phase2_end = t0 + (t_limit - t0) * 0.70
    for n in n_options:
        if time.time() > phase2_end or time.time() > t_limit - 30:
            break
        if n in seen_n or n < 20 or n > 300:
            continue
        seen_n.add(n)
        h_init = best_h.copy()
        if n != len(h_init):
            h_init = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, len(best_h)),
                best_h
            )
        chunk_end = min(time.time() + 90, phase2_end)
        h_r, c5_r = optimize_soft(h_init, n, chunk_end, 0.035)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        # Binary polish
        dx_n = 2.0 / n
        h_b, c5_b = try_binary_local(h_r, n, dx_n, 1000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Refine best with perturbations (remaining time)
    while time.time() < t_limit - 5:
        n = len(best_h)
        h_pert = best_h + np.random.randn(n) * 0.012
        chunk_end = min(time.time() + 45, t_limit - 5)
        h_r, c5_r = optimize_soft(h_pert, n, chunk_end, 0.03)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        # Quick binary check
        dx_n = 2.0 / n
        h_b, c5_b = try_binary_local(h_r, n, dx_n, 500)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = proj(best_h, ts_final)
    best_c5 = c5_eval(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
