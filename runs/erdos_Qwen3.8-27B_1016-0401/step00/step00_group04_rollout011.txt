import numpy as np
import time


def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def local_search_binary(h_init, time_limit, seed, T0=0.006, cool=0.99999):
    rng = np.random.default_rng(seed)
    n = len(h_init)
    h = np.round(np.asarray(h_init, dtype=np.float64)).copy()
    s = int(np.sum(h))
    target = n // 2
    if s > target:
        ones = np.flatnonzero(h == 1)
        h[ones[rng.choice(len(ones), s - target, replace=False)]] = 0
    elif s < target:
        zeros = np.flatnonzero(h == 0)
        h[zeros[rng.choice(len(zeros), target - s, replace=False)]] = 1

    cur_sc = compute_score(h)
    best_sc = cur_sc
    best_h = h.copy()
    T = T0
    no_imp = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0
        h[j] = 1
        ns = compute_score(h)
        d = ns - cur_sc
        if d <= 0 or (T > 1e-10 and rng.random() < np.exp(-d / T)):
            cur_sc = ns
            if ns < best_sc - 1e-15:
                best_sc = ns
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            h[i] = 1
            h[j] = 0
            no_imp += 1
        T *= cool
        if no_imp > 80000:
            break
    return best_h, best_sc

    return np.repeat(h, 2)


def kron_start(n, seed, alpha_override=None):

    if s > target:
        ones = np.flatnonzero(h == 1)

    elif s < target:
        zeros = np.flatnonzero(h == 0)
        h[zeros[rng.choice(len(zeros), target - s, replace=False)]] = 1
    return h


def segment_swap_move(h, rng):
    n = len(h)
    h2 = h.copy()
    a = int(rng.integers(0, n - 2))
    ln = int(rng.integers(2, min(6, n - a)))
    b = int(rng.integers(0, n - ln))
    seg = h2[a:a + ln].copy()
    h2[a:a + ln] = h2[b:b + ln]
    h2[b:b + ln] = seg
    return h2
