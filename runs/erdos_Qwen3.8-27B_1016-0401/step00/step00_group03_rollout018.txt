def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def smooth_obj_and_grad(h, dx, t, penalty):
    m = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    tc = t * corr * dx
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    weights = exp_tc / np.sum(exp_tc)
    obj = (tc_max + np.log(np.sum(exp_tc))) / t
    s = np.sum(h) - m / 2.0
    obj += penalty * s * s

    grad = np.zeros(m)
    for k in range(-m + 1, m):
        w = weights[k + m - 1]
        if w < 1e-10:
            continue
        lo1 = max(0, k)
        hi1 = min(m - 1, m - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 - k:hi1 + 1 - k])
        lo2 = max(0, -k)
        hi2 = min(m - 1, m - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + k:hi2 + 1 + k]
    grad += 2.0 * penalty * s
    return obj, grad
