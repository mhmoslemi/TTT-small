import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    sh = np.clip(beta * (corr - mx), -500.0, 0.0)
    ev = np.exp(sh)
    sv = np.sum(ev)
    val = mx + np.log(sv) / beta
    w = ev / sv
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] += wk * (-h[lag:])
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] += wk * (-h[:n - L])
    return val, grad

def hard_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] += (-h[lag:])
    else:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] += (-h[:n - L])
    return float(corr[k_star]), grad

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smoo_obj(u, n, target, beta, pen):
    val, g = smooth_c5_grad(u, beta)
    sp = np.sum(u) - target
    return val + 0.5 * pen * sp * sp, g + pen * sp / n * np.ones(n)

def hard_obj(u, n, target, pen):
    val, g = hard_c5_grad(u)
    sp = np.sum(u) - target
    return val + 0.5 * pen * sp * sp, g + pen * sp / n * np.ones(n)

def make_candidate_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def make_candidate_sine(n, rng):
    x = np.linspace(0, 2.0 * np.pi * (1 + rng.randint(3)), n, endpoint=False)
    ph = rng.uniform(0, 2 * np.pi)
    h = np.clip(0.5 + 0.3 * np.sin(x + ph) + 0.15 * np.sin(2 * x), 0, 1)
    return h

def optimize_single(h_init, t0, budget_s):
    n = len(h_init)
    target = n / 2.0
    h_work = project_sum(np.clip(h_init, 0.0, 1.0), target)
    bounds = [(0.0, 1.0)] * n
    best_local = h_work.copy()
    best_local_c5 = compute_c5(h_work)

    for beta in [5, 10, 25, 60, 150, 400, 800]:
        if time.time() - t0 > budget_s * 0.82:
            break
        try:
            res = minimize(smoo_obj, h_work, args=(n, target, beta, 800.0),
                           jac=True, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12})
            h_work = project_sum(np.clip(res.x, 0.0, 1.0), target)
            c5_now = compute_c5(h_work)
            if c5_now < best_local_c5:
                best_local_c5 = c5_now
                best_local = h_work.copy()
        except Exception:
            pass

    try:
        res = minimize(hard_obj, h_work, args=(n, target, 800.0),
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-16, 'gtol': 1e-12})
        h_work = project_sum(np.clip(res.x, 0.0, 1.0), target)
        c5_now = compute_c5(h_work)
        if c5_now < best_local_c5:
            best_local_c5 = c5_now
            best_local = h_work.copy()
    except Exception:
        pass

    # Coordinate polish: two-point swaps
    h_polish = best_local.copy()
    c5_polish = best_local_c5
    for _ in range(3000):
        if time.time() - t0 > budget_s * 0.86:
            break
        i = rng_coord.randint(n)
        j = rng_coord.randint(n)
        if i == j:
            continue
        oi, oj = h_polish[i], h_polish[j]
        improved = False
        for d in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, -0.02, -0.05, -0.1, -0.15, -0.2, -0.3]:
            ni = oi + d
            nj = oj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_polish[i], h_polish[j] = ni, nj
            c5_try = compute_c5(h_polish)
            if c5_try < c5_polish - 1e-12:
                c5_polish = c5_try
                improved = True
                break
        if not improved:
            h_polish[i], h_polish[j] = oi, oj
        h_polish = project_sum(h_polish, target)

    if c5_polish < best_local_c5:
        best_local = h_polish.copy()
        best_local_c5 = c5_polish

    return best_local, best_local_c5
