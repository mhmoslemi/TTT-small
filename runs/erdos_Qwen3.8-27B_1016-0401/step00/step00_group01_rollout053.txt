    if m >= 0:

        idx1 = np.arange(0, n-m)
        g[idx1] += (1 - h[idx1 + m])
        # Second term: j in [m, n-1]
        idx2 = np.arange(m, n)
        g[idx2] += -h[idx2 - m]
    else:

        # First term: j in [p, n-1]
        idx1 = np.arange(p, n)
        g[idx1] += (1 - h[idx1 + m])  # idx1 + m = idx1 - p
        # Second term: j in [0, n-1-p]
        idx2 = np.arange(0, n-p)
        g[idx2] += -h[idx2 - m]  # idx2 - m = idx2 + p
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, m):
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        idx1 = np.arange(0, n - m)
        g[idx1] += 1.0 - h[idx1 + m]
        idx2 = np.arange(m, n)
        g[idx2] -= h[idx2 - m]
    else:
        p = -m
        idx1 = np.arange(p, n)
        g[idx1] += 1.0 - h[idx1 - p]
        idx2 = np.arange(0, n - p)
        g[idx2] -= h[idx2 + p]
    return g

def grad_max(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    tol = max(1e-10, mx * 1e-6)
    active = np.where(corr >= mx - tol)[0]
    w = corr[active] - (mx - tol)
    ws = w.sum()
    if ws > 0:
        w = w / ws
    else:
        w = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        m = k - (n - 1)
        grad += w[idx] * grad_lag(h, m)
    return grad, mx

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
        else:
            h = np.clip(h + diff / len(h), 0, 1)
    else:
        room = h
        tot = room.sum()
        if tot > 1e-14:
            h -= room * ((-diff) / tot)
        else:
            h = np.clip(h + diff / len(h), 0, 1)
    h = np.clip(h, 0.0, 1.0)
    d2 = target - h.sum()
    if abs(d2) > 1e-14:
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if mask.sum() > 0:
            h[mask] += d2 / mask.sum()
        else:
            h += d2 / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    best_val = compute_c5(h)
    alpha = 0.5
    dx = 2.0 / n
    m = n - 1
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        val = mx * dx
        if val < best_val:
            best = h.copy()
            best_val = val
        tol_a = max(1e-10, mx * 1e-5)
        active = np.where(corr >= mx - tol_a)[0]
        w = corr[active] - (mx - tol_a)
        ws = w.sum()
        if ws > 0:
            w = w / ws
        else:
            w = np.ones(len(active)) / len(active)
        grad = np.zeros(n)
        for idx, k in enumerate(active):
            mm = k - m
            if mm >= 0:
                i1 = np.arange(0, n - mm)
                grad[i1] += w[idx] * (1.0 - h[i1 + mm])
                i2 = np.arange(mm, n)
                grad[i2] -= w[idx] * h[i2 - mm]
            else:
                p = -mm
                i1 = np.arange(p, n)
                grad[i1] += w[idx] * (1.0 - h[i1 - p])
                i2 = np.arange(0, n - p)
                grad[i2] -= w[idx] * h[i2 + p]
        a = alpha
        h_new = h
        val_new = val

            h = h_new
            alpha = min(alpha * 1.05, 2.0)
        else:
            alpha = a
        if alpha < 1e-12:
            break
    return h, best_val

def coord_descent(h, n, max_rounds=80, time_limit=None, t0=None):
    h = h.copy()
    target = n / 2.0
    best_val = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        improved = False
        for j in range(n):
            best_hj = h[j]
            best_v = best_val
            for frac in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0,
                         max(0, h[j]-0.15), min(1, h[j]+0.15),
                         max(0, h[j]-0.05), min(1, h[j]+0.05)]:
                h2 = h.copy()
                h2[j] = frac
                h2 = project_h(h2, target)
                v = compute_c5(h2)
                if v < best_v - 1e-12:
                    best_v = v
                    best_hj = h2[j]
                    h2_save = h2.copy()
            if best_v < best_val - 1e-12:
                h = h2_save
                best_val = best_v
                improved = True
        if not improved:
            break
    return h, best_val
