        if s > S:

    hp = np.clip(h - lam, 0.0, 1.0)
    resid = S - hp.sum()
    if abs(resid) > 1e-11:

        if free.sum() > 0:

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode="full")

def _try_optimize(obj, x_start, maxiter, time_budget, t0):
    """Run Nelder-Mead with time check, return best x and f."""
    best_x = x_start.copy()
    best_f = obj(x_start)
    # Wrap objective with time check
    state = {'calls': 0}

    def timed_obj(x):
        state['calls'] += 1
        if time.time() - t0 > time_budget:
            return best_f + 1.0
        return obj(x)

    res = optimize.minimize(timed_obj, x_start, method='Nelder-Mead',
                            options={'maxiter': maxiter, 'xatol': 1e-9,
                                     'fatol': 1e-11, 'adaptive': True})
    if not np.any(np.isnan(res.x)):
        f = obj(res.x)
        if f < best_f:
            best_f = f
            best_x = res.x.copy()
    return best_x, best_f


def _coordinate_descent(h, n, S, c5_best, max_rounds, t0, budget_s):

                ht[i] += d
                ht = project_feasible(ht, n, S)
                ct = compute_c5(ht, n)

                    improved = True
