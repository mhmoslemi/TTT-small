    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def project_onto_feasible(h, target_sum):
    """Project h to [0,1] and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum; increase smallest elements first
        idx = np.argsort(h)
        for i in idx:
            room = 1 - h[i]
            if room > 0:
                inc = min(room, diff)
                h[i] += inc
                diff -= inc
                if diff <= 0:
                    break
    else:
        # Need to decrease sum; decrease largest elements first
        idx = np.argsort(h)[::-1]
        for i in idx:
            room = h[i]
            if room > 0:
                dec = min(room, -diff)
                h[i] -= dec
                diff += dec
                if diff >= 0:
                    break
