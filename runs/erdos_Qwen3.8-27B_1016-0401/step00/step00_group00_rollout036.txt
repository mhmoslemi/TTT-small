import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def proj(h, ts):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h *= ts / s
    else:
        h[:] = ts / len(h)
        return h
    h = np.clip(h, 0.0, 1.0)
    d = ts - h.sum()
    if abs(d) > 1e-12:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def directed_search(h, dx, ts, rng, tlim):

    bc5 = compute_c5(bh, dx)
    tmp = bh.copy()
