def compute_c5(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h onto {0<=h[i]<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase: add to elements below 1
            available = 1.0 - h
            if np.sum(available) < 1e-15:
                break
            frac = diff / np.sum(available)
            frac = min(frac, 1.0)
            h = h + frac * available
        else:
            # Need to decrease: subtract from elements above 0
            available = h.copy()
            if np.sum(available) < 1e-15:
                break
            frac = (-diff) / np.sum(available)
            frac = min(frac, 1.0)
            h = h - frac * available
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = diff
        for i in order:
            if abs(remaining) < 1e-14:
                break
            if remaining > 0:
                adj = min(remaining, 1.0 - h[i])
                h[i] += adj
                remaining -= adj
            else:
                adj = max(remaining, -h[i])
                h[i] += adj
                remaining -= adj
    return h

    n = len(h)

    corr = np.correlate(h, comp, mode="full")

        k = mi - (n - 1)

            for i in range(n):
                val = 0.0
                if i >= k:
                    val += 1.0 - h[i - k]
                if i < n - k:
                    val -= h[i + k]
                g[i] = val
        else:
            p = -k
            for i in range(n):
                val = 0.0
                if i < n - p:
                    val += 1.0 - h[i + p]
                if i >= p:
                    val -= h[i - p]
                g[i] = val
        grad += g
    grad /= len(lags)
    return grad, max_val

def optimize_h(h0, n, max_iters=2000, base_step=0.05):
    """Subgradient optimization with adaptive step size"""
    h = h0.copy().astype(float)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for it in range(max_iters):
        grad, c5_raw = subgrad_step(h, step)
        c5 = c5_raw * (2.0 / n)
        if c5 < best_c5:

            best_h = h.copy()
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        if new_c5 > c5:
            step *= 0.7
        else:
            step = min(step * 1.05, 1.0)
        h = h_new
        if step < 1e-6:

    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            avail = 1.0 - h
            if np.sum(avail) < 1e-15:
                break
            h += (diff / np.sum(avail)) * avail
        else:
            avail = h
            if np.sum(avail) < 1e-15:
                break
            h -= ((-diff) / np.sum(avail)) * avail
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        rem = diff
        for i in order:
            if abs(rem) < 1e-14:
                break
            if rem > 0:
                a = min(rem, 1.0 - h[i])
                h[i] += a
                rem -= a
            else:
                a = max(rem, -h[i])
                h[i] += a
                rem -= a
    return h

def subgradient(h):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    mx = np.max(corr)
    lags = np.where(corr >= mx - 1e-12)[0]
    grad = np.zeros(n)
    for mi in lags:
        k = mi - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            lo = k
            hi = n
            for i in range(lo, hi):
                g[i] += 1.0 - h[i - k]
            for i in range(0, n - k):
                g[i] -= h[i + k]
        else:
            p = -k
            for i in range(0, n - p):
                g[i] += 1.0 - h[i + p]
            for i in range(p, n):
                g[i] -= h[i - p]
        grad += g
    grad /= len(lags)
    return grad, mx * (2.0 / n)

def optimize_subgrad(h0, n, max_iters=2000, base_step=0.02, tol=1e-8):
    h = project_feasible(h0.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for it in range(max_iters):
        grad, c5_cur = subgradient(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > c5_cur + 1e-14:
            step *= 0.7
        else:
            step = min(step * 1.08, 0.5)
        h = h_new
        if step < 1e-8:
            break
        if best_c5 < tol:
            break
    return best_h, best_c5

def random_binary_init(n, rng):
    target = n / 2.0
    h = np.zeros(n)
    n_ones = int(target)
    if n_ones > 0:
        idx = rng.choice(n, size=n_ones, replace=False)
        h[idx] = 1.0
    rem = target - n_ones
    if rem > 1e-12:
        remaining = np.setdiff1d(np.arange(n), idx)
        perm = rng.choice(remaining, size=len(remaining), replace=False)
        for i in perm:
            a = min(rem, 1.0)
            h[i] += a
            rem -= a
            if rem < 1e-12:
                break
    return h

def random_continuous_init(n, rng):
    h = rng.random(n)
    return project_feasible(h, n)

def perturb_feasible(h, n, strength, rng):
    h_new = h + strength * rng.standard_normal(n)
    return project_feasible(h_new, n)

def try_binary_round(h, n, rng, max_try=20):
    """Try rounding to 0/1 while maintaining sum, see if C5 improves"""
    c5_cont = compute_c5(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = c5_cont
    for _ in range(max_try):
        h_bin = np.round(h + rng.standard_normal(n) * 0.01)
        h_bin = project_feasible(h_bin, n)
        # Try to make more binary
        frac = h_bin[(h_bin > 0.01) & (h_bin < 0.99)]
        if len(frac) > 0:
            h_bin[h_bin > 0.5] = 1.0
            h_bin[h_bin <= 0.5] = 0.0
            h_bin = project_feasible(h_bin, n)
        c5_b = compute_c5(h_bin)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()
    return best_h, best_c5

        n_starts = 4 if n <= 80 else 2

            if n == n_init and s == 0:

            elif n == n_init and s == 1:

            else:

    if time.time() < deadline - 5:
        n_b = len(best_h)
        strengths = [0.05, 0.1, 0.2, 0.4, 0.8]
        for st in strengths:
            for _ in range(8):
                if time.time() > deadline - 2:

                h_p = perturb_feasible(best_h, n_b, st, rng)
                h_o, c5_o = optimize_subgrad(h_p, n_b, max_iters=800, base_step=0.01)
                if c5_o < best_c5:

                    best_h = h_o.copy()

    # Phase 5: Binary rounding attempts

    best_h = project_feasible(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
