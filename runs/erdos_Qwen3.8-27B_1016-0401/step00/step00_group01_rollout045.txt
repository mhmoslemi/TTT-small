import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_valid(h, n):
    h = h.astype(np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if len(free) == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h[free] += diff / len(free)
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    weights = np.zeros(2 * n - 1)
    tol = max(1e-10, 1e-6 * max_val)

    if weights.sum() == 0:

    weights /= weights.sum()
    
    grad = np.zeros(n)
    active = np.where(weights > 0)[0]
    for m in active:
        d = m - (n - 1)
        part1 = np.zeros(n)
        part2 = np.zeros(n)
        if d >= 0:
            part1[d:] = 1.0 - h[:n - d]
            part2[:n - d] = h[d:]
        else:
            dd = -d
            part1[:n - dd] = 1.0 - h[dd:]
            part2[dd:] = h[:n - dd]
        grad += weights[m] * (part1 - part2)
    return grad
