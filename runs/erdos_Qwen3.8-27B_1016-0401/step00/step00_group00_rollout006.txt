    parent = np.array(initial_h_values, dtype=float)

            if time.time() - t0 > budget_s - 20:
                break

            for r in range(3):
                if time.time() - t0 > budget_s - 15:
                    break
                if r == 0:
                    x0 = h0.copy()
                elif r == 1:
                    x0 = h0 + rng.normal(0, 0.05, n)
                    x0 = norm_h(x0, ts)
                else:
                    x0 = h0 + rng.normal(0, 0.1, n)
                    x0 = norm_h(x0, ts)
                try:
                    res = minimize(lambda x: smooth_obj(x, ts, alpha), x0,
                                 method='SLSQP', bounds=bounds, constraints=constr,
                                 options={'maxiter': maxiter, 'ftol': 1e-10})
                    h_new = norm_h(res.x, ts)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_curr:
                        c5_curr = c5_new
                        h0 = h_new.copy()
                except Exception:
                    pass

        if time.time() - t0 < budget_s - 15:
            for r in range(100):
                if time.time() - t0 > budget_s - 10:
                    break
                h = h0.copy()
                n_h = len(h)
                dx = 2.0 / n_h
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                # Find which lag is maximized
                k_max = np.argmax(corr)
                # Gradient of corr[k_max] with respect to h
                # corr[k_max] = sum_i h[i] * (1-h[i + k_max - n + 1])
                # Actually np.correlate(h, 1-h, 'full')[k] = sum_i h[i] * (1-h)[i+k-(n-1)]
                # Let offset = k - (n-1)
                offset = k_max - (n_h - 1)
                if offset >= 0:

                    # d/dh[j] = (1-h[j]) for j in [0, n-1-offset] (first factor)
                    #            -h[j-offset] for j in [offset, n-1] (second factor)
                    grad = np.zeros(n_h)
                    for i in range(n_h - offset):

                        grad[i + offset] -= h[i] * dx
                else:

                    o = -offset
                    # corr = sum_{i=0}^{n-1-o} h[i+o] * (1 - h[i])
                    # d/dh[j] = (1-h[j-o]) for j in [o, n-1] ... hmm this is getting complex
                    # Let me just use finite differences for the exact max
                    pass
                
                c5_curr = compute_c5(h0)
                if c5_curr < best_c5:
                    best_c5 = c5_curr
                    best_h = h0.copy()
                break  # Just one iteration for now
    
    # Final fix for exact sum
    ts = len(best_h) / 2.0
    res = ts - np.sum(best_h)
    if abs(res) > 1e-9:
        # Spread evenly among non-saturated entries
        if res > 0:
            idx = np.where(best_h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(best_h > 1e-9)[0]
        if len(idx) > 0:
            best_h[idx] += res / len(idx)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, ts, alpha):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    val = cmax + alpha * np.log(np.sum(np.exp((corr - cmax) / alpha)))
    val += 200.0 * (np.sum(h) - ts) ** 2
    return val

def norm_h(h, ts):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (ts / s)
    return np.clip(h, 0.0, 1.0)

def fix_sum(h, ts):
    h = np.clip(h, 0.0, 1.0)
    res = ts - np.sum(h)
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += res / len(idx)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    parent = np.array(initial_h_values, dtype=float)
    np_ = len(parent)
    parent = norm_h(parent, np_ / 2.0)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h)

    sizes = sorted(set([np_, 40, 50, 60, 80, 100, 120, 150, 200]))
    x_old = np.arange(np_) * 2.0 / np_

    for n in sizes:
        if time.time() - t0 > budget_s - 30:
            break
        ts = n / 2.0
        if n == np_:
            h0 = parent.copy()
        else:
            x_new = np.arange(n) * 2.0 / n
            h0 = np.interp(x_new, x_old, parent, left=parent[0], right=parent[-1])
            h0 = np.clip(h0, 0.0, 1.0)
        h0 = fix_sum(h0, ts)
        c5_curr = compute_c5(h0)

        bounds = [(0.0, 1.0)] * n
        constr = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts}]

        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
            if time.time() - t0 > budget_s - 20:
                break
            maxiter = 400 if n <= 80 else 200
            n_restart = 4 if n <= 60 else 3
            for r in range(n_restart):
                if time.time() - t0 > budget_s - 15:
                    break
                if r == 0:
                    x0 = h0.copy()
                elif r == 1:
                    x0 = h0 + rng.normal(0, 0.04, n)
                    x0 = fix_sum(x0, ts)
                elif r == 2:
                    x0 = h0 + rng.normal(0, 0.08, n)
                    x0 = fix_sum(x0, ts)
                else:
                    x0 = h0 + rng.normal(0, 0.15, n)
                    x0 = fix_sum(x0, ts)
                try:
                    res = minimize(lambda x: smooth_obj(x, ts, alpha), x0,
                                   method='SLSQP', bounds=bounds, constraints=constr,
                                   options={'maxiter': maxiter, 'ftol': 1e-12})
                    h_new = fix_sum(res.x, ts)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_curr:
                        c5_curr = c5_new
                        h0 = h_new.copy()
                except Exception:
                    pass

        # Local polish: coordinate-group refinement
        if time.time() - t0 < budget_s - 20:
            group_size = max(5, n // 20)
            for _ in range(3):
                if time.time() - t0 > budget_s - 10:
                    break
                np.random.default_rng()
                perm = rng.permutation(n)
                improved = False
                for gi in range(0, n, group_size):
                    if time.time() - t0 > budget_s - 10:
                        break
                    grp = perm[gi:gi+group_size]
                    x0_grp = h0[grp].copy()
                    def grp_obj(xg):
                        x_full = h0.copy()
                        x_full[grp] = xg
                        x_full = fix_sum(x_full, ts)
                        return smooth_obj(x_full, ts, 0.001)
                    try:
                        res_g = minimize(grp_obj, x0_grp, method='L-BFGS-B',
                                         bounds=[(0,1)]*len(grp),
                                         options={'maxiter': 50})
                        h0[grp] = np.clip(res_g.x, 0, 1)

                        pass
                if not improved:

        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h0.copy()

    # Final exact sum fix
    ts = len(best_h) / 2.0
    best_h = fix_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
