    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_objective(h, t, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx

    m = np.max(vals)

    target = n / 2.0

    return smooth_max + sum_pen
