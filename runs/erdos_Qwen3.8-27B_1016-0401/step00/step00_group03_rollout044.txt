import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + beta * np.log(np.mean(np.exp((corr - m) / beta)))) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / beta)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        s = idx - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        for i in range(lo, hi):
            ni = i - s
            g = (1.0 - h[ni])
            if 0 <= i + s < n:
                g -= h[i + s]
            grad[i] += w[idx] * g
    grad *= dx
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:

    target_sum = n / 2.0

    cons = {'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum,
            'jac': lambda x: np.ones(n)}

    best_h = project_feasible(h_init.copy(), n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    def slsqp_run(h0, beta, maxiter):
        if elapsed() > budget_s * 0.92:
            return h0, compute_c5(h0)
        h0 = project_feasible(h0.copy(), n)

        def obj(x):
            return smooth_obj(x, beta) if beta > 0.01 else compute_c5(x)

        def grad(x):
            if beta > 0.01:
                return smooth_grad(x, beta)
            # Use subgradient of true max
            c, _ = smooth_grad_raw(x)
            return c

            res = minimize(obj, h0, method='SLSQP', jac=grad,
                           bounds=bounds, constraints=[cons],

            hx = project_feasible(res.x, n)
            return hx, compute_c5(hx)

            return h0, compute_c5(h0)

        dx2 = 2.0 / n2
        corr = np.correlate(h, 1.0 - h, mode="full")

                if 0 <= i + s < n2:
                    g -= h[i + s]
                grad[i] += g
        grad /= len(active)
        grad *= dx2
        return mv * dx2, grad

            h_nm = project_feasible(res_nm.x, n)

            if c_nm < best_c5 - 1e-12:
                best_c5 = c_nm
                best_h = h_nm
        except Exception:
            pass

    # Final feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
