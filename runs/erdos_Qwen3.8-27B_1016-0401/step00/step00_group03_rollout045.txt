import numpy as np
import time


def project_hs(v, target):
    """Project v onto {h: 0<=h<=1, sum(h)=target}."""
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full') * dx))


def objgrad(h, alpha):
    """Smoothed max of correlation values and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    vals = corr * dx
    m = vals.max()
    ev = np.exp((vals - m) / alpha)
    w = ev / ev.sum()
    val = m + alpha * np.log(ev.mean())
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx] * dx
        if wk < 1e-13:
            continue
        k = idx - n + 1

            g[k:] -= wk * h[:n - k]
        else:
            k2 = -k
            g[k2:] += wk * (1.0 - h[:n - k2])
            g[:n - k2] -= wk * h[k2:]
    return val, g

            if time.time() > deadline:

        val, g = objgrad(h, alpha)

        h = project_hs(h, target)
        if it % 50 == 49:

    c5 = c5_of(h)

        best = h.copy()
    return best, best_c5


def round_binary(h, target):
    """Round h to binary maintaining sum."""
    n = len(h)
    order = np.argsort(-h)
    k = int(target)
    h_bin = np.zeros(n)
    h_bin[order[:k]] = 1.0
    # Handle fractional case
    if k < n:
        frac = target - k
        if frac > 0:
            h_bin[order[k]] = frac
            if frac > 0 and k + 1 < n:
                h_bin[order[k + 1]] = 1.0 - frac
    return h_bin
