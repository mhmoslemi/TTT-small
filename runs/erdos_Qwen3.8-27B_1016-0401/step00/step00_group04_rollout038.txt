import numpy as np
from scipy.optimize import minimize
import time

_ALPHA = 100.0

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def slexp_max(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return slexp_max(corr, _ALPHA)

def constraint_fun(h):
    return np.sum(h) - len(h) / 2.0

def constraint_jac(h):
    return np.ones(len(h))

def make_bounds(n):
    return [(0.0, 1.0)] * n

def slsqp_optimize(h0, maxiter=400, ftol=1e-14):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = make_bounds(n)
    res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

        for i in range(target - s):
