            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(70, 0.5)
    
    n = len(h0)
    S = n / 2.0
    h0 = normalize_h(h0, S)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP with increasing T (annealing)
    T_schedule = [3, 10, 40, 150, 600, 2000, 8000]
    for T in T_schedule:
        if time.time() - t0 > budget_s * 0.35:
            break
        h_new = slsqp_refine(best_h, n, S, T, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Binary conversion + SA + hill climbing
    t_deadline_sa = t0 + budget_s * 0.75
    t_deadline_hc = t0 + budget_s * 0.90
    
    h_bin = to_binary(best_h, S)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
    
    # SA from binary start
    h_sa, c5_sa = binary_sa(best_h, n, S, rng, 80000, t_deadline_sa, best_c5)
    if c5_sa < best_c5:
        best_h = h_sa
        best_c5 = c5_sa
    
    # Hill climbing
    h_hc_out, c5_hc_out = hill_climb_binary(best_h, n, S, best_c5,
                                             max_sweeps=100, t_deadline=t_deadline_hc)
    if c5_hc_out < best_c5:
        best_h = h_hc_out
        best_c5 = c5_hc_out
    
    # Phase 3: One more SLSQP round from best binary
    if time.time() - t0 < budget_s * 0.92:
        h_cont = best_h.astype(float) + rng.normal(0, 0.05, n)
        h_cont = normalize_h(h_cont, S)
        for T in [50, 300, 1500]:
            if time.time() - t0 > budget_s * 0.94:
                break
            h_new = slsqp_refine(h_cont, n, S, T, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_cont = h_new
    
    # Final check: try binary version of final continuous
    if time.time() - t0 < budget_s * 0.96:
        h_final_bin = to_binary(best_h, S)
        c5_fb = compute_c5(h_final_bin)
        if c5_fb < best_c5:
            best_h = h_final_bin
            best_c5 = c5_fb
        # Quick hill climb
        h_hc2, c5_hc2 = hill_climb_binary(best_h, n, S, best_c5,
                                           max_sweeps=30, t_deadline=t0 + budget_s * 0.97)
        if c5_hc2 < best_c5:
            best_h = h_hc2
            best_c5 = c5_hc2
    
    # Final normalization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n)
