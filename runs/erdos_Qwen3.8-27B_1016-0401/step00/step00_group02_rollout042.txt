import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def subgradient_step(h, n, lr, n_top=5):
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(corrs)[-n_top:]
    lags = top_idx - (n - 1)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += dx * (1.0 - 2.0 * h)
        else:
            sp = np.roll(h, -lag)
            sm = np.roll(h, lag)
            grad += dx * (1.0 - sp - sm)
    grad /= n_top
    h_new = h - lr * grad
    return project_feasible(h_new, n), np.max(corrs)

def nm_optimize(h0, n, max_iter, timeout_fn):
    h = h0.copy()
    c5 = evaluate_c5(h)
    dim = n
    simplex = [h.copy()]
    step_init = 0.15
    for i in range(dim):
        v = h.copy()
        v[i] += step_init if v[i] < 0.9 else -step_init * 0.5
        v = project_feasible(v, n)
        simplex.append(v)
    fvals = [c5] + [evaluate_c5(s) for s in simplex[1:]]
    alpha, gamma, rho, sigma = 1.0, 2.0, 0.5, 0.5
    no_improve = 0
    for it in range(max_iter):
        if timeout_fn() < 3.0:
            break
        order = np.argsort(fvals)
        simplex = [simplex[i] for i in order]
        fvals = [fvals[i] for i in order]
        if fvals[0] < c5:
            c5 = fvals[0]
            h = simplex[0].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break
        if fvals[-1] - fvals[0] < 1e-12:
            break
        centroid = project_feasible(np.mean(simplex[:-1], axis=0), n)
        xr = project_feasible(centroid + alpha * (centroid - simplex[-1]), n)
        fr = evaluate_c5(xr)
        if fr < fvals[0]:
            xe = project_feasible(centroid + gamma * (centroid - simplex[-1]), n)
            fe = evaluate_c5(xe)
            if fe < fr:
                simplex[-1], fvals[-1] = xe, fe
            else:
                simplex[-1], fvals[-1] = xr, fr
        elif fr < fvals[-2]:
            simplex[-1], fvals[-1] = xr, fr
        else:
            xc = project_feasible(centroid + rho * (simplex[-1] - centroid), n)
            fc = evaluate_c5(xc)
            if fc < fvals[-1]:
                simplex[-1], fvals[-1] = xc, fc
            else:
                for i in range(1, dim + 1):
                    simplex[i] = project_feasible(simplex[0] + sigma * (simplex[i] - simplex[0]), n)
                    fvals[i] = evaluate_c5(simplex[i])
    return h, c5

def subgradient_optimize(h0, n, start_lr, min_lr, n_iter, timeout_fn, n_top=5):
    h = h0.copy()
    c5 = evaluate_c5(h)
    lr = start_lr
    patience = 0
    for it in range(n_iter):
        if timeout_fn() < 3.0:
            break
        h_new, c5_new = subgradient_step(h, n, lr, n_top)
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
            patience = 0
        else:
            patience += 1
            if patience > max(50, n_iter // 10):
                lr *= 0.8
                patience = 0
            if lr < min_lr:
                break
    return h, c5

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
