    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h onto {sum = target_sum, 0 <= h <= 1}"""
    h = h.copy()
    for _ in range(50):
        excess = np.sum(h) - target_sum
        if abs(excess) < 1e-10:
            break
        if excess > 0:
            # Need to reduce
            mask = h > 0
            n_reduce = np.sum(mask)
            if n_reduce == 0:

            h[mask] -= reduction
            h = np.clip(h, 0, 1)

            if n_incr == 0:

            h[mask] += increase
            h = np.clip(h, 0, 1)
    return h

        If all active slopes < 0: return -1
        If mixed or zero: return 0"""

        elif np.all(slopes < -1e-10):
            return -1
        else:
            return 0

    s_hi = subgradient_sign(hi)

        return lo

        t_budget_n = remaining * alloc[idx_n] if idx_n < len(alloc) else remaining * 0.1
        n_methods = 4 if n <= 100 else 2
        t_per_method = t_budget_n / (n_methods + 0.3)

        for m in range(n_methods):
            remaining = budget_s - (time.time() - t_start)
            if remaining < 5:
                break
            if m == 0 and n == 70:
                try:
                    h0 = np.array(globals()['initial_h_values'], dtype=float)
                    if len(h0) != n:
                        xo = np.linspace(0, 2, len(h0), endpoint=False)
                        xn = np.linspace(0, 2, n, endpoint=False)
                        h0 = np.interp(xn, xo, h0)
                except (KeyError, NameError, TypeError, ValueError):
                    h0 = make_init(n, 3, rng)
            else:
                h0 = make_init(n, m, rng)
            h_opt, c5 = coord_descent(h0, max_passes=400, time_budget=t_per_method)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n

    # Perturbation restarts on best solution
    remaining = budget_s - (time.time() - t_start)
    if remaining > 20 and best_h is not None:
        n_cur = len(best_h)
        for attempt in range(5):
            remaining = budget_s - (time.time() - t_start)
            if remaining < 10:
                break
            h_pert = best_h + rng.uniform(-0.05, 0.05, n_cur)
            h_pert = project_simplex(h_pert, n_cur / 2.0)
            h_opt, c5 = coord_descent(h_pert, max_passes=200, time_budget=remaining / 6.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n_cur

    # Final validation
    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_simplex(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
