    """Gradient of c[k] w.r.t. h for lag d (d = k - (n-1))"""
    grad = np.zeros(n)

        grad[m + d] += (1.0 - h[m])
        grad[m] += -h[m + d]
    else:

        grad[m + d] += (1.0 - h[m])
        grad[m] += -h[m + d]

    return _c5_val(x)

import numpy as np
from scipy.optimize import minimize
import time

_dx = 0.0
_n_pts = 0  
_tau_cur = 0.0

def _project(h, target_sum):
    # Project onto {0<=x<=1, sum(x)=target_sum}
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return np.max(_corr_vec(h)) * _dx

def _lag_grad(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        m = np.arange(n - d)
        g[m + d] += 1.0 - h[m]
        g[m] += -h[m + d]
    else:
        m = np.arange(-d, n)
        g[m + d] += 1.0 - h[m]
        g[m] += -h[m + d]
    return g

def _obj(x):
    c = _corr_vec(x)
    return np.max(c) * _dx

def _jac(x):
    c = _corr_vec(x)
    idx = np.argmax(c)
    d = idx - (_n_pts - 1)
    return _lag_grad(x, d, _n_pts) * _dx

def _obj_s(x):
    c = _corr_vec(x)
    mx = np.max(c)
    e = np.exp((c - mx) / _tau_cur)
    return (mx + _tau_cur * np.log(np.sum(e))) * _dx

def _jac_s(x):
    c = _corr_vec(x)
    mx = np.max(c)
    e = np.exp((c - mx) / _tau_cur)
    w = e / np.sum(e)
    n = _n_pts
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            g += w[k] * _lag_grad(x, k - (n - 1), n)
    return g * _dx

    global _tau_cur
    t = _n_pts / 2.0
    x0 = _project(x.copy(), t)
    if smooth_tau > 0:
        _tau_cur = smooth_tau
        res = minimize(_obj_s, x0, jac=_jac_s, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * _n_pts,
                       options={'maxiter': maxit, 'ftol': 1e-13, 'gtol': 1e-9})

                       options={'maxiter': maxit, 'ftol': 1e-13, 'gtol': 1e-9})
