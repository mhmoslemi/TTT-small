import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx, dx

def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(corr, temp):
    """Compute log-sum-exp value and its gradient."""
    scaled = corr / temp
    m = np.max(scaled)
    exps = np.exp(scaled - m)
   lse = temp * (m + np.log(np.sum(exps)))
    weights = exps / np.sum(exps)
    return lse, weights

def grad_at_weights(h, weights, dx):
    """Compute weighted sum of gradients of all correlation values."""
    n = len(h)
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if weights[m_idx] < 1e-10:
            continue
        shift = m_idx - (n - 1)
        w = weights[m_idx] * dx
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n - shift]
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    return grad

def optimize_continuous(h0, n, budget_s, time_start):
    """Optimize using projected gradient with adaptive temperature."""
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Adaptive learning rate and temperature
    lr = 0.01
    temp = 0.05  # initial temperature for log-sum-exp
    momentum = 0.9
    v = np.zeros(n)
    
    it = 0
    stagnation = 0
    while time.time() - time_start < budget_s:
        corr, dx = compute_corr(h)
        c5_val = np.max(corr)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Compute gradient
        lse, weights = smooth_max_grad(corr, temp)
        g = grad_at_weights(h, weights, dx)
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_h(h_new, target_sum)
        
        # Check if improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_val + 1e-10:
            h = h_new
        else:
            # Reduce learning rate
            lr *= 0.9
            v *= 0.5
            h = h_new  # still move, but less aggressively
        
        it += 1
        
        # Adapt temperature (slowly decrease)
        if it % 100 == 0:
            temp *= 0.95
            lr = max(lr * 1.02, 0.001)
        
        if stagnation > 2000:
            break
    
    return best_h, best_c5

    h = np.clip(h, 0.0, 1.0)

    scaled = corr / temp
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    denom = np.sum(exps)
    weights = exps / denom
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-9:
            continue
        shift = idx - (n - 1)
        coeff = w * dx
        if shift >= 0:
            s = shift
            if n - s > 0:
                grad[:n - s] += coeff * (1.0 - h[s:])
                grad[s:] -= coeff * h[:n - s]
        else:
            s = -shift
            if n - s > 0:
                grad[s:] += coeff * (1.0 - h[:n - s])
                grad[:n - s] -= coeff * h[s:]
    return grad

def optimize_grad(h0, n, max_time, rng):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_h(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.015
    temp = 0.04
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    t0 = time.time()
    while time.time() - t0 < max_time:
        corr = compute_corr(h)
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        g = grad_smooth_max(h, corr, temp, dx)
        v = momentum * v - lr * g
        h_new = project_h(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 0.02:
            lr *= 0.85
            v *= 0.4
        h = h_new
        if stall > 400:
            h += rng.standard_normal(n) * 0.04
            h = project_h(h, target_sum)
            v = np.zeros(n)
            lr = 0.015
            stall = 0
        if stall > 0 and stall % 80 == 0:
            temp = max(0.004, temp * 0.97)
    return best_h, best_c5

def binary_search(h0, n, max_time, rng):
    if n % 2 != 0:
        return h0, compute_c5(h0)
    target = n // 2
    h = np.zeros(n)
    order = np.argsort(-h0)
    h[order[:target]] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_imp = 0
    t0 = time.time()
    while time.time() - t0 < max_time and no_imp < 8000:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            no_imp = 0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_imp += 1
        if no_imp > 3000:
            for _ in range(4):
                ii = ones[rng.integers(len(ones))]
                jj = zeros[rng.integers(len(zeros))]
                h[ii], h[jj] = h[jj], h[ii]
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_imp = 0
    return best_h, best_c5

def make_init(n, rng):
    h0 = np.zeros(n)
    style = rng.integers(4)
    if style == 0:
        h0[:n // 2] = 1.0
    elif style == 1:
        h0[:n // 4] = 1.0
        h0[3 * n // 4:] = 1.0
    elif style == 2:
        h0[::2] = 1.0
        if n % 2 == 1:
            h0[-1] = 0.0
    else:
        h0 = (np.sin(np.linspace(0, 4 * np.pi, n)) > 0).astype(float)
    h0 += rng.standard_normal(n) * 0.15
    return project_h(h0, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    n_list = [50, 60, 80, 100, 120, 150, 200]
    if h_init is not None:
        n0 = len(h_init)
        if n0 not in n_list:
            n_list.insert(0, n0)

    phase1_end = t_start + budget_s * 0.45
    for n in n_list:
        if time.time() > phase1_end - 5:
            break
        rem = phase1_end - time.time()
        if rem < 8:
            break
        if h_init is not None and n == len(h_init):
            h0 = h_init.copy()
        else:
            h0 = make_init(n, rng)
        sub_t = min(35, rem * 0.85)
        h_opt, c5v = optimize_grad(h0, n, sub_t, rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt.copy()

    phase2_end = t_start + budget_s * 0.78
    n_bin = [n for n in n_list if n % 2 == 0][:6]
    for n in n_bin:
        if time.time() > phase2_end - 5:
            break
        rem = phase2_end - time.time()
        if rem < 6:
            break
        if best_h is not None and len(best_h) == n:
            h0 = best_h.copy()
        elif h_init is not None and n == len(h_init):
            h0 = h_init.copy()
        else:
            h0 = make_init(n, rng)
        sub_t = min(25, rem * 0.6)
        h_bin, c5b = binary_search(h0, n, sub_t, rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()

    if best_h is not None and time.time() < t_start + budget_s - 10:
        nb = len(best_h)
        rem = t_start + budget_s - 10 - time.time()
        if rem > 5:
            h_ref, c5r = optimize_grad(best_h, nb, rem * 0.45, rng)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_ref
        if nb % 2 == 0 and rem > 12:
            hb2, c5b2 = binary_search(best_h, nb, rem * 0.35, rng)
            if c5b2 < best_c5:
                best_c5 = c5b2
                best_h = hb2

    if best_h is None:
        n = 50
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
