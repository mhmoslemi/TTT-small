import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def sum_constraint(h):
    return float(np.sum(h) - len(h) / 2.0)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
        else:
            mask = h > 1e-7
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def slsqp_optimize(h, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': sum_constraint},)
    try:
        res = minimize(eval_c5, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x, eval_c5(res.x)
    except Exception:
        pass
    return h.copy(), eval_c5(h)
