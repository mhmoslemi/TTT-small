import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def ensure_binary(h, target_sum):
    n = len(h)
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    order = np.argsort(h)[::-1]
    result = np.zeros(n, dtype=float)
    result[order[:int(target_sum)]] = 1.0
    return result


def local_search_binary(h, dx, max_rounds=300, time_limit=None, t_start=None):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_score = best_c5
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                score = compute_c5(h, dx)
                if score < best_score - 1e-14:
                    best_score = score
                    best_swap = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_score
        else:
            break
    return h, best_c5


def simulated_annealing(h, dx, n_iter=20000, t_start=None, time_limit=None):
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    t0, t1 = 0.15, 0.001
    for it in range(n_iter):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        temp = t0 * (t1 / t0) ** (it / max(n_iter, 1))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5


def random_binary(n, target_sum):
    h = np.zeros(n, dtype=float)
    positions = np.random.choice(n, size=target_sum, replace=False)
    h[positions] = 1.0
    return h


def structured_initializations(n, target_sum, rng):
    inits = []
    # Arithmetic progression-based
    for step in [2, 3, 4, 5, 7]:
        h = np.zeros(n, dtype=float)
        positions = list(range(0, n, step))
        # Adjust to get exactly target_sum ones
        positions = positions[:target_sum]
        h[positions] = 1.0
        if len(positions) < target_sum:
            extra = np.random.choice([p for p in range(n) if h[p] == 0],
                                      size=target_sum - len(positions), replace=False)
            h[extra] = 1.0
        inits.append(h)
    # Random permutations of first target_sum positions
    for _ in range(5):
        h = np.zeros(n, dtype=float)
        pos = np.sort(rng.choice(n, size=target_sum, replace=False))
        h[pos] = 1.0
        inits.append(h)
    return inits
