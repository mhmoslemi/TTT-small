import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def softmax_value(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    if m <= 1e-15:
        return 0.0
    shifted = (corr - m) / temp
    return float(m + temp * np.log(np.sum(np.exp(shifted))))


def project_h(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(x, target_sum, temp):
    h = project_h(x, target_sum)
    return softmax_value(h, temp)


def obj_exact(x, target_sum):
    h = project_h(x, target_sum)
    return compute_c5(h)


def elapsed_time(t0):
    return time.time() - t0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 47
        h0 = np.zeros(n0)
        h0[:23] = 1.0
        h0[23] = 0.5

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_h(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth L-BFGS-B with decreasing temperature
    temps = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for temp in temps:
        if elapsed_time(t0) > budget_s - 120:
            break
        res = minimize(obj_smooth, best_h, args=(target_sum, temp),
                       method='L-BFGS-B',
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = project_h(res.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Nelder-Mead with exact objective (limited iterations for 47D)
    if elapsed_time(t0) < budget_s - 90:
        res = minimize(obj_exact, best_h, args=(target_sum,),
                       method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-11, 'fatol': 1e-14,
                                'adaptive': True})
        h_new = project_h(res.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Random restarts with local optimization
    n_restarts = 40
    for r in range(n_restarts):
        if elapsed_time(t0) > budget_s - 40:
            break
        scale = 0.03 + 0.02 * (r % 5)
        h_trial = best_h + np.random.normal(0, scale, n)
        h_trial = project_h(h_trial, target_sum)
        for temp in [0.01, 0.003, 0.001]:

                    h_trial = best_h.copy()

            n = n2

        # Nelder-Mead on h2 if time allows
        if elapsed_time(t0) < budget_s - 60 and len(best_h) == n2:
            res = minimize(obj_exact, best_h, args=(n2 / 2.0,),
                           method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-11, 'fatol': 1e-14,
                                    'adaptive': True})
            h2_r = project_h(res.x, n2 / 2.0)
            c5_2r = compute_c5(h2_r)

                best_c5 = c5_2r
                n = n2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
