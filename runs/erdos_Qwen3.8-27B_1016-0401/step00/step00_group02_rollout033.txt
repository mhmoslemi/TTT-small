        else:

        xm = x.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def pnorm(corr, p):
    return float(np.sum(corr**p)**(1.0/p))

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return project_sum(h, new_n / 2.0)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_interp = np.interp(new_x, old_x, h)
    return project_sum(h_interp, new_n / 2.0)

def downsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return project_sum(h, new_n / 2.0)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_interp = np.interp(new_x, old_x, h)
    return project_sum(h_interp, new_n / 2.0)

def optimize_h(h, p=8, max_iter=100, time_limit=60.0):
    n = len(h)
    target_sum = n / 2.0
    rho = 50.0
    lam = 0.0
    t0 = time.time()
    best = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for phase in range(30):
        if time.time() - t0 > time_limit:
            break
        rho = rho  # keep updating
        
        def obj(x):
            c = compute_corr(x)
            base = float(np.sum(c**p)**(1.0/p))
            sv = np.sum(x) - target_sum
            return base + lam * sv + 0.5 * rho * sv * sv
        
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': min(50, max(10, (int)(time_limit - (time.time()-t0)) * 2)),
                                'ftol': 1e-12, 'gtol': 1e-8})
        h_new = np.clip(res.x, 0.0, 1.0)
        sv = np.sum(h_new) - target_sum
        lam += rho * sv
        if abs(sv) > 1e-5:
            rho = min(rho * 1.5, 1e4)
        
        h_new = project_sum(h_new, target_sum)
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

def pnorm_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    sp = float(np.sum(corr**p))
    if sp < 1e-30:
        return np.zeros(n)
    sp_factor = sp ** (1.0/p - 1.0) / p * p  # = sp^(1/p - 1)
    # dF/dh[m] = sp^(1/p - 1) * sum_k c_k^(p-1) * dx * dc_k/dh[m]
    
    # weights for each k
    w = corr**(p-1) * dx * sp**((1.0/p) - 1.0)
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2*n - 1):
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        kn = k - (n - 1)
        # Term 1: (1-h[i]) where i = m-kn, valid for 0<=m-kn<n
        # m ranges from max(0,kn) to min(n-1, n-1+kn)
        m_lo = max(0, kn)
        m_hi = min(n - 1, n - 1 + kn)
        if m_lo <= m_hi:
            i_range = np.arange(m_lo, m_hi + 1) - kn
            grad[m_lo:m_hi+1] += wk * one_minus_h[i_range]
        # Term 2: -h[i+kn] where i = m, valid for 0<=m+kn<n
        # m ranges from max(0,-kn) to min(n-1, n-1-kn)
        m_lo2 = max(0, -kn)
        m_hi2 = min(n - 1, n - 1 - kn)
        if m_lo2 <= m_hi2:
            idx_range = np.arange(m_lo2, m_hi2 + 1) + kn
            grad[m_lo2:m_hi2+1] += wk * (-h[idx_range])
    
    return grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0

        else:
            lo = mid

        return project_sum(h.copy(), new_n * 0.5)

    nx = np.arange(new_n) * (2.0 / new_n)

    return project_sum(hi, new_n * 0.5)
