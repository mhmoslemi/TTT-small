        h += diff2 / n

For binary h at lag 0: h[i]*(1-h[i]) = 0 (since h[i] is 0 or 1). So lag 0 is always 0 for binary.

For lag k>0: sum of h[i+k]*(1-h[i]) over valid i. This counts positions where h has 0 and h shifted by k has 1.

For lag -k (represented as index in full correlation): similar but reversed.

So the full correlation for binary h has 0 at lag 0, and for other lags, counts some kind of mismatch.

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def smooth_obj(h, p, ts):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    if m < 1e-15:
        return 0.0 + 100.0 * (h.sum() - ts) ** 2
    sh = p * (corr - m)
    sm = m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / p
    return sm + 100.0 * (h.sum() - ts) ** 2

def fix_sum(h, n, ts=None):
    if ts is None:
        ts = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = ts - h.sum()
        if abs(diff2) > 1e-12:
            # More careful fix
            if diff2 > 0:
                idx = np.argsort(h)[:int(np.ceil(abs(diff2)))]
            else:
                idx = np.argsort(-h)[:int(np.ceil(abs(-diff2)))]
            # Distribute among the most extreme
            per = diff2 / len(idx) if len(idx) > 0 else 0
            h[idx] = np.clip(h[idx] + per, 0, 1)
    return h

    best_E = E
    best_h = h_bin.copy()

            break

        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        j = zeros[rng.randint(len(zeros))]

        E_new = compute_c5(h_bin)

            E = E_new
            if E < best_E:
                best_E = E
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        T *= 0.99995
    return best_h, best_E

def local_search_binary(h_bin, ts):

    E = compute_c5(h_bin)
    best_E = E
    best_h = h_bin.copy()

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
    return best_h, best_E

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
