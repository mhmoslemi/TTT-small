For a given k (and thus m):
- Term where h[i] is the "left" factor: j=i, contribution to corr is (1-h[i+m]) if 0≤i+m<n
- Term where h[i] is the "right" factor: j+m=i, so j=i-m, contribution is -h[i-m] if 0≤i-m<n

def _project(h, target):

    for _ in range(80):

        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            hi = mid

            lo = mid

def _c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _grad(h, vals, tau, top_k):
    n = len(h)
    dx = 2.0 / n
    cm = vals.max()
    ev = np.exp(tau * (vals - cm))
    ws = ev / ev.sum()
    p = min(top_k, len(ws))
    idx = np.argsort(ws)[-p:]
    ws2 = np.zeros(len(ws))
    ws2[idx] = ws[idx]
    ws2 /= ws2.sum()
    g = np.zeros(n)
    for k in range(len(ws2)):
        wk = ws2[k]
        if wk < 1e-12:
            continue
        m = k - n + 1
        if m >= 0:
            g[:n - m] += wk * dx * (1.0 - h[m:])
            g[m:] -= wk * dx * h[:n - m]
        else:
            g[-m:] += wk * dx * (1.0 - h[:n + m])
            g[:n + m] -= wk * dx * h[-m:]
    return g
