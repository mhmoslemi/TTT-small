def _project_sum(x, s, lo=0.0, hi=1.0):
    n = len(x)
    def _sum_at(lam):
        return np.sum(np.clip(x - lam, lo, hi))
    lo_lam = x.min() - hi - 1.0
    hi_lam = x.max() - lo + 1.0
    for _ in range(80):
        mid = 0.5 * (lo_lam + hi_lam)
        if _sum_at(mid) > s:
            lo_lam = mid
        else:
            hi_lam = mid
    lam = 0.5 * (lo_lam + hi_lam)
    return np.clip(x - lam, lo, hi)

def _get_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def _coord_descent_step(h, i, n):
    old = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode="full")
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode="full")
    h[i] = old
    a = c0
    b = c1 - c0

    x2 = x_lo + gr * (x_hi - x_lo)
    f1 = np.max(a + b * x1)
    f2 = np.max(a + b * x2)

            x2 = x1

            x_lo = x1
            x1 = x2
            f1 = f2
            x2 = x_lo + gr * (x_hi - x_lo)
            f2 = np.max(a + b * x2)
