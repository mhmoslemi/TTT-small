import numpy as np
from scipy.interpolate import interp1d
import time

def project_onto_simplex(v, target_sum):
    n = len(v)
    if n == 0:
        return v.copy()
    lo, hi = -20.0, 20.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    exps = np.exp(shifted)
    sum_exps = exps.sum()
    weights = exps / sum_exps
    smooth_val = max_c + np.log(sum_exps) / tau
    
    grad = np.zeros(n)
    for k in range(m):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            s = shift
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                p1 = np.zeros(n); p1[:n-s] = 1.0 - h[s:]
                p2 = np.zeros(n); p2[s:] = h[:n-s]
                grad += w * (p1 - p2)
        else:
            s = -shift
            p1 = np.zeros(n); p1[s:] = 1.0 - h[:n-s]
            p2 = np.zeros(n); p2[:n-s] = h[s:]
            grad += w * (p1 - p2)
    return smooth_val, grad

def fista_optimize(h0, n, tau_schedule, total_iters, deadline):
    h = project_onto_simplex(h0.copy(), n / 2.0)
    y = h.copy()
    t_mom = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        tau = tau_schedule[min(it // max(1, total_iters // len(tau_schedule)), len(tau_schedule)-1)]
        
        val, grad = smooth_max_and_grad(y, tau)
        # Backtracking line search
        alpha = 1.0
        f_y = val
        for ls in range(30):
            x_trial = y - alpha * grad
            h_trial = project_onto_simplex(x_trial, n / 2.0)
            f_trial, _ = smooth_max_and_grad(h_trial, tau)
            if f_trial <= f_y - 0.5 * alpha * np.sum(grad * (y - h_trial)):
                break
            alpha *= 0.5
        h_new = project_onto_simplex(y - alpha * grad, n / 2.0)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + (t_mom - 1.0) / t_new * (h_new - h)
        y = project_onto_simplex(y, n / 2.0)
        t_mom = t_new
        h = h_new
    
    return best_h, best_c5

def refine_grid(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    h_interp = np.clip(h_interp, 0, 1)
    return project_onto_simplex(h_interp, n_new / 2.0)

def smooth_max_and_grad(h, tau):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)  # 2n-1
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    exps = np.exp(shifted)
    sum_exps = exps.sum()
    weights = exps / sum_exps
    smooth_val = max_c + np.log(sum_exps) / tau
    
    grad = np.zeros(n)
    for k_out in range(m):
        w = weights[k_out]
        if w < 1e-12:
            continue
        shift = k_out - (n - 1)
        # ∂corr/∂h[i] = (1-h[i-shift])*[valid] - h[i+shift]*[valid]
        # Term 1: (1-h[i-shift]) for i where i-shift is in [0,n-1]
        #   i-shift >= 0 → i >= shift
        #   i-shift < n → i < n+shift
        #   So i in [max(0,shift), min(n-1, n-1+shift)]
        # Term 2: -h[i+shift] for i where i+shift is in [0,n-1]
        #   i+shift >= 0 → i >= -shift
        #   i+shift < n → i < n-shift
        #   So i in [max(0,-shift), min(n-1, n-1-shift)]
        
        # Vectorized:
        i_min1 = max(0, shift)
        i_max1 = min(n - 1, n - 1 + shift)
        if i_min1 <= i_max1:
            grad[i_min1:i_max1+1] += w * (1.0 - h[i_min1-shift:i_max1-shift+1])
        
        i_min2 = max(0, -shift)
        i_max2 = min(n - 1, n - 1 - shift)
        if i_min2 <= i_max2:
            grad[i_min2:i_max2+1] -= w * h[i_min2+shift:i_max2+shift+1]
    
    return smooth_val, grad

    L = len(v)
    if L == 0:
        return v.copy()
    lo, hi = -20.0, 20.0
    for _ in range(80):

        else:

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return project_onto_simplex(h_interp, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 5.0

    h_init = None
    try:
        if initial_h_values is not None:
            h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        pass
    if h_init is None or len(h_init) == 0:
        L_init = 42
        h_init = np.zeros(L_init)
        idx = rng.permutation(L_init)[:L_init // 2]
        h_init[idx] = 1.0

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    base_n = len(h_init)
    tau_sched = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]

    # Phase 1: optimize at initial grid
    h_opt = fista_optimize(h_init, base_n, tau_sched, 6000, deadline)
    if h_opt[1] < best_c5:
        best_c5 = h_opt[1]
        best_h = h_opt[0]

    # Random restarts at initial grid
    for restart in range(8):
        if time.time() > deadline:
            break
        scale = 0.12 + 0.03 * restart
        h_rand = best_h + scale * rng.randn(base_n)
        h_rand = project_onto_simplex(h_rand, base_n / 2.0)
        h_opt = fista_optimize(h_rand, base_n, tau_sched, 3000, deadline)
        if h_opt[1] < best_c5:
            best_c5 = h_opt[1]
            best_h = h_opt[0]

    # Phase 2: refine to larger grids
    current_h = best_h
    for factor in [2, 4, 8, 16, 32]:
        n_new = base_n * factor
        if n_new > 800 or time.time() > deadline:
            break
        h_refined = refine_grid(current_h, n_new)
        n_iters = max(800, min(4000, 12000 // factor))
        h_opt = fista_optimize(h_refined, n_new, tau_sched, n_iters, deadline)
        if h_opt[1] < best_c5:
            best_c5 = h_opt[1]
            best_h = h_opt[0]
            current_h = h_opt[0].copy()

        # One random perturbation restart at this grid
        if time.time() < deadline:
            h_rand = refine_grid(current_h, n_new)
            h_rand += 0.08 * rng.randn(n_new)
            h_rand = project_onto_simplex(h_rand, n_new / 2.0)
            h_opt2 = fista_optimize(h_rand, n_new, tau_sched, n_iters // 2, deadline)
            if h_opt2[1] < best_c5:
                best_c5 = h_opt2[1]
                best_h = h_opt2[0]
                current_h = h_opt2[0].copy()

    # Phase 3: try downsampling from best large grid back to smaller grids
    # (sometimes a coarser grid captures the structure better)
    if len(best_h) > base_n and time.time() < deadline:
        for target_n in [base_n, base_n * 2, base_n * 3]:
            if target_n >= len(best_h):
                continue
            x_big = np.linspace(0, 2, len(best_h), endpoint=False)
            x_small = np.linspace(0, 2, target_n, endpoint=False)
            h_down = np.interp(x_small, x_big, best_h)
            h_down = project_onto_simplex(np.clip(h_down, 0, 1), target_n / 2.0)
            h_opt = fista_optimize(h_down, target_n, tau_sched, 3000, deadline)
            if h_opt[1] < best_c5:
                best_c5 = h_opt[1]
                best_h = h_opt[0]

    # Final polish: a few more FISTA iters on the best solution
    if time.time() < deadline:
        L = len(best_h)
        h_opt = fista_optimize(best_h, L, [20.0, 50.0, 100.0, 200.0], 3000, deadline)
        if h_opt[1] < best_c5:
            best_c5 = h_opt[1]
            best_h = h_opt[0]

    # Ensure feasibility
    best_h = project_onto_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
