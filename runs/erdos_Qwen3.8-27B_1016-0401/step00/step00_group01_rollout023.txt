import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_val = np.max(corr)
    
    # Active lags (within tolerance)
    tol = 1e-8
    active = np.where(np.abs(corr - max_val) < tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        tau = k - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + tau < n:
                g += (1.0 - h[j + tau])
            if 0 <= j - tau < n:
                g -= h[j - tau]
            grad[j] += g
    grad /= len(active)
    return grad * dx

def project(h, S):
    """Project onto {x: 0<=x<=1, sum(x)=S}"""
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(h + mid, 0.0, 1.0)
        if np.sum(val) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def optimize_from(start, budget_s, n):
    h = start.copy().astype(np.float64)
    S = n / 2.0
    h = project(h, S)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    
    # Projected subgradient descent
    step = 0.01
    n_iter = 0
    while time.time() - t0 < budget_s:
        g = compute_grad(h)
        h_new = h - step * g
        h_new = project(h_new, S)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            # Decrease step slowly
            step *= 0.999
        else:
            step *= 0.9
        h = h_new
        n_iter += 1
        if step < 1e-10:
            break
    
    return best_h, best_c5

            break

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def subgrad_max(h):
    n = len(h)
    dx = 2.0 / n
    lags = compute_lags(h)
    max_v = lags.max()
    tol = max(1e-8, 1e-6 * max(1.0, max_v))
    active = np.where(lags >= max_v - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        tau = int(idx) - (n - 1)
        jp = np.arange(n) + tau
        jm = np.arange(n) - tau
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        g = np.zeros(n)
        g[mask_p] += 1.0 - h[jp[mask_p]]
        g[mask_m] -= h[jm[mask_m]]
        grad += g
    grad /= len(active)
    return grad * dx

def project(h, S):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        v = np.clip(h + mid, 0.0, 1.0)
        if v.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def opt_grad(h, S, t_budget):
    n = len(h)
    h = project(h.astype(float).copy(), S)
    best, best_c = h.copy(), compute_c5(h)
    step = 0.05
    t0 = time.time()
    stall = 0
    while time.time() - t0 < t_budget:
        g = subgrad_max(h)
        gn = np.linalg.norm(g)
        h2 = project(h - step * g / max(gn, 1e-12), S)
        c = compute_c5(h2)
        if c < best_c - 1e-12:
            best, best_c = h2.copy(), c
            h = h2
            step = min(step * 1.005, 1.0)
            stall = 0
        else:
            step *= 0.88
            stall += 1
        if stall > 500 or step < 1e-12:
            break
    return best, best_c

def opt_coord(h, S, t_budget):
    n = len(h)
    h = project(h.astype(float).copy(), S)
    best, best_c = h.copy(), compute_c5(h)
    t0 = time.time()
    sweep_no_improve = 0
    while time.time() - t0 < t_budget and sweep_no_improve < 4:
        improved = False
        for j in range(n):
            old = h[j]
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands += [old, old * 0.5, (old + 1.0) * 0.5]
            best_cand, best_cv = old, best_c

                h[j] = v

                h[j] = v

def opt_lin(h, S, t_budget):

    h = project(h.astype(float).copy(), S)
    best, best_c = h.copy(), compute_c5(h)
    t0 = time.time()
