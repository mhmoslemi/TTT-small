def smooth_obj(x, n, temp, pen_w):
    x = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    m = vals.max()

    s = np.sum(x)
    f += pen_w * (s - n / 2.0) ** 2
    return f


def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-14:

            mask = h > 0
