import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        err = target - np.sum(h)
        if abs(err) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += err / np.sum(free)
        else:
            h += err / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0) - 15

    best_h = None
    best_c5 = np.inf
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    else:
        best_n = 70
        best_h = (np.arange(best_n) % 4 < 2).astype(float)
        best_h = project(best_h, best_n)

    best_c5, _ = compute_c5(best_h)

    # Phase 1: Powell on initial n
    n = best_n
    h = best_h.copy()
    if rem() > 150:
        def obj1(x):
            return compute_c5(project(x, n))[0]
        try:
            res = minimize(obj1, h.copy(), method='Powell',
                         options={'maxiter': 600, 'xtol': 1e-12, 'ftol': 1e-14})
            h_opt = project(res.x, n)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h, h = c5_opt, h_opt, h_opt
        except Exception:
            pass

    # Phase 2: Random perturbation local search with annealing
    if rem() > 20:
        n = best_n
        h = best_h.copy()
        c5_cur, _ = compute_c5(h)
        T = 0.005
        T_min = 0.0001
        cooldown = 0.9995
        for it in range(500000):
            if rem() < 15:
                break
            k = np.random.randint(1, max(2, n // 12) + 1)
            idx = np.random.choice(n, k, replace=False)
            h_new = h.copy()
            h_new[idx] += np.random.randn(k) * 0.025
            h_new = project(h_new, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < c5_cur or np.random.rand() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
                h = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            T = max(T * cooldown, T_min)

    # Phase 3: Try targeted lag reduction
    if rem() > 30:
        n = best_n
        h = best_h.copy()
        c5_cur, corr = compute_c5(h)
        for sweep in range(200):
            if rem() < 15:
                break
            max_idx = np.argmax(corr)
            k_lag = max_idx - (n - 1)
            if k_lag == 0:
                # Reduce sum h*(1-h): push toward binary
                for _ in range(n):
                    i = np.random.randint(n)
                    if h[i] > 0.5:
                        h[i] += 0.01
                    else:
                        h[i] -= 0.01
                    h = project(h, n)
            else:
                # For lag k, reduce h[i]*(1-h[i+k]) contributions
                for _ in range(n // 2):
                    i = np.random.randint(0, n - abs(k_lag))
                    j = i + abs(k_lag)
                    if h[i] > h[j]:
                        h[i] -= 0.005
                        h[j] += 0.005
                    else:
                        h[i] += 0.005
                        h[j] -= 0.005
                    h = project(h, n)
            c5_new, corr = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h.copy()

    # Phase 4: Try other n values
    for n_try in [64, 80, 96, 112, 128, 160]:
        if rem() < 90:
            break
        h_interp = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                            np.linspace(0, 2, best_n, endpoint=False), best_h)
        h_interp = project(h_interp, n_try)
        c5_try, _ = compute_c5(h_interp)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_interp, n_try
        if rem() < 90:
            break
        def obj_local(x, nn=n_try):
            return compute_c5(project(x, nn))[0]
        try:
            res = minimize(obj_local, h_interp.copy(), method='Powell',
                         options={'maxiter': 300, 'xtol': 1e-12, 'ftol': 1e-14})
            h_opt = project(res.x, n_try)
            c5_opt, _ = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt, n_try
        except Exception:
            pass
        # Quick local search at this n
        if rem() > 60:
            h2 = best_h.copy()
            c5_2, _ = compute_c5(h2)
            for _ in range(5000):
                if rem() < 40:
                    break
                kk = np.random.randint(1, 4)
                idx = np.random.choice(n_try, kk, replace=False)
                h2[idx] += np.random.randn(kk) * 0.02
                h2 = project(h2, n_try)
                c5_new, _ = compute_c5(h2)
                if c5_new < c5_2:
                    c5_2 = c5_new
                    if c5_2 < best_c5:

    # Final polish: one more Nelder-Mead if time
    if rem() > 60:
        n_f = best_n
        def obj_f(x):
            return compute_c5(project(x, n_f))[0]
        try:
            simplex = best_h.copy()
            simplex[0] += 0.05
            simplex = project(simplex, n_f)
            res = minimize(obj_f, best_h.copy(), method='Nelder-Mead',
                         options={'maxiter': 1000, 'xatol': 1e-10, 'fatol': 1e-12})
            h_opt = project(res.x, n_f)

        except Exception:
            pass

    best_h = project(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, best_n
