    h = project(h, target)
    c = compute_c(h)
    best_s = get_score(c, n)
    best_h = h.copy()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    no_imp = 0
    nd = n - 1
    rng = np.random.RandomState(seed)
    while time.time() < t_end:
        cmax = np.max(c)
        r = cmax - np.min(c)
        thresh = cmax - top_frac * (r + 1e-10)
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        g = np.zeros(n)
        for p in active:
            g += subgrad_shift(h, p - nd)
        g /= len(active)
        v = beta * v + (1.0 - beta) * g
        h = project(h - lr * v, target)
        c = compute_c(h)
        s_val = get_score(c, n)
        if s_val < best_s - 1e-9:
            best_s = s_val
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 800:
            lr *= 0.6
            v *= 0.5
            no_imp = 0
            if lr < 1e-6:
                break
        elif no_imp > 200:
            lr = min(0.08, lr * 1.05)
        if int(no_imp) % 500 == 0 and lr > 0.01:
            h = project(h + rng.randn(n) * lr * 0.3, target)
    return best_h, best_s

def upsample(h, factor):
    n = len(h)
    n2 = n * factor
    h2 = np.empty(n2)
    for i in range(n2):
        h2[i] = h[i // factor]
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    target0 = 0.5 * n0
    h0 = project(h0, target0)
    best_h = h0.copy()
    best_score = get_score(compute_c(h0), n0)
    best_n = n0

    # Phase 1: Optimize at original resolution
    t1 = t0 + 0.5 * (budget_s - 30)
    h_opt, s_opt = optimize_h(h0.copy(), t1, seed, top_frac=0.04, lr_init=0.04)
    if s_opt < best_score:
        best_score, best_h, best_n = s_opt, h_opt, len(h_opt)

    # Phase 2: Upsample and refine
    t2 = t0 + 0.85 * (budget_s - 30)
    remaining = t2 - time.time()
    if remaining > 30:
        for factor in [2]:
            n2 = best_n * factor
            if n2 > 800:
                continue
            h2 = upsample(best_h, factor)
            h2 = project(h2, 0.5 * n2)
            h_opt2, s_opt2 = optimize_h(h2, t2, seed + 500, top_frac=0.03, lr_init=0.02)
            if s_opt2 < best_score:
                best_score, best_h, best_n = s_opt2, h_opt2, n2

    # Phase 3: Random restarts and local improvements
    t3 = t0 + budget_s - 5
    rng = np.random.RandomState(seed + 999)
    attempt = 0
    while time.time() < t3:
        n_r = best_n
        h_r = best_h.astype(float)
        strategy = attempt % 3
        if strategy == 0:
            h_r = project(h_r + rng.randn(n_r) * 0.15, 0.5 * n_r)
        elif strategy == 1:
            # Binary-ish restart
            h_r = (rng.rand(n_r) < 0.5).astype(float)
            h_r = project(h_r, 0.5 * n_r)
            h_r = h_r * 0.8 + 0.1  # slightly relax binary
            h_r = project(h_r, 0.5 * n_r)
        else:
            # Block pattern
            block = n_r // 7 if n_r >= 7 else n_r
            h_r = np.zeros(n_r)
            for b in range(0, n_r, block * 2):
                h_r[b:b + block] = 1.0
            h_r = project(h_r, 0.5 * n_r)

        sub_t = min(time.time() + 40, t3)
        h_opt_r, s_opt_r = optimize_h(h_r, sub_t, seed + 1000 + attempt * 37,
                                      top_frac=0.05, lr_init=0.03)
        if s_opt_r < best_score:
            best_score, best_h, best_n = s_opt_r, h_opt_r, n_r
        attempt += 1

    # Final verification
    c_final = compute_c(best_h)
    final_score = get_score(c_final, best_n)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, best_n
