        return h
    h = h + diff / n
    return np.clip(h, 0, 1)

    return np.correlate(h, 1 - h, mode='full') * dx

    corr = _correlate_matrix(h, n, dx)

def project_feasible(h, n):
    """Project h onto {0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0, 1)
