from scipy.optimize import minimize
import time

def evaluate(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    return np.max(corr) * dx

def project(h, target_sum):
    """Project h onto {0≤x≤1, sum(x)=target_sum}."""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Adjust for sum constraint
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Simple: distribute diff evenly, then re-clip and iterate
        for _ in range(100):
            h = h + diff / n
            h_clipped = np.clip(h, 0, 1)
            new_diff = target_sum - np.sum(h_clipped)
            if abs(new_diff) < 1e-10:
                break
            h = h_clipped
            diff = new_diff
        h = np.clip(h, 0, 1)
    return h

def subgradient_descent(h, target_sum, max_iter=5000, lr=None):
    """Minimize max overlap using subgradient method."""
    n = len(h)
    dx = 2.0 / n
    best_c = evaluate(h)
    best_h = h.copy()
    
    if lr is None:
        lr = 0.05 / n
    
    for it in range(max_iter):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode="full")
        max_idx = np.argmax(corr)
        c_val = corr[max_idx] * dx
        
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h.copy()
        
        # Compute subgradient at the argmax lag
        # corr[max_idx] = sum_i h[i + lag] * (1-h[i]) where lag = max_idx - (n-1)
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            # S_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            k = lag
            for i in range(n - k):

                # (1-h[i]) negative term: grad[i] += -h[i+k]
                grad[i+k] += (1 - h[i])
                grad[i] -= h[i+k]
        else:

            k = -lag
            # S_{-k} = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
            for i in range(k, n):
                grad[i] += (1 - h[i-k])
                grad[i-k] -= h[i]
        
        # Project gradient onto constraint space (orthogonal to all-ones vector)
        grad_mean = np.mean(grad)
        grad = grad - grad_mean
        
        # Subgradient step
        step = lr / (1.0 + it * 1.0 / max_iter)
        h = h - step * grad
        h = project(h, target_sum)
        
        # Adapt learning rate
        if c_val > best_c:
            lr *= 0.9
        else:
            lr = min(lr * 1.01, 0.1 / n)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert to binary, maintaining sum constraint."""
    target = n / 2
    h_bin = (h > 0.5).astype(float)
    s = h_bin.sum()
    if s > target:
        # Flip some 1s to 0s (those closest to 0.5 in original)
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        n_flip = int(s - target)
        h_bin[ones_sorted[:n_flip]] = 0
    elif s < target:
        # Flip some 0s to 1s
        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        n_flip = int(target - s)
        h_bin[zeros_sorted[:n_flip]] = 1
    return h_bin

def local_search_binary(h, max_iters=100000):
    """Local search for binary solution: flip pairs of bits."""
    n = len(h)
    best_c = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        # Try random pair flip
        i, j = np.random.choice(n, 2, replace=False)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c <= best_c:
            best_c = c
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    
    return best_h, best_c

def grad_at_max(h, k_star):
    """Gradient of the overlap at lag k_star w.r.t. h."""
    n = len(h)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        k = lag
        # S_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j]: (1-h[j-k]) for j=k..n-1; -h[j+k] for j=0..n-1-k

            for j in range(0, n - k):
                grad[j] -= h[j + k]

        for j in range(0, n - k):
            grad[j] += (1.0 - h[j + k])
        for j in range(k, n):
