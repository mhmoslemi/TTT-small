def project_simplex(v, target_sum):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(v)
    v = np.clip(v, 0, 1)
    # Now adjust to get sum = target_sum
    # Use bisection on offset
    s = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):

        proj = np.clip(v + mid, 0, 1)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo+hi)/2, 0, 1)
    # Final correction
    diff = s - proj.sum()
    if abs(diff) > 1e-10:
        # Distribute difference to elements not at boundary
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj
