    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(x, n):
    target = n / 2.0
    def f(lam):
        return float(np.sum(np.clip(x + lam, 0.0, 1.0)) - target)
    try:
        lam = brentq(f, -3.0, 3.0, xtol=1e-14)
    except ValueError:
        # Fallback: binary search
        lo, hi = -3.0, 3.0
        for _ in range(100):
            mid = (lo + hi) / 2.0
            if f(mid) > 0:
                hi = mid
            else:
                lo = mid
        lam = (lo + hi) / 2.0

        jp = j + k

def grad_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    one_h = 1.0 - h
    if s >= 0:
        if s < n:
            g[:n - s] += one_h[s:]
            g[s:] -= h[:n - s]
    else:

def grad_c5(h, n_top=6):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:n_top]
    vals = corr[idx]
    spread = vals[0] - vals[-1]
    if spread < 1e-10:
        w = np.ones(len(idx)) / len(idx)
    else:
        w = np.exp((vals - vals[0]) / max(spread, 0.5))
        w /= w.sum()
    grad = np.zeros(n)
    for ii, m in enumerate(idx):
        wi = w[ii]
        if wi < 1e-8:
            continue
        s = m - (n - 1)
        grad += wi * grad_for_shift(h, s)
    return grad

def flip_random(h, rng, n_flips):

    h0 = project_h(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: optimize the parent construction
    h1, c1 = optimize_h(h0, max_iters=10000, time_limit=50.0)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: multiple restarts with varying perturbations
    n_restarts = 45
    t_alloc = max(3.0, (budget_s - (time.time() - t_start) - 80.0) / n_restarts)
    for r in range(n_restarts):
        if time.time() - t_start > budget_s - 80.0:
            break
        t_left = budget_s - 80.0 - (time.time() - t_start)
        tl = min(t_alloc, max(2.0, t_left / max(n_restarts - r, 1)))
        frac = r / max(n_restarts - 1, 1)
        scale = 0.02 + 0.35 * frac
        if r % 3 == 0:
            h_p = flip_random(best_h, rng, max(3, n0 // 10))
        else:
            h_p = perturb(best_h, rng, scale)
        h_r, c_r = optimize_h(h_p, max_iters=4000, time_limit=tl)
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r

    # Phase 3: try a couple of different resolutions
    for n_try in [90, 120]:
        if time.time() - t_start > budget_s - 70.0:
            break
        n_cur = len(best_h)
        xo = np.linspace(0.0, 2.0, n_cur)
        xn = np.linspace(0.0, 2.0, n_try)
        h_init = project_h(np.interp(xn, xo, best_h), n_try)
        h3, c3 = optimize_h(h_init, max_iters=8000, time_limit=min(55.0, budget_s - 70.0 - (time.time() - t_start)))
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Final cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    nf = len(best_h)
    best_h = project_h(best_h, nf)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, nf
