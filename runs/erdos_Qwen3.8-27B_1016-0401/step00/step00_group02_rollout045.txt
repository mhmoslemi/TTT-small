def project_onto_feasible(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2} using water-filling."""
    target = n / 2.0
    h = h.copy()
    # Iterate water-filling
    for _ in range(100):
        # First, clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase, only where h < 1
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to decrease, only where h > 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0, 1)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute remaining evenly
    mask = (h > 0) & (h < 1)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)
