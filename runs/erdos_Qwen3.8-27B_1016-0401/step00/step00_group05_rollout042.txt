    if t_rem > 60:
        h_cont = proj_sum(best_h + rng.randn(n) * 0.1, target_sum)
        h4, c4 = optimize_continuous(h_cont, dx, target_sum, min(t_rem * 0.5, 200), rng)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4

    # Phase 5: final binary polish
    t_rem = elapsed - (time.time() - t0)
    if t_rem > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        excess = h_bin2.sum() - target_sum
        if excess > 0:
            idx = np.where(h_bin2 == 1)[0]
            rng.shuffle(idx)
            for i in idx[:int(excess)]:
                h_bin2[i] = 0
        elif excess < 0:
            idx = np.where(h_bin2 == 0)[0]
            rng.shuffle(idx)
            for i in idx[:int(-excess)]:
                h_bin2[i] = 1
        h5, c5v = optimize_binary(h_bin2, dx, target_sum, min(t_rem * 0.8, 200), rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h5

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
