    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    j_star = int(np.argmax(corr))
    grad = np.zeros(n)
    k_lo = max(0, j_star - (n - 1))
    k_hi = min(n - 1, j_star)
    if k_hi >= k_lo:
        idx = np.arange(k_lo, k_hi + 1)
        grad[idx] = 1.0 - 2.0 * h[j_star - idx]
    return grad * dx

def smooth_obj(h):
    n = len(h)
    dx = 2.0 / n
    temp = 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w_sum = w.sum()
    w /= w_sum
    return float(np.dot(w, corr)) * dx

def smooth_grad(h):
    n = len(h)
    dx = 2.0 / n
    temp = 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-10:
            continue
        k_lo = max(0, j - (n - 1))
        k_hi = min(n - 1, j)
        if k_hi >= k_lo:
            idx = np.arange(k_lo, k_hi + 1)
            grad[idx] += w[j] * (1.0 - 2.0 * h[j - idx])
    return grad * dx

def eq_constraint(h):
    return h.sum() - len(h) / 2.0

def eq_jacobian(h):
    return np.ones(len(h))

def project_sum(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = h_p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h_p.sum()
    if abs(diff) > 1e-10:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if mask.any():
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = 70
    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    if h0 is None:
        h0 = np.zeros(n)
        h0[::2] = 1.0

    target_sum = n / 2.0
    h0 = project_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = obj_func(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian}]

    def elapsed():
        return time.time() - t0

    # Phase 1: Optimize from initial point
    try:
        res = minimize(smooth_obj, h0, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.fun < best_c5 + 1e-8:
            hc = project_sum(res.x, target_sum)
            c5c = obj_func(hc)
            if c5c < best_c5:
                best_h = hc
                best_c5 = c5c
    except Exception:
        pass

    # Phase 2: Random restarts with binary initializations
    max_restarts = 200
    for r in range(max_restarts):
        if elapsed() > budget_s - 60:
            break

        h_rand = np.zeros(n)
        ones = rng.choice(n, int(target_sum), replace=False)
        h_rand[ones] = 1.0
        h_rand += rng.normal(0.0, 0.08, n)
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = project_sum(h_rand, target_sum)

        try:
            res = minimize(smooth_obj, h_rand, jac=smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-11})
            c5c = obj_func(project_sum(res.x, target_sum))
            if c5c < best_c5:
                best_h = project_sum(res.x, target_sum)
                best_c5 = c5c
        except Exception:
            pass

    # Phase 3: Refine best solution with hard max gradient
    for _ in range(3):
        if elapsed() > budget_s - 20:
            break
        try:
            res = minimize(obj_func, best_h, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-13})
            hc = project_sum(res.x, target_sum)
            c5c = obj_func(hc)
            if c5c < best_c5:
                best_h = hc
                best_c5 = c5c
        except Exception:
            break

    # Phase 4: Try a few more restarts focused on promising regions
    for r in range(30):
        if elapsed() > budget_s - 10:
            break
        h_pert = best_h + rng.normal(0.0, 0.1, n)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_sum(h_pert, target_sum)
        try:
            res = minimize(smooth_obj, h_pert, jac=smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-11})
            c5c = obj_func(project_sum(res.x, target_sum))
            if c5c < best_c5:
                best_h = project_sum(res.x, target_sum)
                best_c5 = c5c
        except Exception:
            pass

    # Final cleanup
    best_h = project_sum(best_h, target_sum)
    best_c5 = obj_func(best_h)

    return best_h, best_c5, n
