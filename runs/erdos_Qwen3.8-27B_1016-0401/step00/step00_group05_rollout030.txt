def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(v, target_sum):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - float(h.sum())
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / float(free.sum())
            h = np.clip(h, 0.0, 1.0)
    return h


def gradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mx = float(corr.max())

    grad = np.zeros(n)
    for k in active:

            grad[lo1:hi1] += omh[lo1 + j: hi1 + j]
        lo2, hi2 = max(0, j), min(n, n + j)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 - j: hi2 - j]
    grad *= dx / float(len(active))
    return grad
