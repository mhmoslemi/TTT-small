import numpy as np
import time

def score_func(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

def init_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def swap1(h, budget):
    n = len(h)
    h = h.astype(float).copy()
    best, best_s = h.copy(), score_func(h)
    t = time.time()
    while time.time() - t < budget:
        p = np.where(h == 1)[0]
        q = np.where(h == 0)[0]
        if len(p) == 0 or len(q) == 0:
            break
        hit = False
        for i in p:
            if time.time() - t >= budget:
                return best, best_s
            for j in q:
                h[i], h[j] = 0.0, 1.0
                s = score_func(h)
                if s < best_s - 1e-14:
                    best_s, best, hit = s, h.copy(), True
                    break
                h[i], h[j] = 1.0, 0.0
            if hit:
                break
        if not hit:
            break
    return best, best_s

def swap2sa(h, budget, rng):
    cur = h.astype(float).copy()
    cur_s = score_func(cur)
    best, best_s = cur.copy(), cur_s
    t0 = time.time()
    while time.time() - t0 < budget:
        p = np.where(cur > 0.5)[0]
        q = np.where(cur < 0.5)[0]
        if len(p) < 2 or len(q) < 2:
            break
        i1, i2 = rng.choice(p, 2, replace=False)
        j1, j2 = rng.choice(q, 2, replace=False)
        cur[i1], cur[i2], cur[j1], cur[j2] = 0, 0, 1, 1
        s = score_func(cur)
        fr = min(1.0, (time.time() - t0) / max(budget, 1e-6))
        T = 0.005 * (1.0 - fr) + 1e-4
        if s < best_s:
            best_s, best, cur_s = s, cur.copy(), s
        elif rng.random() < np.exp(-(s - cur_s) / T):
            cur_s = s
        else:
            cur[i1], cur[i2], cur[j1], cur[j2] = 1, 1, 0, 0
    return best, best_s

def contsa(h, budget, rng):

def fix_sum(h, target):

    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:

        diff -= step if diff > 0 else -step
