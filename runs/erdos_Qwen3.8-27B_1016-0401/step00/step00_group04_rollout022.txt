from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - t, 0.0, 1.0))
        if s > target_sum:
            lo = t
        else:
            hi = t
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj(h, target_sum, temp):
    hp = project_feasible(h, target_sum)
    n = len(hp)
    dx = 2.0 / n
    corr = np.correlate(hp, 1.0 - hp, mode='full') * dx
    z = corr / temp
    z = z - z.max()
    return temp * (z.max() + np.log(np.sum(np.exp(z))))

def nm_result(h_start, target, maxiter=2000):
    def obj(x):
        return compute_c5(project_feasible(x, target))
    res = minimize(obj, h_start, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10,
                            'fatol': 1e-12, 'initial_simplex_size': 0.08})
    hc = project_feasible(res.x, target)
    return hc, compute_c5(hc)
