                h[i] -= adj
                diff += adj
                if diff >= -1e-12:
                    break
    return h

def _obj(x):
    n = _opt_n
    target = _opt_target
    dx = _opt_dx
    xc = np.clip(x, 0.0, 1.0)
    s = np.sum(xc)
    if s < 1e-15:
        return 1.0
    xp = xc * (target / s)
    xp = np.clip(xp, 0.0, 1.0)
    return compute_c5(xp, dx)

def _con_fun(x):
    return np.sum(x) - _opt_target

def _con_jac(x):
    return np.ones(_opt_n)

def _soft_obj(x, beta):
    n = _opt_n
    target = _opt_target
    dx = _opt_dx
    xc = np.clip(x, 0.0, 1.0)
    s = np.sum(xc)
    if s < 1e-15:
        return 1.0
    xp = xc * (target / s)
    xp = np.clip(xp, 0.0, 1.0)
    corr = np.correlate(xp, 1.0 - xp, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp((corr - m) / beta))) * beta
    return sm * dx

def run(seed=42, budget_s=1100, **kwargs):
    global _opt_n, _opt_dx, _opt_target
    np.random.seed(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except Exception:
        h_init = np.full(45, 0.5)
        n_init = 45

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, 2.0 / n_init)
    best_n = n_init

    n_candidates = sorted(set([n_init, 30, 40, 45, 50, 60, 80, 100, 120, 150, 200]))
    n_candidates = [n for n in n_candidates if n <= 500]

    for n in n_candidates:
        if time.time() - t_start > budget_s * 0.82:
            break
        dx = 2.0 / n
        target = n / 2.0
        _opt_n = n
        _opt_dx = dx
        _opt_target = target

        starts = []
        if n == n_init:
            starts.append(h_init.copy())
        else:
            x_old = np.linspace(0.0, 2.0, n_init)
            x_new = np.linspace(0.0, 2.0, n)
            starts.append(project_feasible(np.interp(x_new, x_old, np.interp(
                np.linspace(0, 2, min(n_init, n)),
                np.linspace(0, 2, n_init), h_init)), n))
            # Also interp directly
            starts.append(project_feasible(np.interp(x_new, x_old, h_init), n))

        for _ in range(4):
            r = np.random.rand(n) * 0.7 + 0.15
            starts.append(project_feasible(r, n))

        x = np.linspace(0.0, 2.0, n)
        starts.append(project_feasible(0.5 + 0.35 * np.sin(2 * np.pi * x / 0.5), n))
        starts.append(project_feasible(0.5 + 0.35 * np.cos(2 * np.pi * x / 0.7), n))
        blk = np.zeros(n)
        blk[0:n//5] = 1.0
        blk[2*n//5:3*n//5] = 1.0
        blk[4*n//5:] = 1.0
        starts.append(project_feasible(blk, n))

        bounds = [(0.0, 1.0)] for _ in range(n)
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

        for h0 in starts:
            if time.time() - t_start > budget_s * 0.88:
                break

            # SLSQP
            try:
                res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
                h_opt = project_feasible(res.x, n)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_opt.copy(), n
            except Exception:
                pass

            # Nelder-Mead (no bounds/constraints handled internally, use projected)
            if time.time() - t_start < budget_s * 0.86:
                try:
                    def _nm_obj(xx):
                        xp = project_feasible(xx, n)
                        return compute_c5(xp, dx)
                    res2 = minimize(_nm_obj, h0, method='Nelder-Mead',
                                    options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
                    h_opt2 = project_feasible(res2.x, n)
                    c5_2 = compute_c5(h_opt2, dx)
                    if c5_2 < best_c5:
                        best_c5, best_h, best_n = c5_2, h_opt2.copy(), n
                except Exception:
                    pass

    # Coordinate descent refinement on best solution
    if time.time() - t_start < budget_s * 0.92:
        n = best_n
        dx = 2.0 / n
        h = best_h.copy()
        c5 = best_c5
        target = n / 2.0

        deltas = [0.03, -0.03, 0.06, -0.06, 0.01, -0.01, 0.1, -0.1, 0.15, -0.15]
        for iteration in range(3000):
            if time.time() - t_start > budget_s * 0.95:
                break
            improved = False
            for i in range(n):
                for d in deltas:
                    h_new = h.copy()
                    h_new[i] = np.clip(h[i] + d, 0.0, 1.0)
                    h_new = project_feasible(h_new, n)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < c5 - 1e-11:
                        h = h_new
                        c5 = c5_new
                        improved = True
                        break
            if not improved:
                # Try smaller steps
                for i in range(n):
                    for d in [0.005, -0.005, 0.002, -0.002]:
                        h_new = h.copy()
                        h_new[i] = np.clip(h[i] + d, 0.0, 1.0)
                        h_new = project_feasible(h_new, n)
                        c5_new = compute_c5(h_new, dx)
                        if c5_new < c5 - 1e-12:
                            h = h_new
                            c5 = c5_new
                            improved = True
                            break
                    if improved:
                        break
            if not improved and iteration > 100:
                break

        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h, n

    # Second refinement pass with finer steps
    if time.time() - t_start < budget_s * 0.97:
        n = best_n
        dx = 2.0 / n
        h = best_h.copy()
        c5 = best_c5
        fine_deltas = [0.008, -0.008, 0.004, -0.004, 0.002, -0.002]
        for iteration in range(2000):
            if time.time() - t_start > budget_s * 0.99:
                break
            improved = False
            for i in range(n):
                for d in fine_deltas:
                    h_new = h.copy()
                    h_new[i] = np.clip(h[i] + d, 0.0, 1.0)
                    h_new = project_feasible(h_new, n)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < c5 - 1e-12:
                        h = h_new
                        c5 = c5_new
                        improved = True
                        break
            if not improved:
                break
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_n = len(best_h)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
