```python
def compute_delta_matrices(h, n):
    num_k = 2*n - 1
    s_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    D1 = np.zeros((n, num_k))
    D0 = np.zeros((n, num_k))
    
    for a in range(n):
        pos_p = a + s_vals  # shape (2n-1,)
        pos_m = a - s_vals  # shape (2n-1,)
        
        valid_p = (pos_p >= 0) & (pos_p < n)
        valid_m = (pos_m >= 0) & (pos_m < n)
        
        # Clip for safe indexing
        cp = np.clip(pos_p, 0, n-1)
        cm = np.clip(pos_m, 0, n-1)
        
        # D1: flip h[a] from 1 to 0
        # ΔC[s] = -(1-h[a+s]) if valid, +h[a-s] if valid
        # But s=0: ΔC[0] = 0
        D1[a] = np.where(valid_p, -(1 - h[cp]), 0.0) + np.where(valid_m, h[cm], 0.0)
        D1[a, s_vals == 0] = 0  # fix s=0 case
        
        # D0: flip h[b] from 0 to 1
        D0[a] = np.where(valid_p, (1 - h[cp]), 0.0) - np.where(valid_m, h[cm], 0.0)
        D0[a, s_vals == 0] = 0
    
    return D1, D0
```

        cp = np.clip(pos_p, 0, n - 1)
        cm = np.clip(pos_m, 0, n - 1)
        D1[a] = np.where(valid_p, -(1.0 - h[cp]), 0.0) + np.where(valid_m, h[cm], 0.0)
        D1[a, zero_s_idx] = 0.0
        D0[a] = np.where(valid_p, 1.0 - h[cp], 0.0) - np.where(valid_m, h[cm], 0.0)
        D0[a, zero_s_idx] = 0.0
    return D1, D0

def local_search_fast(h_init, dx, end_time, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.astype(np.float64).copy()
    # Ensure binary
    h = (h > 0.5).astype(np.float64)
    # Fix sum if needed
    target_sum = n // 2
    s = int(h.sum())
    if s > target_sum:
        ones = np.where(h > 0.5)[0]
        for i in range(s - target_sum):
            h[ones[i]] = 0.0
    elif s < target_sum:
        zeros = np.where(h < 0.5)[0]
        for i in range(target_sum - s):
            h[zeros[i]] = 1.0
    
    full_corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(full_corr) * dx
    best_c5 = c5
    best_h = h.copy()
    D1, D0 = compute_delta_matrices(h, n)
    
    num_restarts = 0
    max_restarts = 200
    
    while time.time() < end_time and num_restarts < max_restarts:
        # Greedy phase
        while time.time() < end_time:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            best_imp = 1e-12
            best_a = -1
            best_b = -1
            
            for a in ones:
                deltas = D1[a][None, :] + D0[zeros]
                new_max = np.max(full_corr[None, :] + deltas, axis=1) * dx
                min_val = np.min(new_max)
                if min_val < c5 - best_imp:
                    best_imp = c5 - min_val
                    best_a = a
                    best_b = zeros[np.argmin(new_max)]
            
            if best_a < 0:
                break
            
            # Accept flip
            h[best_a] = 0.0
            h[best_b] = 1.0
            full_corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(full_corr) * dx
            D1, D0 = compute_delta_matrices(h, n)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
        
        # Perturbation
        num_restarts += 1
        h = best_h.copy()
        c5 = best_c5
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        D1, D0 = compute_delta_matrices(h, n)
        
        n_flips = rng.randint(2, min(10, n // 3))
        for _ in range(n_flips):
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a != b:
                h[a] = 0.0
                h[b] = 1.0
        c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        D1, D0 = compute_delta_matrices(h, n)
    
    return best_h, best_c5

def double_grid(h):
    n = len(h)
    h_new = np.zeros(2 * n)
    h_new[0::2] = h
    h_new[1::2] = h
    return h_new

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.randint(0, 2, size=n).astype(np.float64)
    target = n // 2
    s = int(h.sum())
    if s > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        for i in range(s - target):
            h[ones[i]] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for i in range(target - s):
            h[zeros[i]] = 1.0
    return h
