            h = np.clip(h, 0, 1)

        if mask.sum() > 0:

            h = np.clip(h, 0, 1)
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

            p = -m
            for i in range(p, n):
                grad[i] += wk * dx * (1.0 - h[i - p])
                grad[i - p] -= wk * dx * h[i]
    return grad

def smooth_softmin(h, n, alpha):
    """Compute softmax weights over correlation terms."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = alpha * corr * dx
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    return w, corr

def binary_local_search(b, n, max_time, rng, current_best):
    """Flip-pair local search on binary sequence. Returns improved (b, c5)."""
    c5_b = compute_c5(b)
    improvement = True
    while improvement and time.time() < max_time:
        improvement = False
        pass_best_delta = 1e14
        best_flip = None
        ones_idx = np.where(b == 1)[0]
        zeros_idx = np.where(b == 0)[0]
        for i in ones_idx:
            for j in zeros_idx:
                b[i] = 0
                b[j] = 1
                c5_try = compute_c5(b)
                delta = c5_try - c5_b
                if delta < pass_best_delta:
                    pass_best_delta = delta
                    best_flip = (i, j)
                b[i] = 1
                b[j] = 0
        if best_flip is not None and pass_best_delta < -1e-14:
            i, j = best_flip
            b[i] = 0
            b[j] = 1
            c5_b += pass_best_delta
            improvement = True
            if c5_b < current_best[0]:
                current_best[0] = c5_b
                current_best[1] = b.copy()
        if time.time() > max_time:
            break
    return b, c5_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    deadline = t0 + budget_s

    h_cur = proj_constraint(h0.copy(), target)
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur

    # Phase 1: Projected gradient descent with smooth max
    alpha = 25.0
    step = 0.008
    stagnant = 0
    for it in range(50000):
        if time.time() > deadline - budget_s * 0.4:
            break
        w, corr = smooth_softmin(h_cur, n, alpha)
        grad = weighted_corr_grad(h_cur, n, w)
        grad -= grad.mean()
        
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_try = proj_constraint(h_cur - s * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-14:
                h_cur = h_try
                c5_cur = c5_try
                improved = True
                break
        if improved:
            stagnant = 0
            step = min(step * 1.03, 0.05)
            if c5_cur < c5_best - 1e-13:
                h_best = h_cur.copy()
                c5_best = c5_cur
        else:
            stagnant += 1
            step *= 0.7
            if step < 1e-7:
                break
        
        if stagnant > 200 and it % 300 == 0:
            noise = 0.008 + 0.004 * rng.random()
            h_p = h_cur + rng.normal(0, noise, n)
            h_p = proj_constraint(h_p, target)
            c5_p = compute_c5(h_p)
            if c5_p < c5_cur + 0.02:
                h_cur = h_p
                c5_cur = c5_p
            stagnant = 0
            step = 0.005

    # Phase 2: Binary local search
    t_phase2_end = min(deadline - 2.0, t0 + budget_s * 0.75)
    if time.time() < t_phase2_end:
        b = (h_best > 0.5).astype(float)
        n_ones = int(b.sum())
        if n_ones > n // 2:
            idx1 = np.where(b == 1)[0]
            idx1 = idx1[np.argsort(h_best[idx1])]
            b[idx1[:n_ones - n // 2]] = 0
        elif n_ones < n // 2:
            idx0 = np.where(b == 0)[0]
            idx0 = idx0[np.argsort(-h_best[idx0])]
            b[idx0[:n // 2 - n_ones]] = 1
        c5_b = compute_c5(b)
        best_track = [min(c5_best, c5_b), (h_best if c5_best <= c5_b else b).copy()]
        b, c5_b = binary_local_search(b, n, t_phase2_end, rng, best_track)
        if c5_b < c5_best:
            h_best = b.copy()
            c5_best = c5_b

    # Phase 3: Continuation with different alpha and more iterations
    if time.time() < deadline - 5.0:
        h_cur = h_best.copy()
        c5_cur = c5_best
        alpha = 50.0
        step = 0.003
        for it in range(30000):
            if time.time() > deadline - 3.0:
                break
            w, corr = smooth_softmin(h_cur, n, alpha)
            grad = weighted_corr_grad(h_cur, n, w)
            grad -= grad.mean()
            improved = False
            for s in [step, step * 0.5, step * 0.2]:
                h_try = proj_constraint(h_cur - s * grad, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-14:
                    h_cur = h_try
                    c5_cur = c5_try
                    improved = True
                    break
            if improved:
                step = min(step * 1.02, 0.02)
                if c5_cur < c5_best - 1e-13:
                    h_best = h_cur.copy()
                    c5_best = c5_cur
            else:
                step *= 0.6
                if step < 1e-8:
                    break
        
        # Try binary polish on best continuous
        b2 = (h_best > 0.5).astype(float)
        n_ones = int(b2.sum())
        if n_ones > n // 2:
            idx1 = np.where(b2 == 1)[0]
            idx1 = idx1[np.argsort(h_best[idx1])]
            b2[idx1[:n_ones - n // 2]] = 0
        elif n_ones < n // 2:
            idx0 = np.where(b2 == 0)[0]
            idx0 = idx0[np.argsort(-h_best[idx0])]
            b2[idx0[:n // 2 - n_ones]] = 1
        c5_b2 = compute_c5(b2)
        if c5_b2 < c5_best:
            h_best = b2.copy()
            c5_best = c5_b2
            t_pe = min(deadline - 2.0, time.time() + 20.0)
            best_t = [c5_best, h_best.copy()]
            b2, c5_b2 = binary_local_search(b2, n, t_pe, rng, best_t)
            if c5_b2 < c5_best:
                h_best = b2.copy()
                c5_best = c5_b2

    # Final constraint enforcement
    h_best = proj_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
