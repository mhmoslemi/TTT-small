    corr = np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    smax = logsumexp(T * corr) / T
    cr = np.sum(h) - 0.5 * n
    return smax * dx + pen_w * cr * cr

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()

        if diff > 0:
            room = 1.0 - h
        else:

            i = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
