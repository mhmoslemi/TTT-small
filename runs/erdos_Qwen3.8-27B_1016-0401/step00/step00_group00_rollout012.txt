        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)

        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)

def smax(c, t):
