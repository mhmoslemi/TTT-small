def smooth_c5_obj(x, n, tau):
    h = project_to_feasible(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + tau * np.log(np.exp((corr - m) / tau).sum()) / tau)

def local_powell(h_start, n, maxiter, timeout=None, t0=None):
    try:
        res = minimize(c5_objective, h_start.copy(), args=(n,), method='Powell',
                       options={'maxiter': maxiter, 'xtol': 1e-9, 'ftol': 1e-11})
        h_out = project_to_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h_start, compute_c5(h_start)

def local_lbfgs(h_start, n, tau, maxiter):
    try:
        res = minimize(lambda x: smooth_c5_obj(x, n, tau), h_start.copy(),
                       method='L-BFGS-B', bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_out = project_to_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h_start, compute_c5(h_start)

def coordinate_descent(h_start, n, n_passes=5, delta_base=0.05):
    h = h_start.copy()
    best_val = compute_c5(h)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for d in [delta_base, -delta_base, delta_base*0.5, -delta_base*0.5]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_feasible(h_new, n)
                val = compute_c5(h_new)
                if val < best_val - 1e-12:
                    h = h_new
                    best_val = val
                    improved = True
                    break
        if not improved:
            break
        delta_base *= 0.7
    return h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s * 0.93 - (time.time() - t0)

    n = len(initial_h_values)
    h_best = project_to_feasible(np.asarray(initial_h_values, dtype=float), n)
    best_c5 = compute_c5(h_best)
    no_improve_count = 0
    sigma = 0.15

    while remaining() > 5:
        frac = (time.time() - t0) / budget_s
        # Adaptive sigma
        if no_improve_count > 30:
            sigma = min(sigma * 1.5, 0.3)
            no_improve_count = 0
        elif no_improve_count > 10:
            sigma = min(sigma * 1.2, 0.25)

        r = rng.random()
        if r < 0.35:
            h_try = h_best + rng.normal(0, sigma, n)
        elif r < 0.6:
            k = max(3, int(n * 0.3))
            idx = rng.choice(n, k, replace=False)
            h_try = h_best.copy()
            h_try[idx] = rng.uniform(0, 1, k)
        elif r < 0.75:
            h_try = 1.0 - h_best + rng.normal(0, sigma * 0.5, n)
        elif r < 0.9:
            # Structured: block pattern
            block_sizes = rng.randint(3, 10, size=max(2, n // 5))
            block_sum = block_sizes.sum()
            scales = np.linspace(0.3, 1.5, len(block_sizes))
            h_try = np.zeros(n)
            pos = 0
            for bi, (bs, sc) in enumerate(zip(block_sizes, scales)):
                end = min(pos + bs, n)
                h_try[pos:end] = sc + rng.normal(0, 0.05, end - pos)
                pos = end
            h_try[:n] = h_try[:n]
        else:
            h_try = rng.uniform(0, 1, n)

        h_try = project_to_feasible(h_try, n)

        # Quick coordinate descent (fast)
        h_cd, c5_cd = coordinate_descent(h_try, n, n_passes=3, delta_base=0.08)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            h_best = h_cd
            no_improve_count = 0
            sigma *= 0.85

        # Powell local refinement (slower, more thorough)
        if remaining() > 15:
            iters = 400 if remaining() > 300 else 200
            h_ref, c5_ref = local_powell(h_best, n, maxiter=iters)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                h_best = h_ref
                no_improve_count = 0
                sigma *= 0.85
            else:
                no_improve_count += 1

        # Periodic L-BFGS-B polish with smooth objective
        if remaining() > 30 and int(time.time() - t0) % 30 < 5:
            tau = 0.003 if frac > 0.5 else 0.01
            h_ref2, c5_ref2 = local_lbfgs(h_best, n, tau, maxiter=300)
            if c5_ref2 < best_c5:
                best_c5 = c5_ref2
                h_best = h_ref2
                no_improve_count = 0

    # Final polish: aggressive local search
    if remaining() > 3:
        h_ref, c5_ref = local_powell(h_best, n, maxiter=3000)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            h_best = h_ref
    if remaining() > 2:
        h_ref, c5_ref = local_lbfgs(h_best, n, 0.001, maxiter=1000)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            h_best = h_ref
    if remaining() > 1:
        h_ref, c5_ref = coordinate_descent(h_best, n, n_passes=20, delta_base=0.02)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            h_best = h_ref

    # Ensure exact feasibility
    h_best = project_to_feasible(h_best, n)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, n
