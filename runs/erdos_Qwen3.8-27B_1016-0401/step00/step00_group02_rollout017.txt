                res = lbfgsb_opt(h_best, n, dx, sum_target, 500, 1500)
                hc = fix_sum(np.clip(res.x, 0, 1), sum_target)
                c = eval_c5(hc)
                if c < c5_best:
                    c5_best, h_best = c, hc

    # Phase 5: Try n=180
    if elapsed() < budget_s * 0.8:
        n3 = 180
        h3 = interpolate_h(h_best, n, n3)
        st3 = n3 / 2.0
        dx3 = 2.0 / n3
        h3 = fix_sum(h3, st3)
        c = eval_c5(h3)
        if c < c5_best:
            c5_best = c
            h_best = h3
            n, sum_target, dx = n3, st3, dx3
        if elapsed() < budget_s * 0.92:
            res = lbfgsb_opt(h_best, n, dx, sum_target, 200, 1000)
            hc = fix_sum(np.clip(res.x, 0, 1), sum_target)
            c = eval_c5(hc)
            if c < c5_best:
                c5_best, h_best = c, hc
            res = slsqp_opt(h_best, n, dx, sum_target, 500)
            hc = fix_sum(np.clip(res.x, 0, 1), sum_target)
            c = eval_c5(hc)
            if c < c5_best:
                c5_best, h_best = c, hc

    # Final hill climb
    if elapsed() < budget_s - 5:
        h_best = hill_climb(h_best, sum_target, dx, rng, 200, [0.05, 0.02, 0.01, 0.005])
        c5_final = eval_c5(h_best)
        if c5_final < c5_best:
            c5_best = c5_final

    # Ensure feasibility
    h_best = np.clip(h_best, 0, 1)
    h_best = fix_sum(h_best, len(h_best) / 2.0)
    c5_final = eval_c5(h_best)
    c5_best = min(c5_best, c5_final)

    return (h_best, c5_best, len(h_best))
