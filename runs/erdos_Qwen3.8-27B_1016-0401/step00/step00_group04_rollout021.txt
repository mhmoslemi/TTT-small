def project_feasible(h, target_sum):
    """Project h to {0≤h≤1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1).copy()

def _score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _project(h, s):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = s - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _subgrad(h):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    mx = corrs.max()
    tol = 1e-9
    g = np.zeros(n)
    for k in range(len(corrs)):
        if corrs[k] < mx - tol:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            i1 = np.arange(n - lag)
            i2 = i1 + lag
            g[i1] += 1.0 - h[i2]
            g[i2] -= h[i1]
        else:

            i1 = np.arange(n - L)
            i2 = i1 + L
            g[i2] += 1.0 - h[i1]
            g[i1] -= h[i2]

    old_n = len(h)

    h_new = np.interp(x_new, x_old, h, left=h[-1], right=h[0])
    h_new = _project(h_new, new_n / 2.0)

    corrs = np.correlate(h, 1.0 - h, mode='full')
    mx = corrs.max()

    for k in range(len(corrs)):

        if lag >= 0:
            i1 = np.arange(n - lag)
            g[i1] += 1.0 - h[i1 + lag]
            g[i1 + lag] -= h[i1]
        else:

    n = len(h_bin)
    h = h_bin.copy()
    score = _score(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for _ in range(max_iters):
        best_gain = 1e-10
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                s = _score(h)
                gain = score - s
                h[i] = 1.0
                h[j] = 0.0
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        score -= best_gain
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)

import numpy as np
import time

def _score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _project(h, s):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = s - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _subgrad(h):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    mx = corrs.max()
    g = np.zeros(n)
    for k in range(len(corrs)):
        if corrs[k] < mx - 1e-9:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            i1 = np.arange(n - lag)
            g[i1] += 1.0 - h[i1 + lag]
            g[i1 + lag] -= h[i1]
        else:
            L = -lag
            i1 = np.arange(n - L)
            g[i1 + L] += 1.0 - h[i1]
            g[i1] -= h[i1 + L]
    return g * (2.0 / n)

def _greedy_binary(h_bin, max_rounds=200):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    score = _score(h)
    for _ in range(max_rounds):
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        best_gain = 1e-12
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                s = _score(h)
                gain = score - s
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        score = _score(h)
    return h, score

            lr = min(lr * 1.01, 0.06)
        else:
            lr *= 0.995
        if rng.random() < 0.08:
            idx = rng.integers(n)
            best_h[idx] = np.clip(best_h[idx] + rng.normal(0, 0.2), 0, 1)
            best_h = _project(best_h, target)

    # Phase 2: Greedy binary local search (35% time)
    t2 = t0 + budget_s * 0.50
    while time.time() < t2:
        hb = (best_h >= 0.5).astype(float)
        hb = _fix_binary(hb, n, rng)
        hb, sb = _greedy_binary(hb)
        if sb < best_score:
            best_h, best_score = hb, sb
        # Perturb for diversification
        for _ in range(3):
            x = rng.integers(n)
            hb[x] = 1.0 - hb[x]
        hb = _fix_binary(hb, n, rng)
        hb, sb = _greedy_binary(hb)
        if sb < best_score:
            best_h, best_score = hb, sb

    # Phase 3: Continuous pairwise search (25% time)
    t3 = t0 + budget_s * 0.75
    while time.time() < t3:
        i, j = rng.integers(n, 2)
        if i == j:
            continue
        sgn = 1.0 if best_h[i] >= best_h[j] else -1.0
        eps = rng.uniform(0.01, 0.2) * sgn
        nh = best_h.copy()
        nh[i] = np.clip(nh[i] + eps, 0, 1)
        nh[j] = np.clip(nh[j] - eps, 0, 1)
        nh = _project(nh, target)
        s = _score(nh)
        if s < best_score:
            best_h, best_score = nh, s

    # Phase 4: Fine polish (17% time)
    while time.time() < t0 + budget_s * 0.97:
        i, j = rng.integers(n, 2)
        if i == j:
            continue
        sgn = 1.0 if best_h[i] >= best_h[j] else -1.0
        eps = rng.uniform(0.001, 0.03) * sgn
        nh = best_h.copy()
        nh[i] = np.clip(nh[i] + eps, 0, 1)
        nh[j] = np.clip(nh[j] - eps, 0, 1)
        nh = _project(nh, target)
        s = _score(nh)
        if s < best_score:
            best_h, best_score = nh, s

    best_h = _project(best_h, target)
    best_score = _score(best_h)
    return best_h, best_score, n
