import numpy as np
import time

def eval_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def ls_pass(h, dx):
    c = eval_c5(h, dx)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_cn = c
    bi, bj = -1, -1
    for i in ones:
        for j in zeros:
            h[i] = 0.0
            h[j] = 1.0
            cn = eval_c5(h, dx)
            h[i] = 1.0
            h[j] = 0.0
            if cn < best_cn - 1e-15:
                best_cn = cn
                bi, bj = i, j
    if bi >= 0:
        h[bi] = 0.0
        h[bj] = 1.0
        return best_cn, True
    return c, False

def local_search(h, dx, t_limit):
    c = eval_c5(h, dx)
    while time.time() < t_limit:
        cn, improved = ls_pass(h, dx)
        if not improved:
            break
        c = cn
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    else:
        h0 = np.zeros(70)
        h0[np.random.RandomState(seed).choice(70, 35, replace=False)] = 1.0

    n = len(h0)
    dx = 2.0 / n
    n_ones = int(round(np.sum(h0)))
    h = (h0 > 0.5).astype(np.float64).copy()

    best_h, best_c = h.copy(), eval_c5(h, dx)
    rng = np.random.RandomState(seed + 777)

    # Phase 1: aggressive local search
    ph1_end = t0 + budget_s * 0.12
    best_h, best_c = local_search(best_h, dx, ph1_end)

    # Phase 2: Simulated annealing
    T = 0.004
    T_min = 1e-7
    cool = 0.9998
    ph2_end = t0 + budget_s * 0.78

    while time.time() < ph2_end:
        # Decide single or double swap
        if rng.random() < 0.82:
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            new_h = best_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
        else:
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            idx_i = rng.choice(len(ones), 2, replace=False)
            idx_j = rng.choice(len(zeros), 2, replace=False)
            i1, i2 = ones[idx_i[0]], ones[idx_i[1]]
            j1, j2 = zeros[idx_j[0]], zeros[idx_j[1]]
            if len(set([i1, i2, j1, j2])) < 4:
                i1, i2, j1, j2 = 0, 0, 0, 0
                if len(set([i1, i2])) < 2 or len(set([j1, j2])) < 2:
                    continue
            new_h = best_h.copy()
            new_h[i1] = 0.0
            new_h[i2] = 0.0
            new_h[j1] = 1.0
            new_h[j2] = 1.0

        d = c_new - best_c

            if d < -0.002:
