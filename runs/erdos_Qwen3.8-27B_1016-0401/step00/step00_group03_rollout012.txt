        s = step_sizes[rng.randint(len(step_sizes))]
        current[i] -= s
        current[j] += s
        new_c5 = compute_c5(current)
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = current.copy()
        else:
            current[i] += s
            current[j] -= s
            # Try moving back
            s2 = step_sizes[rng.randint(len(step_sizes))] * 0.5
            current[i] += s2
            current[j] -= s2
            new_c5 = compute_c5(current)
            if new_c5 < current_c5:
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best = current.copy()
            else:
                current[i] -= s2
                current[j] += s2

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load initial solution
    if 'initial_h_values' in globals():
        try:
            h0 = np.array(initial_h_values, dtype=np.float64)
            c5_0 = compute_c5(h0)
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = len(h0)
        except Exception:
            pass

    if best_h is None:
        best_n = 70
        best_h = random_binary(70, 35, rng)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start) - 2

    # Phase 1: Greedy local search on primary (parent) solution
    if time_left() > 60:
        t_g1 = min(time_left() * 0.45, 350.0)
        h1, c5_1, cur1, c5_cur1 = greedy_search(best_h, t_g1, rng)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()
        if c5_cur1 < best_c5:
            best_c5, best_h = c5_cur1, cur1.copy()

    # Phase 2: Simulated annealing from best greedy result
    if time_left() > 40:
        start_h = best_h.copy()
        t_sa = min(time_left() * 0.25, 200.0)
        h2, c5_2, cur2, c5_cur2 = sa_search(start_h, t_sa, rng)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        if c5_cur2 < best_c5:
            best_c5, best_h = c5_cur2, cur2.copy()

    # Phase 3: Second greedy pass + SA from different angle
    if time_left() > 40:
        h3, c5_3, cur3, c5_c3 = greedy_search(best_h, min(time_left() * 0.2, 120.0), rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        if c5_c3 < best_c5:
            best_c5, best_h = c5_c3, cur3.copy()

    # Phase 4: Try other n values with random starts
    n_candidates = [n for n in [50, 60, 80, 90, 100] if n != best_n]
    for n in n_candidates:
        if time_left() < 25:
            break
        for r in range(2):
            if time_left() < 20:
                break
            h_r = random_binary(n, n // 2, rng)
            t_g = min(40.0, time_left() * 0.25)
            h_g, c5_g, cur_g, c5_cg = greedy_search(h_r, t_g, rng)
            if c5_g < best_c5:
                best_c5, best_h, best_n = c5_g, h_g, n
            if c5_cg < best_c5:
                best_c5, best_h, best_n = c5_cg, cur_g, n

    # Phase 5: Fractional polish on best binary solution
    if time_left() > 20:
        # Round best_h to binary if needed for polish
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        if abs(h_bin.sum() - len(h_bin) / 2.0) > 0.01:
            h_bin = best_h.copy()
        else:
            h_bin = best_h.copy()
        t_fp = min(time_left() * 0.3, 100.0)
        h_fp, c5_fp = try_fractional_polish(h_bin, rng, t_fp)
        if c5_fp < best_c5:
            best_c5, best_h = c5_fp, h_fp

    # Phase 6: Final greedy polish
    if time_left() > 15:
        best_h_rounded = np.where(best_h > 0.5, 1.0, np.where(best_h < 0.5, 0.0, best_h))
        h_f, c5_f, _, _ = greedy_search(best_h_rounded, min(time_left() * 0.7, 120.0), rng)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    best_n = len(best_h)
    # Verify sum constraint approximately
    target_sum = best_n / 2.0
    actual_sum = best_h.sum()
    if abs(actual_sum - target_sum) > 0.01:
        # Fix: adjust toward target
        diff = target_sum - actual_sum
        if diff > 0:
            idx = np.argsort(best_h)[::-1]
            for i in idx:
                if abs(best_h[i]) < 1e-9:
                    continue
                adj = min(diff, best_h[i])
                best_h[i] -= adj
                diff -= adj
                if diff < 1e-9:
                    break
        else:
            idx = np.argsort(best_h)
            for i in idx:
                if 1.0 - best_h[i] < 1e-9:
                    continue
                adj = min(-diff, 1.0 - best_h[i])
                best_h[i] += adj
                diff += adj
                if diff > -1e-9:
                    break
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
