            if restart > 0:
                scale = 0.04 * restart
                h_try = project_feasible(h_curr + np.random.standard_normal(n) * scale, target_sum)
            else:
                h_try = h_curr.copy()
            res = minimize(obj_func, h_try, jac=grad_func, args=(temp,),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr - 1e-12:
                h_curr = h_new
                c5_curr = c5_new
    return h_curr, c5_curr


def final_polish(h, n, max_iters):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    c5_curr = compute_c5(h)
    eps = 1e-6
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        grad = np.full(n, dx)
        for j in range(n):
            i1 = j + s
            i2 = j - s
            if 0 <= i1 < n:
                grad[j] -= dx * h[i1]
            if 0 <= i2 < n:
                grad[j] -= dx * h[i2]
        # line search
        for alpha in [0.01, 0.005, 0.002, 0.001]:
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr - 1e-12:
                h = h_new
                c5_curr = c5_new
                break
        else:
            break
    return h, c5_curr


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t0

    def update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    try:
        init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        init = np.full(45, 0.5)
    n_init = len(init)
    init = np.clip(init, 0.0, 1.0)
    init = project_feasible(init, n_init / 2.0)
    update(init, compute_c5(init), n_init)

    temps_list = [0.08, 0.04, 0.02, 0.01, 0.005]
    # Phase 1: optimize at initial resolution
    h_opt, c5_opt = optimize_h(init.copy(), n_init, temps_list, 3, budget_s, 0.15)
    update(h_opt, c5_opt, n_init)

    # Phase 2: try different starting points at initial resolution
    for sp_type in ['block', 'sin']:
        if elapsed() > budget_s * 0.35:
            break
        if sp_type == 'block':
            h_sp = make_two_block(n_int := n_init, rng)
        else:
            h_sp = make_sinusoidal(n_init, rng)
        c5_sp = compute_c5(h_sp)
        if c5_sp < best_c5:
            update(h_sp, c5_sp, n_init)
        h_opt2, c5_opt2 = optimize_h(h_sp, n_init, temps_list[1:], 2, budget_s - elapsed(), 0.1)
        update(h_opt2, c5_opt2, n_init)

    # Phase 3: optimize at higher resolutions
    for mult in [2, 4]:
        nx = n_init * mult
        if elapsed() > budget_s * 0.75:
            break
        h_interp = interp_to(best_h, nx)
        h_interp = project_feasible(h_interp, nx / 2.0)
        c5_interp = compute_c5(h_interp)
        update(h_interp, c5_interp, nx)
        n_temps = max(2, len(temps_list) - mult)
        h_hi, c5_hi = optimize_h(h_interp, nx, temps_list[:n_temps], 2, budget_s - elapsed(), 0.1)
        update(h_hi, c5_hi, nx)

    # Phase 4: final polish on best
    if best_h is not None and elapsed() < budget_s * 0.92:
        h_pol, c5_pol = final_polish(best_h, best_n,
                                     max_iters=int(min(30, (budget_s * 0.05) / max(0.01, elapsed()))) )
        update(h_pol, c5_pol, best_n)

    # Ensure constraints are met
    if best_h is not None:
        best_h = project_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
