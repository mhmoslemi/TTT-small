    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    n = len(best_h)

    # Phase 1: Steepest descent (20% budget)
    deadline1 = t0 + budget_s * 0.20
    h_work = h.copy()
    while time.time() < deadline1:
        h_work, c5 = steepest_descent_binary(h_work, max_iters=50)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_work.copy()
        else:
            break

    # Phase 2: Simulated annealing with restarts (30% budget)
    deadline2 = t0 + budget_s * 0.50
    h_sa = best_h.copy()
    sa_restart = 0
    while time.time() < deadline2 and sa_restart < 30:
        sa_restart += 1
        h_sa = best_h.copy()
        n_steps = 80000
        h_new, c5_new = simulated_annealing(h_sa, 0.03, 0.0005, n_steps, deadline2)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Polish with steepest descent
        h_new, c5_new = steepest_descent_binary(h_new, max_iters=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Multi-restart perturbation (25% budget)
    deadline3 = t0 + budget_s * 0.75
    for restart in range(200):
        if time.time() > deadline3:
            break
        h_pert = best_h.copy()
        n_swap = np.random.randint(2, 8)
        for _ in range(n_swap):
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_pert[i], h_pert[j] = 0.0, 1.0
        h_new, c5_new = steepest_descent_binary(h_pert, max_iters=30)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Small SA around perturbed solution
        if time.time() < deadline3:
            h_new, c5_new = simulated_annealing(h_new, 0.01, 0.001, 10000, deadline3)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_new2, c5_new2 = steepest_descent_binary(h_new, max_iters=10)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

    # Phase 4: Try larger grid (n*2) if beneficial (15% budget)
    deadline4 = t0 + budget_s * 0.95
    if time.time() < deadline4:
        n_new = 2 * n
        h_large = np.zeros(n_new)
        for i in range(n):
            h_large[2 * i] = best_h[i]
            h_large[2 * i + 1] = best_h[i]
        h_new, c5_new = steepest_descent_binary(h_large, max_iters=15)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if time.time() < deadline4:
            h_new, c5_new = simulated_annealing(h_new, 0.02, 0.002, 30000, deadline4)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_new, c5_new = steepest_descent_binary(h_new, max_iters=10)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Final polish with SA + descent (5% budget)
    deadline5 = t0 + budget_s * 0.99
    if time.time() < deadline5:
        h_new, c5_new = simulated_annealing(best_h, 0.005, 0.0001, 20000, deadline5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_new, c5_new = steepest_descent_binary(best_h, max_iters=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Verify constraints
    n_final = len(best_h)
    assert abs(np.sum(best_h) - n_final / 2.0) < 1e-6, "Sum constraint violated"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Range constraint violated"

    return (best_h, float(best_c5), int(n_final))
