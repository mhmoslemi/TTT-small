```python
def analytic_grad(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    # For each output j (0 to 2n-2) and each m0 (0 to n-1):
    # C(j,m0): max(0,j-n+1) <= m0 <= min(j, n-1)
    # contribution: w[j] * dx * (1 - 2*h[j-m0])
    
    for j in range(2*n - 1):
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        if w[j] < 1e-15:
            continue
        for m0 in range(lo, hi + 1):
            grad[m0] += w[j] * dx * (1 - 2 * h[j - m0])
    
    return grad
```

    m = vals.max()
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

    lo, hi = -2.0, 2.0
    for _ in range(60):

            lo = mid
