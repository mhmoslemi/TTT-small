import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def coordinate_descent_step(h, i, dx):
    n = len(h)
    # Set h[i] = 0
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode="full")
    # Set h[i] = 1
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode="full")
    
    a = corr1 - corr0  # coefficient of h[i]
    b = corr0  # constant part
    orig_hi = h[i]
    
    # Minimize max_m (a[m]*x + b[m]) over x in [0,1]
    # Find all candidate x values
    candidates = np.array([0.0, 1.0])
    # Pairwise intersections
    n_lags = len(a)
    for p in range(n_lags):
        for q in range(p+1, n_lags):
            da = a[p] - a[q]
            if abs(da) > 1e-12:
                x = (b[q] - b[p]) / da
                if 0.0 < x < 1.0:
                    candidates = np.append(candidates, x)
    
    # Evaluate at all candidates
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        vals = a * x + b
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_x = x
    
    h[i] = best_x
    return h, best_val * dx

        X = np.where(safe, db / np.where(safe, da, 1.0), -999.0)
    triu_mask = np.triu(np.ones((nl, nl), dtype=bool), 1)
    valid = triu_mask & (X > 1e-12) & (X < 1.0 - 1e-12)
    inter = X[valid]
    candidates = np.concatenate(([0.0, 1.0], inter))
    vals = a[None, :] * candidates[:, None] + b[None, :]
    maxes = np.max(vals, axis=1)
    return candidates[np.argmin(maxes)]

def _coord_descent(h, n, max_sweeps, deadline):
    target = n / 2.0
    h = h.astype(float).copy()
    _project(h, target)
    best_local = _eval_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        changed = False
        for i in range(n):
            new_val = _coord_update(h, i, n)
            if abs(new_val - h[i]) > 1e-11:
                changed = True
            h[i] = new_val
        _project(h, target)
        c5 = _eval_c5(h)
        if c5 < best_local:
            best_local = c5
        if not changed:
            break
    return h, best_local

def _slsqp_polish(h, n, deadline):
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    eps = 1e-5
    t_smooth = 0.01

    t_values = [0.05, 0.02, 0.01, 0.005, 0.002]

    for t_smooth in t_values:
        if time.time() > deadline - 5:
            break

        def objective(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            scaled = corr * dx / t_smooth
            max_s = np.max(scaled)
            if max_s > 500:
                return max_s * t_smooth
            return t_smooth * np.log(np.sum(np.exp(scaled - max_s)))

        def gradient(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            scaled = corr * dx / t_smooth
            max_s = np.max(scaled)
            w = np.exp(scaled - max_s)
            w /= np.sum(w)
            grad = np.zeros(n)
            for i in range(n):
                x[i] += eps
                cp = np.correlate(x, 1.0 - x, mode="full")
                x[i] -= 2 * eps
                cm = np.correlate(x, 1.0 - x, mode="full")
                x[i] += eps
                dc = (cp - cm) / (2 * eps)
                grad[i] = np.dot(w, dc) * dx
            return grad

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                        'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n

        x0 = best_h.copy()
        res = minimize(objective, x0, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 80, 'ftol': 1e-8})
        if res.success and np.all(res.x >= -1e-8) and np.all(res.x <= 1 + 1e-8):
            x_new = np.clip(res.x, 0, 1)
            _project(x_new, target)
            c5 = _eval_c5(x_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except (NameError, AttributeError):
        h_parent = None

    best_h, best_c5, best_n = None, np.inf, 47

    def update_best(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = _eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 1: coordinate descent from parent
    if h_parent is not None:
        n0 = len(h_parent)
        hp, _ = _coord_descent(h_parent.copy(), n0, 500, t_start + 80)
        update_best(hp, n0)

    # Phase 2: random restarts with various n
    n_options = [46, 48, 50, 52, 54, 56, 60, 64, 68]
    n_restarts_per_n = 4
    for n in n_options:
        if time.time() > deadline - 150:
            break
        target = n / 2.0
        for r in range(n_restarts_per_n):
            if time.time() > deadline - 100:
                break
            if n % 2 == 0:
                h_rand = np.zeros(n)
                idx = np.random.choice(n, int(target), replace=False)
                h_rand[idx] = 1.0
            else:
                h_rand = np.random.uniform(0.05, 0.95, n)
                _project(h_rand, target)
            hl, _ = _coord_descent(h_rand, n, 200, min(time.time() + 20, deadline - 60))
            update_best(hl, n)

    # Phase 3: SLSQP polish on best
    if best_h is not None and time.time() < deadline - 30:
        hp2, _ = _slsqp_polish(best_h.copy(), best_n, deadline - 10)
        update_best(hp2, best_n)

    # Phase 4: final coord descent polish
    if best_h is not None and time.time() < deadline - 5:
        hf, _ = _coord_descent(best_h.copy(), best_n, 3000, deadline - 2)
        update_best(hf, best_n)

    # Final evaluation
    final_c5 = _eval_c5(best_h)
    if best_h is None:
        n = 47
        best_h = np.full(n, 0.5)
        final_c5 = _eval_c5(best_h)
        best_n = n

    return best_h, final_c5, best_n
