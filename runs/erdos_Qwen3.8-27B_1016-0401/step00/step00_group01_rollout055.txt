import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        for i in range(i_lo, i_hi):
            j = i + s
            grad[i] += wk * (1.0 - h[j])
            grad[j] -= wk * h[i]
    return grad * dx

def project_sum(h, target):
    """Project h to [0,1]^n with sum(h) = target"""
    n = len(h)
    # Bisection on alpha: find alpha s.t. sum(clip(h - alpha, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pgd_step(h, target, beta, lr, n_inner):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_inner):
        g = compute_grad(h, beta)
        h_new = h - lr * g
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def local_search_binary(h, n, target, max_iter=1000):
    """Local search: try flipping/perturbing individual elements"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        improved = False
        for i in range(n):
            for delta in [0.5, -0.5, 1.0, -1.0, 0.1, -0.1]:
                h_trial = best_h.copy()
                h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
                # Fix sum
                diff = target - np.sum(h_trial)
                if abs(diff) < 1e-10:
                    h_trial = best_h.copy()
                    h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
                else:
                    # Distribute difference
                    others = [j for j in range(n) if j != i]
                    h_trial = best_h.copy()
                    h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
                    # Adjust others proportionally
                    remaining = target - h_trial[i]
                    h_others = best_h[others].copy()
                    h_others = np.clip(h_others * (remaining / max(np.sum(h_others), 1e-10)), 0, 1)
                    h_trial[others] = h_others
                    if abs(np.sum(h_trial) - target) > 1e-6:
                        continue
                
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-10:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5
