    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-16:
            break
        m = (h > 1e-16) & (h < 1.0 - 1e-16)
        c = int(m.sum())
        if c > 0:
            h[m] += d / c
        else:
            idx = np.argmin(h) if d > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + d, 0, 1)
        h = np.clip(h, 0.0, 1.0)

    if abs(d) > 1e-15:

                break
            if d > 0 and h[i] < 1.0:

                h[i] += a; d -= a
            elif d < 0 and h[i] > 0.0:
                a = min(h[i], -d)
                h[i] -= a; d += a
    return h

    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    g = np.zeros(n)
    g[lo:hi + 1] += dx * (1.0 - h[lo + s:hi + 1 + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= dx * h[lo2 - s:hi2 + 1 - s]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def c5_vec(h):

def project(h, n):

    tgt = n / 2.0

        if abs(d) < 1e-16:

            h[m] += d / c
        else:

            h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))
        h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-15:
        for i in np.random.permutation(n):
            if abs(d) < 1e-15:
                break
            if d > 0 and h[i] < 1.0:
                a = min(1.0 - h[i], d)
                h[i] += a
                d -= a
            elif d < 0 and h[i] > 0.0:
                a = min(h[i], -d)
                h[i] -= a
                d += a
    return h

def grad_fs(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi + 1] += dx * (1.0 - h[lo + s:hi + 1 + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= dx * h[lo2 - s:hi2 + 1 - s]
    return g

def local_search(h, n, tl, rng):
    t0 = time.time()
    cur = h.copy()
    bh = h.copy()
    bc = compute_c5(h)
    T = 3e-4
    it = 0
    while time.time() - t0 < tl:
        it += 1
        scale = 0.025 * max(0.2, T / 3e-4)
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = float(rng.uniform(-scale, scale))
        hn = cur.copy()
        hn[i] += d
        hn[j] -= d
        if hn[i] < 0.0 or hn[i] > 1.0 or hn[j] < 0.0 or hn[j] > 1.0:
            hn = project(hn, n)
        cn = compute_c5(hn)
        if cn < bc:
            bc = cn
            bh = hn.copy()
            cur = hn
        elif cn - bc < 0.05:
            if rng.random() < np.exp(-(cn - bc) / T):
                cur = hn
        if it % 60 == 0:
            vals = c5_vec(cur)
            s_star = int(np.argmax(vals))
            shift = s_star - (n - 1)
            g = grad_fs(cur, shift, n)
            gnorm = np.linalg.norm(g)
            if gnorm > 1e-12:
                hn2 = cur - 0.004 * g
                hn2 = project(hn2, n)
                cn2 = compute_c5(hn2)
                if cn2 < bc:
                    bc = cn2
                    bh = hn2.copy()
                cur = bh.copy()
                T = 3e-4
        if it % 2000 == 0:
            cur = project(bh + 0.02 * rng.standard_normal(n), n)
            T = max(T * 0.9, 1e-5)
        T = max(T * 0.999997, 3e-7)
    return bh, bc

    xo = np.linspace(0.0, 2.0, no)
    xn = np.linspace(0.0, 2.0, nn)
    return project(np.interp(xn, xo, h), nn)
