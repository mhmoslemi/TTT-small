import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def fixed_constraint(h, n):
    """Fix h to satisfy sum constraint and bounds"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        else:
            h += diff / n
