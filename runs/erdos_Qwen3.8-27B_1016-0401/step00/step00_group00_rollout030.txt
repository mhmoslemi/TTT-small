import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, T):
    dx = 2.0 / n
    nh = 1.0 - h
    corr = np.correlate(h, nh, mode="full")
    vals = T * corr * dx
    v_max = vals.max()
    e = np.exp(vals - v_max)
    w = e / e.sum()
    obj = (v_max + np.log(e.sum())) / T
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            grad[delta:] += wk * nh[:n - delta]
            grad[:n - delta] -= wk * h[delta:]
        else:
            d = -delta
            grad[:n - d] += wk * nh[d:]
            grad[d:] -= wk * h[:n - d]
    return obj, grad

def optimize_smooth(h0, n, max_time=30.0, T_start=30.0, T_end=500.0):
    t0 = time.time()
    dx = 2.0 / n
    h = h0.astype(np.float64).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.03
    v = np.zeros(n)
    mu = 0.75
    for it in range(100000):
        if time.time() - t0 > max_time:
            break
        frac = min(1.0, it / 3000.0)
        T = T_start + (T_end - T_start) * frac
        obj, g = smooth_obj_grad(h, n, T)
        c5_true = compute_c5(h, dx)
        if c5_true < best_c5:
            best_c5 = c5_true
            best_h = h.copy()
        v = mu * v - lr * g
        h_new = project(h + v, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5_true + 0.004:
            h = h_new
        else:
            v *= 0.4
            h = project(h - 0.3 * lr * g, n)
        lr = max(0.001, 0.03 * (0.9995 ** it))
    c5_end = compute_c5(h, dx)
    if c5_end < best_c5:
        best_h = h.copy()
        best_c5 = c5_end
    return best_h, best_c5

def coordinate_polish(h0, n, dx, max_time=20.0):
    t0 = time.time()
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    steps = [-0.12, -0.06, -0.02, 0.02, 0.06, 0.12]
    for sweep in range(200):
        if time.time() - t0 > max_time:
            break
        improved = False
        for i in range(n):
            for s in steps:
                h_try = h.copy()
                h_try[i] += s
                h_try = project(h_try, n)
                c = compute_c5(h_try, dx)
                if c < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5
