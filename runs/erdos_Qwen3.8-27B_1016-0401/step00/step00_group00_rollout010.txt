import numpy as np
from scipy.optimize import minimize
import time

_g_n = 45
_g_tau = 50.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    s = float(np.sum(h))
    if abs(s - target) < 1e-13:
        return h
    sign = 1.0 if s > target else -1.0
    lo, hi = 0.0, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - sign * mid, 0.0, 1.0)
        if float(np.sum(x)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - sign * (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] = float(np.clip(x[idx] + diff, 0.0, 1.0))
    return x


def obj_func(h):
    n = _g_n
    dx = 2.0 / n
    tau = _g_tau
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    if m <= 1e-15:
