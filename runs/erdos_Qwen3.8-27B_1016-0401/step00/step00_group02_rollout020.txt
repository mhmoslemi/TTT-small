    return float(np.max(corr) * dx)

    d = target - h.sum()
    if abs(d) < 1e-12:
        return h
    # Distribute the correction
    if d > 0:
        idx = np.argsort(h)[:max(1, abs(d))]  # elements with room to increase
    else:
        idx = np.argsort(-h)[:max(1, abs(d))]  # elements with room to decrease
    h[idx] += d / len(idx)
    return np.clip(h, 0, 1)

    n = len(h0)
    target = n / 2.0
    h0 = fix_sum(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing tau
    for tau in [8.0, 15.0, 30.0, 60.0, 120.0]:
        if elapsed() > budget_s * 0.45:
            break
        def f_lbfgs(x):
            x = np.clip(x, 0.0, 1.0)
            return smooth_c5(x, target, n, tau)
        try:
            res = minimize(f_lbfgs, best_h, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = fix_sum(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Nelder-Mead polish
    if elapsed() < budget_s * 0.55:
        def f_nm(x):
            x = np.clip(x, 0.0, 1.0)
            return compute_c5(x) + 200.0 * abs(x.sum() - target)
        try:
            res = minimize(f_nm, best_h, method='Nelder-Mead',
                          options={'maxiter': 30000, 'xatol': 1e-11,
                                   'fatol': 1e-13, 'adaptive': True})
            h_new = fix_sum(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 3: Local search with pair perturbations
    if elapsed() < budget_s * 0.75:
        h = best_h.copy()
        lr = 0.03
        stalled = 0
        for it in range(50000):
            if elapsed() > budget_s * 0.75:
                break
            i, j = int(np.random.randint(n)), int(np.random.randint(n))
            if i == j:
                continue
            delta = np.random.uniform(-lr, lr)
            hi, hj = h[i], h[j]
            ni, nj = hi + delta, hj - delta
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                lr = min(lr + 0.001, 0.08)
                stalled = 0
            else:
                h[i], h[j] = hi, hj
                stalled += 1
                if stalled > 200:
                    lr = max(lr * 0.95, 0.005)
                    stalled = 0
                    if lr <= 0.005:
                        break

    # Phase 4: Binarization attempt
    if elapsed() < budget_s * 0.82:
        h_b = (best_h > 0.5).astype(float)
        # Fix count: need exactly target sum
        curr_sum = h_b.sum()
        if curr_sum < target:
            # Need more: set some 0s to 1
            zeros = np.where(h_b == 0)[0]
            need = int(round(target - curr_sum))
            if need > 0 and len(zeros) >= need:
                h_b[zeros[:need]] = 1.0
        elif curr_sum > target:
            ones = np.where(h_b == 1)[0]
            need = int(round(curr_sum - target))
            if need > 0 and len(ones) >= need:
                h_b[ones[:need]] = 0.0
        # If still not exact, adjust one value
        d = target - h_b.sum()
        if abs(d) > 1e-10:
            h_b[0] += d
            h_b = np.clip(h_b, 0, 1)
        h_b = fix_sum(h_b, n)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_h, best_c5 = h_b.copy(), c5_b

    # Phase 5: Hill-climb the best solution with smaller steps
    if elapsed() < budget_s * 0.92:
        h = best_h.copy()
        for it in range(30000):
            if elapsed() > budget_s * 0.92:
                break
            idx = np.random.randint(n)
            # Try adjusting one element and compensating with another
            jdx = (idx + 1 + np.random.randint(n - 1)) % n
            delta = np.random.uniform(-0.02, 0.02)
            hi, hj = h[idx], h[jdx]
            ni, nj = hi + delta, hj - delta
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[idx], h[jdx] = ni, nj
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[idx], h[jdx] = hi, hj

    # Final fix
    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
