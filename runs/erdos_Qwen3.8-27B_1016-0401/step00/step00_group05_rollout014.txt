    return obj

def soft_max_grad(hv, temp, rho):
    n = len(hv)
    dx = 2.0 / n
    hm = 1.0 - hv
    corr = np.correlate(hv, hm, mode="full")
    m = np.max(corr)
    w = np.exp(temp * (corr - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        # dc[k]/dh[j]: first term (1-h[j-offset]) where j-offset in [0,n-1]
        js = max(0, offset)
        je = min(n, n + offset)
        if js < je:
            grad[js:je] += wk * (1.0 - hv[js - offset:je - offset])
        # second term: -h[j+offset] where j+offset in [0,n-1]
        js2 = max(0, -offset)
        je2 = min(n, n - offset)
        if js2 < je2:
            grad[js2:je2] -= wk * hv[js2 + offset:je2 + offset]
    
    grad *= dx
    # Add constraint gradient
    s = np.sum(hv) - n / 2.0
    grad += 2.0 * rho * s
    return grad

def run(seed=42, budget_s=1100, **kwargs):

np.correlate(h, v, mode="full") where v = 1-h:
The result c has length 2n-1.
c[k] = sum_{t} h[t] * v[t - (k - (n-1))] = sum_{t} h[t] * (1 - h[t - offset])
where offset = k - (n-1) and the sum is over valid t (where 0 ≤ t < n and 0 ≤ t - offset < n).

So: c[k] = sum_{t=max(0,offset)}^{min(n-1, n-1+offset)} h[t] * (1 - h[t - offset])

dc[k]/dh[j]:
- If j is the "t" index: contributes (1 - h[j - offset]) if j - offset is valid
- If j - offset is the "t - offset" index, i.e., j = t + offset, t = j - offset: contributes -h[j - offset] if j - offset is valid... wait no.

Let me be more careful. c[k] = sum_t h[t] * (1 - h[t - offset])

dc[k]/dh[j] = d/dh[j] [sum_t h[t] * (1 - h[t - offset])]

The variable h[j] appears in two places:
1. As h[t] when t = j: contributes (1 - h[j - offset]) [if j - offset is in [0, n-1]]
2. As h[t - offset] when t - offset = j, i.e., t = j + offset: contributes -h[j + offset] [if j + offset is in [0, n-1]]

So: dc[k]/dh[j] = (1 - h[j - offset]) * [0 ≤ j - offset < n] - h[j + offset] * [0 ≤ j + offset < n]

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    correl = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(correl) * dx)

def project_feasible(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        interior = (proj > 1e-14) & (proj < 1.0 - 1e-14)
        ni = int(np.sum(interior))
        if ni > 0:
            proj[interior] += diff / ni
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _sm_obj(hv, temp, rho):
    n = len(hv)
    corr = np.correlate(hv, 1.0 - hv, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(temp * (corr - m)))) / temp
    obj = sm * (2.0 / n)
    s = np.sum(hv) - n * 0.5
    return obj + rho * s * s

def _sm_grad(hv, temp, rho):
    n = len(hv)
    dx = 2.0 / n
    hm = 1.0 - hv
    corr = np.correlate(hv, hm, mode="full")
    m = np.max(corr)
    w = np.exp(temp * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-12:
            continue
        offset = k - (n - 1)
        js = max(0, offset)
        je = min(n, n + offset)
        if js < je:
            grad[js:je] += wk * (1.0 - hv[js - offset:je - offset])
        js2 = max(0, -offset)
        je2 = min(n, n - offset)
        if js2 < je2:
            grad[js2:je2] -= wk * hv[js2 + offset:je2 + offset]
    grad *= dx
    s = np.sum(hv) - n * 0.5
    grad += 2.0 * rho * s
    return grad

def _upsample(h_src, n_src, n_tgt):
    if n_tgt == n_src:
        return h_src.copy()
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    return np.clip(np.interp(x_new, x_old, h_src, period=2.0), 0.0, 1.0)

def _lbbf_pass(h, n, target_sum, t0, budget_s, temp=40.0, rho=50.0):
    def obj(hv):
        return _sm_obj(hv, temp, rho)
    def jac(hv):
        return _sm_grad(hv, temp, rho)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 150, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxls': 30})
        h_out = np.clip(res.x, 0.0, 1.0)
        h_out = project_feasible(h_out, target_sum)
        return h_out
    except Exception:
        return h

def _pg_polish(h, n, target_sum, t0, budget_s):
    step = 0.015
    temp = 60.0
    c5_old = evaluate_c5(h)
    for it in range(800):
        if time.time() - t0 > budget_s - 3:
            break
        g = _sm_grad(h, temp, 0.0)
        h_new = h - step * g
        h_new = project_feasible(h_new, target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_old - 1e-12:
            h = h_new
            c5_old = c5_new
            step = min(step * 1.06, 0.08)
        else:
            step *= 0.6
            if step < 1e-9:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    sizes = sorted(set([n_init, 90, 120, 160, 200, 280]))

    for n_target in sizes:
        if time.time() - t0 > budget_s - 40:
            break
        target_sum = float(n_target) / 2.0

        for restart in range(3):
            if time.time() - t0 > budget_s - 10:
                break
            if n_target == n_init and restart == 0:
                h = h_init.copy()
            elif restart == 0:
                h = _upsample(h_init, n_init, n_target)
            else:
                h = _upsample(h_init, n_init, n_target)
                h += rng.normal(0, 0.03 * (restart), n_target)
                h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)

            # L-BFGS pass
            h = _lbbf_pass(h, n_target, target_sum, t0, budget_s - 5)
            c5_cur = evaluate_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()

            # PG polish
            if time.time() - t0 < budget_s - 8:
                h = _pg_polish(h, n_target, target_sum, t0, budget_s - 3)
                c5_cur = evaluate_c5(h)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, float(n_final) / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_final
