import numpy as np
import time


def c5_val(hv, dx):
    return float(np.max(np.correlate(hv, 1.0 - hv, mode='full')) * dx)


def hill_climb_best(hv, dx, best_c5, best_h, tl):
    while time.time() < tl:
        improved = False
        ones = np.where(hv == 1.0)[0]
        zeros = np.where(hv == 0.0)[0]
        best_local = best_c5
        bi, bj = -1, -1
        for i in ones:
            for j in zeros:
                hv[i], hv[j] = hv[j], hv[i]
                c5 = c5_val(hv, dx)
                hv[i], hv[j] = hv[j], hv[i]
                if c5 < best_local:
                    best_local = c5
                    bi, bj = i, j
        if bi >= 0:
            hv[bi], hv[bj] = hv[bj], hv[bi]
            best_c5 = best_local
            best_h = hv.copy()
            improved = True
        if not improved:
            break
    return best_c5, best_h

    while time.time() < deadline and T > 0.0003:

        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:

        h[i], h[j] = h[j], h[i]
