    lo, hi = -2.0, 2.0

        m = (lo + hi) * 0.5
        s = np.sum(np.clip(x - m, 0.0, 1.0))
        if s > ts:
            lo = m
        else:
            hi = m
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def score(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def optimize(h0, n, dx, t0, t_end, seed):
    rng = np.random.RandomState(seed)
    ts = n / 2.0
    h = h0.copy()
    bh = h.copy()
    bc = score(h, dx)
    alpha = 5.0
    lr = 0.1
    stall = 0
    best_lr = lr

    for it in range(500000):
        if time.time() > t_end:
            break
        if stall > 300:
            break
        if alpha > 500:
            alpha = 500

        corr = np.correlate(h, 1.0 - h, mode="full")
        sc = corr * dx
        mx = np.max(sc)
        ex = np.exp(alpha * (sc - mx))
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(sc)
            w[np.argmax(sc)] = 1.0
        else:
            w = ex / s_ex

        g = wcorr_grad(h, w, n) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        improved = False
        for ls in range(12):
            step = lr / (2.0 ** ls)
            nh = proj(h - step * g, ts)
            nc = score(nh, dx)
            if nc < bc - 1e-13:
                h = nh
                bh = nh.copy()
                bc = nc
                stall = 0
                improved = True
                if step > 0.01:
                    lr = min(lr * 1.1, 0.5)
                break
        if not improved:
            stall += 1
            lr *= 0.5
            if lr < 1e-7:
                h = bh + rng.randn(n) * 0.03
                h = np.clip(h, 0.0, 1.0)
                h = proj(h, ts)
                lr = max(0.05, best_lr * 0.5)
                alpha = max(5.0, alpha * 0.5)
                stall = 0
                if it > 20000:
                    break
        alpha += 0.01

    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 5

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    h0 = proj(np.clip(h0, 0.0, 1.0), ts)

    bh = h0.copy()
    bc = score(h0, dx)

    # Phase 1: Main optimization at initial n
    t_phase1 = t0 + budget_s * 0.35
    bh, bc = optimize(h0, n, dx, t0, t_phase1, seed)

    # Phase 2: Restart perturbations
    for r in range(4):
        if time.time() > t0 + budget_s * 0.60:
            break
        rng = np.random.RandomState(seed + 11 + r)
        hr = bh + rng.randn(n) * (0.03 + 0.01 * r)
        hr = proj(np.clip(hr, 0.0, 1.0), ts)
        hr, rc = optimize(hr, n, dx, time.time(), t0 + budget_s * 0.75, seed + 200 + r)
        if rc < bc:
            bc = rc
            bh = hr

    # Phase 3: Try other n values
    for tn in [100, 60, 140]:
        if time.time() > t0 + budget_s * 0.90:
            break
        tdx = 2.0 / tn
        tts = tn / 2.0
        hi = np.interp(np.linspace(0, 2, tn, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), bh)
        hi = proj(np.clip(hi, 0.0, 1.0), tts)
        hi, rc = optimize(hi, tn, tdx, time.time(), t0 + budget_s * 0.95, seed + 500)
        if rc < bc:
            bc = rc
            bh = hi
            n = tn
            dx = tdx
            ts = tts

    # Phase 4: Final polish
    if time.time() < t_end - 10:
        bh, bc = optimize(bh, n, dx, time.time(), t_end, seed + 999)

    # Ensure feasibility
    bh = np.clip(bh, 0.0, 1.0)
    bh = proj(bh, len(bh) / 2.0)
    n = len(bh)
    bc = score(bh, 2.0 / n)

    return bh, bc, n
