def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    k = abs(k_idx - (n - 1))
    m = k  # magnitude of shift
    grad = np.zeros(n)
    if m > 0:
        grad[m:] += dx * (1.0 - h[:n-m])
        grad[:n-m] -= dx * h[m:]
    elif m == 0:
        # R_0 = dx * sum h[i](1-h[i]) = dx * (sum h - sum h^2)
        grad[:] = dx * (1.0 - 2.0 * h)
    return grad

        diff = target - s
        if abs(diff) < 1e-14:

import numpy as np
import time


def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)


def _shifts(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _grad(h, m):
    n = len(h)
    dx = 2.0 / n
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    g = np.zeros(n)
    g[m:] += dx * (1.0 - h[:n - m])
    g[:n - m] -= dx * h[m:]
    return g


def _proj(h, n):
    t = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mk = h < 1.0
            c = int(mk.sum())
            if c == 0:
                break
            h[mk] = np.minimum(h[mk] + d / c, 1.0)
        else:
            mk = h > 0.0
            c = int(mk.sum())
            if c == 0:
                break
            h[mk] = np.maximum(h[mk] + d / c, 0.0)
    return h


def _rem(t0, budget):
    return budget - (time.time() - t0)


def _subgrad(h, n, iters, t0, budget, rng):
    h = _proj(h.copy(), n)
    best = h.copy()
    bc = _c5(h)
    lr = 0.05
    for i in range(iters):
        if _rem(t0, budget) < 25:
            break
        s = _shifts(h)
        k = int(np.argmax(s))
        m = abs(k - (n - 1))
        if s[k] < bc:
            bc = float(s[k])
            best = h.copy()
        g = _grad(h, m)
        h2 = _proj(h - lr * g, n)
        c2 = _c5(h2)
        if c2 < float(s[k]):
            h = h2
            lr = min(lr * 1.05, 0.12)
        else:
            lr *= 0.92
            if lr < 1e-5:
                lr = 0.05
                h = _proj(h + rng.randn(n) * 0.04, n)
    return best, bc


def _local(h, n, iters, t0, budget, rng):
    h = _proj(h.copy(), n)
    best = h.copy()
    bc = _c5(h)
    cc = bc
    temp = 0.004
    for i in range(iters):
        if _rem(t0, budget) < 25:
            break
        h_save = h.copy()
        j = rng.randint(n)
        h[j] = np.clip(h[j] + rng.uniform(-0.12, 0.12), 0.0, 1.0)
        h = _proj(h, n)
        nc = _c5(h)
        if nc < cc or (nc - cc) < temp * rng.standard_normal():
            cc = nc
            if nc < bc:
                bc = nc
                best = h.copy()
        else:
            h = h_save
        if i > 0 and i % 12000 == 0:
            temp *= 0.91
    return best, bc


def _interp(h_src, n_dst):
    n_src = len(h_src)
    xo = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    xn = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    return np.interp(xn, xo, h_src, period=2.0)
