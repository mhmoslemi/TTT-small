import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, n_points):
    """Project h to satisfy sum(h) = n_points/2, 0 <= h <= 1"""
    target = n_points / 2.0
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n_points
        h = np.clip(h, 0, 1)
        # If still not satisfied, do iterative projection
        for _ in range(100):
            diff = target - np.sum(h)
            if abs(diff) < 1e-15:
                break
            mask = (h > 0) & (h < 1)
            if np.sum(mask) == 0:
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def upsample(h, n_new):
    """Upsample from n_old points to n_new points"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # cell centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    # Linear interpolation
    h_new = np.interp(x_new, x_old, h)
    return project_to_constraint(h_new, n_new)

import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    """Project to 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    # Cell centers
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)

def _grad_active(h, n):
    """Subgradient: gradient of the active (max) constraint."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ak = np.argmax(corr)
    lag = ak - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # c = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx])
        g[idx] -= h[idx + lag]
    else:
        s = -lag
        idx = np.arange(n - s)
        g[idx] += (1.0 - h[idx + s])
        g[idx + s] -= h[idx]
    return g * dx, corr * dx

def _subgrad_step(h, n, lr):
    g, _ = _grad_active(h, n)
    h_new = h - lr * g
    return _project(h_new, n)

def _binary_local_search(h, n, rng, max_iters, T0=0.01, T1=0.001):
    """Simulated annealing on binary sequences."""
    h_b = (np.round(h) > 0.5).astype(float)
    # Repair
    target = n // 2
    if abs(int(h_b.sum()) - target) > 0:
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if h_b.sum() > target:
            idx = rng.choice(ones, int(h_b.sum() - target), replace=False)
            h_b[idx] = 0
        else:
            idx = rng.choice(zeros, int(target - h_b.sum()), replace=False)
            h_b[idx] = 1
    cur_c5 = _eval_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    T = T0
    decay = (T1 / T0) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        h_b[i] = 1.0
        h_b[j] = 0.0
        new_c5 = _eval_c5(h_b)
        T *= decay
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_b.copy()
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
        else:
            h_b[i] = 0.0
            h_b[j] = 1.0
    return best_h, best_c5

import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return _project(np.interp(x_new, x_old, h), n_new)

def _subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ak = int(np.argmax(corr))
    lag = ak - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx])
        g[idx] -= h[idx + lag]
    else:
        s = -lag
        idx = np.arange(n - s)
        g[idx] += (1.0 - h[idx + s])
        g[idx + s] -= h[idx]
    return g * dx

def _subgrad_descend(h, n, lr0, niter, deadline):
    h = h.copy()
    lr = lr0
    for it in range(niter):
        if time.time() > deadline:
            break
        g = _subgrad(h, n)
        h = _project(h - lr * g, n)
        lr *= 0.999
        if lr < 0.003:
            lr = 0.003
    return h

def _make_binary(h, n, rng):

    target = n // 2
    s = int(hb.sum())
    if s > target:
        ones = np.where(hb == 1)[0]
        hb[rng.choice(ones, s - target, replace=False)] = 0.0
    elif s < target:
        zeros = np.where(hb == 0)[0]
        hb[rng.choice(zeros, target - s, replace=False)] = 1.0
    if n % 2 == 1:
        # Need sum = n/2 which is x.5, so set one value to 0.5
        # Actually target = n/2 = 22.5 for n=45
        # We need to adjust: set one random position to 0.5
        if int(hb.sum()) != int(target):
            if hb.sum() < target:
                zeros = np.where(hb == 0)[0]
                if len(zeros) > 0:
                    hb[rng.integers(len(zeros))] = 0.5
            elif hb.sum() > target:
                ones = np.where(hb == 1)[0]
                if len(ones) > 0:
                    hb[rng.integers(len(ones))] = 0.5
    return hb

def _binary_sa(hb, n, rng, max_iters, deadline):
    cur_c5 = _eval_c5(hb)
    best_h = hb.copy()
    best_c5 = cur_c5
    T = cur_c5 * 0.05
    T_min = cur_c5 * 0.002
    decay = (T_min / T) ** (1.0 / max(max_iters, 1))
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = int(zeros[rng.integers(len(zeros))])
        j = int(ones[rng.integers(len(ones))])
        hb[i], hb[j] = 1.0, 0.0
        new_c5 = _eval_c5(hb)
        T = max(T * decay, T_min)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = hb.copy()
        else:
            hb[i], hb[j] = 0.0, 1.0
    return best_h, best_c5

def _perturb(h, n, rng, frac):
    h2 = h.copy()
    n_pert = max(2, int(n * frac))
    idx = rng.choice(n, n_pert, replace=False)
    h2[idx] = np.clip(h2[idx] + rng.normal(0, 0.2, n_pert), 0, 1)
    return _project(h2, n)
