    current_sum = np.sum(h)

    # If sum too high, reduce elements most close to 1
    # If sum too low, increase elements most close to 0
    diff = target_sum - current_sum

        order = np.argsort(h)

            room = 1.0 - h[i]

            diff -= add

    dx = 2.0 / n

    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_cube_sum(vec, target_sum):
    lo = float(vec.min()) - 1.0
    hi = float(vec.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(vec - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.clip(vec - theta, 0.0, 1.0)

def smooth_obj(h, alpha, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    val = m + np.log(np.sum(e)) / alpha
    val += 50.0 * (np.sum(h) - target_sum) ** 2
    return val

def smooth_grad(h, alpha, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    grad *= dx
    grad += 100.0 * (np.sum(h) - target_sum)
    return grad

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def binary_local_search(h, n_iters=2000):
    n = len(h)
    target = n // 2
    best = (h > 0.5).astype(float)
    # Ensure exactly target ones
    n1 = int(best.sum())
    if n1 > target:
        inds = np.where(best == 1)[0]
        np.random.shuffle(inds)
        best[inds[:n1-target]] = 0
    elif n1 < target:
        inds = np.where(best == 0)[0]
        np.random.shuffle(inds)
        best[inds[:target-n1]] = 1
    best_val = evaluate_c5(best)
    for _ in range(n_iters):
        i = np.random.randint(n)
        trial = best.copy()
        if trial[i] > 0.5:
            if best.sum() - 1 < target:
                continue
            trial[i] = 0
        else:
            if best.sum() + 1 > target:
                continue
            trial[i] = 1
        val = evaluate_c5(trial)
        if val < best_val:
            best = trial
            best_val = val
    return best, best_val
