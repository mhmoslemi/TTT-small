For the pair move (j, m) with h[j]+=δ, h[m]-=δ:
The change in the max is approximately:
δ * (max_k ∂A_k/∂h[j] - max_k ∂A_k/∂h[m]) for the active k's

If max_k ∂A_k/∂h[j] < max_k ∂A_k/∂h[m], then increasing h[j] and decreasing h[m] helps.

This gives a heuristic for which pairs to try first.

def _c5_val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _grad_one(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        v = 0.0
        jp = j + s
        if 0 <= jp < n:
            v += 1.0 - h[jp]
        jm = j - s
        if 0 <= jm < n:
            v -= h[jm]
        g[j] = v
    return g

def _fix_sum(h, target):

    rem = target - h.sum()
    if abs(rem) > 1e-12:

        if len(idx) > 0:
            # Distribute evenly
            per = rem / len(idx)
            h[idx] += per
            h = np.clip(h, 0.0, 1.0)
            # Handle any remaining
            rem2 = target - h.sum()
            if abs(rem2) > 1e-12 and len(idx) > 0:
                h[idx[0]] += rem2
    return h

    # For each j, compute (1-h[j+s]) if valid, minus h[j-s] if valid

        g[lo:hi] += 1.0 - h[lo + s:hi + s]
    # j-s valid: 0<=j-s<n => s<=j<n+s
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - s:hi2 - s]
    return g

def _optimal_delta(h, A, im, jm, n):
    """Find optimal δ for h[jm] -= δ, h[im] += δ"""
    max_d = min(h[jm], 1.0 - h[im])
    if max_d < 1e-10:
        return 0.0
    
    # Compute D_k = effect of the move on each A_k
    # D_k = grad_k[im] - grad_k[jm] (since h[im]+=δ, h[jm]-=δ)
    # Effect on A_k: δ * (∂A_k/∂h[im] - ∂A_k/∂h[jm])
    
    # Compute gradients
    D = np.zeros(len(A))
    nA = len(A)
    for k in range(nA):
        s = k - (n - 1)
        d_im = 0.0  # ∂A_k/∂h[im]
        if 0 <= im + s < n:
            d_im += 1.0 - h[im + s]
        if 0 <= im - s < n:
            d_im -= h[im - s]
        
        d_jm = 0.0  # ∂A_k/∂h[jm]
        if 0 <= jm + s < n:
            d_jm += 1.0 - h[jm + s]
        if 0 <= jm - s < n:
            d_jm -= h[jm - s]
        
        D[k] = d_im - d_jm
    
    # A + δ*D, find δ that minimizes max
    # Candidate δ values:
    # 0, max_d
    # (A[k2] - A[k1]) / (D[k1] - D[k2]) for pairs where D[k1] != D[k2]
    
    candidates = [0.0, max_d]
    # Only consider pairs that could be optimal
    for k1 in range(nA):
        for k2 in range(k1+1, nA):
            dD = D[k1] - D[k2]
            if abs(dD) > 1e-12:
                d = (A[k2] - A[k1]) / dD
                if 0 < d < max_d:
                    candidates.append(d)
    
    # Evaluate at candidates
    best_d, best_v = 0.0, A.max()
    for d in candidates:
        d = max(0.0, min(d, max_d))
        # Quick evaluation
        new_max = (A + d * D).max()
        if new_max < best_v:
            best_v = new_max
            best_d = d
    
    return best_d
