            # After change: h[idx] = t, h[n-1] = target_sum - t - sum(h[0..n-2] except idx)
            sum_except_idx = h[:n-1].sum() - old_t  # sum of h[0..n-2] excluding idx
            # h[n-1] = target_sum - sum_except_idx - t
            # 0 <= h[n-1] <= 1 → target_sum-1-sum_except_idx <= t <= target_sum-sum_except_idx
            t_lo = max(0.0, target_sum - 1.0 - sum_except_idx)
            t_hi = min(1.0, target_sum - sum_except_idx)
            
            if t_lo >= t_hi - 1e-12:
                continue
            
            # Build grid
            ng = grid_size
            if t_hi - t_lo < 0.05:
                ng = max(ng, 20)
            grid = np.linspace(t_lo, t_hi, ng)
            
            best_t = old_t
            best_val = c5_score(h, dx)  # current value
            
            for t in grid:
                h[idx] = t
                h[n-1] = target_sum - h[:n-1].sum()
                if h[n-1] < 0 or h[n-1] > 1:
                    continue
                val = c5_score(h, dx)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            h[idx] = best_t
            h[n-1] = target_sum - h[:n-1].sum()
            
            if best_val < best_c5 - 1e-10:
                best_c5 = best_val
                best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
import time

def c5_score(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, dx, target_sum, time_budget, rng, grid_size=40):
    n = len(h)
    h = h.copy()
    h[:n-1] = np.clip(h[:n-1], 0.0, 1.0)
    h[n-1] = target_sum - h[:n-1].sum()
    if h[n-1] < -1e-6 or h[n-1] > 1.0 + 1e-6:
        h = fix_sum(h, target_sum)
    start = time.time()
    best_c5 = c5_score(h, dx)
    best_h = h.copy()
    sweep = 0
    while time.time() - start < time_budget:
        sweep += 1
        improved = False
        order = rng.permutation(n - 1)
        for idx in order:
            if time.time() - start > time_budget:
                break
            old_t = h[idx]
            sum_except = h[:n-1].sum() - old_t
            t_lo = max(0.0, target_sum - 1.0 - sum_except)
            t_hi = min(1.0, target_sum - sum_except)
            if t_lo >= t_hi - 1e-12:
                continue
            ng = min(grid_size, max(10, int(40 / max(t_hi - t_lo, 0.01))))
            grid = np.linspace(t_lo, t_hi, ng)
            best_t = old_t
            best_val = c5_score(h, dx)
            for t in grid:
                h[idx] = t
                h[n-1] = target_sum - h[:n-1].sum()
                if h[n-1] < -1e-9 or h[n-1] > 1.0 + 1e-9:

                best_h = h.copy()
                improved = True
        if not improved and sweep > 2:
            break
    return best_h, best_c5
