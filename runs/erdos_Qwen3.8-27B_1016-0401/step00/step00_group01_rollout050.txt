import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr)) * dx

def soft_max_val(h, dx, beta):
    g = 1.0 - h
    vals = np.correlate(h, g, mode='full') * dx
    m = vals.max()
    s = beta * (vals - m)
    if s.max() > 500:
        return m
    return m + np.log(np.exp(s).sum()) / beta

def _obj_pen(x):
    return soft_max_val(x, _OPT['dx'], _OPT['beta']) + 200.0 * (np.sum(x) - _OPT['target']) ** 2

def _obj_slsqp(x):
    return soft_max_val(x, _OPT['dx'], 1000.0)

def _con_eq(x):
    return np.sum(x) - _OPT['target']

def proj(h, target):
    n = len(h)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        free = (p > 1e-9) & (p < 1.0 - 1e-9)
        if free.sum() > 0:
            p[free] += diff / free.sum()
        else:
            p += diff / n
        p = np.clip(p, 0.0, 1.0)
    return p

def local_search(h, dx, rng, n_iter):
    n = len(h)
    best = compute_c5(h, dx)
    hc = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.04, 0.04)
        ht = hc.copy()
        ht[i] -= delta
        ht[j] += delta
        if 0.0 <= ht[i] <= 1.0 and 0.0 <= ht[j] <= 1.0:
            c5 = compute_c5(ht, dx)
            if c5 < best:
                hc = ht
                best = c5
    return hc, best

def local_search_multi(h, dx, rng, n_iter):
    n = len(h)
    best = compute_c5(h, dx)
    hc = h.copy()
    for _ in range(n_iter):
        mode = rng.randint(3)
        if mode == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(-0.03, 0.03)
            ht = hc.copy()
            ht[i] -= delta
            ht[j] += delta
        elif mode == 1:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            k = rng.randint(n)
            if k == i or k == j:
                continue
            d1 = rng.uniform(-0.02, 0.02)
            d2 = rng.uniform(-0.02, 0.02)
            ht = hc.copy()
            ht[i] -= d1
            ht[j] += d1
            ht[k] -= d2
            ht[k + 1 % n] += d2
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(-0.01, 0.01)
            ht = hc.copy()
            ht[i] -= delta
            ht[j] += delta
        valid = True
        for idx in range(n):
            if ht[idx] < -1e-9 or ht[idx] > 1.0 + 1e-9:
                valid = False
                break
        if not valid:
            continue
        ht = np.clip(ht, 0.0, 1.0)
        c5 = compute_c5(ht, dx)
        if c5 < best:
            hc = ht
            best = c5
    return hc, best

def _elapsed(t0):
    return time.time() - t0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = None
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()

    best_c5 = float('inf')
    best_h = None
    best_n = 47

    n_list = [47, 50, 55, 64, 80, 100]
    if h0 is not None:
        n_list.insert(0, len(h0))
    n_list = sorted(set(n_list))

    for n in n_list:
        if _elapsed(t0) > budget_s - 60:
            break
        dx = 2.0 / n
        target = n / 2.0

        starts = []
        if h0 is not None and n == len(h0):
            starts.append(h0.copy())
        starts.append(np.full(n, 0.5))
        for f in [1, 2, 3, 5, 7, 11]:
            h_s = 0.5 + 0.45 * np.sin(2.0 * np.pi * f * np.arange(n) / n)
            starts.append(proj(np.clip(h_s, 0.0, 1.0), target))
        h_b = np.zeros(n)
        h_b[::2] = 1.0
        starts.append(proj(h_b, target))
        for ratio in [0.3, 0.4, 0.6, 0.7]:
            h_blk = np.full(n, 0.5)
            h_blk[:int(n * ratio)] = 0.85
            h_blk[int(n * ratio):] = 0.15
            starts.append(proj(h_blk, target))
        for _ in range(3):
            starts.append(proj(rng.rand(n), target))

        for si, h_init in enumerate(starts):
            if _elapsed(t0) > budget_s - 20:
                break
            h_cur = h_init.copy()
            max_iter = 200 if si < 5 else 100
            for beta in [10, 50, 200, 1000]:
                if _elapsed(t0) > budget_s - 10:
                    break
                _OPT['dx'] = dx
                _OPT['beta'] = beta
                _OPT['target'] = target
                try:
                    res = minimize(_obj_pen, h_cur, method='L-BFGS-B',
                                 bounds=[(0.0, 1.0)] * n,
                                 options={'maxiter': max_iter, 'ftol': 1e-15})
                    h_cur = proj(res.x, target)
                except Exception:
                    pass

            h_cur, c5 = local_search(h_cur, dx, rng, 2000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
                best_n = n

    if best_h is not None and _elapsed(t0) < budget_s - 5:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        _OPT['dx'] = dx
        _OPT['beta'] = 1000
        _OPT['target'] = target
        try:
            res = minimize(_obj_slsqp, best_h, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': _con_eq}],
                         options={'maxiter': 500, 'ftol': 1e-14})
            h_try = proj(res.x, target)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        except Exception:
            pass

        if _elapsed(t0) < budget_s - 5:
            h_cur, c5 = local_search_multi(best_h, dx, rng, 8000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur

    if best_h is None:
        n = 47
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / n)
        best_n = n

    return best_h, best_c5, best_n
