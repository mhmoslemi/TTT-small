        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        n_init = 45
        h_init = project_feasible(rng.uniform(0, 1, n_init), n_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: DE on n_init
    t_d = t0 + budget_s * 0.12
    h_de, c_de = de_optimize(n_init, seed, t_d, popsize=20, maxiter=4000)
    if c_de < best_c5:
        best_c5 = c_de
        best_h = h_de

    # Phase 2: Local search on best
    t_d = t0 + budget_s * 0.22
    h_ls, c_ls = local_search(best_h, n_init, rng, 30000, t_d)
    if c_ls < best_c5:
        best_c5 = c_ls
        best_h = h_ls

    # Phase 3: Multiple random restarts at n_init
    t_d = t0 + budget_s * 0.35
    n_restarts = 12
    for r in range(n_restarts):
        if frac_left() < 0.08:
            break
        if r % 3 == 0:
            h_s = structured_init(n_init, rng)
        elif r % 3 == 1:
            h_s = project_feasible(best_h + rng.normal(0, 0.15, n_init), n_init)
        else:
            h_s = project_feasible(rng.uniform(0, 1, n_init), n_init)
        t_sub = min(t_d, t0 + budget_s * 0.9)
        h_r, c_r = local_search(h_s, n_init, rng, 8000, t_sub)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r

    # Phase 4: Try increasing n
    n_candidates = [50, 75, 100, 150, 200, 300]
    for ci, n_try in enumerate(n_candidates):
        if frac_left() < 0.05:
            break
        if n_try <= len(best_h):
            continue
        n_left = len(n_candidates) - ci
        t_alloc = budget_s * 0.55 / len(n_candidates)
        t_dead = min(time.time() + t_alloc, t0 + budget_s * 0.92)

        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)

        h_t, c_t = de_optimize(n_try, seed + 100 * ci, t_dead, popsize=12, maxiter=2000)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t
        elif h_t is None:
            h_t = h_interp

        t_ls = min(time.time() + t_alloc * 0.4, t0 + budget_s * 0.93)
        h_t2, c_t2 = local_search(h_t, n_try, rng, 15000, t_ls)
        if c_t2 < best_c5:
            best_c5 = c_t2
            best_h = h_t2

    # Phase 5: Final intensive polish
    t_final = t0 + budget_s * 0.98
    h_pol, c_pol = local_search(best_h, len(best_h), rng, 100000, t_final)
    if c_pol < best_c5:
        best_c5 = c_pol
        best_h = h_pol

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
