from scipy.optimize import minimize
import time

def center_sum(h, target):
    """Adjust h so that sum(h) = target, keeping values in [0,1]"""
    x = h.copy().astype(float)
    n = len(x)
    for _ in range(200):
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-10:
            break
        free = (x > 1e-8) & (x < 1.0 - 1e-8)
        if free.sum() == 0:
            # If all at bounds, spread among closest to bounds
            if diff > 0:
                # Need to increase, pick those closest to 1
                idx = np.argsort(1.0 - x)[:max(1, int(n*0.1))]
                x[idx] += diff / len(idx)
            else:
                idx = np.argsort(x)[:max(1, int(n*0.1))]
                x[idx] += diff / len(idx)
        else:
            x[free] += diff / free.sum()
    return np.clip(x, 0.0, 1.0)

def compute_c5(h):
    """Compute the exact C5 value"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    """Smooth approximation of max correlation"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    # log-sum-exp for numerical stability
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / alpha

    return np.interp(xs_new, xs_old, h)


def projected_grad(h, lr_init, n_iter, t_limit):
    x = h.copy()
    n = len(x)
    target = n / 2.0
    dx = 2.0 / n
    best_x = x.copy()
    best_f = compute_c5(x)
    lr = lr_init
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        k_max = np.argmax(corr)
        s = k_max - (n - 1)
        idx = np.arange(n)
        grad = np.zeros(n)
        p1 = idx + s
        v1 = (p1 >= 0) & (p1 < n)
        grad[v1] += (1.0 - x[p1[v1]])
        p2 = idx - s
        v2 = (p2 >= 0) & (p2 < n)
        grad[v2] -= x[p2[v2]]
        grad *= dx
        x = x - lr * grad
        x = np.clip(x, 0.0, 1.0)
        x = center_sum(x, target)
        f = compute_c5(x)
        if f < best_f:
            best_f = f
            best_x = x.copy()
        if it % 200 == 0 and lr > 1e-6:
            lr *= 0.85
    return best_x


def coord_descent(h, n_sweeps, t_limit):
    x = h.copy()
    n = len(x)
    target = n / 2.0
    dx = 2.0 / n
    best_f = compute_c5(x)
    for sw in range(n_sweeps):
        if time.time() > t_limit:
            break
        improved = False
        for j in range(n):
            old = x[j]
            best_val = old
            best_local_f = compute_c5(x)
            for cand in [old * 1.1, old * 0.9, old + 0.05, old - 0.05, 0.0, 1.0]:
                if cand < 0 or cand > 1:
                    continue
                x[j] = cand
                x = center_sum(x, target)
                lf = compute_c5(x)
                if lf < best_local_f - 1e-12:
                    best_local_f = lf
                    best_val = cand
            x[j] = best_val
            x = center_sum(x, target)
            if best_val != old:
                improved = True
        if compute_c5(x) < best_f:
            best_f = compute_c5(x)
        if not improved:
            break
    return x


def run(seed=42, budget_s=1100, **kwargs):
    global _g_n, _g_alpha, _g_target
    np.random.seed(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.random.uniform(0.3, 0.7, 42)
    n_best = len(h_best)
    h_best = center_sum(h_best, n_best / 2.0)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with increasing smoothness
    for alpha in [10.0, 30.0, 80.0, 200.0]:
        if time.time() > t_start + budget_s * 0.28:
            break
        h_opt = do_optimize(h_best, alpha, 1000, 1e-10)
        c5_opt = compute_c5(h_opt)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

    # Phase 2: Projected gradient for escape
    t_pg_end = t_start + budget_s * 0.48
    if time.time() < t_pg_end:
        h_pg = projected_grad(h_best, 0.04, 30000, t_pg_end)
        c5_pg = compute_c5(h_pg)
        if c5_pg < c5_best:
            h_best = h_pg
            c5_best = c5_pg

    # Phase 3: Coordinate descent polish
    t_cd_end = t_start + budget_s * 0.58
    if time.time() < t_cd_end:
        h_cd = coord_descent(h_best, 5, t_cd_end)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5_best:
            h_best = h_cd
            c5_best = c5_cd

    # Phase 4: Random restarts
    n0 = len(h_best)
    t_rs_end = t_start + budget_s * 0.75
    for trial in range(60):
        if time.time() > t_rs_end:
            break
        r = np.random.random()
        if r < 0.25:
            h_rand = np.random.uniform(0.0, 1.0, n0)
        elif r < 0.5:
            h_rand = h_best + np.random.normal(0, 0.15, n0)
        elif r < 0.75:
            h_rand = np.random.uniform(0.2, 0.8, n0)
            h_rand[:n0 // 2] += 0.2
            h_rand[n0 // 2:] -= 0.2
        else:
            period = np.random.choice([n0 // 3, n0 // 4, n0 // 5])
            base = np.random.uniform(0, 1, period)
            h_rand = np.tile(base, n0 // period + 1)[:n0]
        h_rand = center_sum(np.clip(h_rand, 0, 1), n0 / 2.0)
        h_opt = do_optimize(h_rand, 50.0, 500, 1e-9)
        c5_opt = compute_c5(h_opt)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

    # Phase 5: Grid refinement
    t_ref_end = t_start + budget_s * 0.9
    for n_try in [84, 168, 336]:
        if time.time() > t_ref_end:
            break
        h_up = upscale_linear(h_best, n_try)
        h_up = center_sum(h_up, n_try / 2.0)
        c5_init = compute_c5(h_up)
        if c5_init < c5_best + 0.015:
            h_up = do_optimize(h_up, 40.0, 500, 1e-9)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best:
                h_best = h_up
                c5_best = c5_up

    # Phase 6: Final polish
    if time.time() < t_end - 15:
        h_best = do_optimize(h_best, 150.0, 800, 1e-11)
        c5_best = compute_c5(h_best)
    if time.time() < t_end - 5:
        h_best = coord_descent(h_best, 3, t_end - 2)
        c5_best = compute_c5(h_best)

    h_best = center_sum(h_best, len(h_best) / 2.0)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
