def project_onto_constraint(h, n_points):

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def shift_array(h, k):
    n = len(h)
    out = np.zeros(n)
    if k >= 0:
        if k < n:
            out[:n-k] = h[k:]
    else:
        if -k < n:
            out[-k:] = h[:n+k]
    return out

def compute_gradient(h, n_points, T=30.0):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c_max = np.max(corr)
    weights = np.exp((corr - c_max) * T)
    wsum = weights.sum()
    if wsum > 0:
        weights /= wsum
    else:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    
    n = n_points
    grad = dx * np.ones(n)
    for m, w in enumerate(weights):
        if w < 1e-12:
            continue
        k = m - (n - 1)
        if abs(k) >= n:
            continue
        grad -= w * dx * (shift_array(h, k) + shift_array(h, -k))

def projected_gradient_descent(h, n_points, n_iters=500, T=30.0):

    step = 0.01

        grad = compute_gradient(h, n_points, T)

        for _ in range(20):

                best_h = new_h.copy()
            if new_val < evaluate_fitness(h, n_points) * 0.9999:

            alpha *= 0.5

import numpy as np
import time

def evaluate_fitness(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def project_onto_constraint(h, n_points):
    target = n_points / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def shift_arr(h, k):
    n = len(h)
    out = np.zeros(n)
    if 0 <= k < n:
        out[:n - k] = h[k:]
    elif -n < k < 0:
        out[-k:] = h[:n + k]
    return out

def compute_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    cmax = np.max(corr)
    w = np.exp((corr - cmax) * T)
    ws = w.sum()
    if ws > 1e-300:
        w /= ws
    else:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        if abs(k) >= n:
            continue
        grad -= w[m] * dx * (shift_arr(h, k) + shift_arr(h, -k))
    return grad

def optimize_pgd(h, n, max_iters, T0, Tg, time_limit, seed=0):

            break

        gn = np.linalg.norm(grad)
