```python
import numpy as np
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for m in active:
        l = m - (n - 1)
        # First term: d/dh[j] of sum_i h[i]*(1-h[i+l]) = (1-h[j+l]) if 0<=j+l<n
        if l >= 0:
            j_lo, j_hi = 0, n - 1 - l
        else:
            j_lo, j_hi = -l, n - 1
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += (1.0 - h[idx + l])
        # Second term: -h[j-l] if 0<=j-l<n
        if l >= 0:
            j_lo2, j_hi2 = l, n - 1
        else:
            j_lo2, j_hi2 = 0, n - 1 + l
        if j_lo2 <= j_hi2:
            idx = np.arange(j_lo2, j_hi2 + 1)
            grad[idx] -= h[idx - l]
    grad *= dx / len(active)
    return grad

def project_constrained(h, target_sum):
    n = len(h)
    # Find t such that sum(clip(h+t, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h + t, 0.0, 1.0)

def optimize_pg(h_init, time_limit, start_time, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h_init)
    target_sum = n / 2.0
    best_h = project_constrained(h_init.copy(), target_sum)
    best_c5 = c5_value(best_h)
    
    lr = 0.05
    it = 0
    no_improve = 0
    
    while time.time() - start_time < time_limit:
        grad = c5_gradient(best_h)
        # Try a few step sizes
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            new_h = project_constrained(best_h - trial_lr * grad, target_sum)
            new_c5 = c5_value(new_h)
            if new_c5 < best_c5 - 1e-10:
                best_h = new_h
                best_c5 = new_c5
                improved = True
                lr = trial_lr
                break
        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Add noise to escape
                noise = rng.normal(0, 0.05, n)
                best_h = project_constrained(best_h + noise, target_sum)
                best_c5 = c5_value(best_h)
                lr = max(0.001, lr * 0.5)
                no_improve = 0
        it += 1
    return best_h, best_c5

def coordinate_descent(h, time_limit, start_time, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = c5_value(best_h)
    
    while time.time() - start_time < time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        a, b = best_h[i], best_h[j]
        # Move: h[i] = a + d, h[j] = b - d
        # Constraints: 0 <= a+d <= 1, 0 <= b-d <= 1
        # So d in [max(-a, -(1-b)), min(1-a, b)]
        d_lo = max(-a, b - 1.0)
        d_hi = min(1.0 - a, b)
        if d_lo >= d_hi:
            continue
        # Try a few values of d
        mid = 0.5 * (d_lo + d_hi)
        for d in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1, 0.001, -0.001, mid]:
            if d < d_lo or d > d_hi:
                continue
            trial = best_h.copy()
            trial[i] = a + d
            trial[j] = b - d
            c = c5_value(trial)
            if c < best_c5 - 1e-12:
                best_h = trial
                best_c5 = c
                break
    return best_h, best_c5

    rng = np.random.default_rng(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 45
        h_init = np.full(n, 0.5)

    best_h = project_constrained(h_init.copy(), target_sum)
    best_c5 = c5_value(best_h)
    
    # Phase 1: Gradient descent from initial
    t1 = start_time + 0.4 * budget_s
    best_h, best_c5 = optimize_pg(best_h, t1, start_time, seed)
    
    # Phase 2: Coordinate descent refinement
    t2 = start_time + 0.6 * budget_s
    best_h, best_c5 = coordinate_descent(best_h, t2, start_time, seed + 1)
    
    # Phase 3: Multiple restarts with gradient descent
    t_end = start_time + budget_s - 5
    for restart in range(20):
        remaining = t_end - time.time()
        if remaining < 5:

        # Create perturbed initial
        noise = rng.normal(0, 0.1, n)
```
