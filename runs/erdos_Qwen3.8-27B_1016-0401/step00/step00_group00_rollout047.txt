    # Phase 4: Final fine-tune with subgradient + tiny noise
    t_end4 = t0 + budget_s * 0.97
    h = best_h.copy()
    lr = 0.3
    while time.time() < t_end4:
        grad, _ = subgrad_at_max(h, n)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-15:
            break
        scale = lr / gnorm
        h_new = h - scale * grad + 0.005 * scale * np.random.randn(n)
        h_new = clamp_project(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        lr *= 0.999

    best_h = clamp_project(best_h, n)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return (best_h, final_c5, n)
