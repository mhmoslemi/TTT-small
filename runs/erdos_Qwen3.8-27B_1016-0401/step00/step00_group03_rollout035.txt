import time

def evaluate_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def make_binary(n, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n//2]] = 1.0
    return h

def checkerboard(n):
    h = np.zeros(n)
    h[::2] = 1.0
    s = h.sum()
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        for idx in ones[:int(s - target)]:
            h[idx] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:int(target - s)]:
            h[idx] = 1
    return h

    h = h.copy()

    zeros = np.where(h <= 0.5)[0]

        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

import numpy as np
import time

def evaluate_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def _make_binary(n, seed=0):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _checkerboard(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return h

def _block_pattern(n):
    h = np.zeros(n)
    h[:n // 4] = 1.0
    h[n // 2 : n // 2 + n // 4] = 1.0
    return h

def _hill_climb(h, dx, rng, n_iters):
    n = len(h)
    h = h.astype(float).copy()
    bc = evaluate_c5(h, dx)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for _ in range(n_iters):
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 1.0, 0.0
        c = evaluate_c5(h, dx)
        if c < bc:
            bc = c
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
        else:
            h[i], h[j] = 0.0, 1.0
    return h, bc

def _sa(h, dx, rng, t_limit):
    n = len(h)
    h = h.astype(float).copy()
    bc = evaluate_c5(h, dx)
    bh = h.copy()
    T = 0.04
    end = time.time() + t_limit
    cnt = 0
    no_improve = 0
    while time.time() < end:
        if cnt % 5000 == 0:
            if no_improve > 50:
                T = max(T, 0.02)
                no_improve = 0
            T = max(5e-4, T * 0.998)
        if rng.random() < 0.15 and n >= 4:
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            if len(oi) >= 2 and len(zi) >= 2:
                i1, i2 = rng.choice(oi, 2, replace=False)
                j1, j2 = rng.choice(zi, 2, replace=False)
                h[i1], h[i2], h[j1], h[j2] = 0, 0, 1, 1
                c = evaluate_c5(h, dx)
                d = c - bc
                if d < 0 or rng.random() < np.exp(-d / T):
                    if c < bc:
                        bc, bh, no_improve = c, h.copy(), 0
                    else:
                        no_improve += 1
                else:
                    h[i1], h[i2], h[j1], h[j2] = 1, 1, 0, 0
            else:
                cnt += 1
                continue
        else:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if h[i] == h[j]:
                continue
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            c = evaluate_c5(h, dx)
            d = c - bc
            if d < 0 or rng.random() < np.exp(-d / T):

                    bc, bh, no_improve = c, h.copy(), 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        cnt += 1
    return bh, bc

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-9:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
        if abs(diff) < 1e-9:
            break
    return h
