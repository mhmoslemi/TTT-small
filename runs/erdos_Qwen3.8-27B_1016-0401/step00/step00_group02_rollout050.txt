def _make_ineq_jac(n_pts, dx):
    def jac(vars_):
        h = vars_[:n_pts]
        n_p = n_pts
        m = 2 * n_p - 1
        J = np.zeros((m, n_p + 1))
        for k in range(m):
            L = k - (n_p - 1)  # lag
            for j in range(n_p):
                jL = j + L
                jnL = j - L
                val = 0.0
                if 0 <= jL < n_p:
                    val += (1.0 - h[jL])
                if 0 <= jnL < n_p:
                    val -= h[jnL]
                J[k, j] = -val * dx
            J[k, n_p] = 1.0
        return J
    return jac

def compute_c5(h):

    h = np.clip(h, 0.0, 1.0)

    diff = target - s

            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)

def _slsqp_solve(h0, n_pts, time_limit):

    dx = 2.0 / n
    
    def obj(vars_):
        return vars_[n]

        h = vars_[:n]
        t = vars_[n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr * dx
    
    def ineq_jac(vars_):
        h = vars_[:n]
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        for k in range(m):
            L = k - (n - 1)
            row = J[k]
            for j in range(n):
                v = 0.0
                jL = j + L
                if 0 <= jL < n:
                    v += 1.0 - h[jL]
                jnL = j - L
                if 0 <= jnL < n:
                    v -= h[jnL]
                row[j] = -v * dx
            row[n] = 1.0
        return J
    
    def eq(vars_):
        return np.sum(vars_[:n]) - n / 2.0
    
    def eq_jac(vars_):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}
    ]

                   options={'maxiter': 2000, 'ftol': 1e-10})

        c5 = compute_c5(h_sol)
        if c5 < compute_c5(h0):

    return h0.copy(), compute_c5(h0)

    return h_sol, compute_c5(h_sol)

import numpy as np
from scipy.optimize import minimize
import time

_N = 45

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def proj_feasible(h):
    n = _N
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def t_obj(x):
    return x[_N]

def t_ineq(x):
    n = _N
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr * dx

def t_ineq_jac(x):
    n = _N
    h = x[:n]
    sx = 2.0 / n
    m = 2 * n - 1
    J = np.zeros((m, n + 1))
    for k in range(m):
        L = k - (n - 1)
        for j in range(n):
            v = 0.0
            ip = j + L
            if 0 <= ip < n:
                v += 1.0 - h[ip]
            im = j - L
            if 0 <= im < n:
                v -= h[im]
            J[k, j] = -v * sx
        J[k, n] = 1.0
    return J

def t_eq(x):
    return np.sum(x[:_N]) - _N / 2.0

def t_eq_jac(x):
    j = np.zeros(_N + 1)
    j[:_N] = 1.0
    return j

def run_slsqp(h0):
    global _N
    n = len(h0)
    _N = n
    bounds = [(0.0, 1.0)] * (n + 1)
    x0 = np.append(h0.copy(), compute_c5(h0))
    cons = [
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac},
        {'type': 'ineq', 'fun': t_ineq, 'jac': t_ineq_jac}
    ]
    res = minimize(t_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
    h_sol = np.clip(res.x[:n], 0.0, 1.0)
    h_sol = proj_feasible(h_sol)
    return h_sol, compute_c5(h_sol)

def nm_penalized(h):
    n = _N
    h_c = np.clip(h, 0.0, 1.0)
    c5 = compute_c5(h_c)
    pen = 500.0 * (np.sum(h_c) - n / 2.0) ** 2
    out_of = np.sum(np.maximum(0, h - 1.0) ** 2) + np.sum(np.maximum(0, -h) ** 2)
    pen += 500.0 * out_of
    return c5 + pen

def run_nm(h0):
    global _N
    n = len(h0)
    _N = n
    res = minimize(nm_penalized, h0, method='Nelder-Mead',
                   options={'maxiter': 20000, 'fatol': 1e-13, 'xatol': 1e-13,
                            'adaptive': True})
    h_sol = np.clip(res.x, 0.0, 1.0)
    h_sol = proj_feasible(h_sol)
    return h_sol, compute_c5(h_sol)

def run_cd(h0):
    """Cyclic coordinate descent on h with projection"""
    global _N
    n = _N
    target = n / 2.0
    h = np.array(h0, dtype=float).copy()
    c5_cur = compute_c5(h)
    for it in range(5000):
        improved = False
        for j in range(n):
            for delta in [0.05, 0.1, 0.2, 0.5]:
                for sign in [1, -1]:
                    h2 = h.copy()
                    h2[j] = np.clip(h[j] + sign * delta, 0.0, 1.0)
                    diff = target - np.sum(h2)
                    h2 = np.clip(h2 + diff / n, 0.0, 1.0)
                    c5_new = compute_c5(h2)
                    if c5_new < c5_cur - 1e-12:
                        h = h2
                        c5_cur = c5_new
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 5.0

    else:

        h_init[22] = 0.5
    n_init = len(h_init)

    h_init = proj_feasible(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def still_go():
        return time.time() < deadline

    def update(hs, c5s):
        nonlocal best_h, best_c5
        if c5s < best_c5:
            best_h = hs.copy()
            best_c5 = c5s
    # Can't use nonlocal without closure - just check inline

    # Phase 1: SLSQP from initial
    if still_go():

    # Phase 2: NM polish

    # Phase 3: CD polish
    if still_go():

    # Phase 4: Perturbation restarts
    for i in range(25):
        if not still_go():

        scale = 0.03 + 0.2 * (i % 6) / 6.0
        h_pert = best_h + rng.randn(n_init) * scale
        h_pert = proj_feasible(h_pert)
        h_s, c5_s = run_slsqp(h_pert)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
        if still_go() and c5_s < best_c5 + 0.003:
            h_s2, c5_s2 = run_nm(h_s)
            if c5_s2 < best_c5:
                best_h, best_c5 = h_s2, c5_s2
            h_s3, c5_s3 = run_cd(best_h)
            if c5_s3 < best_c5:
                best_h, best_c5 = h_s3, c5_s3

    # Phase 5: Random structured starts
    for i in range(10):
        if not still_go():
            break
        # Block-structured random
        n_blocks = rng.randint(3, 10)
        block_sizes = np.full(n_blocks, n_init // n_blocks)
        block_sizes[:n_init % n_blocks] += 1
        h_struct = np.zeros(n_init)
        offsets = np.cumsum([0] + block_sizes[:-1])
        for gi, (off, sz) in enumerate(zip(offsets, block_sizes)):

                h_struct[off:off + sz] = 1.0
        h_struct = proj_feasible(h_struct)
        h_s, c5_s = run_slsqp(h_struct)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 6: Multigrid - higher resolution
    remaining = deadline - time.time()
    if remaining > 100:
        for n_hi in [90, 180]:
            if remaining < 60:

            x_old = np.linspace(0, 2, n_init, endpoint=False)

            h_hi = np.interp(x_new, x_old, best_h)
            _N = n_hi
            h_hi = proj_feasible(h_hi)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:

            if remaining > 50:

            if remaining > 30:

                if c5_s2 < best_c5:

        if c5_s < best_c5:

    best_h = proj_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
