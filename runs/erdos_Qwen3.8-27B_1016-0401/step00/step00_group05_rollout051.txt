import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(h, lam=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr * lam - m * lam))) / lam)

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(1.0 - h[i], diff)
                if add <= 0:
                    break
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(h[i], -diff)
                if sub <= 0:
                    break
                h[i] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return h

def binarize_h(h, n):
    h_b = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_b == 1.0)[0]
        order = np.argsort(h[ones])
        for idx in order[n_ones - target:]:
            h_b[ones[idx]] = 0.0
    elif n_ones < target:
        zeros = np.where(h_b == 0.0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target - n_ones]:
            h_b[zeros[idx]] = 1.0
    return h_b

def binary_local_search(h, t0, budget_s, max_time=60.0):
    n = len(h)
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    improved_count = 0
    while (time.time() - t0 < budget_s - 20) and improved_count < 5000:
        if (time.time() - t0) > min(t0 + budget_s - 20, t0 + max_time + 12):
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-13
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                gain = c5 - c5_new
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c5 -= best_gain
            improved_count += 1
        else:
            break
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = budget_s - 12

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    n0 = 70
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        pass
    if h0 is None:
        h0 = np.ones(n0) * 0.5
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n0)

    c5_cur = compute_c5(h0)
    best_h, best_c5, best_n = h0.copy(), c5_cur, n0

    # Phase 1: Continuous optimization with increasing lambda
    work_h = h0.copy()
    for lam in [20, 50, 100, 200, 500]:
        if time.time() - t0 > deadline:
            break
        res = minimize(smooth_max_obj, work_h.copy(), args=(lam,),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * best_n,
                      options={'maxiter': 400, 'ftol': 1e-13})
        h_opt = fix_sum(np.clip(res.x, 0.0, 1.0), best_n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5 = h_opt.copy(), c5_opt
            work_h = h_opt.copy()

    # Phase 2: Binarize from best continuous + local search
    if time.time() - t0 < deadline:
        h_bin = binarize_h(best_h, best_n)
        h_bin, c5_bin = binary_local_search(h_bin, t0, budget_s, max_time=80.0)
        if c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin.copy(), c5_bin, best_n

    # Phase 3: Try other n values
    for n_t in [80, 90, 100, 120, 60]:
        if time.time() - t0 > deadline:
            break
        h_t = fix_sum(interp_to_n(best_h, n_t), n_t)
        for lam in [80, 200]:
            if time.time() - t0 > deadline:

            res = minimize(smooth_max_obj, h_t.copy(), args=(lam,),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_t,
                          options={'maxiter': 300, 'ftol': 1e-13})
            h_opt = fix_sum(np.clip(res.x, 0.0, 1.0), n_t)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5 - 1e-10:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_t
                h_t = h_opt.copy()

    # Phase 4: Random restarts in continuous space
    n_r = best_n
    for restart in range(8):
        if time.time() - t0 > deadline:
            break
        h_r = best_h.copy()
        noise = np.random.randn(n_r) * 0.15
        h_r = fix_sum(np.clip(h_r + noise, 0.0, 1.0), n_r)
        res = minimize(smooth_max_obj, h_r, args=(200,),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_r,
                      options={'maxiter': 500, 'ftol': 1e-13})
        h_opt = fix_sum(np.clip(res.x, 0.0, 1.0), n_r)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_r

    # Phase 5: Final binary local search on best
    if time.time() - t0 < deadline:
        h_bin = binarize_h(best_h, best_n)
        h_bin, c5_bin = binary_local_search(h_bin, t0, budget_s, max_time=60.0)
        if c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin.copy(), c5_bin, best_n

    # Phase 6: Final continuous polish
    if time.time() - t0 < deadline:
        for lam in [300, 800]:
            if time.time() - t0 > deadline:
                break
            res = minimize(smooth_max_obj, best_h.copy(), args=(lam,),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * best_n,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            h_opt = fix_sum(np.clip(res.x, 0.0, 1.0), best_n)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5 - 1e-10:
                best_h, best_c5 = h_opt.copy(), c5_opt

    return best_h, best_c5, best_n
