import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_active_grad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    c5 = float(corr[k_star] * dx)
    s = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        i1 = j + s
        i2 = j - s
        if 0 <= i1 < n:
            grad[j] += (1.0 - h[i1])
        if 0 <= i2 < n:
            grad[j] -= h[i2]
    grad *= dx
    return c5, grad

def smooth_c5_obj(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau)

def smooth_c5_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp(tau * (corr - m))
    w /= w.sum()
    s_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k_idx, s in enumerate(s_vals):
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        for j in range(n):
            i1 = j + s
            i2 = j - s
            g = 0.0
            if 0 <= i1 < n:
                g += (1.0 - h[i1])
            if 0 <= i2 < n:
                g -= h[i2]
            grad[j] += wi * g
    return grad * dx

def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(100):
        diff = x.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        x = x - diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    return x

def polish_coordinate(h, dx, target_sum, n_iters=50):
    n = len(h)
    for it in range(n_iters):
        improved = False
        for j in range(n):
            others = h.sum() - h[j]
            lo = max(0.0, target_sum - min(1.0, others))
            hi = min(1.0, target_sum - others)
            if lo > hi:
                lo, hi = 0.0, 1.0
            best_val = h[j]
            best_c5 = compute_c5(h, dx)
            for trial in np.linspace(lo, hi, 21):
                h[j] = trial
                h = project_feasible(h, target_sum)
                c = compute_c5(h, dx)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_val = h[j]
            h[j] = best_val
            if best_c5 < compute_c5(np.array(h), dx) + 1e-12:
                improved = True
        if not improved:
            break
    return h
