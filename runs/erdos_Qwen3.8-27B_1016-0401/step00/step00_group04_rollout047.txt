def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    return np.max(corr) * dx

def smooth_max(vals, T):
    """Compute smooth max and its gradient w.r.t. input indices"""
    shifted = T * vals
    max_val = shifted.max()
    exp_shifted = np.exp(shifted - max_val)
    log_sum = np.log(exp_shifted.sum()) + max_val
    result = log_sum / T
    probs = exp_shifted / exp_shifted.sum()
    return result, probs

def objective(h_flat, T, n):
    """Objective: smooth max of overlaps + penalty for constraint violation"""
    h = h_flat
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    overlaps = corr * dx
    sm, probs = smooth_max(overlaps, T)
    return sm

def objective_grad(h_flat, T, n):
    """Gradient of smooth max objective"""
    h = h_flat
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    overlaps = corr * dx
    sm, probs = smooth_max(overlaps, T)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max(vals, T):
    s = T * vals
    mx = s.max()
    e = np.exp(s - mx)
    ss = e.sum()
    if ss < 1e-300:
        return vals.max(), np.zeros_like(vals)
    return (np.log(ss) + mx) / T, e / ss

def obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    ov = get_overlaps(h)
    f, w = smooth_max(ov, T)
    g = np.zeros(n)
    for m in range(len(ov)):
        wm = w[m]
        if wm < 1e-14:
            continue
        t = m - n + 1
        a, b = max(0, t), min(n, n + t)
        if b > a:
            g[a:b] += wm * (1.0 - h[a - t:b - t])
        a, b = max(0, -t), min(n, n - t)
        if b > a:
            g[a:b] += wm * (-h[a + t:b + t])
    return f, g * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(300):
        d = h.sum() - target
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h > 1e-12
            if not m.any():
                break
            h[m] -= d / m.sum()
        else:
            m = h < 1.0 - 1e-12
            if not m.any():
                break
            h[m] += -d / m.sum()
    return np.clip(h, 0.0, 1.0)

def grad_desc(h, n, T0, T1, n_steps, lr0, rng, record_time):
    h = h.copy()
    best = h.copy()
    best_f = compute_c5(h)
    lr = lr0
    mom = np.zeros(n)
    beta = 0.85
    for it in range(n_steps):
        if record_time() < 0:
            break
        frac = it / max(1, n_steps - 1)
        T = T0 * (T1 / T0) ** frac
        f, g = obj_and_grad(h, T)
        mom = beta * mom + (1 - beta) * g
        h_new = project(h - lr * mom, n)
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best = h_new.copy()
            best_f = f_new
            h = h_new
            lr = min(lr * 1.03, 0.4)
        else:
            h = h_new
            lr *= 0.82
        if it % 300 == 150:
            h = project(h + rng.normal(0, 0.03, n), n)
    return best, best_f

def coord_refine(h, n, rng, record_time):
    h = h.copy()
    best_f = compute_c5(h)
    for rnd in range(30):
        if record_time() < 0:
            break
        improved = False
        for j in range(n):
            for delta in [0.15, -0.15, 0.05, -0.05]:
                h2 = h.copy()
                h2[j] += delta
                h2 = project(h2, n)
                f2 = compute_c5(h2)
                if f2 < best_f - 1e-10:
                    h = h2
                    best_f = f2
                    improved = True
                    break
        if not improved:
            break
    return h, best_f

def make_pattern(n, kind, rng):
    if kind == 0:
        return np.full(n, 0.5)
    elif kind == 1:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(np.pi * x)
        return project(np.clip(h, 0, 1), n)
    elif kind == 2:
        h = np.full(n, 0.2)
        w = n // 8
        h[:w] = 0.8
        h[n - w:] = 0.8
        return project(h, n)
    elif kind == 3:
        return project(np.clip(np.full(n, 0.5) + rng.normal(0, 0.15, n), 0, 1), n)
    else:
        h = np.zeros(n)
        k = max(1, n // 7)
        for i in range(0, n, 2 * k):
            h[i:i + k] = 1.0
        return project(h, n)

    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h, best_c5, best_n = None, 1.0, 0

    def remaining():
        return budget_s - (time.time() - t0) - 15

    try:

    except NameError:
        h_init = None

    n_base = len(h_init) if h_init is not None else 42

    if h_init is not None:

    # Phase 2: try multiple starting patterns at n_base
    patterns = [0, 1, 2, 3, 4]
    for p in patterns:

            break

        if cp < best_c5:
            best_c5, best_h, best_n = cp, hp, n_base

    ref_n_list = [n_base + 21, n_base + 42, n_base + 63]

        if nr > 500:

        if t_avail > 20:
            steps = max(200, int(t_avail * 500))

            best_h, c_pol = coord_refine(best_h, best_n, rng, remaining)
            if c_pol < best_c5:
                best_c5 = c_pol

    # Final feasibility check
    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
