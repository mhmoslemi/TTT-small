from scipy.optimize import minimize
import time

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    m = float(np.max(vals))
    if T < 1e-12:
        return m
    shifted = np.clip((vals - m) / T, -60.0, 0.0)
    return m + T * float(np.log(np.sum(np.exp(shifted))))

def project_constraint(h, n):
    for _ in range(100):
        d = n / 2.0 - np.sum(h)
        if abs(d) < 1e-14:
            break
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(h_start, n, T, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_constraint(res.x, n)
        return h_new
    except Exception:
        return h_start.copy()

    bounds = [(0.0, 1.0)] * n

        pen = 200.0 * abs(np.sum(x) - n / 2.0)
        return smooth_obj(xc, T) + pen
    try:
        res = minimize(obj, h_start, method='Powell', bounds=bounds,
                       options={'maxiter': maxiter, 'xtol': 1e-12, 'ftol': 1e-14})
        h_new = project_constraint(res.x, n)
        return h_new
    except Exception:
        return h_start.copy()

def local_search(h, n, max_rounds=5):
    best = h.copy()
    best_c = eval_c5(best)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            old = best[i]
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                test = best.copy()
                test[i] = np.clip(old + delta, 0.0, 1.0)
                test = project_constraint(test, n)
                c = eval_c5(test)
                if c < best_c - 1e-12:
                    best = test
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h = initial_h_values.astype(float).copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(best_h)

    # Phase 1: SLSQP with temperature annealing
    temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for T in temps:
        if elapsed() > budget_s - 200:
            break
        h_new = slsqp_solve(best_h, n, T, maxiter=500)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Local search refinement
    if elapsed() < budget_s - 100:
        best_h, best_c5 = local_search(best_h, n, max_rounds=3)

    # Phase 3: Random perturbations + re-optimization
    for attempt in range(30):
        if elapsed() > budget_s - 100:
            break
        scale = 0.03 + 0.04 * np.random.rand()
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = project_constraint(h_pert, n)
        h_new = slsqp_solve(h_pert, n, 0.001, maxiter=250)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Powell refinement
    if elapsed() < budget_s - 50:
        h_new = powell_solve(best_h, n, T=0.0005, maxiter=800)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Final local search
    if elapsed() < budget_s - 20:
        best_h, best_c5 = local_search(best_h, n, max_rounds=5)

    # Phase 6: One more SLSQP at very low temp
    if elapsed() < budget_s - 10:
        h_new = slsqp_solve(best_h, n, 0.00001, maxiter=300)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    return (best_h, best_c5, n)
