def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    # Find active shifts
    max_val = np.max(corr)
    active = np.where(np.isclose(corr, max_val, atol=1e-12))[0]
    # Compute gradient for active shifts
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            # F = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # dF/dh[j] = (1-h[j-s]) for j in [s, n-1] - h[j+s] for j in [0, n-1-s]
            idx1 = np.arange(s, n)
            g[idx1] += (1.0 - h[idx1 - s])
            idx2 = np.arange(0, n - s)
            g[idx2] -= h[idx2 + s]
        else:
            t = -s
            # F = sum_{j=0}^{n-1-t} h[j]*(1-h[j+t])
            # dF/dh[j] = (1-h[j+t]) for j in [0, n-1-t] - h[j-t] for j in [t, n-1]
            idx1 = np.arange(0, n - t)
            g[idx1] += (1.0 - h[idx1 + t])
            idx2 = np.arange(t, n)
            g[idx2] -= h[idx2 - t]
        grad += g
    grad *= dx / len(active)
    return c5, grad

def project_feasible(h):
    """Project h to [0,1]^n with sum = n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Fix sum
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Distribute correction
        if diff > 0:
            # Need to increase; add to elements below 1
            room = 1.0 - h
            mask = room > 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            # Need to decrease; subtract from elements above 0
            room = h
            mask = room > 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize(h_init, max_iter=5000, lr_init=0.01):
    """Gradient descent with projection."""
    n = len(h_init)
    h = h_init.copy()
    h = project_feasible(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()

        if step_c5 > c5:
            lr *= 0.9
        else:
            lr = min(lr * 1.05, 0.1)

    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_grad_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Active set: shifts at the max
    active = np.where(corr >= max_val - 1e-8)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s >= 0:
            # gradient of f_s
            j = np.arange(s, n)
            grad[j] += (1.0 - h[j - s])
            j2 = np.arange(0, n - s)
            grad[j2] -= h[j2 + s]
        else:
            t = -s
            j = np.arange(0, n - t)
            grad[j] += (1.0 - h[j + t])
            j2 = np.arange(t, n)
            grad[j2] -= h[j2 - t]
    grad /= len(active)
    grad *= dx
    return grad, max_val * dx

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if diff > 0:

        else:

        if not mask.any():

        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def soft_max_grad(h, beta):
    """Compute soft max of overlap and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Soft max: (1/beta) * log(sum(exp(beta * corr)))
    # For numerical stability, subtract max
    mc = np.max(corr)
    exps = np.exp(beta * (corr - mc))
    soft = (mc + np.log(np.sum(exps))) / beta * dx
    weights = exps / np.sum(exps)
    # Gradient: weighted sum of gradients of each corr[k]
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-12:
            continue
        s = idx - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            j = np.arange(s, n)
            g[j] = (1.0 - h[j - s])
            j2 = np.arange(0, n - s)
            g[j2] -= h[j2 + s]
        else:
            t = -s
            j = np.arange(0, n - t)
            g[j] = (1.0 - h[j + t])
            j2 = np.arange(t, n)
            g[j2] -= h[j2 - t]
        grad += weights[idx] * g
    grad *= dx
    return soft, grad

def project(h):

        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_flip_matrix(h, n):
    """D[j, m] = change in corr[m] for 0→1 at j (binary h)."""
    D = np.zeros((n, 2*n-1))
    shifts = np.arange(-(n-1), n)  # s values for m=0,...,2n-2
    for j in range(n):
        for idx_s, s in enumerate(shifts):
            if s == 0:
                continue
            m = (n-1) + s
            effect = 0.0
            # j as first factor: h[j+s] is the "other"
            if 0 <= j+s < n:
                effect += (1.0 - h[j+s])
            # j as second factor: h[j-s] is the "other"
            if 0 <= j-s < n:
                effect -= h[j-s]
            D[j, m] = effect
    return D

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode="full")

    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def flip_matrix(h, n):
    D = np.zeros((n, 2 * n - 1))
    for j in range(n):
        for s in range(-(n-1), n):
            if s == 0:
                continue
            m = (n - 1) + s
            e = 0.0
            if 0 <= j + s < n:
                e += 1.0 - h[j + s]
            if 0 <= j - s < n:
                e -= h[j - s]
            D[j, m] = e
    return D

def local_search(h0, time_limit=60, seed=42):
    n = len(h0)
    h = to_binary(h0)
    rng = np.random.default_rng(seed)
    start = time.time()
    
    ov = compute_overlaps(h)
    cur_max = np.max(ov)
    best_h = h.copy()
    best_max = cur_max
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    T = 0.05
    T_min = 1e-8
    cool = 0.99995
    
    while time.time() - start < time_limit:
        D = flip_matrix(h, n)
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        
        # Compute new max for all pairs
        Dz = D[zeros]
        Do = D[ones]
        # (nz, no, 2n-1)
        nm = (ov[None, None, :] + Dz[:, None, :] - Do[None, :, :]).max(axis=2)

            bz = zeros[fi[0]]
            bo = ones[fi[1]]

            o_pos = np.where(ones == bo)[0][0]

            ones = np.delete(ones, o_pos)
            zeros = np.append(zeros, bo)
            ones = np.append(ones, bz)
            ov = compute_overlaps(h)

                best_h = h.copy()
