import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    c_shifted = beta * (c - c_max)
    return c_max + float(np.log(np.sum(np.exp(c_shifted)))) / beta

def sum_constraint(h, n):
    return np.sum(h) - n / 2.0

def initial_construction(n):
    h = np.zeros(n)
    x = np.arange(n) * 2.0 / n
    mask = (x < 0.5) | ((x >= 1.0) & (x < 1.5))
    h[mask] = 1.0
    s = np.sum(h)
    if s > 0:
        h *= (n / 2.0) / s
    return np.clip(h, 0, 1)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new)
