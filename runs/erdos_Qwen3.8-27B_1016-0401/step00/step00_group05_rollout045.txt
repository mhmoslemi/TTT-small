        g = np.zeros(n)
        for k in ai:
            g += grad_lag(h, int(k), n)
        g *= dx / max(len(ai), 1)
        vel = 0.85 * vel - alpha * g
        h_new = h + vel
        h_new = project_normalize(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5_new
            alpha = min(alpha * 1.02, 0.06)
            stall = 0
        else:
            stall += 1
            vel *= 0.5
            if stall % 40 == 0:
                alpha *= 0.92
            # Periodic local moves
            if stall % 500 == 499 and time.time() < t_end - 2:
                h_lm, c5_lm = local_moves(h, n, dx, rng, 20)
                if c5_lm < best_c5:
                    h = h_lm
                    best_c5 = c5_lm
                    stall = 0
                    alpha = 0.012
            if stall > patience:
                s = min(0.12, 0.005 + 0.05 * (stall / patience))
                h_try = h + rng.normal(0, s, n)
                h_try = project_normalize(h_try, n)
                c5_t = compute_c5(h_try, dx)
                if c5_t < best_c5:
                    h = h_try
                    best_c5 = c5_t
                alpha = 0.012
                vel[:] = 0
                stall = 0
    return h, best_c5

def spectral_reseed(h, n, dx, rng):
    H = fft(h.astype(complex))
    freq = np.fft.fftfreq(n)
    weight = np.exp(-4.0 * freq * freq)
    pert = (rng.normal(0, 0.03, n) + 1j * rng.normal(0, 0.03, n)) * weight
    h_new = np.real(ifft(H + pert))
    h_new = project_normalize(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 70

    h_init = None
    n_init = 0
    try:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
        h_init = np.clip(h_init, 0.0, 1.0)
        n_init = len(h_init)
    except Exception:
        pass

    if h_init is None:
        n_init = 70
        h_init = np.zeros(n_init)
        for i in range(n_init):
            h_init[i] = 1.0 if (i % 7 < 3) else 0.0
        h_init = project_normalize(h_init, n_init)

    # Phase 1: optimize initial n
    t_p1 = t0 + budget_s * 0.35
    dx1 = 2.0 / n_init
    h1, c5_1 = main_optimize(h_init, n_init, dx1, rng, t_p1)
    best_h, best_c5, best_n = h1, c5_1, n_init

    # Phase 2: try other n values via interpolation
    for n2 in [100, 140]:
        if time.time() > t0 + budget_s - 15:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h1, period=2.0)
        h2 = project_normalize(h2, n2)
        dx2 = 2.0 / n2
        deadline = t0 + budget_s - 8
        h2, c5_2 = main_optimize(h2, n2, dx2, rng, deadline, patience=3000)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Phase 3: polish best solution
    if time.time() < t0 + budget_s - 3:
        dx3 = 2.0 / best_n
        # Try spectral reseed + optimize
        h3 = spectral_reseed(best_h, best_n, dx3, rng)
        c5_3 = compute_c5(h3, dx3)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
        h3, c5_3 = main_optimize(best_h, best_n, dx3, rng, t0 + budget_s - 1, patience=2000)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Final local polish
    if time.time() < t0 + budget_s - 0.5:
        dx_f = 2.0 / best_n
        best_h, c5_f = local_moves(best_h, best_n, dx_f, rng, 50)
        if c5_f < best_c5:
            best_c5 = c5_f

    return best_h, best_c5, best_n
