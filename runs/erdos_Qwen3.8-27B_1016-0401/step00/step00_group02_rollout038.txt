from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, temp=0.02):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    m = np.max(corr)
    # smooth max: m + temp * log(sum(exp((corr-m)/temp)) / (2n-1))
    # = (1/(2n-1)) * log-sum-exp((corr-m)/temp) * temp + m
    vals = (corr - m) / temp
    log_sum = temp * (np.max(vals) + np.log(np.sum(np.exp(vals - np.max(vals))))) / (2*n-1) + m
    return float(log_sum * dx)

def project_sum(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2} using bisection."""
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    h = np.clip(params, 0, 1)
    val = smooth_c5(h, temp)
    s = np.sum(h) - n / 2.0
    val += penalty * s * s
    return val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, temp):
    c = corr / temp
    c -= np.max(c)
    return float(temp * np.log(np.sum(np.exp(c))))

def c5_smooth(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, temp) * dx

def c5_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cs = corr / temp
    cs -= np.max(cs)
    W = np.exp(cs)
    W /= np.sum(W)
    grad = np.zeros(n)

            ipk2 = j + k
