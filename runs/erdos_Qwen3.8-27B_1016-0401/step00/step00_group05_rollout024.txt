def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0, 1, out=h)
    return h

def cordescent_pass_fast(h, dx, n_grid=101):

    grid = np.linspace(0, 1, n_grid)

        x0 = h[i]
        A = np.zeros(2 * n - 1)
        for d in range(-(n - 1), n):

            if 0 <= i + d < n:
