import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)


class C5Problem:
    def __init__(self, n, target_sum, alpha):
        self.n = n
        self.target_sum = target_sum
        self.alpha = alpha
        self.dx = 2.0 / n

    def objective(self, h):
        corr = np.correlate(h, 1.0 - h, mode="full") * self.dx
        return smooth_max(corr, self.alpha)

    def con_fun(self, h):
        return np.sum(h) - self.target_sum

    def con_jac(self, h):
        return np.ones(self.n)

    def get_constraints(self):
        return [{'type': 'eq', 'fun': self.con_fun, 'jac': self.con_jac}]

    def get_bounds(self):
        return [(0.0, 1.0)] * self.n


def do_slsqp(h_init, n, target_sum, alpha, maxiter):
    prob = C5Problem(n, target_sum, alpha)
    res = minimize(prob.objective, h_init, method='SLSQP',
                   constraints=prob.get_constraints(),
                   bounds=prob.get_bounds(),
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project_feasible(res.x, target_sum)


def local_search(h_best, best_c5, n, target_sum, t_start, budget_s, frac=1.0):
    np.random.seed(int(time.time() * 1000) % 100000)
    h = h_best.copy()
    c = best_c5
    lr = 0.06
    deadline = budget_s * frac
    for it in range(20000):
        if (time.time() - t_start) > deadline:
            break
        k = np.random.randint(1, 6)

        h_t = h.copy()
        h_t[idx] += np.random.uniform(-lr, lr, k)
        h_t = project_feasible(h_t, target_sum)
        c_t = c5_value(h_t)
        if c_t < c:
            h, c = h_t, c_t
            lr = min(lr * 1.05, 0.1)
    return h, c


def binary_push(h, n, target_sum):
    """Push values toward 0 or 1 while preserving sum."""

        h2 = project_feasible(h2, target_sum)
        if np.allclose(h2, h):
            break
        h = h2.copy()
    return h, c5_value(h2)
