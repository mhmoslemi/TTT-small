def _project(h, n):

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-14:

            else:

            h = np.clip(h, 0.0, 1.0)

            break
        if nt <= bn:
            continue
        xo = np.linspace(0, 2, bn, endpoint=False) + 1.0 / bn
        xn = np.linspace(0, 2, nt, endpoint=False) + 1.0 / nt
        hi = np.interp(xn, xo, bh, period=2.0)
        hi = _proj(hi, nt)
        ta = min(35, max(8, (budget_s - elapsed() - 40) * 0.2))
        if ta < 5:
            break
        ht, ct = _optimize(hi, ta)
        if ct < bc:
            bh, bc, bn = ht, ct, nt
    
    # Phase 4: multi-start with perturbations
    t_rem = budget_s - elapsed() - 15
    if t_rem > 40:
        nc = bn
        attempts = 0
        while elapsed() < budget_s - 15 and attempts < 200:
            attempts += 1
            ta = max(4, min(12, (budget_s - elapsed() - 15) / max(1, (200 - attempts) * 0.3)))
            for es in [0.01, 0.03, 0.08]:
                if elapsed() > budget_s - 15:
                    break
                hp = bh + es * np.random.randn(nc)
                hp = _proj(hp, nc)
                ht, ct = _optimize(hp, ta, active_frac=0.1)
                if ct < bc:
                    bh, bc, bn = ht, ct, len(ht)
    
    # Phase 5: final polish
    t_rem = budget_s - elapsed() - 3
    if t_rem > 10 and bn <= 140:
        h5, c5f = _optimize(bh, min(30, t_rem), active_frac=0.08)
        if c5f < bc:
            bh, bc = h5, c5f
        if bn <= 100:
            hs, cs = _scipy_polish(bh, min(15, max(0, budget_s - elapsed() - 2)))
            if cs < bc:
                bh, bc = hs, cs
    
    # Final projection and evaluation
    bh = _proj(bh, bn)
    # Force exact sum
    diff = bn * 0.5 - bh.sum()
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(bh - 0.5))
        bh[idx] += diff
        bh = np.clip(bh, 0.0, 1.0)
    bc = _c5(bh)
    return (bh, bc, bn)
