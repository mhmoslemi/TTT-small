import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(80):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(v, beta):
    m = float(v.max())
    return m + float(np.log(np.mean(np.exp(beta * (v - m))))) / beta

def nm_objective(x, n, dx, beta):
    h = project_feasible(x, n)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(vals, beta)

def direct_objective(x, n, dx):
    h = project_feasible(x, n)
    return compute_c5(h, dx)

            break
