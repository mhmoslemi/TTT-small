    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        found = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best_h, best_score
                oi, oj = h[i], h[j]
                for d in deltas:
                    for sgn in (1.0, -1.0):
                        ni = oi - sgn * d
                        nj = oj + sgn * d
                        if ni < -1e-9 or ni > 1.0 + 1e-9:
                            continue
                        if nj < -1e-9 or nj > 1.0 + 1e-9:
                            continue
                        ni = max(0.0, min(1.0, ni))
                        nj = max(0.0, min(1.0, nj))
                        h[i], h[j] = ni, nj
                        s = _evaluate(h)
                        if s < best_score - 1e-12:
                            best_score = s
                            best_h = h.copy()
                            found = True
                        else:
                            h[i], h[j] = oi, oj
        if not found:
            break
    return best_h, best_score


def _local_random(h, n, rng, deadline, max_rounds=500, samples_per_round=500):
    best_score = _evaluate(h)
    best_h = h.copy()
    deltas = (0.5, 0.3, 0.2, 0.1, 0.05, 0.02)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        found = False
        for _ in range(samples_per_round):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            oi, oj = h[i], h[j]
            for d in deltas:
                for sgn in (1.0, -1.0):
                    ni = oi - sgn * d
                    nj = oj + sgn * d
                    if ni < -1e-9 or ni > 1.0 + 1e-9:
                        continue
                    if nj < -1e-9 or nj > 1.0 + 1e-9:
                        continue
                    ni = max(0.0, min(1.0, ni))
                    nj = max(0.0, min(1.0, nj))
                    h[i], h[j] = ni, nj
                    s = _evaluate(h)
                    if s < best_score - 1e-12:
                        best_score = s
                        best_h = h.copy()
                        found = True
                    else:
                        h[i], h[j] = oi, oj
            if found:
                break
        if not found:
            break
    return best_h, best_score


def _try_binary_round(h, n):
    """Try rounding to binary and adjusting."""
    h_bin = (h >= 0.5).astype(float)
    target = n / 2.0
    s = np.sum(h_bin)
    if s > target:
        excess = int(s - target)
        if excess <= target:
            sorted_idx = np.argsort(h_bin * h)
            for idx in sorted_idx[:excess]:
                h_bin[idx] = 0.0
    elif s < target:
        deficit = int(target - s)
        if deficit <= n - s:
            sorted_idx = np.argsort(-h * (1.0 - h_bin))
            for idx in sorted_idx[:deficit]:
                h_bin[idx] = 1.0
    if abs(np.sum(h_bin) - target) < 1e-9:
        return h_bin
    h_bin = _project(h_bin, n)
    return h_bin


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    best_h = h_init.copy()
    best_score = _evaluate(best_h)

    # Phase 1: Gradient descent (30% budget)
    dl1 = t0 + budget_s * 0.28
    best_h, best_score = _gd_refine(best_h, n, 8.0, 20000, rng, dl1)

    # Try binary rounding
    h_bin = _try_binary_round(best_h, n)
    s_bin = _evaluate(h_bin)
    if s_bin < best_score:
        best_score = s_bin
        best_h = h_bin

    # Phase 2: Exhaustive local search (25%)
    dl2 = t0 + budget_s * 0.55
    best_h, best_score = _local_exhaustive(best_h, n, dl2, max_rounds=50)

    # Phase 3: More GD + local (25%)
    dl3 = t0 + budget_s * 0.82
    for attempt in range(40):
        if time.time() > dl3:
            break
        noise_scale = 0.03 + 0.07 * rng.rand()
        h_try = best_h + rng.randn(n) * noise_scale
        h_try = _project(h_try, n)
        sub_dl = min(dl3, t0 + budget_s * 0.85)
        h_try, _ = _gd_refine(h_try, n, 5.0, 500, rng, sub_dl)
        h_try, s_try = _local_exhaustive(h_try, n, dl3, max_rounds=10)
        if s_try < best_score:
            best_score = s_try
            best_h = h_try.copy()

    # Phase 4: Final local polish (13%)
    dl4 = t0 + budget_s * 0.98
    best_h, best_score = _local_exhaustive(best_h, n, dl4, max_rounds=100)

    # Try binary at the end
    h_bin2 = _try_binary_round(best_h, n)
    s_bin2 = _evaluate(h_bin2)
    if s_bin2 < best_score:
        best_score = s_bin2
        best_h = h_bin2

    # Final safety
    best_h = _project(best_h, n)
    final_score = _evaluate(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, len(best_h)
