```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    if n < 2:
        return 0.0
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr)) * dx

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def proj_sum(h, target):
    """Project h to {0<=h<=1} while making sum(h)=target using bisection."""
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s < target:
            low = mid
        else:
            high = mid
    h_proj = np.clip(h + (low + high) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Distribute remaining diff to elements not at boundary
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
    return np.clip(h_proj, 0.0, 1.0)

def gradient_for_shift(h, shift_idx, n):
    """Compute gradient of overlap for a specific correlate index."""
    # correlate(h, 1-h, mode="full")[shift_idx]
    # = sum_t h[t] * (1-h[t + shift_idx - (n-1)]) for valid t
    lag = shift_idx - (n - 1)
    grad = np.zeros(n)
    # h[t] appears with coefficient (1-h[t+lag]) if t+lag in [0,n)
    # h[t+lag] appears with coefficient -h[t] if t in [0,n)
    # First term: for t in [0,n), t+lag in [0,n)
    t_range = range(max(0, -lag), min(n, n - lag))
    for t in t_range:
        grad[t] += (1.0 - h[t + lag])
        grad[t + lag] += (-h[t])
    return grad

def optimize_h(h_init, n, budget_s, start_time):
    """Optimize h using projected subgradient method."""
    target_sum = n / 2.0
    h = h_init.copy()
    h = proj_sum(h, target_sum)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Adaptive learning rate
    lr = 0.01
    no_improve = 0
    patience = 500
    
    while time.time() - start_time < budget_s - 10:
        overlaps = get_overlaps(h)
        max_idx = np.argmax(overlaps)
        current_c5 = overlaps[max_idx]
        
        if current_c5 < best_c5 - 1e-10:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

            h = best_h + np.random.normal(0, lr * 0.5, n)
            h = proj_sum(h, target_sum)
            no_improve = 0
            lr *= 0.9

        grad = gradient_for_shift(h, max_idx, n)

        h_new = proj_sum(h_new, target_sum)

        new_c5 = compute_c5(h_new)

            lr *= 0.999
        else:
            lr *= 0.95
        
        if lr < 1e-7:
            lr = 1e-7
```

```python
def project_feasible(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum} using bisection on shift."""
    lo, hi = -2.0, 2.0
```

```python
def coordinate_step(h, j, n):
    """Find optimal h[j] using 1D convex PL minimization."""
    current_overlaps = get_all_overlaps(h)  # shape (2n-1,)
    
    n_overlaps = 2 * n - 1
    slopes = np.zeros(n_overlaps)
    
    for k in range(n_overlaps):
        lag = k - (n - 1)
        a = 0.0
        jl = j - lag
        if 0 <= jl < n:
            a += (1.0 - h[jl])
        jr = j + lag
        if 0 <= jr < n:
            a += (-h[jr])
        slopes[k] = a
    
    # b_k = O_k - a_k * h[j]
    intercepts = current_overlaps - slopes * h[j]
    
    # Now find min_{x in [0,1]} max_k (a_k * x + b_k)
    # This is 1D: evaluate at x=0, x=1, and all intersection points in (0,1)
    best_val = float('inf')
    best_x = 0.0
    
    # Check boundaries
    for x_check in [0.0, 1.0]:
        val = np.max(slopes * x_check + intercepts)
        if val < best_val:
            best_val = val
            best_x = x_check
    
    # Check all pairwise intersections
    # For lines i and j: x* = (b_j - b_i) / (a_i - a_j) when a_i != a_j
    # Only check if x* in (0,1) and both lines are at the top at x*
    # For efficiency, just check all O(n^2) pairs and evaluate the max at x*
    for i in range(n_overlaps):
        for l in range(i+1, n_overlaps):
            da = slopes[i] - slopes[l]
            if abs(da) < 1e-15:
                continue
            x_star = (intercepts[l] - intercepts[i]) / da
            if 0 < x_star < 1:
                val = np.max(slopes * x_star + intercepts)
                if val < best_val:
                    best_val = val
                    best_x = x_star
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr)) * dx

def get_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_p < 1.0
        else:
            mask = h_p > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_p = h_p.copy()
            h_p[mask] += diff / cnt
    return np.clip(h_p, 0.0, 1.0)

def subgrad_descent(h, n, start_time, time_limit, seed, top_k=3):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = project_feasible(h.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = 0.05
    patience = 0
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(500000):
        if time.time() - start_time > time_limit:
            break
        
        overlaps = get_all_overlaps(h)
        cur = float(np.max(overlaps))
        
        if cur < best_c5 - 1e-11:
            best_c5 = cur
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Compute weighted subgradient from top-k shifts
        top_k_actual = min(top_k, 2*n-1)
        top_idx = np.argsort(overlaps)[-top_k_actual:]
        weights = np.exp(10.0 * (overlaps[top_idx] - overlaps[top_idx[-1]]))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = idx - (n - 1)
            g_ = np.zeros(n)
            if lag >= 0:
                g_[lag:] += (1.0 - h[:n-lag])
                g_[:n-lag] += -h[lag:]
            else:
                m = -lag
                g_[:n-m] += (1.0 - h[m:])
                g_[m:] += -h[:n-m]
            grad += w * g_
        
        # Momentum
        momentum = mu * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < cur:
            h = h_new
        else:
            # Still take the step (for exploration) but with smaller effective step
            h = 0.5 * h + 0.5 * h_new
            h = project_feasible(h, target)
        
        # Learning rate
        if patience > 500:
            lr *= 0.7
            momentum *= 0.5
            patience = 0
            if lr < 0.001:
                # Restart with perturbation
                h = best_h + rng.normal(0, 0.03, n)
                h = project_feasible(h, target)
                momentum = np.zeros(n)
                lr = 0.02
        else:
            lr *= 0.9997
    
    return best_h, best_c5

def coordinate_descent(h, n, start_time, time_limit, seed, n_sweeps=50):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = project_feasible(h.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

        if time.time() - start_time > time_limit:
            break

        for j in order:
            if time.time() - start_time > time_limit:
                break

            overlaps = get_all_overlaps(h)
            n_ov = 2 * n - 1

                lag = k - (n - 1)
                a = 0.0
                jl = j - lag
                if 0 <= jl < n:
                    a += 1.0 - h[jl]
                jr = j + lag
                if 0 <= jr < n:
                    a += -h[jr]
                slopes[k] = a
            intercepts = overlaps - slopes * h[j]
            
            # Find min of max_k (a_k*x + b_k) over x in [0,1]
            best_x = h[j]
            best_val = float(np.max(slopes * h[j] + intercepts))
            
            # Check boundaries and intersections
            candidates = [0.0, 1.0]
            for i in range(n_ov):
                for l in range(i+1, n_ov):

                    if abs(da) > 1e-12:

                    best_x = xc

            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
jl_idx = np.where(jl_valid, j - lags, 0)
jr_idx = np.where(jr_valid, j + lags, 0)

slopes = np.where(jl_valid, 1.0 - h[jl_idx], 0.0) + np.where(jr_valid, -h[jr_idx], 0.0)
```

```python
    h_cd = best_h if len(best_h) == n1 else project_feasible(upsample(best_h, n1), n1 / 2.0)
    h, c = coordinate_descent(h_cd, n1, start, t4, seed + 3, n_sweeps=200)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 5: Try 4*n0 if time allows
    n2 = n0 * 4
    if time.time() - start < budget_s * 0.68:
        h_up2 = project_feasible(upsample(best_h, n2), n2 / 2.0)
        t5 = start + min(budget_s - 80, 850)
        h, c = subgrad_descent(h_up2, n2, start, t5, seed + 4, top_k=5)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 6: Final polish with coordinate descent
    if time.time() - start < budget_s - 40:
        nf = len(best_h)
        t6 = start + budget_s - 10
        h, c = coordinate_descent(best_h, nf, start, t6, seed + 5, n_sweeps=100)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
