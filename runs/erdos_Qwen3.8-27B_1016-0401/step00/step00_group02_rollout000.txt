    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 1e-14:
            mask = h < 1.0 - 1e-14

    h[0] += target - s
    h = np.clip(h, 0.0, 1.0)
