import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

def _lse_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    ev = np.exp(beta * (corr - m))
    return m + np.log(ev.sum()) / beta

def _lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    ev = np.exp(beta * (corr - m))
    z = ev.sum()
    w = ev / z
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-17:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        for i in range(lo, hi):
            j = i + s
            g[j] += wk * (1.0 - x[i])
            g[i] -= wk * x[j]
    return g * dx

def _cfn(x):
    return np.sum(x) - _g_target

def _cjn(x):
    return np.ones(len(x))

def _project(h, target):
    lo_lam = -3.0
    hi_lam = 3.0
    for _ in range(100):
        mid = (lo_lam + hi_lam) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo_lam = mid
        else:
            hi_lam = mid
    return np.clip(h - (lo_lam + hi_lam) * 0.5, 0.0, 1.0)

def _pattern(n, p1, p0):
    p = p1 + p0
    h = np.array([1.0 if (i % p) < p1 else 0.0 for i in range(n)])
    return h

def _tm(n):
    h = np.zeros(n)
    for i in range(n):
        v, c = i, 0
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 0 else 0.0
    return h

    global _g_target
    _g_target = target
    try:
        res = minimize(_lse_obj, h_init, args=(beta,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(res.x, target)
    except Exception:
        return h_init
