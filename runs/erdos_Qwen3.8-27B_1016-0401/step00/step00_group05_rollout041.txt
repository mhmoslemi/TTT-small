def subgradient_step(h, dx, lr, n):
    """One step of projected subgradient descent on the max objective"""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c5 = np.max(corr) * dx
    
    # Identify active lags (within some tolerance of max)
    threshold = np.max(corr) - 0.01 * np.max(corr)
    active = corr >= threshold
    
    # Compute subgradient
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        k = s + (n-1)  # index in correlation array
        if not active[k]:
            continue
        # For shift s: c[s+1] = sum_j h[j] * (1-h[j+s]) for valid j
        j_min = max(0, -s)
        j_max = min(n-1, n-1-s)
        for j in range(j_min, j_max+1):
            j2 = j + s
            if 0 <= j2 < n:
                # h[j] * (1-h[j+s]): gradient wrt h[j] is (1-h[j+s])
                grad[j] += (1 - h[j2]) * active[k]
                # gradient wrt h[j+s] is -h[j]
                grad[j2] += -h[j] * active[k]
    
    grad *= dx
    grad /= len(np.where(active)[0])
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    h_new += (n/2.0 - np.sum(h_new)) / n
    h_new = np.clip(h_new, 0, 1)
    
    return h_new, c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return np.max(compute_correlation(h)) * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, dx, temp):
    corr = compute_correlation(h) * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def subgradient_step(h, n, lr, temp):
    corr = compute_correlation(h)
    cmax = np.max(corr)
    weights = np.exp((corr - cmax) / (0.05 * n + 1e-10))
    sw = np.sum(weights)
    if sw < 1e-30:
        return h, cmax
    weights /= sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        j_min = max(0, -s)
        j_max = min(n - 1, n - 1 - s)
        js = np.arange(j_min, j_max + 1)
        js2 = js + s
        grad[js] += w * (1.0 - h[js2])
        grad[js2] -= w * h[js]
    h_new = h - lr * grad
    h_new = project_feasible(h_new)
    return h_new, cmax
