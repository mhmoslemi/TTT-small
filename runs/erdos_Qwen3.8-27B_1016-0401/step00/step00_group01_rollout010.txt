import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def project_all(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > target:
            hi = mid
        else:
            lo = mid
    p = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - p.sum()
    free = (p > 1e-6) & (p < 1.0 - 1e-6)
    if free.sum() > 0:
        p[free] += diff / free.sum()
        p = np.clip(p, 0.0, 1.0)
    return p

def smooth_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    m = vals.max()
    obj = m + np.log(np.sum(np.exp((vals - m) * alpha))) / alpha
    w = np.exp((vals - m) * alpha)

                g += 1.0 - h[i + shift]
            if 0 <= i - shift < n:
                g -= h[i - shift]
            grad[i] += w[k] * g
    return obj, grad
