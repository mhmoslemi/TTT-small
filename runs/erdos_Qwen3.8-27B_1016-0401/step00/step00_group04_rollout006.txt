import time

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def c5_score(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Find lambda via bisection
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    p = np.clip(h - lam, 0.0, 1.0)

    residual = target - p.sum()
    if abs(residual) > 1e-12:

            p = np.clip(p, 0.0, 1.0)

    g = np.zeros(n)
    for i in range(n):

        j = i - d
        if 0 <= j < n:
            g[i] += 1.0 - h[j]
        # Second factor: -h[i+d] if i+d in [0,n-1]
        j = i + d
        if 0 <= j < n:
            g[i] -= h[j]

        # Start from best interpolated
        if best_h is not None and best_n != ni:
            h_start = project_feasible(interpolate_h(best_h, best_n, ni), ni)
        else:
            h_start = np.zeros(ni)
            h_start::2 = 1.0
            h_start = project_feasible(h_start, ni)
        h_opt, s_opt = optimize_n(h_start, ni, budget_here, rng)
        if s_opt < best_s:
            best_h, best_s, best_n = h_opt, s_opt, ni

    # Phase 3: golden-ratio start
    for ni in [best_n, best_n + 2]:
        if time.time() - t0 > budget_s - 30:
            break
        h_gold = np.zeros(ni)
        for i in range(ni):
            frac = (i * phi) % 1.0
            h_gold[i] = 1.0 if frac < 0.5 else 0.0
        h_gold = project_feasible(h_gold, ni)
        h_opt, s_opt = optimize_n(h_gold, ni, 100, rng)
        if s_opt < best_s:
            best_h, best_s, best_n = h_opt, s_opt, ni

    # Phase 4: final refinement at best_n
    if best_h is not None:
        rem = budget_s - (time.time() - t0) - 5
        if rem > 20:
            h_opt, s_opt = optimize_n(best_h, best_n, rem, rng)
            if s_opt < best_s:
                best_h, best_s, best_n = h_opt, s_opt, ni

    # Phase 5: try slightly larger n
    if best_h is not None and time.time() - t0 < budget_s - 30:
        for dn in [2, 4, 6]:
            nt = best_n + dn
            if nt > 250 or time.time() - t0 > budget_s - 15:
                break
            h_int = project_feasible(interpolate_h(best_h, best_n, nt), nt)
            h_opt, s_opt = optimize_n(h_int, nt, 80, rng)
            if s_opt < best_s:
                best_h, best_s, best_n = h_opt, s_opt, nt

    # Final verification
    if best_h is None:
        n_def = 42
        best_h = np.zeros(n_def)
        best_h::2 = 1.0
        best_h = project_feasible(best_h, n_def)
        best_s = c5_score(best_h, n_def)
        best_n = n_def

    return best_h, best_s, best_n
