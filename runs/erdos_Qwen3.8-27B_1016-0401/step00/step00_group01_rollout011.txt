    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for beta in betas:
        stuck = 0
        for it in range(iters_per_beta):
            if time.time() > deadline:
                return best_h, best_c5
            _, g = lse_obj_grad(h, beta)
            cur_c5 = compute_c5(h)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                stuck = 100
                break
            improved = False
            lr_t = lr
            for _ in range(10):
                h_cand = project_box_sum(h - lr_t * g, S)
                c5_cand = compute_c5(h_cand)
                if c5_cand < cur_c5 - 1e-14:
                    h = h_cand
                    cur_c5 = c5_cand
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
                lr_t *= 0.5
                if lr_t < 1e-9:
                    break
            if improved:
                lr = min(lr * 1.08, 10.0)
                stuck = 0
            else:
                lr = max(lr * 0.4, 1e-7)
                stuck += 1
                if stuck > 80:
                    lr = lr0 * 0.3
                    h = project_box_sum(h + 0.05 * (np.random.RandomState(it).randn(n)), S)
                    stuck = 0
    return best_h, best_c5

def random_feasible(n, rng):
    h = rng.rand(n)
    return project_box_sum(h, n / 2.0)

def block_init(n, rng):
    h = np.zeros(n)
    nb = max(2, n // 10)
    for _ in range(2 * nb):
        s = rng.randint(0, n)
        l = max(1, n // (2 * nb) + rng.randint(-1, 2))
        h[s:min(n, s + l)] = 1.0
    return project_box_sum(h, n / 2.0)

def upsample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    deadline = t_start + budget_s

    best_h = None
    best_c5 = 1.0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        if n0 < 2:
            raise ValueError
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)
    except Exception:
        n0 = 47
        best_h = random_feasible(n0, rng)
        best_c5 = compute_c5(best_h)

    # Phase 1: LSE PGD from initial (45% budget)
    dl1 = t_start + budget_s * 0.45
    best_h, best_c5 = lse_pgd(best_h, [5, 15, 40, 100, 200], 400, 0.2, t_start, dl1)

    # Phase 2: restarts at various n (35% budget)
    dl2 = t_start + budget_s * 0.80
    n_opts = [n0, n0 + 1 if n0 % 2 == 0 else n0 - 1, max(50, n0 * 2), 120, 200]
    for r, n_try in enumerate(n_opts):
        if time.time() > dl2:
            break
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 10:
            n_try = 10
        if r == 0:
            h_try = block_init(n_try, rng)
        elif r == 1:
            h_try = random_feasible(n_try, rng)
        elif r == 2:
            h_try = upsample(best_h, n_try)
            h_try = project_box_sum(h_try, n_try / 2.0)
        elif r == 3:
            h_try = block_init(n_try, rng)
        else:
            h_try = random_feasible(n_try, rng)
        h_try = project_box_sum(h_try, n_try / 2.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
        h_opt, c5_opt = lse_pgd(h_try, [10, 40, 120], 300, 0.1, t_start, dl2 - (time.time() - t_start) * 0.1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: final high-beta refinement (20% budget)
    if time.time() < deadline - 5 and best_h is not None:
        n_best = len(best_h)
        best_h, best_c5 = lse_pgd(best_h, [100, 300, 800], 800, 0.01, t_start, deadline - 2)

    # Final sanity check and exact projection
    n_f = len(best_h)
    best_h = project_box_sum(best_h, n_f / 2.0)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_f)
