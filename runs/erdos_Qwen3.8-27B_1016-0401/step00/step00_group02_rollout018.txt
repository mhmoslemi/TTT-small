    for _ in range(4):

    return noise

def block_initial(rng, n):
    h0 = np.zeros(n)
    nblocks = rng.randint(3, 10)
    for b in range(nblocks):
        s = int(b * n / nblocks)
        e = int((b + 1) * n / nblocks)
        h0[s:e] = rng.uniform(0.15, 0.85)
    return h0

def sinusoidal_initial(rng, n):
    freq = rng.uniform(1.0, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n) / n
    h0 = 0.5 + 0.25 * np.sin(2 * np.pi * freq * x + phase)
    h0 += rng.randn(n) * 0.03
    return h0

def try_polish(h, bounds, cons, iter_per_call, tol):
    res = minimize(compute_c5, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': iter_per_call, 'ftol': tol})
    return res

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    h_best = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            h_best = np.array(initial_h_values, dtype=float).copy()
        except Exception:
            h_best = None
    
    if h_best is None or len(h_best) == 0:
        h_best = np.ones(45) * 0.5
    
    n = len(h_best)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Optimize from initial
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    
    if time.time() - t0 < budget_s * 0.6:
        res = try_polish(h_best, bounds, cons, 1500, 1e-10)
        if res.fun < c5_best:
            h_best = res.x.copy()
            c5_best = res.fun
            n = len(h_best)
    
    # Phase 2: Multiple restarts with diverse inits
    n_restarts = 25
    for attempt in range(n_restarts):
        if time.time() - t0 > budget_s * 0.85:
            break
        
        if attempt == 0:
            h0 = h_best + smooth_noise(rng, n, 0.04)
        elif attempt % 3 == 1:
            h0 = block_initial(rng, n)
        elif attempt % 3 == 2:
            h0 = sinusoidal_initial(rng, n)
        else:
            h0 = h_best + smooth_noise(rng, n, 0.08)
            h0 = 0.5 * h0 + 0.5 * np.flip(h0)
        
        h0 = np.clip(h0, 0, 1)
        h0 = fix_sum(h0)
        h0 = np.clip(h0, 0, 1)
        
        bounds_a = [(0.0, 1.0)] * n
        cons_a = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
        
        res = try_polish(h0, bounds_a, cons_a, 600, 1e-9)
        if res.fun < c5_best:
            h_best = res.x.copy()
            c5_best = res.fun
            n = len(h_best)
    
    # Phase 3: Try different resolutions
    for n_try in [55, 65, 80]:
        if time.time() - t0 > budget_s * 0.93:
            break
        
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, h_best)
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        bounds_b = [(0.0, 1.0)] * n_try
        cons_b = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
        
        res = try_polish(h_new, bounds_b, cons_b, 800, 1e-9)
        if res.fun < c5_best:
            h_best = res.x.copy()
            c5_best = res.fun
            n = n_try
    
    # Phase 4: Fine-tune with coordinate descent
    if time.time() - t0 < budget_s * 0.96:
        for _ in range(50):
            if time.time() - t0 > budget_s * 0.96:
                break
            improved = False
            for i in range(n):
                if time.time() - t0 > budget_s * 0.96:
                    break
                old = h_best[i]
                best_val = compute_c5(h_best)
                for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    new = old + delta
                    if new < 0 or new > 1:
                        continue
                    h_try = h_best.copy()
                    h_try[i] = new
                    h_try = fix_sum(h_try)
                    h_try = np.clip(h_try, 0, 1)
                    val = compute_c5(h_try)
                    if val < best_val:
                        best_val = val
                        h_best = h_try.copy()
                        improved = True
                        break
            if not improved:
                break
    
    # Phase 5: Final high-precision polish
    if time.time() - t0 < budget_s * 0.99:
        n = len(h_best)
        bounds_c = [(0.0, 1.0)] * n
        cons_c = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
        res = try_polish(h_best, bounds_c, cons_c, 3000, 1e-12)
        if res.fun < c5_best:
            h_best = res.x.copy()
            c5_best = res.fun
    
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
