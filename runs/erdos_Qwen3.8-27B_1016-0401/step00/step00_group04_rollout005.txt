    np.clip(h0, 0, 1, out=h0)

    h0 += diff / n

        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        # Softmax approximation of max
        m = np.max(corr)
        return T_smooth * np.log(np.sum(np.exp((corr - m) / T_smooth))) * dx + m * dx

        f0 = obj(x)

            g[i] = (obj(x) - f0) / eps
