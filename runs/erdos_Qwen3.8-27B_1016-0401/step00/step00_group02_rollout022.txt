    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, new_n)
    new_h = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    new_h = np.clip(new_h, 0, 1)
    # Adjust to meet constraint
    diff = target - np.sum(new_h)

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    x_old = np.arange(len(h), dtype=float)
    x_new = np.linspace(0, len(h) - 1, new_n)
    new_h = np.interp(x_new, x_old, h)
    return new_h

    n = len(h)
    h = h.astype(float).copy()
    curr = eval_c5(h)
    best = curr
    best_h = h.copy()
    T = 0.003
    T_floor = 1e-5
    cool = 0.9998
    t0 = time.time()
    no_improve = 0
    threshold = 100000

        j = rng.randint(n)
        if i == j:
            continue

                best_h = h.copy()
                no_improve = 0

        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        for _ in range(200):
            if time.time() - t0 > max_time:
                break
            i = one_idx[rng.randint(len(one_idx))]
            j = zero_idx[rng.randint(len(zero_idx))]
            h[i] = 0
            h[j] = 1
            new = eval_c5(h)
            if new < curr:
                curr = new
                if new < best:
                    best = new
                    best_h = h.copy()
                improved = True
            else:
                h[i] = 1
                h[j] = 0
        if not improved:
            # Random perturbation: swap 3 random pairs
            for _ in range(3):
                one_idx = np.where(h == 1)[0]
                zero_idx = np.where(h == 0)[0]
                i = one_idx[rng.randint(len(one_idx))]
                j = zero_idx[rng.randint(len(zero_idx))]
                h[i] = 0
                h[j] = 1
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    s = np.sum(h0)
    if s > 1e-10:
        h0 *= (n0 / 2.0) / s
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = eval_c5(h0)

    # Stage 1: continuous SA at original n (~100s)
    remaining = budget_s - (time.time() - t0)
    t1 = min(100.0, remaining * 0.10)
    if remaining > t1 + 300:
        h1, c1 = sa_continuous(h0, t1, seed=seed)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Stage 2: binary SA at n=90 with multiple restarts
    n2 = 90
    remaining = budget_s - (time.time() - t0)
    if remaining > 250:
        n_restarts = 3
        per = min(140.0, (remaining - 200.0) / max(n_restarts, 1))
        if per > 20:
            for r in range(n_restarts):
                if time.time() - t0 > budget_s * 0.60:
                    break
                if r == 0:
                    start_h = upsample(best_h, n2)
                else:
                    rr = np.random.RandomState(seed + 100 + r * 7)
                    start_h = (rr.random(n2) < 0.5).astype(float)
                hb, cb = sa_binary(start_h, per, seed=seed + 10 + r * 3)
                if cb < best_c5:
                    best_c5 = cb
                    best_h = hb.copy()
                # Quick greedy refine on best
                if time.time() - t0 < budget_s * 0.58:
                    hb2, cb2 = greedy_binary_refine(best_h, min(30.0, per * 0.25), seed=seed + 50 + r)
                    if cb2 < best_c5:
                        best_c5 = cb2
                        best_h = hb2.copy()

    # Stage 3: binary SA at n=180
    n3 = 180
    remaining = budget_s - (time.time() - t0)
    if remaining > 130:
        n_restarts = 2
        per = min(180.0, (remaining - 40.0) / max(n_restarts, 1))
        if per > 25:
            for r in range(n_restarts):
                if time.time() - t0 > budget_s * 0.93:
                    break
                if r == 0:
                    start_h = upsample(best_h, n3)
                else:
                    rr = np.random.RandomState(seed + 300 + r * 11)
                    start_h = (rr.random(n3) < 0.5).astype(float)
                hb, cb = sa_binary(start_h, per, seed=seed + 30 + r * 3)
                if cb < best_c5:
                    best_c5 = cb
                    best_h = hb.copy()

    # Final greedy polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10 and len(best_h) % 2 == 0:
        hb, cb = greedy_binary_refine(best_h, min(50.0, remaining - 5), seed=seed + 999)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()

    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
