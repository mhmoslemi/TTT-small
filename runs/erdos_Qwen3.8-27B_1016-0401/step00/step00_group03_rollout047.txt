import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def corr_and_max_lag(h):
    n = len(h)
    corr = correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(corr))
    return corr, s_star

def grad_for_lag(h, k):
    """Gradient of corr at lag k (where k = s - n + 1) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    j = np.arange(n)
    jmk = j - k
    jpk = j + k
    v1 = (jmk >= 0) & (jmk < n)
    v2 = (jpk >= 0) & (jpk < n)
    grad[v1] += (1.0 - h[jmk[v1]])
    grad[v2] -= h[jpk[v2]]
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

        c = correlate(x, 1.0 - x, mode='full')
        return float(np.max(c)) * (2.0 / len(x))
    
    bounds = [(0.0, 1.0)] * n

def compute_c5(h):

def grad_for_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    j = np.arange(n)
    jmk = j - k
    jpk = j + k
    v1 = (jmk >= 0) & (jmk < n)

    grad[v1] += (1.0 - h[jmk[v1]])
    grad[v2] -= h[jpk[v2]]
    return grad


def soft_max_grad_total(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / T)
    wsum = w.sum()
    if wsum < 1e-300:
        s_star = int(np.argmax(corr))
        k = s_star - (n - 1)
        return grad_for_lag(h, k) * dx
    w /= wsum
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:

        grad += w[s] * grad_for_lag(h, k)
