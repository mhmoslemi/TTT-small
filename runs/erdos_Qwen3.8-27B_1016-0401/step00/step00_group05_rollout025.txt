        j1 = min(n - 1, n - 1 - s)
        if j0 <= j1:
            grad[j0:j1 + 1] += wi * (1.0 - h[j0 + s:j1 + s + 1])
        k0 = max(0, s)
        k1 = min(n - 1, n - 1 + s)
        if k0 <= k1:
            grad[k0:k1 + 1] -= wi * h[k0 - s:k1 - s + 1]
    return grad * dx

def project_feasible(p, S):
    n = len(p)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(p - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(p - (lo + hi) * 0.5, 0, 1)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-8
            cnt = float(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
                np.minimum(h, 1.0, out=h)
        else:
            mask = h > 1e-8
            cnt = float(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
                np.maximum(h, 0.0, out=h)
    return h

def optimize_from(h0, n, T_init, budget, seed=0, binarize=False):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    h = h0.copy()
    S = n / 2.0
    lr = 0.05
    T = T_init
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    stuck = 0

    while time.time() - t_start < budget:
        if time.time() - t_start > budget:
            break
        grad = smooth_grad(h, T)
        if binarize:
            grad -= 0.3 * (1.0 - 2.0 * h)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, S)
        c5 = evaluate_c5(h_new)

        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h_new.copy()
            stuck = 0
            h = h_new
            lr = min(0.12, lr * 1.01)
        else:
            stuck += 1
            if stuck % 30 == 0:
                lr = max(0.005, lr * 0.6)
                h = h + 0.08 * lr * 20 * rng.randn(n)
                h = project_feasible(h, S)
            else:
                h = h_new

        T = min(80.0, T + 0.05)
        if T > 80:
            T = 80

    return best_h, best_c5

def make_start(n, kind, seed):
    rng = np.random.RandomState(seed)
    if kind == 0:
        x = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        x[idx] = 1.0
        return x
    if kind == 1:
        t = np.linspace(0, 1, n, endpoint=False)
        freq = rng.choice([1, 2, 3])
        phase = rng.randn() * 2
        return 0.5 + 0.45 * np.sin(2 * np.pi * freq * t + phase)
    if kind == 2:
        x = rng.randn(n)
        kernel = np.ones(7) / 7.0
        return np.convolve(x, kernel, mode='same')
    if kind == 3:
        block = n // 4
        x = np.zeros(n)
        x[0:block] = 1.0
        x[2*block:3*block] = 1.0
        return x + 0.1 * rng.randn(n)
    x = rng.rand(n)
    x[x < 0.5] = 0.0
    x[x >= 0.5] = 1.0
    return x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    # Get initial construction
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=np.float64)
    else:
        h_init = np.ones(70) * 0.5
    n_init = len(h_init)
    S_init = n_init / 2.0
    h_init = project_feasible(h_init, S_init)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    best_n = n_init

    # Phase 1: Optimize from initial (35% budget)
    b1 = max(10, (t0 + budget_s * 0.35) - time.time())
    h1, c5_1 = optimize_from(h_init, n_init, 25.0, b1, seed=seed, binarize=False)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n_init

    # Phase 2: Diverse restarts (45% budget)
    n_restarts = 6
    t_end2 = t0 + budget_s * 0.80
    rb = max(5, (t_end2 - time.time()) / n_restarts)
    for r in range(n_restarts):
        if time.time() > t_end2:
            break
        start_kind = r % 5
        h_r = make_start(n_init, start_kind, seed + 1000 + r)
        if r >= 3:
            h_r = 0.6 * best_h + 0.4 * h_r
        h_r = project_feasible(h_r, S_init)
        T_r = 20.0 + 5.0 * r
        use_bin = (r >= 4)
        hr, cr = optimize_from(h_r, n_init, T_r, rb, seed=seed + 2000 + r, binarize=use_bin)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, n_init

    # Phase 3: Fine-tune with high T (20% budget)
    t_end3 = t0 + budget_s - 5
    b3 = max(5, t_end3 - time.time())
    hf, cf = optimize_from(best_h, best_n, 50.0, b3, seed=seed + 5000, binarize=True)
    if cf < best_c5:
        best_h, best_c5 = hf, cf

    # Final feasibility check
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
