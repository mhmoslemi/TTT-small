            best_c5 = c5_new
            best_h = h_new

    # Phase 2: L-BFGS-B with penalty (faster per iteration)
    for beta in [50.0, 100.0, 300.0]:
        if time.time() - t0 > budget_s * 0.45:
            break
        for lam in [1.0, 10.0, 50.0]:
            if time.time() - t0 > budget_s * 0.5:
                break
            res = minimize(obj_pen_grad, best_h, args=(beta, lam), method='L-BFGS-B',
                           bounds=bounds0, options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_h(res.x, target0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 3: Random restarts
    for attempt in range(12):
        if time.time() - t0 > budget_s * 0.6:
            break
        noise_scale = 0.02 + 0.01 * attempt
        h_start = best_h + rng.normal(0, noise_scale, n0)
        h_start = project_h(h_start, target0)
        beta_r = 50.0 + 30.0 * (attempt % 5)
        res = minimize(obj_grad, h_start, args=(beta_r,), method='SLSQP',
                       bounds=bounds0, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-12})
        h_new = project_h(res.x, target0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Different resolutions
    for n_try in [50, 60, 80, 100, 140, 200, 350]:
        if time.time() - t0 > budget_s * 0.8:
            break
        target_n = n_try / 2.0
        bounds_n = [(0.0, 1.0)] * n_try
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_start = np.interp(x_new, x_old, best_h)
        h_start = project_h(h_start, target_n)

        for beta in [100.0, 300.0]:
            if time.time() - t0 > budget_s * 0.85:
                break
            res = minimize(obj_pen_grad, h_start, args=(beta, 50.0),
                           method='L-BFGS-B', bounds=bounds_n,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_h(res.x, target_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_start = h_new

    # Phase 5: Projected subgradient with momentum
    if time.time() - t0 < budget_s * 0.95:
        n_b = len(best_h)
        target_b = n_b / 2.0
        h_cur = best_h.copy()
        v = np.zeros(n_b)
        mu = 0.9
        lr = 0.003
        for it in range(2000):
            if time.time() - t0 > budget_s * 0.98:
                break
            _, grad = c5_softmax_grad(h_cur, 500.0)
            v = mu * v + (1 - mu) * grad
            h_new = h_cur - lr * v
            h_new = project_h(h_new, target_b)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_cur = h_new
                lr = min(lr * 1.01, 0.015)
            else:
                lr *= 0.92

    # Final: try SLSQP one more time on best
    if best_h is not None and time.time() - t0 < budget_s * 0.99:
        n_f = len(best_h)
        b_f = [(0.0, 1.0)] * n_f
        res = minimize(obj_grad, best_h, args=(1000.0,), method='SLSQP',
                       bounds=b_f, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        h_new = project_h(res.x, n_f / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    return best_h, best_c5, len(best_h)
