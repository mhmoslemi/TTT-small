import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, temp):
    m = np.max(corr)
    if m < -1e15:
        return 0.0
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
            continue
        step = diff / nf
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, dx, target_sum, max_iter, time_limit, seed=0, binary=True):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5 = compute_c5(h, dx)
    t_start = time.time()
    if binary:
        ones = np.where(h > 0.5)[0].tolist()
        zeros = np.where(h < 0.5)[0].tolist()
    else:
        ones = zeros = None

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        if binary:
            if not ones or not zeros:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, dx)
            if new_c5 < c5 - 1e-14:
                c5 = new_c5
                ones[i], ones[j] = j, i
                zeros[i], zeros[j] = j, i
            else:
                h[i], h[j] = h[j], h[i]
        else:
            i = rng.randint(n)
            old = h[i]
            delta = rng.uniform(-0.15, 0.15)
            h[i] = np.clip(old + delta, 0.0, 1.0)
            h = project_sum(h, target_sum)
            new_c5 = compute_c5(h, dx)
            if new_c5 < c5 - 1e-14:
                c5 = new_c5
            else:
                h[i] = old
                h = project_sum(h, target_sum)
    return h, c5
