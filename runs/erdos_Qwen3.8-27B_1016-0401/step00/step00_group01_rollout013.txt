def project_sum(h, target):
    """Adjust h to have the correct sum while staying in [0,1]."""
    h = h.copy()
    diff = target - np.sum(h)

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_sum(h, target):
    h = h.copy()
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:

        h[mask] += diff / cnt
