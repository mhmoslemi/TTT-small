def project_sum_box(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Find lambda such that sum(clamp(h + lambda, 0, 1)) = target_sum
    lo, hi = -h.max() + target_sum/n - h.min() + 1, h.max() + 1
    # Actually simpler: lambda ranges from -max(h) to 1-min(h)
    lo = -h.max()
    hi = 1.0 - h.min()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)/2, 0, 1)
