import numpy as np

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        err = target - h.sum()
        if abs(err) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + err / n, 0, 1)
        else:
            h[free] += err / free.sum()
            h = np.clip(h, 0, 1)
    # Final correction
    h = np.clip(h, 0, 1)
    err = target - h.sum()
    if abs(err) > 1e-10:
        idx = np.argsort(h)
        if err > 0:

import numpy as np

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(100):
        err = target - h.sum()
        if abs(err) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += err / free.sum()
        else:
            h += err / n
        h = np.clip(h, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-11:
        if err > 0:
            for i in range(n - 1, -1, -1):
                room = 1.0 - h[i]
                if room > 1e-12:
                    a = min(room, err)
                    h[i] += a
                    err -= a
                    if err < 1e-13:
                        break
        else:
            for i in range(n):
                room = h[i]
                if room > 1e-12:
                    a = min(room, -err)
                    h[i] -= a
                    err += a
                    if err > -1e-13:
                        break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)
