- Term where i+k=j, i.e., i=j-k: gives -h[j-k] if j-k is valid

For the max, the subgradient is the gradient of the active (maximizing) correlation.

For the soft max: gradient is (1/α) * Σ_k exp(α*c_k) * ∇c_k / Σ_k exp(α*c_k)

This is a weighted average of the individual gradients, weighted by softmax(α*c_k).

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_step(h, n, alpha, lr):
    corr = np.correlate(h, 1.0 - h, mode="full")
    ac = alpha * corr
    ac -= np.max(ac)
    w = np.exp(ac)
    sw = np.sum(w)
    w = w / sw
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        if k >= 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n - kk])
            grad[:n - kk] -= wk * h[kk:]

    h_new = project_feasible(h_new, n)

    return (best_h, best_c5, n_final)
