import numpy as np
import time
from scipy.optimize import minimize

def evaluate_h(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr * dx), corr * dx

def repair(h, n):
    """Ensure h satisfies constraints: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target) < 1e-10:
        return h
    # Adjust
    diff = target - current_sum
    # Distribute adjustment
    if diff > 0:
        # Need to add value, prefer adding to positions with h < 1
        idx = np.argsort(h)  # smallest first
        for i in idx:
            if diff <= 0:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        # Need to subtract value, prefer subtracting from positions with h > 0
        idx = np.argsort(-h)  # largest first
        for i in idx:
            if diff >= 0:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def binary_local_search(h, n, improvements=100):
    """Local search by swapping 0s and 1s in binary h."""
    h = np.array(h, dtype=float)
    # Make it binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    c5, _ = evaluate_h(h)
    
    for _ in range(improvements):
        improved = False
        best_improve = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5, _ = evaluate_h(h)
                improve = c5 - new_c5
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
                h[i] = 1
                h[j] = 0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c5 = c5 - best_improve
            improved = True
        if not improved:
            break
    return h, c5

def simulated_annealing(h, n, start_temp=1.0, end_temp=0.01, steps=10000):
    """Simulated annealing on binary h."""
    h = np.array(h, dtype=float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    c5, _ = evaluate_h(h)
    best_h = h.copy()
    best_c5 = c5
    
    for step in range(steps):
        temp = start_temp * (end_temp / start_temp) ** (step / steps)
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:

        h[i] = 0; h[j] = 1
        new_c5, _ = evaluate_h(h)

        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):

            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    return best_h, best_c5

    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    return corr

def local_search_binary(h, max_iters=5000, patience=200):
    """Greedy local search on binary h by single swaps."""
    h = h.copy()
    n = len(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    c5 = np.max(fast_correlate(h))
    best_c5 = c5
    no_improve = 0

        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:

                new_c5 = np.max(fast_correlate(h))

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def c5_raw(h):
    """Return max of raw correlation (without dx)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))

def ls_binary(h, max_iters=5000):
    """Local search: try all single 0-1 swaps, take best improvement."""
    h = np.round(h).astype(np.float64)
    n = len(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    c5 = c5_raw(h)
    
    for _ in range(max_iters):
        best_d = 0.0
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nc5 = c5_raw(h)
                d = c5 - nc5
                if d > best_d:
                    best_d = d
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        oi = ones.index(best_i)
        ones[oi] = best_j
        oz = zeros.index(best_j)
        zeros[oz] = best_i
        c5 -= best_d
    
    return h, c5_raw(h)

def sa_binary(h, steps=10000, t0=0.3, t1=0.003, seed=0):
    """Simulated annealing with single-swap moves."""
    rng = np.random.default_rng(seed)
    h = np.round(h).astype(np.float64)
    n = len(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    c5 = c5_raw(h)
    best_h = h.copy()
    best_c5 = c5
    
    for s in range(steps):
        t = t0 * (t1 / t0) ** (s / max(steps - 1, 1))
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        nc5 = c5_raw(h)
        d = nc5 - c5
        if d < 0 or rng.random() < np.exp(-d / max(t, 1e-15)):
            c5 = nc5
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, c5_raw(best_h)
