    n = len(h)
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    # Smoothed max
    beta_c = beta * c
    max_bc = np.max(beta_c)
    exp_bc = np.exp(beta_c - max_bc)
    w = exp_bc / np.sum(exp_bc)  # softmax weights
    
    # Gradient: sum_m w[m] * grad_c[m]
    grad = np.zeros(n)
    for m in range(2*n-1):
        if w[m] < 1e-10:
            continue
        delta = m - n + 1
        # valid i range
        i_start = max(0, -delta)
        i_end = min(n-1, n-1-delta)
        i_range = range(i_start, i_end+1)
        for i in i_range:
            grad[i] += w[m] * (1 - h[i + delta])
            grad[i + delta] -= w[m] * h[i]
    
    return grad, np.max(c)

    # Phase 2: Directed local search using gradient info
    if time.time() - t0 < budget_s - 60:
        h = best_h.copy()
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        g = _grad_smoothed(h, np.ones(2 * n - 1) / (2 * n - 1), n)
        g -= g.mean()
        order = np.argsort(np.abs(g))[::-1]
        for it in range(100000):
            if time.time() - t0 > budget_s - 60:
                break
            si = order[rng.randint(len(order))]
            # Choose partner with opposite sign gradient
            candidates = np.where((np.sign(g) * (-1) == np.sign(g[si])) & (np.arange(n) != si))[0]
            if len(candidates) == 0:
                break
            sj = candidates[rng.randint(len(candidates))]
            step = rng.choice([-1.0, 1.0]) * rng.uniform(0.0003, 0.015)
            if not (0.0 <= h[si] + step <= 1.0 and 0.0 <= h[sj] - step <= 1.0):
                continue
            h[si] += step
            h[sj] -= step
            c5 = _compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[si] -= step
                h[sj] += step

    # Phase 3: Fine random search with annealing
    if time.time() - t0 < budget_s - 3:
        h = best_h.copy()
        temp = 0.003
        for it in range(800000):
            if time.time() - t0 > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            s = rng.choice([-1.0, 1.0]) * rng.uniform(0.0002, 0.01)
            if not (0.0 <= h[i] + s <= 1.0 and 0.0 <= h[j] - s <= 1.0):
                continue
            h[i] += s
            h[j] -= s
            c5 = _compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            elif c5 < best_c5 + temp and rng.rand() < 0.05:
                pass
            else:
                h[i] -= s
                h[j] += s
            temp *= 0.99995

    # Final projection to ensure constraint satisfaction
    best_h = _project(best_h, tsum)
    best_c5 = _compute_c5(best_h, n)
    return best_h, best_c5, n
