def project_feasible(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = n / 2.0 - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    """Compute C5 = max_k corr[k] * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Compute gradient of output entry (shift) of np.correlate(h, 1-h, full)."""
    g = np.zeros(n)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = n / 2.0 - s
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0 - 1e-14) if diff > 0 else (h > 1e-14)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def grad_entry(h, shift, n):
    g = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n, n - shift)
    j1 = max(0, i_lo + shift)
    j2 = min(n, i_hi + shift)
    if j1 < j2:
        g[j1:j2] += (1.0 - h[j1 - shift:j2 - shift])
    if i_lo < i_hi:
        g[i_lo:i_hi] -= h[i_lo + shift:i_hi + shift]
    return g

def subgradient_descent(h, n, rng, max_iters, time_limit):
    h_cur = h.copy()
    best_c5 = eval_c5(h_cur)
    best_h = h_cur.copy()
    lr = 0.3
    momentum = np.zeros(n)
    mu = 0.9
    stagnant = 0

    for it in range(max_iters):
        if time.time() > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        c5_val = float(np.max(corr) * 2.0 / n)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_cur.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 800:
            h_cur = best_h + rng.randn(n) * 0.08
            h_cur = project_feasible(h_cur, n)
            lr = 0.3
            momentum = np.zeros(n)
            stagnant = 0
            continue

        max_val = np.max(corr)
        max_idx = np.where(corr >= max_val - 1e-9)[0]
        k_cap = min(len(max_idx), 3)
        grad = np.zeros(n)
        for ki in max_idx[:k_cap]:
            shift = ki - (n - 1)
            grad += grad_entry(h_cur, shift, n)
        grad /= k_cap

        step = lr * (2.0 / n)
        momentum = mu * momentum + (1 - mu) * grad
        h_new = h_cur - step * momentum
        h_new = project_feasible(h_new, n)
        c5_new = eval_c5(h_new)
        if c5_new < c5_val:

            lr = min(lr * 1.08, 3.0)
        else:

            if lr <= 0.011:
                h_cur = best_h + rng.randn(n) * 0.03
                h_cur = project_feasible(h_cur, n)
                lr = 0.3
                momentum = np.zeros(n)
    return best_h, best_c5

def smoothed_max_grad(h, n, beta=60.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    shifted = beta * (corr - mx)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    top_k = 10
    top_idx = np.argsort(corr)[-top_k:]
    for ki in top_idx:
        if weights[ki] < 1e-8:
            continue
        shift = ki - (n - 1)
        grad += weights[ki] * grad_entry(h, shift, n)
    return grad, mx * 2.0 / n
