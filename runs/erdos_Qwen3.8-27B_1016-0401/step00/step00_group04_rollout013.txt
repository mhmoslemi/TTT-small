```python
    #   term 2: i+m-n+1=j → -h[j-m+n-1] if 0 <= j-m+n-1 < n
    
    m = k_max
    # Vectorized:
    idx1 = np.arange(n) + m - n + 1  # j + m - n + 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    
    idx2 = np.arange(n) - m + n - 1  # j - m + n - 1
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    
    h_new = h - lr * grad
    return project_feasible(h_new, n * 0.5)
```
