            h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    bc = float('inf')  # best c5
    bh = None
    bn = 0
    
    # Get parent
    try:
        hp = np.array(initial_h_values, float)
    except:
        hp = None
    
    def try_update(h, n):
        nonlocal bc, bh, bn
        c = c5_of(h, n)
        if c < bc:
            bc, bh, bn = c, h.copy(), n
        return c
    
    # N values
    ns = []
    if hp is not None:
        ns.append(len(hp))
    for v in [50, 60, 70, 80, 90, 100, 120]:
        if v not in ns:
            ns.append(v)
    ns.sort()
    
    for n in ns:
        if time.time() - t0 > budget_s - 30:
            break
        tgt = n / 2.0
        tl = budget_s - (time.time() - t0) - 30
        n_left = len(ns) - ns.index(n)
        tn = min(180, tl / max(1, n_left))
        
        # Build starts
        starts = []
        if hp is not None and n == len(hp):
            starts.append(hp.copy())
        # binary block
        h1 = np.zeros(n); h1[:n//2] = 1.0; starts.append(h1)
        # binary alternating
        starts.append((np.arange(n) % 2).astype(float))
        # binary pattern: 1,1,0,0,1,1,0,0...
        pat = np.tile([1,1,0,0], n//4 + 1)[:n]
        starts.append(pat.astype(float))
        # sinusoidal
        starts.append(np.clip(0.5 + 0.45*np.sin(2*np.pi*np.arange(n)*3.0/n), 0, 1))
        # random binary
        hr = (rng.random(n) < 0.5).astype(float)
        hr = proj(hr, tgt)
        starts.append(hr)
        
        for si, s in enumerate(starts):
            if time.time() - t0 > budget_s - 20:
                break
            h = proj(s.copy(), tgt)
            c = try_update(h, n)
            
            # Subgradient with Nesterov momentum
            lr0 = 0.08
            for phase in range(6):
                lr = lr0 * (0.75 ** phase)
                if lr < 0.001:
                    break
                if time.time() - t0 > budget_s - 20:
                    break
                y = h.copy()
                for it in range(20000):
                    if time.time() - t0 > budget_s - 20:
                        break
                    corr = np.correlate(y, 1.0 - y, mode="full")
                    mi = int(np.argmax(corr))
                    lag = mi - (n - 1)
                    # subgradient at y for active lag
                    g = np.zeros(n)
                    if lag >= 0:
                        m = n - lag
                        g[:m] += 1.0 - y[lag:]
                        if lag > 0:
                            g[lag:] -= y[:m]
                        else:
                            g -= y
                    else:
                        ll = -lag
                        m = n - ll
                        g[ll:] += 1.0 - y[:m]
                        g[:m] -= y[ll:]
                    h_new = proj(y - lr * g, tgt)
                    # Nesterov
                    y = h_new + 0.9 * (h_new - h)
                    y = np.clip(y, 0.0, 1.0)
                    h = h_new
                    if it % 5000 == 4999:
                        # perturbation
                        h += rng.normal(0, 0.03, n)
                        h = proj(h, tgt)
                        y = h.copy()
                    # check
                    if it % 1000 == 999:
                        c = try_update(h, n)
                # end phase
            # end phases
            
            # Local polish: try small perturbations
            for _ in range(50):
                if time.time() - t0 > budget_s - 15:
                    break
                h2 = h + rng.normal(0, 0.02, n)
                h2 = proj(h2, tgt)
                c = try_update(h2, n)
                if c < c5_of(h, n):
                    h = h2
            
    # Final subgradient polish on best solution
    if bh is not None:
        n = bn
        tgt = n / 2.0
        h = bh.copy()
        c = try_update(h, n)
        lr = 0.01
        for it in range(200000):
            if time.time() - t0 > budget_s - 5:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            mi = int(np.argmax(corr))
            lag = mi - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                m = n - lag
                g[:m] += 1.0 - h[lag:]
                if lag > 0:
                    g[lag:] -= h[:m]
                else:
                    g -= y
            else:
                ll = -lag
                m = n - ll
                g[ll:] += 1.0 - h[:m]
