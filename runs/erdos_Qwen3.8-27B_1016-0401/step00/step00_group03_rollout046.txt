import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_sum(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum}"""
    n = len(h)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def grad_corr_at(h, k):
    """Gradient of corr at shift k (where shift ranges -(n-1) to n-1)"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i] += (1.0 - h[i + k])
            grad[i + k] += (-h[i])
    else:
        kk = -k
        for i in range(kk, n):
            grad[i] += (1.0 - h[i + kk])
            grad[i + kk] += (-h[i])
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    # Ensure feasibility
    h0 = project_sum(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimization using subgradient + projection
    step_size = 0.1
    for iteration in range(10000):

        c = np.correlate(h0, 1.0 - h0, mode='full')
        max_idx = np.argmax(c)

        # Also consider sum constraint gradient (all ones)

        h_new = h0 - step_size * g
        h_new = project_sum(h_new, target_sum)

            h0 = h_new.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_onto_constraint(h, target_sum):
    """Project onto {x: 0≤x_i≤1, sum(x)=target_sum} via bisection on shift"""
    n = len(h)
    s0 = np.sum(h)
    if abs(s0 - target_sum) < 1e-12:
        return np.clip(h, 0, 1)
    # We need sum(clip(h + alpha, 0, 1)) = target_sum
    lo, hi = min(0, target_sum - s0) - 1, max(0, target_sum - s0) + 1
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2
    return np.clip(h + alpha, 0, 1)

def objective_smooth(h, T, penalty_w, target_sum):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * c * dx
    Tc_max = np.max(Tc)
    exp_Tc = np.exp(Tc - Tc_max)
    logsumexp_val = (Tc_max + np.log(np.sum(exp_Tc))) / T * dx
    sum_h = np.sum(h)
    penalty = penalty_w * (sum_h - target_sum)**2
    return logsumexp_val + penalty

def grad_smooth(h, T, penalty_w, target_sum):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

        k = idx - (n-1)  # shift
        if w[idx] < 1e-15:
            continue

        # corr(k) = sum_i h[i]*(1-h[i+k]) for valid i
        # d/dh[j]: (1-h[j+k])*I(0≤j+k<n) - h[j-k]*I(0≤j-k<n)

                grad[i] += w[idx] * (1.0 - h[i+k])
                grad[i+k] += w[idx] * (-h[i])
        else:
            kk = -k
            for i in range(kk, n):
                grad[i] += w[idx] * (1.0 - h[i+kk])
                grad[i+kk] += w[idx] * (-h[i])
    
    grad *= dx
    # Add penalty gradient
    grad += 2 * penalty_w * (np.sum(h) - target_sum)
    return grad

def _proj(h, ts):
    """Project onto {x: 0<=x_i<=1, sum(x)=ts}"""
    lo, hi = -2.0, 2.0

        if np.sum(np.clip(h + m, 0, 1)) < ts:

    tc = T * c * dx
    tc -= tc.max()

    val += pw * (np.sum(h) - ts) ** 2
