import numpy as np
import time

def c5_eval(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_01(x, n, rng):
    target = n // 2
    result = np.zeros(n)
    idx = np.argsort(x)[::-1]
    result[idx[:target]] = 1.0
    return result

def to_vertex_pattern(h, rng):
    n = len(h)
    if n % 2 == 0:
        return project_01(h, n, rng)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    frac = np.where((h > 0.01) & (h < 0.99))[0]
    zeros = np.where(h <= 0.01)[0]
    h[ones] = 1.0
    h[zeros] = 0.0
    target_ones = (n - 1) // 2
    count = int((h > 0.5).sum())
    if count > target_ones:
        extra = ones[:count - target_ones] if count - target_ones <= len(ones) else np.array([])
        for i in extra:
            h[i] = 0.0
            count -= 1
        if count > target_ones:
            ones2 = np.where(h > 0.5)[0]
            for i in ones2[:count - target_ones]:
                h[i] = 0.0
    elif count < target_ones:
        zeros2 = np.where(h == 0.0)[0]
        rng.shuffle(zeros2)
        for i in zeros2[:target_ones - count]:
            h[i] = 1.0
    if len(frac) > 0:
        h[frac[0]] = 0.5
    else:
        h[np.where(h == 0.0)[0][0]] = 0.5
    s = h.sum()
    if abs(s - n / 2.0) > 1e-6:
        diff = n / 2.0 - s
        if diff > 0:
            z = np.where(h == 0.0)[0]
            if len(z) > 0:
                h[z[0]] += diff
            else:
                h[np.argmin(h)] += diff
        else:
            o = np.where(h == 1.0)[0]
            if len(o) > 0:
                h[o[0]] += diff
            else:
                h[np.argmax(h)] += diff
    h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(h_start, end_time, rng, t0):
    h = h_start.copy().astype(float)
    n = len(h)
    current = c5_eval(h)
    best = h.copy()
    best_val = current
    it = 0
    max_iters = 2000000
    
    while it < max_iters:
        it += 1
        if time.time() >= end_time:
            break
        frac = min(1.0, it / 200000.0)
        T = 0.25 * (0.003 / 0.25) ** frac
        if T < 0.001:
            T = 0.001
        
        ones = np.where(h > 0.9)[0]
        zeros = np.where(h < 0.1)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        old_i, old_j = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        new_val = c5_eval(h)
        delta = new_val - current
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
        else:
            h[i], h[j] = old_i, old_j
    return best, best_val

def greedy_local_search(h, end_time):
    h = h.copy().astype(float)
    current = c5_eval(h)
    best = h.copy()
    best_val = current
    while time.time() < end_time:
        improved = False
        ones = list(np.where(h > 0.9)[0])
        zeros = list(np.where(h < 0.1)[0])
        if len(ones) == 0 or len(zeros) == 0:

        for i in ones:
            if time.time() >= end_time:
                return best, best_val
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                nv = c5_eval(h)
                if nv < current - 1e-12:
                    current = nv
                    improved = True
                    if nv < best_val:
                        best_val = nv
                        best = h.copy()
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if improved:
                break
        if not improved:

def interpolate_h(src, n_new):
    n_src = len(src)
    x_old = np.linspace(0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, src)

        if diff > 0:
            z = np.where(best_h < 1.0)[0]
