import numpy as np
from scipy import optimize
import time

_CONST = [0.0]


def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)


def _softmax(v, tau):
    m = v.max()
    return float(m + tau * np.log(np.sum(np.exp((v - m) / tau))))


def _obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return _softmax(corr, tau)


def _grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    w = np.exp((corr - m) / tau)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += wk * dx
            g[:n - s] -= wk * dx * h[s:]
            g[s:] -= wk * dx * h[:n - s]
        else:
            ss = -s
            g[ss:] += wk * dx
            g[ss:] -= wk * dx * h[:n - ss]
            g[:n - ss] -= wk * dx * h[ss:]
    return g


def _cons(x):
    return np.sum(x) - _CONST[0]


def _project(h, target):
    n = len(h)
    h = h + (target - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

                            method='SLSQP', bounds=bnds, constraints=cons,
                            options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    return np.clip(res.x, 0.0, 1.0)
