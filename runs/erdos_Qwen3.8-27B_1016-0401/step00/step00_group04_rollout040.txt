def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, j, n):
    g = np.zeros(n)
    lo1 = max(0, -j)
    hi1 = min(n, n - j)
    if hi1 > lo1:
        m = np.arange(lo1, hi1)
        g[m] += 1.0 - h[m + j]
    lo2 = max(0, j)
    hi2 = min(n, n + j)
    if hi2 > lo2:
        m = np.arange(lo2, hi2)
        g[m] -= h[m - j]
    return g
