def projected_grad_descent(h_init, n, alpha=50.0, lr=0.01, max_iter=5000, time_limit=100):
    t0 = time.time()
    target = n / 2.0
    h = h_init.copy()
    h = project_constraint(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        val, grad = compute_grad_smooth_max(h, alpha)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_constraint(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            # Reduce learning rate
            lr *= 0.9
            if lr < 1e-6:
                break
    
    return best_h, best_c5

For each pair (j, k), optimize the pair jointly:

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if not np.any(mask):
            mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, target, alpha=40.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    val = cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha
    pen = 1e4 * (np.sum(h) - target) ** 2
    return val + pen

        h_b = np.where(best_h > 0.5, 1.0, 0.0)
        if n % 2 == 1:
            # Need exactly one 0.5 (or combination)
            s = np.sum(h_b)
            diff = target - s
            if diff == 0.5:
                idx = np.argmin(best_h)
                h_b[idx] = 0.5
            elif diff == -0.5:
                idx = np.argmax(best_h)
                h_b[idx] = 0.5
            elif diff != 0:
                # Adjust
                if diff > 0:
                    for k in range(int(diff) + 1):
                        idx = np.where(h_b == 0.0)[0]
                        if len(idx) > 0:
                            h_b[idx[k]] = 1.0
                    if abs(np.sum(h_b) - target) > 1e-9:
                        idx = np.argmin(h_b)
                        h_b[idx] = target - (np.sum(h_b) - h_b[idx])
        h_b = project_feasible(h_b, target)
        c5_b = compute_c5(h_b)
        # Binary pair-flip search
        h_b2, c5_b2 = binary_ls(h_b, target, time_limit=min(90, tl() * 0.4), rng=rng)
        if c5_b2 < best_c5:
            best_h, best_c5 = h_b2, c5_b2

    # Phase 5: Random restarts with perturbation
    for _ in range(8):
        if tl() < 30:
            break
        h_re = best_h + 0.15 * rng.randn(n)
        h_re = project_feasible(h_re, target)
        h_ls, c5_ls = local_search(h_re, target, time_limit=min(40, tl() * 0.3), rng=rng)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 6: Try higher resolution (interpolate to 2n)
    if tl() > 80:
        n2 = n * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_feasible(h_interp, target2)
        try:
            def obj3(hf):
                return smooth_obj(hf, target2, 50.0)
            res = minimize(obj3, h_interp, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n2,
                          options={'maxiter': 300, 'ftol': 1e-13, 'maxfun': 3000})
            h_new = project_feasible(res.x, target2)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
        if tl() > 30 and len(best_h) == n2:
            h_ls, c5_ls = local_search(best_h, target2, time_limit=min(80, tl() * 0.5), rng=rng)
            if c5_ls < best_c5:
                best_h, best_c5 = h_ls, c5_ls

    # Final polish
    if tl() > 10:
        h_ls, c5_ls = local_search(best_h, len(best_h) / 2.0, time_limit=min(30, tl() * 0.6), rng=rng)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Ensure valid output
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
