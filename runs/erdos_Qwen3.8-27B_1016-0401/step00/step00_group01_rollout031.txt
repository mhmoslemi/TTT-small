import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    """Compute C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    h = h.copy()
    n = len(h)
    # Simple projection: adjust uniformly, then clip and iterate
    for _ in range(50):
        excess = np.sum(h) - target_sum
        if abs(excess) < 1e-10:
            break
        if excess > 0:
            # Need to decrease
            can_decrease = h > 1e-10
            if not np.any(can_decrease):
                break
            delta = min(excess / np.sum(can_decrease), min(h[can_decrease] * 0.5))
            h[can_decrease] -= delta
        else:
            # Need to increase
            can_increase = h < 1 - 1e-10
            if not np.any(can_increase):
                break
            delta = min(-excess / np.sum(can_increase), (1 - h[can_increase]).max() * 0.5)
            h[can_increase] += delta
    h = np.clip(h, 0, 1)
    return h

def coordinate_descent(h, max_iters=100, tol=1e-10):
    """Optimize h via coordinate descent with sum constraint."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    c5_old = eval_c5(h)
    
    for it in range(max_iters):
        improved = False
        for j in range(n):
            # Try values of h[j] in [0,1]
            # Must maintain sum constraint: adjust by transferring to a partner
            # Partner: the index with most "room" in the opposite direction
            old_val = h[j]
            
            # Find partner: if we increase h[j], decrease partner; if decrease, increase partner
            # For now, just try changing h[j] and reproject
            best_t = old_val
            best_c5 = c5_old
            
            # Grid search for t
            for t in np.linspace(0, 1, 21):
                h_test = h.copy()
                h_test[j] = t
                h_test = project_to_constraint(h_test, target_sum)
                c5_test = eval_c5(h_test)
                if c5_test < best_c5 - 1e-15:
                    best_c5 = c5_test
                    best_t = t
                    h[j] = h_test[j]  # save the projected result
                    h = h_test.copy()
            
            if h[j] != old_val:
                improved = True
            c5_old = eval_c5(h)
        
        if not improved:
            break
    
    return h, c5_old

Let me be more careful about the coordinate descent. The objective is:
C5(h) = max_k c_k(h) * dx

where c_k(h) = sum_{valid i} h[i] * (1-h[i+k])

For a given j, when we change h[j] to t, the change in c_k is:
- For each k where j participates: c_k changes by Δ_k * t where Δ_k is the coefficient
- c_k(t) = c_k_base + Δ_k * t (linear in t)

So C5(t) = max_k (c_k_base + Δ_k * t) * dx

This is a piecewise linear convex function in t. Its minimum is at a breakpoint or at t=0 or t=1.

The breakpoints occur where (c_i_base + Δ_i * t) = (c_j_base + Δ_j * t), i.e.,
t = (c_j_base - c_i_base) / (Δ_i - Δ_j)

So I can:
1. Compute c_k_base (correlation with h[j]=0) and Δ_k (coefficient) for all k
2. Find all breakpoints in [0,1]
3. Evaluate C5 at all breakpoints and at 0, 1
4. Pick the minimum

This is exact for the coordinate step! And it's O(n²) per coordinate (for computing all c_k and finding breakpoints among O(n²) pairs).

Actually, computing all 93 correlation values from scratch is O(n²) = 2209. Finding breakpoints among C(93,2) ≈ 4300 pairs is fast. Evaluating at up to 4300+2 points is also fast.

Total per pass: 47 * (2209 + 4300 + 4300) ≈ 47 * 11000 ≈ 515k ops. Very fast!

            break
        T = T_init * (T_fin / T_init) ** (it / max(max_steps, 1))
        j = rng.randint(n)
        m = rng.randint(n)
        if j == m:
            continue
        d = (2.0 * rng.rand() - 1.0) * 0.15
        h[j] += d
        h[m] -= d
        if h[j] < 0 or h[j] > 1 or h[m] < 0 or h[m] > 1:
            h[j] -= d
            h[m] += d
            continue
        c5_new = eval_c5(h)
        dc = c5_new - c5_curr
        if dc < 0 or rng.rand() < np.exp(-dc / max(T, 1e-12)):
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[j] -= d
            h[m] += d
    best_h = project_sum(best_h, target)
    return best_h, eval_c5(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    except:
        h = np.zeros(47)
        h[:24] = 1.0
    n = len(h)
    target = n / 2.0
    h = project_sum(h, target)
    h_best = h.copy()
    c5_best = eval_c5(h_best)

    # Phase 1: Coordinate descent with strong penalty
    t_lim = t0 + budget_s * 0.12
    for _ in range(5000):
        if time.time() > t_lim:
            break
        h = coord_pass(h, M=500.0, n_grid=41)
        h = project_sum(h, target)
        c5 = eval_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()

    # Phase 2: Coordinate descent with moderate penalty
    t_lim = t0 + budget_s * 0.25
    h = h_best.copy()
    for _ in range(5000):
        if time.time() > t_lim:
            break
        h = coord_pass(h, M=10.0, n_grid=81)
        h = project_sum(h, target)
        c5 = eval_c5(h)
        if c5 < c5_best:
            c5_best, h_best = c5, h.copy()

    # Phase 3: Local search (pairwise swaps)
    t_lim = t0 + budget_s * 0.50
    h = h_best.copy()
    h, c5 = local_search(h, max_steps=200000, time_limit=t_lim - time.time(), start_time=time.time(), rng=rng)
    if c5 < c5_best:
        c5_best, h_best = c5, h.copy()

    # Phase 4: Simulated annealing
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 120:
        h = h_best.copy()
        t_lim_sa = min(t0 + budget_s * 0.78, time.time() + t_rem * 0.65)
        h, c5 = anneal(h, T_init=0.02, T_fin=0.00005, max_steps=int(t_rem * 3000),
                       time_limit=t_lim_sa - time.time(), start_time=time.time(), rng=rng)
        if c5 < c5_best:
            c5_best, h_best = c5, h.copy()

    # Phase 5: Final coordinate descent polish
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 20:
        h = h_best.copy()
        for _ in range(3000):
            if time.time() - t0 > budget_s - 5:
                break
            h = coord_pass(h, M=1.0, n_grid=101)
            h = project_sum(h, target)
            c5 = eval_c5(h)
            if c5 < c5_best:
                c5_best, h_best = c5, h.copy()

    # Phase 6: Final local search
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 15:
        h = h_best.copy()
        h, c5 = local_search(h, max_steps=50000, time_limit=min(t_rem - 3, 60), start_time=time.time(), rng=rng)
        if c5 < c5_best:
            c5_best, h_best = c5, h.copy()

    h_best = project_sum(h_best, target)
    c5_best = eval_c5(h_best)
    return h_best, c5_best, n
