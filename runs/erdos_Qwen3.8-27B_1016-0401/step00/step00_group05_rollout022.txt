import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def to_binary(h):
    h_b = (h > 0.5).astype(float)
    # Fix count
    n = len(h)
    target = n // 2
    ones = np.sum(h_b == 1)
    if ones > target:
        # Remove some ones
        idx = np.where(h_b == 1)[0]
        # Remove those with highest h value (least contribution)
        to_remove = idx[np.argsort(-h[idx])[:ones - target]]
        h_b[to_remove] = 0.0
    elif ones < target:
        idx = np.where(h_b == 0)[0]
        to_add = idx[np.argsort(-h[idx])[:target - ones]]
        h_b[to_add] = 1.0
    return h_b

def to_binary(h):
    h_b = (h > 0.5).astype(float)
    n = len(h)
    target = n // 2
    ones_count = int(np.sum(h_b))
    if ones_count > target:
        idx = np.where(h_b == 1.0)[0]
        to_rm = idx[np.argsort(-h[idx])[:ones_count - target]]
        h_b[to_rm] = 0.0

def steepest_descent_binary(h, max_sweeps=200, time_limit=60):
    n = len(h)
