import numpy as np
from scipy.optimize import minimize
import time

def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_smooth(h, alpha=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (corr - m) / dx))) * dx

def _project(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute equally
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
