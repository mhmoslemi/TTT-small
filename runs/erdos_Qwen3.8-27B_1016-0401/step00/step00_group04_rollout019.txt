def gradient_move(h, rng, n):
    """Move in the negative subgradient direction"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_star = np.argmax(corr)
    # Subgradient
    s = np.zeros(n)
    shift = m_star - (n - 1)  # actual shift value
    for j in range(n):
        jm = j - shift
        jp = j + shift
        if 0 <= jm < n:
            s[j] += (1 - h[jm])
        if 0 <= jp < n:
            s[j] -= h[jp]
    s *= dx
    
    # Move in -s direction, paired to maintain sum
    i_candidates = np.where(s > 0.1)[0]  # decrease h here
    j_candidates = np.where(s < -0.1)[0]  # increase h here
    if len(i_candidates) == 0 or len(j_candidates) == 0:
        return h, False
    i = rng.choice(i_candidates)
    j = rng.choice(j_candidates)
    delta = min(0.05, abs(s[i]), abs(s[j]))
    h[i] -= delta
    h[j] += delta
    return h, (0 <= h[i] <= 1 and 0 <= h[j] <= 1)
