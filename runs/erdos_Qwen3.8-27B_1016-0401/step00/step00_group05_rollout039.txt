def project_onto_simplex(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -max(h), max(1.0 - h)
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)/2, 0, 1)
