        # Binary round + local search at this resolution
        h_up_bin = (h_up >= 0.5).astype(float)
        n_ones = int(np.sum(h_up_bin == 1))
        t_ones = n_try // 2
        if n_ones > t_ones:
            oi = np.where(h_up_bin == 1)[0]
            h_up_bin[oi[:n_ones - t_ones]] = 0
        elif n_ones < t_ones:
            zi = np.where(h_up_bin == 0)[0]
            h_up_bin[zi[:t_ones - n_ones]] = 1

        time_left = budget_s * 0.85 - elapsed()
        if time_left > 10:
            max_t4 = min(80, time_left * 0.5)
            h_up_bin, c5_ub = binary_ls(h_up_bin, n_try, max_t4)
            try_update(h_up_bin, c5_ub)

    # Phase 5: Final polish on best solution
    time_left = budget_s - elapsed() - 10
    if time_left > 20 and best_h is not None:
        n_best = len(best_h)
        h_fin = best_h.copy()
        c5_fin = compute_c5(h_fin)
        h_fin, c5_fin = coord_descent(h_fin, n_best / 2.0, time_left * 0.5)
        try_update(h_fin, c5_fin)

        h_fin_bin = (h_fin >= 0.5).astype(float)
        n_ones = int(np.sum(h_fin_bin == 1))
        t_ones = n_best // 2
        if n_ones > t_ones:
            oi = np.where(h_fin_bin == 1)[0]
            h_fin_bin[oi[:n_ones - t_ones]] = 0
        elif n_ones < t_ones:
            zi = np.where(h_fin_bin == 0)[0]
            h_fin_bin[zi[:t_ones - n_ones]] = 1
        max_t5 = max(5, time_left * 0.3)
        h_fin_bin, c5_fb = binary_ls(h_fin_bin, n_best, max_t5)
        try_update(h_fin_bin, c5_fb)

    # Validate and return
    n_ret = len(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = project_feasible(best_h, n_ret / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_ret
