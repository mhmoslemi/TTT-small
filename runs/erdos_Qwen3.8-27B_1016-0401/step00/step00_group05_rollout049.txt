def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_objective(h, T, target_sum, penalty_w):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    # Log-sum-exp for numerical stability
    m = np.max(vals)
    smooth_max = m + np.log(np.mean(np.exp((vals - m) * T))) / T
    penalty = penalty_w * (np.sum(h) - target_sum) ** 2
    return smooth_max + penalty

def obj_wrapper(h, T, target_sum, penalty_w):
    h_clipped = np.clip(h, 0, 1)
    return smooth_objective(h_clipped, T, target_sum, penalty_w)

def coordinate_descent(h, dx, target_sum, n_iters=100):
    n = len(h)
    for _ in range(n_iters):
        improved = False
        for i in range(n):
            # Try small perturbations in each direction
            best_val = h[i]
            best_obj = compute_c5(h)
            # Actually, this is O(n^2) per step which is OK for n=70
            for delta in [-1, -0.5, -0.1, 0.1, 0.5, 1]:
                new_val = h[i] + delta / n  # small step
                if 0 <= new_val <= 1:
                    h[i] = new_val
                    # Adjust sum constraint approximately
                    obj = compute_c5(h)
                    if obj < best_obj:
                        best_obj = obj
                        best_val = new_val
                        improved = True
            h[i] = best_val
        if not improved:
            break
    # Fix sum constraint
    diff = target_sum - np.sum(h)

        # Distribute adjustment
        mask = (h < 1) if diff > 0 else (h > 0)
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

    return float(np.max(corr)) * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max(vals, T):
    m = np.max(vals)
    if T < 1:
        return m
    shifted = (vals - m) * T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.exp(shifted).sum()) / T

def make_obj(T, target_sum, pen_w):
    def obj(h):
        vals = get_corr(h)
        sm = smooth_max(vals, T)
        pen = pen_w * (np.sum(h) - target_sum) ** 2
        return sm + pen
    return obj

def project(h, target_sum):

    return corr * dx

    return float(np.max(_corr_vals(h)))

    m = np.max(vals)
    if T < 1:
        return m
    shifted = np.clip((vals - m) * T, -400, 400)
    return m + np.log(np.exp(shifted).sum()) / T

    vals = _corr_vals(h)
    sm = _smooth_max(vals, T)
    pen = pen_w * (np.sum(h) - target_sum) ** 2
    return sm + pen

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
