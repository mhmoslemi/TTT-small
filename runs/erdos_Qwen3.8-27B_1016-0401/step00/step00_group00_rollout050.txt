def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s == 0:
        h[:] = 0.5
        return h
    # First try uniform scaling
    h2 = h * (target / s)
    if h2.max() <= 1.0 + 1e-12:
        h2 = np.clip(h2, 0, 1)
    else:
        # Need water-filling style adjustment
        h2 = np.clip(h * (target / s), 0, 1)
        # Iterate to fix sum
        for _ in range(100):
            diff = target - h2.sum()
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                mask = h2 < 1.0
            else:
                mask = h2 > 0.0
            if not mask.any():
                break
            step = diff / mask.sum()
            h2[mask] = np.clip(h2[mask] + step, 0, 1)
    # Final fix-up
    diff = target - h2.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(-h2) if diff < 0 else np.argsort(h2)

                remaining -= adj

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(x, n, T):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.max(corr)
    if T > 0:
        shifted = T * (corr - c)
        c = c + np.log(np.sum(np.exp(shifted))) / T
    pen = 500.0 * (h.sum() - n / 2.0) ** 2
    return c + pen
