def adjust_binary_vector(arr, desired_ones):
    """Adjust the binary vector to have exactly desired_ones."""
    n = len(arr)
    n_ones = sum(arr)
    if n_ones != desired_ones:
        # Find the positions to change.
        if n_ones < desired_ones:
            # need to add ones.
            zeros = [i for i in range(n) if arr[i]==0]
            to_ones = np.random.choice(zeros, desired_ones - n_ones, replace=False)
            arr[to_ones] = 1
        else:
            # need to remove ones.
            ones = [i for i in range(n) if arr[i]==1]
            to_zeros = np.random.choice(ones, n_ones - desired_ones, replace=False)
            arr[to_zeros] = 0
