    n_list = sorted(set(n_list))
    
    for n in n_list:
        if time.time() - start_time > budget_s * 0.85:
            break
        if n % 2 != 0:
            continue
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Build starting point
        if n == n_init and init_h is not None:
            h0 = adjust_sum(init_h.copy(), target_sum)
        else:
            # Resample init_h
            x_old = np.linspace(0, 2, len(init_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init_h, left=init_h[0], right=init_h[-1])
            h0 = np.clip(h0, 0, 1)
            h0 = adjust_sum(h0, target_sum)
        
        # SLSQP continuous optimization
        for beta in [30, 80, 150]:
            if time.time() - start_time > budget_s * 0.85:
                break
            h_cont = slsqp_optimize(h0, dx, target_sum, beta,
                                     time_limit=min(40, budget_s * 0.04))
            c5c = compute_c5(h_cont, dx)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_cont.copy()
                best_n = n
            h0 = h_cont
        
        # Round to binary and do local search
        h_bin = (h0 >= 0.5).astype(float)
        h_bin = adjust_sum(h_bin, target_sum)
        c5b = compute_c5(h_bin, dx)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()
            best_n = n
        
        # Binary local search (best improvement)
        h_bs, c5bs = binary_local_search(h_bin, dx, max_iters=3000,
                                          time_limit=min(45, budget_s * 0.04))
        if c5bs < best_c5:
            best_c5 = c5bs
            best_h = h_bs.copy()
            best_n = n
        
        # Simulated annealing from best binary
        h_sa, c5sa = simulated_anneal_binary(h_bs, dx, n_iter=5000,
                                              time_limit=min(40, budget_s * 0.03))
        if c5sa < best_c5:
            best_c5 = c5sa
            best_h = h_sa.copy()
            best_n = n
        
        # Another round of local search after SA
        h_ls2, c5ls2 = binary_local_search(h_sa, dx, max_iters=2000,
                                            time_limit=min(30, budget_s * 0.03))
        if c5ls2 < best_c5:
            best_c5 = c5ls2
            best_h = h_ls2.copy()
            best_n = n
        
        # Continue with continuous from best binary
        h0 = adjust_sum(h_ls2.copy(), target_sum)
        # Add small noise to escape
        h0 += np.random.randn(n) * 0.05
        h0 = np.clip(h0, 0, 1)
        h0 = adjust_sum(h0, target_sum)
        h_cont2 = slsqp_optimize(h0, dx, target_sum, 100,
                                  time_limit=min(30, budget_s * 0.03))
        c5c2 = compute_c5(h_cont2, dx)
        if c5c2 < best_c5:
            best_c5 = c5c2
            best_h = h_cont2.copy()
            best_n = n
        
        # Update init_h for next n (resample best)
        init_h = best_h.copy()
    
    # Final: try one more pass on best_n with random restarts
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        ts = n_best / 2.0
        for rt in range(5):
            if time.time() - start_time > budget_s * 0.92:
                break
            # Random perturbation
            h_r = best_h.copy()
            for _ in range(min(5, n_best // 10)):
                idx = np.random.randint(n_best)
                h_r[idx] = 1.0 - h_r[idx]
            h_r = adjust_sum(h_r, ts)
            # Greedy repair: flip bits to reduce max
            c5r = compute_c5(h_r, dx_best)
            h_r, c5r = binary_local_search(h_r, dx_best, max_iters=1000,
                                            time_limit=min(20, budget_s * 0.02))
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_r.copy()
    
    # Validate
    if best_h is None:
        best_h = adjust_sum(init_h, len(init_h) / 2.0)
        best_c5 = compute_c5(best_h, 2.0 / len(best_h))
        best_n = len(best_h)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    
    return (best_h, best_c5, best_n)
