                c5t = compute_c5(h_trial)

                if gain > best_gain + 1e-12:
                    best_gain = gain
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            c5_cur = compute_c5(h_cur)
            improved = True
        if not improved:
            break
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = 0

    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = len(init)
        best_h = init.copy()
        best_c5 = compute_c5(init)
        best_n = n0
    except Exception:
        n0 = 70
        best_h = np.zeros(n0)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = n0

    n_candidates = sorted(set([n0, 60, 70, 80, 100]))

    for n_target in n_candidates:
        if time.time() - t0 > budget_s * 0.82:
            break

        if n_target == best_n and best_h is not None and len(best_h) == n_target:
            h_curr = best_h.copy()
        else:
            src = best_h if best_h is not None else np.zeros(n0)
            h_curr = np.interp(np.linspace(0, 1, n_target),
                               np.linspace(0, 1, len(src)), src)

        ts = n_target / 2.0
        h_curr = project_box_sum(h_curr, ts)

        # Projected gradient descent with annealed temperature
        for t_val in [10, 30, 100, 300, 800]:
            if time.time() - t0 > budget_s * 0.82:
                break
            lr = 0.02
            for it in range(4000):
                if time.time() - t0 > budget_s * 0.82:
                    break
                obj, grad = smooth_obj_grad(h_curr, t_val)
                h_curr -= lr * grad
                h_curr = project_box_sum(h_curr, ts)
            c5_chk = compute_c5(h_curr)
            if c5_chk < best_c5:
                best_c5 = c5_chk
                best_h = h_curr.copy()
                best_n = n_target

        # Scipy SLSQP polish
        if time.time() - t0 > budget_s * 0.85:
            break

        def slsqp_obj(x):
            return smooth_obj_grad(x, 300)[0]

        bounds = [(0.0, 1.0)] * n_target
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                 'jac': lambda x: np.ones(n_target)}]
        try:
            res = minimize(slsqp_obj, h_curr, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_target
        except Exception:
            pass

        # Binary rounding + local search
        if time.time() - t0 > budget_s * 0.85:
            break
        h_bin = (h_curr > 0.5).astype(float)
        # Fix sum
        cur_sum = h_bin.sum()
        ts = n_target / 2.0
        if abs(cur_sum - ts) > 0.5:
            pass
        else:
            # Adjust to get exactly ts ones
            h_bin = project_box_sum(h_bin, ts)
            h_bin = np.round(h_bin)
            # Ensure exact count
            n_ones = int(np.sum(h_bin > 0.5))
            if n_ones > ts:
                idx1 = np.where(h_bin > 0.5)[0]
                for k in range(n_ones - int(ts)):
                    h_bin[idx1[k]] = 0.0
            elif n_ones < ts:
                idx0 = np.where(h_bin < 0.5)[0]
                for k in range(int(ts) - n_ones):
                    h_bin[idx0[k]] = 1.0
            h_b, c5_b = binary_local_search(h_bin, max_iters=300)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
                best_n = n_target

    # Final polish: small random perturbations
    if time.time() - t0 < budget_s * 0.92:
        n = len(best_h)
        ts = n / 2.0
        for _ in range(1000):
            if time.time() - t0 > budget_s * 0.97:
                break
            scale = rng.uniform(0.001, 0.02)
            h_trial = best_h + rng.normal(0, scale, n)
            h_trial = project_box_sum(h_trial, ts)
            c5t = compute_c5(h_trial)
            if c5t < best_c5 - 1e-12:
                best_c5 = c5t
                best_h = h_trial

    return best_h, best_c5, best_n
