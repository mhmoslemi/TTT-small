        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mid = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mid) > 0:
            h[mid] += diff / np.sum(mid)
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_value(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def c5_smooth(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / beta


def obj_fn(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    v = c5_smooth(h, n, beta)
    sv = np.sum(h) - n / 2.0
    return v + pen * sv * sv


def obj_slsqp(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    return c5_smooth(h, n, beta)


def con_fun(x, n):
    return np.sum(x) - n / 2.0


def thue_morse_h(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = 1.0 - t.astype(float)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    n0 = len(initial_h_values)
    h0 = project_feasible(np.array(initial_h_values, dtype=float), n0)
    best_h = h0.copy()
    best_c5 = c5_value(best_h, n0)

    bounds_box = [(0.0, 1.0)] * n0

    # Phase 1: Progressive smoothing via L-BFGS-B at n0
    alpha_schedule = [3, 8, 15, 30, 60, 120, 250, 500, 1000]
    cur = h0.copy()
    for a in alpha_schedule:
        if time.time() - t0 > budget_s * 0.30:
            break
        res = minimize(obj_fn, cur, args=(n0, a, 5000.0), method='L-BFGS-B',
                       bounds=bounds_box,
                       options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 50})
        hn = project_feasible(res.x, n0)
        cn = c5_value(hn, n0)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn.copy()
            cur = hn.copy()

    # Phase 1b: SLSQP refinement at high beta
    for a in [200, 500, 1000]:
        if time.time() - t0 > budget_s * 0.40:
            break
        cons = {'type': 'eq', 'fun': con_fun, 'args': (n0,)}
        res = minimize(obj_slsqp, best_h, args=(n0, a), method='SLSQP',
                       bounds=bounds_box, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        hn = project_feasible(res.x, n0)
        cn = c5_value(hn, n0)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn.copy()

    # Phase 2: Random restarts and blended starts at n0
    n_start_points = [best_h.copy(), thue_morse_h(n0)]
    for i in range(15):
        n_start_points.append(project_feasible(rng.random(n0) * 0.6 + best_h * 0.4, n0))

    for sp in n_start_points:
        if time.time() - t0 > budget_s * 0.65:
            break
        for a in [80, 200]:
            if time.time() - t0 > budget_s * 0.65:
                break
            cons = {'type': 'eq', 'fun': con_fun, 'args': (n0,)}
            res = minimize(obj_slsqp, sp, args=(n0, a), method='SLSQP',
                           bounds=bounds_box, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12})
            hn = project_feasible(res.x, n0)
            cn = c5_value(hn, n0)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()

    # Phase 3: Larger grid sizes
    for n_try in [60, 90, 135]:
        if time.time() - t0 > budget_s * 0.90:
            break
        xo = np.arange(n0) / n0
        xn = np.arange(n_try) / n_try
        h_up = np.interp(xn, xo, best_h)
        h_up = project_feasible(h_up, n_try)
        bnd = [(0.0, 1.0)] * n_try
        for a in [100, 300, 800]:
            if time.time() - t0 > budget_s * 0.93:
                break
            cons = {'type': 'eq', 'fun': con_fun, 'args': (n_try,)}
            res = minimize(obj_slsqp, h_up, args=(n_try, a), method='SLSQP',
                           bounds=bnd, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-12})
            hn = project_feasible(res.x, n_try)
            cn = c5_value(hn, n_try)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
                n0 = n_try

    # Phase 4: Final local polish at best grid size
    n_final = len(best_h)
    bnd_f = [(0.0, 1.0)] * n_final
    for a in [200, 500, 1000]:
        if time.time() - t0 > budget_s * 0.97:
            break
        cons = {'type': 'eq', 'fun': con_fun, 'args': (n_final,)}
        res = minimize(obj_slsqp, best_h, args=(n_final, a), method='SLSQP',
                       bounds=bnd_f, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-13})
        hn = project_feasible(res.x, n_final)
        cn = c5_value(hn, n_final)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = c5_value(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
