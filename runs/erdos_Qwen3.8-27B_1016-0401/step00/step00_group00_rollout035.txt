```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, n):
    target = n / 2.0
    # Find lambda: sum(clip(h - lambda, 0, 1)) = target
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def subgradient_step(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    # Get top-k indices
    indices = np.argsort(corr)[-top_k:]
    active = indices[corr[indices] >= max_val - 1e-6]
    if len(active) == 0:
        active = indices[:1]
    
    grad = np.zeros(n)
    for m in active:
        tau = int(m) - (n - 1)
        # gradient contribution: (1-h[j+tau]) - h[j-tau]
        j_lo1 = max(0, -tau)
        j_hi1 = min(n, n - tau)
        if j_lo1 < j_hi1:
            grad[j_lo1:j_hi1] += (1.0 - h[j_lo1+tau:j_hi1+tau])
        j_lo2 = max(0, tau)
        j_hi2 = min(n, n + tau)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= h[j_lo2-tau:j_hi2-tau]
    
    grad *= dx / float(len(active))
    return grad, max_val * dx

def optimize_sg(h, n, max_time, rng, lr_init=0.02):
    """Subgradient descent with projection"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()
    stall = 0
    
    while time.time() - t0 < max_time:
        grad, c5 = subgradient_step(h, n)
        h_new = h - lr * grad
        h_new = project_constraint(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 0.2)

            stall += 1

                h = best_h + rng.normal(0, 0.05, n)
                h = project_constraint(h, n)
                lr = lr_init
                stall = 0
            else:
                h = h_new
                lr *= 0.97
        
        if time.time() - t0 > max_time:
            break
    
    return best_h, best_c5

def random_h(n, rng):
    """Generate random h satisfying constraints"""
    h = rng.uniform(0, 1, n)
    h = project_constraint(h, n)
    return h

def smooth_h(h, n, window=3):

    if window <= 1:
        return h.copy()
    kernel = np.ones(window) / window

    h_smooth = np.convolve(h_padded, kernel, mode='valid')[:n]
    return project_constraint(h_smooth, n)
```

```python
        jhi2 = min(n, -tau + n)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= h[jlo2+tau:jhi2+tau]
    grad *= dx / n_active
    return grad, max_val * dx

                lr = lr_init
                stall = 0
            else:

        if time.time() - t0 > max_time:
            break
    return best_h, best_c5

    return project_constraint(h, n)
```

Alternative: use the smooth approximation (log-sum-exp) with scipy's L-BFGS-B or SLSQP.

The idea: minimize LSE(h) = (1/β) * log(Σ_k exp(β * corr_k)) * dx

subject to 0 ≤ h_i ≤ 1 and Σ h_i = n/2.

This is a smooth (for large β) differentiable function. We can use scipy.optimize.minimize with constraints.

But the constraint Σ h_i = n/2 makes it a bit tricky. We can parameterize: h = [h_1, ..., h_{n-1}, n/2 - Σ_{i<n} h_i] and require the last to be in [0,1].

```python
def lse_objective(h_flat, n, beta):
    h = h_flat.copy()
    h = project_constraint(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Log-sum-exp
    max_corr = np.max(corr)
    lse = max_corr + np.log(np.sum(np.exp(corr - max_corr)))
    return lse / beta * dx

def lse_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = np.exp(corr - np.max(corr))
    weights /= np.sum(weights)
    # Gradient of LSE is weighted average of individual gradients
    grad = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] < 1e-10:
            continue
        tau = m - (n - 1)
        jlo1 = max(0, tau)
        jhi1 = min(n, tau + n)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += weights[m] * (1.0 - h[jlo1-tau:jhi1-tau])
        jlo2 = max(0, -tau)
        jhi2 = min(n, -tau + n)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= weights[m] * h[jlo2+tau:jhi2+tau]
    grad *= dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, n):
    target = n / 2.0
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def sg_gradient(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    idx = np.argsort(corr)[-top_k:]
    thr = max_val - max(1e-8, max_val * 1e-6)
    active = idx[corr[idx] >= thr]
    if len(active) == 0:
        active = idx[:1]
    grad = np.zeros(n)
    for m in active:
        tau = int(m) - (n - 1)
        jlo1 = max(0, tau)
        jhi1 = min(n, tau + n)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += (1.0 - h[jlo1 - tau:jhi1 - tau])
        jlo2 = max(0, -tau)
        jhi2 = min(n, -tau + n)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= h[jlo2 + tau:jhi2 + tau]
    grad *= dx / float(len(active))
    return grad

def lse_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    exp_c = np.exp(corr - mx)
    weights = exp_c / float(np.sum(exp_c))
    grad = np.zeros(n)
    cutoff = 1e-8
    for m in range(len(corr)):
        w = weights[m]
        if w < cutoff:
            continue
        tau = m - (n - 1)
        jlo1 = max(0, tau)
        jhi1 = min(n, tau + n)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += w * (1.0 - h[jlo1 - tau:jhi1 - tau])
        jlo2 = max(0, -tau)
        jhi2 = min(n, -tau + n)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= w * h[jlo2 + tau:jhi2 + tau]
    grad *= dx
    return grad

def optimize_lse(h, n, max_time, rng, beta_start=10.0, beta_end=200.0, lr_init=0.05):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = lr_init
    stall = 0
    while time.time() - t0 < max_time:
        prog = min(1.0, (time.time() - t0) / max_time)
        beta = beta_start * (beta_end / beta_start) ** prog
        grad = lse_gradient(h, n, beta)
        h = h - lr * grad
        h = project_constraint(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.005, 0.3)
        else:
            stall += 1
            if stall > 40:
                h = best_h + rng.normal(0, 0.04 * (1.0 + stall / 80.0), n)
                h = project_constraint(h, n)
                lr = lr_init
                stall = 0
            else:
                lr *= 0.985
        if time.time() - t0 > max_time:
            break
    return best_h, best_c5

def optimize_sg(h, n, max_time, rng, lr_init=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(h)
```
