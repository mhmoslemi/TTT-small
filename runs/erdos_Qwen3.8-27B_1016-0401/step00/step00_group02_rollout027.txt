def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, dx, scale=50.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    if scale < 1e6:
        val = m + np.log(np.sum(np.exp(scale * (corr - m)))) / scale
    else:
        val = m
    return val * dx

    target = n / 2.0

        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        h = h - diff / n
        h = np.clip(h, 0, 1)

        if res > 0:

        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += res / len(idx)
            h = np.clip(h, 0, 1)

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exps = np.exp(scale * (corr - m))
    val = m + np.log(exps.sum()) / scale

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _sopt(h, dx, sc):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(sc * (corr - m))
    return (m + np.log(e.sum()) / sc) * dx

def _f_slsqp(x):
    return _sopt(np.clip(x, 0.0, 1.0), _G_DX, _G_SCALE)

def _ceq(x):
    return x.sum() - 0.5 * _G_N

def _ceqj(x):
    return np.ones(len(x))

def _proj(h, n):
    tgt = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        h -= d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - h.sum()
    if abs(r) > 1e-15:
        msk = (h < 1.0) if r > 0 else (h > 0.0)
        if msk.any():
            h[msk] += r / msk.sum()
            np.clip(h, 0.0, 1.0, out=h)
