```python
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, mode='full') * dx"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def evaluate(h, n):
    dx = 2.0 / n
    return compute_c5(h, dx)

def local_search_binary(h, n, max_iter=1000):
    """Local search on binary representation"""
    dx = 2.0 / n
    h = h.copy()
    current_c5 = compute_c5(h, dx)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for iteration in range(max_iter):
        improved = False
        # Try swapping a 1 and a 0
        best_improvement = 0
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                new_c5 = compute_c5(h, dx)
                h[i], h[j] = 1, 0
                if current_c5 - new_c5 > best_improvement:
                    best_improvement = current_c5 - new_c5
                    best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            h[i], h[j] = 0, 1
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            current_c5 -= best_improvement
            improved = True
        else:
            break
    return h, current_c5
```

```python
    Returns a matrix grad[k, j] = d(result[k])/d(h[j])"""
    # For each k and j:
    # d(result[k])/d(h[j]) = [j+k-n+1 in [0,n-1]]*(1-h[j+k-n+1]) - [j-k+n-1 in [0,n-1]]*h[j-k+n-1]
```

```python
def logsumexp_c5(h, n, t):
    result = np.correlate(h, 1-h, mode="full")
    max_r = np.max(result)
    if t * max_r > 500:
        log_sum = np.log(np.sum(np.exp(t * (result - max_r)))) + t * max_r
    else:
        log_sum = np.log(np.sum(np.exp(t * result)))

    # Gradient
```

```python
def compute_gradient_vectorized(h, n, p):
    """p[k] is the weight for lag k (already includes the (2/n) factor)
    Returns gradient vector of length n."""
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-12:
            continue
        d = k - (n-1)
        # (1-h[j+d]⁺) - h[j-d]⁺
        if d >= 0:
            # 1-h[j+d] is valid for j = 0, ..., n-1-d
            term1 = np.zeros(n)
            term1[:n-d] = 1 - h[d:]
            # h[j-d] is valid for j = d, ..., n-1
            term2 = np.zeros(n)
            term2[d:] = h[:n-d]
        else:
            dd = -d
            # 1-h[j-d] = 1-h[j+dd] is valid for j = 0, ..., n-1-dd
            term1 = np.zeros(n)
            term1[:n-dd] = 1 - h[dd:]
            # h[j+d] = h[j-dd] is valid for j = dd, ..., n-1
            term2 = np.zeros(n)
            term2[dd:] = h[:n-dd]
        grad += p[k] * (term1 - term2)
    return grad * (2.0/n)
```

```python
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 0 and s < n:
        h = h + (n/2.0 - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1-h, mode="full")
    max_c = np.max(corr)
    exp_vals = np.exp(t * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    F = (2.0/n) * ((1.0/t) * np.log(sum_exp) + max_c)

    grad = np.zeros(n)
```

```python
import numpy as np
import time

def compute_c5(h, n):
    return np.max(np.correlate(h, 1-h, mode="full")) * (2.0/n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n/2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_grad(h, n, t):
    corr = np.correlate(h, 1-h, mode="full")
    m = np.max(corr)
    exps = np.exp(t*(corr - m))
    s = np.sum(exps)
    F = (2.0/n)*(np.log(s)/t + m)
    p = exps/s
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-14:
            continue
        d = k - n + 1
        if d >= 0:
            t1 = np.zeros(n); t1[:n-d] = 1.0-h[d:]
            t2 = np.zeros(n); t2[d:] = h[:n-d]
        else:
            dd = -d
            t1 = np.zeros(n); t1[:n-dd] = 1.0-h[dd:]
            t2 = np.zeros(n); t2[dd:] = h[:n-dd]
        grad += p[k]*(t1-t2)
    grad *= (2.0/n)
    return F, grad

def gd_optimize(h, n, time_limit, t0=2.0, t1=80.0, base_lr=0.02):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    max_iter = 8000
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        frac = it/max_iter
        t = t0 + (t1-t0)*frac
        lr = base_lr * (1.0 - 0.8*frac)
        F, grad = lse_grad(h, n, t)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = lr * grad / gnorm  # normalize step
        for scale in [step, 0.5*step, 0.1*step, 0.01*step]:
            h_new = project_feasible(h + scale, n)
            c5_new = compute_c5(h_new, n)
            if c5_new <= best_c5 + 1e-7:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                break
    return h, best_c5

def local_search(h, n, max_passes=200):
```
