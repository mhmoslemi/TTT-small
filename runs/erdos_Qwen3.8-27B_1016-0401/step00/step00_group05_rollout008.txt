def project_feasible(h, n):
    h = np.clip(h.copy(), 0, 1)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:

        h += diff / n
        h = np.clip(h, 0, 1)
        # Fix residual

            if abs(diff) < 1e-14:

                    break

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if diff > 1e-13:
        idx = np.argsort(1.0 - h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff < 1e-14:
                break
    elif diff < -1e-13:
        idx = np.argsort(h)
        for i in idx:
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
            if diff > -1e-14:
                break
    return h

def optimize_pair(h, i, j):
    n = len(h)
    dx = 2.0 / n
    c = h[i] + h[j]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    span = t_hi - t_lo
    if span < 1e-14:
        return 0.0
    h_a = h.copy()
    h_a[i] = t_lo
    h_a[j] = c - t_lo
    corr_a = np.correlate(h_a, 1.0 - h_a, mode="full")
    h_b = h.copy()
    h_b[i] = t_hi
    h_b[j] = c - t_hi
    corr_b = np.correlate(h_b, 1.0 - h_b, mode="full")
    delta = corr_b - corr_a
    K = len(corr_a)
    ca_diff = corr_a[np.newaxis, :] - corr_a[:, np.newaxis]
    d_diff = delta[np.newaxis, :] - delta[:, np.newaxis]
    safe = np.abs(d_diff) > 1e-14
    s_mat = np.zeros((K, K))
    s_mat[safe] = ca_diff[safe] / (-d_diff[safe])
    valid = safe & (s_mat > 1e-9) & (s_mat < 1.0 - 1e-9)
    all_s = np.concatenate(([0.0, 1.0], s_mat[valid]))
    corr_all = corr_a[np.newaxis, :] + all_s[:, np.newaxis] * delta[np.newaxis, :]
    max_vals = np.max(corr_all, axis=1)
    best_idx = int(np.argmin(max_vals))
    s_best = all_s[best_idx]
    h[i] = t_lo + s_best * span
    h[j] = c - h[i]
    return float(max_vals[best_idx] * dx)

def pair_sweep(h, n, rng):
    order = rng.permutation(n)
    for idx in range(n // 2):
        i = int(order[2 * idx])
        j = int(order[2 * idx + 1])
        optimize_pair(h, i, j)
    return h

def round_vertices(h, n):
    h_r = np.zeros(n)
    h_r[h > 0.75] = 1.0
    h_r[(h >= 0.25) & (h <= 0.75)] = 0.5
    h_r = project_feasible(h_r, n)
    return h_r

def interpolate_h(h_from, n_from, n_to):
    x_old = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x_new = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    h_new = np.interp(x_new, x_old, h_from)
    return project_feasible(h_new, n_to)
