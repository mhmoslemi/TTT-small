    return float(np.max(corr * dx))

    corr = np.correlate(h, 1.0 - h, mode="full") * dx

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_max_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) * T
    shifted = np.clip(shifted, -80, 80)
    return m + np.log(np.sum(np.exp(shifted))) / T

def project_h(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def try_optimize(h_start, n, T, maxiter=300):
    h_start = project_h(h_start, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]

        return smooth_max_obj(x, T, n)

        h = project_h(res.x, n)
        return h, evaluate_c5(h)
    except Exception:
        h = project_h(h_start, n)
