import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, dx, alpha):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_step(h0, dx, target, n, alpha, maxiter=80, tol=1e-9):
    def obj(x):
        return _smooth_obj(x, dx, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, maxiter=maxiter, tol=tol)
    if res.success and np.all(res.x >= -1e-8) and np.all(res.x <= 1 + 1e-8):
        hx = np.clip(res.x, 0, 1)
        hx += (target - hx.sum()) / n
        hx = np.clip(hx, 0, 1)
        if abs(hx.sum() - target) < 1e-6:
            return hx, _c5(hx, dx)
    return h0, _c5(h0, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        n = 70
        h = np.zeros(n)
        h[0::2] = 1.0

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h, target)

    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with increasing alpha
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]
    for alpha in alphas:
        if tl() < 30:
            break
        for sub in range(3):
            if tl() < 20:
                break
            best_h, c5_val = _slsqp_step(best_h, dx, target, n, alpha,
                                          maxiter=60, tol=1e-8)
            if c5_val < best_c5:
                best_c5 = c5_val

    # Phase 2: Gradient descent refinement with smooth max
    if tl() > 40:
        alpha = 50.0
        lr = 0.01
        for it in range(800):
            if tl() < 30:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
            m = np.max(corr)
            s = alpha * (corr - m)
            w = np.exp(s) / np.sum(np.exp(s))
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                mk = k - (n - 1)
                wk = w[k]
                if mk >= 0:
                    idx1 = np.arange(mk, n)
                    grad[idx1] += wk * (1.0 - best_h[idx1 - mk])
                    idx2 = np.arange(0, n - mk)
                    grad[idx2] -= wk * best_h[idx2 + mk]
                else:
                    mp = -mk
                    idx1 = np.arange(0, n - mp)
                    grad[idx1] += wk * (1.0 - best_h[idx1 + mp])
                    idx2 = np.arange(mp, n)
                    grad[idx2] -= wk * best_h[idx2 - mp]
            grad *= dx
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            direction = -grad / gnorm
            for _ in range(25):
                nh = _project(best_h + lr * direction, target)
                nc5 = _c5(nh, dx)
                if nc5 < best_c5 - 1e-12:
                    best_h = nh
                    best_c5 = nc5
                    break
                lr *= 0.7
            else:
                lr *= 0.5
                if lr < 1e-7:
                    break
            lr = min(lr * 1.1, 0.05)

            h_try[j] -= d
        h_try = _project(h_try, target)
        c5_try = _c5(h_try, dx)
        if c5_try < best_c5 + 0.01:

                                         maxiter=40, tol=1e-8)

                best_c5 = c5_ref

    if tl() > 30:
        for alpha in [200.0, 500.0]:
            if tl() < 20:

                                          maxiter=50, tol=1e-10)
            if c5_val < best_c5:

    best_h += (target - best_h.sum()) / n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _c5(best_h, dx)

    return best_h.tolist(), float(best_c5), n
