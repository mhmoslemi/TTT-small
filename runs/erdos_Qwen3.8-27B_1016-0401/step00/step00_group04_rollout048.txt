import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 50.0
_g_target = 0.0
_g_n = 0

def _obj(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    a = _g_alpha
    m = a * np.max(corr)
    ac = a * corr - m
    val = dx * (m + np.log(np.sum(np.exp(ac)))) / a
    pen = 10.0 * (np.sum(x) - _g_target) ** 2
    return val + pen

def _grad(x):
    n = _g_n
    dx = 2.0 / n
    a = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = a * np.max(corr)
    ac = a * corr - m
    es = np.exp(ac)
    w = es / np.sum(es)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - x[:n - s])
            grad[:n - s] -= wk * x[s:]
        else:
            t = -s
            grad[:n - t] += wk * (1.0 - x[t:])
            grad[t:] -= wk * x[:n - t]
    grad *= dx
    grad += 20.0 * (np.sum(x) - _g_target)
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    idx = np.arange(n_new) * n_old / float(n_new)
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n_old - 1)
    frac = idx - i0
    h_new = h[i0] * (1.0 - frac) + h[i1] * frac
    return make_feasible(h_new, n_new)

def nlp_solve(h, n, alpha, max_iter):
    global _g_alpha, _g_target, _g_n
    _g_alpha = alpha
    _g_target = n / 2.0
    _g_n = n
    h = make_feasible(h.copy(), n)
    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_out = make_feasible(res.x, n)
    except Exception:
        h_out = h
    return h_out

def binary_round(h, n):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(3):
        improved = False
        for i in range(n):
            if 0.15 < best[i] < 0.85:
                for val in (0.0, 1.0):
                    ht = best.copy()
                    ht[i] = val
                    ht = make_feasible(ht, n)
                    c5 = compute_c5(ht)
                    if c5 < best_c5 - 1e-10:
                        best_c5 = c5
                        best = ht.copy()
                        improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = None

    best_h = h0.copy()

    # Phase 1: NLP at native resolution, increasing alpha
    for alpha in [30.0, 60.0, 120.0, 250.0]:
        if budget_s - (time.time() - t0) < 50:

        h_try = nlp_solve(best_h, n, alpha, 400)

        hb, cb = binary_round(best_h, n)
        if cb < best_c5 - 1e-10:

            best_h = hb.copy()
        h_try = nlp_solve(best_h, n, 200.0, 300)
        c5 = compute_c5(h_try)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
