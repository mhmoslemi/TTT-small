Another idea: think of it as a scheduling problem. We have n positions, each can hold mass in [0,1], total mass = n/2. We want to arrange the mass so that for every shift, the overlap between h and (1-h shifted) is small.

Let me think about this differently. The overlap at lag k is:
O_k = sum_i h[i] * (1 - h[i+k])

We want max_k O_k * dx to be small.

Note that: sum_k (for k=0 to n-1) O_k = sum_k sum_i h[i]*(1-h[i+k])
= sum_i h[i] * sum_k (1-h[i+k])
= sum_i h[i] * (n - sum_k h[i+k])
= n * sum_i h[i] - sum_i h[i] * sum_k h[i+k]
= n * n/2 - sum_i h[i] * (sum over all h values that are "reachable" from i)

import time

def eval_c5(h, dx):
    cor = np.correlate(h, 1-h, mode="full")
    return np.max(cor) * dx

def project(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            m = h < 1.0
            if m.any():
                h[m] += diff / m.sum()
                h = np.clip(h, 0, 1)
        else:
            m = h > 0.0
            if m.any():
                h[m] += diff / m.sum()
                h = np.clip(h, 0, 1)
    # Fine-tune
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            m = h < 1.0
            if m.any():
                h[m[0]] += min(diff, 1-h[m[0]])
        else:
            m = h > 0.0
            if m.any():
                h[m[0]] += max(diff, -h[m[0]])
    return h

def grad_c5(h, dx, temp):
    n = len(h)
    ov = np.correlate(h, 1-h, mode="full") * dx
    w = np.exp((ov - ov.max()) / temp)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        lag = idx - (n - 1)
        wi = w[idx] * dx
        if lag == 0:
            grad += wi * (1 - 2*h)
        elif lag > 0:
            # O_lag = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j] = [j < n-lag]*(1-h[j]) - [j >= lag]*h[j-lag]
            for j in range(n):
                g = 0.0
                if j < n - lag:
                    g += (1 - h[j])
                if j >= lag:
                    g -= h[j - lag]
                grad[j] += wi * g
        else:
            k = -lag
            # O_-k = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
            # d/dh[j] = [j >= k]*(1-h[j-k]) - [j < n-k]*h[j+k]
            for j in range(n):
                g = 0.0
                if j >= k:
                    g += (1 - h[j-k])
                if j < n - k:
                    g -= h[j+k]
                grad[j] += wi * g
    return grad
