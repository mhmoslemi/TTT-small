import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_all_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        mask2 = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask2) > 0:
            h[mask2] += diff2 / np.sum(mask2)
            h = np.clip(h, 0.0, 1.0)
    return h

def numerical_grad(h, eps=1e-5):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h[i] += eps
        cp = np.max(np.correlate(h, 1.0 - h, mode="full"))
        h[i] -= 2.0 * eps
        cm = np.max(np.correlate(h, 1.0 - h, mode="full"))
        h[i] += eps
        grad[i] = (cp - cm) / (2.0 * eps)
    return grad

def opt_subgrad(h, n_iter=400, lr0=0.04, decay=0.997, rng=None):
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for it in range(n_iter):
        grad = numerical_grad(h)
        h = h - lr * grad
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= decay
        if lr < 1e-5:
            break
    return best_h, best_c5

def local_search(h, n_trials=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_try = h.copy()
        np_ = rng.randint(1, 4)
        idx = rng.choice(n, size=np_, replace=False)
        h_try[idx] += rng.uniform(-0.15, 0.15, size=np_)
        h_try = project_feasible(h_try, n)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            h = h_try
    return h, best_c5

def binarize_attempt(h, n_trials=200, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        idx = rng.randint(n)
        h_try = h.copy()
        if h_try[idx] > 0.5:
            h_try[idx] = 1.0
        else:
            h_try[idx] = 0.0
        h_try = project_feasible(h_try, n)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            h = h_try
    return h, best_c5
