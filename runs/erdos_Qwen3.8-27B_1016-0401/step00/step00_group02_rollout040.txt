def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Iteratively fix the sum
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        if diff > 0:
            # Need to increase some values

                break
            h[mask] -= (-diff) / np.sum(mask)

    return h

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            inc = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + inc, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            dec = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - dec, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def updown_sample(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_local_search(h_bin, res, rng, max_iters, deadline):
    c5_cur = evaluate_c5(h_bin)
    h_work = h_bin.copy()
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(0, res))
        j = int(rng.integers(0, res))
        if i == j or h_work[i] == h_work[j]:
            continue
        h_work[i], h_work[j] = h_work[j], h_work[i]
        c5_new = evaluate_c5(h_work)
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
        else:
            h_work[i], h_work[j] = h_work[j], h_work[i]
    return h_work, c5_cur

def nm_optimize(x0, res, rng, deadline, max_iter=5000):
    def obj(h_flat):
        h = project_feasible(h_flat, res)
        return evaluate_c5(h)
    best_x = x0.copy()
    best_f = obj(best_x)
    n_restarts = 6
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if r == 0:
            x_start = best_x.copy()
        else:
            x_start = best_x + rng.standard_normal(res) * 0.12
            x_start = project_feasible(x_start, res)
        res_opt = minimize(obj, x_start, method='Nelder-Mead',
                          options={'maxiter': max_iter, 'xatol': 1e-10,
                                   'fatol': 1e-12, 'initial_simplex_size': 0.08})
        h_opt = project_feasible(res_opt.x, res)
        f_opt = evaluate_c5(h_opt)
        if f_opt < best_f:
            best_f = f_opt
            best_x = h_opt.copy()
    return best_x, best_f

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.97

    best_h = None
    best_c5 = np.inf
    best_n = 0

    h_start = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)

    resolutions = [45, 90, 135, 180, 270]

    for res in resolutions:
        if time.time() > deadline:
            break

        if h_start is not None:
            h0 = updown_sample(h_start, res)
        else:
            h0 = np.full(res, 0.5)
        h0 = project_feasible(h0, res)

        # Continuous Nelder-Mead optimization
        if time.time() < deadline:
            h_opt, c5_opt = nm_optimize(h0, res, rng, deadline, max_iter=4000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = res

        # Binary local search
        if res % 2 == 0 and time.time() < deadline:
            n_tries = 4
            for trial in range(n_tries):
                if time.time() > deadline:
                    break
                if trial == 0:
                    h_b = np.where(h_opt >= 0.5, 1.0, 0.0)
                else:
                    h_b = np.zeros(res)
                    idx = rng.permutation(res)
                    h_b[idx[:res // 2]] = 1.0
                h_b, c5_b = binary_local_search(h_b, res, rng,
                                                max_iters=5000, deadline=deadline)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    best_n = res
                    # Refine the continuous solution from this binary
                    if time.time() < deadline:
                        h_ref, c5_ref = nm_optimize(h_b, res, rng, deadline, max_iter=3000)
                        if c5_ref < best_c5:
                            best_c5 = c5_ref
                            best_h = h_ref.copy()
                            best_n = res

    # Final polish
    if best_h is not None and time.time() < deadline:
        final_res = best_n
        if time.time() < deadline:
            h_fin, c5_fin = nm_optimize(best_h, final_res, rng, deadline, max_iter=8000)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    return (best_h, best_c5, best_n)
