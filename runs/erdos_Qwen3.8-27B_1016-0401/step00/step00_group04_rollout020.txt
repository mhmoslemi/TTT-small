```python
import numpy as np
import time

def project_bounded_simplex(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(v)
    lo, hi = v.min() - target_sum, v.max() - 1
    # Actually: clamp(v - lam, 0, 1) should sum to target_sum
    # f(lam) = sum(clamp(v_i - lam, 0, 1)) is decreasing in lam
    # f(-inf) = n, f(+inf) = 0
    lo = -1 - v.max()  # definitely sum > target_sum
    hi = 1 - v.min() + 1  # definitely sum < target_sum
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_grad_c5(h, n):
    """Compute subgradient of max_k corr[k] (without dx factor)"""
    corr = np.correlate(h, 1-h, mode="full")
    max_val = np.max(corr)
    # Active set: all k where corr[k] is close to max
    active = corr >= max_val - 1e-10
    # For simplicity, use weighted average of gradients of active terms
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        shift = k - (n - 1)
        # c_k = sum_i h[i+shift] * (1-h[i]) for valid i
        i_min = max(0, -shift)
        i_max = min(n, n - shift)
        if i_min >= i_max:
            continue
        # Gradient w.r.t. h[j]:
        # d/dh[j] sum_{i=i_min}^{i_max-1} h[i+shift]*(1-h[i])
        # = (1-h[j-shift]) if j-shift in [i_min, i_max-1] i.e. j in [i_min+shift, i_max-1+shift]
        #   minus h[j+shift] if j in [i_min, i_max-1]
        
        # First part: j in [i_min+shift, i_max-1+shift] intersect [0, n-1]
        j1_lo = max(0, i_min + shift)
        j1_hi = min(n-1, i_max - 1 + shift)
        if j1_lo <= j1_hi:
            # grad[j] += (1 - h[j-shift]) for j in [j1_lo, j1_hi]
            idx = np.arange(j1_lo, j1_hi + 1)
            h_idx = idx - shift
            grad[idx] += (1 - h[h_idx])
        
        # Second part: j in [i_min, i_max-1] intersect [0, n-1]
        j2_lo = max(0, i_min)
        j2_hi = min(n-1, i_max - 1)
        if j2_lo <= j2_hi:
            idx = np.arange(j2_lo, j2_hi + 1)
            h_idx = idx + shift
            grad[idx] -= h[h_idx]
    
    return grad, corr

def projected_gradient_descent(h, n, dx, max_iter=5000, lr_init=0.01):
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    lr = lr_init
    
    for it in range(max_iter):
        h = np.clip(h, 0, 1)
        # Compute gradient
        grad, corr = compute_grad_c5(h, n)
        # Take step
        h_new = h - lr * grad
        # Project
        h_new = project_bounded_simplex(h_new, target_sum)
        
        # Line search (simple)
        val_new = compute_c5(h_new, dx)
        if val_new < best_val:
            best_h = h_new.copy()
            best_val = val_new
            h = h_new
        else:
            # Backtrack
            h = np.clip(0.5 * h + 0.5 * best_h, 0, 1)
            h = project_bounded_simplex(h, target_sum)
        
        # Decay learning rate
        lr *= 0.999
        
        if lr < 1e-8:
            break
    
    return best_h, best_val
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def project_bounded_simplex(v, target_sum):
    n = len(v)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def eval_and_grad(h, temp=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / temp)
    w /= w.sum()
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        wk = w[k]
        shift = k - (n - 1)
        i_min = max(0, -shift)
        i_max = min(n, n - shift)
        if i_min >= i_max:
            continue
        # d c_k / dh[j] = (1-h[j-shift]) [j in valid range for i+shift] 
        #                  - h[j+shift] [j in valid range for i]
        # First: j-shift in [i_min, i_max-1] → j in [i_min+shift, i_max-1+shift]
        j1lo = max(0, i_min + shift)
        j1hi = min(n - 1, i_max - 1 + shift)
        if j1lo <= j1hi:
            idx = np.arange(j1lo, j1hi + 1)
            grad[j1lo:j1hi + 1] += wk * (1.0 - h[idx - shift])
        # Second: j in [i_min, i_max-1]
        j2lo = max(0, i_min)
        j2hi = min(n - 1, i_max - 1)
        if j2lo <= j2hi:
            idx = np.arange(j2lo, j2hi + 1)
            grad[j2lo:j2hi + 1] -= wk * h[idx + shift]
    val = float(np.max(corr) * dx)
    return val, grad * dx

def pgd_optimize(h0, n, max_iter=3000, lr0=0.02, temp0=0.01):
    target = n / 2.0
    h = h0.copy()
    h = project_bounded_simplex(h, target)
    best_h = h.copy()
    best_val = eval_c5(h)
    lr = lr0
    temp = temp0
    
    for it in range(max_iter):
        val, grad = eval_and_grad(h, temp)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_bounded_simplex(h_new, target)
        new_val = eval_c5(h_new)
        
        if new_val <= best_val + 1e-10:
            h = h_new
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            lr *= 0.7
            if lr < 1e-10:
                lr = 1e-10
            continue
        
        # Decay
        if it % 200 == 199:
            lr *= 0.9
            temp *= 0.8
            temp = max(temp, 1e-5)
    
    return best_h, best_val

def refine_grid(h_old, n_new):
    """Upsample h from n_old to n_new points, preserving integral"""
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=1.0)  # periodic interp
    # Actually just linear interp
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_bounded_simplex(h_new, n_new / 2.0)
    return h_new

def random_binary_init(n, rng):
    """Random binary vector with sum = n/2"""
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def try_scipy_optimize(h0, n, timeout=60):

    target = n / 2.0
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = eval_c5(h0)
    
    def objective(z):

        corr = np.correlate(h, 1.0 - h, mode="full")
        # Softmax of max
        m = np.max(corr)
        s = np.sum(np.exp(corr / 0.01))

        if v_res < best_val:
```

```python
import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def project_bounded_simplex(v, target_sum):
    n = len(v)
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def eval_and_grad(h, temp=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        shift = k - (n - 1)
        i_min = max(0, -shift)
        i_max = min(n, n - shift)
        if i_min >= i_max:
            continue
        j1lo = max(0, i_min + shift)
        j1hi = min(n - 1, i_max - 1 + shift)
        if j1lo <= j1hi:
            idx = np.arange(j1lo, j1hi + 1)
            grad[j1lo:j1hi + 1] += wk * (1.0 - h[idx - shift])
        j2lo = max(0, i_min)
        j2hi = min(n - 1, i_max - 1)
        if j2lo <= j2hi:
            idx = np.arange(j2lo, j2hi + 1)
            grad[j2lo:j2hi + 1] -= wk * h[idx + shift]
    val = float(max_c * dx)
    return val, grad

def pgd_optimize(h0, n, max_iter=5000, lr0=0.0, temp0=0.01):
    if n % 2 == 0:
        target = n / 2.0
    else:
        target = n / 2.0
    if lr0 == 0.0:
        lr0 = 1.5 / n
    lam = project_bounded_simplex(h0, target)
    h = lam.copy()
    best_h = h.copy()
    best_val = eval_c5(h)
    lr = lr0
    temp = temp0
    for it in range(max_iter):
        val, grad = eval_and_grad(h, temp)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        h_new = h - lr * grad
        h_new = project_bounded_simplex(h_new, target)
        new_val = eval_c5(h_new)
        if new_val <= best_val + 1e-12:
            h = h_new
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            lr *= 0.6
            if lr < 1e-12:
                break
            continue
        if (it + 1) % 300 == 0:
            lr *= 0.85
            temp *= 0.7
            temp = max(temp, 2e-5)
    return best_h, best_val

def refine_grid(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_bounded_simplex(h_new, n_new / 2.0)
    return h_new

    h[ones] = 1.0
    return h

def local_perturb(h, n, rng, n_trials=200):
    target = n / 2.0
    best_h = h.copy()
    best_val = eval_c5(h)
    for _ in range(n_trials):
        h2 = best_h.copy()
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        amt = rng.uniform(0.05, 0.3)
        h2[i] += amt
        h2[j] -= amt
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_bounded_simplex(h2, target)
        v2 = eval_c5(h2)
        if v2 < best_val - 1e-12:
            best_val = v2
            best_h = h2.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_val = 10.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = random_binary_init(42, rng)
    n_init = len(h_init)
    h_init = project_bounded_simplex(h_init, n_init / 2.0)

    # Phase 1: PGD at initial grid
    t_limit1 = t0 + budget_s * 0.25
    h_opt, val = pgd_optimize(h_init, n_init, max_iter=8000, lr0=3.0 / n_init, temp0=0.01)
    if time.time() > t_limit1:
        h_opt, val = pgd_optimize(h_init, n_init, max_iter=3000, lr0=3.0 / n_init, temp0=0.01)
    if val < best_val:
        best_val = val
        best_h = h_opt.copy()

    # Local perturb
    h_p, val_p = local_perturb(best_h, len(best_h), rng, n_trials=500)
    if val_p < best_val:
        best_val = val_p
        best_h = h_p.copy()

    # Phase 2: scipy
    if time.time() - t0 < budget_s * 0.5:
        h_s, val_s = try_scipy(best_h, len(best_h), max_iter=200)
        if val_s < best_val:
            best_val = val_s
            best_h = h_s.copy()

    # Phase 3: Grid refinement
    if time.time() - t0 < budget_s * 0.75:
        for n_try in [64, 80, 96, 120, 144]:
            if time.time() - t0 > budget_s * 0.75:
                break
            h_ref = refine_grid(best_h, n_try)
            h_new, val_new = pgd_optimize(h_ref, n_try, max_iter=4000, lr0=2.0 / n_try, temp0=0.008)
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()
            h_p2, val_p2 = local_perturb(best_h, len(best_h), rng, n_trials=300)
            if val_p2 < best_val:
                best_val = val_p2
                best_h = h_p2.copy()

    # Phase 4: Random restarts
    if time.time() - t0 < budget_s * 0.9:
        for n_try in [42, 56, 64, 80, 100]:
            if time.time() - t0 > budget_s * 0.9:
                break
            h_rand = random_binary_init(n_try, rng)
            h_new, val_new = pgd_optimize(h_rand, n_try, max_iter=4000, lr0=2.5 / n_try, temp0=0.012)
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()

    # Final polish
    if time.time() - t0 < budget_s * 0.97:
        n_b = len(best_h)
        h_fin, val_fin = pgd_optimize(best_h, n_b, max_iter=5000, lr0=0.8 / n_b, temp0=0.003)
        if val_fin < best_val:
            best_val = val_fin
            best_h = h_fin.copy()
        h_p3, val_p3 = local_perturb(best_h, n_b, rng, n_trials=500)
        if val_p3 < best_val:
            best_val = val_p3
            best_h = h_p3.copy()

    h_final = project_bounded_simplex(best_h, len(best_h) / 2.0)
    final_val = eval_c5(h_final)
    return h_final, final_val, len(h_final)
```
