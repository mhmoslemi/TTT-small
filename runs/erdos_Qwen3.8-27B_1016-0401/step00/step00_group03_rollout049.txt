import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def obj_grad(x, dx, n, target, pw):
    corr = np.correlate(x, 1.0 - x, mode="full")
    c5v = float(np.max(corr)) * dx
    ks = int(np.argmax(corr))
    d = ks
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jm = j - d
        jp = j + d
        if 0 <= jm < n:
            g += 1.0 - x[jm]
        if 0 <= jp < n:
            g += -x[jp]
        grad[j] = g
    grad *= dx
    s = np.sum(x) - target
    c5v += pw * s * s
    if abs(s) > 1e-15:
        grad += 2.0 * pw * s
    return c5v, grad

def obj_only(x, dx, n, target, pw):
    v, _ = obj_grad(x, dx, n, target, pw)
    return v

def sum_constraint(x, target):
    return np.sum(x) - target

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            tr = room.sum()
            if tr < 1e-14:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff * room / tr)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            th = h[mask].sum()
            if th < 1e-14:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff * h[mask] / th)
    return np.clip(h, 0.0, 1.0)

def make_random_h(n, rng):
    half = n // 2
    pos = rng.choice(n, size=half, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    h += rng.randn(n) * 0.1
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def make_periodic_h(n, period):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < period // 2 else 0.0
    return project_feasible(h, n)

        target = n / 2.0
        pw = 100.0
        bounds = [(0.0, 1.0)] * n

        starts = []
        if best_h is not None and len(best_h) == n:
            starts.append(best_h.copy())
        for r in range(3):
            starts.append(make_random_h(n, rng))
        if n >= 12:
            starts.append(make_periodic_h(n, 4))
        if n >= 12:
            starts.append(make_periodic_h(n, 3))
        if n >= 10:
            starts.append(make_periodic_h(n, 5))

        for h_start in starts:
            if time.time() - t0 > budget_s - 10:
                break
            try:
                res = minimize(obj_grad, h_start.copy(), method='L-BFGS-B',
                               args=(dx, n, target, pw),
                               bounds=bounds, jac=True,
                               options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})

                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
            except Exception:
                pass

        if best_n == n and best_h is not None and time.time() - t0 < budget_s - 40:
            cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (target,)}]
            try:
                res2 = minimize(obj_only, best_h.copy(), method='SLSQP',
                                args=(dx, n, target, pw),
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 300, 'ftol': 1e-12})
                h_new2 = np.clip(res2.x, 0.0, 1.0)

            except Exception:

        for attempt in range(8):

            scale = 0.05 / (attempt + 1)
            h_try = np.clip(best_h + rng.randn(n) * scale, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            try:

                               bounds=bounds, jac=True,
                               options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    if best_h is not None:
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
