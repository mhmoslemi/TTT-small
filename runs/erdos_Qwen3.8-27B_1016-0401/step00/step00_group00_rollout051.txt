minimize t

This is a SOCP or MILP problem (actually, the constraints corr_k ≤ t/dx are quadratic, so it's a QP or non-convex QP).

Wait, corr_k = sum_i h[i] * (1-h[i+k]) for valid i. This is sum h[i] - sum h[i]*h[i+k]. The first term is linear, the second is quadratic (bilinear). So the constraint is:

This is a non-convex quadratic constraint (because of the -h[i]*h[i+k] term, which makes it non-convex).

    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):

        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_objective(n):
    target = n / 2.0
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        s = h.sum()
        if s < 1e-12:
            return 1.0
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        return compute_c5(h)
    return obj

        return x0.copy()

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def project_to_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def objective_fn(x, n):
    target = n / 2.0
    h = np.clip(x, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return 1.0
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return compute_c5(h)


def nm_optimize(x0, n, maxiter=20000):
    try:
        res = minimize(objective_fn, x0, args=(n,), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return x0.copy()


def es_optimize(x0, n, sigma_init, time_limit, rng):
    mu, lam = 4, 16
    sigma = sigma_init
    x_curr = x0.copy()
    f_curr = objective_fn(x0, n)
    best_x = x0.copy()
    best_f = f_curr
    t_start = time.time()
    stalled = 0
    while time.time() - t_start < time_limit:
        offspring = []
        for _ in range(lam):
            x = x_curr + sigma * rng.standard_normal(n)
            x = project_to_feasible(x)
            f = objective_fn(x, n)
            offspring.append((f, x))
        offspring.sort(key=sort_by_first)
        if offspring[0][0] < best_f:
            best_x = offspring[0][1].copy()
            best_f = offspring[0][0]
            stalled = 0
        else:
            stalled += 1
        sel = np.array([x for _, x in offspring[:mu]])
        x_curr = np.mean(sel, axis=0)
        x_curr = project_to_feasible(x_curr)
        spread = offspring[-1][0] - offspring[0][0]
        if spread < 1e-10:
            sigma *= 0.7
        elif offspring[0][0] < f_curr:
            sigma *= 1.06
        else:
            sigma *= 0.92
        sigma = min(max(sigma, 0.0008), 1.0)
        f_curr = objective_fn(x_curr, n)
        if stalled > 80:
            x_curr = best_x + 0.12 * rng.standard_normal(n)
            x_curr = project_to_feasible(x_curr)
            sigma = 0.08
            stalled = 0
    return best_x, best_f


def sort_by_first(t):
    return t[0]


def init_binary(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    n_ones = int(target)
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] = 0.5
    return project_to_feasible(h)

    phase = rng.rand() * 6.2832
    freq = rng.randint(2, 6)
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.4 * np.sin(6.2832 * float(freq) * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_to_feasible(h)


def init_random(n, rng):
    h = rng.rand(n)
    return project_to_feasible(h)
