def project_onto_feasible(h, target_sum):
    # Project h onto {0<=x<=1, sum(x)=target_sum}
    # Use water-filling: find lambda such that sum(clip(h-lambda, 0, 1)) = target_sum
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix rounding
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_c = np.exp(beta * shifted)
    sum_exp = np.sum(exp_c)
    obj = dx * (max_c + np.log(sum_exp) / beta)
    w = dx * exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        m = i - (n - 1)
        if m >= 0:
            idx1 = np.arange(m, n)
            grad[idx1] += wi * (1.0 - h[idx1 - m])
            idx2 = np.arange(n - m)
            grad[idx2] -= wi * h[idx2 + m]
        else:
            p = -m
            idx1 = np.arange(n - p)
            grad[idx1] += wi * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= wi * h[idx2 - p]
    return obj, grad


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return np.clip(proj, 0.0, 1.0)


def _cons_fun(x, ts):
    return np.sum(x) - ts


def _cons_jac(x):
    return np.ones(len(x))


def optimize_smooth(h0, dx, ts, t_limit):
    n = len(h0)
    h = project_feasible(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    for beta in [100.0, 300.0, 1000.0, 5000.0]:
        if time.time() > t_limit:
            break
        cons = [{'type': 'eq', 'fun': partial(_cons_fun, ts=ts), 'jac': _cons_jac}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj_grad, h, args=(dx, beta),
                           jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass

    beta = 5000.0
    lr = 1.0
    stall = 0
    for it in range(500000):
        if time.time() > t_limit:
            break
        _, grad = smooth_obj_grad(h, dx, beta)
        h_new = project_feasible(h - lr * grad, ts)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.05, 5.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.92
        h = h_new
        if stall > 3000 or lr < 1e-7:
            break

    return best_h, best_c5


def binary_local_search(h0, dx, t_limit):
    n = len(h0)
    ts = n // 2
    best = (h0 > 0.5).astype(float).copy()
    oc = int(np.sum(best))
    if oc > ts:
        idxs = np.where(best == 1)[0]
        np.random.shuffle(idxs)
        for i in idxs[:oc - ts]:
            best[i] = 0.0
    elif oc < ts:
        idxs = np.where(best == 0)[0]
        np.random.shuffle(idxs)
        for i in idxs[:ts - oc]:
            best[i] = 1.0
    best_c5 = compute_c5(best, dx)

    for _ in range(200):
        if time.time() > t_limit:
            break
        improved = False
        zeros = np.where(best == 0)[0]
        ones = np.where(best == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        zlist = list(zeros)
        olist = list(ones)
        for zi in zlist:
            if time.time() > t_limit:
                return best, best_c5
            for oi in olist:
                if zi == oi:
                    continue
                best[zi] = 1.0
                best[oi] = 0.0
                c5_new = compute_c5(best, dx)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                best[zi] = 0.0
                best[oi] = 1.0
            if improved:
                break
        if not improved:
            break

    return best, best_c5
