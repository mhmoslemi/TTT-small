```python
    target = best_n / 2.0
    
    # Nelder-Mead
    def nm_objective(x):

        pen = 100.0 * (np.sum(xc) - target) ** 2

                   options={'maxiter': 5000, 'fatol': 1e-12, 'xatol': 1e-12})
    best_h = project_constraint(np.clip(res.x, 0, 1), target)
    best_c5 = evaluate_c5(best_h)
    
    if time_left() < phase1_budget * 0.3:
        pass  # skip remaining phase 1
    else:
        # Coordinate descent multiple passes
        for _ in range(3):
            if time_left() < 50:
                break
            best_h, best_c5 = coordinate_descent(best_h, target, 2000, rng)
        
        # Simulated annealing
        if time_left() > 50:
            best_h, best_c5 = anneal(best_h, target, 2000, rng)
    
    # Phase 2: Multi-resolution
    resolutions = [2, 2, 2]  # 45->90->180->360
    for fac in resolutions:
        if time_left() < 80:

        h_up = upsample(best_h, fac)
        new_n = len(h_up)
        target = new_n / 2.0
        phase_budget = min(time_left() * 0.5, 400)
        
        h_cur = h_up.copy()
        c5_cur = evaluate_c5(h_cur)
        if c5_cur < best_c5:

        n_rounds = max(2, int(phase_budget / 100))
        for r in range(n_rounds):
            if time_left() < 30:

            if r == 0:
                h_start = h_up
            else:
                sigma = 0.01 * (1 + 0.5 * r)
                h_start = project_constraint(h_up + rng.normal(0, sigma, new_n), target)
            
            iters = max(500, int(phase_budget / n_rounds / 0.05))  # rough
            h_try, c5_try = coordinate_descent(h_start, target, iters, rng)
            if c5_try < c5_cur:
                h_cur, c5_cur = h_try, c5_try
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
        
        best_h = h_cur.copy()
        best_n = new_n
    
    # Final refinement
    if time_left() > 20:
        target = best_n / 2.0
        best_h, best_c5 = coordinate_descent(best_h, target, 1000, rng)
    
    # Ensure correctness
    best_h = project_constraint(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, best_n
```

```python
                   options={'maxiter': n_iter_max, 'rhobeg': 0.1, 'tol': 1e-10})
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_constraint(h_new, n_new / 2.0)

def make_triangle(n):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return project_constraint(h, n / 2.0)

def coordinate_descent(h, target, iters, rng):
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(best)
    deltas = np.array([0.005, 0.02, 0.05, 0.1, -0.005, -0.02, -0.05, -0.1])
    for it in range(iters):
        improved = False
        idx = rng.integers(n)
        for d in deltas[rng.integers(len(deltas))]:
            h_try = best.copy()
            h_try[idx] += d
            h_try = project_constraint(h_try, target)
            val = evaluate_c5(h_try)
            if val < best_val - 1e-14:
                best = h_try
                best_val = val
                improved = True
                break
        if not improved and it % 100 == 0:
            h_try = best + rng.normal(0, 0.02, n)
            h_try = project_constraint(h_try, target)
            val = evaluate_c5(h_try)
            if val < best_val:
                best = h_try
                best_val = val
    return best, best_val

def anneal(h, target, iters, rng):
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(best)
    cur = best.copy()
    cur_val = best_val
    T = 0.005
    for it in range(iters):
        T *= 0.9995
        i = rng.integers(n)
        delta = rng.normal(0, 0.04)
        h_try = cur.copy()
        h_try[i] += delta
        h_try = project_constraint(h_try, target)
        val = evaluate_c5(h_try)
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / max(T, 1e-8)):
            cur = h_try
            cur_val = val
        if val < best_val:
            best = h_try
            best_val = val
    return best, best_val

def nm_optimize(h, target, maxiter=3000):
    n = len(h)
    def obj(x):
        xc = np.clip(x, 0.0, 1.0)
        pen = 50.0 * (np.sum(xc) - target) ** 2
        return evaluate_c5(xc) + pen
    res = minimize(obj, h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'fatol': 1e-13, 'xatol': 1e-13})
    h_opt = project_constraint(np.clip(res.x, 0.0, 1.0), target)
    return h_opt, evaluate_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - t0))

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_triangle(45)

    best_n = len(best_h)
    best_c5 = evaluate_c5(best_h)
    target = best_n / 2.0

    # Phase 1: optimize at starting resolution
    h_nm, c5_nm = nm_optimize(best_h, target, maxiter=4000)
    if c5_nm < best_c5:
        best_h, best_c5 = h_nm, c5_nm
    best_h, best_c5 = coordinate_descent(best_h, target, 3000, rng)
    if time_left() > 60:
        best_h, best_c5 = anneal(best_h, target, 3000, rng)
    best_h, best_c5 = coordinate_descent(best_h, target, 2000, rng)

    # Try triangle as alternative start
    if time_left() > 100:
        h_tri = make_triangle(best_n)
        h_tri, c5_tri = nm_optimize(h_tri, target, maxiter=3000)
        h_tri, c5_tri = coordinate_descent(h_tri, target, 2000, rng)
        if c5_tri < best_c5:
            best_h, best_c5 = h_tri, c5_tri

    # Phase 2: multi-resolution
    factors = [2, 2, 2]
    for fac in factors:
        if time_left() < 100:
            break
        h_up = upsample(best_h, fac)
        new_n = len(h_up)
        target = new_n / 2.0
        c5_up = evaluate_c5(h_up)
        h_cur = h_up.copy()
        c5_cur = c5_up
        if c5_cur < best_c5:
            best_h, best_c5 = h_cur, c5_cur.copy() if False else (h_up.copy(), c5_cur)

        phase_time = min(time_left() * 0.55, 300)
        n_rounds = max(3, int(phase_time / 60))

        for r in range(n_rounds):
            if time_left() < 40:
                break
            if r == 0:
                h_start = h_up.copy()
            else:
                sigma = 0.008 * (1.0 + 0.4 * r)
                h_start = project_constraint(h_up + rng.normal(0, sigma, new_n), target)

            iters = max(800, int(4000 / max(1, new_n // 90)))
            h_try, c5_try = coordinate_descent(h_start, target, iters, rng)
            if c5_try < c5_cur:
                h_cur, c5_cur = h_try, c5_try
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try

        # Final NM at this resolution
        if time_left() > 50:
            h_nm2, c5_nm2 = nm_optimize(h_cur, target, maxiter=2000)
            if c5_nm2 < c5_cur:
                h_cur, c5_cur = h_nm2, c5_nm2
            if c5_cur < best_c5:
                best_h, best_c5 = h_cur.copy(), c5_cur

        best_h = h_cur.copy()
        best_n = new_n

    # Phase 3: final refinement
    if time_left() > 30:
        target = best_n / 2.0
        best_h, best_c5 = coordinate_descent(best_h, target, 3000, rng)
        best_h, best_c5 = anneal(best_h, target, 1500, rng)
        best_h, best_c5 = coordinate_descent(best_h, target, 2000, rng)

    # Ensure feasibility
    best_h = project_constraint(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
```
