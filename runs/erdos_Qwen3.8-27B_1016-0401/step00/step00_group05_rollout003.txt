    except Exception:
        return h.copy()

def local_search_2coord(h, n, target, rng, max_iters, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    step = 0.05
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Adaptive step
        if it % 5000 == 0 and it > 0:
            step *= 0.85
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        delta = rng.uniform(-step, step)
        h_try = h_curr.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h_curr = h_try
            best_c5 = c5_try
        elif it % 5000 == 2500:
            # Random restart kick
            idx = rng.choice(n, size=3, replace=False)
            h_curr[idx] += rng.uniform(-0.03, 0.03, size=3)
            h_curr = make_feasible(h_curr, n, target)
            best_c5 = compute_c5(h_curr)
    return h_curr, best_c5

def try_binary_base(n, rng, target):
    """Create a random binary solution with target sum, then smooth."""
    ones = int(round(target))
    h = np.zeros(n)
    idx = rng.permutation(n)[:ones]
    h[idx] = 1.0
    # Add small noise and smooth
    h += rng.uniform(-0.05, 0.05, n)
    h = make_feasible(h, n, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(70) * 0.5
    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, n, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [20, 40, 80, 160, 320, 640, 1280]:
        if time.time() > deadline - 60:
            break
        h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Local search
    if time.time() < deadline - 40:
        h_ls, c5_ls = local_search_2coord(best_h, n, target, rng, 30000, deadline - 40)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls

    # Phase 3: SLSQP refinement
    if time.time() < deadline - 30:
        for alpha in [500, 1000, 2000]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Try alternative starting points (binary-based)
    for trial in range(3):
        if time.time() > deadline - 50:
            break
        h_alt = try_binary_base(n, rng, target)
        # Optimize from this start
        for alpha in [30, 100, 400]:
            if time.time() > deadline - 50:
                break
            h_alt = slsqp_optimize(h_alt, n, target, alpha, maxiter=150)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_h = h_alt
            best_c5 = c5_alt

    # Phase 5: Final local search polish
    if time.time() < deadline - 10:
        h_ls, c5_ls = local_search_2coord(best_h, n, target, rng, 15000, deadline - 5)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls

    # Final validation
    best_h = make_feasible(best_h, n, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h.tolist(), best_c5, n
