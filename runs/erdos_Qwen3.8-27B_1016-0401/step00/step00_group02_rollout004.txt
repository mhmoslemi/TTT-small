import numpy as np
from scipy.optimize import minimize
import time

_n_current = 45


def project_to_feasible(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2} via bisection on threshold."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        proj = np.clip(h - tau, 0.0, 1.0)
        if proj.sum() > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def c5_value(h):
    """Compute C5 = max(corr * dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def objective_fn(x):
    """Objective with projection, uses global _n_current."""
    x = project_to_feasible(x, _n_current)
    return c5_value(x)


def time_left(t0, budget_s):
    return budget_s - (time.time() - t0)

    global _n_current
