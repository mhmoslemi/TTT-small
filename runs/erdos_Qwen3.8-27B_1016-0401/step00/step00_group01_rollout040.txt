def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_and_lags(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return c5_and_lags.max_val(h, dx), vals

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.clip(h[mask] + step, 0, 1)
        else:
            mask = h > 0.0001
            if mask.sum() == 0:
                break
            step = max(diff / mask.sum(), -0.5)
            h[mask] = np.clip(h[mask] + step, 0, 1)
    return h

    target = n / 2.0

             'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=cons,

        if res.success or res.fun < c5_value(h0, dx):
            return project_sum(res.x, target)
    except Exception:
        pass
    return project_sum(h0, target)
