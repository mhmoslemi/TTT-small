    n = len(h)
    dx = 2.0 / n
    comp = 1 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def local_search(h, n, rng, max_iters=100000):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    current_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h)

            # Update ones/zeros

            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            ones = np.append(ones, j)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except:
        h_init = None

    best_h = None
    best_c5 = 1.0

    # Phase 1: SA from initial construction
    if h_init is not None and len(h_init) > 0:
        h_b = to_binary(h_init)
        tl = min(250, budget_s * 0.25)
        h_opt, c5 = sa_search(h_b, rng, tl)
        best_h, best_c5 = h_opt, c5

    # Phase 2: Try greedy init + SA at n=70
    n70 = 70
    h_g = greedy_init(n70, rng)
    tl2 = min(150, (budget_s - 30 - (time.time() - t_start)) * 0.3)
    if tl2 > 10:
        h_opt, c5 = sa_search(h_g, rng, tl2)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 3: Random restarts at n=70
    for attempt in range(8):
        if time.time() - t_start > budget_s - 120:
            break
        h_r = np.zeros(n70)
        h_r[rng.choice(n70, n70 // 2, replace=False)] = 1.0
        tl3 = min(50, (budget_s - 120 - (time.time() - t_start)) / (8 - attempt))
        if tl3 < 5:
            break
        h_opt, c5 = sa_search(h_r, rng, tl3)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 4: Try larger n
    for n in [140, 200, 300, 500]:
        if time.time() - t_start > budget_s - 60:
            break
        target = n // 2
        h = np.zeros(n)
        h[rng.choice(n, target, replace=False)] = 1.0
        tl4 = min(60, (budget_s - 60 - (time.time() - t_start)) / 3)
        if tl4 < 5:
            break
        h_opt, c5 = sa_search(h, rng, tl4)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 5: Continuous refinement on best binary
    if best_h is not None and time.time() - t_start < budget_s - 20:
        tl5 = min(80, budget_s - 20 - (time.time() - t_start))
        h_c, c5 = continuous_refine(best_h, rng, tl5)
        if c5 < best_c5:
            best_h, best_c5 = h_c, c5

    if best_h is None:
        n = 70
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
