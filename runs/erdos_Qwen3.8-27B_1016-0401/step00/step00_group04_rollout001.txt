def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_binary(h):
    n = len(h)
    target = n // 2
    h = np.round(np.clip(h, 0.0, 1.0))
