def pgd_opt(h0, n_target, iters, tau0, tau1, lr):
    h = project_feasible(h0.copy(), n_target)
    best_h = h.copy()
    best_c5 = c5_of(h)
    for it in range(iters):
        frac = (it + 0.1) / iters
        tau = tau0 * (tau1 / tau0) ** frac
        g = grad_smooth(h, tau)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        step = lr * g / gn  # normalized step
        h = project_feasible(h - step, n_target)
        c5 = c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Initial
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 2.0 / 45)
    n = len(h_init)
    n_target = n / 2.0
    h_init = project_feasible(h_init, n_target)
    
    best_h = h_init.copy()
    best_c5 = c5_of(h_init)
    
    # Phase 1: SLSQP with decreasing tau
    taus = [10.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1]
    for tau in taus:
        if time.time() - t0 > budget_s * 0.25:
            break
        h_new = slsqp_opt(best_h, n_target, tau, maxiter=300)
        c5 = c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
    
    # Phase 2: PGD refinement
    if time.time() - t0 < budget_s * 0.4:
        best_h, best_c5 = pgd_opt(best_h, n_target, 300, 2.0, 0.01, 0.3)
    
    # Phase 3: Basin hopping
    amps = [0.4, 0.25, 0.5, 0.3, 0.2, 0.35, 0.15, 0.45, 0.2, 0.1]
    for amp in amps:
        if time.time() - t0 > budget_s * 0.7:
            break
        h_pert = project_feasible(best_h + amp * rng.standard_normal(n), n_target)
        h_cand = slsqp_opt(h_pert, n_target, 1.0, maxiter=200)
        c5 = c5_of(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand
            # Continue optimizing from new best
            h_cand2 = slsqp_opt(best_h, n_target, 0.3, maxiter=200)
            c52 = c5_of(h_cand2)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h_cand2

    rem = budget_s - (time.time() - t0)
