def c5_value(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h


class C5Objective:
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        h = project_h(x, self.n)
        return c5_value(h, self.n)

    current = c5_value(h, n)
