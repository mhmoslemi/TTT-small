    return float(np.max(corr) * dx)

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500)))) / beta

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

        return _soft_max_obj(x, beta)

        eps = 1e-7
        f0 = obj(x)

            xp = x.copy()
            xp[i] += eps
            g[i] = (obj(xp) - f0) / eps - 0  # forward diff
        return g

        return np.sum(x) - n / 2.0

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _corr_vec(h):

    return np.correlate(h, 1.0 - h, mode="full") * dx


def _softmax_obj(h, beta):
    c = _corr_vec(h)
    m = c.max()
    sh = np.clip(beta * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(sh))) / beta


def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(1000):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_opt(h0, beta, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n}]
    res = minimize(_softmax_obj, h0, args=(beta,),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})

def _cd_lagrangian(h, n_sweeps=100, rng=None):
    n = len(h)
    h = h.copy()
    lam = 0.0
    target = 0.5 * n
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng else np.arange(n)
        for i in order:
            # Compute A_k and B_k for all k
            # corr[j] = sum_i h[i]*(1-h[i-lag_j])
            # ∂corr[j]/∂h[i] = (1-h[i-lag_j])*I(0<=i-lag_j<n) - h[i+lag_j]*I(0<=i+lag_j<n)
            
            lags = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
            
            # B[k] for each output index k (0..2n-2)
            lag_k = lags  # shape (2n-1,)
            
            # For coordinate i:
            # term1: (1-h[i-lag_k]) where 0<=i-lag_k<n
            idx1 = i - lag_k  # shape (2n-1,)
            valid1 = (idx1 >= 0) & (idx1 < n)
            # term2: h[i+lag_k] where 0<=i+lag_k<n
            idx2 = i + lag_k
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            B = np.zeros(2*n-1)
            B[valid1] += 1.0 - h[idx1[valid1]]
            B[valid2] -= h[idx2[valid2]]
            
            # A[k] = corr[k] - B[k]*h[i]
            corr_all = np.correlate(h, 1.0-h, mode="full") * (2.0/n)
            A = corr_all - B * h[i]
            
            # Minimize max_k (A[k] + B[k]*v) + lam*v over v in [0,1]
            # = max_k (A[k] + (B[k]+lam)*v)
            B2 = B + lam
            
            # Find kink points: v = (A[k]-A[j])/(B2[j]-B2[k]) for B2[k]!=B2[j]
            # Plus v=0, v=1
            candidates = [0.0, 1.0]
            
            # To avoid O(K^2) pairwise computation, use a simpler approach:
            # The function is convex piecewise linear. 
            # We can do a ternary search or just evaluate at many points.
            # For K=89 (n=45), pairwise is 89*88/2 ≈ 3900, which is fine.
            
            # Actually, let's just do a fine grid search + refinement
            # Or use the fact that the minimum is at a kink or boundary.
            
            # Efficient approach: the value at v is max_k (A[k]+B2[k]*v)
            # At the optimum v*, the subgradient [min active slope, max active slope] contains 0
            # i.e., there exist active k's with B2[k]*v*+A[k] = f(v*) and B2 takes both signs
            # Or v* is at a boundary
            
            # Simple approach: evaluate at O(K) critical points
            # Sort by slope, compute upper envelope using convex hull trick
            # But that's overkill. Let's just evaluate at pairwise intersections.
            
            K = 2*n - 1
            # Only consider pairs where B2 values differ
            for kk in range(K):
                for jj in range(kk+1, K):
                    if abs(B2[kk] - B2[jj]) < 1e-15:
                        continue
                    v = (A[jj] - A[kk]) / (B2[kk] - B2[jj])
                    if 0.0 <= v <= 1.0:
                        candidates.append(v)
            
            # Evaluate at all candidates
            best_v = 0.0
            best_val = np.inf
            for v in candidates:
                val = np.max(A + B2 * v)
                if val < best_val:
                    best_val = val
                    best_v = v
            
            h[i] = best_v
        
        # Update lambda
        s = h.sum()
        diff = target - s
        if abs(diff) > 1e-6:
            # Subgradient: if sum > target, increase lam; else decrease
            lam += 0.5 * diff  # step size
        # More robust: bisection would be better, but this should work
    
    return h

    m = c.max()
    s = np.exp(np.clip(beta * (c - m), -500.0, 500.0))

import numpy as np
from scipy.optimize import minimize
import time


def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _sm(h, beta):
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h))
    m = c.max()
    s = np.exp(np.clip(beta * (c - m), -500.0, 500.0))
    return m + np.log(s.sum()) / beta


def _fix(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(1000):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        msk = (h < 1.0) if d > 0 else (h > 0.0)
        c = msk.sum()
        if c == 0:
            break
        h[msk] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h


def _opt(h, beta, nit):
    n = len(h)
    res = minimize(_sm, h, args=(beta,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq',
                                 'fun': lambda x: x.sum() - 0.5 * n}],
                   options={'maxiter': nit, 'ftol': 1e-12})
    return res.x


def _ls(h, niter, rng):
    n = len(h)
    h = h.copy()
    best = _c5(h)
    st = 0.15
    stall = 0
    for _ in range(niter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.02, st)
        if h[i] < h[j]:
            d = -d
        if not (0.0 <= h[i] + d <= 1.0 and 0.0 <= h[j] - d <= 1.0):
            continue
        h[i] += d
        h[j] -= d
        c = _c5(h)
        if c < best:
            best = c
            stall = 0
        else:
            h[i] -= d
            h[j] += d
            stall += 1
            st = max(0.002, st * 0.997)
            if stall > 5000:
                st = 0.1
                stall = 0
    return h, best

    x1 = np.arange(nn) * (2.0 / nn)
    return _fix(np.interp(x1, x0, h), nn)
