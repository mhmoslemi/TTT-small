    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return (1.0/alpha) * logsumexp(alpha * corr * dx)

                c5_new = c5_value(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                break

    # Phase 4: Random restarts + local optimize
    if best_h is not None:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        remaining = budget_s - (time.time() - t0)
        n_restarts = max(3, int(remaining / 25))
        for i in range(n_restarts):
            if time.time() - t0 > budget_s - 20:
                break
            scale = 0.1 + 0.25 * ((i % 4) / 3.0)
            h_try = best_h + rng.randn(n) * scale
            h_try = fix_sum(h_try, n)
            if i % 3 == 2:
                h_try = snap_binary(h_try, n)
            for alpha in [80, 300]:
                if time.time() - t0 > budget_s - 20:
                    break
                try:
                    res = minimize(smooth_obj, h_try, args=(alpha, 2000.0),
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 1500, 'ftol': 1e-14, 'gtol': 1e-12})
                    h_new = fix_sum(res.x, n)
                    c5_new = c5_value(h_new)
                    if c5_new < best_c5:
                        best_h, best_c5 = h_new.copy(), c5_new
                        h_try = h_new
                except Exception:
                    break

    # Phase 5: Final refinement with very high alpha
    if time.time() - t0 < budget_s - 5 and best_h is not None:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        for alpha in [2000, 6000]:
            if time.time() - t0 > budget_s - 3:
                break
            try:
                res = minimize(smooth_obj, best_h, args=(alpha, 5000.0),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-15})
                h_new = fix_sum(res.x, n)
                c5_new = c5_value(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass

    # Final cleanup
    best_h = fix_sum(best_h, len(best_h))
    final_c5 = c5_value(best_h)
    return best_h, final_c5, len(best_h)
