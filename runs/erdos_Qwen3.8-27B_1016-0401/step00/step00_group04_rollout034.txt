import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(h, n, rng):
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        for _ in range(n_ones - target):
            idx = ones_idx[rng.randint(len(ones_idx))]
            h_bin[idx] = 0.0
            ones_idx = np.where(h_bin == 1)[0]
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        for _ in range(target - n_ones):
            idx = zeros_idx[rng.randint(len(zeros_idx))]
            h_bin[idx] = 1.0
            zeros_idx = np.where(h_bin == 0)[0]
    return h_bin

def max_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def local_search(h, n, rng, deadline):
    dx = 2.0 / n
    best_h = h.copy()
    best_mc = max_corr(h)
    stuck = 0

    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        cur_mc = max_corr(h)
        best_flip = None
        best_nm = cur_mc

        if n <= 64:
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    nm = max_corr(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    if nm < best_nm:
                        best_nm = nm
                        best_flip = (i, j)
        else:
            npairs = min(800, len(ones) * len(zeros))
            for _ in range(npairs):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                nm = max_corr(h)
                h[i] = 1.0
                h[j] = 0.0
                if nm < best_nm:
                    best_nm = nm
                    best_flip = (i, j)

        if best_flip is not None:
            i, j = best_flip
            h[i] = 0.0
            h[j] = 1.0
            if best_nm < best_mc:
                best_mc = best_nm
                best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            npert = min(stuck + 1, 5, len(ones) - 1, len(zeros) - 1)
            if npert <= 0:
                break
            for _ in range(npert):
                oi = rng.randint(len(ones))
                zi = rng.randint(len(zeros))
                h[ones[oi]] = 0.0
                h[zeros[zi]] = 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

    return best_h, best_mc * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(42)
        h0[:21] = 1.0
    n0 = len(h0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    starts = []
    starts.append(make_binary(h0, n0, rng))
    h_alt = np.zeros(n0)
    h_alt[::2] = 1.0
    starts.append(make_binary(h_alt, n0, rng))
    h_alt2 = np.zeros(n0)
    h_alt2[1::2] = 1.0
    starts.append(make_binary(h_alt2, n0, rng))
    for s in range(6):
        h_r = np.zeros(n0)
        h_r[:n0 // 2] = 1.0
        rng.shuffle(h_r)
        starts.append(h_r.copy())

    phase1_end = start + budget_s * 0.45
    time_per = (phase1_end - start) / len(starts)
    for sp in starts:
        t_end = start + time_per
        if time.time() > t_end:
            continue
        h_opt, c5_opt = local_search(sp.copy(), n0, rng, t_end)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n0

    phase2_end = start + budget_s * 0.82
    larger_ns = sorted(set([n0 + 2, n0 + 4, n0 + 8, n0 + 12, 50, 60, 80, 100]))
    for n_try in larger_ns:
        if n_try < n0 or n_try % 2 != 0 or time.time() > phase2_end:
            continue
        h_try = np.zeros(n_try)
        h_try[:n_try // 2] = 1.0
        rng.shuffle(h_try)
        t_end = min(phase2_end, time.time() + 90)
        h_opt, c5_opt = local_search(h_try, n_try, rng, t_end)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    if best_h is not None and time.time() < start + budget_s * 0.96:
        h_opt, c5_opt = local_search(best_h.copy(), best_n, rng, start + budget_s * 0.96)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if best_h is None:
        best_h = make_binary(h0, n0, rng)
        best_c5 = compute_c5(best_h, n0)
        best_n = n0

    # Verify constraint
    s = float(np.sum(best_h))
    if abs(s - best_n / 2.0) > 1e-9:
        best_h = make_binary(best_h, best_n, rng)
        best_c5 = compute_c5(best_h, best_n)

    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
