```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, n):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s == 0:
        h = np.zeros(n)
        h[:int(target)] = 1.0
        if target - int(target) > 0:
            h[int(target)] = target - int(target)
        return h
    # If sum is too high, reduce; if too low, increase
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    # Distribute the difference
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room < diff * (1 + 1e-10):
            h[:] = 1.0
            h[n-1] = target - (n-1)
            return h
        h += room * (diff / total_room)
    else:
        # Need to decrease sum: reduce elements that are > 0
        space = h.copy()
        total_space = np.sum(space)
        if total_space < -diff * (1 + 1e-10):
            h[:] = 0.0
            h[:int(target)] = 1.0
            return h
        h += space * (diff / total_space)
    return np.clip(h, 0.0, 1.0)

def optimize_slsqp(h0, time_limit):
    n = len(h0)
    dx = 2.0 / n
    h0 = project_feasible(h0.copy(), n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        return c5_eval(h, dx)
    
    def constraint_fun(h):
        return np.sum(h) - n / 2.0
    
    def constraint_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    best_h = h0.copy()
    best_c5 = c5_eval(h0, dx)
    start = time.time()
    
    result = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 300, 'ftol': 1e-12, 'iprint': 0})
    
    if result.x is not None:
        h_new = project_feasible(result.x, n)
        c5_new = c5_eval(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5, time.time() - start

def upsample_block(h, factor):
```

```python
    for r in np.linspace(0, 1, 11):
        h_r = (h >= r).astype(float)
        h_r = project_feasible(h_r, n)
        yield h_r
    # Also try keeping top-k ones and adjusting
    k = int(n / 2)
    idx = np.argsort(h)[::-1]
    for extra in range(0, 4):
        for rem in range(0, 4):
            h_b = np.zeros(n)
            h_b[idx[:k + extra]] = 1.0
            h_b = project_feasible(h_b, n)
            yield h_b

def random_pattern(n, rng):
    h = np.zeros(n)
    k = n // 2
    perm = rng.permutation(n)
    h[perm[:k]] = 1.0
    if n % 2 == 1:
        h[perm[k]] = 0.5
    return project_feasible(h, n)

def periodic_pattern(n, period, ones_per_period, rng):
    h = np.zeros(n)
    for start in range(0, n, period):
        offset = rng.random()
        for k in range(ones_per_period):
            pos = int(start + (offset + k / max(ones_per_period,1)) * period) % n
            h[pos] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    deadline = t_start + budget_s * 0.97

    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 45
        h_init = np.zeros(n_init)
        h_init[:n_init//2] = 1.0
        if n_init % 2 == 1:
            h_init[n_init//2] = 0.5

    best_h = h_init.copy()
    best_c5 = c5_eval(best_h, 2.0 / len(best_h))
    best_n = len(best_h)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return deadline - time.time()

    # Phase 1: Optimize at initial resolution
    if n_init <= 100 and remaining() > 5:
        h_opt, c5_opt = optimize_slsqp(h_init, min(15, remaining()))
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_init

    # Phase 2: Multi-resolution SLSQP for small n
    for target_n in [60, 90, 120]:
        if remaining() < 10:
            break
        if best_n <= target_n:
            h_curr = upsample(best_h, max(1, target_n // best_n)) if best_n % 2 == 0 or target_n % best_n == 0 else None
            if h_curr is None or len(h_curr) != target_n:
                h_curr = np.zeros(target_n)
                # Interpolate
                x_old = np.linspace(0, 2, best_n)
                x_new = np.linspace(0, 2, target_n)
                h_curr = np.interp(x_new, x_old, best_h)
        else:
            break
        h_curr = project_feasible(h_curr, target_n)
        t_alloc = min(10, remaining() * 0.3)
        h_opt, c5_opt = optimize_slsqp(h_curr, t_alloc)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, target_n

    # Phase 3: Gradient optimization at higher resolutions
    for target_n in [180, 270, 360, 540, 720]:
        if remaining() < 15:
            break
        if best_n == target_n:
            h_curr = best_h.copy()
        elif best_n > target_n:
            break
        else:
            # Upsample
            factor = target_n // best_n if target_n % best_n == 0 else 1
            if factor > 1:
                h_curr = upsample(best_h, factor)
                if len(h_curr) != target_n:
                    x_old = np.linspace(0, 2, len(h_curr))
                    x_new = np.linspace(0, 2, target_n)
                    h_curr = np.interp(x_new, x_old, h_curr)
            else:
                x_old = np.linspace(0, 2, best_n)
                x_new = np.linspace(0, 2, target_n)
                h_curr = np.interp(x_new, x_old, best_h)
        h_curr = project_feasible(h_curr, target_n)
        t_alloc = min(30, remaining() * 0.35)
        h_opt, c5_opt = optimize_projgrad(h_curr, target_n, t_alloc)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, target_n

    # Phase 4: Random restarts at best resolution
    if remaining() > 20:
        n_try = best_n
        n_restarts = int(min(15, remaining() / max(n_try, 1) * 5))
        n_restarts = max(n_restarts, 3)
        for r in range(n_restarts):
            if remaining() < 5:
                break
            if r == 0:
                h_r = random_pattern(n_try, rng)
            elif r == 1:
                h_r = random_pattern(n_try, rng)
                h_r = upsample(h_r, 1)
            else:
                # Blend random with best
                h_rand = random_pattern(n_try, rng)
                mix = 0.3 + 0.4 * rng.random()
                h_r = mix * h_rand + (1 - mix) * best_h
                h_r = project_feasible(h_r, n_try)
            t_alloc = min(5, remaining() * 0.6 / max(n_try, 1))
            h_opt, c5_opt = optimize_projgrad(h_r, n_try, t_alloc)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Phase 5: Binary rounding attempts
    if remaining() > 5:
        for n_bin in [best_n, best_n * 2 if best_n * 2 <= 900 else best_n]:
            if remaining() < 3:
                break
            h_b = np.zeros(n_bin)
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_bin)
            h_interp = np.interp(x_new, x_old, best_h)
            for thresh in np.linspace(0.2, 0.8, 7):
                h_b = (h_interp >= thresh).astype(float)
                h_b = project_feasible(h_b, n_bin)
                c5_b = c5_eval(h_b, 2.0 / n_bin)
                if c5_b < best_c5:
                    best_h, best_c5, best_n = h_b, c5_b, n_bin

    # Phase 6: Final fine-tuning at best resolution
    if remaining() > 8 and best_n <= 200:
        t_alloc = min(remaining() * 0.6, 20)
        h_opt, c5_opt = optimize_slsqp(best_h, t_alloc)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: One more gradient pass
    if remaining() > 5:
        t_alloc = min(remaining() * 0.5, 15)
        h_opt, c5_opt = optimize_projgrad(best_h, best_n, t_alloc)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = c5_eval(best_h, 2.0 / best_n)
    best_c5 = final_c5

    return (best_h, best_c5, best_n)
```
