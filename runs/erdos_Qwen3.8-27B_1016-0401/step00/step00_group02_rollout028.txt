def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not mask.any():
        return h
    if diff > 0:
        room = 1.0 - h[mask]
    else:
        room = h[mask]
    total = room.sum()
    if total < abs(diff) * (1 + 1e-10):
        h[mask] += np.sign(diff) * room
    else:
        h[mask] += np.sign(diff) * abs(diff) * room / total
    return np.clip(h, 0.0, 1.0)

def greedy_local_search(h, target_sum, deltas, max_rounds=10000, time_limit=None):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not mask.any():
        return h
    if diff > 0:
        room = 1.0 - h[mask]
    else:
        room = h[mask]
    total = room.sum()
    if total <= 0:
        return h
    h[mask] += np.sign(diff) * min(abs(diff), total) * (room / total)
    return np.clip(h, 0.0, 1.0)

def greedy_ls(h, target, deltas, max_rounds, time_limit=None):
    n = len(h)
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        prev_c5 = best_c5
        for i in range(n):
            ov = h[i]
            for d in deltas:
                v = ov + d
                if v < -1e-12 or v > 1.0 + 1e-12:
                    continue
                h2 = h.copy()
                h2[i] = min(1.0, max(0.0, v))
                h2 = repair_sum(h2, target)
                c5 = compute_c5(h2)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    h = h2
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def sa_opt(h, target, n_iters, T0, T1, time_limit=None):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_h = h.copy()
    cur_c5 = best_c5
    for it in range(n_iters):
        if time_limit and time.time() > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        npert = 1 if np.random.random() < 0.6 else 2
        idxs = np.random.choice(n, npert, replace=False)
        saved = [cur_h[i] for i in idxs]
        for i in idxs:
            cur_h[i] += np.random.uniform(-0.5, 0.5)
        cur_h = repair_sum(np.clip(cur_h, 0, 1), target)
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and np.random.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            for i, s in zip(idxs, saved):
                cur_h[i] = s
            cur_h = repair_sum(cur_h, target)
            cur_c5 = compute_c5(cur_h)
    return best_h, best_c5

def swap_search(h, target, n_iters, time_limit=None):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if time_limit and time.time() > time_limit:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if abs(h[i] - h[j]) < 1e-8:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = best_h[i], best_h[j]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    else:

    n = len(h)

    h = repair_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: Greedy local search (coarse deltas)
    tl = t0 + budget_s * 0.12
    h, c5 = greedy_ls(h, target, [-0.5, -0.3, -0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.3, 0.5],
                      3000, time_limit=tl)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n

    # Phase 2: Simulated annealing
    tl = t0 + budget_s * 0.35
    h, c5 = sa_opt(best_h.copy(), target, 800000, 0.06, 0.002, time_limit=tl)
    if c5 < best_c5:

    # Phase 3: Greedy (fine deltas) after SA
    tl = t0 + budget_s * 0.50
    h, c5 = greedy_ls(best_h.copy(), target,
                      [-0.2, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.2],
                      5000, time_limit=tl)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n

    # Phase 4: Pair swap
    tl = t0 + budget_s * 0.65
    h, c5 = swap_search(best_h.copy(), target, 8000000, time_limit=tl)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n

    # Phase 5: Second SA round
    tl = t0 + budget_s * 0.78
    h, c5 = sa_opt(best_h.copy(), target, 500000, 0.03, 0.001, time_limit=tl)
    if c5 < best_c5:

    # Phase 6: Try 2x upsample
    tl = t0 + budget_s * 0.90
    n2 = best_n * 2
    tgt2 = n2 / 2.0
    h2 = np.repeat(best_h, 2)
    h2 = repair_sum(h2, tgt2)
    h2, c5 = greedy_ls(h2, tgt2, [-0.3, -0.1, -0.05, 0.05, 0.1, 0.3], 800, time_limit=tl)
    if c5 < best_c5:

        h2 = np.repeat(best_h, 2)
        h2 = repair_sum(h2, tgt2)

    # Final cleanup
    best_h = repair_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
