import numpy as np
import time

def _eval(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _compute_S(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[n - 1:]

def _subgrad(h, n):
    S = _compute_S(h)
    vals = S.copy()
    top_k = min(5, n)
    idx_sorted = np.argsort(vals)[-top_k:]
    idx_sorted = idx_sorted[idx_sorted[idx_sorted >= 0]]
    if len(idx_sorted) == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for d in idx_sorted:
        d = int(d)
        term1 = np.zeros(n)
        if d > 0:
            term1[d:] = 1.0 - h[:n - d]
        else:
            term1[:] = 1.0 - h[:]
        term2 = np.zeros(n)
        if d < n:
            term2[:n - d] = -h[d:]
        grad += term1 + term2
    return grad / top_k

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-(np.minimum(1.0 - h, 1.0)))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0 and h[idx] < 1.0:
                adj = min(diff, 1.0 - h[idx])
                h[idx] += adj
                diff -= adj
            elif diff < 0 and h[idx] > 0.0:
                adj = max(diff, -h[idx])
                h[idx] += adj
                diff -= adj
            if abs(diff) < 1e-12:
                break

            cur_c = c_trial
        it += 1
        if it % 500 == 0:
            T = max(0.0005, T * 0.98)
            step_size = max(0.01, step_size * 0.98)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8
    best_h = None
    best_c5 = 1e9
    best_n = 47

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    # Phase A: optimize at n=47 starting from parent
    n = 47
    dx = 2.0 / n
    if h_parent is not None and len(h_parent) == n:
        h0 = _project(h_parent.copy(), n)
    else:
        h0 = _make_init(n, rng)

    t_sg_end = min(deadline, t0 + budget_s * 0.3)
    h_opt = _subgrad_opt(h0, n, dx, rng, t_sg_end)
    c5 = _eval(h_opt, dx)
    best_h, best_c5, best_n = h_opt.copy(), c5, n

    t_an_end = min(deadline, t0 + budget_s * 0.5)
    h_ans, c5_a = _anneal(h_opt, n, dx, rng, time.time(), t_an_end)
    if c5_a < best_c5:
        best_h, best_c5 = h_ans.copy(), c5_a

    # Phase B: try a different random init at n=47
    if time.time() < deadline - 60:
        h_rand = _make_init(n, rng)
        t_end_b = min(deadline, time.time() + budget_s * 0.15)
        h_opt2 = _subgrad_opt(h_rand, n, dx, rng, min(t_end_b, time.time() + 20))
        c5_2 = _eval(h_opt2, dx)
        if c5_2 < best_c5:
            best_h, best_c5 = h_opt2.copy(), c5_2
            h_sub = h_opt2
        else:
            h_sub = h_opt
        t_an2 = min(deadline, time.time() + 30)
        h_a2, c5_a2 = _anneal(h_sub, n, dx, rng, time.time(), t_an2)
        if c5_a2 < best_c5:
            best_h, best_c5 = h_a2.copy(), c5_a2

    # Phase C: try n=94 (even, allows purely binary)
    n2 = 94
    dx2 = 2.0 / n2
    if time.time() < deadline - 60:
        if best_h is not None:
            h0_2 = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, len(best_h)), best_h)
            h0_2 = _project(h0_2, n2)
        else:
            h0_2 = _make_init(n2, rng)
        t_sg2 = min(deadline - 20, time.time() + 30)
        h_opt3 = _subgrad_opt(h0_2, n2, dx2, rng, t_sg2)
        c5_3 = _eval(h_opt3, dx2)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h_opt3.copy(), c5_3, n2
        t_an3 = min(deadline - 5, time.time() + 40)
        h_a3, c5_a3 = _anneal(h_opt3, n2, dx2, rng, time.time(), t_an3)
        if c5_a3 < best_c5:
            best_h, best_c5, best_n = h_a3.copy(), c5_a3, n2

    # Phase D: final refinement on best solution
    if best_h is not None and time.time() < deadline - 5:
        n_final = best_n
        dx_final = 2.0 / n_final
        t_final = deadline
        h_fin, c5_fin = _anneal(best_h, n_final, dx_final, rng, time.time(), t_final)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.copy(), c5_fin

    # Final check
    if best_h is None:
        best_h = np.ones(47) * 0.5
        best_c5 = 0.5
        best_n = 47

    best_c5 = _eval(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
