                    vals = slopes * t + corr0
                    vals[center_idx] = center_val + t * (1 - t)
                    F_vals[k] = np.max(vals)

                best_t = t_grid[best_k]
                
                # Refine with local search
                if 0 < best_k < len(t_grid) - 1:

                            vals[center_idx] = center_val + t_new * (1 - t_new)
                            if np.max(vals) < F_vals[best_k]:
                                F_vals[best_k] = np.max(vals)
                                best_t = t_new
                        else:
                            break
                    # More precise refinement
                    for _ in range(30):
                        t1 = best_t - 0.01
                        t2 = best_t + 0.01
                        t1 = max(0, min(1, t1))
                        t2 = max(0, min(1, t2))
                        if t1 == t2:
                            break
                        vals1 = slopes * t1 + corr0
                        vals1[center_idx] = center_val + t1*(1-t1)
                        vals2 = slopes * t2 + corr0
                        vals2[center_idx] = center_val + t2*(1-t2)
                        v1 = np.max(vals1)
                        v2 = np.max(vals2)
                        if v1 < v2:
                            best_t = t1
                        elif v2 < v1:
                            best_t = t2
                        else:
                            break
                
                best_h[i] = best_t
            
            # Enforce constraint: sum(best_h) = n/2
            # Note: coordinate descent might violate this
            s = best_h.sum()
            if abs(s - target_sum) > 1e-6:
                # Simple redistribution
                best_h += (target_sum - s) / n
                best_h = np.clip(best_h, 0, 1)
                # Fix remaining error by adjusting a few entries
                diff = target_sum - best_h.sum()
                for _ in range(n):
                    if abs(diff) < 1e-9:
                        break
                    j = rng.randint(n)
                    adj = diff * 0.5
                    old = best_h[j]
                    best_h[j] = np.clip(old + adj, 0, 1)
                    diff = target_sum - best_h.sum()

            cur_c5 = compute_c5(best_h, dx)
        
        # Occasionally try random restart / perturbation
        if outer % 10 == 5 and time.time() - t0 < budget_s * 0.6:

            perturb_h = best_h.copy()
            mask = rng.random(n) < 0.3
            perturb_h[mask] = np.clip(perturb_h[mask] + rng.uniform(-0.5, 0.5, mask.sum()), 0, 1)
            # Fix constraint
            perturb_h += (target_sum - perturb_h.sum()) / n
            perturb_h = np.clip(perturb_h, 0, 1)
            p_c5 = compute_c5(perturb_h, dx)
            if p_c5 < best_c5:
                best_h = perturb_h
                best_c5 = p_c5
    
    # Final constraint check
    best_h += (target_sum - best_h.sum()) / n
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h, dx)
    
    return best_h, best_c5, n

3. Handle the constraint separately after each pass

Option 3 is simplest but might hurt convergence. Let me use a penalty: add a large penalty (sum(h) - n/2)² to the objective.

Let me just use the penalty approach: add (sum(h) - n/2)² * M to the objective, with M growing over iterations.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, target_sum, dx, rng, n_iters=30):
    n = len(h)
    center_idx = n - 1
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iters):

        rng.shuffle(order)

            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = h_save
            
            slopes = corr1 - corr0

            F_vals = np.empty(len(t_grid))

                F_vals[k] = np.max(vals)
            
            bk = int(np.argmin(F_vals))

            for _ in range(25):

                hi = best_t + 0.02
                lo = max(0.0, min(1.0, lo))
                hi = max(0.0, min(1.0, hi))
                if hi - lo < 1e-6:

                t_mid = (lo + hi) / 2.0
