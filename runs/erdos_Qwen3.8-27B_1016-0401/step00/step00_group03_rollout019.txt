            h_rand = rng.uniform(0, 1, n_try)
            h_rand = project_feasible(h_rand, target_try)
            starts.append(h_rand)

        for h_start in starts:
            if not time_ok(50):
                break
            h_opt, c_opt = projected_gradient_opt(h_start, target_try, 2000, 0.5, 0.008, time_ok)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt
            # Local polish
            if time_ok(30):
                h_ls, c_ls = local_search(h_opt, target_try, time_ok, max_rounds=15)
                if c_ls < best_c5:
                    best_c5, best_h = c_ls, h_ls

    # Phase 3: Final polish at best resolution
    if time_ok(30):
        n_best = len(best_h)
        target_best = n_best / 2.0
        h_pol, c_pol = projected_gradient_opt(best_h, target_best, 1000, 0.2, 0.003, time_ok)
        if c_pol < best_c5:
            best_c5, best_h = c_pol, h_pol
        if time_ok(20):
            h_ls2, c_ls2 = local_search(best_h, target_best, time_ok, max_rounds=20)
            if c_ls2 < best_c5:
                best_c5, best_h = c_ls2, h_ls2

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    final_c5 = compute_c5(best_h)
    # Re-project to ensure exact sum
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
