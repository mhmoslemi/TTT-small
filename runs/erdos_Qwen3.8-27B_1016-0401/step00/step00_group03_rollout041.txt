import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def grad_single_lag(h, d):
    """Gradient of correlation at lag corresponding to offset d = k - n + 1"""
    n = len(h)
    grad = np.zeros(n)
    # term1: (1-h[m+d]) for valid m+d in [0,n)
    lo1 = max(0, -d)
    hi1 = min(n, n - d)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
    # term2: -h[m-d] for valid m-d in [0,n)
    lo2 = max(0, d)
    hi2 = min(n, n + d)
    if hi2 > lo2:
        grad[lo2:hi2] += (-h[lo2 - d:hi2 - d])
    return grad

def objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find all active lags (within numerical tolerance)
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        d = k - n + 1
        grad += grad_single_lag(h, d)
    grad /= len(active)
    return max_val * dx, grad * dx

        else:
