With a = h, v = (1-h), length n:

When we swap:
- h[a] was 1, becomes 0
- h[b] was 0, becomes 1

Change in corr[k] due to h[a] 1→0:
- As first index: h[i+k] where i+k=a, i=a-k: old contribution h[a]*(1-h[a-k]) = (1-h[a-k]), new: 0. Δ = -(1-h[a-k]) = h[a-k]-1 (if 0≤a-k<n)
- As second index: h[i+k] where i=a: old contribution h[a+k]*(1-h[a]) = 0, new: h[a+k]*1 = h[a+k]. Δ = h[a+k] (if 0≤a+k<n)

Change in corr[k] due to h[b] 0→1:
- As first index: i+k=b, i=b-k: old h[b]*(1-h[b-k]) = 0, new: 1*(1-h[b-k]) = (1-h[b-k]). Δ = (1-h[b-k])

import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _fix_h(h, n, seed=0):
    """Make h binary with sum = n//2."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        for _ in range(cur - target):
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0
    elif cur < target:
        for _ in range(target - cur):
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1
    return h

def _hill_climb(h, rng, time_limit):
    """First-improvement hill climbing with random restarts on stuck."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    cur_c5 = best_c5
    
    stuck = 0
    while time.time() < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        
        h[a] = 0
        h[b] = 1
        new_c5 = _eval_c5(h)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            stuck = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
            stuck += 1
            if stuck > 100:
                # Perturbation: make 3-5 random swaps
                h = best_h.copy()
                for _ in range(rng.randint(3, 6)):
                    ones_p = np.where(h == 1)[0]
                    zeros_p = np.where(h == 0)[0]
                    h[ones_p[rng.randint(len(ones_p))]] = 0
                    h[zeros_p[rng.randint(len(zeros_p))]] = 1
                cur_c5 = _eval_c5(h)
                stuck = 0
    
    return best_h, best_c5

    rng = np.random.RandomState(seed)
    
    h0 = np.array(initial_h_values, dtype=np.float64)
    n = len(h0)
    h_init = _fix_h(h0, n, seed)
    
    best_h = h_init.copy()
    best_c5 = _eval_c5(best_h)
    
    # Phase 1: Hill climbing at initial n (50% of time)
    t1 = start + budget_s * 0.5
    while time.time() < t1:
        h_try = best_h.copy()
        h_try, c5_try = _hill_climb(h_try, rng, min(time.time() + 20, t1 - 1))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    # Phase 2: Try 2n grid (30% of time)
    t2 = start + budget_s * 0.8
    n2 = n * 2
    h2 = np.zeros(n2)
