import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def local_search(h, n, max_iters=50000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        # Try small transfer
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta < 1e-6:
            continue
        delta = rng.uniform(0.01, max_delta)
        
        # Forward: i -> j
        trial = h.copy()
        trial[i] -= delta
        trial[j] += delta
        c = compute_c5(trial)
        if c < best_c5:
            h = trial
            best_c5 = c
            continue
        
        # Backward: j -> i
        max_delta2 = min(h[j], 1.0 - h[i])
        if max_delta2 < 1e-6:
            continue
        delta2 = rng.uniform(0.01, max_delta2)
        trial = h.copy()
        trial[j] -= delta2
        trial[i] += delta2
        c = compute_c5(trial)
        if c < best_c5:
            h = trial
            best_c5 = c
    
    return h, best_c5

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if p[k] < 1e-10:

def make_feasible(h, n):

        else:

            if s > 1e-15:
