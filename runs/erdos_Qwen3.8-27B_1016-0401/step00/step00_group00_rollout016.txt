    n = len(h)
    lo, hi = float(h.min()) - 2.0, float(h.max()) + 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid

    n = len(h)

        m = n - shift
        if m > 0:
            g[:m] += 1.0 - h[shift:]
            g[shift:shift + m] -= h[:m]
    else:

        g[s:] += 1.0 - h[:n - s]
