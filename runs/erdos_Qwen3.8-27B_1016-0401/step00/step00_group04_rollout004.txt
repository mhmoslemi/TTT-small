    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / tau

    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # Smooth max
    m = np.max(corr)
    shifted = (corr - m) * tau
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (m + np.log(sum_exp) / tau) * dx
    weights = exp_vals / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-8:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_hi <= i_lo:
            continue
        idx_a = np.arange(i_lo + s, i_hi + s)
        idx_b = np.arange(i_lo, i_hi)
        grad[idx_a] += w * (1.0 - h[idx_b])
        grad[idx_b] += w * (-h[idx_a])
    grad *= dx  # scale by dx since obj includes dx
    
    return obj, grad

def project_bounded(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def optimize(h0, time_limit=60.0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    tau = 100.0
    lr = 0.02
    t0 = time.time()
    patience = 0
    
    for it in range(20000):
        if time.time() - t0 > time_limit:
            break
        
        obj, grad = smooth_obj_and_grad(h, tau)
        
        # Gradient descent step
        h_new = h - lr * grad
        h_new = project_bounded(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience > 200:
            lr *= 0.9
            patience = 0
        elif patience < 50 and lr < 0.1:
            lr *= 1.02
        
        h = h_new
        
        # Slowly increase tau (sharpen the max approximation)
        if it % 100 == 0:
            tau += 2.0
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = (corr - m) * tau
    sh -= np.max(sh)
    ev = np.exp(sh)
    se = np.sum(ev)
    w = ev / se
    obj = (m + np.log(se + 1e-300) / tau) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-9:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_hi <= i_lo:
            continue
        ia = np.arange(i_lo + s, i_hi + s)
        ib = np.arange(i_lo, i_hi)
        grad[ia] += wk * (1.0 - h[ib])
        grad[ib] += wk * (-h[ia])
    grad *= dx
    return obj, grad

def project_bounded(h, target):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def optimize_h(h0, time_limit=60.0, tau0=80.0):
    n = len(h0)
    target = n / 2.0
    h = h0.astype(float).copy()
    h = project_bounded(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = tau0
    lr = 0.03
    v = np.zeros(n)
    t0 = time.time()
    stale = 0
    for it in range(50000):
        if time.time() - t0 > time_limit:
            break
        _, grad = smooth_obj_grad(h, tau)
        v = 0.85 * v - lr * grad
        h_new = project_bounded(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        if stale > 300:
            lr *= 0.85
            v *= 0.5
            stale = 0
        elif stale < 100 and lr < 0.15:
            lr *= 1.03
        h = h_new
        if it % 200 == 0 and it > 500:
            tau = min(tau + 3.0, 500.0)
    return best_h, best_c5

def binary_search_refine(h, target, time_limit=30.0):
    n = len(h)
    h_b = np.where(h > 0.5, 1.0, 0.0)
    cur_nc1 = int(np.sum(h_b == 1))
    if cur_nc1 != target:
        if cur_nc1 > target:
            idx1 = np.where(h_b == 1.0)[0]
            order = np.argsort(h_b[idx1])
            for i in order[target - cur_nc1:]:
                h_b[idx1[i]] = 0.0
        else:
            idx0 = np.where(h_b == 0.0)[0]
            order = np.argsort(-h[idx0])
            for i in order[:target - cur_nc1]:
                h_b[idx0[i]] = 1.0
    best_h = h_b.copy()
    best_c5 = compute_c5(h_b)
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_swap_c5 = best_c5
        best_pair = None
        for ii in range(len(ones)):
            if time.time() - t0 > time_limit:
                break
            oi = ones[ii]
            h_b[oi] = 0.0
            for jj in range(len(zeros)):
                zi = zeros[jj]
                h_b[zi] = 1.0
                c5 = compute_c5(h_b)
                h_b[zi] = 0.0
                if c5 < best_swap_c5 - 1e-10:
                    best_swap_c5 = c5
                    best_pair = (oi, zi)
            h_b[oi] = 1.0
        if best_pair is not None:
            oi, zi = best_pair
            h_b[oi] = 0.0
            h_b[zi] = 1.0
            best_h = h_b.copy()
            best_c5 = best_swap_c5
            improved = True
    return best_h, best_c5

    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_bounded(h_new, n_new / 2.0)
