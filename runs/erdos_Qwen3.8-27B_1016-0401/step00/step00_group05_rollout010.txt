    grad = np.zeros(n)

import numpy as np
import time

def _eval(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def _proj(h, s):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h - m, 0.0, 1.0)) > s:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def _subgrad(h, beta=50.0):
    """Softmax-weighted subgradient for smoother optimization."""
    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    r_max = np.max(r)
    w = np.exp(beta * (r - r_max))
    w /= np.sum(w)
    
    g = np.zeros(n)
    # Only process lags with significant weight
    threshold = np.max(w) * 0.01
    active = np.where(w > threshold)[0]
    
    for m in active:
        k = m - n + 1
        wm = w[m]
        if k >= 0:
            g[k:] += wm * (1.0 - h[:n-k])
            g[:n-k] -= wm * h[k:]
        else:
            g[:n+k] += wm * (1.0 - h[-k:])
            g[-k:] -= wm * h[:n+k]
    return g, r

import numpy as np
import time

def _eval(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _proj(h, s):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h - m, 0.0, 1.0)) > s:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _coord_step(h, j, s):
    n = len(h)
    n_r = 2 * n - 1
    hj = h[j]
    B = np.zeros(n_r)
    for k in range(0, n):
        m = n - 1 + k
        if j >= k:
            B[m] += 1.0 - h[j - k]
        if j < n - k:
            B[m] += -h[j + k]
    for kp in range(1, n):
        m = n - 1 - kp
        if j < n - kp:
            B[m] += 1.0 - h[j + kp]
        if j >= kp:
            B[m] += -h[j - kp]
    c_cur = np.correlate(h, 1.0 - h, mode="full")
    ts = np.linspace(0.0, 1.0, 400)
    cv = c_cur[None, :] + B[None, :] * (ts[:, None] - hj)
    mx = np.max(cv, axis=1)
    bi = np.argmin(mx)
    if bi > 0 and bi < len(ts) - 1:
        ts2 = np.linspace(ts[bi - 1], ts[bi + 1], 200)
        cv2 = c_cur[None, :] + B[None, :] * (ts2[:, None] - hj)
        mx2 = np.max(cv2, axis=1)
        h[j] = ts2[np.argmin(mx2)]
    else:
        h[j] = ts[bi]
    return _proj(h, s)

def _grad_step(h, s, lr, beta=40.0):
    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    rmax = np.max(r)
    w = np.exp(beta * (r - rmax))
    w /= np.sum(w)
    thr = np.max(w) * 0.01
    active = np.where(w > thr)[0]
    g = np.zeros(n)
    for m in active:
        k = m - n + 1
        wm = w[m]
        if k >= 0:
            g[k:] += wm * (1.0 - h[:n - k])
            g[:n - k] -= wm * h[k:]
        else:
            g[:n + k] += wm * (1.0 - h[-k:])
            g[-k:] -= wm * h[:n + k]
    h_new = _proj(h - lr * g, s)
    return h_new

    t0 = time.time()
    dl = t0 + budget_s

    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Coordinate descent
    for sweep in range(2000):
        if time.time() > dl - 60:
            break
        for j in range(n):
            if time.time() > dl - 60:
                break
            h = _coord_step(h, j, s)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: Gradient + coordinate hybrid with restarts
    n_restarts = 0
    for restart in range(100):
        if time.time() > dl - 40:

        h = best_h.copy()
        # Add perturbation
        scale = 0.02 + 0.06 * rng.random()
        h = best_h + rng.normal(0, scale, n)
        h = _proj(h, s)

        lr = 0.3
        for it in range(3000):
            if time.time() > dl - 40:
                break
            if it % 10 == 0:
                # Gradient step
                h = _grad_step(h, s, lr, beta=30.0)
            else:
                # Coordinate step
                j = it % n
                h = _coord_step(h, j, s)
            c5 = _eval(h)
            if c5 < best_c5 - 1e-12:
                best_h = h.copy()
                best_c5 = c5
                lr = min(lr * 1.01, 0.8)

    final_c5 = _eval(best_h)
    return best_h, final_c5, n
