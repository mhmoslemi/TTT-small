def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def grad_c5(h, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(c))
    s = idx - (n - 1)
    ks = np.arange(n)
    grad = np.zeros(n)
    v1 = (ks - s >= 0) & (ks - s < n)
    grad[v1] += (1.0 - h[ks[v1] - s])
    v2 = (ks + s >= 0) & (ks + s < n)
    grad[v2] += (-h[ks[v2] + s])
