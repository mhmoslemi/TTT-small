import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr) * dx)

def c5_gradient(h, n):
    """Return (c5_value, gradient) for h of length n."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        for i in range(i_lo, i_hi + 1):
            j1 = i + s
            j2 = i
            grad[j1] += (1.0 - h[j2])
            grad[j2] += (-h[j1])
    
    n_active = len(active)
    grad /= n_active
    grad *= dx
    return max_val * dx, grad

def optimize_gradient(h0, n, max_iter=200, tol=1e-8):
    """SLSQP optimization of C5."""
    def obj(v):
        val, _ = c5_gradient(v, n)
        return val
    
    def obj_grad(v):
        val, g = c5_gradient(v, n)
        return val, g
    
    def eq_constraint(v):
        return np.sum(v) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_constraint}]
    
    result = minimize(obj_grad, h0, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': max_iter, 'ftol': tol})
    return result.x, result.fun

def local_search(h, n, max_iters=5000):
    """Local improvement by pairwise swaps."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        improved = False
        # Try small perturbations
        for i in range(n):
            for j in range(i+1, n):
                # Try setting h[i] to 1 and h[j] to 0 (or vice versa)
                # Actually, try swapping: if h[i]=1, h[j]=0, try h[i]=0, h[j]=1
                if abs(h[i] - 1) < 1e-6 and abs(h[j]) < 1e-6:
                    h_new = h.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-10:
                        h = h_new
                        best_c5 = c5_new
                        improved = True
                        break
                elif abs(h[i]) < 1e-6 and abs(h[j] - 1) < 1e-6:
                    h_new = h.copy()
                    h_new[i] = 1
                    h_new[j] = 0
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-10:
                        h = h_new
                        best_c5 = c5_new
                        improved = True
                        break
                else:

                    diff = min(h[i], 1-h[j]) * 0.1

∂output[m]/∂h[j] = (1-h[j-shift]) * I(j-shift ∈ [lo,hi]) + (-h[j+shift]) * I(j ∈ [lo,hi])

def c5_obj_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        for i in range(lo, hi + 1):
            j1 = i + s
            j2 = i
            grad[j1] += (1.0 - h[j2])
            grad[j2] += (-h[j1])

    grad *= dx

def c5_obj_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        for i in range(lo, hi + 1):
            j1, j2 = i + s, i
            grad[j1] += 1.0 - h[j2]
            grad[j2] -= h[j1]
    grad /= max(len(active), 1)
    grad *= dx
    return max_val * dx, grad

    return c5_obj_grad(v, _nglob)

    return np.sum(v) - _nglob / 2.0

import numpy as np
from scipy.optimize import minimize
import time

_nglob = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_obj_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        for i in range(lo, hi + 1):
            j1 = i + s
            j2 = i
            grad[j1] += 1.0 - h[j2]
            grad[j2] -= h[j1]
    k = max(len(active), 1)
    grad /= k
    grad *= dx
    return max_val * dx, grad

def _obj(v):
    return c5_obj_grad(v, _nglob)

def _eq(v):
    return np.sum(v) - _nglob / 2.0

def _eq_j(v):
    return np.ones(_nglob)

def project_feasible(h, n):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > n / 2.0:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_solve(h0, n, max_it):

                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_it, 'ftol': 1e-14})
    x = project_feasible(res.x, n)
