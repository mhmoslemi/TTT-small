import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_subgrad(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    idx = int(np.argmax(corr))
    val = corr[idx] * dx
    lag = idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i + lag] += (1.0 - h[i]) * dx
            grad[i] -= h[i + lag] * dx
    else:
        m = -lag
        for i in range(n - m):
            grad[i] += (1.0 - h[i + m]) * dx
            grad[i + m] -= h[i] * dx
    return val, grad

    h = np.clip(h, 0.0, 1.0)

        d = target - h.sum()

        else:

        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    h = h.copy()
