import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def steepest_descent(h, dx, deadline, max_steps=10000):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h, dx)
    tau = np.arange(-(n-1), n)
    steps = 0
    
    while steps < max_steps:
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        old_corr = np.correlate(h, 1.0 - h, mode='full')
        E = np.zeros((len(ones), 2*n-1))
        for ii, i in enumerate(ones):
            idx = i + tau
            m = (idx >= 0) & (idx < n)
            E[ii, m] = h[idx[m]]
            idx = i - tau
            m = (idx >= 0) & (idx < n)
            E[ii, m] -= (1.0 - h[idx[m]])
        
        F = np.zeros((len(zeros), 2*n-1))
        for jj, j in enumerate(zeros):
            idx = j - tau
            m = (idx >= 0) & (idx < n)
            F[jj, m] = (1.0 - h[idx[m]])
            idx = j + tau
            m = (idx >= 0) & (idx < n)
            F[jj, m] -= h[idx[m]]
        
        best_delta = 1e-15
        best_ii, best_jj = -1, -1
        for ii in range(len(ones)):
            G = old_corr + E[ii]
            for jj in range(len(zeros)):
                nm = np.max(G + F[jj])
                d = nm * dx - current_c5
                if d < best_delta:
                    best_delta = d
                    best_ii, best_jj = ii, jj
        
        if best_ii >= 0:
            h[ones[best_ii]] = 0.0
            h[zeros[best_jj]] = 1.0
            current_c5 += best_delta
            steps += 1
        else:
            break
    return h, current_c5

def random_swap(h, rng):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    i = rng.choice(ones)
    j = rng.choice(zeros)
    h[i] = 0.0
    h[j] = 1.0
    return h

def simulated_annealing(h, dx, deadline, t_init=0.01, cooling=0.9995, rng=None):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    temp = t_init
    
    while time.time() < deadline:
        temp *= cooling
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-12)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def steepest_descent(h, dx, deadline, max_steps=50000):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h, dx)
    tau = np.arange(-(n-1), n)
    steps = 0
    
    while steps < max_steps:
        if time.time() > deadline:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        old_corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Precompute E (effect of removing 1 at each one-position)
        E = np.zeros((n_ones, 2*n-1))
        for ii in range(n_ones):
            i = ones[ii]
            idx = i + tau
            m = (idx >= 0) & (idx < n)
            E[ii, m] = h[idx[m]]
            idx = i - tau
            m = (idx >= 0) & (idx < n)
            E[ii, m] -= (1.0 - h[idx[m]])
        
        # Precompute F (effect of adding 1 at each zero-position)
        F = np.zeros((n_zeros, 2*n-1))
        for jj in range(n_zeros):
            j = zeros[jj]
            idx = j - tau
            m = (idx >= 0) & (idx < n)
            F[jj, m] = (1.0 - h[idx[m]])
            idx = j + tau
            m = (idx >= 0) & (idx < n)
            F[jj, m] -= h[idx[m]]
        
        # Find best swap
        best_delta = 1e-15
        best_ii, best_jj = -1, -1
        for ii in range(n_ones):
            G = old_corr + E[ii]
            new_maxes = np.max(G[None, :] + F, axis=1)
            bb = np.argmin(new_maxes)
            d = new_maxes[bb] * dx - current_c5
            if d < best_delta:
                best_delta = d
                best_ii, best_jj = ii, bb
        
        if best_ii >= 0:
            h[ones[best_ii]] = 0.0
            h[zeros[best_jj]] = 1.0
            current_c5 += best_delta
            steps += 1
        else:
            break
    
    return h, current_c5

    best_c5 = current_c5
