import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def penalized_obj(x, n, T, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    shifted = T * (c - m)
    logsum = m + np.log(np.sum(np.exp(shifted))) / T
    sv = np.sum(x) - 0.5 * n
    return logsum * dx + pen * sv * sv


def penalized_grad(x, n, T, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    shifted = T * (c - m)
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for i in range(n):
        lo = max(0, n - 1 - i)
        hi = min(2 * n - 2, n - 1 + i)
        ks = np.arange(lo, hi + 1)
        j1 = i - ks + (n - 1)
        j2 = i + ks - (n - 1)
        grad[i] = np.dot(w[lo:hi + 1], 1.0 - x[j1]) - np.dot(w[lo:hi + 1], x[j2])
    grad *= T * dx
    sv = np.sum(x) - 0.5 * n
    grad += 2.0 * pen * sv
    return grad


def project_constraint(h):
    n = len(h)
    target = 0.5 * n
    for _ in range(1000):
        sv = target - np.sum(h)
        if abs(sv) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if not np.any(free):
            free = np.ones(n, dtype=bool)
        h[free] += sv / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n) * 0.5 + 0.25

    opts = {'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': 0}
    res = minimize(penalized_obj, h_start.copy(), jac=penalized_grad,
                   args=(n, T, pen), method='L-BFGS-B', bounds=bounds, options=opts)
    h_new = project_constraint(res.x)
    if time.time() - t0 > time_limit:
        return h_new
    return h_new


def interpolate_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h)
