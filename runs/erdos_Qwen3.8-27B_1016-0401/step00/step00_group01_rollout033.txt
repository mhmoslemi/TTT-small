def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum}"""
    def f(lam):
        proj = np.clip(h - lam, 0, 1)
        return proj.sum() - target_sum
    lo, hi = h.min() - 2, h.max() + 2
    # Use bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    ec = np.exp(corr - m)
    p = ec / ec.sum()
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        lag = li - (n - 1)
        w = beta * p[li]
        if w < 1e-10 or lag == 0:
            continue
        if lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            d = -lag
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
    return grad
