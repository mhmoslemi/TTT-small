    """Just the max correlation (without dx scaling) for speed."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Adjust: if diff > 0, increase from smallest; if diff < 0, decrease from largest
    if diff > 0:
        idx = np.argsort(h)  # smallest first
        for i in idx:
            room = 1.0 - h[i]
            adj = min(room, diff)
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    else:
        idx = np.argsort(-h)  # largest first
        for i in idx:
            room = h[i]
            adj = min(room, -diff)
            h[i] -= adj
            diff += adj
            if abs(diff) < 1e-12:
                break

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def maxcorr(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            adj = min(1.0 - h[i], diff)
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            adj = min(h[i], -diff)
            h[i] -= adj
            diff += adj
            if abs(diff) < 1e-12:
                break
    return h
