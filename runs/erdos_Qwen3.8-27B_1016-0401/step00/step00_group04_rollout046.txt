    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    # Bisection on lambda: sum(clamp(h - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(v, target_sum):
    lo = -v.max() - 1.0
    hi = 1.0 - v.min() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def grad_lag(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L == 0:
        g = dx * (1.0 - 2.0 * h)
    elif 0 < L < n:
        g[:n - L] = dx * (1.0 - h[L:])
        g[L:] -= dx * h[:n - L]
    elif -n < L < 0:
        p = -L
        g[p:] = dx * (1.0 - h[:n - p])
        g[:n - p] -= dx * h[p:]
    return g

def optimize(h0, max_time, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0.astype(float).copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    lr = 0.04
    beta = 40.0
    stall = 0
    max_stall = 800

    while time.time() - t0 < max_time:
        ov = compute_overlaps(h)
        mx = float(np.max(ov))
        if mx < best_c5 - 1e-13:
            best_c5 = mx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            break

        shifted = ov - mx
        w = np.exp(beta * shifted)
        w /= w.sum()
        idx = np.where(w > 0.005)[0]
        if len(idx) == 0:
            idx = np.array([np.argmax(ov)])
        # Cap for speed
        if len(idx) > 30:
            top = np.argsort(w[idx])[-30:]
            idx = idx[top]

        grad = np.zeros(n)
        for i in idx:
            L = int(i) - (n - 1)
            if -n < L < n:
                grad += w[i] * grad_lag(h, L)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn

        h_trial = h - lr * grad
        h_trial = project_feasible(h_trial, target)
        c5t = compute_c5(h_trial)

        if c5t <= best_c5 + 1e-12:
            h = h_trial
            lr = min(lr * 1.003, 0.15)
            beta = min(beta * 1.015, 300.0)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = h.copy()
                stall = 0
        else:
            lr *= 0.82
            if lr < 0.003:
                h = project_feasible(best_h + 0.05 * rng.randn(n), target)
                lr = 0.04
                beta = 40.0
                stall = min(stall, 400)

    return best_h, best_c5

def refine_grid(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)
