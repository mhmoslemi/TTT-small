import numpy as np
from scipy.optimize import minimize

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(np.abs(corr)) * dx)

def soft_max_c5(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use log-sum-exp for numerical stability
    x = T * np.abs(corr)
    x_max = np.max(x)
    return (x_max + np.log(np.sum(np.exp(x - x_max)))) / T * dx

def objective(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = T * np.abs(corr)
    x_max = np.max(x)
    return (x_max + np.log(np.sum(np.exp(x - x_max)))) / T

    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:

        for _ in range(100):

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(np.abs(corr)) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = T * np.abs(corr)
    x_max = np.max(x)
    return (x_max + np.log(np.sum(np.exp(x - x_max)))) / T

    target = n * 0.5

        if abs(diff) < 1e-14:
            break

            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def soft_obj(h, T=100.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = T * corr
    x_max = x.max()
    return (x_max / T) + (np.log(np.exp(x - x_max).sum()) / T)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-13: break
            h = h + d * r / s
        else:
            s = h.sum()
            if s < 1e-13: break
            h = h + d * h / s
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        i = int(np.where(1.0-h == (1.0-h).max())[0][0]) if d > 0 else int(np.where(h == h.min())[0][0])
        h[i] = np.clip(h[i] + d, 0, 1)
    return h

    b = [(0.0, 1.0)] * n

            s = h.sum()
            if s < 1e-14:

            h += d * h / s

    bc5 = eval_c5(bh)
    return bh, bc5, bn
