import numpy as np
from scipy.optimize import minimize
import time

_n = 70
_beta = 100.0

def _build_h(d, n):
    h = np.zeros(n)
    h[:-1] = 0.5 + d
    h[-1] = 0.5 - np.sum(d)
    return h

def _obj(d):
    n = _n
    h = _build_h(d, n)
    neg = np.maximum(0.0, -h)
    pos = np.maximum(0.0, h - 1.0)
    pen = 1e5 * (np.sum(neg * neg) + np.sum(pos * pos))
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sc = _beta * (corr - m)
    sc = np.clip(sc, -500.0, 500.0)
    return float(m + np.log(np.sum(np.exp(sc))) / _beta + pen)

def _wf(h, lm):
    return np.sum(np.clip(h + lm, 0.0, 1.0))

def _proj(h, n):
    t = n * 0.5
    s0 = _wf(h, 0.0)
    if abs(s0 - t) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    if _wf(h, lo) > t:
        lo, hi = hi, lo
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if _wf(h, mid) < t:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
