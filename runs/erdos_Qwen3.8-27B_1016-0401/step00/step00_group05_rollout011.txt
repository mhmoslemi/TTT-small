- A point where f_lin(D) = f_quad(D) and g is minimized (the "crossover")
- Or at the minimum of f_lin(D) if f_lin > f_quad everywhere
- Or at a boundary

        lam = (lo + hi) * 0.5
        h = np.minimum(h + lam, 1)
    h = np.clip(h, 0, 1)
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] += residual / mask.sum()
    return np.clip(h, 0, 1)

def optimize_h_single(h, n, max_sweeps, deadline):
    target_sum = n / 2.0
    h = project_onto_constraint(h, target_sum)
    c5 = compute_c5(h)
    grid = np.linspace(0.0, 1.0, 50)
    
    s_arr = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_change = False
        for j in range(n):
            ov = h[j]
            c_current = np.correlate(h, 1 - h, mode='full')
            C_n1 = c_current[n - 1]
            
            js = j - s_arr
            jp = j + s_arr
            valid_js = (js >= 0) & (js < n) & (s_arr != 0)
            valid_jp = (jp >= 0) & (jp < n) & (s_arr != 0)
            
            alpha = np.zeros(2 * n - 1)
            alpha[valid_js] += (1.0 - h[js[valid_js]])
            alpha[valid_jp] -= h[jp[valid_jp]]
            
            # Evaluate at grid points
            D = grid - ov
            lin_vals = c_current[None, :] + D[:, None] * alpha[None, :]
            quad_vals = C_n1 + D * (1.0 - 2.0 * ov) - D * D
            lin_vals[:, n - 1] = quad_vals
            obj_all = np.max(lin_vals, axis=1)
            best_idx = np.argmin(obj_all)
            best_t = grid[best_idx]
            
            h[j] = best_t
            if abs(best_t - ov) > 1e-13:
                any_change = True
        
        h = project_onto_constraint(h, target_sum)
        new_c5 = compute_c5(h)
        if not any_change and new_c5 >= c5 - 1e-12:
            c5 = new_c5
            break
        c5 = new_c5
    
    h = project_onto_constraint(h, target_sum)
    c5 = compute_c5(h)
    return h, c5
