    B = one_minus_h_ql - h_qml  # (n_zeros, 2n-1)
    
    # All new correlations: (n_ones, n_zeros, 2n-1)
    all_new = corr[None, None, :] + A[:, None, :] + B[None, :, :]
    all_max = all_new.max(axis=2)  # (n_ones, n_zeros)
    
    current_max = corr.max()
    
    # Find best
    best_val = all_max.min()
    if best_val >= current_max - 1e-12:
        return None, None, None
    
    idx = np.unravel_index(np.argmin(all_max), all_max.shape)
    i = ones[idx[0]]
    j = zeros[idx[1]]
    new_corr = corr + A[idx[0]] + B[idx[1]]
    
    return i, j, new_corr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = np.inf
    
    global initial_h_values
    has_parent = 'initial_h_values' in dir() and initial_h_values is not None
    parent_n = len(initial_h_values) if has_parent else 70
    
    n_list = [parent_n] if has_parent else [70]
    for n in [68, 72, 66, 74, 64, 76, 60, 80, 56, 84, 52, 88, 48, 92, 44, 96, 40, 100, 36]:
        if n >= 20 and n not in n_list:
            n_list.append(n)
    
    for n in n_list:
        t_rem = budget_s - (time.time() - t0)
        if t_rem < 20:
            break
        
        n_restarts = 1
        if t_rem > 300:
            n_restarts = 3
        elif t_rem > 150:
            n_restarts = 2
        
        for rs in range(n_restarts):
            t_rem = budget_s - (time.time() - t0)
            if t_rem < 10:
                break
            
            # Initialize
            if rs == 0 and has_parent and n == parent_n:
                vals = np.asarray(initial_h_values, dtype=np.float64)
                h = np.zeros(n)
                order = np.argsort(-vals)
                h[order[:n//2]] = 1.0
            else:
                h = np.zeros(n)
                pos = rng.choice(n, size=n//2, replace=False)
                h[pos] = 1.0
            
            # Optimize
            corr = compute_corr(h)
            cm = float(corr.max())
            stuck = 0
            iterations = 0
            
            while time.time() - t0 < budget_s - 5:
                i, j, new_corr = greedy_step(h, corr)
                iterations += 1
                
                if i is not None:
                    h[i], h[j] = h[j], h[i]
                    corr = new_corr
                    cm = float(corr.max())
                    stuck = 0
                else:
                    stuck += 1
                    if stuck >= 15:
                        # Perturbation
                        n_pert = rng.randint(1, 6)
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        for _ in range(n_pert):
                            a = ones[rng.randint(len(ones))]
                            b = zeros[rng.randint(len(zeros))]
                            h[a], h[b] = h[b], h[a]
                        corr = compute_corr(h)
                        cm = float(corr.max())
                        stuck = 0
            
            c5 = cm * 2.0 / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    if best_h is None:
        n = 70
        h = np.zeros(n)
        h[:n//2] = 1
        best_h = h
        best_c5 = compute_c5(h)
    
    return best_h, best_c5, len(best_h)
