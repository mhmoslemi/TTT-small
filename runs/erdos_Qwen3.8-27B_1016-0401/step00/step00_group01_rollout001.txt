        d = target - h.sum()

            break

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += d / free.sum()
        else:
            a = np.argmin(np.abs(h - 0.5))
            h[a] = 0.5
            h = np.clip(h, 0, 1)
    return np.clip(h, 0, 1)

def coord_descent(h, max_sweeps=200, rng=None, verbose=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h)
    rng_local = rng if rng is not None else np.random.RandomState(0)
    for sweep in range(max_sweeps):
        improved = False
        order = rng_local.permutation(n)
        for ii in range(n):
            i = order[ii]
            for jj in range(ii + 1, n):
                j = order[jj]
                ai, aj = h[i], h[j]
                t_lo = max(-ai, aj - 1.0) + 1e-12
                t_hi = min(1.0 - ai, aj) - 1e-12
                if t_hi <= t_lo:
                    continue
                fracs = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
                tands = t_lo + fracs * (t_hi - t_lo)
                tands = np.clip(tands, t_lo, t_hi)
                cur_val = compute_c5(h)
                bt, bv = 0.0, cur_val
                for t in tands:
                    if abs(t) < 1e-13:
                        continue
                    h[i], h[j] = ai + t, aj - t
                    v = compute_c5(h)
                    if v < bv - 1e-13:
                        bv, bv_ = v, v
                        bt = t
                    h[i], h[j] = ai, aj
                if bt != 0.0:
                    h[i], h[j] = ai + bt, aj - bt
                    if compute_c5(h) < best - 1e-10:
                        best = compute_c5(h)
                        improved = True
        if not improved:
            break
    h = project_sum(h, target)
    return h, compute_c5(h)

def block_perturb(h, rng, strength=0.1):
    n = len(h)
    target = n / 2.0
    h2 = h.copy()
    for _ in range(rng.randint(1, 5)):
        idx = rng.randint(0, n)
        h2[idx] += rng.uniform(-strength, strength)
    h2 = project_sum(h2, target)
    return h2

def random_init(n, rng):
    h = rng.uniform(0, 1, n)
    return project_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
