NumPy's correlate(a, v, mode='full') returns an array c of length len(a)+len(v)-1.

From the NumPy documentation: "The linear cross-correlation of the two inputs is defined as:
c_k = \sum_n a_{n+k} \overline{v_n}"

For mode='full', c is computed for all valid shifts. The indexing is such that c[0] corresponds to the shift where a is shifted to the right of v as much as possible, and c[-1] corresponds to the opposite extreme.

More precisely, for len(a) = len(v) = n:
c has length 2n-1.
c[n-1+k] = \sum_i a[i+k] * v[i] for k = -(n-1), ..., n-1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    val = mc + np.log(np.sum(np.exp(shifted))) / T
    return val * (2.0 / n)

def soft_max_grad(h, n, T):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return grad * dx

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                s = min(diff, 1.0 - h[i])
                h[i] += s
            else:
                s = min(-diff, h[i])
                h[i] -= s
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
    return h

def _make_constraints(n, target):
    cons = [{'type': 'eq', 'fun': lambda x, ts=target: x.sum() - ts,
             'jac': lambda x, ns=n: np.ones(ns)}]
    return cons

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(70)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    cons = _make_constraints(n, target)

    # Phase 1: Temperature annealing from parent
    for T in [5, 10, 20, 40, 80, 160, 320]:
        if elapsed() > budget_s - 120:
            break
        def obj_p1(x):
            return soft_max_obj(x, n, T)
        def jac_p1(x):
            return soft_max_grad(x, n, T)
        try:
            res = minimize(obj_p1, h_best, method='SLSQP', jac=jac_p1,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-14})
            if res.fun < 100:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 2: Random restarts
    n_restarts = 0
    while elapsed() < budget_s - 200:
        n_restarts += 1
        r = n_restarts
        if r % 5 == 1:
            h_init = (rng.random(n) >= 0.5).astype(float)
        elif r % 5 == 2:
            h_init = np.zeros(n)
            h_init[::2] = 1.0
            h_init = np.roll(h_init, int(rng.integers(-10, 11)))
        elif r % 5 == 3:
            h_init = h_best + rng.normal(0, 0.2, n)
        elif r % 5 == 4:
            h_init = h_best + rng.normal(0, 0.08, n)
        else:
            h_init = (rng.random(n) > 0.4).astype(float)

        h_init = fix_constraint(h_init, target)
        T = min(20 + r * 15, 500)
        def obj_p2(x):
            return soft_max_obj(x, n, T)
        def jac_p2(x):
            return soft_max_grad(x, n, T)
        try:
            res = minimize(obj_p2, h_init, method='SLSQP', jac=jac_p2,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-14})
            if res.fun < 100:
                h_new = fix_constraint(res.x.copy(), target)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: Refine at higher resolution
    for n_hi in [140, 210, 280]:
        if elapsed() > budget_s - 80:
            break
        n_old = len(h_best)
        if n_hi <= n_old:
            continue
        target_hi = n_hi / 2.0
        x_old = np.arange(n_old) * (2.0 / n_old)
        x_new = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = fix_constraint(h_hi, target_hi)

        bounds_hi = [(0.0, 1.0)] * n_hi
        cons_hi = _make_constraints(n_hi, target_hi)

        for T in [30, 60, 120, 250]:
            if elapsed() > budget_s - 40:
                break
            def obj_hi(x):
                return soft_max_obj(x, n_hi, T)
            def jac_hi(x):
                return soft_max_grad(x, n_hi, T)
            try:
                res = minimize(obj_hi, h_hi, method='SLSQP', jac=jac_hi,
                              bounds=bounds_hi, constraints=cons_hi,
                              options={'maxiter': 1500, 'ftol': 1e-14})
                if res.fun < 100:
                    h_new = fix_constraint(res.x.copy(), target_hi)
                    c5 = compute_c5(h_new)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = h_new
                        h_hi = h_new.copy()
            except Exception:
                pass

    # Final check
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, len(h_best))
