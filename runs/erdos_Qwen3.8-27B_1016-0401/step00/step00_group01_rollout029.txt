    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_subgradient(h, dx):
    """Subgradient of C5 w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        if shift == 0:
            grad += (1.0 - 2.0 * h)
        else:
            # Part 1: h[j] as first factor, need j+shift in [0,n-1]
            lo1 = max(0, -shift)
            hi1 = min(n-1, n-1-shift)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += (1.0 - h[lo1+shift:hi1+1+shift])
            # Part 2: h[j] as second factor (in 1-h[j]), need j-shift in [0,n-1]
            lo2 = max(0, shift)
            hi2 = min(n-1, n-1+shift)
            if lo2 <= hi2:
                grad[lo2:hi2+1] += (-h[lo2-shift:hi2+1-shift])
    return grad * dx / len(active)  # average over active (for smoothness)

            grad += (1.0 - 2.0 * h)

            hi = min(n - 1, n - 1 - s)

                grad[lo:hi + 1] += (1.0 - h[lo + s:hi + 1 + s])
            lo2 = max(0, s)
            hi2 = min(n - 1, n - 1 + s)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] += (-h[lo2 - s:hi2 + 1 - s])

def smooth_max_c5(h, dx, alpha=200.0):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = corr - np.max(corr)
    return np.max(corr) + np.log(np.sum(np.exp(alpha * shifted))) / alpha
