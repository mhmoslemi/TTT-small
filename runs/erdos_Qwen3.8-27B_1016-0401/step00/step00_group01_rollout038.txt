import numpy as np
import time

def project(h, target_sum):
    """Project h onto {0≤x≤1, sum(x)=target_sum}"""
    n = len(h)
    # First clip
    h = np.clip(h, 0, 1)
    # Then adjust sum
    # Binary search for offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr

def compute_grad(h, corr, active_mask):
    n = len(h)
    grad = np.zeros(n)
    nactive = np.sum(active_mask)
    for k in np.where(active_mask)[0]:
        lag = k - (n - 1)
        for j in range(n):
            g = 1.0
            if lag != 0:
                if 0 <= j + lag < n:
                    g -= h[j + lag]
                if 0 <= j - lag < n:
                    g -= h[j - lag]
            else:
                g = 1.0 - 2.0 * h[j]
            grad[j] += g
    grad /= nactive

def project_onto_simplex(h, target_sum):

    if n == 0:
        return h
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

def compute_grad(h, corr, tol=1e-6):

    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n - lag] += 1.0 - h[lag:]
            grad[lag:] -= h[:n - lag]
        else:
            l = -lag
            grad[l:] += 1.0 - h[:n - l]
            grad[:n - l] -= h[l:]
    grad /= len(active)
    return grad

def gd_step(h, grad, lr, target_sum):
    h_new = h - lr * grad
    return project_onto_simplex(h_new, target_sum)

def get_grad(h, corr, tol_frac=0.01):
    n = len(h)
    mv = np.max(corr)
    thresh = mv * (1.0 - tol_frac) if mv > 0 else mv - tol_frac

    while time.time() - ts < 0.55 * t_alloc:
        corr = get_corr(h)
        g = get_grad(h, corr)
        h = project(h - lr * g, tgt)
        c5 = get_c5(h)
        if c5 < bc5:
            bc5 = c5
            bh = h.copy()
        lr *= 0.997
        if lr < 1e-5:
            lr = 0.5 / np.sqrt(n)
    tls = time.time()
    T = 0.01
    while time.time() - tls < 0.45 * t_alloc:
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = 0.12 * rng.randn()
        h2 = h.copy()
        h2[i1] += d
        h2[i2] -= d
        h2 = project(h2, tgt)
        c52 = get_c5(h2)
        if c52 < bc5:
            bc5 = c52
            bh = h2.copy()
            h = h2
            T = max(T * 0.995, 1e-5)
        elif c52 < bc5 + T:
            if rng.rand() < min(1.0, np.exp((bc5 - c52) / max(T, 1e-8))):
                h = h2
    # Final GD polish
    lr2 = 0.3 / np.sqrt(n)
    tfin = time.time()
    for _ in range(200):
        if time.time() - tfin > 3.0:
            break
        corr = get_corr(h)
        g = get_grad(h, corr)
        h2 = project(h - lr2 * g, tgt)
        c52 = get_c5(h2)
        if c52 < bc5:
            bc5 = c52
            bh = h2.copy()
        h = h2
    return bh, bc5

def init_bin(n, sv):
    rng = np.random.RandomState(sv)
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    if n % 2 == 1:
        h[idx[n // 2]] = 0.5
    return project(h, 0.5 * n)

def init_blk(n, sv):
    rng = np.random.RandomState(sv)
    nb = rng.randint(3, 9)
    w = np.rint(rng.dirichlet(np.ones(nb)) * n).astype(int)
    w[-1] = n - w[:-1].sum()
    h = np.zeros(n)
    p = 0
    for i, s in enumerate(w):
        if s > 0 and i % 2 == 0:
            h[p:p + s] = 1.0
        p += s
    return project(h, 0.5 * n)

def init_sin(n, sv):
    rng = np.random.RandomState(sv)
    x = np.linspace(0, 2.0, n, endpoint=False)
    f = rng.uniform(1.5, 6.0)
    ph = rng.uniform(0, 2.0 * np.pi)
    h = np.clip(0.5 + 0.45 * np.sin(np.pi * f * x + ph), 0.0, 1.0)
    return project(h, 0.5 * n)

def init_wave(n, sv):
    rng = np.random.RandomState(sv)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(1.0, 8.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += 0.3 * np.sin(np.pi * f * x + ph)
    h = np.clip(0.5 + h, 0.0, 1.0)
    return project(h, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    bc5 = 1e9
    bh = None

    try:
        _ihv = np.array(initial_h_values, dtype=float)
        if len(_ihv) > 0:
            _ni = len(_ihv)
            _ihv = project(_ihv.copy(), 0.5 * _ni)
            _ci = get_c5(_ihv)
            if _ci < bc5:
                bc5 = _ci
                bh = _ihv.copy()
            _rem = budget_s - (time.time() - t0)
            if _rem > 40:
                _h, _c = optimize_from(_ihv, min(_rem * 0.18, 100), seed)
                if _c < bc5:
                    bc5 = _c
                    bh = _h
    except (NameError, Exception):
        pass

    _ns = [47, 48, 50, 54, 60, 64, 72, 80]
    _inits = [init_bin, init_blk, init_sin, init_wave]
    for _n in _ns:
        if time.time() - t0 > budget_s - 30:
            break
        for _init in _inits:
            if time.time() - t0 > budget_s - 20:
                break
            for _tr in range(2):
                if time.time() - t0 > budget_s - 12:
                    break
                _rem = budget_s - (time.time() - t0) - 25
                if _rem < 8:
                    break
                _h0 = _init(_n, seed + _n * 7 + _tr * 31)
                _ta = min(_rem * 0.22, 40)
                _h, _c = optimize_from(_h0, _ta, seed + _n * 10 + _tr * 100)
                if _c < bc5:
                    bc5 = _c
                    bh = _h

    if bh is not None:
        _rem = budget_s - (time.time() - t0)
        if _rem > 20:
            _h, _c = optimize_from(bh, min(_rem * 0.5, 120), seed + 7777)
            if _c < bc5:
                bc5 = _c
                bh = _h
        _rem2 = budget_s - (time.time() - t0)
        if _rem2 > 15:
            _h, _c = optimize_from(bh, min(_rem2 * 0.6, 80), seed + 8888)
            if _c < bc5:
                bc5 = _c
                bh = _h

    if bh is None:
        bh = np.zeros(48)
        bh[:24] = 1.0
        bc5 = get_c5(bh)

    return bh, get_c5(bh), len(bh)
