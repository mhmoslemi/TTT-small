def make_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        h = np.full(n, 0.5)
        return h
    # Scale
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix up
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:

        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
