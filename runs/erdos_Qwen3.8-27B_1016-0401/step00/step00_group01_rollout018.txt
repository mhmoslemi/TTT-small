from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def obj_and_grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    # Log-sum-exp for smooth max
    max_s = np.max(scaled)
    exp_s = np.exp(T * (scaled - max_s))
    sum_exp = np.sum(exp_s)
    obj = max_s + np.log(sum_exp) / T
    p = exp_s / sum_exp  # softmax weights
    
    # Gradient
    n_corr = len(corr)  # 2n-1
    shifts = np.arange(n_corr) - (n - 1)  # -(n-1) to (n-1)
    js = np.arange(n)
    
    grad = np.zeros(n)

        if p[m] < 1e-10:

        s = shifts[m]
        jp = js + s
        jm = js - s
        mask_p = (jp >= 0) & (jp < n)

            grad[mask_p] += p[m] * (1.0 - h[jp[mask_p]])
        if np.any(mask_m):
            grad[mask_m] -= p[m] * h[jm[mask_m]]
    grad *= dx
    return obj, grad

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    # If already at target, done
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Need to adjust: if current_sum > target_sum, decrease; else increase
    # Binary search on the shift
    # After clipping and shifting by t, sum = sum(clip(h+t, 0, 1)) = target_sum
    
    lo, hi = -1.0, 1.0
    if current_sum > target_sum:
        lo, hi = -1.0, 0.0
    else:
        lo, hi = 0.0, 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h_new = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_new)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)
