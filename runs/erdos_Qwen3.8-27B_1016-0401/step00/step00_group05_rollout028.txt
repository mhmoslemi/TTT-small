import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_polytope(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def subgrad_for_lag(h, k):
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        m = -k
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g


def optimize_h_subgrad(h0, max_iters, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_polytope(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.025
    no_improve = 0
    t0 = time.time()
    vel = np.zeros(n)
    mu = 0.6

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = compute_full_corr(h)
        idx_max = int(np.argmax(corr))
        sg = subgrad_for_lag(h, idx_max - n + 1)

        top2 = np.argpartition(corr, -3)[-3:]
        for ti in top2:
            if corr[ti] > corr[idx_max] * 0.85 and ti != idx_max:
                sg += 0.25 * subgrad_for_lag(h, int(ti) - n + 1)

        vel = mu * vel + (1 - mu) * sg
        h_new = project_polytope(h - alpha * vel, target_sum)
        c5_new = compute_c5(h_new)

            no_improve = 0
            alpha = min(alpha * 1.015, 0.07)

            no_improve += 1
            alpha *= 0.985

    return best_h, best_c5


def local_search(h, n_trials, rng):
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_trials):
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        delta = rng.uniform(-0.05, 0.05)
        hv = best_h.copy()
        hv[i] += delta
        hv[j] -= delta
        if 0.0 <= hv[i] <= 1.0 and 0.0 <= hv[j] <= 1.0:
            c5_try = compute_c5(hv)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = hv
    return best_h, best_c5


def binarize_push(h, target_sum):
    """Push values toward 0 or 1 while maintaining sum."""
    for _ in range(5):
        sorted_idx = np.argsort(h)
        lo_idx = sorted_idx[:int(np.sum(h < 0.5))]
        hi_idx = sorted_idx[int(np.sum(h < 0.5)):]
        h2 = h.copy()
        h2[lo_idx] = 0.0
        h2[hi_idx] = 1.0
