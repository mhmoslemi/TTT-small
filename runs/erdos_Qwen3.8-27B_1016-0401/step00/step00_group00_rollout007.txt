from scipy.optimize import minimize
import time

def c5_objective(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    return np.max(corr)

def get_gradient(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    k_max = np.argmax(corr)
    # Gradient of corr[k_max] with respect to h
    L = k_max - n + 1  # lag
    grad = np.zeros(n)
    for m in range(n):
        # Contribution from h[m] term: (1 - h[m+L]) if m+L in [0,n-1]
        j1 = m + L
        if 0 <= j1 < n:
            grad[m] += (1.0 - h[j1])
        # Contribution from h[j+L] term where j+L = m, i.e., j = m-L
        j2 = m - L
        if 0 <= j2 < n:
            grad[m] -= h[j2]
    grad *= dx
    return grad

def optimize_h(h0, n_target, max_iter=1000, tol=1e-10):
    # Resample h0 to n_target points
    x0 = np.linspace(0, 2, len(h0), endpoint=False) + 1.0/(2*len(h0))
    x1 = np.linspace(0, 2, n_target, endpoint=False) + 1.0/(2*n_target)
    h_resampled = np.interp(x1, x0, h0)
    
    # Fix sum

    h_resampled = h_resampled - (np.sum(h_resampled) - target_sum) / n_target
    h_resampled = np.clip(h_resampled, 0, 1)
    # Re-fix sum after clipping
    diff = target_sum - np.sum(h_resampled)
    h_resampled += diff / n_target
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_target/2.0}]
    bounds = [(0, 1)] * n_target
    
    result = minimize(c5_objective, h_resampled, method='SLSQP', 
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': tol})
    return result.x, result.fun

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h) * dx))

def grad_corr_k(h, k):
    n = len(h)
    L = k - n + 1
    grad = np.zeros(n)
    if L >= 0:
        grad[:n-L] += (1.0 - h[L:])
        grad[L:] -= h[:n-L]
    else:
        Lp = -L
        grad[Lp:] += (1.0 - h[:n-Lp])
        grad[:n-Lp] -= h[Lp:]
    return grad * (2.0 / n)

def lse_obj(h, beta):
    n = len(h)
    cx = compute_corr(h) * (2.0 / n)
    mc = np.max(cx)
    if beta < 1e-6:
        return mc
    return mc + np.log(np.sum(np.exp(beta * (cx - mc)))) / beta

def lse_grad(h, beta):
    n = len(h)
    cx = compute_corr(h) * (2.0 / n)
    mc = np.max(cx)
    if beta < 1e-6:
        k = int(np.argmax(cx))
        return grad_corr_k(h, k)
    sh = beta * (cx - mc)
    es = np.exp(sh)
    sw = np.sum(es)
    sm = es / sw
    g = np.zeros(n)
    for k in range(len(sm)):
        if sm[k] > 1e-10:
            g += sm[k] * grad_corr_k(h, k)
    return g

def resample_to(h0, n_new):
    x0 = np.linspace(0.0, 2.0, len(h0), endpoint=False) + 1.0/(2.0*len(h0))
    x1 = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0/(2.0*n_new)
    h_new = np.interp(x1, x0, h0)
    target = n_new / 2.0
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    h_new += diff / n_new
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, beta=50.0, maxiter=500):
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    fun = lambda x: lse_obj(x, beta)
    jac = lambda x: lse_grad(x, beta)
    res = minimize(fun, h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    h_best = np.clip(res.x, 0.0, 1.0)
    h_best = fix_sum(h_best, n)
    return h_best, c5_value(h_best)
