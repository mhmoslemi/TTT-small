def project_feasible(h, n):

    for _ in range(200):

        s = np.sum(h)

def subgradient_step(h, n):
    """Compute subgradient of C5 at h and return (value, subgrad)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    val = corr[k_star] * dx
    # shift: corr index k_star corresponds to lag = k_star - (n-1)
    lag = k_star - (n - 1)
    g = np.zeros(n)
    # Term: sum_i h[i]*(1-h[i+lag]) for valid i
    if lag >= 0:
        # i ranges from 0 to n-1-lag
        # g[i] += (1 - h[i+lag]) * dx
        g[:n-lag] += (1.0 - h[lag:]) * dx
        # g[i+lag] += -h[i] * dx
        g[lag:] -= h[:n-lag] * dx
    else:

        a = -lag

        g[a:] += (1.0 - h[:n-a]) * dx

def upsample(h, new_n):
    n = len(h)
    old_x = np.arange(n) * 2.0 / n
    new_x = np.arange(new_n) * 2.0 / new_n
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    try:
        init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        init = np.full(45, 0.5)

    n0 = len(init)
    init = project_feasible(init, n0)
    best_h = init.copy()
    best_c5 = compute_c5(init)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Nelder-Mead at n0
    def nm_obj(x):
        h = project_feasible(x, n0)
        return compute_c5(h)

    try:
        res = minimize(nm_obj, init.copy(), method='Nelder-Mead',
                       options={'maxiter': 6000, 'xatol': 1e-7, 'fatol': 1e-10, 'adaptive': True})
        h_opt = project_feasible(res.x, n0)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
    except Exception:
        pass

    # Phase 2: L-BFGS-B with smooth max at n0
    if elapsed() < budget_s * 0.4:
        temps = [0.005, 0.002, 0.001, 0.0005]
        for temp in temps:
            if elapsed() > budget_s * 0.45:
                break
            def lbfgs_obj(x, t=temp):
                return smooth_max_obj(x, n0, t)
            try:
                res = minimize(lbfgs_obj, best_h.copy(), method='L-BFGS-B',
                               bounds=[(0, 1)] * n0,
                               options={'maxiter': 500, 'ftol': 1e-12, 'maxfun': 5000})
                h_opt = project_feasible(res.x, n0)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 3: Subgradient at n0
    if elapsed() < budget_s * 0.6:
        tl = min(budget_s * 0.15, 120.0)
        h_sg, c5_sg = subgradient_optimize(best_h, n0, max_iter=50000, time_limit=tl)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 4: Upscale and optimize at finer resolutions
    for target_n in [90, 180, 360]:
        if elapsed() > budget_s * 0.8:
            break
        h_up = upsample(best_h, target_n)
        h_up = project_feasible(h_up, target_n)
        # Subgradient at this resolution
        tl = min(budget_s * (0.8 - elapsed() / budget_s) * 0.4, 180.0)
        if tl > 5:
            h_sg, c5_sg = subgradient_optimize(h_up, target_n, max_iter=80000, time_limit=tl)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
        # L-BFGS polish if not too large
        if target_n <= 90 and elapsed() < budget_s * 0.8:
            for temp in [0.001, 0.0005]:
                if elapsed() > budget_s * 0.85:
                    break
                def lbfgs_obj2(x, tn=target_n, t=temp):
                    return smooth_max_obj(x, tn, t)
                try:
                    res = minimize(lbfgs_obj2, best_h.copy(), method='L-BFGS-B',
                                   bounds=[(0, 1)] * target_n,
                                   options={'maxiter': 300, 'ftol': 1e-12, 'maxfun': 3000})
                    h_opt = project_feasible(res.x, target_n)
                    c5_val = compute_c5(h_opt)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_opt.copy()
                except Exception:
                    pass

    # Phase 5: Final Nelder-Mead polish at best resolution (if small enough)
    if elapsed() < budget_s * 0.9 and len(best_h) <= 90:
        n_best = len(best_h)
        def final_nm(x):
            h = project_feasible(x, n_best)
            return compute_c5(h)
        try:
            res = minimize(final_nm, best_h.copy(), method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-11, 'adaptive': True})
            h_opt = project_feasible(res.x, n_best)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
