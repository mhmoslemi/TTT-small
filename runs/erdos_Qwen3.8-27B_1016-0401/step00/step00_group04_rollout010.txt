```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def compute_active_gradient(h, m):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - m + n - 1
        idx2 = j + m - n + 1
        val = 0.0
        if 0 <= idx1 <= n - 1:
            val += (1.0 - h[idx1])
        if 0 <= idx2 <= n - 1:
            val -= h[idx2]
        grad[j] = dx * val
    return grad

def project_feasible(z, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_descent(h, n, iterations, initial_lr=0.01):
    h = h.copy()
    lr = initial_lr
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(iterations):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        # Find all near-max indices for subgradient
        threshold = max_val - 1e-6
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for m in active:
            grad += compute_active_gradient(h, m)
        grad /= len(active)
        
        h = project_feasible(h - lr * grad, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive learning rate
        if it % 100 == 99:
            lr *= 0.95
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Get initial
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimize at n0
    h_opt, c5_opt = subgradient_descent(best_h, n0, 50000, 0.05)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt
    
    elapsed = time.time() - start
    # Try refinement at 2*n0 if time allows
    if budget_s - elapsed > 60:

        # Interpolate

        h1 = project_feasible(h1, n1)

        if c5_opt < best_c5:

    return best_h, best_c5, len(best_h)
```

```python
    lo, hi = -0.1, 1.1

        if h.sum() > target:

        else:

    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def corr_weights(h_cur, m, n):
    """For correlation index m, return the weight vector w where corr[m] = h @ w"""
    w = np.zeros(n)
    for i in range(n):
        idx = i - m + n - 1
        if 0 <= idx <= n - 1:
            w[i] = 1.0 - h_cur[idx]
    return w

def lp_linearize(h_cur, n, solver='ECOS'):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    W = np.zeros((n_corr, n))
    for m in range(n_corr):
        for i in range(n):
            idx = i - m + n - 1
            if 0 <= idx <= n - 1:
                W[m, i] = 1.0 - h_cur[idx]
    constraints.append(cp.multiply(dx, W @ h) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)

        h_val = project_feasible(h_val, n)
```

```python
        idx2 = j + m_vals - n + 1
```

```python
import numpy as np
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(z, n):
    target = n / 2.0
    lo, hi = -0.5, 1.5
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def build_W(h_cur, n):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    W = np.zeros((n_corr, n))
    idx = np.arange(n)
    for m in range(n_corr):
        off = idx - m + n - 1
        valid = (off >= 0) & (off < n)
        W[m, valid] = 1.0 - h_cur[off[valid]]
    return W, dx

def lp_step(h_cur, n, solver=cp.ECOS):
    W, dx = build_W(h_cur, n)
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    constraints.append((dx * W @ h) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver, max_time=5)
    except Exception:
        return h_cur.copy(), compute_c5(h_cur)
    if h.value is None:
        return h_cur.copy(), compute_c5(h_cur)
    h_new = np.clip(h.value, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, float(compute_c5(h_new))

def smooth_grad(h, n, eps=5.0):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    s = np.max(corr)
    ev = np.exp(eps * (corr - s))
    w = ev / ev.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for j in range(n):
        m_vals = np.arange(n_corr)
        i1 = j - m_vals + n - 1
        i2 = j + m_vals - n + 1
        v1 = (i1 >= 0) & (i1 < n)
        v2 = (i2 >= 0) & (i2 < n)
        c1 = np.where(v1, w * (1.0 - h[np.clip(i1, 0, n - 1)]), 0.0)
        c2 = np.where(v2, w * h[np.clip(i2, 0, n - 1)], 0.0)
        grad[j] = dx * (c1.sum() - c2.sum())
    return grad

def grad_descent(h, n, max_time, start_lr=0.02, eps=5.0):
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = start_lr
    v = np.zeros(n)
    mu = 0.85
    t0 = time.time()
    it = 0
    while time.time() - t0 < max_time:
        it += 1
        g = smooth_grad(h, n, eps)
        v = mu * v - lr * g
        h = project_feasible(h + v, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        if it % 2000 == 1999:
            lr *= 0.85
            mu *= 0.99
    return best, best_c5

def coord_descent(h, n, iters=30):
    h = h.copy()
    bp = best = compute_c5(h)
    for _ in range(iters):
        for j in range(n):
            k = (j + 1) % n
            for d in (-0.04, -0.02, -0.01, 0.01, 0.02, 0.04):
                h[j] += d
                h[k] -= d
                if 0 <= h[j] <= 1 and 0 <= h[k] <= 1:
                    v = compute_c5(h)
                    if v < bp:
                        bp = v
                        best = h.copy()
                    else:
                        h[j] -= d
                        h[k] += d
                else:
                    h[j] -= d
                    h[k] += d
    return best

def random_feasible(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.random(n)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def remaining():
        return max(0, budget_s - (time.time() - t_start))

    n0 = len(initial_h_values)

        if c5_cd < best_c5:

            best_h = h_cd.copy()

        for s in range(5):
            if remaining() < 20:

            h_r = random_feasible(n0, seed + 1000 + s)

            for it in range(40):

                    best_h = h_r.copy()

    # Phase 5: Upscale to 2*n0

                best_h = h1.copy()

        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    return best_h, best_c5, len(best_h)
```
