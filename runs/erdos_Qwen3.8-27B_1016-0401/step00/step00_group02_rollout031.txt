    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def binary_refine(h, rng, max_steps, time_limit):
    """Local search with SA for binary h"""
    n = len(h)
    h = (h >= 0.5).astype(float)
    # Fix sum to n/2
    deficit = int(n/2) - int(h.sum())
    if deficit > 0:
        zeros = np.where(h == 0)[0]
        chosen = rng.choice(zeros, size=min(deficit, len(zeros)), replace=False)
        h[chosen] = 1.0
    elif deficit < 0:
        ones = np.where(h == 1)[0]
        chosen = rng.choice(ones, size=min(-deficit, len(ones)), replace=False)
        h[chosen] = 0.0
    
    cur = np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0/n)
    best_local = h.copy()
    best_local_c5 = cur
    
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        for _ in range(20):
            i = int(rng.integers(len(ones)))
            j = int(rng.integers(len(zeros)))
            i, j = ones[i], zeros[j]
            h[i], h[j] = 0.0, 1.0
            new_c = np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0/n)
            if new_c < cur:
                cur = new_c
                if cur < best_local_c5:
                    best_local_c5 = cur
                    best_local = h.copy()
                improved = True
                break
            else:
                h[i], h[j] = 1.0, 0.0
        
        if not improved:
            # SA move
            T_sa = 0.005 * max(0.1, 1.0 - step/max_steps)
            i, j = ones[rng.integers(len(ones))], zeros[rng.integers(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_c = np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0/n)
            if rng.random() < math.exp(-max(0, new_c - cur) / T_sa):
                cur = new_c
            else:
                h[i], h[j] = 1.0, 0.0
    
    return best_local, best_local_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_obj(x, T):
    h = np.clip(x, 0.0, 1.0)

def normalize_h(h, n):

    return h

def fix_binary_sum(h, n, rng):
