        h_new, c5_new = optimize_local(h_bin, n, lam, maxiter=400)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Try larger grid if time permits
    if elapsed() < budget_s - 120:
        for n2 in [60, 90]:
            if elapsed() > budget_s - 60:
                break
            h_up = project_feasible(upsample_h(best_h, n, n2), n2)
            c5_up = compute_c5(h_up)
            # Optimize on finer grid
            h_new, c5_new = multi_start_optimize(h_up, n2, lam, rng, budget_s, elapsed)
            if c5_new < best_c5:
                best_h, best_c5, n = h_new, c5_new, n2

    # Phase 5: Final polish
    if elapsed() < budget_s - 30:
        h_new, c5_new = optimize_local(best_h, n, 50.0, maxiter=600, ftol=1e-14)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Ensure exact feasibility
    best_h = project_feasible(best_h, n)
    # Fix sum to exact n/2
    diff = np.sum(best_h) - n / 2.0
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if mask.sum() > 0:
            best_h[mask] -= diff / mask.sum()
    best_h = np.clip(best_h, 0.0, 1.0)
    # Final correction for rounding
    diff = np.sum(best_h) - n / 2.0
    if abs(diff) > 1e-15:
        best_h[np.argmin(np.abs(best_h - 0.5))] -= diff
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)
    return (best_h, float(best_c5), n)
