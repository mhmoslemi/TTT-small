    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()

    h = (h > 0.5).astype(float)
    n = len(h)

    target = n // 2

        for i in idx[len(idx)-(s-target):]:
            h[i] = 0

        for i in idx[:target-s]:
            h[i] = 1

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h):
    n = len(h)
    h2 = np.empty(2 * n)
    h2[0::2] = h
    h2[1::2] = h
    return h2

def binary_local_search(h, deadline, seed=0):
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float).copy()
    n = len(h)
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        idx1 = np.where(h == 1)[0]
        h[idx1[len(idx1) - (s - target):]] = 0.0
    elif s < target:
        idx0 = np.where(h == 0)[0]
        h[idx0[:target - s]] = 1.0
    c5 = compute_c5(h)
    while time.time() < deadline:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        improved = False
        n_tries = min(8000, max(1000, 500000 // max(1, n)))
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            nc = compute_c5(h)
            if nc < c5:
                c5 = nc
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return h, c5

def continuous_optimize(h_init, deadline, beta=100.0):
    h = h_init.astype(float).copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_onto_constraint(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 2.0 / n
    stagnation = 0
    offsets = np.arange(-(n - 1), n)
    while time.time() < deadline and lr > 1e-12:
        corr = np.correlate(h, 1.0 - h, mode="full")
        c = corr * dx
        c_max = np.max(c)
        e = np.exp(beta * (c - c_max))
        p = e / np.sum(e)
        grad = np.zeros(n)
        for j in range(n):
            v1 = (j + offsets >= 0) & (j + offsets < n)
            v2 = (j - offsets >= 0) & (j - offsets < n)
            if np.any(v1):
                grad[j] += np.dot(p[v1], (1.0 - h[j + offsets[v1]]))
            if np.any(v2):
                grad[j] -= np.dot(p[v2], h[j - offsets[v2]])
        grad *= dx
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr *= 0.99
            stagnation = 0
        else:
            lr *= 0.7
            stagnation += 1
            if stagnation > 30:
                break
    return best_h, best_c5

def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(42)
        h_init[::2] = 1.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return max(0.0, budget_s - 15.0 - (time.time() - t0))

    # Phase 1: binary search at initial resolution
    n0 = len(best_h)
    dl1 = t0 + min(40.0, budget_s * 0.05)
    h1, c5_1 = binary_local_search(best_h, dl1, seed)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: progressive grid doubling
    h_cur = best_h.copy()
    c5_cur = best_c5
    for target_n in [84, 168, 336, 672]:
        if rem() < 50 or target_n > 1000:
            break
        # Upsample
        h_up = h_cur.copy()
        while len(h_up) < target_n:
            h_up = upsample(h_up)
        h_up = h_up[:target_n]
        h_up = project_onto_constraint(h_up, target_n / 2.0)
        # Binary local search
        dl_bin = min(t0 + budget_s * 0.55, time.time() + min(rem() * 0.35, 120.0))
        if time.time() < dl_bin:
            h_b, c5_b = binary_local_search(h_up, dl_bin, seed + target_n)
            if c5_b < c5_cur:
                h_cur, c5_cur = h_b, c5_b
        # Continuous optimization
        if rem() > 40 and target_n <= 336:
            dl_cont = min(t0 + budget_s * 0.75, time.time() + rem() * 0.3)
            h_c, c5_c = continuous_optimize(h_cur, dl_cont, beta=80.0)
            if c5_c < c5_cur:
                h_cur, c5_cur = h_c, c5_c
            # Also try rounding
            h_rnd = (h_c > 0.5).astype(float)
            s = int(np.sum(h_rnd))
            tgt = target_n // 2
            if s != tgt:
                if s > tgt:
                    o1 = np.where(h_rnd == 1)[0]
                    h_rnd[o1[s - tgt:]] = 0.0
                else:
                    o0 = np.where(h_rnd == 0)[0]
                    h_rnd[o0[:tgt - s]] = 1.0
            c5_r = compute_c5(h_rnd)
            if c5_r < c5_cur:
                h_cur, c5_cur = h_rnd, c5_r
        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur

    # Phase 3: Try alternative starting structures and upsample
    if rem() > 80:
        for alt_seed in [seed + 100, seed + 200, seed + 300]:
            if rem() < 40:
                break
            n_alt = 42
            h_alt = random_binary(n_alt, alt_seed)
            dl_alt = min(t0 + budget_s * 0.7, time.time() + min(rem() * 0.2, 30.0))
            h_alt, c5_alt = binary_local_search(h_alt, dl_alt, alt_seed)
            # Upsample to 168
            h_alt_up = upsample(h_alt)
            h_alt_up = upsample(h_alt_up)
            h_alt_up = h_alt_up[:168]
            h_alt_up = project_onto_constraint(h_alt_up, 84.0)
            dl_alt2 = min(t0 + budget_s * 0.8, time.time() + min(rem() * 0.25, 60.0))
            if time.time() < dl_alt2:
                h_alt_up, c5_alt2 = binary_local_search(h_alt_up, dl_alt2, alt_seed + 1)
            if c5_alt < c5_alt2:
                h_alt_up, c5_alt2 = upsample(upsample(upsample(h_alt)))[:168], c5_alt
                h_alt_up = project_onto_constraint(h_alt_up, 84.0)
            if c5_alt2 < best_c5:
                best_h, best_c5 = h_alt_up, c5_alt2

    # Phase 4: Final continuous polish at best resolution
    if rem() > 30 and len(best_h) <= 336:
        dl_f = t0 + budget_s * 0.95
        h_f, c5_f = continuous_optimize(best_h, dl_f, beta=60.0)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Final: quick binary polish
    if rem() > 15:
        dl_fin = t0 + budget_s * 0.96
        h_fin, c5_fin = binary_local_search(best_h, dl_fin, seed + 7777)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
