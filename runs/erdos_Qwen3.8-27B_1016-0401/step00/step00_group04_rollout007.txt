    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:target]] = 1.0
    return h

def all_swaps(h, time_limit=30.0):
    n = len(h)
    cur = h.copy().astype(np.float64)
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    t0 = time.time()
    while time.time() - t0 < time_limit:
        oi = np.where(cur > 0.999)[0]
        zi = np.where(cur < 0.001)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        bm = None
        bd = 1e-14
        for i in oi:
            cur[i] = 0.0
            for j in zi:
                cur[j] = 1.0
                c5 = compute_c5(cur)
                d = cur_c5 - c5
                if d > bd:
                    bd = d
                    bm = (i, j)
                cur[j] = 0.0
            cur[i] = 1.0
        if bm is None:
            break
        i, j = bm
        cur[i] = 0.0
        cur[j] = 1.0
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best = cur.copy()
    return best, best_c5

def sa(h, time_limit=100.0, seed=42):
    rng = np.random.default_rng(seed)
    cur = h.copy().astype(np.float64)
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    t0 = time.time()
    Ts, Te = 0.04, 0.0003
    while time.time() - t0 < time_limit:
        f = min(1.0, (time.time() - t0) / max(1e-9, time_limit))
        T = Ts * (Te / Ts) ** f
        oi = np.where(cur > 0.999)[0]
        zi = np.where(cur < 0.001)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = int(rng.choice(oi))
        j = int(rng.choice(zi))
        cur[i] = 0.0
        cur[j] = 1.0
        nc = compute_c5(cur)
        d = nc - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_c5

def random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    T0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h0 = np.zeros(42)
        h0[::2] = 1.0

    bh = h0.copy()
    bc = compute_c5(bh)

    # Binary at original grid
    hb = to_binary(h0)
    cb = compute_c5(hb)
    if cb < bc:
        bc, bh = cb, hb.copy()
    n0 = len(bh)

    # Multi-start: try a few random inits at n0, keep best
    rem = budget_s - (time.time() - T0)
    t_rs = min(10.0, rem * 0.01)
    if t_rs > 1:
        for trial in range(3):
            if time.time() - T0 > budget_s - 100:
                break
            hr = random_binary(n0, rng)
            hr_g, cr_g = all_swaps(hr, min(t_rs / 3, 5.0))
            if cr_g < bc:
                bc, bh = cr_g, hr_g

    # Greedy at n0
    rem = budget_s - (time.time() - T0)
    t = min(30.0, rem * 0.03)
    if t > 1:
        h1, c1 = all_swaps(bh, t)
        if c1 < bc:
            bc, bh = c1, h1

    # SA round 1 at n0
    rem = budget_s - (time.time() - T0)
    t = min(150.0, rem * 0.14)
    if t > 5:
        h2, c2 = sa(bh, t, seed)
        if c2 < bc:
            bc, bh = c2, h2

    # SA round 2
    rem = budget_s - (time.time() - T0)
    t = min(150.0, rem * 0.14)
    if t > 5:
        h3, c3 = sa(bh, t, seed + 17)
        if c3 < bc:
            bc, bh = c3, h3

    # Grid doubling to 2*n0
    rem = budget_s - (time.time() - T0)
    if rem > 120:
        ht = np.concatenate([bh, bh])
        htb = to_binary(ht)
        t = min(50.0, rem * 0.05)
        if t > 2:
            h4, c4 = all_swaps(htb, t)
            if c4 < bc:
                bc, bh = c4, h4
        rem = budget_s - (time.time() - T0)
        if rem > 100:
            t = min(200.0, rem * 0.2)
            if t > 10:
                h5, c5v = sa(h4, t, seed + 41)
                if c5v < bc:
                    bc, bh = c5v, h5

    # Grid doubling again if we got good results
    rem = budget_s - (time.time() - T0)
    n_cur = len(bh)
    if rem > 100 and n_cur <= 84:
        ht2 = np.concatenate([bh, bh])
        htb2 = to_binary(ht2)
        t = min(40.0, rem * 0.05)
        if t > 2:
            h6, c6 = all_swaps(htb2, t)
            if c6 < bc:
                bc, bh = c6, h6
        rem = budget_s - (time.time() - T0)
        if rem > 80:
            t = min(150.0, rem * 0.2)
            if t > 10:
                h7, c7 = sa(h6, t, seed + 97)
                if c7 < bc:
                    bc, bh = c7, h7

    # Continue SA at best grid
    rem = budget_s - (time.time() - T0)
    if rem > 30:
        h8, c8 = sa(bh, rem * 0.7, seed + 200)
        if c8 < bc:
            bc, bh = c8, h8

    # Final greedy polish
    rem = budget_s - (time.time() - T0)
    if rem > 3:
        h9, c9 = all_swaps(bh, min(rem * 0.5, 20.0))
        if c9 < bc:
            bc, bh = c9, h9

    return bh, bc, len(bh)
