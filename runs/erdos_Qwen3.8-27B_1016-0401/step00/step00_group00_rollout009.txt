def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / alpha)

def proj_bounded(h, target_sum):
    n = len(h)
    x = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(300):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = x < (1.0 - 1e-12)
        else:
            free = x > 1e-12
        if not free.any():
            free = np.ones(n, dtype=bool)
        nf = free.sum()
        per = diff / nf
        if diff > 0:
