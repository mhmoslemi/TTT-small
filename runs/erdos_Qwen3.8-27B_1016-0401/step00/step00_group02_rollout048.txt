    h_new = project_feasible(res.x, n)
    c5_new = eval_c5(h_new)
    if c5_new < best_c5:
        best_h, best_c5 = h_new, c5_new

    # Phase 2: L-BFGS-B with softmax + penalty
    for beta, gamma in [(20, 100.0), (50, 500.0), (100, 2000.0), (300, 10000.0)]:
        if time.time() - t0 > budget_s * 0.45:
            break

        def p_obj(x, b=beta, g=gamma):
            v, _ = penalized_sm(x, b, g)
            return v

        def p_jac(x, b=beta, g=gamma):
            _, gr = penalized_sm(x, b, g)
            return gr

        res2 = minimize(p_obj, best_h, jac=p_jac, method='L-BFGS-B',
                        bounds=[(0, 1)] * n,
                        options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0})
        h_lb = project_feasible(res2.x, n)
        c5_lb = eval_c5(h_lb)
        if c5_lb < best_c5:
            best_h, best_c5 = h_lb, c5_lb

    # Phase 3: Try higher resolution n2
    n2 = n * 2
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n2)
    h_interp = np.interp(x_new, x_old, best_h)
    h_interp = project_feasible(h_interp, n2)
    c5_interp = eval_c5(h_interp)
    if c5_interp < best_c5 + 0.001:
        best_h2 = h_interp
        best_c5_2 = c5_interp

        def nm_obj2(x):
            return eval_c5(project_feasible(x, n2))

        res5 = minimize(nm_obj2, best_h2, method='Nelder-Mead',
                        options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
        h2_new = project_feasible(res5.x, n2)
        c5_2_new = eval_c5(h2_new)
        for beta2, gamma2 in [(50, 500.0), (200, 5000.0)]:
            if time.time() - t0 > budget_s * 0.7:
                break

            def p_obj2(x, b=beta2, g=gamma2):
                v, _ = penalized_sm(x, b, g)
                return v

            def p_jac2(x, b=beta2, g=gamma2):
                _, gr = penalized_sm(x, b, g)
                return gr

            res6 = minimize(p_obj2, best_h2, jac=p_jac2, method='L-BFGS-B',
                            bounds=[(0, 1)] * n2,
                            options={'maxiter': 1500, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0})
            h6 = project_feasible(res6.x, n2)
            c6 = eval_c5(h6)
            if c6 < best_c5_2:
                best_h2, best_c5_2 = h6, c6
        if best_c5_2 < best_c5:
            best_h, best_c5, n = best_h2, best_c5_2, n2

    # Phase 4: Perturbation restarts on best
    for it in range(60):
        if time.time() - t0 > budget_s * 0.92:
            break
        scale = 0.015 + 0.06 * (it / 60.0)
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = project_feasible(h_pert, len(best_h))

        def nm_obj_p(x):
            return eval_c5(project_feasible(x, len(best_h)))

        res3 = minimize(nm_obj_p, h_pert, method='Nelder-Mead',
                        options={'maxiter': 1500, 'xatol': 1e-13, 'fatol': 1e-15, 'adaptive': True})
        h_r = project_feasible(res3.x, len(best_h))
        c5_r = eval_c5(h_r)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 5: Final polish
    if time.time() - t0 < budget_s * 0.96:
        nn = len(best_h)
        def fin_obj(x, b=800.0, g=50000.0):
            v, _ = penalized_sm(x, b, g)
            return v
        def fin_jac(x, b=800.0, g=50000.0):
            _, gr = penalized_sm(x, b, g)
            return gr
        res7 = minimize(fin_obj, best_h, jac=fin_jac, method='L-BFGS-B',
                        bounds=[(0, 1)] * nn,
                        options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-13, 'iprint': 0})
        h_fin = project_feasible(res7.x, nn)
        c5_fin = eval_c5(h_fin)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    return (best_h, best_c5, len(best_h))
