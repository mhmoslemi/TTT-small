def smooth_max_grad(h, n, T=0.3):
    dx_val = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    weights = np.exp(T * shifted)
    wsum = weights.sum()

        if weights[k] / wsum < 1e-6:
            continue
        s = k - (n - 1)
        g += (weights[k] / wsum) * grad_shift(h, s)
    return g, corr.max() * dx_val
