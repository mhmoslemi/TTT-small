    #   dC[-e]/dh[j] = (1-h[j+e])*(j in [0,n-e-1]) - h[j-e]*(j in [e,n-1])
    #   (by symmetry of the formula with roles swapped)

    # dC[d]/dh[j] = (1-h[j-d])*(0<=j-d<n)*(0<=j<n) - h[j+d]*(0<=j<n)*(0<=j+d<n)
