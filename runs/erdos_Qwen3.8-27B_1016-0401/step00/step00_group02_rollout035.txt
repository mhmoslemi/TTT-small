    return (1.0/lam) * np.log(np.sum(np.exp(lam * vals)))

    dx = 2.0 / n
    # Compute all correlation values
    comp = 1 - h
    corr = np.correlate(h, comp, mode="full") * dx

    """Optimize using SLSQP."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(h):
        comp = 1 - h
        corr = np.correlate(h, comp, mode="full") * dx
        return smooth_max(corr, 50)
    
    def jac(h):
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros(n)
        for i in range(n):
            h2 = h.copy()
            h2[i] += eps
            # Renormalize
            diff = np.sum(h2) - target
            h2 -= diff / n
            h2 = np.clip(h2, 0, 1)
            grad[i] = (obj(h2) - obj(h)) / eps

        {'type': 'ineq', 'fun': lambda h: 1 - h, 'jac': lambda h: -np.eye(n)},

    bounds = [(0, 1)] * n

    # Project to feasible set
    h_opt = np.clip(res.x, 0, 1)
