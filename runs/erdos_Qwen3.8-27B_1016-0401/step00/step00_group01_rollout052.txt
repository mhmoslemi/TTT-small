import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project_box_sum(h, S, n):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binary_local_search(h, n, max_iters, time_limit_fn):
    """Local search on binary sequence: try all single swaps, take best."""
    best = h.copy()
    best_val = compute_c5(best)
    for it in range(max_iters):
        if time_limit_fn():
            break
