    corr = correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return correlate(h, 1.0 - h, mode='full') * dx

def upsample(h, n_new):
    n = len(h)
    x_old = np.arange(n) * 2.0 / n
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    h_new += (n_new / 2.0 - h_new.sum()) / n_new
    return np.clip(h_new, 0, 1)

def smooth_h(h, sigma):
    n = len(h)
    center = n // 2
    kx = np.concatenate([np.arange(center + 1), -np.arange(center - 1, 0, -1)])
    kernel = np.exp(-kx.astype(float)**2 / (2.0 * sigma**2))
    kernel /= kernel.sum()
    from numpy.fft import rfft, irfft
    h_s = irfft(rfft(h) * rfft(kernel), n)
    h_s = np.clip(h_s, 0, 1)
    h_s += (n / 2.0 - h_s.sum()) / n
    return np.clip(h_s, 0, 1)

def pair_descent(h, rng, t_start, t_limit):
    n = len(h)
    h = h.astype(float).copy()
    best = eval_c5(h)
    deltas = np.array([0.03, 0.07, 0.15, 0.3])
    n_try = 6 * n
    stall = 0

    while time.time() - t_start < t_limit:
        improved = False
        for _ in range(n_try):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                if h[i] >= d and h[j] + d <= 1.0:
                    h[i] -= d; h[j] += d
                    c = eval_c5(h)
                    if c < best - 1e-14:
                        best = c; improved = True
                    else:
                        h[i] += d; h[j] -= d
                if h[j] >= d and h[i] + d <= 1.0:
                    h[i] += d; h[j] -= d
                    c = eval_c5(h)
                    if c < best - 1e-14:
                        best = c; improved = True
                    else:
                        h[i] -= d; h[j] += d
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
            deltas = deltas * 0.6
            if deltas[0] < 1e-5:
                break
    return h, best

def worst_shift_refine(h, rng, t_start, t_limit):
    n = len(h)
    h = h.astype(float).copy()
    best = eval_c5(h)

    while time.time() - t_start < t_limit:
        corrs = get_corrs(h)
        k_worst = int(np.argmax(corrs))
        shift = k_worst - (n - 1)

        improved = False
        n_try = 8 * n
        for _ in range(n_try):
            i = rng.randint(n)
            if i + shift >= 0 and i + shift < n:
                j = i + shift
            elif i - shift >= 0 and i - shift < n:
                j = i - shift
            else:
                j = rng.randint(n)
            if i == j:
                continue
            d = 0.02 + 0.15 * rng.random()
            # Align: make h[i] and h[j] more similar to increase h[i]*h[j]
            if h[i] < 0.5:
                di = d if h[i] + d <= 1 else 1 - h[i]
                dj = d if h[j] - d >= 0 else h[j]
                if di > 0 and dj > 0:
                    h[i] += di; h[j] -= dj
                    c = eval_c5(h)
                    if c < best - 1e-14:
                        best = c; improved = True
                    else:
                        h[i] -= di; h[j] += dj
            else:
                dj = d if h[j] + d <= 1 else 1 - h[j]
                di = d if h[i] - d >= 0 else h[i]
                if dj > 0 and di > 0:
                    h[j] += dj; h[i] -= di
                    c = eval_c5(h)
                    if c < best - 1e-14:
                        best = c; improved = True
                    else:
                        h[j] -= dj; h[i] += di

        if not improved:
            break
    return h, best

def block_flip(h, rng, t_start, t_limit):
    n = len(h)
    h = h.astype(float).copy()
    best = eval_c5(h)
    block_sizes = [2, 3, 4, 5]

    while time.time() - t_start < t_limit:
        improved = False
        for _ in range(4 * n):
            size = int(rng.choice(block_sizes))
            i = rng.randint(n - size + 1)
            target = rng.choice([0.0, 1.0])
            orig = h[i:i+size].copy()
            h[i:i+size] = target
            c = eval_c5(h)
            if c < best - 1e-14:
                best = c; improved = True
            else:
                h[i:i+size] = orig
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.clip(np.asarray(initial_h_values, dtype=float), 0, 1)
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = eval_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: coordinate descent at initial resolution
    if rem() > 30:
        h, c5 = pair_descent(h_init, rng, time.time(), min(150, rem() * 0.25))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: worst-shift refinement
    if rem() > 20:
        h, c5 = worst_shift_refine(best_h, rng, time.time(), min(80, rem() * 0.15))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: block flips
    if rem() > 20:
        h, c5 = block_flip(best_h, rng, time.time(), min(60, rem() * 0.1))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: try higher resolutions
    for n2 in [2 * n0, 4 * n0, 8 * n0]:
        if rem() < 40 or n2 > 600:
            break
        h2 = upsample(best_h, n2)
        h2 = smooth_h(h2, 1.0)
        h2, c5_2 = pair_descent(h2, rng, time.time(), min(100, rem() * 0.2))
        h2, c5_2 = worst_shift_refine(h2, rng, time.time(), min(40, rem() * 0.1))
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 5: final polish at best resolution
    if rem() > 15:
        h, c5 = pair_descent(best_h, rng, time.time(), min(80, rem() * 0.25))
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    if rem() > 10:
        h, c5 = worst_shift_refine(best_h, rng, time.time(), min(30, rem() * 0.25))
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    if rem() > 5:
        h, c5 = block_flip(best_h, rng, time.time(), min(15, rem() * 0.3))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Final constraint enforcement
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, n)
