                hc2, c52 = _try_optimize(best_h, T_alt, n, target, bounds, deadline - 40)

                    best_c5 = c52
                    best_h = hc2.copy()
    
    # Final polish with low T
    if time.time() < deadline - 5:
        best_h = _polish(best_h, 10.0, n, target, 40)
    best_h = _proj_sum(best_h, target)
    best_c5 = _compute_c5(best_h)
    
    return (best_h, best_c5, n)
