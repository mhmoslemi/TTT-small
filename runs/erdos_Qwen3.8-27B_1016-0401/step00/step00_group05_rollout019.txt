    c = np.correlate(h, 1.0 - h, mode='full') * dx
    p_max = int(np.argmax(c))
    d = p_max - (n - 1)
    grad = np.zeros(n)
    js = max(0, d)
    je = min(n, d + n)
    if js < je:
        grad[js:je] += dx * (1.0 - h[js - d:je - d])
    js2 = max(0, -d)
    je2 = min(n, n - d)
    if js2 < je2:
        grad[js2:je2] += -dx * h[js2 + d:je2 + d]
    return grad

def optimize_h(h0, t_limit, alpha_start=20.0, alpha_max=300.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    alpha = alpha_start
    lr = 0.08
    stall = 0
    while time.time() - t0 < t_limit:
        g = c5_smooth_grad(h, alpha)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        improved = False
        for _ in range(8):
            h_new = h - lr * g
            h_new = project_h(h_new, n)
            c5_new = eval_c5(h_new)
            if c5_new < eval_c5(h) - 1e-12:
                h = h_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                break
            lr *= 0.5
        if not improved:
            stall += 1
        else:
            lr = min(lr * 1.03, 0.5)
        if stall > 30:
            if alpha < alpha_max:
                alpha = min(alpha * 1.4, alpha_max)
                lr = 0.08
            else:
                pert = np.random.randn(n) * 0.08
                pert -= pert.mean()
                h = project_h(h + pert, n)
                lr = 0.06
            stall = 0
        alpha = min(alpha * 1.003, alpha_max)
    return best_h, best_c5

def optimize_h_worst(h0, t_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    lr = 0.05
    stall = 0
    while time.time() - t0 < t_limit:
        g = worst_lag_grad(h)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        improved = False
        for _ in range(6):
            h_new = h - lr * g
            h_new = project_h(h_new, n)
            c5_new = eval_c5(h_new)
            if c5_new < eval_c5(h) - 1e-12:
                h = h_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                lr = min(lr * 1.04, 0.3)
                break
            lr *= 0.5
        if not improved:
            stall += 1
            if stall > 40:
                pert = np.random.randn(n) * 0.05
                pert -= pert.mean()
                h = project_h(h + pert, n)
                lr = 0.05
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return max(0, budget_s - elapsed())

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(70, 0.5)
    n0 = len(h0)
    h0 = project_h(h0, n0)

    best_h = h0.copy()
    best_c5 = eval_c5(h0)

    # Phase 1: Optimize at original resolution
    t_lim = min(remaining() * 0.3, 280)
    if t_lim > 20:
        h1, c1 = optimize_h(h0.copy(), t_lim, 20, 250)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Multi-resolution
    n_candidates = sorted(set([n0, 80, 100, 140, 200, 300]))
    for nt in n_candidates:
        if nt == n0:
            continue
        if remaining() < 80:
            break
        t_lim = min(remaining() * 0.12, 120, remaining() - 60)
        if t_lim < 15:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, nt)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_h(hi, nt)
        h2, c2 = optimize_h(hi, t_lim, 15, 200)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Worst-lag refinement at best n
    if remaining() > 50:
        t_lim = min(remaining() * 0.2, 100, remaining() - 30)
        h3, c3 = optimize_h_worst(best_h.copy(), t_lim)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Restart loop
    nb = len(best_h)
    while remaining() > 40:
        t_lim = min(35, remaining() - 20)
        if t_lim < 10:
            break
        scale = 0.04 + 0.04 * np.random.rand()
        hp = best_h + scale * np.random.randn(nb)
        hp = project_h(hp, nb)
        h4, c4 = optimize_h(hp, t_lim, 30, 300)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Final polish
    if remaining() > 15:
        h5, c5 = optimize_h_worst(best_h.copy(), min(remaining() - 5, 15))
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    best_h = project_h(best_h, len(best_h))
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
