```python
import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0

def _constraint_fun(x):
    return x.sum() - _g_target

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, T):
    m = float(np.max(vals))
    shifted = T * (vals - m)
    return m + float(np.log(np.sum(np.exp(shifted)))) / T

def project_feasible(x, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def gradient_corr(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        nm = n - k
        grad[0:nm] += 1.0 - h[k:n]
        grad[k:n] -= h[0:nm]
    else:
        kn = -k
        nm = n - kn
        grad[kn:n] += 1.0 - h[0:nm]
        grad[0:nm] -= h[kn:n]
    return grad

def slsqp_optimize(h, n, target, dx, T, maxiter=600):
    global _g_target
    _g_target = target
    def obj(h):
        return smooth_max(np.correlate(h, 1.0 - h, mode="full") * dx, T)
    bounds = [(0, 1)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=[{'type': 'eq', 'fun': _constraint_fun}],
                  options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, target)

def subgradient_step(h, n, target, dx, lr):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_max = int(np.argmax(corr))
    k = k_max - (n - 1)
    grad = gradient_corr(h, k)
    h_new = project_feasible(h - lr * grad * dx, target)
    return h_new, eval_c5(h_new)

def nm_objective_factory(target, n):
    def obj(x):
        hf = np.empty(n)
        hf[:-1] = x
        hf[-1] = target - x.sum()
        if np.any(hf < -0.2) or np.any(hf > 1.2):
            return 10.0 + np.sum((np.clip(hf, 0, 1) - hf) ** 2)
        return eval_c5(np.clip(hf, 0, 1))
    return obj

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.ones(45) * 0.5

    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)

    # Phase 1: SLSQP with increasing smoothness
    for T in [15, 30, 60, 120, 250, 500, 1000]:
        if time.time() - t0 > budget_s * 0.15:
            break
        h_new = slsqp_optimize(best_h, n, target, dx, T, maxiter=1000)
        c5 = eval_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_new

    # Phase 2: Subgradient with adaptive step
    if time.time() - t0 < budget_s * 0.5:
        h = best_h.copy()
        base_lr = 0.15
        stall = 0
        for it in range(20000):
            if time.time() - t0 > budget_s * 0.5:
                break
            lr = base_lr / (1.0 + 0.0002 * it)
            h_new, c5_new = subgradient_step(h, n, target, dx, lr)
            c5_cur = eval_c5(h)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                stall = 0
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                stall += 1
                if stall > 200:
                    base_lr *= 0.7
                    stall = 0
                    if base_lr < 1e-6:
                        break

    # Phase 3: Nelder-Mead polish in reduced space
    if time.time() - t0 < budget_s * 0.65:
        obj_nm = nm_objective_factory(target, n)
        h_red = best_h[:-1].copy()
        res = minimize(obj_nm, h_red, method='Nelder-Mead',
                      options={'maxiter': 30000, 'xatol': 1e-11,
                               'fatol': 1e-14, 'adaptive': True})
        h_new = np.empty(n)
        h_new[:-1] = res.x
        h_new[-1] = target - res.x.sum()
        h_new = project_feasible(h_new, target)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    while time.time() - t0 < budget_s * 0.96:
        scale = 0.01 + 0.03 * rng.rand()
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    return (best_h, best_c5, n)
```
