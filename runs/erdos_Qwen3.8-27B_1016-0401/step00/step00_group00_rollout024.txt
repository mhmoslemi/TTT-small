For the gradient:
∂F_α/∂h[j] = Σ_k exp(α * c_k * dx) / Σ exp(α * c_k * dx) * (∂c_k/∂h[j]) * dx

This gives a weighted gradient where more weight is on larger c_k.

```python
def smooth_objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    comp = 1 - h
    corr = np.correlate(h, comp, mode="full")
    vals = alpha * corr * dx
    vals -= np.max(vals)  # for numerical stability
    weights = np.exp(vals)
    weights /= weights.sum()
    
    obj = (1.0/alpha) * (np.log(np.sum(np.exp(vals))) + alpha * np.max(corr*dx))
    # Actually: smooth max = (1/alpha)*log(sum(exp(alpha*x_i)))
    # = (1/alpha)*log(sum(exp(alpha*(x_i - max_x)))) + max_x
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = k - (n-1)
        lo = max(0, -s)
        hi = min(n, n-s)
        for i in range(lo, hi):
            j2 = i + s
            if 0 <= j2 < n:
                grad[i] += weights[k] * (1 - h[j2])
                grad[j2] -= weights[k] * h[i]
    grad *= dx
    
    return obj, grad
```

```python
        if abs(d) < 1e-14:
```

```python
    exps = np.exp(alpha * (xc - xm))
```
