```python
            break
        t_frac = it / max_iter
        lr = lr_init * 0.5 * (1 + np.cos(np.pi * t_frac))
        lr = max(lr, 1e-5)
        g = get_subgrad(h)
        h_new = project(h - lr * g, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Occasionally add noise
        if it % 5000 == 2000:
            h = project(h + 0.02 * rng.standard_normal(n), target)
    
    # Phase 2: SLSQP on smoothed objective
    if time_left() > 120:
        for temp in [50.0, 20.0, 10.0, 5.0]:
            if time_left() < 60:
                break
            scale = 1.0 / temp
            def objective(hv):
                corr = np.correlate(hv, 1.0 - hv, mode='full')
                # Smooth max: (1/scale)*log(sum(exp(scale*corr)))
                # But for minimization, we want to minimize the smooth max
                max_c = np.max(corr)
                shifted = corr - max_c
                return (max_c + np.log(np.sum(np.exp(shifted * scale))) / scale) * dx
            
            def objective_grad(hv):
                corr = np.correlate(hv, 1.0 - hv, mode='full')
                max_c = np.max(corr)
                shifted = corr - max_c
                weights = np.exp(shifted * scale) / np.sum(np.exp(shifted * scale))
                grad = np.zeros(n)
                for pos in range(len(corr)):
                    w = weights[pos]
                    if w < 1e-12:
                        continue
                    lag = pos - (n - 1)
                    for j in range(n):
                        jp = j + lag
                        if 0 <= jp < n:
                            grad[j] += w * (1.0 - hv[jp])
                        jm = j - lag
                        if 0 <= jm < n:
                            grad[j] -= w * hv[jm]
                return grad * dx
            
            cons = [{'type': 'eq', 'fun': lambda hv: hv.sum() - target}]
            bnds = [(0.0, 1.0)] * n
            res = minimize(objective, best_h.copy(), jac=objective_grad,
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': 100, 'ftol': 1e-12})
            if res.fun < best_c5 or res.success:
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = res.x.copy()
    
    # Phase 3: Fine-tuning with smaller subgradient steps
    if time_left() > 30:
        for lr in [0.002, 0.001, 0.0005, 0.0002]:
            if time_left() < 30:
                break
            h = best_h.copy()
            for it in range(50000):
                if time_left() < 30:

                g = get_subgrad(h)
                h = project(h - lr * g, target)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5, n
```

```python
def get_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = np.max(corr)
    active = np.where(corr >= mv - 1e-9)[0]
    grad = np.zeros(n)
    for pos in active:
        lag = pos - (n - 1)
        # (1-h[j+lag]) for valid j: j+lag in [0,n-1], so j in [-lag, n-1-lag]
        j_start = max(0, -lag)
        j_end = min(n, n - lag)  # exclusive
        if j_end > j_start:
            grad[j_start:j_end] += 1.0 - h[j_start + lag:j_end + lag]
        # -h[j-lag] for valid j: j-lag in [0,n-1], so j in [lag, n-1+lag]
        j_start2 = max(0, lag)
        j_end2 = min(n, n + lag)  # exclusive
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= h[j_start2 - lag:j_end2 - lag]
    return grad / len(active) * dx
```

- For the first term: we need j such that 0 ≤ j < n and 0 ≤ j+lag < n

  - And h[j+lag] ranges from h[max(0,-lag)+lag] to h[min(n,n-lag)-1+lag]
  - If lag ≥ 0: j from 0 to n-lag-1, h[lag] to h[n-1]

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def get_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = np.max(corr)
    active = np.where(corr >= mv - 1e-9)[0]
    grad = np.zeros(n)
    for pos in active:
        lag = pos - (n - 1)
        js = max(0, -lag)
        je = min(n, n - lag)
        if je > js:
            grad[js:je] += 1.0 - h[js + lag:je + lag]
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if je2 > js2:
            grad[js2:je2] -= h[js2 - lag:je2 - lag]
    return grad / len(active) * dx

def project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0 - 1e-13) if d > 0 else (h > 1e-13)
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(hv, n, dx, scale):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(shifted * scale))) / scale) * dx

def smooth_grad(hv, n, dx, scale):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(shifted * scale) / np.sum(np.exp(shifted * scale))
    grad = np.zeros(n)
    for pos in range(len(corr)):
        w = weights[pos]
        if w < 1e-14:
            continue
        lag = pos - (n - 1)
        js = max(0, -lag)
        je = min(n, n - lag)
        if je > js:
            grad[js:je] += w * (1.0 - hv[js + lag:je + lag])
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if je2 > js2:
            grad[js2:je2] -= w * hv[js2 - lag:je2 - lag]
    return grad * dx

def subgrad_optimize(h, n, target, dx, lr_init, n_iter, rng, time_check, best_ref):
    h = h.copy()
    best_c5 = best_ref[0]
    best_h = best_ref[1]
    for it in range(n_iter):
        if time_check():
            break
        frac = it / max(n_iter, 1)
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, 1e-6)
        g = get_subgrad(h)
        h = project(h - lr * g, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 5000 == 3000:
            h = project(h + 0.03 * rng.standard_normal(n), target)
    return best_c5, best_h

def try_slsqp(h0, n, target, dx, scale, time_check, best_ref):
    best_c5 = best_ref[0]
    best_h = best_ref[1]
    def obj(hv):
        return smooth_obj(hv, n, dx, scale)
    def obj_grad(hv):
        return smooth_grad(hv, n, dx, scale)
    cons = [{'type': 'eq', 'fun': lambda hv: hv.sum() - target, 'jac': lambda hv: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-12})
        if res.x is not None:
            c5r = compute_c5(res.x)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = res.x.copy()
    except Exception:
        pass
    return best_c5, best_h
```
