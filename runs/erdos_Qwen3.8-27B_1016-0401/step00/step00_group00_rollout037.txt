def project_onto_constraint(h, S):

        diff = S - h.sum()
        if abs(diff) < 1e-12:
            break
        # Find offset to add
        # We need sum(clip(h + t, 0, 1)) = S
        # Binary search on t
        lo, hi = -1.0, 1.0
        for _ in range(60):

            if val < S:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo+hi)/2, 0, 1)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return c.max(), c
