```python
import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(corr_full(h)) * dx

def grad_max_corr(h):
    """Gradient of max_k c_k with respect to h"""
    n = len(h)
    c = corr_full(h)  # length 2n-1
    k_max = np.argmax(c)
    # Convert k_max to lag: lag = k_max - (n-1)
    lag = k_max - (n - 1)
    
    g = np.zeros(n)
    for m in range(n):
        # Contribution from h[m] as first factor: lag = m - n_index
        # c_lag = sum_n h[n+lag] * (1-h[n])
        # h[m] is first factor when n + lag = m => n = m - lag
        n1 = m - lag
        if 0 <= n1 < n:
            g[m] += (1 - h[n1])
        # h[m] is second factor when n = m
        n2 = m
        if 0 <= lag + n2 < n:
            g[m] -= h[n2 + lag]
    
    return g * (2.0 / n)  # include dx factor

def project_onto_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) < 1e-10:
        return h
    # Water-filling
    diff = target - current
    if diff > 0:
        # Need to increase sum: increase zeros more
        h2 = h + diff / n
        return np.clip(h2, 0, 1)
    else:
        h2 = h + diff / n
        return np.clip(h2, 0, 1)

def gradient_descent(h0, n, n_iter=5000, lr=0.01):
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_s = score(h)
    
    for i in range(n_iter):
        g = grad_max_corr(h)
        h -= lr * g
        h = project_onto_feasible(h, n)
        s = score(h)
        if s < best_s:
            best_s = s
            best_h = h.copy()
    
    return best_h, best_s
```

```python
import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(corr_full(h)) * dx

def sa_search(h, n, rng, time_limit=60.0, T_init=0.03):
    h = h.copy().astype(float)
    best_h = h.copy()
    cur_s = score(h)
    best_s = cur_s
    T = T_init
    decay = 0.9995

    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h < 0.5)[0]

    start = time.time()
    n_iters = 0
    while time.time() - start < time_limit and T > 1e-9:
        idx_i = rng.integers(len(ones_list))
        idx_j = rng.integers(len(zeros_list))
        i = ones_list[idx_i]
        j = zeros_list[idx_j]

        h[i] = 0.0
        h[j] = 1.0
        new_s = score(h)
        delta = new_s - cur_s

        accept = delta <= 0
        if not accept and T > 1e-9:
            if delta / T < 50.0:
                accept = rng.random() < np.exp(-delta / T)
            else:
                accept = False

        if accept:
            cur_s = new_s
            ones_list[idx_i] = j
            zeros_list[idx_j] = i
            if cur_s < best_s - 1e-14:
                best_s = cur_s
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

        T *= decay
        n_iters += 1

    return best_h, best_s

def gradient_step(h, n, lr=0.02, n_inner=200):
    """Gradient descent on continuous relaxation targeting max correlation lag"""
    h = h.copy().astype(float)
    best_h = h.copy()
    best_s = score(h)
    dx = 2.0 / n

    for _ in range(n_inner):
        c = corr_full(h)
        k_max = np.argmax(c)
        lag = k_max - (n - 1)
        g = np.zeros(n)
        for m in range(n):
            n1 = m - lag
            if 0 <= n1 < n:
                g[m] += (1.0 - h[n1])
            n2 = m + lag
            if 0 <= n2 < n:
                g[m] -= h[n2]
        h_new = h - lr * g * dx
        # Project to 0<=h<=1 and sum=n/2
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        s = score(h_new)
        if s < best_s:
            best_s = s
            best_h = h_new.copy()
        if s <= best_s + 1e-10:
            h = h_new
        else:
            lr *= 0.8
    return best_h, best_s

    target = n // 2
    h_sorted = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[h_sorted[:target]] = 1.0
    return h_bin
```
