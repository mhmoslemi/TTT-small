import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_soft(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + tau * np.log(np.sum(np.exp((corr - m) / tau) + 1e-30)))

def enforce_constraint(h, n):

            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)

        idx = int(np.argmax(h < 1.0) if np.any(h < 1.0) else 0)
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def sum_cons(h):
    return h.sum() - len(h) / 2.0

def sum_jac(h):
    return np.ones(len(h))

def slsqp_opt(h0, n, maxiter=400, ftol=1e-13):
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': sum_cons, 'jac': sum_jac}]
    res = minimize(compute_c5, h0, method='SLSQP',
                   constraints=cons, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h = enforce_constraint(res.x, n)
    v = compute_c5(h)
    return h, v

def nm_opt(h0, n, maxiter=3000):

        hc = np.clip(x, 0, 1)
        pen = 0.5 * (hc.sum() - n / 2.0) ** 2

    res = minimize(penalty_obj, h0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-12})
