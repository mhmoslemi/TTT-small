    n = len(h)
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    c5 = c.max()

            L = -lag

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    return np.max(c) * dx


def project_constraint(h, target_sum):
    lo, hi = -1.5, 2.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]

            continue

            grad[:n - lag] -= wk * h[lag:]

            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad, cmax * (2.0 / n)
