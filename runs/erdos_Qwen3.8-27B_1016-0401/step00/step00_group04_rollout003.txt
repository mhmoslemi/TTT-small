import numpy as np
import time

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_all_overlaps(h):
    """Return all correlation values * dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return np.max(compute_all_overlaps(h))

def subgrad_lag(h, k):
    """Subgradient of C[k] w.r.t. h, where k is in [-(n-1), n-1]"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)

    hi2 = min(n, n - k)
    if lo2 < hi2:
        idx2 = np.arange(lo2, hi2)
        grad[idx2] -= h[idx2 + k]

Actually, let me reconsider the subgradient computation. For the full correlation, we have 2n-1 lags. The "max" over all these is what we're minimizing. At any point, typically only a few lags are near the maximum.

A key insight: the subgradient of max(f_1, ..., f_m) at a point where f_j is the unique max is ∇f_j. If there's a tie, it's any convex combination.

For robustness, let me use a "smoothing" approach:
f_ε(h) = ε * log(sum_k exp(C[k]/ε))

This is a smooth approximation of max_k C[k], and its gradient is:
∇f_ε = sum_k p_k * ∇C[k] where p_k = exp(C[k]/ε) / sum_l exp(C[l]/ε)

This is the softmax-weighted average of subgradients. As ε → 0, it approaches the subgradient of the max.

This is much more stable than using just one active constraint.

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def smooth_gradient(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = C.max()
    # Softmax weights (shifted for stability)
    shifted = (C - cmax) / epsilon
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(C)
        w[np.argmax(C)] = 1.0
    else:
        w = w / w_sum
    # Compute weighted subgradient
    grad = np.zeros(n)
    for ki in range(len(C)):
        if w[ki] < 1e-8:
            continue
        k = ki - (n - 1)
        wi = w[ki]
        # dC[k]/dh[j] = dx * [(1-h[j-k]) if 0<=j-k<n else 0 - h[j+k] if 0<=j+k<n else 0]
        # Vectorized:
        # Term 1: for j in [max(0,k), min(n, k+n)-1], add wi*dx*(1-h[j-k])
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:
            idx1 = np.arange(lo1, hi1)
            grad[idx1] += wi * dx * (1.0 - h[idx1 - k])
        # Term 2: for j in [max(0,-k), min(n, n-k)-1], subtract wi*dx*h[j+k]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= wi * dx * h[idx2 + k]
    return grad, cmax

import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def active_subgradient(h, C, top_k=8):
    n = len(h)
    dx = 2.0 / n
    cmax = C.max()
    # Softmax over top lags for stability
    order = np.argsort(C)[::-1]
    top_idx = order[:top_k]
    top_C = C[top_idx]
    shifted = (top_C - cmax) / 0.015
    w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for kk in range(top_k):
        if w[kk] < 0.005:
            continue
        k = int(top_idx[kk]) - (n - 1)
        wi = w[kk]
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:
            idx1 = np.arange(lo1, hi1)
            grad[idx1] += wi * dx * (1.0 - h[idx1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= wi * dx * h[idx2 + k]
    return grad
