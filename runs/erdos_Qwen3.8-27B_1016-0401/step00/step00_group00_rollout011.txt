import numpy as np
import time

def _project(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2}"""
    target = n / 2.0
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(p)
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return p

def _c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_c5(h, n):
    """Compute subgradient of C5 w.r.t. h (gradient of max lag correlation)"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    k = k_max - (n - 1)  # actual lag
    grad = np.zeros(n)
    for j in range(n):
        # Contribution from h[j] as the first factor: (1 - h[j+k]) if j+k in [0,n)
        if 0 <= j + k < n:
            grad[j] += (1.0 - h[j + k])
        # Contribution from h[j] as the second factor (h[i+k] = h[j], so i = j-k): -h[j-k] if j-k in [0,n)
        if 0 <= j - k < n:
            grad[j] -= h[j - k]
    return grad

import numpy as np

def _project(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2} via bisection on shift."""
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, 'full') * dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_active(h, n):
    """Analytical subgradient of C5 (gradient of max-lag correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr)) - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        ak = -k
        grad[ak:] += (1.0 - h[:n - ak])
        grad[:n - ak] -= h[ak:]
    return grad

def _pgd(h, n, dx, max_iters=500, lr_init=0.05):
    """Projected gradient descent minimizing C5."""
    h_curr = h.copy()
    best = h_curr.copy()
    best_val = _c5(h_curr, dx)
    lr = lr_init
    stall = 0
    for it in range(max_iters):

            best_val = c5_curr
            best = h_curr.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        grad = _grad_active(h_curr, n)
        h_new = _project(h_curr - lr * grad, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_curr - 1e-12:
            h_curr = h_new
            lr = min(lr * 1.05, 0.5)

    """Generate Thue-Morse 0/1 sequence of length n."""
    tm = [0]
    for _ in range(12):
        tm = tm + [1 - x for x in tm]
        if len(tm) >= n:
            break
    return np.array(tm[:n], dtype=float)

def _cma_es(mean, n, dx, rng, sigma_init=0.08, max_gamma=300):
    """Simplified CMA-ES for minimizing C5."""
    sigma = sigma_init
    n_pop = max(12, int(3 + 3 * np.log(n)))
    C = np.eye(n)
    best = mean.copy()
    best_val = _c5(mean, dx)
    mu = n_pop // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights /= np.sum(weights)
    c_cov = 1.0 / (5 * n + 5)
    d_cov = 1.0 + 2.0 * max((mu - 1) / (n + 1), 0.0) + c_cov

    for _ in range(max_gamma):
        pop = []
        for _ in range(n_pop):
            z = rng.standard_normal(n)
            try:
                step = np.linalg.cholesky(C) @ z
            except np.linalg.LinAlgError:
                step = z
            pop.append(_project(mean + sigma * step, n))
        vals = np.array([_c5(p, dx) for p in pop])
        order = np.argsort(vals)
        new_mean = np.zeros(n)
        for i in range(mu):
            new_mean += weights[i] * pop[order[i]]
        # Rank-1 covariance update
        d = (new_mean - mean) / sigma
        C = (1 - c_cov * d_cov * mu) * C + c_cov * d_cov * (mu + 1 - mu) * np.outer(d, d)
        mean = new_mean
        sigma *= np.exp(0.5 * (np.mean(vals[order[:mu]]) / (best_val + 1e-10) - 1.0) / (n + 2))
        sigma = np.clip(sigma, 0.005, 0.5)
        if vals[order[0]] < best_val:
            best_val = vals[order[0]]
