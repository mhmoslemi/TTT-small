    Tc = T * corr
    m = np.max(Tc)
    e = np.exp(Tc - m)
    s = np.sum(e)
    val = (m + np.log(s)) / T
    w = e / s
    return val, w

    n = len(h)
    grad = np.zeros(n)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(compute_corr_full(h)) * (2.0 / len(h)))

def smooth_max(corr, T):
    Tc = T * corr
    m = np.max(Tc)
    e = np.exp(Tc - m)
    s = np.sum(e)
    if s < 1e-300:
        n = len(corr)
        v = np.zeros(n); v[np.argmax(corr)] = 1.0
        return corr[np.argmax(corr)], v
    return (m + np.log(s)) / T, e / s

    n = len(h)
    grad = np.zeros(n)
    ms, me = max(0, k), min(n, n + k)
    if ms < me:
        grad[ms:me] += (1.0 - h[ms-k:me-k])
    ms2, me2 = max(0, -k), min(n, n - k)
    if ms2 < me2:
        grad[ms2:me2] -= h[ms2+k:me2+k]
    return grad
