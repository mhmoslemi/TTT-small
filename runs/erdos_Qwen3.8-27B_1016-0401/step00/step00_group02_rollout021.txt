import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def resample_nn(h, n_new):
    n_old = len(h)
    idx = np.minimum((np.arange(n_new) * n_old) // n_new, n_old - 1)
    return h[idx].copy()

def make_binary_feasible(h, n, rng):
    vals = (h >= 0.5).astype(float)
    target = n // 2
    cur = int(vals.sum())
    if cur > target:
        idx = np.where(vals == 1)[0]
        rng.shuffle(idx)
        vals[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(vals == 0)[0]
        rng.shuffle(idx)
        vals[idx[:target - cur]] = 1
    return vals

def sa_optimize(h, rng, t_deadline, T0=0.05, cool_rate=0.997, reheat_T=0.02):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T0
    iters = 0
    while time.time() < t_deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta <= 0 or (T > 1e-9 and rng.rand() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        iters += 1
        if iters % 800 == 0:
            T *= cool_rate
            if T < reheat_T * 0.5:
                T = reheat_T
    return best_h, best_c5

def polish(h, rng, t_deadline):
    n = len(h)
    h = h.copy()

    changed = True

        zeros = list(np.where(h == 0)[0])
        rng.shuffle(ones)
        rng.shuffle(zeros)

                new_c5 = compute_c5(h)
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    changed = True
                    ones = list(np.where(h == 1)[0])
                    zeros = list(np.where(h == 0)[0])
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
    return h, best_c5

def periodic_construct(n, period, rng):
    h = np.zeros(n)
    duty = max(1, period // 2)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return make_binary_feasible(h, n, rng)

def random_construct(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h
