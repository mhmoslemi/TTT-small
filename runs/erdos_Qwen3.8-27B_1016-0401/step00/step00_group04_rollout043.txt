        top_idx = np.argsort(ov)[::-1][:min(25, len(ov))]
        for idx in top_idx:
            if w[idx] < 1e-8:
                continue
            l = int(idx) - (n - 1)
            g += w[idx] * _grad_l(h, l)

        # Adaptive step
        h2 = _proj(h - lr * g, n)
        c2 = _c5(h2)

        if c2 < cur - 1e-12:
            h = h2
            lr = min(lr * 1.03, 4.0)
        else:
            lr *= 0.7
            if lr < 0.002:
                h = _proj(best_h + rng.standard_normal(n) * 0.06, n)
                lr = 0.3
                stall = int(stall * 0.5)

        if stall > 500:
            # Larger kick
            h = _proj(best_h + rng.standard_normal(n) * 0.3, n)
            if _c5(h) > best + 0.02:
                h = best_h.copy()
            lr = 0.4
            stall = 0
            # Gradually increase alpha for sharper max
            alpha = min(alpha * 1.05, 80.0)

    return best_h, best

def _try_binary_start(n, rng, t_end):
    """Optimize from a random binary starting point."""
    idx = rng.choice(n, size=n // 2, replace=False)
    hb = np.zeros(n)
    hb[idx] = 1.0
    return _optimize(hb, n, t_end, rng)

def _try_intermediate_start(n, rng, t_end):
    """Start from a smooth bell-like shape modulated to integral 1."""
    x = np.linspace(0, 1, n)
    hb = 0.5 + 0.4 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    hb = _proj(hb, n)
    return _optimize(hb, n, t_end, rng)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    results = []

    # Phase 1: optimize the parent construction (highest priority)
    if h0 is not None and len(h0) >= 10:
        n = len(h0)
        t_end1 = t0 + budget_s * 0.5
        if t_end1 > t0 + 400:
            t_end1 = t0 + 400
        bh, bc = _optimize(h0, n, t_end1, rng)
        results.append((bh, bc, n))

    # Phase 2: other resolutions and starts
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 20
    if remaining > 40:
        n_candidates = [42, 56, 72, 84, 100, 120]
        if h0 is not None:
            n0 = len(h0)
            n_candidates = [x for x in n_candidates if x != n0]
            # Also try same n from different starts
            n_candidates.insert(0, n0)

        per_trial = max(25, remaining / (len(n_candidates) * 2 + 2))

        for n_try in n_candidates[:6]:
            if time.time() - t0 > budget_s - 30:
                break
            t_end_a = time.time() + per_trial
            bh, bc = _try_binary_start(n_try, rng, t_end_a)
            results.append((bh, bc, n_try))

            if time.time() - t0 > budget_s - 30:
                break
            t_end_b = time.time() + per_trial
            bh, bc = _try_intermediate_start(n_try, rng, t_end_b)
            results.append((bh, bc, n_try))

    # Phase 3: final polish of the best result
    results.sort(key=lambda x: x[1])
    if results:
        best_h, best_c, best_n = results[0]
        t_end3 = t0 + budget_s - 10
        if time.time() < t_end3 - 10:
            bh, bc = _optimize(best_h, best_n, t_end3, rng)
            if bc < best_c:
                results[0] = (bh, bc, best_n)

    return results[0]
