import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5_val(h, n):
    return float(np.max(_corr(h, n)))

def _smax(x, b):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(b * (x - m)))) / b

def _sgrad(h, n, b):
    dx = 2.0 / n
    c = _corr(h, n)
    m = np.max(c)
    w = np.exp(b * (c - m))
    w /= w.sum()
    omh = 1.0 - h
    g = np.empty(n)
    for p in range(n):
        g[p] = np.dot(w[p:p + n][::-1], omh) - np.dot(w[n - 1 - p:2 * n - 1 - p], h)
    return g * dx

def _feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    d = n * 0.5 - h.sum()
    if abs(d) > 1e-12:
        si = np.argsort(h)
        if d > 0:
            for i in si:
                s = min(1.0 - h[i], d)
                h[i] += s
                d -= s
                if d < 1e-12:
                    break
        else:
            for i in si[::-1]:
                s = min(h[i], -d)
                h[i] -= s
                d += s
                if d > -1e-12:
                    break
    return h

    if n == n0:
        return hp.copy()
    xp = np.linspace(0, 2, n0, endpoint=False)
    xn = np.linspace(0, 2, n, endpoint=False)
    return _feasible(np.interp(xn, xp, hp), n)

def _slsqp(h, n, beta, mi=200):
    def f(x):
        return _smax(_corr(x, n), beta)
    def g(x):
        return _sgrad(x, n, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n

                   constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})
    return _feasible(res.x, n)

def _proj_grad(h, n, beta, ni=400, lr0=0.05):

            c = _c5_val(ho, nb)
            if c < bc:
                bc = c
                bh = ho.copy()
        except Exception:
            pass

    # Final sum correction
    bh = _feasible(bh, nb)
    bc = _c5_val(bh, nb)
    return bh, bc, nb
