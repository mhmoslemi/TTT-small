import numpy as np
from scipy.optimize import minimize
import time


def project_to_feasible(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            add_per = diff / np.sum(mask)
            h = np.where(mask, np.minimum(h + add_per, 1.0), h)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            sub_per = (-diff) / np.sum(mask)
            h = np.where(mask, np.maximum(h - sub_per, 0.0), h)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def objective(x, n_points):
    h = project_to_feasible(x, n_points)
    return compute_c5(h)
