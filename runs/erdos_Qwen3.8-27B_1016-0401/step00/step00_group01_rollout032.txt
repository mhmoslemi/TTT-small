import numpy as np
import time

def compute_c5_only(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = 0.5 * n
    for _ in range(30):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] = np.clip(h[m] + d / c, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmin(h))
        h[i] = float(np.clip(h[i] + d, 0.0, 1.0))
    return h

def make_bin_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    k = n // 2
    h[rng.choice(n, k, replace=False)] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] += 0.5
        h = proj(h, n)
    return h
