    hi = y.max() + 2.0
    for _ in range(60):

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def slsqp_smooth(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh - sh.max())  # numerical stability
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh_shifted = sh - sh.max()

            w = sw[oi]
            if w < 1e-11:
                continue
            grad += w * shift_grad(x, n, oi - center)

import numpy as np
from scipy.optimize import minimize
import time

_ss = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _smooth_obj(x):
    T = _ss['T']
    dx = _ss['dx']
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _smooth_jac(x):
    n = _ss['n']
    T = _ss['T']
    dx = _ss['dx']
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    for oi in range(2 * n - 1):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad(x, n, oi - center)
    return grad * dx

def _topk_obj(x):
    dx = _ss['dx']
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def _topk_jac(x):
    n = _ss['n']
    K = _ss['K']
    dx = _ss['dx']
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    tv = corr[top_idx]
    w = (tv / max(mc, 1e-15))**8
    ws = w.sum()
    if ws < 1e-15:

    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(K):
        grad += w[i] * shift_grad(x, n, top_idx[i] - center)
    return grad * dx

def _eq_fun(x):
    return x.sum() - _ss['target']

def _eq_jac(x):
    return np.ones(len(x))

def do_slsqp_smooth(h, n, T, maxiter):
    _ss['n'] = n
    _ss['T'] = T
    _ss['dx'] = 2.0 / n
    _ss['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n

    try:

                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:

    _ss['K'] = K
    _ss['dx'] = 2.0 / n
    _ss['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_topk_obj, h, method='SLSQP', jac=_topk_jac,
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h.copy()

def grad_method(h_init, n, t0, deadline, alpha0=0.35, n_iter=400000):
    dx = 2.0 / n
    center = n - 1
    h = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    for it in range(n_iter):
        if time.time() - t0 > deadline:
            break
        frac = it / float(n_iter)
        K = max(1, int(1 + 14 * (1.0 - frac)**1.3))
        beta = 0.85 if it > 0 else 0.0
        h_y = h + beta * (h - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            tv = corr[top_idx]
            w = (tv / max(mc, 1e-15))**6
            ws = w.sum()
            w = w / ws if ws > 1e-15 else np.ones(K) / K
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx
        alpha = alpha0 / (1.0 + 0.004 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h
        h = h_new
        if (it + 1) % 8000 == 0:
            c = compute_c5(h)
            if c < c5_best:
                c5_best = c
                h_best = h.copy()
    c_final = compute_c5(h)
    if c_final < c5_best:
        c5_best = c_final
        h_best = h.copy()
    return h_best, c5_best

def perturb(h, n, rng, mode, scale=1.0):
    x = np.arange(n) * 2.0 / n
    if mode % 6 == 0:
        f = 0.2 + 0.8 * rng.random()
        h_p = h + 0.03 * scale * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode % 6 == 1:
        f = 0.5 + 2.0 * rng.random()
        h_p = h + 0.02 * scale * np.cos(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode % 6 == 2:
        nblk = 3 + rng.integers(0, 5)
        h_p = h.copy()
        for _ in range(nblk):
            ci = rng.integers(0, n)
            w = rng.integers(max(2, n // 20), max(3, n // 4))
            lo_i = max(0, ci - w // 2)
            hi_i = min(n, ci + w // 2 + 1)
            h_p[lo_i:hi_i] += rng.uniform(-0.04, 0.04) * scale
    elif mode % 6 == 3:
        h_p = h + rng.normal(0, 0.015 * scale, n)
    elif mode % 6 == 4:
        h_p = h.copy()
        for _ in range(3 + rng.integers(0, 4)):
            i = rng.integers(0, n)
            h_p[i] = 1.0 - h_p[i]
    else:
        f1 = 0.3 + rng.random()
        f2 = 0.7 + rng.random()
        h_p = h + 0.02 * scale * (np.sin(2*np.pi*f1*x) + 0.5*np.cos(2*np.pi*f2*x))
    return h_p

def resample(h, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth polish
    p1_end = t0 + 100
    for T in [40, 16, 6]:
        if time.time() > p1_end:
            break
        h_new = do_slsqp_smooth(h_best, n, T, 4000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 2: Gradient method with momentum
    p2_end = t0 + 350
    h_g, c5_g = grad_method(h_best, n, t0, p2_end, 0.35, 500000)
    if c5_g < c5_best:
        c5_best = c5_g
        h_best = h_g
    h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP smooth refinement
    p3_end = t0 + 500
    for T in [3, 2]:
        if time.time() > p3_end:
            break
        h_new = do_slsqp_smooth(h_best, n, T, 5000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    # Phase 4: Diversification with temperature schedule
    p4_end = t0 + budget_s - 200
    n_try = 50
    for pi in range(n_try):
        if time.time() > p4_end:
            break
        scale = 1.0 * (1.0 - 0.01 * pi) + 0.3
        h_p = perturb(h_best, n, rng, pi, scale)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp_topk(h_p, n, 5, 2000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new
            if time.time() < p4_end - 40:
                h_new2 = do_slsqp_smooth(h_best, n, 6, 3000)
                h_new2 = project_feasible(h_new2, n)
                c2 = compute_c5(h_new2)
                if c2 < c5_best:
                    c5_best = c2
                    h_best = h_new2

    # Phase 5: Try alternate resolutions
    p5_end = t0 + budget_s - 100
    for n2 in [n // 2, n * 3 // 2, n * 2]:
        if time.time() > p5_end:
            break
        if n2 < 56 or n2 > 900:
            continue
        h_alt = resample(h_best, n, n2)
        h_alt = project_feasible(h_alt, n2)
        for K2 in [4, 8]:
            if time.time() > p5_end - 20:
                break
            h_new = do_slsqp_topk(h_alt, n2, K2, 4000)
            h_new = project_feasible(h_new, n2)
            c_alt = compute_c5(h_new)
            if c_alt < c5_best:
                c5_best = c_alt
                h_best = h_new
                n = n2

    # Phase 6: Final polish
    if time.time() < t0 + budget_s - 20:
        h_new = do_slsqp_topk(h_best, n, 3, 5000)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new
        if time.time() < t0 + budget_s - 10:
            h_new = do_slsqp_smooth(h_best, n, 2, 4000)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new)
            if c < c5_best:
                c5_best = c
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
