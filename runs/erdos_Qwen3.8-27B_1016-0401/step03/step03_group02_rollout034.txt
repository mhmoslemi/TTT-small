import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(-h)
        for i in idx:
            room = (1.0 - h[i]) if diff > 0 else h[i]
            step = min(abs(diff), room)
            if step <= 1e-15:
                continue
            h[i] += step if diff > 0 else -step
            diff -= step if diff > 0 else step
            if abs(diff) < 1e-13:
                break
    return h

def sm_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += w * (1.0 - h[:n - o])
            grad[:n - o] += -w * h[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += -w * h[:n - p]
    return grad * dx

def adam_opt(h, target, T, n_steps, lr, deadline):
    n = len(h)
    x = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x, best_c5 = x.copy(), compute_c5(x)
    for t in range(n_steps):
        if time.time() > deadline:
            break
        g = sm_grad(x, T)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mc = m / (1 - b1 ** (t + 1))
        vc = v / (1 - b2 ** (t + 1))
        x_new = x - lr * mc / (np.sqrt(vc) + eps)
        x_new = fix_constraint(np.clip(x_new, 0, 1), target)
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_c5, best_x = c5, x_new.copy()
        x = x_new
    return best_x, best_c5

def _eq_fun(x):
    return x.sum() - _GLOBAL_TARGET

def _eq_jac(x):
    return np.ones(len(x))

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g * dx

def slsqp_lag(h, lag_idx, target, bounds, maxiter):
    global _GLOBAL_TARGET
    _GLOBAL_TARGET = target
    cons = [dict(type='eq', fun=_eq_fun, jac=_eq_jac)]
    try:
        res = minimize(lag_obj, h, args=(lag_idx,), method='SLSQP',
                       jac=lag_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

def binary_sa(h, target, n_tries, rng, deadline):
    n = len(h)
    h_cur = np.where(h >= 0.5, 1.0, 0.0)
    n_ones = int(h_cur.sum())
    t_ones = int(target)
    if n_ones > t_ones:
        idx = np.where(h_cur == 1.0)[0]
        rng.shuffle(idx)
        h_cur[idx[:n_ones - t_ones]] = 0.0
    elif n_ones < t_ones:
        idx = np.where(h_cur == 0.0)[0]
        rng.shuffle(idx)
        h_cur[idx[:t_ones - n_ones]] = 1.0
    c5_cur = compute_c5(h_cur)
    best_h, best_c5 = h_cur.copy(), c5_cur
    temp = 0.004
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ii = ones[rng.integers(len(ones))]
        jj = zeros[rng.integers(len(zeros))]
        h_cur[ii] = 0.0
        h_cur[jj] = 1.0
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_cur.copy()
        else:
            h_cur[ii] = 1.0
            h_cur[jj] = 0.0
        temp *= 0.9997
    return best_h, best_c5
