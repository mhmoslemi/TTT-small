```python
def smoothed_max(corr, T, K=50):
    """Smoothed max using top-K log-sum-exp."""
    n_corr = len(corr)
    if K >= n_corr:
        idx = np.arange(n_corr)
    else:
        idx = np.argpartition(corr, -K)[-K:]
    c_sel = corr[idx]
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    return mc + np.log(e.sum()) / T, idx
```

```python
    # Phase 1: Coordinate descent
    h_best = coord_descent(h_best, n, 5, t0, t0 + 60)
    c5_best = compute_c5(h_best)
    
    # Phase 2: L-BFGS-B coarse-to-fine
    for T, K, mi in [(64, 30, 800), (16, 20, 1200), (4, 15, 1500), (1, 10, 2000)]:
        if time.time() > tlim - 500:
            break
        _G['T'], _G['K'], _G['t0'], _G['tlim'] = T, K, t0, tlim - 30
        h_new = do_lbfgsb(h_best, n, mi)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    
    # Phase 3: Coordinate descent
    h_best = coord_descent(h_best, n, 5, t0, tlim - 400)
    c5_best = compute_c5(h_best)
    
    # Phase 4: SLSQP
    for T, K, mi in [(4, 20, 2000), (1.5, 15, 2500)]:
        if time.time() > tlim - 250:
            break
        _G['T'], _G['K'], _G['t0'], _G['tlim'] = T, K, t0, tlim - 20
        h_new = do_slsqp(h_best, n, mi)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    
    # Phase 5: Diverse restarts
    n_rest = 40
    for pi in range(n_rest):
        rem = tlim - time.time()
        if rem < 25:
            break
        if pi % 4 == 0:
            h_p = h_best + rng.normal(0, 0.015, n)
        elif pi % 4 == 1:
            x = np.arange(n, dtype=float)
            fq = rng.uniform(1, 6)
            ph = rng.uniform(0, 2 * np.pi)
            h_p = h_best + 0.03 * np.sin(2 * np.pi * fq * x / n + ph)
        elif pi % 4 == 2:
            h_p = h_best.copy()
            idx = rng.choice(n, 20, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            h_p = h_best + rng.normal(0, 0.008, n)
        h_p = project_feasible(h_p, n)
        h_p = coord_descent(h_p, n, 2, t0, t0 + rem - 20)
        _G['T'], _G['K'], _G['t0'], _G['tlim'] = 4, 20, t0, t0 + rem - 10
        h_new = do_lbfgsb(h_p, n, 800)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    
    # Phase 6: Final coordinate descent + L-BFGS-B polish
    h_best = coord_descent(h_best, n, 8, t0, tlim - 10)
    c5 = compute_c5(h_best)
    if c5 < c5_best:
        c5_best = c5
    
    _G['T'], _G['K'], _G['t0'], _G['tlim'] = 1, 10, t0, tlim - 5
    h_new = do_lbfgsb(h_best, n, 2000)
    c5 = compute_c5(h_new)
    if c5 < c5_best:
        c5_best, h_best = c5, h_new
    
    # Phase 7: Try smaller n
    rem = tlim - time.time()
    if rem > 80:
        x_old = np.arange(n) * (2.0 / n)
        for n_try in [200, 240]:
            rem = tlim - time.time()
            if rem < 40:
                break
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, h_best)
            h_try = project_feasible(h_try, n_try)
            h_try = coord_descent(h_try, n_try, 4, t0, t0 + rem - 10)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best:
                c5_best, h_best, n = c5_try, h_try, n_try
    
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
```
