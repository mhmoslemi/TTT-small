    g = np.zeros(n)

def project_feasible(y, n):
    target = n / 2.0

        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]

    # Phase 6: Final ultra-fine polish (8% time)
    if time.time() < budget_s - 8:
        hn = slsqp_full(h_best, n, 0.2, 4, 5000, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
        hr2, cr2 = block_coordinate(h_best, n, 20, min(budget_s - 3, time.time() + 20), seed + 99)
        if cr2 < c5_best:
            c5_best, h_best = cr2, hr2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
