def compute_c5(h):

    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def opt_1d(B, C):
    """Minimize max_i(B[i] + C[i]*x) over x in [0,1]."""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = B + C * mid
        fmax = vals.max()
        active = vals >= fmax - 1e-14
        if not active.any():
            return mid
        cmin = C[active].min()
        cmax = C[active].max()
        if cmin > 1e-10:
            hi = mid
        elif cmax < -1e-10:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def opt_1d(B, C):
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = B + C * mid
        fmax = vals.max()
        active = vals >= fmax - 1e-14
        if not active.any():
            return mid
        cmin = C[active].min()
        cmax = C[active].max()
        if cmin > 1e-10:
            hi = mid
        elif cmax < -1e-10:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def coord_pass_fast(h, n):
    oh = 1.0 - h
    corr = np.correlate(h, oh, mode="full")
    m = np.arange(-(n - 1), n)
    for j in range(n):
        I_L = (j + m >= 0) & (j + m < n)
        I_R = (j - m >= 0) & (j - m < n)
        jp = j + m
        jm = j - m
        h_jp = np.zeros(2 * n - 1)
        h_jm = np.zeros(2 * n - 1)
        h_jp[I_L] = h[jp[I_L]]
        h_jm[I_R] = h[jm[I_R]]
        C = (1.0 - h_jp) * I_L.astype(float) - h_jm * I_R.astype(float)
        B = corr - h[j] * C
        x_opt = opt_1d(B, C)
        corr += (x_opt - h[j]) * C
        h[j] = x_opt
    return h

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_S_N = 0
_S_TGT = 0.0
_S_T = 1.0
_S_K = 20

def _s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_S_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_S_T * (c - m))
    return (m + np.log(e.sum()) / _S_T) * (2.0 / _S_N)

def _s_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_S_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_S_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_S_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _S_N, ci - (_S_N - 1))
    return grad * (2.0 / _S_N)

def _s_cf(x):
    return x.sum() - _S_TGT

def _s_cj(x):
    return np.ones(_S_N)

def _slsqp(h, n, T, top_k, maxiter, ftol):
    global _S_N, _S_TGT, _S_T, _S_K
    _S_N = n
    _S_TGT = n / 2.0
    _S_T = float(T)
    _S_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_s_obj, h, method='SLSQP', jac=_s_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _s_cf, 'jac': _s_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

def _nesterov(h_init, n, t_end, rng, top_k=8):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(200000):
        if time.time() > t_end:
            break
        beta = 1.0 if it == 0 else float(it) / (it + 1.5)
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 0.8 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 20000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a
