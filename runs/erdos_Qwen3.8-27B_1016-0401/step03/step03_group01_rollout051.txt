import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def corr_and_grad(h, corr, K, dx):
    """Compute subgradient from top-K lags."""
    n = len(h)
    center = n - 1
    j = np.arange(n)
    idx = np.argpartition(corr, -K)[-K:]
    grad = np.zeros(n)
    for oi in idx:
        shift = int(oi) - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= h[j[m2] - s2]
    return grad * dx, idx

def fista_adaptive(h_init, n, t_end, max_iters=100000):
    """FISTA with adaptive top-K and backtracking."""
    dx = 2.0 / n
    y = h_init.copy()
    z = h_init.copy()
    t_mom = 1.0
    lr = 0.05
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        c5_curr = float(np.max(corr) * dx)
        
        # Adaptive K: all lags within 1% of max
        threshold = np.max(corr) * 0.99
        K = max(int(np.sum(corr >= threshold)), 3)
        K = min(K, 20)  # cap at 20
        
        grad, _ = corr_and_grad(y, corr, K, dx)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        z = y + beta * (y - z)
        y_new = z - lr * grad
        y_new = project_feasible(y_new, n)
        
        # Check improvement (every 500 iters)
        if (it + 1) % 500 == 0:
            c5_check = compute_c5(y_new)
            if c5_check < c5_best:
                c5_best = c5_check
                h_best = y_new.copy()
                stall = 0
                lr = min(lr * 1.01, 0.15)
            else:
                stall += 1
                lr *= 0.95
                if stall > 5:
                    # Reset momentum
                    z = y_new.copy()
                    t_mom = 1.0
                    stall = 0
                    lr = 0.03
        y = y_new
        t_mom = t_new
    
    return h_best, c5_best

def slsqp_smooth(h, n, T, K, maxiter, ftol, t_end):
    """SLSQP on smooth top-K approximation."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(K, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        e = np.exp(T * (topk - mc))
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(K, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        e = np.exp(T * (topk - mc))
        sw = e / e.sum()
        grad = np.zeros(n)
        center = n - 1
        j = np.arange(n)
        for oi in range(mk):
            w = sw[oi]
            if w < 1e-11:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w * (1.0 - x[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= w * x[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += w * (1.0 - x[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= w * x[j[m2] - s2]
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coordinate_descent(h, n, t_end, rng):
    """Optimize small subsets of coordinates."""
    dx = 2.0 / n
    c5 = compute_c5(h)
    for _ in range(50000):
        if time.time() > t_end:
            break
        # Pick random subset
        size = rng.integers(5, 15)
        idx = rng.choice(n, size, replace=False)
        idx_sorted = np.sort(idx)
        
        # SLSQP on subset
        h_sub = h[idx_sorted].copy()
        
        def sub_obj(x):
            h_tmp = h.copy()
            h_tmp[idx_sorted] = x
            return compute_c5(h_tmp)
        
        def sub_jac(x):
            h_tmp = h.copy()
            h_tmp[idx_sorted] = x
            corr = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
            k_max = np.argmax(corr)
            grad_full = np.zeros(n)
            center = n - 1
            j = np.arange(n)
            shift = int(k_max) - center
            if shift >= 0:
                m1 = j >= shift
                grad_full[m1] += (1.0 - h_tmp[j[m1] - shift])
                m2 = (j + shift) < n
                grad_full[m2] -= h_tmp[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad_full[m1] += (1.0 - h_tmp[j[m1] + s2])
                m2 = j >= s2
                grad_full[m2] -= h_tmp[j[m2] - s2]
            return (grad_full * dx)[idx_sorted]
        
        try:
            res = minimize(sub_obj, h_sub, method='SLSQP', jac=sub_jac,
                          bounds=[(0,1)]*size, maxiter=100, ftol=1e-14)
            h_new = h.copy()
            h_new[idx_sorted] = res.x
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                c5 = c5_new
                h = h_new
        except Exception:
            pass
    return h, c5

        h_best = h_f

    # Phase 2: SLSQP smooth polish (25% budget)
    rem = budget_s - (time.time() - t0)
    t2 = time.time() + rem * 0.30
    for T, K in [(16, 12), (8, 10), (4, 8), (2, 6), (1, 5)]:
        if time.time() > t2:
            break
        _smo_params = (n, T, K)
        _smj_params = (n, T, K)
        h_new = slsqp_smooth(h_best, n, T, K, 4000, 1e-18, t2)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Another FISTA pass with tighter focus
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        t3 = time.time() + rem * 0.25
        h_f2, c5_f2 = fista_adaptive(h_best, n, t3)
        h_f2 = project_feasible(h_f2, n)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < c5_best:
            c5_best = c5_f2
            h_best = h_f2

    # Phase 4: Perturbation + re-optimize loop
    rem = budget_s - (time.time() - t0)
    n_perts = max(3, int(rem / 25))
    for pi in range(n_perts):
        if time.time() - t0 > budget_s - 15:
            break
        strength = 0.02 + 0.03 * (pi % 3) / 3.0
        h_p = perturb(h_best, n, rng, strength)
        _smo_params = (n, 4, 8)
        _smj_params = (n, 4, 8)
        h_new = slsqp_smooth(h_p, n, 4, 8, 2000, 1e-16, None)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Resolution upscaling
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, h_best)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up)
        n_curr = n
        if c5_up < c5_best - 0.001:
            c5_best = c5_up
            h_best = h_up
            n_curr = n2
        _smo_params = (n_curr, 8, 10)
        _smj_params = (n_curr, 8, 10)
        t5 = time.time() + rem * 0.5
        h_new = slsqp_smooth(h_best, n_curr, 8, 10, 3000, 1e-17, None)
        h_new = project_feasible(h_new, n_curr)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            n = n_curr
        _smo_params = (n, 2, 6)
        _smj_params = (n, 2, 6)
        h_new = slsqp_smooth(h_best, n, 2, 6, 2000, 1e-17, None)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final FISTA polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_f3, c5_f3 = fista_adaptive(h_best, len(h_best), time.time() + rem * 0.6)
        h_f3 = project_feasible(h_f3, len(h_f3))
        c5_f3 = compute_c5(h_f3)
        if c5_f3 < c5_best:
            c5_best = c5_f3
            h_best = h_f3
            n = len(h_best)

    h_best = project_feasible(h_best, n)
    c5_check = compute_c5(h_best)
    c5_final = min(c5_check, c5_best)
    return (h_best, c5_final, n)
