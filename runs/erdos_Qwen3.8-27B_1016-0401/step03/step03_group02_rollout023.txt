import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    """Project h onto {0<=h<=1, sum=h_target} using bisection."""
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = h_p.sum() - target
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            h_p[mask] -= diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def softmax_gradient(h, T, top_m=100):
    """Compute gradient of the T-softmax-smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    # Get top m lags
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        corr_top = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        corr_top = corr
    mc = corr_top.max()
    shifted = T * (corr_top - mc)
    e = np.exp(shifted - shifted.max())
    w = e / e.sum()
    grad = np.zeros(n)
    for i, lag_idx in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-12:
            continue
        o = lag_idx - center
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] -= wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx, float(mc * dx)

def fista_smoothed(h, T_start, T_end, n_iters, top_m=100, target=None):
    """FISTA for the smoothed max correlation."""
    if target is None:
        target = len(h) / 2.0
    n = len(h)
    h_old = h.copy()
    z = h.copy()
    t_k = 1.0
    # Lipschitz constant estimate
    L = 4.0 / n  # rough bound
    step = 1.0 / L
    
    for k in range(n_iters):
        T_k = T_start * (T_end / T_start) ** (k / max(n_iters - 1, 1))
        grad, _ = softmax_gradient(z, T_k, top_m)
        z_new = z - step * grad
        z_new = fix_constraint(z_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_new = z_new + ((t_k - 1.0) / t_new) * (z_new - z_old)
        h_new = fix_constraint(h_new, target)
        z_old = z
        z = z_new
        h_old = h_new
        t_k = t_new
    return z

def sa_binary(h_init, n_flips, T_init, T_final, rng):
    """Simulated annealing on binary h."""
    n = len(h_init)
    target = n // 2
    h = (h_init > 0.5).astype(float)
    # Ensure exactly target ones
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        excess = rng.choice(ones, len(ones)-target, replace=False)
        h[excess] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        zeros = np.where(h == 0)[0]
        to_add = rng.choice(zeros, deficit, replace=False)
        h[to_add] = 1
    
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for it in range(n_flips):
        T = T_init * (T_final / T_init) ** (it / max(n_flips - 1, 1))
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        h[i], h[j] = 0.0, 1.0
        ones_set.remove(i); ones_set.add(j)
        zeros_set.add(i); zeros_set.remove(j)
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            ones_set.add(i); ones_set.remove(j)
            zeros_set.remove(i); zeros_set.add(j)
    
    return h_best, c5_best

def eq_fun(x, _target):
    return x.sum() - _target

def lag_obj(x, lag_idx, n):
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx, n):
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] -= x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] -= x[:n-p]
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = h_p.sum() - target
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        if mask.sum() > 0:
            h_p[mask] -= diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def grad_single_lag(h, lag_idx, n):
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] -= h[:n-p]
    return g * dx

def softmax_grad(h, T, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        ct = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        ct = corr
    mc = ct.max()
    sh = T * (ct - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-12:
            continue
        o = top_idx[i] - center
        if o == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif o > 0:
            grad[o:] += wi * (1.0 - h[:n-o])
            grad[:n-o] -= wi * h[o:]
        else:
            p = -o
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return grad * dx

def pgrad_smoothed(h, T_start, T_end, n_iters, top_m, target):
    n = len(h)
    step_base = 0.3 / (n * T_start + 1.0)
    x = h.copy()
    x_ymm = h.copy()
    t_k = 1.0
    for k in range(n_iters):
        frac = k / max(n_iters - 1, 1)
        T_k = T_start * (T_end / T_start) ** frac
        g = softmax_grad(x, T_k, top_m)
        step = step_base / (1.0 + 0.01 * k)
        z = x - step * g
        z = project_onto_feasible(z, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = z + ((t_k - 1.0) / t_new) * (z - x_ymm)
        x_new = project_onto_feasible(x_new, target)
        x_ymm = x
        x = x_new
        t_k = t_new
    return x
