import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        grad[m2] -= h[j[m2] - s]
    return grad

    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    
    def obj(x):

        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)

        return grad * dx
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def proximal_subgrad(h_init, n, t0, time_limit, alpha0=0.08):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_sg.copy()
    h_avg = h_sg.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()

    for it in range(500000):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h_sg, 1.0 - h_sg, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -3)[-3:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-15)) ** 10
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(3) / 3
        else:
            w = w / ws
        g = np.zeros(n)
        for i in range(3):
            g += w[i] * shift_grad(h_sg, n, top_idx[i] - center)
        g *= dx

        beta = 2.0 / (3.0 + float(it))
        alpha = alpha0 * beta
        h_new = h_sg - alpha * g
        h_new = project_feasible(h_new, n)

        h_prev = h_sg.copy()
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))

        if (it + 1) % 8000 == 0:
            h_ap = project_feasible(h_avg, n)
            c5a = compute_c5(h_ap)
            if c5a < c5_best:
                c5_best = c5a
                h_best = h_ap.copy()
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()

    return h_best, c5_best

def multi_start_refine(h_best, n, c5_best, rng, t0, remaining):
    for trial in range(6):
        if time.time() - t0 > budget_s_global:
            break
        sigma = 0.01 * (trial + 1)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        for T in [4, 16]:
            if time.time() - t0 > budget_s_global - 60:
                break
            h_new = slsqp_smooth(h_p, n, T, 3000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
    return h_best, c5_best

budget_s_global = 1100

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP broad-to-sharp (opposite of previous)
    for T in [2, 4, 8, 16]:
        if time.time() - t0 > budget_s - 550:
            break
        h_new = slsqp_smooth(h_best, n, T, 6000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Proximal subgradient with top-3 focusing
    h_sg, c5_sg = proximal_subgrad(h_best, n, t0, budget_s - 400)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 3: Sharp SLSQP (large T = close to true max)
    for T in [32, 64]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = slsqp_smooth(h_best, n, T, 10000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Top-K SLSQP focused on worst shifts
    for K in [2, 3, 5, 10]:
        if time.time() - t0 > budget_s - 120:
            break
        h_new = slsqp_topk(h_best, n, K, 8000, 1e-18, pwr=16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Multi-start with micro-perturbations
    elapsed = time.time() - t0
    if elapsed < budget_s - 100:
        h_best, c5_best = multi_start_refine(h_best, n, c5_best, rng, t0, budget_s - elapsed)

    # Phase 6: Final ultra-sharp polish
    if time.time() - t0 < budget_s - 40:
        h_new = slsqp_smooth(h_best, n, 128, 12000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
