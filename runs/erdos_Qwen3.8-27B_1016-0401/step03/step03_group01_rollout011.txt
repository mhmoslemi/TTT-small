import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def c5_grad_active(h, n, active_idx):
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for oi in active_idx:
        shift = oi - center
        if shift >= 0:
            grad[shift:] += (1.0 - h[:n - shift])
            grad[:n - shift] -= h[shift:]
        else:
            s = -shift
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    grad *= dx / len(active_idx)
    return grad

def find_active(h, n, k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argpartition(corr, -k)[-k:]

def fist_opt(h_init, n, t_end, k_active=5, lr0=0.04, recompute=25):

def _slsqp_obj_fn(x, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    e = np.exp(sh)
    e_sum = e.sum()
    obj = (mc + np.log(e_sum) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    for oi in range(mk):
        w = e[oi] / e_sum
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            grad[shift:] += w * (1.0 - x[:n - shift])
            grad[:n - shift] -= w * x[shift:]
        else:
            s = -shift
            grad[:n - s] += w * (1.0 - x[s:])
            grad[s:] -= x[:n - s]
    grad *= dx
    return obj, grad

def _slsqp_con_fn(x):
    return x.sum() - x.size / 2.0

def _slsqp_con_jac(x):
    return np.ones(x.size)

def slsqp_opt(h, n, T, k, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fn, 'jac': _slsqp_con_jac}]

    def obj_wrapped(x):
        return _slsqp_obj_fn(x, n, T, k)[0]

    def jac_wrapped(x):
        return _slsqp_obj_fn(x, n, T, k)[1]

    try:
        res = minimize(obj_wrapped, h.copy(), method='SLSQP', jac=jac_wrapped,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
