import time

def compute_c5(h):

def project_feasible(y, n):

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m = j >= shift
        if m.any():
            grad[m] += (1.0 - h[j[m] - shift])

        m2 = j >= s

def sm_obj_grad(x, n, T, K, t0, tlim, flag):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
    else:
        topk = np.arange(len(corr))
        c_sel = corr
    mc = float(c_sel.max())
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    s = e.sum()
    val = (mc + np.log(s) / T) * dx
    sw = e / s
    grad = np.zeros(n)
    for ii in range(len(topk)):
        w = sw[ii]
        if w < 1e-14:
            continue
        grad += w * shift_grad(x, n, topk[ii] - center)

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    flag = [False]
    def obj(x):
        if time.time() - t0 > tlim:
            flag[0] = True
        v, _ = sm_obj_grad(x, n, T, K, t0, tlim, flag)
        return v
    def jac(x):
        if time.time() - t0 > tlim:
            flag[0] = True
        _, g = sm_obj_grad(x, n, T, K, t0, tlim, flag)
        return g
    bounds = [(0.0, 1.0)] * n

        center = n - 1

def diversify(h, n, rng):
    """Generate diverse perturbations."""
    x = np.arange(n, dtype=float)
    mode = int(rng.integers(0, 5))
    if mode == 0:
        # Sinusoidal
        pert = np.zeros(n)
        for _ in range(6):
            f = rng.uniform(0.5, 10)
            p = rng.uniform(0, 2*np.pi)
            a = rng.uniform(0.01, 0.05)
            pert += a * np.sin(2*np.pi*f*x/n + p)
        return h + pert
    elif mode == 1:
        # Gaussian noise
        return h + rng.normal(0, 0.005 + 0.01*(mode), n)
    elif mode == 2:
        # Block flip
        h2 = h.copy()
        bs = n // 10
        bi = int(rng.integers(0, 10))
        h2[bi*bs:(bi+1)*bs] = 1.0 - h2[bi*bs:(bi+1)*bs]
        return h2 + rng.normal(0, 0.003, n)
    elif mode == 3:
        # Swapped segments
        h2 = h.copy()
        L = n // 7
        i = int(rng.integers(0, n-L))
        j = int(rng.integers(0, n-L))
        if abs(i-j) > L:
            h2[i:i+L], h2[j:j+L] = h2[j:j+L], h2[i:i+L]
        return h2
    else:
        # High-frequency jitter
        h2 = h.copy()
        mask = rng.random(n) < 0.1
        h2[mask] += rng.uniform(-0.1, 0.1, mask.sum())
        return h2
